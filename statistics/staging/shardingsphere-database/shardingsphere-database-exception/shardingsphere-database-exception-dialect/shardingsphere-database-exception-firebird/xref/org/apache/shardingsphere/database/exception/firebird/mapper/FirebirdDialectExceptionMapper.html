49  /**
50   * Firebird dialect exception mapper.
51   */
52  public final class FirebirdDialectExceptionMapper implements SQLDialectExceptionMapper {
53      
54      @Override
55      public SQLException convert(final SQLDialectException sqlDialectException) {
56          if (sqlDialectException instanceof UnknownDatabaseException) {
57              return toSQLException(FirebirdVendorError.UNAVAILABLE_DATABASE, ((UnknownDatabaseException) sqlDialectException).getDatabaseName());
58          }
59          if (sqlDialectException instanceof DatabaseCreateExistsException) {
60              return toSQLException(FirebirdVendorError.DATABASE_ALREADY_EXISTS, ((DatabaseCreateExistsException) sqlDialectException).getDatabaseName());
61          }
62          if (sqlDialectException instanceof DatabaseDropNotExistsException) {
63              return toSQLException(FirebirdVendorError.UNAVAILABLE_DATABASE, ((DatabaseDropNotExistsException) sqlDialectException).getDatabaseName());
64          }
65          if (sqlDialectException instanceof AccessDeniedException) {
66              return toSQLException(FirebirdVendorError.LOGIN_FAILED);
67          }
68          if (sqlDialectException instanceof LoginErrorException) {
69              return toSQLException(FirebirdVendorError.LOGIN_ERROR);
70          }
71          if (sqlDialectException instanceof InvalidBatchHandleException) {
72              return toSQLException(FirebirdVendorError.INVALID_BATCH_HANDLE);
73          }
74          if (sqlDialectException instanceof BatchTooBigException) {
75              return toSQLException(FirebirdVendorError.BATCH_TOO_BIG);
76          }
77          if (sqlDialectException instanceof BatchAlreadyOpenedException) {
78              return toSQLException(FirebirdVendorError.BATCH_ALREADY_OPENED);
79          }
80          if (sqlDialectException instanceof InvalidBatchParameterVersionException) {
81              InvalidBatchParameterVersionException ex = (InvalidBatchParameterVersionException) sqlDialectException;
82              return toSQLException(FirebirdVendorError.INVALID_BATCH_PARAMETER_VERSION, ex.getVersion(), ex.getExpectedVersion());
83          }
84          if (sqlDialectException instanceof BatchParametersRequiredException) {
85              return toSQLException(FirebirdVendorError.BATCH_PARAMETERS_REQUIRED);
86          }
87          if (sqlDialectException instanceof InvalidBatchMessageFormatException) {
88              return toSQLException(FirebirdVendorError.SQLDA_ERROR);
89          }
90          if (sqlDialectException instanceof InvalidStatementHandleException) {
91              return toSQLException(FirebirdVendorError.INVALID_STATEMENT_HANDLE);
92          }
93          if (sqlDialectException instanceof InvalidTransactionHandleException) {
94              return toSQLException(FirebirdVendorError.INVALID_TRANSACTION_HANDLE);
95          }
96          if (sqlDialectException instanceof ExcessTransactionsException) {
97              return toSQLException(FirebirdVendorError.EXCESS_TRANSACTIONS, ((ExcessTransactionsException) sqlDialectException).getMaxTransactions());
98          }
99          if (sqlDialectException instanceof DialectSQLParsingException) {
100             return toSQLException(FirebirdVendorError.DYNAMIC_SQL_ERROR, sqlDialectException.getMessage());
101         }
102         if (sqlDialectException instanceof TableExistsException) {
103             return toSQLException(FirebirdVendorError.TABLE_ALREADY_EXISTS, ((TableExistsException) sqlDialectException).getTableName());
104         }
105         if (sqlDialectException instanceof InvalidParameterValueException) {
106             return toSQLException(FirebirdVendorError.CHARSET_NOT_FOUND, ((InvalidParameterValueException) sqlDialectException).getParameterValue());
107         }
108         if (sqlDialectException instanceof InvalidSegstrHandleException) {
109             return toSQLException(FirebirdVendorError.INVALID_SEGSTR_HANDLE);
110         }
111         if (sqlDialectException instanceof InvalidSegstrIdException) {
112             return toSQLException(FirebirdVendorError.INVALID_SEGSTR_ID);
113         }
114         if (sqlDialectException instanceof CannotUpdateOldBlobException) {
115             return toSQLException(FirebirdVendorError.CANNOT_UPDATE_OLD_BLOB);
116         }
117         if (sqlDialectException instanceof InvalidSegstrTypeException) {
118             return toSQLException(FirebirdVendorError.INVALID_SEGSTR_TYPE);
119         }
120         return new UnknownSQLException(sqlDialectException).toSQLException();
121     }
122     
123     private SQLException toSQLException(final VendorError vendorError, final Object... messageArgs) {
124         return new SQLException(String.format(vendorError.getReason(), messageArgs), vendorError.getSqlState().getValue(), vendorError.getVendorCode());
125     }
126     
127     @Override
128     public String getDatabaseType() {
129         return "Firebird";
130     }
131 }
