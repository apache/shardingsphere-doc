66  final class SQLStatementObjectExtractor {
67      
68      private static final Set<String> NAMESPACE_PRIVILEGE_OBJECT_TYPES = Set.of("DATABASE", "SCHEMA");
69      
70      Collection<SQLStatementObjectName> extract(final SQLStatement sqlStatement) {
71          Collection<SQLStatementObjectName> result = new LinkedHashSet<>();
72          extractDirectTargets(sqlStatement, result);
73          TableExtractor tableExtractor = new TableExtractor();
74          tableExtractor.extractTablesFromSQLStatement(sqlStatement);
75          addTableSegments(tableExtractor.getTableContext(), result);
76          sqlStatement.getAttributes().findAttribute(TableSQLStatementAttribute.class).ifPresent(attribute -> addTables(attribute.getTables(), result));
77          extractMergeTables(sqlStatement, result);
78          extractCommonTableExpressionTables(sqlStatement, result);
79          removeCommonTableExpressionAliases(sqlStatement, result);
80          extractDCLTargets(sqlStatement, result);
81          extractQualifiedFunctions(sqlStatement, result);
82          return result;
83      }
84      
85      private void extractDCLTargets(final SQLStatement sqlStatement, final Collection<SQLStatementObjectName> result) {
86          Collection<PrivilegeObjectSegment> privilegeObjects;
87          if (sqlStatement instanceof GrantStatement grant) {
88              privilegeObjects = grant.getPrivilegeObjects();
89          } else if (sqlStatement instanceof RevokeStatement revoke) {
90              privilegeObjects = revoke.getPrivilegeObjects();
91          } else {
92              return;
93          }
94          for (PrivilegeObjectSegment each : privilegeObjects) {
95              if (!each.getIdentifiers().isEmpty()) {
96                  result.add(NAMESPACE_PRIVILEGE_OBJECT_TYPES.contains(each.getObjectType())
97                          ? SQLStatementObjectName.fromNamespace(each.getIdentifiers())
98                          : SQLStatementObjectName.from(each.getIdentifiers()));
99              }
100         }
101     }
102     
103     private void extractQualifiedFunctions(final SQLStatement sqlStatement, final Collection<SQLStatementObjectName> result) {
104         new SQLStatementTreeWalker(each -> {
105         }, expression -> {
106             if (expression instanceof FunctionSegment function) {
107                 if (null != function.getOwner()) {
108                     result.add(SQLStatementObjectName.from(Optional.of(function.getOwner()), new IdentifierValue(function.getFunctionName())));
109                 }
110             }
111         }).walk(sqlStatement);
112     }
113     
114     private void extractDirectTargets(final SQLStatement sqlStatement, final Collection<SQLStatementObjectName> result) {
115         if (sqlStatement instanceof InsertStatement insert) {
116             insert.getTable().ifPresent(optional -> addTable(optional, result));
117         } else if (sqlStatement instanceof UpdateStatement update) {
118             addTableSegment(update.getTable(), result);
119         } else if (sqlStatement instanceof DeleteStatement delete) {
120             addTableSegment(delete.getTable(), result);
121         } else if (sqlStatement instanceof MergeStatement merge) {
122             addTableSegment(merge.getTarget(), result);
123         } else if (sqlStatement instanceof CreateTableStatement createTableStatement) {
124             addTable(createTableStatement.getTable(), result);
125             createTableStatement.getLikeTable().ifPresent(table -> addTable(table, result));
126             createTableStatement.getSelectStatement().ifPresent(select -> addSelectTables(select, result));
127         } else if (sqlStatement instanceof AlterTableStatement alterTableStatement) {
128             addTable(alterTableStatement.getTable(), result);
129         } else if (sqlStatement instanceof CreateViewStatement createViewStatement) {
130             addTable(createViewStatement.getView(), result);
131         } else if (sqlStatement instanceof AlterViewStatement alterViewStatement) {
132             addTable(alterViewStatement.getView(), result);
133             alterViewStatement.getRenameView().ifPresent(view -> addTable(view, result));
134             alterViewStatement.getSelect().ifPresent(select -> addSelectTables(select, result));
135         }
136         extractIndexTargets(sqlStatement, result);
137         extractDatabaseTargets(sqlStatement, result);
138         extractSchemaTargets(sqlStatement, result);
139         extractSequenceTargets(sqlStatement, result);
140     }
141     
142     private void extractIndexTargets(final SQLStatement sqlStatement, final Collection<SQLStatementObjectName> result) {
143         if (sqlStatement instanceof CreateIndexStatement createIndexStatement) {
144             addIndex(createIndexStatement.getIndex(), result);
145         } else if (sqlStatement instanceof AlterIndexStatement alterIndexStatement) {
146             alterIndexStatement.getIndex().ifPresent(index -> addIndex(index, result));
147             alterIndexStatement.getRenameIndex().ifPresent(index -> addIndex(index, result));
148         } else if (sqlStatement instanceof DropIndexStatement dropIndexStatement) {
149             for (IndexSegment each : dropIndexStatement.getIndexes()) {
150                 addIndex(each, result);
151             }
152         }
153     }
154     
155     private void extractDatabaseTargets(final SQLStatement sqlStatement, final Collection<SQLStatementObjectName> result) {
156         if (sqlStatement instanceof CreateDatabaseStatement createDatabaseStatement) {
157             addNamespaceName(createDatabaseStatement.getDatabaseName(), result);
158         } else if (sqlStatement instanceof AlterDatabaseStatement alterDatabaseStatement) {
159             addNamespaceName(alterDatabaseStatement.getDatabaseName(), result);
160             alterDatabaseStatement.getRenameDatabaseName().ifPresent(name -> addNamespaceName(name, result));
161         } else if (sqlStatement instanceof DropDatabaseStatement dropDatabaseStatement) {
162             addNamespaceName(dropDatabaseStatement.getDatabaseName(), result);
163         }
164     }
165     
166     private void extractSchemaTargets(final SQLStatement sqlStatement, final Collection<SQLStatementObjectName> result) {
167         if (sqlStatement instanceof CreateSchemaStatement createSchemaStatement) {
168             createSchemaStatement.getSchemaName().ifPresent(schema -> addNamespaceIdentifier(schema, result));
169         } else if (sqlStatement instanceof AlterSchemaStatement alterSchemaStatement) {
170             addNamespaceIdentifier(alterSchemaStatement.getSchemaName(), result);
171             alterSchemaStatement.getRenameSchema().ifPresent(schema -> addNamespaceIdentifier(schema, result));
172         } else if (sqlStatement instanceof DropSchemaStatement dropSchemaStatement) {
173             for (IdentifierValue each : dropSchemaStatement.getSchemaNames()) {
174                 addNamespaceIdentifier(each, result);
175             }
176         }
177     }
178     
179     private void extractSequenceTargets(final SQLStatement sqlStatement, final Collection<SQLStatementObjectName> result) {
180         if (sqlStatement instanceof CreateSequenceStatement createSequenceStatement) {
181             addName(createSequenceStatement.getSequenceName(), result);
182         } else if (sqlStatement instanceof AlterSequenceStatement alterSequenceStatement) {
183             addName(alterSequenceStatement.getSequenceName(), result);
184         } else if (sqlStatement instanceof DropSequenceStatement dropSequenceStatement) {
185             for (Object each : dropSequenceStatement.getSequenceNames()) {
186                 if (each instanceof String sequenceName) {
187                     addName(sequenceName, result);
188                 } else if (each instanceof SimpleTableSegment simpleTable) {
189                     addTable(simpleTable, result);
190                 } else {
191                     throw new MCPUnsupportedSQLStatementException();
192                 }
193             }
194         }
195     }
196     
197     private void extractMergeTables(final SQLStatement sqlStatement, final Collection<SQLStatementObjectName> result) {
198         if (sqlStatement instanceof MergeStatement merge) {
199             addTableSegment(merge.getSource(), result);
200         }
201     }
202     
203     private void extractCommonTableExpressionTables(final SQLStatement sqlStatement, final Collection<SQLStatementObjectName> result) {
204         findWithSegment(sqlStatement).ifPresent(with -> {
205             for (CommonTableExpressionSegment each : with.getCommonTableExpressions()) {
206                 TableExtractor extractor = new TableExtractor();
207                 extractor.extractTablesFromSelect(each.getSubquery().getSelect());
208                 addTableSegments(extractor.getTableContext(), result);
209             }
210         });
211     }
212     
213     private void addSelectTables(final SelectStatement selectStatement, final Collection<SQLStatementObjectName> result) {
214         TableExtractor extractor = new TableExtractor();
215         extractor.extractTablesFromSelect(selectStatement);
216         addTableSegments(extractor.getTableContext(), result);
217     }
218     
219     private void removeCommonTableExpressionAliases(final SQLStatement sqlStatement, final Collection<SQLStatementObjectName> objectNames) {
220         Set<String> aliases = new HashSet<>();
221         collectCommonTableExpressionAliases(sqlStatement, aliases);
222         objectNames.removeIf(each -> !each.isQualified() && aliases.contains(each.getObjectName().toUpperCase(Locale.ENGLISH)));
223     }
224     
225     private void collectCommonTableExpressionAliases(final SQLStatement sqlStatement, final Collection<String> result) {
226         findWithSegment(sqlStatement).ifPresent(with -> {
227             for (CommonTableExpressionSegment each : with.getCommonTableExpressions()) {
228                 each.getAliasName().ifPresent(alias -> result.add(alias.toUpperCase(Locale.ENGLISH)));
229                 collectCommonTableExpressionAliases(each.getSubquery().getSelect(), result);
230             }
231         });
232         if (sqlStatement instanceof CreateTableStatement createTableStatement) {
233             createTableStatement.getSelectStatement().ifPresent(each -> collectCommonTableExpressionAliases(each, result));
234         } else if (sqlStatement instanceof CreateViewStatement createViewStatement) {
235             collectCommonTableExpressionAliases(createViewStatement.getSelect(), result);
236         } else if (sqlStatement instanceof AlterViewStatement alterViewStatement) {
237             alterViewStatement.getSelect().ifPresent(each -> collectCommonTableExpressionAliases(each, result));
238         }
239     }
240     
241     private Optional<WithSegment> findWithSegment(final SQLStatement sqlStatement) {
242         if (sqlStatement instanceof SelectStatement select) {
243             return select.getWith();
244         }
245         if (sqlStatement instanceof InsertStatement insert) {
246             return insert.getWith();
247         }
248         if (sqlStatement instanceof UpdateStatement update) {
249             return update.getWith();
250         }
251         if (sqlStatement instanceof DeleteStatement delete) {
252             return delete.getWith();
253         }
254         return sqlStatement instanceof MergeStatement merge ? merge.getWith() : Optional.empty();
255     }
256     
257     private void addTableSegment(final TableSegment tableSegment, final Collection<SQLStatementObjectName> result) {
258         if (tableSegment instanceof SimpleTableSegment simpleTable) {
259             addTable(simpleTable, result);
260         } else if (tableSegment instanceof JoinTableSegment joinTable) {
261             addTableSegment(joinTable.getLeft(), result);
262             addTableSegment(joinTable.getRight(), result);
263         } else if (tableSegment instanceof SubqueryTableSegment subqueryTable) {
264             TableExtractor extractor = new TableExtractor();
265             extractor.extractTablesFromSelect(subqueryTable.getSubquery().getSelect());
266             addTableSegments(extractor.getTableContext(), result);
267         } else if (tableSegment instanceof DeleteMultiTableSegment deleteMultiTable) {
268             addTables(deleteMultiTable.getActualDeleteTables(), result);
269             addTableSegment(deleteMultiTable.getRelationTable(), result);
270         }
271     }
272     
273     private void addTableSegments(final Collection<TableSegment> tableSegments, final Collection<SQLStatementObjectName> result) {
274         for (TableSegment each : tableSegments) {
275             addTableSegment(each, result);
276         }
277     }
278     
279     private void addTables(final Collection<SimpleTableSegment> tables, final Collection<SQLStatementObjectName> result) {
280         for (SimpleTableSegment each : tables) {
281             addTable(each, result);
282         }
283     }
284     
285     private void addTable(final SimpleTableSegment table, final Collection<SQLStatementObjectName> result) {
286         if (null == table) {
287             return;
288         }
289         result.add(SQLStatementObjectName.from(table.getOwner(), table.getTableName().getIdentifier()));
290     }
291     
292     private void addIndex(final IndexSegment index, final Collection<SQLStatementObjectName> result) {
293         result.add(SQLStatementObjectName.from(index.getOwner(), index.getIndexName().getIdentifier()));
294     }
295     
296     private void addName(final String name, final Collection<SQLStatementObjectName> result) {
297         if (null != name && !name.isEmpty()) {
298             result.add(SQLStatementObjectName.fromNormalizedName(name));
299         }
300     }
301     
302     private void addNamespaceName(final String name, final Collection<SQLStatementObjectName> result) {
303         if (null != name && !name.isEmpty()) {
304             result.add(SQLStatementObjectName.fromNormalizedNamespaceName(name));
305         }
306     }
307     
308     private void addNamespaceIdentifier(final IdentifierValue identifier, final Collection<SQLStatementObjectName> result) {
309         if (null != identifier) {
310             result.add(SQLStatementObjectName.fromNamespace(identifier));
311         }
312     }
