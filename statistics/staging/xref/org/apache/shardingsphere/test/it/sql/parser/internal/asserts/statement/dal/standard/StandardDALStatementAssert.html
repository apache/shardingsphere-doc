59  /**
60   * Standard DAL statement assert.
61   */
62  @NoArgsConstructor(access = AccessLevel.PRIVATE)
63  public final class StandardDALStatementAssert {
64      
65      /**
66       * Assert DAL statement is correct with expected parser result.
67       *
68       * @param assertContext assert context
69       * @param actual actual DAL statement
70       * @param expected expected DAL statement test case
71       */
72      public static void assertIs(final SQLCaseAssertContext assertContext, final DALStatement actual, final SQLParserTestCase expected) {
73          if (actual instanceof EmptyStatement) {
74              EmptyStatementAssert.assertIs(assertContext, (EmptyStatement) actual, (EmptyStatementTestCase) expected);
75          } else if (actual instanceof ExplainStatement) {
76              ExplainStatementAssert.assertIs(assertContext, (ExplainStatement) actual, (ExplainStatementTestCase) expected);
77          } else if (actual instanceof SetStatement) {
78              SetParameterStatementAssert.assertIs(assertContext, (SetStatement) actual, (SetParameterStatementTestCase) expected);
79          } else if (actual instanceof ShowStatement) {
80              ShowStatementAssert.assertIs(assertContext, (ShowStatement) actual, (ShowStatementTestCase) expected);
81          } else if (actual instanceof ShowBuildIndexStatement) {
82              ShowBuildIndexStatementAssert.assertIs(assertContext, (ShowBuildIndexStatement) actual, (ShowBuildIndexStatementTestCase) expected);
83          } else if (actual instanceof DorisAlterSqlBlockRuleStatement) {
84              DorisAlterSqlBlockRuleStatementAssert.assertIs(assertContext, (DorisAlterSqlBlockRuleStatement) actual, (DorisAlterSqlBlockRuleStatementTestCase) expected);
85          } else if (actual instanceof DorisDropSqlBlockRuleStatement) {
86              DorisDropSqlBlockRuleStatementAssert.assertIs(assertContext, (DorisDropSqlBlockRuleStatement) actual, (DorisDropSqlBlockRuleStatementTestCase) expected);
87          } else if (actual instanceof DorisShowSqlBlockRuleStatement) {
88              DorisShowSqlBlockRuleStatementAssert.assertIs(assertContext, (DorisShowSqlBlockRuleStatement) actual, (DorisShowSqlBlockRuleStatementTestCase) expected);
89          } else if (actual instanceof DorisShowRoutineLoadTaskStatement) {
90              DorisShowRoutineLoadTaskStatementAssert.assertIs(assertContext, (DorisShowRoutineLoadTaskStatement) actual, (DorisShowRoutineLoadTaskStatementTestCase) expected);
91          } else if (actual instanceof DorisShowRoutineLoadStatement) {
92              DorisShowRoutineLoadStatementAssert.assertIs(assertContext, (DorisShowRoutineLoadStatement) actual, (DorisShowRoutineLoadStatementTestCase) expected);
93          } else if (actual instanceof ShowAlterTableStatement) {
94              ShowAlterTableStatementAssert.assertIs(assertContext, (ShowAlterTableStatement) actual, (ShowAlterTableStatementTestCase) expected);
95          }
96      }
97  }
