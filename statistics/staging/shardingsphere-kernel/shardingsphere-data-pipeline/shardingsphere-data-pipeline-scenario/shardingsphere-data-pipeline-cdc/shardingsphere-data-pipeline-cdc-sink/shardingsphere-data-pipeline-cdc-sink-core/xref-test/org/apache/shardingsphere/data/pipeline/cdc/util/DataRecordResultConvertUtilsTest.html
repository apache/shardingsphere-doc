38  class DataRecordResultConvertUtilsTest {
39      
40      @ParameterizedTest
41      @MethodSource("dataChangeTypeTestCases")
42      void assertConvertDataRecordToRecordWithNonInsertTypes(final PipelineSQLOperationType operationType, final Record.DataChangeType expectedDataChangeType) throws InvalidProtocolBufferException {
43          DataRecord dataRecord = new DataRecord(operationType, "test_schema", "t_user", UniqueKeyIngestPosition.ofInteger(Range.closed(BigInteger.valueOf(5L), BigInteger.valueOf(10L))), 1);
44          dataRecord.addColumn(new NormalColumn("id", 1L, 2L, true, true));
45          dataRecord.setCommitTime(123L);
46          Record actualRecord = DataRecordResultConvertUtils.convertDataRecordToRecord("logic_db", "test_schema", dataRecord);
47          assertThat(actualRecord.getMetaData().getDatabase(), is("logic_db"));
48          assertThat(actualRecord.getMetaData().getSchema(), is("test_schema"));
49          assertThat(actualRecord.getMetaData().getTable(), is("t_user"));
50          assertThat(actualRecord.getTransactionCommitMillis(), is(123L));
51          assertThat(actualRecord.getDataChangeType(), is(expectedDataChangeType));
52          assertThat(actualRecord.getBefore(0).getValue().unpack(Int64Value.class).getValue(), is(1L));
53          assertThat(actualRecord.getAfter(0).getValue().unpack(Int64Value.class).getValue(), is(2L));
54      }
55      
56      private static Stream<Arguments> dataChangeTypeTestCases() {
57          return Stream.of(
58                  Arguments.of(PipelineSQLOperationType.INSERT, Record.DataChangeType.INSERT),
59                  Arguments.of(PipelineSQLOperationType.UPDATE, Record.DataChangeType.UPDATE),
60                  Arguments.of(PipelineSQLOperationType.DELETE, Record.DataChangeType.DELETE),
61                  Arguments.of(PipelineSQLOperationType.SELECT, Record.DataChangeType.UNKNOWN));
62      }
