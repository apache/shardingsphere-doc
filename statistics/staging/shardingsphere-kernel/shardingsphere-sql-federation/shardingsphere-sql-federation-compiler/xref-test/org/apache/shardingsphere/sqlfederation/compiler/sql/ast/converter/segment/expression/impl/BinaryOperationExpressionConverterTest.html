56  @ExtendWith(AutoMockExtension.class)
57  @StaticMockSettings(ExpressionConverter.class)
58  class BinaryOperationExpressionConverterTest {
59      
60      private final DatabaseType databaseType = TypedSPILoader.getService(DatabaseType.class, "FIXTURE");
61      
62      @Test
63      void assertConvertThrowsWhenIsOperatorUnsupportedLiteral() {
64          LiteralExpressionSegment left = new LiteralExpressionSegment(0, 0, 1);
65          LiteralExpressionSegment right = new LiteralExpressionSegment(0, 0, "unknown");
66          BinaryOperationExpression expression = new BinaryOperationExpression(0, 0, left, right, "IS", "");
67          assertThrows(IllegalStateException.class, () -> BinaryOperationExpressionConverter.convert(expression));
68      }
69      
70      @Test
71      void assertConvertFlattensSqlNodeListOperands() {
72          LiteralExpressionSegment left = new LiteralExpressionSegment(0, 0, 1);
73          LiteralExpressionSegment right = new LiteralExpressionSegment(0, 0, 2);
74          BinaryOperationExpression expression = new BinaryOperationExpression(0, 0, left, right, "+", "");
75          SqlNode leftNode = mock(SqlNode.class);
76          SqlNode firstRightNode = mock(SqlNode.class);
77          SqlNode secondRightNode = mock(SqlNode.class);
78          when(ExpressionConverter.convert(left)).thenReturn(Optional.of(leftNode));
79          when(ExpressionConverter.convert(right)).thenReturn(Optional.of(new SqlNodeList(Arrays.asList(firstRightNode, secondRightNode), SqlParserPos.ZERO)));
80          SqlBasicCall actual = BinaryOperationExpressionConverter.convert(expression);
81          assertThat(actual.getOperator(), is(SqlStdOperatorTable.PLUS));
82          assertThat(actual.getOperandList(), is(Arrays.asList(leftNode, firstRightNode, secondRightNode)));
83      }
84      
85      @Test
86      void assertConvertQuantifySubqueryFallsBackToOriginalOperator() {
87          LiteralExpressionSegment left = new LiteralExpressionSegment(0, 0, 1);
88          QuantifySubqueryExpression right = new QuantifySubqueryExpression(0, 0, new SubquerySegment(0, 0, SelectStatement.builder().databaseType(databaseType).build(), "sub"), "ALL");
89          SqlNode leftNode = mock(SqlNode.class);
90          SqlNode rightNode = mock(SqlNode.class);
91          when(ExpressionConverter.convert(left)).thenReturn(Optional.of(leftNode));
92          when(ExpressionConverter.convert(right)).thenReturn(Optional.of(rightNode));
93          SqlBasicCall actual = BinaryOperationExpressionConverter.convert(new BinaryOperationExpression(0, 0, left, right, "AND", ""));
94          assertThat(actual.getOperator(), is(SqlStdOperatorTable.AND));
95          assertThat(actual.getOperandList(), is(Arrays.asList(leftNode, rightNode)));
96      }
97      
98      @ParameterizedTest(name = "{0}")
99      @MethodSource("provideIsOperatorCases")
100     void assertConvertIsOperatorVariants(final String displayName, final String rightLiteral, final Supplier<SqlNode> leftSupplier,
101                                          final SqlOperator expectedOperator, final Boolean expectedBoolean) {
102         LiteralExpressionSegment left = new LiteralExpressionSegment(0, 0, 1);
103         LiteralExpressionSegment right = new LiteralExpressionSegment(0, 0, rightLiteral);
104         SqlNode leftNode = leftSupplier.get();
105         when(ExpressionConverter.convert(left)).thenReturn(Optional.of(leftNode));
106         SqlBasicCall actual = BinaryOperationExpressionConverter.convert(new BinaryOperationExpression(0, 0, left, right, "IS", ""));
107         assertThat(actual.getOperator(), is(expectedOperator));
108         if (null == expectedBoolean) {
109             assertThat(actual.getOperandList(), is(Collections.singletonList(leftNode)));
110         } else {
111             SqlLiteral operand = (SqlLiteral) actual.getOperandList().get(0);
112             assertThat(operand.getValueAs(Boolean.class), is(expectedBoolean));
113         }
114     }
115     
116     @ParameterizedTest(name = "{0} {1}")
117     @MethodSource("provideQuantifyOperators")
118     void assertConvertQuantifySubquery(final String operator, final String quantifyOperator, final SqlOperator expectedOperator) {
119         LiteralExpressionSegment left = new LiteralExpressionSegment(0, 0, 1);
120         QuantifySubqueryExpression right = new QuantifySubqueryExpression(0, 0, new SubquerySegment(0, 0, SelectStatement.builder().databaseType(databaseType).build(), "sub"), quantifyOperator);
121         SqlNode leftNode = mock(SqlNode.class);
122         SqlNode rightNode = mock(SqlNode.class);
123         when(ExpressionConverter.convert(left)).thenReturn(Optional.of(leftNode));
124         when(ExpressionConverter.convert(right)).thenReturn(Optional.of(rightNode));
125         SqlBasicCall actual = BinaryOperationExpressionConverter.convert(new BinaryOperationExpression(0, 0, left, right, operator, ""));
126         assertThat(actual.getOperator(), is(expectedOperator));
127         assertThat(actual.getOperandList(), is(Arrays.asList(leftNode, rightNode)));
128     }
129     
130     private static Stream<Arguments> provideIsOperatorCases() {
131         return Stream.of(
132                 Arguments.of("IS NULL keeps operand", "NULL", (Supplier<SqlNode>) () -> mock(SqlNode.class), SqlStdOperatorTable.IS_NULL, null),
133                 Arguments.of("IS NOT NULL keeps operand", "NOT NULL", (Supplier<SqlNode>) () -> mock(SqlNode.class), SqlStdOperatorTable.IS_NOT_NULL, null),
134                 Arguments.of("IS FALSE zero -> false", "FALSE", (Supplier<SqlNode>) () -> SqlLiteral.createExactNumeric("0", SqlParserPos.ZERO), SqlStdOperatorTable.IS_FALSE, false),
135                 Arguments.of("IS FALSE null numeric -> false", "FALSE", (Supplier<SqlNode>) () -> {
136                     SqlNumericLiteral literal = mock(SqlNumericLiteral.class);
137                     when(literal.getValueAs(Long.class)).thenReturn(null);
138                     when(literal.getParserPosition()).thenReturn(SqlParserPos.ZERO);
139                     return literal;
140                 }, SqlStdOperatorTable.IS_FALSE, false),
141                 Arguments.of("IS NOT FALSE non-zero -> true", "NOT FALSE", (Supplier<SqlNode>) () -> SqlLiteral.createExactNumeric("2", SqlParserPos.ZERO), SqlStdOperatorTable.IS_NOT_FALSE, true),
142                 Arguments.of("IS TRUE non-numeric keeps operand", "TRUE", (Supplier<SqlNode>) () -> mock(SqlNode.class), SqlStdOperatorTable.IS_TRUE, null),
143                 Arguments.of("IS NOT TRUE zero -> false", "NOT TRUE", (Supplier<SqlNode>) () -> SqlLiteral.createExactNumeric("0", SqlParserPos.ZERO), SqlStdOperatorTable.IS_NOT_TRUE, false));
144     }

146     private static Stream<Arguments> provideQuantifyOperators() {
147         return Stream.of(
148                 Arguments.of("=", "ALL", SqlStdOperatorTable.ALL_EQ),

152                 Arguments.of(">=", "ALL", SqlStdOperatorTable.ALL_GE),
153                 Arguments.of(">=", "SOME", SqlStdOperatorTable.SOME_GE),
154                 Arguments.of("<", "ALL", SqlStdOperatorTable.ALL_LT),
155                 Arguments.of("<", "SOME", SqlStdOperatorTable.SOME_LT),
156                 Arguments.of("<=", "ALL", SqlStdOperatorTable.ALL_LE),
157                 Arguments.of("<=", "SOME", SqlStdOperatorTable.SOME_LE),
158                 Arguments.of("!=", "ALL", SqlStdOperatorTable.ALL_NE),
159                 Arguments.of("<>", "SOME", SqlStdOperatorTable.SOME_NE));
160     }
