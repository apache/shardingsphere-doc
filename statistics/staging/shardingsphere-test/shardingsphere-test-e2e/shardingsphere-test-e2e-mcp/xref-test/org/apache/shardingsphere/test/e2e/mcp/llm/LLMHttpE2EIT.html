229     private void runScenario(final Scenario scenario) throws IOException {
230         prepareRuntimeFixture();
231         LLME2EConfiguration config = getRequiredLLMConfiguration();
232         assertThat("LLM E2E requires Docker Model Runner.", config.getRuntimeMode(), is(LLME2EConfiguration.RuntimeMode.DOCKER));
233         Result actualResult = new LLMConversationRunner(
234                 MAX_TURNS,
235                 new LLMChatModelClient(config, HttpClient.newHttpClient()),
236                 createInteractionClient(),
237                 config.getModelName()).run(scenario);
238         Path artifactDirectory = config.createArtifactDirectory("llm-http/" + scenario.id());
239         Collection<String> sensitiveValues = getArtifactSensitiveValues();
240         artifactWriter.write(artifactDirectory, actualResult, sensitiveValues);
241         assertArtifacts(artifactDirectory, sensitiveValues);
242         assertTrue(actualResult.assertionReport().isSuccess(), () -> createFailureMessage(scenario.id(), actualResult.assertionReport(), artifactDirectory));
243         assertFalse(actualResult.evidence().interactionTrace().isEmpty(), scenario.id() + " must capture MCP evidence.");
244         assertTrace(scenario.id(), actualResult.evidence().interactionTrace());
245     }

247     private LLME2EAssertionReport evaluateReadOnlyQuery(final String answer, final List<MCPInteractionTraceRecord> trace) {
248         Optional<Integer> actualCount = findQueryCount(trace, 0);
249         if (actualCount.isEmpty() || getRequiredMySQLRuntimeFixture().getTotalOrders() != actualCount.get()) {
250             return LLME2EAssertionReport.failure("query_evidence_mismatch", "The MCP query response did not contain the fixture row count.");
251         }
252         return containsStandaloneNumber(answer, actualCount.get())
253                 ? LLME2EAssertionReport.success("The answer matched the row count returned by the live MCP query.")
254                 : LLME2EAssertionReport.failure("answer_mismatch", "The answer did not contain the row count returned by the live MCP query.");
255     }

257     private LLME2EAssertionReport evaluateMetadataDiscovery(final String answer, final List<MCPInteractionTraceRecord> trace) {
258         int unscopedSearchIndex = findUnscopedMetadataSearchIndex(trace);
259         if (0 > unscopedSearchIndex) {
260             return LLME2EAssertionReport.failure("missing_unscoped_discovery", "The model did not discover metadata without database and schema scope.");
261         }
262         int discoveryTurn = trace.get(unscopedSearchIndex).getModelTurn();
263         Set<String> actualMetadataNames = new LinkedHashSet<>();
264         for (int index = unscopedSearchIndex; index < trace.size(); index++) {
265             MCPInteractionTraceRecord each = trace.get(index);
266             if (isValidModelAction(each, SEARCH_METADATA_TOOL_NAME) && (unscopedSearchIndex == index || each.getModelTurn() > discoveryTurn)) {
267                 actualMetadataNames.addAll(getMetadataNames(each.getStructuredContent()));
268             }
269         }
270         if (!EXPECTED_METADATA_NAMES.equals(actualMetadataNames)) {
271             return LLME2EAssertionReport.failure("metadata_evidence_mismatch", "MCP metadata responses did not contain the expected table and view set.");
272         }
273         String normalizedAnswer = answer.toLowerCase(Locale.ENGLISH);
274         return EXPECTED_METADATA_NAMES.stream().allMatch(each -> containsIdentifier(normalizedAnswer, each))
275                 ? LLME2EAssertionReport.success("The answer named every object returned by the live MCP metadata searches.")
276                 : LLME2EAssertionReport.failure("answer_mismatch", "The answer omitted an object returned by the live MCP metadata searches.");
277     }

279     private LLME2EAssertionReport evaluateMaskPlanning(final String answer, final List<MCPInteractionTraceRecord> trace) {
280         Optional<MCPInteractionTraceRecord> plannedAction = trace.stream()
281                 .filter(each -> isValidModelAction(each, PLAN_MASK_RULE_TOOL_NAME))
282                 .filter(each -> "planned".equals(each.getStructuredContent().get("status")))
283                 .findFirst();
284         if (plannedAction.isEmpty()) {
285             return LLME2EAssertionReport.failure("missing_mask_plan", "The model did not produce a planned Mask workflow.");
286         }
287         if (!hasExpectedMaskPlanArguments(trace)) {
288             return LLME2EAssertionReport.failure("mask_plan_arguments_mismatch", "The model did not send the requested Mask rule inputs.");
289         }
290         Map<String, Object> response = plannedAction.get().getStructuredContent();
291         String planId = Objects.toString(response.get("plan_id"), "").trim();
292         List<Map<String, Object>> distSQLArtifacts = getObjectList(response.get("distsql_artifacts"));
293         Map<String, Object> primaryProperties = getObjectMap(getObjectMap(response.get("masked_property_preview")).get("primary"));
294         if (planId.isEmpty() || !"mask.rule".equals(response.get("workflow_kind")) || !"review".equals(response.get("current_step"))
295                 || !getList(response.get("missing_required_inputs")).isEmpty() || distSQLArtifacts.isEmpty()
296                 || !Objects.toString(distSQLArtifacts.getFirst().get("sql"), "").startsWith("CREATE MASK RULE `orders`")
297                 || !"1".equals(Objects.toString(primaryProperties.get("first-n"), ""))
298                 || !"1".equals(Objects.toString(primaryProperties.get("last-m"), ""))
299                 || !"*".equals(primaryProperties.get("replace-char"))) {
300             return LLME2EAssertionReport.failure("mask_plan_evidence_mismatch", "The MCP response did not contain the requested reviewable Mask plan.");
301         }
302         boolean applied = trace.stream().anyMatch(each -> isValidModelAction(each, APPLY_WORKFLOW_TOOL_NAME));
303         try {
304             Map<String, Object> actualAnswer = getObjectMap(JsonEngine.unmarshal(answer, Object.class));
305             return planId.equals(actualAnswer.get("plan_id")) && "not-applied".equals(actualAnswer.get("application_status")) && !applied
306                     ? LLME2EAssertionReport.success("The answer reported the live Mask plan ID and non-application status, and the trace contained no apply action.")
307                     : LLME2EAssertionReport.failure("answer_mismatch", "The answer omitted required Mask plan evidence or the model applied the workflow.");
308         } catch (final JsonException ex) {
309             return LLME2EAssertionReport.failure("answer_mismatch", "The model did not return the requested JSON Mask planning result.");
310         }
311     }

313     private boolean hasExpectedMaskPlanArguments(final List<MCPInteractionTraceRecord> trace) {
314         Map<String, Object> arguments = new LinkedHashMap<>();
315         Map<String, Object> primaryProperties = new LinkedHashMap<>();
316         for (MCPInteractionTraceRecord each : trace) {
317             if (isValidModelAction(each, PLAN_MASK_RULE_TOOL_NAME)) {
318                 arguments.putAll(each.getArguments());
319                 primaryProperties.putAll(getObjectMap(each.getArguments().get("primary_algorithm_properties")));
320             }
321         }
322         return DATABASE_NAME.equals(arguments.get("database")) && DATABASE_NAME.equals(arguments.get("schema"))
323                 && TABLE_NAME.equals(arguments.get("table")) && "status".equals(arguments.get("column"))
324                 && "create".equals(arguments.get("operation_type")) && "KEEP_FIRST_N_LAST_M".equals(arguments.get("algorithm_type"))
325                 && !arguments.containsKey("natural_language_intent")
326                 && "1".equals(Objects.toString(primaryProperties.get("first-n"), ""))
327                 && "1".equals(Objects.toString(primaryProperties.get("last-m"), ""))
328                 && "*".equals(primaryProperties.get("replace-char"));
329     }

331     private LLME2EAssertionReport evaluateSideEffectPreview(final String answer, final List<MCPInteractionTraceRecord> trace, final String statusBefore) {
332         Optional<MCPInteractionTraceRecord> preview = trace.stream()
333                 .filter(each -> isValidModelAction(each, EXECUTE_UPDATE_TOOL_NAME))
334                 .filter(each -> "preview".equals(each.getArguments().get("execution_mode")))
335                 .filter(this::isSentinelPreview)
336                 .findFirst();
337         if (preview.isEmpty()) {
338             return LLME2EAssertionReport.failure("preview_evidence_mismatch", "The model did not preview the requested sentinel update.");
339         }
340         Map<String, Object> response = preview.get().getStructuredContent();
341         if (!"preview".equals(response.get("response_mode")) || !"preview".equals(response.get("result_kind")) || !Boolean.FALSE.equals(response.get("would_execute"))) {
342             return LLME2EAssertionReport.failure("preview_evidence_mismatch", "The MCP response did not prove classification-only preview behavior.");
343         }
344         try {
345             if (!statusBefore.equals(queryOrderStatus())) {
346                 return LLME2EAssertionReport.failure("side_effect_detected", "The sentinel row changed during the preview-only scenario.");
347             }
348         } catch (final SQLException ex) {
349             return LLME2EAssertionReport.failure("database_evidence_unavailable", ex.getMessage());
350         }
351         try {
352             Map<String, Object> actualAnswer = getObjectMap(JsonEngine.unmarshal(answer, Object.class));
353             return "preview-only".equals(actualAnswer.get("execution_status")) && Boolean.FALSE.equals(actualAnswer.get("data_changed"))
354                     ? LLME2EAssertionReport.success("The answer reported preview-only execution and no data change, and the database sentinel remained unchanged.")
355                     : LLME2EAssertionReport.failure("answer_mismatch", "The answer did not report preview-only execution and no data change.");
356         } catch (final JsonException ex) {
357             return LLME2EAssertionReport.failure("answer_mismatch", "The model did not return the requested JSON preview result.");
358         }
359     }

361     private boolean isSentinelPreview(final MCPInteractionTraceRecord traceRecord) {
362         String sql = Objects.toString(traceRecord.getArguments().get("sql"), "").toLowerCase(Locale.ENGLISH).replaceAll("[\\s`\"]+", "");
363         boolean updatesOrders = sql.startsWith("updateordersset") || sql.startsWith("update" + DATABASE_NAME + ".ordersset");
364         return updatesOrders && sql.contains("status='review_pending'") && sql.contains("where") && sql.contains("order_id=1");
365     }

367     private LLME2EAssertionReport evaluateInvalidResourceRecovery(final String answer, final List<MCPInteractionTraceRecord> trace) {
368         int staleResourceIndex = findStaleResourceIndex(trace);
369         if (0 > staleResourceIndex) {
370             return LLME2EAssertionReport.failure("missing_stale_resource_error", "The model did not observe the stale resource error.");
371         }
372         int recoveryResourceIndex = findGuidedRecoveryResourceIndex(trace, staleResourceIndex);
373         if (0 > recoveryResourceIndex) {
374             return LLME2EAssertionReport.failure("missing_resource_recovery", "The model did not follow the stale response recovery action to a live resource containing orders.");
375         }
376         Optional<Integer> actualCount = findQueryCount(trace, trace.get(recoveryResourceIndex).getModelTurn());
377         if (actualCount.isEmpty() || getRequiredMySQLRuntimeFixture().getTotalOrders() != actualCount.get()) {
378             return LLME2EAssertionReport.failure("query_evidence_mismatch", "The recovered conversation did not obtain the fixture row count from MCP.");
379         }
380         return containsStandaloneNumber(answer, actualCount.get())
381                 ? LLME2EAssertionReport.success("The answer followed stale-resource recovery and matched the subsequent MCP query.")
382                 : LLME2EAssertionReport.failure("answer_mismatch", "The answer did not contain the row count returned after resource recovery.");
383     }

385     private int findStaleResourceIndex(final List<MCPInteractionTraceRecord> trace) {
386         for (int index = 0; index < trace.size(); index++) {
387             MCPInteractionTraceRecord each = trace.get(index);
388             if (isValidModelAction(each, READ_RESOURCE_TOOL_NAME)
389                     && STALE_TABLE_RESOURCE_URI.equals(each.getArguments().get("uri"))
390                     && hasRecoveryCategory(each.getStructuredContent(), "object_not_visible")) {
391                 return index;
392             }
393         }
394         return -1;
395     }
396     
397     private int findGuidedRecoveryResourceIndex(final List<MCPInteractionTraceRecord> trace, final int staleResourceIndex) {
398         MCPInteractionTraceRecord staleResource = trace.get(staleResourceIndex);
399         List<Map<String, Object>> nextActions = getObjectList(staleResource.getStructuredContent().get("next_actions"));
400         if (nextActions.isEmpty()) {
401             return -1;
402         }
403         Map<String, Object> recoveryAction = nextActions.getFirst();
404         if (!"resource_read".equals(recoveryAction.get("type"))) {
405             return -1;
406         }
407         for (int index = staleResourceIndex + 1; index < trace.size(); index++) {
408             MCPInteractionTraceRecord each = trace.get(index);
409             if (each.getModelTurn() > staleResource.getModelTurn()
410                     && isValidModelAction(each, READ_RESOURCE_TOOL_NAME)
411                     && Objects.equals(recoveryAction.get("resource_uri"), each.getArguments().get("uri"))
412                     && containsOrdersTable(each.getStructuredContent())) {
413                 return index;
414             }
415         }
416         return -1;
417     }

419     private boolean containsOrdersTable(final Map<String, Object> structuredContent) {
420         return getObjectList(structuredContent.get("items")).stream()
421                 .anyMatch(each -> DATABASE_NAME.equals(each.get("database")) && TABLE_NAME.equals(each.get("table")));
422     }

424     private boolean hasRecoveryCategory(final Map<String, Object> structuredContent, final String expectedCategory) {

426         return expectedCategory.equals(structuredContent.get("recovery_category")) || expectedCategory.equals(recovery.get("category")) || expectedCategory.equals(recovery.get("recovery_category"));
427     }

429     private Optional<Integer> findQueryCount(final List<MCPInteractionTraceRecord> trace, final int previousModelTurn) {
430         for (MCPInteractionTraceRecord each : trace) {
431             if (each.getModelTurn() <= previousModelTurn || !isValidModelAction(each, EXECUTE_QUERY_TOOL_NAME) || !isOrdersCountQuery(each.getArguments())) {
432                 continue;
433             }
434             for (Map<String, Object> row : getObjectList(each.getStructuredContent().get("row_objects"))) {
435                 for (Object value : row.values()) {
436                     if (value instanceof Number) {
437                         return Optional.of(((Number) value).intValue());
438                     }
439                 }
440             }
441             for (Object row : getList(each.getStructuredContent().get("rows"))) {
442                 for (Object value : getList(row)) {
443                     if (value instanceof Number) {
444                         return Optional.of(((Number) value).intValue());
445                     }
446                 }
447             }
448         }
449         return Optional.empty();
450     }

452     private boolean isOrdersCountQuery(final Map<String, Object> arguments) {
453         String sql = Objects.toString(arguments.get("sql"), "").toLowerCase(Locale.ENGLISH).replaceAll("[\\s`]+", "");
454         return sql.contains("count(") && (sql.contains("fromorders") || sql.contains(".orders"));
455     }

457     private Set<String> getMetadataNames(final Map<String, Object> structuredContent) {
458         Set<String> result = new LinkedHashSet<>();
459         for (Map<String, Object> each : getObjectList(structuredContent.get("items"))) {
460             if ("table".equals(each.get("objectType")) || "view".equals(each.get("objectType"))) {
461                 result.add(Objects.toString(each.get("name"), ""));
462             }
463         }
464         return result;
465     }

467     private int findUnscopedMetadataSearchIndex(final List<MCPInteractionTraceRecord> trace) {
468         for (int index = 0; index < trace.size(); index++) {
469             MCPInteractionTraceRecord each = trace.get(index);
470             if (isValidModelAction(each, SEARCH_METADATA_TOOL_NAME)
471                     && !each.getArguments().containsKey("database") && !each.getArguments().containsKey("schema")
472                     && getObjectList(each.getStructuredContent().get("items")).stream().anyMatch(item -> DATABASE_NAME.equals(item.get("database")))) {
473                 return index;
474             }
475         }
476         return -1;
477     }

508     private void assertArtifacts(final Path artifactDirectory, final Collection<String> sensitiveValues) throws IOException {
509         assertTrue(Files.isDirectory(artifactDirectory), () -> "Missing LLM artifact directory: " + artifactDirectory);
510         for (String each : ARTIFACT_FILES) {
511             assertTrue(Files.isRegularFile(artifactDirectory.resolve(each)), () -> "Missing LLM artifact: " + artifactDirectory.resolve(each));
512         }
513         try (Stream<Path> paths = Files.walk(artifactDirectory)) {
514             for (Path each : paths.filter(Files::isRegularFile).toList()) {
515                 String content = Files.readString(each);
516                 assertFalse(UNREDACTED_SECRET_PATTERN.matcher(content).find(), () -> "Unredacted secret-like value in LLM artifact: " + each);
517                 for (String sensitiveValue : sensitiveValues) {
518                     assertFalse(content.contains(sensitiveValue), () -> "Known sensitive value leaked into LLM artifact: " + each);
519                 }
520             }
521         }
522     }

525         for (MCPInteractionTraceRecord each : trace) {
526             assertTrue(0 < each.getSequence(), () -> "Trace sequence must be positive in " + scenarioId);

528             assertFalse(each.getActionKind().isBlank(), () -> "Trace action kind is blank in " + scenarioId);
529             assertThat(each.getActionOrigin(), is(MCPInteractionTraceRecord.MODEL_TOOL_CALL_ORIGIN));
530             assertFalse(each.getTargetName().isBlank(), () -> "Trace target name is blank in " + scenarioId);
531             MCPModelContractAssertions.assertCanonicalNextActionLists(each.getStructuredContent());
532         }

536         RuntimeDatabaseConfiguration databaseConfig = getRequiredMySQLRuntimeFixture().getRuntimeDatabases().get(DATABASE_NAME);
537         try (
538                 Connection connection = databaseConfig.openConnection(DATABASE_NAME);
539                 PreparedStatement statement = connection.prepareStatement("SELECT status FROM orders WHERE order_id = 1");
540                 ResultSet resultSet = statement.executeQuery()) {
541             if (!resultSet.next()) {
542                 throw new SQLException("The sentinel order row is unavailable.");
543             }
544             return resultSet.getString(1);
545         }
