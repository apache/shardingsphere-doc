18  package org.apache.shardingsphere.database.connector.core.metadata.identifier;
19  
20  import lombok.AccessLevel;
21  import lombok.NoArgsConstructor;
22  import org.apache.shardingsphere.database.connector.core.metadata.database.enums.QuoteCharacter;
23  import org.apache.shardingsphere.database.connector.core.spi.DatabaseTypedSPILoader;
24  import org.apache.shardingsphere.database.connector.core.type.DatabaseType;
25  
26  import javax.sql.DataSource;
27  import java.util.Collection;
28  import java.util.Optional;
29  
30  /**
31   * Identifier normalize engine.
32   */
33  @NoArgsConstructor(access = AccessLevel.PRIVATE)
34  public final class IdentifierNormalizeEngine {
35      
36      /**
37       * Resolve identifier case policy.
38       *
39       * @param databaseType database type
40       * @param dataSource data source
41       * @param identifierScope identifier scope
42       * @return identifier case policy
43       */
44      public static IdentifierCasePolicy resolvePolicy(final DatabaseType databaseType, final DataSource dataSource, final IdentifierScope identifierScope) {
45          return DatabaseTypedSPILoader.findService(IdentifierCasePolicyProvider.class, databaseType)
46                  .map(each -> each.provide(new IdentifierCasePolicyProviderContext(databaseType, dataSource)))
47                  .orElseGet(IdentifierCasePolicyFactory::newInsensitivePolicySet)
48                  .getPolicy(identifierScope);
49      }
50      
51      /**
52       * Normalize identifier.
53       *
54       * @param policy identifier case policy
55       * @param identifier identifier
56       * @return normalized identifier
57       */
58      public static String normalize(final IdentifierCasePolicy policy, final String identifier) {
59          QuoteCharacter quoteCharacter = QuoteCharacter.getQuoteCharacter(identifier);
60          String unwrappedIdentifier = quoteCharacter.unwrap(identifier);
61          return policy.normalizeForDefinition(unwrappedIdentifier, quoteCharacter);
62      }
63      
64      /**
65       * Find matched stored identifier.
66       *
67       * @param storedNames stored identifier names
68       * @param policy identifier case policy
69       * @param identifier identifier
70       * @return matched stored identifier
71       */
72      public static Optional<String> findMatchedIdentifier(final Collection<String> storedNames, final IdentifierCasePolicy policy, final String identifier) {
73          QuoteCharacter quoteCharacter = QuoteCharacter.getQuoteCharacter(identifier);
74          String unwrappedIdentifier = quoteCharacter.unwrap(identifier);
75          return storedNames.stream().filter(each -> policy.matches(each, unwrappedIdentifier, quoteCharacter)).findFirst();
76      }
77  }
