53      @ParameterizedTest(name = "{0}")
54      @MethodSource("putArguments")
55      void assertPut(final String name, final String inputDataSourceName, final Collection<String> initialTables, final Collection<String> expectedTables, final boolean expectedDataNodePresent) {
56          SingleRuleConfiguration ruleConfig = new SingleRuleConfiguration(new LinkedList<>(initialTables), null);
57          Map<String, Collection<DataNode>> singleTableDataNodes = createSingleTableDataNodes(Collections.emptyList());
58          SingleTableMapperRuleAttribute tableMapperRuleAttribute = new SingleTableMapperRuleAttribute(singleTableDataNodes.values());
59          SingleMutableDataNodeRuleAttribute ruleAttribute = createRuleAttribute(ruleConfig, Collections.singleton("foo_ds"), singleTableDataNodes, tableMapperRuleAttribute);
60          ruleAttribute.put(inputDataSourceName, "foo_schema", "foo_tbl");
61          assertThat(ruleConfig.getTables(), is(new LinkedList<>(expectedTables)));
62          if (expectedDataNodePresent) {
63              assertTrue(singleTableDataNodes.containsKey("foo_tbl"));
64              assertTrue(tableMapperRuleAttribute.getLogicTableNames().contains("foo_tbl"));
65              DataNode actualDataNode = singleTableDataNodes.get("foo_tbl").iterator().next();
66              assertThat(actualDataNode.getDataSourceName(), is(inputDataSourceName));
67              assertThat(actualDataNode.getSchemaName(), is("foo_schema"));
68              assertThat(actualDataNode.getTableName(), is("foo_tbl"));
69          } else {
70              assertFalse(singleTableDataNodes.containsKey("foo_tbl"));
71              assertFalse(tableMapperRuleAttribute.getLogicTableNames().contains("foo_tbl"));
72          }
73      }

77          DataNode dataNode = new DataNode("foo_ds", "foo_schema", "foo_tbl");

81          SingleTableMapperRuleAttribute tableMapperRuleAttribute = new SingleTableMapperRuleAttribute(singleTableDataNodes.values());
82          SingleMutableDataNodeRuleAttribute ruleAttribute = createRuleAttribute(ruleConfig, Collections.singleton("foo_ds"), singleTableDataNodes, tableMapperRuleAttribute);
83          ruleAttribute.remove("foo_schema", "foo_tbl");
84          assertFalse(singleTableDataNodes.containsKey("foo_tbl"));
85          assertFalse(ruleConfig.getTables().contains(SingleTableLoadUtils.getDataNodeString(DATABASE_TYPE, "foo_ds", "foo_schema", "foo_tbl")));
86          assertFalse(tableMapperRuleAttribute.getLogicTableNames().contains("foo_tbl"));

104     @ParameterizedTest(name = "{0}")
105     @MethodSource("removeWithSchemaNamesArguments")
106     void assertRemoveWithSchemaNames(final String name, final Collection<DataNode> initialDataNodes, final Collection<String> schemaNames,
107                                      final int expectedDataNodeCount, final Collection<String> expectedTables, final boolean expectedTableInMap, final boolean expectedLogicTableExists) {
108         SingleRuleConfiguration configuration = new SingleRuleConfiguration(new LinkedList<>(new LinkedHashSet<>(createDataNodeStrings(initialDataNodes))), null);
109         Map<String, Collection<DataNode>> singleTableDataNodes = createSingleTableDataNodes(initialDataNodes);
110         SingleTableMapperRuleAttribute tableMapperRuleAttribute = new SingleTableMapperRuleAttribute(singleTableDataNodes.values());
111         SingleMutableDataNodeRuleAttribute ruleAttribute = createRuleAttribute(configuration, Collections.singleton("foo_ds"), singleTableDataNodes, tableMapperRuleAttribute);
112         ruleAttribute.remove(schemaNames, "foo_tbl");
113         assertThat(new LinkedList<>(configuration.getTables()), is(new LinkedList<>(expectedTables)));
114         if (expectedTableInMap) {
115             assertTrue(singleTableDataNodes.containsKey("foo_tbl"));
116             assertThat(singleTableDataNodes.get("foo_tbl").size(), is(expectedDataNodeCount));
117         } else {
118             assertFalse(singleTableDataNodes.containsKey("foo_tbl"));
119         }
120         assertThat(tableMapperRuleAttribute.getLogicTableNames().contains("foo_tbl"), is(expectedLogicTableExists));
121     }

146         assertTrue(singleTableDataNodes.isEmpty());
147         assertTrue(ruleConfig.getTables().isEmpty());
148         assertTrue(tableMapperRuleAttribute.getLogicTableNames().isEmpty());

152     @MethodSource("findTableDataNodeArguments")
153     void assertFindTableDataNode(final String name, final Collection<DataNode> initialDataNodes, final String schemaName, final String tableName,
154                                  final boolean expectedPresent, final String expectedDataSourceName) {
155         SingleRuleConfiguration ruleConfig = new SingleRuleConfiguration(new LinkedList<>(createDataNodeStrings(initialDataNodes)), null);
156         Map<String, Collection<DataNode>> singleTableDataNodes = createSingleTableDataNodes(initialDataNodes);
157         SingleTableMapperRuleAttribute tableMapperRuleAttribute = new SingleTableMapperRuleAttribute(singleTableDataNodes.values());
158         SingleMutableDataNodeRuleAttribute ruleAttribute = createRuleAttribute(ruleConfig, Collections.singleton("foo_ds"), singleTableDataNodes, tableMapperRuleAttribute);
159         Optional<DataNode> actual = ruleAttribute.findTableDataNode(schemaName, tableName);
160         if (expectedPresent) {
161             assertTrue(actual.isPresent());
162             assertThat(actual.get().getDataSourceName(), is(expectedDataSourceName));
163             assertThat(actual.get().getTableName(), is("foo_tbl"));
164         } else {
165             assertFalse(actual.isPresent());
166         }

192     private static Stream<Arguments> putArguments() {
193         String allTablesNode = SingleTableLoadUtils.getAllTablesNodeStr(DATABASE_TYPE);
194         String allTablesNodeFromDataSource = SingleTableLoadUtils.getAllTablesNodeStrFromDataSource(DATABASE_TYPE, "foo_ds", "foo_schema");
195         String dataNodeString = SingleTableLoadUtils.getDataNodeString(DATABASE_TYPE, "foo_ds", "foo_schema", "foo_tbl");
196         String existingDataNodeString = SingleTableLoadUtils.getDataNodeString(DATABASE_TYPE, "bar_ds", "bar_schema", "bar_table");
197         return Stream.of(
198                 Arguments.of("skip unmanaged data source", "bar_ds",
199                         Collections.singleton(existingDataNodeString), Collections.singleton(existingDataNodeString), false),
200                 Arguments.of("skip table config when all tables wildcard exists", "foo_ds",
201                         Collections.singleton(allTablesNode), Collections.singleton(allTablesNode), true),
202                 Arguments.of("skip table config when data source wildcard exists", "foo_ds",
203                         Collections.singleton(allTablesNodeFromDataSource), Collections.singleton(allTablesNodeFromDataSource), true),
204                 Arguments.of("append table config when absent", "foo_ds",
205                         Collections.singleton(existingDataNodeString), Arrays.asList(existingDataNodeString, dataNodeString), true),
206                 Arguments.of("do not append duplicated table config", "foo_ds",
207                         Collections.singleton(dataNodeString), Collections.singleton(dataNodeString), true));
208     }

210     private static Stream<Arguments> removeWithSchemaNamesArguments() {
211         DataNode fooDataNode = new DataNode("foo_ds", "foo_schema", "foo_tbl");
212         DataNode barDataNode = new DataNode("foo_ds", "bar_schema", "foo_tbl");
213         DataNode otherDataSourceNode = new DataNode("bar_ds", "bar_schema", "foo_tbl");
214         return Stream.of(
215                 Arguments.of("keep data node when schema not matched", Collections.singleton(fooDataNode), Collections.singleton("baz_schema"), 1,
216                         Collections.singleton(SingleTableLoadUtils.getDataNodeString(DATABASE_TYPE, "foo_ds", "foo_schema", "foo_tbl")), true, true),
217                 Arguments.of("keep data node when schema differs only by case", Collections.singleton(fooDataNode), Collections.singleton("FOO_SCHEMA"), 1,
218                         Collections.singleton(SingleTableLoadUtils.getDataNodeString(DATABASE_TYPE, "foo_ds", "foo_schema", "foo_tbl")), true, true),
219                 Arguments.of("remove matched schema and keep others", Arrays.asList(fooDataNode, barDataNode), Collections.singleton("foo_schema"), 1,
220                         Collections.singleton(SingleTableLoadUtils.getDataNodeString(DATABASE_TYPE, "foo_ds", "bar_schema", "foo_tbl")), true, true),
221                 Arguments.of("remove matched schema and keep another data source", Arrays.asList(fooDataNode, otherDataSourceNode), Collections.singleton("foo_schema"), 1,
222                         Collections.singleton(SingleTableLoadUtils.getDataNodeString(DATABASE_TYPE, "bar_ds", "bar_schema", "foo_tbl")), true, true),
223                 Arguments.of("remove all matched schemas", Collections.singleton(fooDataNode), Collections.singleton("foo_schema"), 0, Collections.emptyList(), false, false));
224     }

226     private static Stream<Arguments> findTableDataNodeArguments() {
227         DataNode dataNode = new DataNode("foo_ds", "foo_schema", "foo_tbl");
228         return Stream.of(
229                 Arguments.of("return data node when schema matched", Collections.singleton(dataNode), "foo_schema", "foo_tbl".toUpperCase(), true, "foo_ds"),
230                 Arguments.of("return empty when schema differs only by case", Collections.singleton(dataNode), "FOO_SCHEMA", "foo_tbl", true, "foo_ds"),
231                 Arguments.of("return empty when schema mismatched", Collections.singleton(dataNode), "bar_schema", "foo_tbl", false, ""),
232                 Arguments.of("return empty when table not exists", Collections.emptyList(), "foo_schema", "bar_table", false, ""));
233     }

235     private static Map<String, Collection<DataNode>> createSingleTableDataNodes(final Collection<DataNode> initialDataNodes) {
236         Map<String, Collection<DataNode>> result = new LinkedHashMap<>(1, 1F);
237         if (!initialDataNodes.isEmpty()) {
238             result.put("foo_tbl", new LinkedHashSet<>(initialDataNodes));
239         }
240         return result;
241     }
242     
243     private static SingleMutableDataNodeRuleAttribute createRuleAttribute(final SingleRuleConfiguration configuration, final Collection<String> dataSourceNames,
244                                                                           final Map<String, Collection<DataNode>> singleTableDataNodes,
245                                                                           final SingleTableMapperRuleAttribute tableMapperRuleAttribute) {
246         return new SingleMutableDataNodeRuleAttribute(mock(SingleRule.class), configuration, new LinkedHashSet<>(dataSourceNames), singleTableDataNodes, DATABASE_TYPE, tableMapperRuleAttribute);
247     }
248     
249     private static Collection<String> createDataNodeStrings(final Collection<DataNode> dataNodes) {
250         return dataNodes.stream().map(each -> SingleTableLoadUtils.getDataNodeString(DATABASE_TYPE, each.getDataSourceName(), each.getSchemaName(), each.getTableName())).collect(Collectors.toList());
251     }
