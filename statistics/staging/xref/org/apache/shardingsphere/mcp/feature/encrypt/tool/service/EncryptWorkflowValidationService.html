49      private final WorkflowValidationSupport validationSupport = new WorkflowValidationSupport();

51      private final EncryptRuleInspectionService ruleInspectionService = new EncryptRuleInspectionService();
52      
53      @Override
54      public ValidationReport validate(final WorkflowContextSnapshot snapshot, final MCPFeatureQueryFacade queryFacade) {
55          ValidationReport result = new ValidationReport();
56          EncryptWorkflowRequest request = (EncryptWorkflowRequest) snapshot.getRequest();
57          List<Map<String, Object>> encryptRules = ruleInspectionService.queryEncryptRules(queryFacade, request.getDatabase(), request.getTable());
58          queryFacade.checkDatabaseCapability(request.getDatabase());
59          result.setRuleValidation(validateRules(snapshot, request, encryptRules, result, queryFacade));
60          result.setOverallStatus(validationSupport.resolveOverallStatus(result.getRuleValidation()));
61          return result;
62      }
63      
64      private ValidationSection validateRules(final WorkflowContextSnapshot snapshot,
65                                              final EncryptWorkflowRequest request, final List<Map<String, Object>> encryptRules, final ValidationReport validationReport,
66                                              final MCPFeatureQueryFacade queryFacade) {
67          Optional<List<Map<String, Object>>> expectedRules = getExpectedRules(snapshot);
68          if (expectedRules.isPresent()) {
69              return validateExpectedRules(snapshot, expectedRules.get(), encryptRules, validationReport, queryFacade);
70          }
71          Optional<Map<String, Object>> actualRule = findEncryptRule(snapshot, encryptRules, queryFacade);
72          if (WorkflowLifecycleUtils.isDropWorkflow(snapshot)) {
73              if (actualRule.isEmpty()) {
74                  return new ValidationSection(WorkflowLifecycle.STATUS_PASSED, List.of(), "Encrypt rule has been removed.");
75              }
76              Map<String, Object> maskedActualRule = createMaskedRules(snapshot, List.of(actualRule.get())).getFirst();
77              validationReport.getMismatches().add(validationSupport.createMismatch(WorkflowIssueCode.RULE_STATE_MISMATCH, "rule", "no encrypt rule",
78                      String.valueOf(maskedActualRule),
79                      "Encrypt rule still exists after drop.", "Drop the encrypt rule again or investigate the failure."));
80              return new ValidationSection(WorkflowLifecycle.STATUS_FAILED, maskedActualRule, "Encrypt rule still exists.");
81          }
82          if (actualRule.isEmpty()) {
83              validationReport.getMismatches().add(validationSupport.createMismatch(WorkflowIssueCode.RULE_STATE_MISMATCH, "rule", snapshot.getRequest().getColumn(), "",
84                      "Encrypt rule is missing.", "Generate the encrypt rule artifact again and re-run validation."));
85              return new ValidationSection(WorkflowLifecycle.STATUS_FAILED, List.of(), "Encrypt rule is missing.");
86          }
87          Map<String, Object> actualRuleValue = actualRule.get();
88          List<Map<String, Object>> mismatches = new LinkedList<>();
89          addIdentifierMismatch(mismatches, queryFacade, request.getDatabase(), "cipher_column", request.getOptions().getCipherColumnName(),
90                  WorkflowRuleValueUtils.getRuleValue(actualRuleValue, "cipher_column"), "Cipher column mapping does not match.");
91          addIdentifierMismatch(mismatches, queryFacade, request.getDatabase(), "assisted_query_column",
92                  Boolean.TRUE.equals(request.getOptions().getRequiresEqualityFilter()) ? request.getOptions().getAssistedQueryColumnName() : "",
93                  WorkflowRuleValueUtils.getRuleValue(actualRuleValue, "assisted_query_column"), "Assisted-query column mapping does not match.");
94          addIdentifierMismatch(mismatches, queryFacade, request.getDatabase(), "like_query_column",
95                  Boolean.TRUE.equals(request.getOptions().getRequiresLikeQuery()) ? request.getOptions().getLikeQueryColumnName() : "",
96                  WorkflowRuleValueUtils.getRuleValue(actualRuleValue, "like_query_column"), "LIKE-query column mapping does not match.");
97          addAlgorithmTypeMismatch(mismatches, "encryptor_type", request.getAlgorithmType(),
98                  WorkflowRuleValueUtils.getRuleValue(actualRuleValue, "encryptor_type"), "Encrypt algorithm type does not match.");
99          addAlgorithmTypeMismatch(mismatches, "assisted_query_type",
100                 Boolean.TRUE.equals(request.getOptions().getRequiresEqualityFilter()) ? request.getOptions().getAssistedQueryAlgorithmType() : "",
101                 WorkflowRuleValueUtils.getRuleValue(actualRuleValue, "assisted_query_type"), "Assisted-query algorithm type does not match.");
102         addAlgorithmTypeMismatch(mismatches, "like_query_type",
103                 Boolean.TRUE.equals(request.getOptions().getRequiresLikeQuery()) ? request.getOptions().getLikeQueryAlgorithmType() : "",
104                 WorkflowRuleValueUtils.getRuleValue(actualRuleValue, "like_query_type"), "LIKE-query algorithm type does not match.");
105         Map<String, Object> maskedActualRule = createMaskedRules(snapshot, List.of(actualRuleValue)).getFirst();
106         if (!mismatches.isEmpty()) {
107             validationReport.getMismatches().addAll(mismatches);
108             return new ValidationSection(WorkflowLifecycle.STATUS_FAILED, maskedActualRule, "Encrypt rule configuration does not match.");
109         }
110         return new ValidationSection(WorkflowLifecycle.STATUS_PASSED, maskedActualRule, createPassedRuleMessage(snapshot));
111     }
112     
113     private Optional<List<Map<String, Object>>> getExpectedRules(final WorkflowContextSnapshot snapshot) {
114         return Optional.ofNullable(snapshot.getFeatureData()).map(RuleWorkflowFeatureData::getExpectedRules);
115     }
116     
117     private ValidationSection validateExpectedRules(final WorkflowContextSnapshot snapshot, final List<Map<String, Object>> expectedRules, final List<Map<String, Object>> actualRules,
118                                                     final ValidationReport validationReport, final MCPFeatureQueryFacade queryFacade) {
119         List<Map<String, Object>> mismatches = createExpectedRuleMismatches(snapshot, expectedRules, actualRules, queryFacade);
120         if (!mismatches.isEmpty()) {
121             validationReport.getMismatches().addAll(mismatches);
122             return new ValidationSection(WorkflowLifecycle.STATUS_FAILED, createMaskedRules(snapshot, actualRules), "Encrypt table rule state does not match the planned state.");
123         }
124         return new ValidationSection(WorkflowLifecycle.STATUS_PASSED, createMaskedRules(snapshot, actualRules), createPassedRuleStateMessage(snapshot));
125     }
126     
127     private String createPassedRuleMessage(final WorkflowContextSnapshot snapshot) {
128         return WorkflowSecretReferenceUtils.hasSecretReferences(snapshot.getRequest())
129                 ? "Encrypt rule matches the planned non-sensitive columns and algorithms; sensitive properties are present and masked."
130                 : "Encrypt rule matches the planned columns and algorithms.";
131     }
132     
133     private String createPassedRuleStateMessage(final WorkflowContextSnapshot snapshot) {
134         return WorkflowSecretReferenceUtils.hasSecretReferences(snapshot.getRequest())
135                 ? "Encrypt table rule state matches the planned non-sensitive state; sensitive properties are present and masked."
136                 : "Encrypt table rule state matches the planned state.";
137     }
138     
139     private List<Map<String, Object>> createExpectedRuleMismatches(final WorkflowContextSnapshot snapshot, final List<Map<String, Object>> expectedRules,
140                                                                    final List<Map<String, Object>> actualRules, final MCPFeatureQueryFacade queryFacade) {
141         List<Map<String, Object>> result = new LinkedList<>();
142         String databaseName = snapshot.getRequest().getDatabase();
143         for (Map<String, Object> each : expectedRules) {
144             String expectedColumn = WorkflowRuleValueUtils.getRuleValue(each, "logic_column");
145             Optional<Map<String, Object>> actualRule = findRuleByColumn(actualRules, queryFacade, databaseName, expectedColumn);
146             if (actualRule.isEmpty()) {
147                 result.add(validationSupport.createMismatch(WorkflowIssueCode.RULE_STATE_MISMATCH, "rule", formatFieldValue("logic_column", expectedColumn), "",
148                         "Expected encrypt rule column is missing.", "Re-apply the intended encrypt rule."));
149                 continue;
150             }
151             addExpectedRuleValueMismatches(result, snapshot, each, actualRule.get(), queryFacade);
152         }
153         for (Map<String, Object> each : actualRules) {
154             String actualColumn = WorkflowRuleValueUtils.getRuleValue(each, "logic_column");
155             if (findRuleByColumn(expectedRules, queryFacade, databaseName, actualColumn).isEmpty()) {
156                 result.add(validationSupport.createMismatch(WorkflowIssueCode.RULE_STATE_MISMATCH, "rule", "no extra encrypt rule column",
157                         formatFieldValue("logic_column", actualColumn), "Unexpected encrypt rule column exists.", "Inspect concurrent rule changes before retrying validation."));
158             }
159         }
160         return result;
161     }
162     
163     private Optional<Map<String, Object>> findRuleByColumn(final List<Map<String, Object>> rules, final MCPFeatureQueryFacade queryFacade,
164                                                            final String databaseName, final String column) {
165         return rules.stream()
166                 .filter(each -> queryFacade.isSameIdentifier(databaseName, IdentifierScope.COLUMN, column, WorkflowRuleValueUtils.getRuleValue(each, "logic_column"))).findFirst();
167     }
168     
169     private void addExpectedRuleValueMismatches(final List<Map<String, Object>> mismatches, final WorkflowContextSnapshot snapshot, final Map<String, Object> expectedRule,
170                                                 final Map<String, Object> actualRule, final MCPFeatureQueryFacade queryFacade) {
171         String databaseName = snapshot.getRequest().getDatabase();
172         addIdentifierMismatch(mismatches, queryFacade, databaseName, "cipher_column", WorkflowRuleValueUtils.getRuleValue(expectedRule, "cipher_column"),
173                 WorkflowRuleValueUtils.getRuleValue(actualRule, "cipher_column"), "Cipher column mapping does not match.");
174         addIdentifierMismatch(mismatches, queryFacade, databaseName, "assisted_query_column", WorkflowRuleValueUtils.getRuleValue(expectedRule, "assisted_query_column"),
175                 WorkflowRuleValueUtils.getRuleValue(actualRule, "assisted_query_column"), "Assisted-query column mapping does not match.");
176         addIdentifierMismatch(mismatches, queryFacade, databaseName, "like_query_column", WorkflowRuleValueUtils.getRuleValue(expectedRule, "like_query_column"),
177                 WorkflowRuleValueUtils.getRuleValue(actualRule, "like_query_column"), "LIKE-query column mapping does not match.");
178         addAlgorithmTypeMismatch(mismatches, "encryptor_type", WorkflowRuleValueUtils.getRuleValue(expectedRule, "encryptor_type"),
179                 WorkflowRuleValueUtils.getRuleValue(actualRule, "encryptor_type"), "Encrypt algorithm type does not match.");
180         addAlgorithmTypeMismatch(mismatches, "assisted_query_type", WorkflowRuleValueUtils.getRuleValue(expectedRule, "assisted_query_type"),
181                 WorkflowRuleValueUtils.getRuleValue(actualRule, "assisted_query_type"), "Assisted-query algorithm type does not match.");
182         addAlgorithmTypeMismatch(mismatches, "like_query_type", WorkflowRuleValueUtils.getRuleValue(expectedRule, "like_query_type"),
183                 WorkflowRuleValueUtils.getRuleValue(actualRule, "like_query_type"), "LIKE-query algorithm type does not match.");
184         addPropertyMismatch(mismatches, snapshot, "encryptor_props", expectedRule.get("encryptor_props"), actualRule.get("encryptor_props"), "Encrypt algorithm properties do not match.");
185         addPropertyMismatch(mismatches, snapshot, "assisted_query_props", expectedRule.get("assisted_query_props"), actualRule.get("assisted_query_props"),
186                 "Assisted-query algorithm properties do not match.");
187         addPropertyMismatch(mismatches, snapshot, "like_query_props", expectedRule.get("like_query_props"), actualRule.get("like_query_props"),
188                 "LIKE-query algorithm properties do not match.");
189     }

191     private void addIdentifierMismatch(final List<Map<String, Object>> mismatches, final MCPFeatureQueryFacade queryFacade, final String databaseName,
192                                        final String fieldName, final String expected, final String actual, final String impact) {
193         if (queryFacade.isSameIdentifier(databaseName, IdentifierScope.COLUMN, expected, actual)) {
194             return;
195         }
196         mismatches.add(validationSupport.createMismatch(WorkflowIssueCode.RULE_STATE_MISMATCH, "rule", formatFieldValue(fieldName, expected), formatFieldValue(fieldName, actual), impact,
197                 "Re-apply the intended encrypt rule."));
198     }
199     
200     private void addPropertyMismatch(final List<Map<String, Object>> mismatches, final WorkflowContextSnapshot snapshot, final String fieldName,
201                                      final Object expected, final Object actual, final String impact) {
202         Map<String, String> expectedProperties = WorkflowAlgorithmUtils.createPropertyMap(expected);
203         Map<String, String> actualProperties = WorkflowAlgorithmUtils.createPropertyMap(actual);
204         String algorithmRole = getAlgorithmRole(fieldName);
205         if (WorkflowSecretReferenceUtils.matchesManualPlaceholderProperties(expectedProperties, actualProperties, snapshot.getRequest(), algorithmRole)) {
206             return;
207         }
208         mismatches.add(validationSupport.createMismatch(WorkflowIssueCode.RULE_STATE_MISMATCH, "rule",
209                 formatFieldValue(fieldName, WorkflowArtifactMaskUtils.maskPropertyMap(expectedProperties, snapshot.getPropertyRequirements(), snapshot.getRequest(),
210                         algorithmRole)),
211                 formatFieldValue(fieldName, WorkflowArtifactMaskUtils.maskPropertyMap(actualProperties, snapshot.getPropertyRequirements(), snapshot.getRequest(),
212                         algorithmRole)),
213                 impact,
214                 "Re-apply the intended encrypt rule."));
215     }
216     
217     private List<Map<String, Object>> createMaskedRules(final WorkflowContextSnapshot snapshot, final List<Map<String, Object>> rules) {
218         List<Map<String, Object>> result = new LinkedList<>();
219         for (Map<String, Object> each : rules) {
220             Map<String, Object> rule = new LinkedHashMap<>(each);
221             rule.put("encryptor_props", WorkflowArtifactMaskUtils.maskPropertyMap(WorkflowAlgorithmUtils.createPropertyMap(each.get("encryptor_props")), snapshot.getPropertyRequirements(),
222                     snapshot.getRequest(), EncryptFeatureDefinition.ALGORITHM_ROLE_PRIMARY));
223             rule.put("assisted_query_props",
224                     WorkflowArtifactMaskUtils.maskPropertyMap(WorkflowAlgorithmUtils.createPropertyMap(each.get("assisted_query_props")), snapshot.getPropertyRequirements(),
225                             snapshot.getRequest(), EncryptFeatureDefinition.ALGORITHM_ROLE_ASSISTED_QUERY));
226             rule.put("like_query_props", WorkflowArtifactMaskUtils.maskPropertyMap(WorkflowAlgorithmUtils.createPropertyMap(each.get("like_query_props")), snapshot.getPropertyRequirements(),
227                     snapshot.getRequest(), EncryptFeatureDefinition.ALGORITHM_ROLE_LIKE_QUERY));
228             result.add(rule);
229         }
230         return result;
231     }
232     
233     private Optional<Map<String, Object>> findEncryptRule(final WorkflowContextSnapshot snapshot, final List<Map<String, Object>> encryptRules,
234                                                           final MCPFeatureQueryFacade queryFacade) {
235         return encryptRules.stream()
236                 .filter(each -> queryFacade.isSameIdentifier(snapshot.getRequest().getDatabase(), IdentifierScope.COLUMN, snapshot.getRequest().getColumn(),
237                         WorkflowRuleValueUtils.getRuleValue(each, "logic_column")))
238                 .findFirst();
239     }
240     
241     private void addAlgorithmTypeMismatch(final List<Map<String, Object>> mismatches, final String fieldName, final String expected, final String actual, final String impact) {
242         if (matchesValue(expected, actual)) {
243             return;
244         }
245         mismatches.add(validationSupport.createMismatch(WorkflowIssueCode.RULE_STATE_MISMATCH, "rule", formatFieldValue(fieldName, expected), formatFieldValue(fieldName, actual), impact,
246                 "Re-apply the intended encrypt rule."));
247     }
248     
249     private boolean matchesValue(final String expected, final String actual) {
250         return expected.equalsIgnoreCase(actual);
251     }
252     
253     private String getAlgorithmRole(final String fieldName) {
254         if ("assisted_query_props".equals(fieldName)) {
255             return EncryptFeatureDefinition.ALGORITHM_ROLE_ASSISTED_QUERY;
256         }
257         if ("like_query_props".equals(fieldName)) {
258             return EncryptFeatureDefinition.ALGORITHM_ROLE_LIKE_QUERY;
259         }
260         return EncryptFeatureDefinition.ALGORITHM_ROLE_PRIMARY;
261     }
262     
263     private String formatFieldValue(final String fieldName, final Object value) {
264         return String.format("%s=%s", fieldName, value);
265     }
