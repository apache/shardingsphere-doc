163     @Test
164     void assertResetRateLimitWhenSessionCloses() {
165         String previous = System.getProperty(MCPRuntimeProtectionPolicy.MAX_COMPLETION_REQUESTS_PER_MINUTE_PROPERTY);
166         try {
167             System.setProperty(MCPRuntimeProtectionPolicy.MAX_COMPLETION_REQUESTS_PER_MINUTE_PROPERTY, "1");
168             MCPRuntimeContext runtimeContext = createRuntimeContext(mock(WorkflowSessionContext.class));
169             MCPCompletionService completionService = createCompletionService(runtimeContext, List.of(new SizedCompletionHandler()));
170             MCPCompletionTargetDescriptor descriptor = createDescriptor("foo_prompt", "value", 50);
171             completionService.complete("session-1", descriptor, "value", "", new LinkedHashMap<>());
172             assertThrows(MCPUnavailableException.class, () -> completionService.complete("session-1", descriptor, "value", "", new LinkedHashMap<>()));
173             new MCPSessionExecutionCoordinator(runtimeContext.getSessionManager()).closeSession("session-1");
174             runtimeContext.getSessionManager().createSession(new MCPSessionIdentity("session-1", "", "", Map.of()));
175             assertDoesNotThrow(() -> completionService.complete("session-1", descriptor, "value", "", new LinkedHashMap<>()));
176         } finally {
177             restoreProperty(MCPRuntimeProtectionPolicy.MAX_COMPLETION_REQUESTS_PER_MINUTE_PROPERTY, previous);
178         }
179     }

181     @Test
182     void assertHandlerRunsOutsideSessionLock() throws InterruptedException, ExecutionException, TimeoutException {
183         BlockingCompletionHandler handler = new BlockingCompletionHandler();
184         MCPRuntimeContext runtimeContext = createRuntimeContext(mock(WorkflowSessionContext.class));
185         MCPCompletionService completionService = createCompletionService(runtimeContext, List.of(handler));
186         ExecutorService executorService = Executors.newFixedThreadPool(2);
187         try {
188             Future<MCPCompletionResult> completionFuture = executorService.submit(
189                     () -> completionService.complete("session-1", createDescriptor("foo_prompt", "value", 50), "value", "", new LinkedHashMap<>()));
190             assertTrue(handler.awaitStarted());
191             assertFalse(completionFuture.isDone());
192             Future<String> sessionLockFuture = executorService.submit(
193                     () -> new MCPSessionExecutionCoordinator(runtimeContext.getSessionManager()).executeWithSessionLock("session-1", () -> "available"));
194             assertThat(sessionLockFuture.get(1L, TimeUnit.SECONDS), is("available"));
195             handler.release();
196             completionFuture.get(1L, TimeUnit.SECONDS);
197         } finally {
198             handler.release();
199             executorService.shutdownNow();
200         }
201     }

208     private MCPCompletionResult complete(final WorkflowSessionContext workflowSessionContext, final MCPCompletionTargetDescriptor descriptor, final String argumentName, final String prefix,
209                                          final Map<String, String> contextArguments, final Collection<MCPCompletionHandler<?>> completionHandlers) {
210         return createCompletionService(createRuntimeContext(workflowSessionContext), completionHandlers).complete("session-1", descriptor, argumentName, prefix, contextArguments);
211     }

213     private MCPCompletionService createCompletionService(final MCPRuntimeContext runtimeContext, final Collection<MCPCompletionHandler<?>> completionHandlers) {
214         try (MockedStatic<ShardingSphereServiceLoader> mocked = mockStatic(ShardingSphereServiceLoader.class)) {
215             MCPHandlerProvider handlerProvider = mock(MCPHandlerProvider.class);
216             when(handlerProvider.getCompletionHandlers()).thenReturn(completionHandlers);
217             mocked.when(() -> ShardingSphereServiceLoader.getServiceInstances(MCPHandlerProvider.class)).thenReturn(List.of(handlerProvider));
218             return new MCPCompletionService(runtimeContext);
219         }
220     }

226     private MCPRuntimeContext createRuntimeContext(final WorkflowSessionContext workflowSessionContext) {
227         MCPDatabaseCapabilityProvider databaseCapabilityProvider = mock(MCPDatabaseCapabilityProvider.class);
228         when(databaseCapabilityProvider.getDatabaseProfiles()).thenReturn(List.of(
229                 new RuntimeDatabaseProfile("logic_db", "FixtureDB", "1.0", TransactionCapability.LOCAL_WITH_SAVEPOINT,
230                         new DatabaseIdentifierContext(IdentifierCasePolicyFactory.newInsensitivePolicySet())),
231                 new RuntimeDatabaseProfile("warehouse", "FixtureWarehouseDB", "2.0", TransactionCapability.LOCAL_WITH_SAVEPOINT,
232                         new DatabaseIdentifierContext(IdentifierCasePolicyFactory.newInsensitivePolicySet()))));
233         MCPSessionManager sessionManager = new MCPSessionManager(Collections.emptyMap());
234         sessionManager.createSession(new MCPSessionIdentity("session-1", "", "", Map.of()));
235         MCPRuntimeContext result = mock(MCPRuntimeContext.class);
236         when(result.getDatabaseCapabilityProvider()).thenReturn(databaseCapabilityProvider);
237         when(result.getWorkflowSessionContext("session-1")).thenReturn(workflowSessionContext);
238         when(result.getSessionManager()).thenReturn(sessionManager);
239         return result;
240     }

335     private static final class BlockingCompletionHandler implements MCPCompletionHandler<MCPFeatureRequestContext> {
336         
337         private final CountDownLatch started = new CountDownLatch(1);
338         
339         private final CountDownLatch released = new CountDownLatch(1);
340         
341         @Override
342         public Class<MCPFeatureRequestContext> getContextType() {
343             return MCPFeatureRequestContext.class;
344         }
345         
346         @Override
347         public boolean supports(final MCPCompletionRequest request) {
348             return "value".equals(request.getArgumentName());
349         }
350         
351         @Override
352         public MCPCompletionHandlerResult complete(final MCPFeatureRequestContext handlerContext, final MCPCompletionRequest request) {
353             started.countDown();
354             try {
355                 if (!released.await(1L, TimeUnit.SECONDS)) {
356                     throw new IllegalStateException("Timed out while waiting for test coordination.");
357                 }
358             } catch (final InterruptedException ex) {
359                 Thread.currentThread().interrupt();
360                 throw new IllegalStateException(ex);
361             }
362             return new MCPCompletionHandlerResult(List.of());
363         }
364         
365         private boolean awaitStarted() throws InterruptedException {
366             return started.await(1L, TimeUnit.SECONDS);
367         }
368         
369         private void release() {
370             released.countDown();
371         }
