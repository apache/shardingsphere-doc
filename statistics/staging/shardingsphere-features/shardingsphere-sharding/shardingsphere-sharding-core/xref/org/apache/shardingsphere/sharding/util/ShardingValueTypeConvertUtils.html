48      /**
49       * Convert collection to target type.
50       *
51       * @param sourceCollection source collection
52       * @param targetType target type
53       * @return converted collection
54       */
55      public static Collection<Comparable<?>> convertCollectionType(final Collection<Comparable<?>> sourceCollection, final Class<?> targetType) {
56          Collection<Comparable<?>> result = new LinkedList<>();
57          for (Comparable<?> value : sourceCollection) {
58              result.add(convertToTargetType(value, targetType));
59          }
60          return result;
61      }
62      
63      /**
64       * Convert value to target type.
65       *
66       * @param <T> type of exception
67       * @param value source value
68       * @param targetType target type class
69       * @return converted value
70       * @throws ClassCastException if conversion is not supported
71       */
72      @SuppressWarnings("unchecked")
73      public static <T> T convertToTargetType(final Comparable<?> value, final Class<?> targetType) {
74          if (null == value) {
75              return null;
76          }
77          if (targetType.isInstance(value)) {
78              return (T) value;
79          }
80          if (Integer.class == targetType) {
81              return (T) convertToInteger(value);
82          } else if (Long.class == targetType) {
83              return (T) convertToLong(value);
84          } else if (Short.class == targetType) {
85              return (T) convertToShort(value);
86          } else if (Byte.class == targetType) {
87              return (T) convertToByte(value);
88          } else if (Double.class == targetType) {
89              return (T) convertToDouble(value);
90          } else if (Float.class == targetType) {
91              return (T) convertToFloat(value);
92          } else if (BigDecimal.class == targetType) {
93              return (T) convertToBigDecimal(value);
94          } else if (BigInteger.class == targetType) {
95              return (T) convertToBigInteger(value);
96          } else if (Boolean.class == targetType) {
97              return (T) convertToBoolean(value);
98          } else if (Character.class == targetType) {
99              return (T) convertToCharacter(value);
100         } else if (String.class == targetType) {
101             return (T) value.toString();
102         } else if (Date.class == targetType) {
103             return (T) convertToDate(value);
104         } else if (java.sql.Date.class == targetType) {
105             return (T) convertToSqlDate(value);
106         } else if (Time.class == targetType) {
107             return (T) convertToSqlTime(value);
108         } else if (Timestamp.class == targetType) {
109             return (T) convertToTimestamp(value);
110         } else if (LocalDate.class == targetType) {
111             return (T) convertToLocalDate(value);
112         } else if (LocalTime.class == targetType) {
113             return (T) convertToLocalTime(value);
114         } else if (LocalDateTime.class == targetType) {
115             return (T) convertToLocalDateTime(value);
116         } else if (Instant.class == targetType) {
117             return (T) convertToInstant(value);
118         } else if (Year.class == targetType) {
119             return (T) convertToYear(value);
120         } else if (YearMonth.class == targetType) {
121             return (T) convertToYearMonth(value);
122         } else if (MonthDay.class == targetType) {
123             return (T) convertToMonthDay(value);
124         } else if (Duration.class == targetType) {
125             return (T) convertToDuration(value);
126         }
127         throw new ClassCastException("Unsupported type conversion from " + value.getClass().getName() + " to " + targetType.getName());
128     }

130     private static Integer convertToInteger(final Comparable<?> value) {
131         if (value instanceof Number) {
132             return ((Number) value).intValue();
133         }
134         return Integer.parseInt(value.toString());
135     }

172     private static BigDecimal convertToBigDecimal(final Comparable<?> value) {
173         if (value instanceof BigDecimal) {
174             return (BigDecimal) value;
175         }
176         if (value instanceof BigInteger) {
177             return new BigDecimal((BigInteger) value);
178         }
179         if (value instanceof Byte || value instanceof Short || value instanceof Integer || value instanceof Long) {
180             return BigDecimal.valueOf(((Number) value).longValue());
181         }
182         if (value instanceof Number) {
183             return BigDecimal.valueOf(((Number) value).doubleValue());
184         }
185         return new BigDecimal(value.toString());
186     }
187     
188     private static BigInteger convertToBigInteger(final Comparable<?> value) {
189         if (value instanceof BigInteger) {
190             return (BigInteger) value;
191         }
192         if (value instanceof BigDecimal) {
193             return ((BigDecimal) value).toBigInteger();
194         }
195         if (value instanceof Number) {
196             return BigInteger.valueOf(((Number) value).longValue());
197         }
198         return new BigInteger(value.toString());
199     }

201     private static Boolean convertToBoolean(final Comparable<?> value) {
202         if (value instanceof Boolean) {
203             return (Boolean) value;
204         }
205         if (value instanceof Number) {
206             return ((Number) value).intValue() != 0;
207         }
208         String strValue = value.toString().trim();
209         if ("true".equalsIgnoreCase(strValue) || "1".equals(strValue)) {
210             return Boolean.TRUE;
211         }
212         if ("false".equalsIgnoreCase(strValue) || "0".equals(strValue)) {
213             return Boolean.FALSE;
214         }
215         return Boolean.parseBoolean(strValue);
216     }

218     private static Character convertToCharacter(final Comparable<?> value) {
219         if (value instanceof Character) {
220             return (Character) value;
221         }
222         if (value instanceof Number) {
223             return (char) ((Number) value).intValue();
224         }
225         String strValue = value.toString();
226         return strValue.isEmpty() ? '\0' : strValue.charAt(0);
227     }

229     private static Date convertToDate(final Comparable<?> value) {
230         if (value instanceof Date) {
231             return (Date) value;
232         }
233         if (value instanceof LocalDateTime) {
234             return Date.from(((LocalDateTime) value).atZone(ZoneId.systemDefault()).toInstant());
235         }
236         if (value instanceof LocalDate) {
237             return Date.from(((LocalDate) value).atStartOfDay(ZoneId.systemDefault()).toInstant());
238         }
239         if (value instanceof Instant) {
240             return Date.from((Instant) value);
241         }
242         if (value instanceof Number) {
243             return new Date(((Number) value).longValue());
244         }
245         return Date.from(parseInstant(value.toString()));
246     }
247     
248     private static java.sql.Date convertToSqlDate(final Comparable<?> value) {
249         if (value instanceof java.sql.Date) {
250             return (java.sql.Date) value;
251         }
252         if (value instanceof Date) {
253             return new java.sql.Date(((Date) value).getTime());
254         }
255         if (value instanceof LocalDate) {
256             return java.sql.Date.valueOf((LocalDate) value);
257         }
258         if (value instanceof LocalDateTime) {
259             return java.sql.Date.valueOf(((LocalDateTime) value).toLocalDate());
260         }
261         if (value instanceof Number) {
262             return new java.sql.Date(((Number) value).longValue());
263         }
264         return new java.sql.Date(parseInstant(value.toString()).toEpochMilli());
265     }

268         if (value instanceof Time) {

286     private static Timestamp convertToTimestamp(final Comparable<?> value) {
287         if (value instanceof Timestamp) {
288             return (Timestamp) value;
289         }
290         if (value instanceof Date) {
291             return new Timestamp(((Date) value).getTime());
292         }
293         if (value instanceof LocalDateTime) {
294             return Timestamp.valueOf((LocalDateTime) value);
295         }
296         if (value instanceof Instant) {
297             return Timestamp.from((Instant) value);
298         }
299         if (value instanceof Number) {
300             return new Timestamp(((Number) value).longValue());
301         }
302         return Timestamp.from(parseInstant(value.toString()));
303     }

305     private static LocalDate convertToLocalDate(final Comparable<?> value) {
306         if (value instanceof LocalDate) {
307             return (LocalDate) value;
308         }
309         if (value instanceof LocalDateTime) {
310             return ((LocalDateTime) value).toLocalDate();
311         }
312         if (value instanceof java.sql.Date) {
313             return ((java.sql.Date) value).toLocalDate();
314         }
315         if (value instanceof Date) {
316             return ((Date) value).toInstant().atZone(ZoneId.systemDefault()).toLocalDate();
317         }
318         if (value instanceof Instant) {
319             return ((Instant) value).atZone(ZoneId.systemDefault()).toLocalDate();
320         }
321         if (value instanceof Number) {
322             return Instant.ofEpochMilli(((Number) value).longValue()).atZone(ZoneId.systemDefault()).toLocalDate();
323         }
324         return parseLocalDate(value.toString());
325     }

327     private static LocalTime convertToLocalTime(final Comparable<?> value) {
328         if (value instanceof LocalTime) {
329             return (LocalTime) value;
330         }
331         if (value instanceof LocalDateTime) {
332             return ((LocalDateTime) value).toLocalTime();
333         }
334         if (value instanceof Time) {
335             return ((Time) value).toLocalTime();
336         }
337         if (value instanceof Date) {
338             return ((Date) value).toInstant().atZone(ZoneId.systemDefault()).toLocalTime();
339         }
340         if (value instanceof Number) {
341             return Instant.ofEpochMilli(((Number) value).longValue()).atZone(ZoneId.systemDefault()).toLocalTime();
342         }
343         return LocalTime.parse(value.toString());
344     }

346     private static LocalDateTime convertToLocalDateTime(final Comparable<?> value) {
347         if (value instanceof LocalDateTime) {
348             return (LocalDateTime) value;
349         }
350         if (value instanceof LocalDate) {
351             return ((LocalDate) value).atStartOfDay();
352         }
353         if (value instanceof Date) {
354             return ((Date) value).toInstant().atZone(ZoneId.systemDefault()).toLocalDateTime();
355         }
356         if (value instanceof Timestamp) {
357             return ((Timestamp) value).toLocalDateTime();
358         }
359         if (value instanceof Instant) {
360             return LocalDateTime.ofInstant((Instant) value, ZoneId.systemDefault());
361         }
362         if (value instanceof Number) {
363             return Instant.ofEpochMilli(((Number) value).longValue()).atZone(ZoneId.systemDefault()).toLocalDateTime();
364         }
365         return parseLocalDateTime(value.toString());
366     }

368     private static Instant convertToInstant(final Comparable<?> value) {
369         if (value instanceof Instant) {
370             return (Instant) value;
371         }
372         if (value instanceof Date) {
373             return ((Date) value).toInstant();
374         }
375         if (value instanceof LocalDateTime) {
376             return ((LocalDateTime) value).atZone(ZoneId.systemDefault()).toInstant();
377         }
378         if (value instanceof Number) {
379             return Instant.ofEpochMilli(((Number) value).longValue());
380         }
381         return parseInstant(value.toString());
382     }

426     private static Instant parseInstant(final String value) {
427         try {
428             return Instant.parse(value);
429         } catch (final DateTimeParseException ignored) {
430             return parseInstantByLocalDateTime(value);
431         }
432     }
433     
434     private static Instant parseInstantByLocalDateTime(final String value) {
435         try {
436             return LocalDateTime.parse(value).atZone(ZoneId.systemDefault()).toInstant();
437         } catch (final DateTimeException ignored) {
438             return LocalDate.parse(value).atStartOfDay(ZoneId.systemDefault()).toInstant();
439         }
440     }
441     
442     private static LocalDate parseLocalDate(final String value) {
443         try {
444             return LocalDate.parse(value);
445         } catch (final DateTimeParseException ignored) {
446             return LocalDateTime.parse(value).toLocalDate();
447         }
448     }
449     
450     private static LocalDateTime parseLocalDateTime(final String value) {
451         try {
452             return LocalDateTime.parse(value);
453         } catch (final DateTimeParseException ignored) {
454             return Instant.parse(value).atZone(ZoneId.systemDefault()).toLocalDateTime();
455         }
456     }
457     
458     private static LocalTime parseLocalTime(final String value) {
459         try {
460             return LocalTime.parse(value);
461         } catch (final DateTimeParseException ignored) {
462             return LocalDateTime.parse(value).toLocalTime();
463         }
464     }
465     
466     private static Duration convertToDuration(final Comparable<?> value) {
467         if (value instanceof Duration) {
468             return (Duration) value;
469         }
470         if (value instanceof Number) {
471             return Duration.ofMillis(((Number) value).longValue());
472         }
473         return Duration.parse(value.toString());
474     }
