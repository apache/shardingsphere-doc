34  class WorkflowContextSnapshotTest {
35      
36      @Test
37      void assertCopyCreatesDetachedSnapshot() {
38          WorkflowContextSnapshot originalSnapshot = createSnapshot();
39          WorkflowContextSnapshot actualSnapshot = originalSnapshot.copy();
40          assertThat(actualSnapshot.getPlanId(), is("plan-1"));
41          assertThat(actualSnapshot.getWorkflowKind(), is(WorkflowKind.valueOf("encrypt.rule")));
42          originalSnapshot.getRequest().setTable("archived_orders");
43          originalSnapshot.getClarifiedIntent().getClarificationMessages().add("another question");
44          originalSnapshot.getClarifiedIntent().getUnresolvedFields().add("schema");
45          originalSnapshot.getClarifiedIntent().getInferredValues().put("requires_decrypt", false);
46          originalSnapshot.getInteractionPlan().getValidationStrategy().put("layers", List.of("rule"));
47          getStringList(originalSnapshot.getIssues().get(0).getDetails(), "missing_fields").add("schema");
48          getStringList(originalSnapshot.getValidationReport().getMismatches().get(0), "layers").add("rule");
49          originalSnapshot.getAlgorithmCandidates().clear();
50          originalSnapshot.getPropertyRequirements().clear();
51          originalSnapshot.getResourceUriTemplates().clear();
52          originalSnapshot.getRuleArtifacts().clear();
53          assertThat(actualSnapshot.getRequest().getTable(), is("orders"));
54          assertThat(actualSnapshot.getClarifiedIntent().getClarificationMessages(), is(List.of("provide schema")));
55          assertTrue(actualSnapshot.getClarifiedIntent().getUnresolvedFields().isEmpty());
56          assertTrue(actualSnapshot.getClarifiedIntent().getInferredValues().isEmpty());
57          assertThat(actualSnapshot.getFeatureData(), not(sameInstance(originalSnapshot.getFeatureData())));
58          assertThat(actualSnapshot.getInteractionPlan().getValidationStrategy().get("layers"), is(List.of("rule")));
59          assertThat(actualSnapshot.getIssues().get(0).getDetails().get("missing_fields"), is(List.of("column")));
60          assertThat(actualSnapshot.getValidationReport().getMismatches().get(0).get("layers"), is(List.of("rule")));
61          assertThat(actualSnapshot.getAlgorithmCandidates().size(), is(1));
62          assertThat(actualSnapshot.getPropertyRequirements().size(), is(1));
63          assertThat(actualSnapshot.getResourceUriTemplates(), is(List.of("shardingsphere://workflow/test-resource")));
64          assertThat(actualSnapshot.getRuleArtifacts().size(), is(1));
65      }
66      
67      @Test
68      void assertClearPlanningStateRemovesTransientArtifacts() {
69          WorkflowContextSnapshot snapshot = createSnapshot();
70          RuleWorkflowFeatureData featureData = snapshot.getFeatureData();
71          assertThat(snapshot.getFeatureData(), is(featureData));
72          snapshot.clearPlanningState();
73          assertTrue(snapshot.getIssues().isEmpty());
74          assertTrue(snapshot.getAlgorithmCandidates().isEmpty());
75          assertTrue(snapshot.getPropertyRequirements().isEmpty());
76          assertTrue(snapshot.getResourceUriTemplates().isEmpty());
77          assertTrue(snapshot.getRuleArtifacts().isEmpty());
78          assertThat(snapshot.getFeatureData(), is(featureData));
79          assertNull(snapshot.getValidationReport());
80      }
81      
82      private WorkflowContextSnapshot createSnapshot() {
83          WorkflowContextSnapshot result = new WorkflowContextSnapshot();
84          result.setPlanId("plan-1");
85          result.setWorkflowKind(WorkflowKind.valueOf("encrypt.rule"));
86          result.setStatus("planned");
87          WorkflowRequest request = new WorkflowRequest();
88          request.setDatabase("logic_db");
89          request.setSchema("public");
90          request.setTable("orders");
91          request.setColumn("phone");
92          result.setRequest(request);
93          ClarifiedIntent clarifiedIntent = new ClarifiedIntent();
94          clarifiedIntent.getClarificationMessages().add("provide schema");
95          result.setClarifiedIntent(clarifiedIntent);
96          result.setFeatureData(new RuleWorkflowFeatureData(List.of(Map.of("rule", "encrypt_rule"))));
97          InteractionPlan interactionPlan = new InteractionPlan();
98          interactionPlan.setCurrentStep("review");
99          interactionPlan.getValidationStrategy().put("layers", List.of("rule"));
100         result.setInteractionPlan(interactionPlan);
101         Map<String, Object> issueDetails = new LinkedHashMap<>(1, 1F);
102         issueDetails.put("missing_fields", new LinkedList<>(List.of("column")));
103         result.getIssues().add(new WorkflowIssue("code", "error", "stage", "message", "action", true, issueDetails));
104         result.getAlgorithmCandidates().add(AlgorithmCandidate.builder().algorithmRole("primary").algorithmType("AES")
105                 .supportsDecrypt(true).supportsEquivalentFilter(true).supportsLike(false).recommendationScore(90).recommendationReason("reason").riskNotes("").build());
106         result.getPropertyRequirements().add(new AlgorithmPropertyRequirement("primary", "key", true, true, "desc", ""));
107         result.getResourceUriTemplates().add("shardingsphere://workflow/test-resource");
108         result.getRuleArtifacts().add(new RuleArtifact("create", "CREATE ENCRYPT RULE t"));
109         ValidationReport validationReport = new ValidationReport();
110         Map<String, Object> mismatch = new LinkedHashMap<>(2, 1F);
111         mismatch.put("code", "mismatch");
112         mismatch.put("layers", new LinkedList<>(List.of("rule")));
113         validationReport.getMismatches().add(mismatch);
114         result.setValidationReport(validationReport);
115         return result;
116     }

120         return (List<String>) source.get(key);
