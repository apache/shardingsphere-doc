43  import static org.hamcrest.MatcherAssert.assertThat;

48  class PostgreSQLCommandPacketFactoryTest {
49      
50      @ParameterizedTest(name = "{0}")
51      @MethodSource("nonExtendedProtocolCases")
52      void assertNewInstanceWithNonExtendedProtocol(final String name, final PostgreSQLCommandPacketType commandPacketType,
53                                                    final int packetTypeValue, final byte[] packetBody, final Class<? extends PostgreSQLCommandPacket> expectedPacketClass) {
54          assertThat(PostgreSQLCommandPacketFactory.newInstance(commandPacketType, createPayload(packetTypeValue, packetBody)), isA(expectedPacketClass));
55      }
56      
57      @Test
58      void assertNewInstanceWithExtendedProtocolWithoutPackets() {
59          assertTrue(((PostgreSQLAggregatedCommandPacket) PostgreSQLCommandPacketFactory.newInstance(
60                  PostgreSQLCommandPacketType.PARSE_COMMAND, new PostgreSQLPacketPayload(Unpooled.buffer(), StandardCharsets.UTF_8))).getPackets().isEmpty());
61      }
62      
63      @ParameterizedTest(name = "{0}")
64      @MethodSource("extendedProtocolCases")
65      void assertNewInstanceWithExtendedProtocol(final String name, final int packetTypeValue, final byte[] packetBody,
66                                                 final Class<? extends PostgreSQLCommandPacket> expectedPacketClass) {
67          PostgreSQLCommandPacket actual = PostgreSQLCommandPacketFactory.newInstance(PostgreSQLCommandPacketType.PARSE_COMMAND, createPayload(packetTypeValue, packetBody));
68          List<PostgreSQLCommandPacket> actualPackets = ((PostgreSQLAggregatedCommandPacket) actual).getPackets();
69          assertThat(actualPackets.size(), is(1));
70          assertThat(actualPackets.get(0), isA(expectedPacketClass));
71      }
72      
73      private static PostgreSQLPacketPayload createPayload(final int packetTypeValue, final byte[] packetBody) {
74          ByteBuf byteBuf = Unpooled.buffer(1 + Integer.BYTES + packetBody.length);
75          byteBuf.writeByte(packetTypeValue);
76          byteBuf.writeInt(Integer.BYTES + packetBody.length);
77          byteBuf.writeBytes(packetBody);
78          return new PostgreSQLPacketPayload(byteBuf, StandardCharsets.UTF_8);
79      }
80      
81      private static Stream<Arguments> nonExtendedProtocolCases() {
82          return Stream.of(
83                  Arguments.of("simple_query", PostgreSQLCommandPacketType.SIMPLE_QUERY, (int) PostgreSQLCommandPacketType.SIMPLE_QUERY.getValue(),
84                          createQueryBody(), PostgreSQLComQueryPacket.class),
85                  Arguments.of("termination", PostgreSQLCommandPacketType.TERMINATE, (int) PostgreSQLCommandPacketType.TERMINATE.getValue(),
86                          new byte[0], PostgreSQLComTerminationPacket.class),
87                  Arguments.of("unsupported_password", PostgreSQLCommandPacketType.PASSWORD, (int) PostgreSQLCommandPacketType.PASSWORD.getValue(),
88                          new byte[0], PostgreSQLUnsupportedCommandPacket.class));
89      }
90      
91      private static byte[] createQueryBody() {
92          byte[] queryBytes = "SELECT 1".getBytes(StandardCharsets.UTF_8);
93          ByteBuf byteBuf = Unpooled.buffer(queryBytes.length + 1);
94          byteBuf.writeBytes(queryBytes);
95          byteBuf.writeByte(0);
96          byte[] result = new byte[byteBuf.readableBytes()];
97          byteBuf.getBytes(0, result);
98          return result;
99      }
100     
101     private static Stream<Arguments> extendedProtocolCases() {
102         return Stream.of(
103                 Arguments.of("parse", (int) PostgreSQLCommandPacketType.PARSE_COMMAND.getValue(), new byte[]{0, 0}, PostgreSQLComParsePacket.class),
104                 Arguments.of("bind", (int) PostgreSQLCommandPacketType.BIND_COMMAND.getValue(), new byte[]{0, 0}, PostgreSQLComBindPacket.class),
105                 Arguments.of("describe", (int) PostgreSQLCommandPacketType.DESCRIBE_COMMAND.getValue(), new byte[]{'S', 0}, PostgreSQLComDescribePacket.class),
106                 Arguments.of("execute", (int) PostgreSQLCommandPacketType.EXECUTE_COMMAND.getValue(), new byte[]{0, 0, 0, 0, 0}, PostgreSQLComExecutePacket.class),
107                 Arguments.of("sync", (int) PostgreSQLCommandPacketType.SYNC_COMMAND.getValue(), new byte[0], PostgreSQLComSyncPacket.class),
108                 Arguments.of("close", (int) PostgreSQLCommandPacketType.CLOSE_COMMAND.getValue(), new byte[]{'S', 0}, PostgreSQLComClosePacket.class),
109                 Arguments.of("flush", (int) PostgreSQLCommandPacketType.FLUSH_COMMAND.getValue(), new byte[0], PostgreSQLComFlushPacket.class));
110     }
