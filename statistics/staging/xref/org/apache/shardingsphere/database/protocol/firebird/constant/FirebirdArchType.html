18  package org.apache.shardingsphere.database.protocol.firebird.constant;
19  
20  import com.google.common.base.Preconditions;
21  import lombok.Getter;
22  import lombok.RequiredArgsConstructor;
23  import org.apache.commons.lang3.SystemUtils;
24  
25  import java.util.HashMap;
26  import java.util.Map;
27  import java.util.stream.Stream;
28  
29  /**
30   * Firebird architecture type.
31   */
32  @RequiredArgsConstructor
33  @Getter
34  public enum FirebirdArchType {
35      
36      ARCH_GENERIC(1, ""),
37      ARCH_SUN(3, "SO"),
38      ARCH_SUN4(8, "S4"),
39      ARCH_SUNX86(9, "SI"),
40      ARCH_HPUX(10, "HU"),
41      ARCH_RT(14, "PA"),
42      ARCH_INTEL_32(29, "WI"),
43      ARCH_LINUX_LOONG(36, "LL"),
44      ARCH_LINUX(36, "LI"),
45      ARCH_FREEBSD(37, "FB"),
46      ARCH_NETBSD(38, "NB"),
47      ARCH_DARWIN_PPC(39, "UP"),
48      ARCH_WINNT_64(40, "WI"),
49      ARCH_DARWIN_X64_ARM(41, "UA"),
50      ARCH_DARWIN_X64(41, "UI"),
51      ARCH_DARWIN_PPC64(42, "UP"),
52      ARCH_ARM(43, "NP"),
53      ARCH_MAX(44, "");
54      
55      public static final FirebirdArchType ARCHITECTURE;
56      
57      private static final Map<Integer, FirebirdArchType> FIREBIRD_ARCH_TYPE_CACHE = new HashMap<>();
58      
59      private final int code;
60      
61      private final String identifier;
62      
63      static {
64          for (FirebirdArchType each : values()) {
65              FIREBIRD_ARCH_TYPE_CACHE.put(each.code, each);
66          }
67          // setup user architecture
68          if (SystemUtils.IS_OS_SUN_OS) {
69              switch (SystemUtils.OS_ARCH.toLowerCase()) {
70                  case "sparc":
71                      ARCHITECTURE = ARCH_SUN4;
72                      break;
73                  case "i386":
74                  case "amd64":
75                      ARCHITECTURE = ARCH_SUNX86;
76                      break;
77                  default:
78                      ARCHITECTURE = ARCH_SUN;
79              }
80          } else if (SystemUtils.IS_OS_HP_UX) {
81              ARCHITECTURE = ARCH_HPUX;
82          } else if (SystemUtils.IS_OS_AIX) {
83              ARCHITECTURE = ARCH_RT;
84          } else if (SystemUtils.IS_OS_LINUX) {
85              ARCHITECTURE = SystemUtils.OS_ARCH.toLowerCase().contains("loong") ? ARCH_LINUX_LOONG : ARCH_LINUX;
86          } else if (SystemUtils.IS_OS_FREE_BSD) {
87              ARCHITECTURE = ARCH_FREEBSD;
88          } else if (SystemUtils.IS_OS_NET_BSD) {
89              ARCHITECTURE = ARCH_NETBSD;
90          } else if (SystemUtils.IS_OS_MAC || SystemUtils.OS_NAME.toLowerCase().contains("darwin")) {
91              switch (SystemUtils.OS_ARCH.toLowerCase()) {
92                  case "ppc":
93                  case "powerpc":
94                      ARCHITECTURE = ARCH_DARWIN_PPC;
95                      break;
96                  case "ppc64":
97                      ARCHITECTURE = ARCH_DARWIN_PPC64;
98                      break;
99                  case "arm":
100                 case "aarch64":
101                 case "aarch32":
102                     ARCHITECTURE = ARCH_DARWIN_X64_ARM;
103                     break;
104                 default:
105                     ARCHITECTURE = ARCH_DARWIN_X64;
106             }
107         } else if (SystemUtils.IS_OS_WINDOWS && "amd64".equalsIgnoreCase(SystemUtils.OS_ARCH)) {
108             ARCHITECTURE = ARCH_WINNT_64;
109         } else if ("i386".equalsIgnoreCase(SystemUtils.OS_ARCH)) {
110             ARCHITECTURE = ARCH_INTEL_32;
111         } else if (Stream.of("arm", "aarch64", "aarch32").anyMatch(SystemUtils.OS_ARCH::equalsIgnoreCase)) {
112             ARCHITECTURE = ARCH_ARM;
113         } else {
114             ARCHITECTURE = ARCH_GENERIC;
115         }
116     }
117     
118     /**
119      * Value of.
120      *
121      * @param code arch type code
122      * @return Firebird arch type
123      */
124     public static FirebirdArchType valueOf(final int code) {
125         FirebirdArchType result = FIREBIRD_ARCH_TYPE_CACHE.get(code);
126         Preconditions.checkNotNull(result, "Cannot find '%d' in arch type", code);
127         return result;
128     }
129     
130     /**
131      * Determines if the architecture is valid.
132      *
133      * @param arch architecture type
134      * @return is the architecture valid
135      */
136     public static boolean isValid(final FirebirdArchType arch) {
137         return arch == ARCH_GENERIC || arch == ARCHITECTURE;
138     }
139 }
