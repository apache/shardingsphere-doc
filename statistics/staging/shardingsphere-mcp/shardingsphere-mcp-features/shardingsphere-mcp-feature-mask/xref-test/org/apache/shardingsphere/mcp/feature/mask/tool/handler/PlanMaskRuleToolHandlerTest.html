56      @Test
57      void assertHandlePlanMaskRule() {
58          try (MockedConstruction<MaskWorkflowPlanningService> mockedConstruction = mockConstruction(MaskWorkflowPlanningService.class)) {
59              PlanMaskRuleToolHandler handler = new PlanMaskRuleToolHandler();
60              MaskWorkflowPlanningService planningService = mockedConstruction.constructed().getFirst();
61              when(planningService.plan(any(), any(), any(), any())).thenReturn(createSnapshot("plan-1", "planned"));
62              WorkflowContextFixture fixture = createWorkflowContextFixture();
63              MCPSuccessPayload actual = handler.handle(fixture.requestContext, Map.of(
64                      "database", "logic_db",
65                      "table", "orders",
66                      "column", "phone",
67                      "algorithm_type", "KEEP_FIRST_N_LAST_M"));
68              assertThat(actual.toPayload().get("plan_id"), is("plan-1"));
69              ArgumentCaptor<WorkflowRequest> requestCaptor = ArgumentCaptor.forClass(WorkflowRequest.class);
70              verify(planningService).plan(eq(fixture.workflowSessionContext), eq(fixture.metadataQueryFacade), eq(fixture.queryFacade), requestCaptor.capture());
71              WorkflowRequest actualRequest = requestCaptor.getValue();
72              assertThat(actualRequest.getAlgorithmType(), is("KEEP_FIRST_N_LAST_M"));
73          }
74      }

76      @Test
77      void assertHandlePlanMaskRuleWithMaskedArtifacts() {
78          try (MockedConstruction<MaskWorkflowPlanningService> mockedConstruction = mockConstruction(MaskWorkflowPlanningService.class)) {
79              PlanMaskRuleToolHandler handler = new PlanMaskRuleToolHandler();
80              when(mockedConstruction.constructed().getFirst().plan(any(), any(), any(), any())).thenReturn(createDetailedSnapshot());
81              MCPSuccessPayload actual = handler.handle(createWorkflowContextFixture().requestContext, Map.of(
82                      "database", "logic_db",
83                      "table", "orders",
84                      "column", "phone"));
85              Map<String, Object> actualPayload = actual.toPayload();
86              assertThat(((Map<?, ?>) ((Map<?, ?>) actualPayload.get("masked_property_preview")).get("primary")).get("first-n"), is("3"));
87              assertTrue(String.valueOf(((Map<?, ?>) ((List<?>) actualPayload.get("distsql_artifacts")).getFirst()).get("sql")).contains("keep_first_n_last_m"));
88              List<?> actualResourcesToRead = (List<?>) actualPayload.get("resources_to_read");
89              List<String> actualResourceUris = extractResourceUris(actualResourcesToRead);
90              assertTrue(actualResourceUris.contains("shardingsphere://features/mask/algorithms"));
91              assertTrue(actualResourceUris.contains("shardingsphere://features/mask/databases/logic_db/rules"));
92              assertTrue(actualResourceUris.contains("shardingsphere://features/mask/databases/logic_db/tables/orders/rules"));
93              assertThat(findResourceKind(actualResourcesToRead, "shardingsphere://features/mask/algorithms"), is("algorithm"));
94              assertThat(findResourceKind(actualResourcesToRead, "shardingsphere://features/mask/databases/logic_db/rules"), is("rule"));
95              assertThat(findResourceKind(actualResourcesToRead, "shardingsphere://features/mask/databases/logic_db/tables/orders/rules"), is("rule"));
96              assertFalse(actualResourceUris.contains("shardingsphere://databases/logic_db/schemas/public/tables/orders/columns"));
97              Map<?, ?> actualNextAction = (Map<?, ?>) ((List<?>) actualPayload.get("next_actions")).getFirst();
98              assertThat(actualNextAction.get("type"), is("tool_call"));
99              assertThat(actualNextAction.get("tool_name"), is("database_gateway_apply_workflow"));
100             assertThat(((Map<?, ?>) actualNextAction.get("arguments")).get("plan_id"), is("plan-1"));
101             assertThat(((Map<?, ?>) actualNextAction.get("arguments")).get("execution_mode"), is("preview"));
102         }
103     }

105     @Test
106     void assertHandlePlanMaskRuleWithSecretReference() {
107         try (MockedConstruction<MaskWorkflowPlanningService> mockedConstruction = mockConstruction(MaskWorkflowPlanningService.class)) {
108             PlanMaskRuleToolHandler handler = new PlanMaskRuleToolHandler();
109             MaskWorkflowPlanningService planningService = mockedConstruction.constructed().getFirst();
110             WorkflowContextSnapshot snapshot = createSnapshot("plan-1", "planned");
111             when(planningService.plan(any(), any(), any(), any())).thenAnswer(invocation -> {
112                 snapshot.setRequest(invocation.getArgument(3));
113                 return snapshot;
114             });
115             WorkflowContextFixture fixture = createWorkflowContextFixture();
116             Map<String, Object> actualPayload = handler.handle(fixture.requestContext, Map.of(
117                     "database", "logic_db",
118                     "primary_algorithm_properties", Map.of("replace-char", Map.of("secret_ref", "placeholder://secret-value-1")))).toPayload();
119             Map<?, ?> actualMaskedPropertyPreview = (Map<?, ?>) actualPayload.get("masked_property_preview");
120             assertThat(((Map<?, ?>) actualMaskedPropertyPreview.get("primary")).get("replace-char"), is("******"));
121             assertFalse(String.valueOf(actualPayload).contains("secret_reference:"));
122             ArgumentCaptor<WorkflowRequest> requestCaptor = ArgumentCaptor.forClass(WorkflowRequest.class);
123             verify(planningService).plan(eq(fixture.workflowSessionContext), eq(fixture.metadataQueryFacade), eq(fixture.queryFacade), requestCaptor.capture());
124             WorkflowRequest actualRequest = requestCaptor.getValue();
125             assertThat(actualRequest.getPrimaryAlgorithmProperties().get("replace-char"), is("secret_reference:primary.replace-char"));
126             assertFalse(actualRequest.getSecretReferences("primary").get("replace-char").isMalformed());
127         }
128     }

130     private WorkflowContextSnapshot createSnapshot(final String planId, final String status) {
131         WorkflowContextSnapshot result = new WorkflowContextSnapshot();
132         result.setPlanId(planId);
133         result.setWorkflowKind(MaskFeatureDefinition.WORKFLOW_KIND);
134         result.setStatus(status);
135         result.setRequest(new WorkflowRequest());
136         result.setClarifiedIntent(new ClarifiedIntent());
137         result.setInteractionPlan(createInteractionPlan());
138         return result;
139     }
140     
141     private WorkflowContextSnapshot createDetailedSnapshot() {
142         WorkflowRequest request = new WorkflowRequest();
143         request.setDatabase("logic_db");
144         request.setSchema("public");
145         request.setTable("orders");
146         request.getPrimaryAlgorithmProperties().put("first-n", "3");
147         request.getPrimaryAlgorithmProperties().put("last-m", "2");
148         request.getPrimaryAlgorithmProperties().put("replace-char", "*");
149         WorkflowContextSnapshot result = createSnapshot("plan-1", "planned");
150         result.setRequest(request);
151         result.getPropertyRequirements().add(new AlgorithmPropertyRequirement("primary", "first-n", true, false, "from", ""));
152         result.getRuleArtifacts().add(new RuleArtifact("create", "CREATE MASK RULE `orders` (TYPE(NAME='keep_first_n_last_m'))"));
153         result.setInteractionPlan(createInteractionPlan());
154         return result;
155     }
156     
157     private InteractionPlan createInteractionPlan() {
158         InteractionPlan result = new InteractionPlan();
159         result.setCurrentStep("review");
160         result.setDeliveryMode("interactive");
161         result.setExecutionMode("review-then-execute");
162         return result;
163     }

165     private List<String> extractResourceUris(final List<?> resources) {
166         return resources.stream().map(each -> (String) ((Map<?, ?>) each).get("uri")).toList();
167     }
168     
169     private String findResourceKind(final List<?> resources, final String uri) {
170         for (Object each : resources) {
171             Map<?, ?> resource = (Map<?, ?>) each;
172             if (uri.equals(resource.get("uri"))) {
173                 return (String) resource.get("resource_kind");
174             }
175         }
176         return "";
177     }
