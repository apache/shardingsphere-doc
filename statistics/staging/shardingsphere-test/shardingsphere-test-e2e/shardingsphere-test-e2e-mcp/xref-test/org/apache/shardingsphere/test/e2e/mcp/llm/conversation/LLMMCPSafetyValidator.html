18  package org.apache.shardingsphere.test.e2e.mcp.llm.conversation;
19  
20  import java.util.Map;
21  import java.util.Objects;
22  import java.util.Optional;
23  
24  final class LLMMCPSafetyValidator {
25      
26      private static final String EXECUTION_MODE_PREVIEW = "preview";
27      
28      Optional<ValidationFailure> validate(final String actionName, final Map<String, Object> arguments) {
29          if (LLMConversationRunner.READ_RESOURCE_TOOL_NAME.equals(actionName) && Objects.toString(arguments.get("uri"), "").trim().isEmpty()) {
30              return Optional.of(new ValidationFailure("invalid_tool_arguments", "Model returned an empty resource URI."));
31          }
32          if ("database_gateway_execute_query".equals(actionName) && isExplain(arguments)) {
33              return Optional.of(new ValidationFailure("invalid_tool_arguments",
34                      "Model routed EXPLAIN SQL to database_gateway_execute_query instead of database_gateway_execute_explain_query."));
35          }
36          if ("database_gateway_execute_update".equals(actionName) && !EXECUTION_MODE_PREVIEW.equals(Objects.toString(arguments.get("execution_mode"), ""))) {
37              return Optional.of(new ValidationFailure("unsafe_sql_execution_attempted",
38                      "Model attempted to execute side-effecting SQL in an LLM E2E scenario."));
39          }
40          return Optional.empty();
41      }
42      
43      private boolean isExplain(final Map<String, Object> arguments) {
44          String sql = Objects.toString(arguments.get("sql"), "").trim();
45          return "EXPLAIN".equalsIgnoreCase(sql) || 7 < sql.length() && sql.regionMatches(true, 0, "EXPLAIN", 0, 7)
46                  && (Character.isWhitespace(sql.charAt(7)) || '(' == sql.charAt(7));
47      }
48      
49      record ValidationFailure(String failureType, String message) {
50      }
51  }
