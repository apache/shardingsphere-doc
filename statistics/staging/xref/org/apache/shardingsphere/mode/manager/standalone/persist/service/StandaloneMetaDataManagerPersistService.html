91      @Override
92      public void createSchema(final String databaseName, final String schemaName) {
93          ShardingSphereMetaData metaData = metaDataContextManager.getMetaDataContexts().getMetaData();
94          ShardingSphereDatabase database = metaData.getDatabase(databaseName);
95          database.addSchema(new ShardingSphereSchema(schemaName));
96          metaData.getGlobalRuleMetaData().getRules().forEach(each -> ((GlobalRule) each).refresh(metaData.getAllDatabases(), GlobalRuleChangedType.SCHEMA_CHANGED));
97          metaDataPersistService.getDatabaseMetaDataFacade().getSchema().add(databaseName, schemaName);
98      }

100     @Override
101     public void alterSchema(final AlterSchemaPOJO alterSchemaPOJO) {
102         ShardingSphereMetaData metaData = metaDataContextManager.getMetaDataContexts().getMetaData();
103         ShardingSphereDatabase database = metaData.getDatabase(alterSchemaPOJO.getDatabaseName());
104         putSchemaMetaData(database, alterSchemaPOJO.getSchemaName(), alterSchemaPOJO.getRenameSchemaName(), alterSchemaPOJO.getLogicDataSourceName());
105         removeSchemaMetaData(database, alterSchemaPOJO.getSchemaName());
106         metaData.getGlobalRuleMetaData().getRules().forEach(each -> ((GlobalRule) each).refresh(metaData.getAllDatabases(), GlobalRuleChangedType.SCHEMA_CHANGED));
107         ShardingSphereSchema alteredSchema = database.getSchema(alterSchemaPOJO.getRenameSchemaName());
108         String databaseName = alterSchemaPOJO.getDatabaseName();
109         String alteredSchemaName = alterSchemaPOJO.getRenameSchemaName();
110         if (alteredSchema.isEmpty()) {
111             metaDataPersistService.getDatabaseMetaDataFacade().getSchema().add(databaseName, alteredSchemaName);
112         }
113         metaDataPersistService.getDatabaseMetaDataFacade().getTable().persist(databaseName, alteredSchemaName, alteredSchema.getAllTables());
114         metaDataPersistService.getDatabaseMetaDataFacade().getView().persist(databaseName, alteredSchemaName, alteredSchema.getAllViews());
115         metaDataPersistService.getDatabaseMetaDataFacade().getSchema().drop(databaseName, alterSchemaPOJO.getSchemaName());
116     }

118     private void putSchemaMetaData(final ShardingSphereDatabase database, final String schemaName, final String renamedSchemaName, final String logicDataSourceName) {
119         ShardingSphereSchema schema = database.getSchema(schemaName);
120         ShardingSphereSchema renamedSchema = new ShardingSphereSchema(renamedSchemaName, schema.getAllTables(), schema.getAllViews());
121         database.addSchema(renamedSchema);
122         addDataNode(database, logicDataSourceName, schemaName, schema.getAllTables());
123     }
124     
125     private void addDataNode(final ShardingSphereDatabase database, final String logicDataSourceName, final String schemaName, final Collection<ShardingSphereTable> tobeAddedTables) {
126         for (ShardingSphereTable each : tobeAddedTables) {
127             if (!Strings.isNullOrEmpty(logicDataSourceName) && TableRefreshUtils.isSingleTable(each.getName(), database)) {
128                 database.getRuleMetaData().getAttributes(MutableDataNodeRuleAttribute.class).forEach(rule -> rule.put(logicDataSourceName, schemaName, each.getName()));
129             }
130         }
131     }
132     
133     private void addDataNode(final ShardingSphereDatabase database, final String logicDataSourceName, final String schemaName, final Collection<ShardingSphereTable> toBeAddedTables,
134                              final Collection<ShardingSphereView> toBeAddedViews) {
135         addTablesToDataNode(database, schemaName, logicDataSourceName, toBeAddedTables);
136         addViewsToDataNode(database, schemaName, logicDataSourceName, toBeAddedTables, toBeAddedViews);
137     }
138     
139     private void addTablesToDataNode(final ShardingSphereDatabase database, final String schemaName, final String logicDataSourceName, final Collection<ShardingSphereTable> toBeAddedTables) {
140         for (ShardingSphereTable each : toBeAddedTables) {
141             if (!Strings.isNullOrEmpty(logicDataSourceName) && TableRefreshUtils.isSingleTable(each.getName(), database)) {
142                 database.getRuleMetaData().getAttributes(MutableDataNodeRuleAttribute.class).forEach(rule -> rule.put(logicDataSourceName, schemaName, each.getName()));
143             }
144             database.getSchema(schemaName).putTable(each);
145         }
146     }
147     
148     private void addViewsToDataNode(final ShardingSphereDatabase database, final String schemaName, final String logicDataSourceName,
149                                     final Collection<ShardingSphereTable> toBeAddedTables, final Collection<ShardingSphereView> toBeAddedViews) {
150         for (ShardingSphereView view : toBeAddedViews) {
151             if (!Strings.isNullOrEmpty(logicDataSourceName) && TableRefreshUtils.isSingleTable(view.getName(), database)) {
152                 database.getRuleMetaData().getAttributes(MutableDataNodeRuleAttribute.class).forEach(each -> each.put(logicDataSourceName, schemaName, view.getName()));
153             }
154             toBeAddedTables.stream().filter(each -> each.getName().toLowerCase().equals(view.getName())).findFirst().ifPresent(optional -> database.getSchema(schemaName).putTable(optional));
155             database.getSchema(schemaName).putView(view);
156         }
157     }
158     
159     private void removeSchemaMetaData(final ShardingSphereDatabase database, final String schemaName) {
160         ShardingSphereSchema schema = new ShardingSphereSchema(schemaName, database.getSchema(schemaName).getAllTables(), database.getSchema(schemaName).getAllViews());
161         database.dropSchema(schemaName);
162         removeDataNode(database.getRuleMetaData().getAttributes(MutableDataNodeRuleAttribute.class), Collections.singleton(schemaName),
163                 schema.getAllTables().stream().map(ShardingSphereTable::getName).collect(Collectors.toSet()));
164     }
165     
166     private void removeDataNode(final Collection<MutableDataNodeRuleAttribute> ruleAttributes, final Collection<String> schemaNames, final Collection<String> tobeRemovedTables) {
167         tobeRemovedTables.forEach(each -> ruleAttributes.forEach(rule -> rule.remove(schemaNames, each)));
168     }
169     
170     private void removeDataNode(final ShardingSphereDatabase database, final String schemaName, final Collection<String> tobeRemovedTables, final Collection<String> tobeRemovedViews) {
171         removeTablesToDataNode(database, schemaName, tobeRemovedTables);
172         removeViewsToDataNode(database, schemaName, tobeRemovedTables, tobeRemovedViews);
173     }
174     
175     private void removeDataNode(final Collection<MutableDataNodeRuleAttribute> ruleAttributes, final String schemaName, final Collection<String> tobeRemovedTables) {
176         tobeRemovedTables.forEach(each -> ruleAttributes.forEach(rule -> rule.remove(schemaName, each)));
177     }
178     
179     private void removeTablesToDataNode(final ShardingSphereDatabase database, final String schemaName, final Collection<String> toBeDroppedTables) {
180         removeDataNode(database.getRuleMetaData().getAttributes(MutableDataNodeRuleAttribute.class), schemaName, toBeDroppedTables);
181         toBeDroppedTables.forEach(each -> database.getSchema(schemaName).removeTable(each));
182     }
183     
184     private void removeViewsToDataNode(final ShardingSphereDatabase database, final String schemaName, final Collection<String> toBeDroppedTables, final Collection<String> toBeDroppedViews) {
185         removeDataNode(database.getRuleMetaData().getAttributes(MutableDataNodeRuleAttribute.class), schemaName, toBeDroppedViews);
186         ShardingSphereSchema schema = database.getSchema(schemaName);
187         toBeDroppedTables.forEach(schema::removeTable);
188         toBeDroppedViews.forEach(schema::removeView);
189     }

191     @Override
192     public void dropSchema(final String databaseName, final Collection<String> schemaNames) {
193         Collection<String> tobeRemovedTables = new LinkedList<>();
194         Collection<String> tobeRemovedSchemas = new LinkedList<>();
195         ShardingSphereMetaData metaData = metaDataContextManager.getMetaDataContexts().getMetaData();
196         ShardingSphereDatabase database = metaData.getDatabase(databaseName);
197         for (String each : schemaNames) {
198             ShardingSphereSchema schema = new ShardingSphereSchema(each, database.getSchema(each).getAllTables(), database.getSchema(each).getAllViews());
199             database.dropSchema(each);
200             Optional.of(schema).ifPresent(optional -> tobeRemovedTables.addAll(optional.getAllTables().stream().map(ShardingSphereTable::getName).collect(Collectors.toSet())));
201             tobeRemovedSchemas.add(each.toLowerCase());
202         }
203         removeDataNode(database.getRuleMetaData().getAttributes(MutableDataNodeRuleAttribute.class), new HashSet<>(tobeRemovedSchemas), new HashSet<>(tobeRemovedTables));
204         metaData.getGlobalRuleMetaData().getRules().forEach(each -> ((GlobalRule) each).refresh(metaData.getAllDatabases(), GlobalRuleChangedType.SCHEMA_CHANGED));
205     }

207     @Override
208     public void alterSchemaMetaData(final AlterSchemaMetaDataPOJO alterSchemaMetaDataPOJO) {
209         String databaseName = alterSchemaMetaDataPOJO.getDatabaseName();
210         String schemaName = alterSchemaMetaDataPOJO.getSchemaName();
211         ShardingSphereMetaData metaData = metaDataContextManager.getMetaDataContexts().getMetaData();
212         ShardingSphereDatabase database = metaData.getDatabase(databaseName);
213         addDataNode(database, alterSchemaMetaDataPOJO.getLogicDataSourceName(), schemaName, alterSchemaMetaDataPOJO.getAlteredTables(), alterSchemaMetaDataPOJO.getAlteredViews());
214         removeDataNode(database, schemaName, alterSchemaMetaDataPOJO.getDroppedTables(), alterSchemaMetaDataPOJO.getDroppedViews());
215         metaData.getGlobalRuleMetaData().getRules().forEach(each -> ((GlobalRule) each).refresh(metaData.getAllDatabases(), GlobalRuleChangedType.SCHEMA_CHANGED));
216         metaDataPersistService.getDatabaseMetaDataFacade().getTable().persist(databaseName, schemaName, alterSchemaMetaDataPOJO.getAlteredTables());
217         metaDataPersistService.getDatabaseMetaDataFacade().getView().persist(databaseName, schemaName, alterSchemaMetaDataPOJO.getAlteredViews());
218         alterSchemaMetaDataPOJO.getDroppedTables().forEach(each -> metaDataPersistService.getDatabaseMetaDataFacade().getTable().drop(databaseName, schemaName, each));
219         alterSchemaMetaDataPOJO.getDroppedViews().forEach(each -> metaDataPersistService.getDatabaseMetaDataFacade().getView().delete(databaseName, schemaName, each));
220     }

222     @Override
223     public void registerStorageUnits(final String databaseName, final Map<String, DataSourcePoolProperties> toBeRegisteredProps) throws SQLException {
224         SwitchingResource switchingResource = metaDataContextManager.getResourceSwitchManager().switchByRegisterStorageUnit(metaDataContextManager.getMetaDataContexts()
225                 .getMetaData().getDatabase(databaseName).getResourceMetaData(), toBeRegisteredProps);
226         ShardingSphereDatabase changedDatabase = new MetaDataContextsFactory(metaDataPersistService, metaDataContextManager.getComputeNodeInstanceContext()).createChangedDatabase(
227                 databaseName, false, switchingResource, null, metaDataContextManager.getMetaDataContexts());
228         metaDataContextManager.getMetaDataContexts().getMetaData().putDatabase(changedDatabase);
229         metaDataContextManager.getMetaDataContexts().getMetaData().getGlobalRuleMetaData().getRules()
230                 .forEach(each -> ((GlobalRule) each).refresh(metaDataContextManager.getMetaDataContexts().getMetaData().getAllDatabases(), GlobalRuleChangedType.DATABASE_CHANGED));
231         metaDataContextManager.getMetaDataContexts().getMetaData().getDatabase(databaseName).getAllSchemas()
232                 .forEach(each -> {
233                     if (each.isEmpty()) {
234                         metaDataPersistService.getDatabaseMetaDataFacade().getSchema().add(databaseName, each.getName());
235                     }
236                     metaDataPersistService.getDatabaseMetaDataFacade().getTable().persist(databaseName, each.getName(), each.getAllTables());
237                 });
238         DataSourceUnitPersistService dataSourceService = metaDataPersistService.getDataSourceUnitService();
239         metaDataPersistService.getMetaDataVersionPersistService()
240                 .switchActiveVersion(dataSourceService.persist(databaseName, toBeRegisteredProps));
241         clearServiceCache();
242     }

244     @Override
245     public void alterStorageUnits(final String databaseName, final Map<String, DataSourcePoolProperties> toBeUpdatedProps) throws SQLException {
246         SwitchingResource switchingResource = metaDataContextManager.getResourceSwitchManager().switchByAlterStorageUnit(metaDataContextManager.getMetaDataContexts().getMetaData()
247                 .getDatabase(databaseName).getResourceMetaData(), toBeUpdatedProps);
248         ShardingSphereDatabase changedDatabase = new MetaDataContextsFactory(metaDataPersistService, metaDataContextManager.getComputeNodeInstanceContext()).createChangedDatabase(
249                 databaseName, true, switchingResource, null, metaDataContextManager.getMetaDataContexts());
250         metaDataContextManager.getMetaDataContexts().getMetaData().putDatabase(changedDatabase);
251         metaDataContextManager.getMetaDataContexts().getMetaData().getGlobalRuleMetaData().getRules()
252                 .forEach(each -> ((GlobalRule) each).refresh(metaDataContextManager.getMetaDataContexts().getMetaData().getAllDatabases(), GlobalRuleChangedType.DATABASE_CHANGED));
253         DataSourceUnitPersistService dataSourceService = metaDataPersistService.getDataSourceUnitService();
254         metaDataPersistService.getMetaDataVersionPersistService()
255                 .switchActiveVersion(dataSourceService.persist(databaseName, toBeUpdatedProps));
256         switchingResource.closeStaleDataSources();
257         clearServiceCache();
258     }

260     @Override
261     public void unregisterStorageUnits(final String databaseName, final Collection<String> toBeDroppedStorageUnitNames) throws SQLException {
262         SwitchingResource switchingResource = metaDataContextManager.getResourceSwitchManager().switchByUnregisterStorageUnit(metaDataContextManager.getMetaDataContexts().getMetaData()
263                 .getDatabase(databaseName).getResourceMetaData(), toBeDroppedStorageUnitNames);
264         MetaDataContexts reloadMetaDataContexts = new MetaDataContextsFactory(metaDataPersistService, metaDataContextManager.getComputeNodeInstanceContext()).createBySwitchResource(
265                 databaseName, false, switchingResource, metaDataContextManager.getMetaDataContexts());
266         metaDataPersistService.persistReloadDatabaseByDrop(databaseName, reloadMetaDataContexts.getMetaData().getDatabase(databaseName),
267                 metaDataContextManager.getMetaDataContexts().getMetaData().getDatabase(databaseName));
268         metaDataContextManager.dropSchemas(databaseName, reloadMetaDataContexts.getMetaData().getDatabase(databaseName),
269                 metaDataContextManager.getMetaDataContexts().getMetaData().getDatabase(databaseName));
270         metaDataContextManager.getMetaDataContexts().update(reloadMetaDataContexts);
271         switchingResource.closeStaleDataSources();
272         clearServiceCache();
273     }

275     @Override
276     public void alterSingleRuleConfiguration(final String databaseName, final Collection<RuleConfiguration> ruleConfigs) throws SQLException {
277         ruleConfigs.removeIf(each -> !each.getClass().isAssignableFrom(SingleRuleConfiguration.class));
278         Collection<MetaDataVersion> metaDataVersions = metaDataPersistService.getDatabaseRulePersistService()
279                 .persist(metaDataContextManager.getMetaDataContexts().getMetaData().getDatabase(databaseName).getName(), ruleConfigs);
280         metaDataPersistService.getMetaDataVersionPersistService().switchActiveVersion(metaDataVersions);
281         metaDataContextManager.getDatabaseRuleConfigurationManager().alterRuleConfiguration(databaseName, ruleConfigs.iterator().next());
282         clearServiceCache();
283     }

285     @Override
286     public void alterRuleConfiguration(final String databaseName, final RuleConfiguration toBeAlteredRuleConfig) throws SQLException {
287         if (null == toBeAlteredRuleConfig) {
288             return;
289         }
290         Collection<MetaDataVersion> metaDataVersions = metaDataPersistService.getDatabaseRulePersistService()
291                 .persist(metaDataContextManager.getMetaDataContexts().getMetaData().getDatabase(databaseName).getName(), Collections.singleton(toBeAlteredRuleConfig));
292         metaDataPersistService.getMetaDataVersionPersistService().switchActiveVersion(metaDataVersions);
293         for (MetaDataVersion each : metaDataVersions) {
294             // TODO double check here, when ruleItemEvent not existed or not AlterRuleItemEvent @haoran
295             Optional<RuleChangedItem> ruleItemChanged = buildAlterRuleItemChanged(databaseName, each, Type.UPDATED);
296             if (ruleItemChanged.isPresent() && ruleItemChanged.get() instanceof AlterRuleItem) {
297                 metaDataContextManager.getRuleItemManager().alterRuleItem((AlterRuleItem) ruleItemChanged.get());
298             }
299         }
300         clearServiceCache();
301     }
302     
303     private Optional<RuleChangedItem> buildAlterRuleItemChanged(final String databaseName, final MetaDataVersion metaDataVersion, final Type type) {
304         return ruleItemChangedBuilder.build(databaseName, metaDataVersion.getActiveVersionNodePath(), metaDataVersion.getNextActiveVersion(), type);
305     }

307     @Override
308     public void removeRuleConfigurationItem(final String databaseName, final RuleConfiguration toBeRemovedRuleConfig) throws SQLException {
309         if (null == toBeRemovedRuleConfig) {
310             return;
311         }
312         Collection<MetaDataVersion> metaDataVersions = metaDataPersistService.getDatabaseRulePersistService().delete(databaseName, Collections.singleton(toBeRemovedRuleConfig));
313         for (MetaDataVersion metaDataVersion : metaDataVersions) {
314             Optional<RuleChangedItem> ruleItemChanged = buildAlterRuleItemChanged(databaseName, metaDataVersion, Type.DELETED);
315             // TODO double check here, when ruleItemEvent not existed or not AlterRuleItemEvent @haoran
316             if (ruleItemChanged.isPresent() && ruleItemChanged.get() instanceof DropRuleItem) {
317                 metaDataContextManager.getRuleItemManager().dropRuleItem((DropRuleItem) ruleItemChanged.get());
318             }
319         }
320         clearServiceCache();
321     }

343     @Override
344     public void createTable(final String databaseName, final String schemaName, final ShardingSphereTable table, final String logicDataSourceName) {
345         ShardingSphereMetaData metaData = metaDataContextManager.getMetaDataContexts().getMetaData();
346         ShardingSphereDatabase database = metaData.getDatabase(databaseName);
347         addTableToDataNode(database, schemaName, logicDataSourceName, table);
348         metaData.getGlobalRuleMetaData().getRules().forEach(each -> ((GlobalRule) each).refresh(metaData.getAllDatabases(), GlobalRuleChangedType.SCHEMA_CHANGED));
349         metaDataPersistService.getDatabaseMetaDataFacade().getTable().persist(databaseName, schemaName, Collections.singleton(table));
350     }

352     @Override
353     public void dropTables(final String databaseName, final String schemaName, final Collection<String> tableNames) {
354         ShardingSphereMetaData metaData = metaDataContextManager.getMetaDataContexts().getMetaData();
355         ShardingSphereDatabase database = metaData.getDatabase(databaseName);
356         removeTablesToDataNode(database, schemaName, tableNames);
357         metaData.getGlobalRuleMetaData().getRules().forEach(each -> ((GlobalRule) each).refresh(metaData.getAllDatabases(), GlobalRuleChangedType.SCHEMA_CHANGED));
358         tableNames.forEach(each -> metaDataPersistService.getDatabaseMetaDataFacade().getTable().drop(databaseName, schemaName, each));
359     }

361     private void addTableToDataNode(final ShardingSphereDatabase database, final String schemaName, final String logicDataSourceName, final ShardingSphereTable table) {
362         if (!Strings.isNullOrEmpty(logicDataSourceName) && TableRefreshUtils.isSingleTable(table.getName(), database)) {
363             database.getRuleMetaData().getAttributes(MutableDataNodeRuleAttribute.class).forEach(rule -> rule.put(logicDataSourceName, schemaName, table.getName()));
364         }
365         database.getSchema(schemaName).putTable(table);
366     }
