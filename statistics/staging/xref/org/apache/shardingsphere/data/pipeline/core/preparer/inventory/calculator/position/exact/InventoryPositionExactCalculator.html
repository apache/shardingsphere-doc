20  import lombok.AccessLevel;
21  import lombok.NoArgsConstructor;
22  import lombok.extern.slf4j.Slf4j;
23  import org.apache.shardingsphere.data.pipeline.core.datasource.PipelineDataSource;
24  import org.apache.shardingsphere.data.pipeline.core.exception.job.SplitPipelineJobByUniqueKeyException;
25  import org.apache.shardingsphere.data.pipeline.core.ingest.position.IngestPosition;
26  import org.apache.shardingsphere.data.pipeline.core.ingest.position.type.pk.PrimaryKeyIngestPosition;
27  import org.apache.shardingsphere.data.pipeline.core.sqlbuilder.sql.PipelinePrepareSQLBuilder;
28  import org.apache.shardingsphere.infra.metadata.database.schema.QualifiedTable;
29  
30  import java.sql.Connection;
31  import java.sql.PreparedStatement;
32  import java.sql.ResultSet;
33  import java.sql.SQLException;
34  import java.util.LinkedList;
35  import java.util.List;
36  
37  /**
38   * Inventory position exact calculator.
39   */
40  @NoArgsConstructor(access = AccessLevel.NONE)
41  @Slf4j
42  public final class InventoryPositionExactCalculator {
43      
44      /**
45       * Get positions by table unique key range.
46       *
47       * @param <T> type of unique key
48       * @param qualifiedTable qualified table
49       * @param uniqueKey unique key
50       * @param shardingSize sharding size
51       * @param dataSource data source
52       * @param positionHandler position handler
53       * @return positions
54       * @throws SplitPipelineJobByUniqueKeyException if an error occurs while splitting table by unique key
55       */
56      public static <T> List<IngestPosition> getPositions(final QualifiedTable qualifiedTable, final String uniqueKey, final int shardingSize,
57                                                          final PipelineDataSource dataSource, final DataTypePositionHandler<T> positionHandler) {
58          List<IngestPosition> result = new LinkedList<>();
59          PrimaryKeyIngestPosition<T> firstPosition = getFirstPosition(qualifiedTable, uniqueKey, shardingSize, dataSource, positionHandler);
60          result.add(firstPosition);
61          result.addAll(getLeftPositions(qualifiedTable, uniqueKey, shardingSize, firstPosition, dataSource, positionHandler));
62          return result;
63      }
64      
65      private static <T> PrimaryKeyIngestPosition<T> getFirstPosition(final QualifiedTable qualifiedTable, final String uniqueKey, final int shardingSize,
66                                                                      final PipelineDataSource dataSource, final DataTypePositionHandler<T> positionHandler) {
67          String firstQuerySQL = new PipelinePrepareSQLBuilder(dataSource.getDatabaseType())
68                  .buildSplitByUniqueKeyRangedSQL(qualifiedTable.getSchemaName(), qualifiedTable.getTableName(), uniqueKey, false);
69          try (
70                  Connection connection = dataSource.getConnection();
71                  PreparedStatement preparedStatement = connection.prepareStatement(firstQuerySQL)) {
72              preparedStatement.setLong(1, shardingSize);
73              try (ResultSet resultSet = preparedStatement.executeQuery()) {
74                  if (!resultSet.next()) {
75                      log.info("No any record, return. First query SQL: {}", firstQuerySQL);
76                      return positionHandler.createIngestPosition(null, null);
77                  }
78                  long count = resultSet.getLong(2);
79                  T minValue = positionHandler.readColumnValue(resultSet, 3);
80                  T maxValue = positionHandler.readColumnValue(resultSet, 1);
81                  log.info("First records count: {}, min value: {}, max value: {}, sharding size: {}, first query SQL: {}", count, minValue, maxValue, shardingSize, firstQuerySQL);
82                  if (0 == count) {
83                      return positionHandler.createIngestPosition(null, null);
84                  }
85                  return positionHandler.createIngestPosition(minValue, maxValue);
86              }
87          } catch (final SQLException ex) {
88              throw new SplitPipelineJobByUniqueKeyException(qualifiedTable.getTableName(), uniqueKey, ex);
89          }
90      }
91      
92      private static <T> List<IngestPosition> getLeftPositions(final QualifiedTable qualifiedTable, final String uniqueKey,
93                                                               final int shardingSize, final PrimaryKeyIngestPosition<T> firstPosition,
94                                                               final PipelineDataSource dataSource, final DataTypePositionHandler<T> positionHandler) {
95          List<IngestPosition> result = new LinkedList<>();
96          T lowerValue = firstPosition.getEndValue();
97          long recordsCount = 0;
98          String laterQuerySQL = new PipelinePrepareSQLBuilder(dataSource.getDatabaseType())
99                  .buildSplitByUniqueKeyRangedSQL(qualifiedTable.getSchemaName(), qualifiedTable.getTableName(), uniqueKey, true);
100         try (
101                 Connection connection = dataSource.getConnection();
102                 PreparedStatement preparedStatement = connection.prepareStatement(laterQuerySQL)) {
103             for (int i = 0; i < Integer.MAX_VALUE; i++) {
104                 positionHandler.setPreparedStatementValue(preparedStatement, 1, lowerValue);
105                 preparedStatement.setLong(2, shardingSize);
106                 try (ResultSet resultSet = preparedStatement.executeQuery()) {
107                     if (!resultSet.next()) {
108                         break;
109                     }
110                     int count = resultSet.getInt(2);
111                     if (0 == count) {
112                         log.info("Done. Later records count: {}, last lower value: {}, sharding size: {}, later query SQL: {}", recordsCount, lowerValue, shardingSize, laterQuerySQL);
113                         break;
114                     }
115                     recordsCount += count;
116                     T minValue = positionHandler.readColumnValue(resultSet, 3);
117                     T maxValue = positionHandler.readColumnValue(resultSet, 1);
118                     result.add(positionHandler.createIngestPosition(minValue, maxValue));
119                     lowerValue = maxValue;
120                 }
121             }
122         } catch (final SQLException ex) {
123             throw new SplitPipelineJobByUniqueKeyException(qualifiedTable.getTableName(), uniqueKey, ex);
124         }
125         return result;
126     }
127 }
