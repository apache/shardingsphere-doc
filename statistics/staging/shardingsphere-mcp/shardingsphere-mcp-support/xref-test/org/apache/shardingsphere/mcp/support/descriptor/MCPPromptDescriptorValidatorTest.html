18  package org.apache.shardingsphere.mcp.support.descriptor;
19  
20  import org.apache.shardingsphere.mcp.api.capability.prompt.MCPPromptDescriptor;
21  import org.junit.jupiter.params.ParameterizedTest;
22  import org.junit.jupiter.params.provider.Arguments;
23  import org.junit.jupiter.params.provider.MethodSource;
24  
25  import java.util.Collection;
26  import java.util.List;
27  import java.util.Map;
28  
29  import static org.hamcrest.MatcherAssert.assertThat;
30  import static org.hamcrest.Matchers.is;
31  import static org.junit.jupiter.api.Assertions.assertThrows;
32  
33  class MCPPromptDescriptorValidatorTest {
34      
35      @ParameterizedTest(name = "{0}")
36      @MethodSource("invalidPrompts")
37      void assertValidateWithInvalidPrompt(final String name, final Collection<MCPPromptDescriptor> descriptors, final Collection<MCPPromptTemplateBinding> bindings,
38                                           final String expectedMessage) {
39          IllegalStateException actual = assertThrows(IllegalStateException.class, () -> MCPPromptDescriptorValidator.validate(descriptors, bindings));
40          assertThat(actual.getMessage(), is(expectedMessage));
41      }
42      
43      private static Collection<Arguments> invalidPrompts() {
44          MCPPromptDescriptor prompt = new MCPPromptDescriptor("test_prompt", "Test Prompt", "Guide the model through a test prompt.", List.of(), Map.of());
45          MCPPromptTemplateBinding binding = new MCPPromptTemplateBinding("test_prompt", "META-INF/shardingsphere-mcp/prompts/fixture-uri-template.md");
46          return List.of(
47                  Arguments.of("unsupported placeholder", List.of(prompt),
48                          List.of(new MCPPromptTemplateBinding("test_prompt", "META-INF/shardingsphere-mcp/prompts/fixture-single-brace-placeholder.md")),
49                          "Prompt template `META-INF/shardingsphere-mcp/prompts/fixture-single-brace-placeholder.md` contains unsupported model-facing placeholder `{database}`."),
50                  Arguments.of("missing binding", List.of(prompt), List.of(), "Prompt `test_prompt` must declare an internal template binding."),
51                  Arguments.of("duplicate prompt", List.of(prompt, prompt), List.of(binding), "Duplicate MCP prompt descriptor `test_prompt`."),
52                  Arguments.of("undeclared placeholder", List.of(prompt), List.of(new MCPPromptTemplateBinding("test_prompt", "META-INF/shardingsphere-mcp/prompts/fixture-markdown-resource.md")),
53                          "Prompt template `META-INF/shardingsphere-mcp/prompts/fixture-markdown-resource.md` has undeclared placeholder `database`."),
54                  Arguments.of("client form metadata is not a list", List.of(new MCPPromptDescriptor("test_prompt", "Test Prompt", "Guide the model through a test prompt.", List.of(),
55                          Map.of("org.apache.shardingsphere/client-form-only-arguments", "database"))), List.of(binding),
56                          "Prompt `test_prompt` metadata `org.apache.shardingsphere/client-form-only-arguments` must be a list."));
57      }
58  }
