84      void assertSetIndexForItemsByIndexOrderBy() {
85          SelectStatement selectStatement = new SQL92SelectStatement();
86          selectStatement.setOrderBy(new OrderBySegment(0, 0, Collections.singletonList(createOrderByItemSegment(INDEX_ORDER_BY))));
87          selectStatement.setProjections(createProjectionsSegment());
88          TableNameSegment tableNameSegment = new TableNameSegment(0, 0, new IdentifierValue("table"));
89          tableNameSegment.setTableBoundInfo(new TableSegmentBoundInfo(new IdentifierValue("foo_db"), new IdentifierValue("foo_schema")));
90          selectStatement.setFrom(new SimpleTableSegment(tableNameSegment));
91          ShardingSphereDatabase database = mockDatabase();
92          SelectStatementContext selectStatementContext =
93                  new SelectStatementContext(createShardingSphereMetaData(database), Collections.emptyList(), selectStatement, "foo_db", Collections.emptyList());

95          assertThat(selectStatementContext.getOrderByContext().getItems().iterator().next().getIndex(), is(4));

292     private OrderByItemSegment createOrderByItemSegment(final String type) {
293         switch (type) {
294             case INDEX_ORDER_BY:
295                 return new IndexOrderByItemSegment(0, 0, 4, OrderDirection.ASC, NullsOrderType.FIRST);
296             case COLUMN_ORDER_BY_WITH_OWNER:
297                 ColumnSegment columnSegment = new ColumnSegment(0, 0, new IdentifierValue("name"));
298                 OwnerSegment owner = new OwnerSegment(0, 0, new IdentifierValue("table"));
299                 owner.setTableBoundInfo(new TableSegmentBoundInfo(new IdentifierValue("foo_db"), new IdentifierValue("foo_schema")));
300                 columnSegment.setOwner(owner);
301                 return new ColumnOrderByItemSegment(columnSegment, OrderDirection.ASC, NullsOrderType.FIRST);
302             case COLUMN_ORDER_BY_WITH_ALIAS:
303                 return new ColumnOrderByItemSegment(new ColumnSegment(0, 0, new IdentifierValue("n")), OrderDirection.ASC, NullsOrderType.FIRST);
304             default:
305                 return new ColumnOrderByItemSegment(new ColumnSegment(0, 0, new IdentifierValue("id")), OrderDirection.ASC, NullsOrderType.FIRST);
306         }
307     }

309     private ProjectionsSegment createProjectionsSegment() {
310         ProjectionsSegment result = new ProjectionsSegment(0, 0);
311         result.setDistinctRow(true);
312         result.getProjections().addAll(Arrays.asList(getColumnProjectionSegmentWithoutOwner(), getColumnProjectionSegmentWithoutOwner(true), getColumnProjectionSegmentWithoutOwner(false)));
313         return result;
314     }
315     
316     private ProjectionSegment getColumnProjectionSegmentWithoutOwner() {
317         ColumnSegment columnSegment = new ColumnSegment(0, 0, new IdentifierValue("name"));
318         OwnerSegment owner = new OwnerSegment(0, 0, new IdentifierValue("table"));
319         owner.setTableBoundInfo(new TableSegmentBoundInfo(new IdentifierValue("foo_db"), new IdentifierValue("foo_schema")));
320         columnSegment.setOwner(owner);
321         return new ColumnProjectionSegment(columnSegment);
322     }
323     
324     private ProjectionSegment getColumnProjectionSegmentWithoutOwner(final boolean hasAlias) {
325         ColumnSegment columnSegment = new ColumnSegment(0, 0, new IdentifierValue(hasAlias ? "name" : "id"));
326         ColumnProjectionSegment columnProjectionSegment = new ColumnProjectionSegment(columnSegment);
327         columnProjectionSegment.setAlias(new AliasSegment(0, 0, new IdentifierValue(hasAlias ? "n" : null)));
328         return columnProjectionSegment;
329     }
