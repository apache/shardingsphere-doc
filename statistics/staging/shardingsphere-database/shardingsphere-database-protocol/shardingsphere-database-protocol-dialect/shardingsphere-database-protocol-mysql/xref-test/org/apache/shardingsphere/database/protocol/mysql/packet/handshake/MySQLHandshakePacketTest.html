48  @ExtendWith(MockitoExtension.class)
49  class MySQLHandshakePacketTest {
50      
51      private static final String EXPECTED_SERVER_VERSION = DatabaseProtocolServerInfo.getDefaultProtocolVersion(TypedSPILoader.getService(DatabaseType.class, "MySQL"));
52      
53      private static final int SECURE_CONNECTION_CAPABILITY_FLAGS_LOWER = MySQLCapabilityFlag.calculateHandshakeCapabilityFlagsLower();
54      
55      private static final int PLUGIN_AUTH_CAPABILITY_FLAGS_UPPER = MySQLCapabilityFlag.CLIENT_PLUGIN_AUTH.getValue() >> 16;
56      
57      private static final byte[] FOO_AUTH_PLUGIN_DATA_PART_1 = {106, 105, 55, 122, 117, 98, 115, 109};
58      
59      private static final byte[] BAR_AUTH_PLUGIN_DATA_PART_2 = {68, 102, 53, 122, 65, 49, 84, 79, 85, 115, 116, 113};
60      
61      @Mock
62      private MySQLPacketPayload payload;
63      
64      @ParameterizedTest(name = "{0}")
65      @MethodSource("constructWithPayloadArguments")
66      void assertConstructWithPayload(final String name, final int capabilityFlagsLower, final int capabilityFlagsUpper,
67                                      final byte[] expectedAuthPluginDataPart2, final String expectedAuthPluginName) {
68          when(payload.readInt1()).thenReturn(MySQLConstants.PROTOCOL_VERSION, MySQLConstants.DEFAULT_CHARSET.getId(), 0);
69          when(payload.readStringNul()).thenReturn(EXPECTED_SERVER_VERSION, MySQLAuthenticationMethod.NATIVE.getMethodName());
70          when(payload.readStringNulByBytes()).thenReturn(FOO_AUTH_PLUGIN_DATA_PART_1, BAR_AUTH_PLUGIN_DATA_PART_2);
71          when(payload.readInt4()).thenReturn(1000);
72          when(payload.readInt2()).thenReturn(capabilityFlagsLower, MySQLStatusFlag.SERVER_STATUS_AUTOCOMMIT.getValue(), capabilityFlagsUpper);
73          MySQLHandshakePacket actual = new MySQLHandshakePacket(payload);
74          assertThat(actual.getServerVersion(), is(EXPECTED_SERVER_VERSION));
75          assertThat(actual.getCapabilityFlagsLower(), is(capabilityFlagsLower));
76          assertThat(actual.getConnectionId(), is(1000));
77          assertThat(actual.getCharacterSet(), is(MySQLConstants.DEFAULT_CHARSET.getId()));
78          assertThat(actual.getStatusFlag(), is(MySQLStatusFlag.SERVER_STATUS_AUTOCOMMIT));
79          assertThat(actual.getCapabilityFlagsUpper(), is(capabilityFlagsUpper));
80          assertThat(actual.getAuthPluginData().getAuthenticationPluginDataPart1(), is(FOO_AUTH_PLUGIN_DATA_PART_1));
81          assertThat(actual.getAuthPluginData().getAuthenticationPluginDataPart2(), is(expectedAuthPluginDataPart2));
82          verify(payload).skipReserved(10);
83          if (null == expectedAuthPluginName) {
84              assertNull(actual.getAuthPluginName());
85          } else {
86              assertThat(actual.getAuthPluginName(), is(expectedAuthPluginName));
87          }
88      }
89      
90      @Test
91      void assertConstructWithInvalidProtocolVersion() {
92          when(payload.readInt1()).thenReturn(MySQLConstants.PROTOCOL_VERSION + 1);
93          assertThrows(IllegalArgumentException.class, () -> new MySQLHandshakePacket(payload));
94      }
95      
96      @Test
97      void assertConstructWithSSLEnabled() {
98          MySQLHandshakePacket actual = new MySQLHandshakePacket(1, true, new MySQLAuthenticationPluginData());
99          assertThat(actual.getCapabilityFlagsLower() & MySQLCapabilityFlag.CLIENT_SSL.getValue(), is(MySQLCapabilityFlag.CLIENT_SSL.getValue()));
100         assertThat(actual.getAuthPluginName(), is(MySQLAuthenticationMethod.CACHING_SHA2_PASSWORD.getMethodName()));
101     }
102     
103     @Test
104     void assertConstructWithoutSSLEnabled() {
105         MySQLHandshakePacket actual = new MySQLHandshakePacket(1, false, new MySQLAuthenticationPluginData());
106         assertThat(actual.getCapabilityFlagsLower() & MySQLCapabilityFlag.CLIENT_SSL.getValue(), is(0));
107     }
108     
109     @Test
110     void assertSetAuthPluginName() {
111         mockPayloadForConstruct();
112         MySQLHandshakePacket actual = new MySQLHandshakePacket(payload);
113         actual.setAuthPluginName(MySQLAuthenticationMethod.NATIVE);
114         assertThat(actual.getAuthPluginName(), is(MySQLAuthenticationMethod.NATIVE.getMethodName()));
115         assertThat(actual.getCapabilityFlagsUpper() & PLUGIN_AUTH_CAPABILITY_FLAGS_UPPER, is(PLUGIN_AUTH_CAPABILITY_FLAGS_UPPER));
116     }
117     
118     @Test
119     void assertWriteWithSecureConnectionAndPluginAuth() {
120         MySQLAuthenticationPluginData authPluginData = new MySQLAuthenticationPluginData(FOO_AUTH_PLUGIN_DATA_PART_1, BAR_AUTH_PLUGIN_DATA_PART_2);
121         MySQLHandshakePacket actual = new MySQLHandshakePacket(1000, false, authPluginData);
122         actual.write((PacketPayload) payload);
123         verify(payload).writeInt1(MySQLConstants.PROTOCOL_VERSION);
124         verify(payload).writeStringNul(EXPECTED_SERVER_VERSION);
125         verify(payload).writeInt4(1000);
126         verify(payload).writeStringNul(new String(authPluginData.getAuthenticationPluginDataPart1()));
127         verify(payload).writeInt2(SECURE_CONNECTION_CAPABILITY_FLAGS_LOWER);
128         verify(payload).writeInt1(MySQLConstants.DEFAULT_CHARSET.getId());
129         verify(payload).writeInt2(MySQLStatusFlag.SERVER_STATUS_AUTOCOMMIT.getValue());
130         verify(payload).writeInt2(MySQLCapabilityFlag.calculateHandshakeCapabilityFlagsUpper());
131         verify(payload).writeInt1(authPluginData.getAuthenticationPluginData().length + 1);
132         verify(payload).writeReserved(10);
133         verify(payload).writeStringNul(new String(authPluginData.getAuthenticationPluginDataPart2()));
134         verify(payload).writeStringNul(MySQLAuthenticationMethod.CACHING_SHA2_PASSWORD.getMethodName());
135     }
136     
137     @Test
138     void assertWriteWithoutSecureConnectionAndPluginAuth() {
139         mockPayloadForConstruct();
140         MySQLHandshakePacket actual = new MySQLHandshakePacket(payload);
141         actual.write((PacketPayload) payload);
142         verify(payload).writeInt1(MySQLConstants.PROTOCOL_VERSION);
143         verify(payload).writeStringNul(EXPECTED_SERVER_VERSION);
144         verify(payload).writeInt4(1000);
145         verify(payload).writeStringNul(new String(FOO_AUTH_PLUGIN_DATA_PART_1));
146         verify(payload, times(2)).writeInt2(0);
147         verify(payload).writeInt1(MySQLConstants.DEFAULT_CHARSET.getId());
148         verify(payload).writeInt2(MySQLStatusFlag.SERVER_STATUS_AUTOCOMMIT.getValue());
149         verify(payload).writeInt1(0);
150         verify(payload).writeReserved(10);
151         verify(payload, never()).writeStringNul(new String(BAR_AUTH_PLUGIN_DATA_PART_2));
152         verify(payload, never()).writeStringNul(MySQLAuthenticationMethod.CACHING_SHA2_PASSWORD.getMethodName());
153     }
154     
155     private void mockPayloadForConstruct() {
156         when(payload.readInt1()).thenReturn(MySQLConstants.PROTOCOL_VERSION, MySQLConstants.DEFAULT_CHARSET.getId(), 0);
157         when(payload.readStringNul()).thenReturn(EXPECTED_SERVER_VERSION, MySQLAuthenticationMethod.NATIVE.getMethodName());
158         when(payload.readStringNulByBytes()).thenReturn(FOO_AUTH_PLUGIN_DATA_PART_1, BAR_AUTH_PLUGIN_DATA_PART_2);
159         when(payload.readInt4()).thenReturn(1000);
160         when(payload.readInt2()).thenReturn(0, MySQLStatusFlag.SERVER_STATUS_AUTOCOMMIT.getValue(), 0);
161     }
162     
163     private static Stream<Arguments> constructWithPayloadArguments() {
164         return Stream.of(
165                 Arguments.of("secure_connection_and_plugin_auth", SECURE_CONNECTION_CAPABILITY_FLAGS_LOWER, PLUGIN_AUTH_CAPABILITY_FLAGS_UPPER,
166                         BAR_AUTH_PLUGIN_DATA_PART_2, MySQLAuthenticationMethod.NATIVE.getMethodName()),
167                 Arguments.of("secure_connection_without_plugin_auth", SECURE_CONNECTION_CAPABILITY_FLAGS_LOWER, 0, BAR_AUTH_PLUGIN_DATA_PART_2, null),
168                 Arguments.of("no_secure_connection_and_no_plugin_auth", 0, 0, new byte[0], null));
169     }
