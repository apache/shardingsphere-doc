52  @ExtendWith(AutoMockExtension.class)
53  @StaticMockSettings(ExpressionConverter.class)
54  class AggregationProjectionConverterTest {
55      
56      @Test
57      void assertConvertReturnsEmptyForNullSegment() {
58          assertFalse(AggregationProjectionConverter.convert(null).isPresent());
59      }
60      
61      @Test
62      void assertConvertBuildsDistinctAggregationWithAliasAndStarParameter() {
63          AggregationDistinctProjectionSegment segment = new AggregationDistinctProjectionSegment(0, 0, AggregationType.COUNT, "COUNT(*)", "COUNT(*)");
64          segment.setAlias(new AliasSegment(0, 0, new IdentifierValue("alias")));
65          Optional<SqlNode> actual = AggregationProjectionConverter.convert(segment);
66          assertTrue(actual.isPresent());
67          SqlBasicCall asCall = (SqlBasicCall) actual.orElse(null);
68          assertThat(asCall.getOperator(), is(SqlStdOperatorTable.AS));
69          SqlBasicCall innerCall = (SqlBasicCall) asCall.getOperandList().get(0);
70          assertThat(innerCall.getOperator(), is(SqlStdOperatorTable.COUNT));
71          assertNotNull(innerCall.getFunctionQuantifier());
72          assertThat(innerCall.getOperandList().size(), is(1));
73          assertThat(innerCall.getOperandList().get(0), isA(SqlIdentifier.class));
74          SqlIdentifier starIdentifier = (SqlIdentifier) innerCall.getOperandList().get(0);
75          assertTrue(starIdentifier.isStar());
76          SqlIdentifier aliasIdentifier = (SqlIdentifier) asCall.getOperandList().get(1);
77          assertThat(aliasIdentifier.names, is(Collections.singletonList("alias")));
78      }
79      
80      @Test
81      void assertConvertBuildsAggregationWithParametersAndSeparator() {
82          AggregationProjectionSegment segment = new AggregationProjectionSegment(0, 0, AggregationType.SUM, "SUM(expr)", "|");
83          ExpressionSegment firstParam = mock(ExpressionSegment.class);
84          ExpressionSegment secondParam = mock(ExpressionSegment.class);
85          segment.getParameters().addAll(Arrays.asList(firstParam, secondParam));
86          SqlNode firstNode = mock(SqlNode.class);
87          SqlNode secondNode = mock(SqlNode.class);
88          when(ExpressionConverter.convert(firstParam)).thenReturn(Optional.of(firstNode));
89          when(ExpressionConverter.convert(secondParam)).thenReturn(Optional.of(secondNode));
90          Optional<SqlNode> actual = AggregationProjectionConverter.convert(segment);
91          SqlBasicCall sqlBasicCall = (SqlBasicCall) actual.orElse(null);
92          assertNotNull(sqlBasicCall);
93          assertThat(sqlBasicCall.getOperator(), is(SqlStdOperatorTable.SUM));
94          assertThat(sqlBasicCall.getOperandList().size(), is(3));
95          assertThat(sqlBasicCall.getOperandList().get(0), is(firstNode));
96          assertThat(sqlBasicCall.getOperandList().get(1), is(secondNode));
97          assertThat(sqlBasicCall.getOperandList().get(2), isA(SqlLiteral.class));
98          assertNull(sqlBasicCall.getFunctionQuantifier());
99      }
100     
101     @Test
102     void assertConvertBuildsAggregationWithParametersWithoutSeparator() {
103         AggregationProjectionSegment segment = new AggregationProjectionSegment(0, 0, AggregationType.MAX, "MAX(expr)");
104         ExpressionSegment param = mock(ExpressionSegment.class);
105         segment.getParameters().add(param);
106         SqlNode expectedNode = mock(SqlNode.class);
107         when(ExpressionConverter.convert(param)).thenReturn(Optional.of(expectedNode));
108         Optional<SqlNode> actual = AggregationProjectionConverter.convert(segment);
109         SqlBasicCall sqlBasicCall = (SqlBasicCall) actual.orElse(null);
110         assertNotNull(sqlBasicCall);
111         assertThat(sqlBasicCall.getOperator(), is(SqlStdOperatorTable.MAX));
112         assertThat(sqlBasicCall.getOperandList().size(), is(1));
113         assertThat(sqlBasicCall.getOperandList().get(0), is(expectedNode));
114         assertNull(sqlBasicCall.getFunctionQuantifier());
115     }
116     
117     @Test
118     void assertConvertThrowsExceptionForUnsupportedOperator() {
119         IllegalStateException ex = assertThrows(IllegalStateException.class,
120                 () -> AggregationProjectionConverter.convert(new AggregationProjectionSegment(0, 0, AggregationType.PRODUCT, "PRODUCT(expr)")));
121         assertThat(ex.getMessage(), is("Unsupported SQL operator: `PRODUCT`"));
122     }
