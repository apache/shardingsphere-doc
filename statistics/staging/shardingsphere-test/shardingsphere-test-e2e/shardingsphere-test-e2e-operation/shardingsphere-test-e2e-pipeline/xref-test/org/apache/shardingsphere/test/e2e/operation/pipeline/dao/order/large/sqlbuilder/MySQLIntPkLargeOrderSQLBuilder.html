90      @Override
91      public List<Object[]> generateInsertData(final KeyGenerateAlgorithm keyGenerateAlgorithm, final int recordCount) {
92          List<Object[]> result = new ArrayList<>(recordCount);
93          for (int i = 0; i < recordCount; i++) {
94              int randomInt = PipelineCaseHelper.generateInt(-100, 100);
95              Object orderId = keyGenerateAlgorithm.generateKeys(mock(AlgorithmSQLContext.class), 1).iterator().next();
96              int randomUnsignedInt = PipelineCaseHelper.generateInt(0, 100);
97              LocalDateTime now = LocalDateTime.now();
98              Object[] params = {
99                      orderId, PipelineCaseHelper.generateInt(0, 100), PipelineCaseHelper.generateString(6),
100                     randomInt, randomInt, randomInt, randomUnsignedInt, randomUnsignedInt, randomUnsignedInt, randomUnsignedInt,
101                     PipelineCaseHelper.generateFloat(), PipelineCaseHelper.generateDouble(), BigDecimal.valueOf(PipelineCaseHelper.generateDouble()),
102                     now, now, now.toLocalDate(), now.toLocalTime(), Year.now().getValue(),
103                     "1", "t", "e", "s", "t",
104                     PipelineCaseHelper.generateString(2), "☠️x☺️x✋x☹️", PipelineCaseHelper.generateString(1),
105                     "1", "2",
106                     0 == i ? PipelineCaseHelper.generateJsonLiteralNull() : PipelineCaseHelper.generateJsonString(32, false)
107             };
108             result.add(params);
109         }
110         return result;
111     }

115         return "MySQL";
116     }
