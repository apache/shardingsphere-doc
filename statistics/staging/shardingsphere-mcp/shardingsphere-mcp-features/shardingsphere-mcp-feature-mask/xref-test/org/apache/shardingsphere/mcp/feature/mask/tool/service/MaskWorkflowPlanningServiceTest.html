206     void assertPlanStopsOnBlockingAlgorithmIssue() {
207         MaskRuleInspectionService ruleInspectionService = mock(MaskRuleInspectionService.class);
208         when(ruleInspectionService.queryMaskRules(any(), any(), any())).thenReturn(List.of());
209         when(ruleInspectionService.queryMaskAlgorithms(any())).thenReturn(List.of());
210         MaskAlgorithmRecommendationService algorithmRecommendationService = mock(MaskAlgorithmRecommendationService.class);
211         when(algorithmRecommendationService.recommendMaskAlgorithms(any(), any(), any())).thenAnswer(invocation -> {
212             List<WorkflowIssue> issues = invocation.getArgument(2);
213             issues.add(new WorkflowIssue(WorkflowIssueCode.ALGORITHM_NOT_FOUND, "error", "selecting-algorithm", "missing", "fix", false, Map.of()));
214             return List.of();
215         });
216         WorkflowContextSnapshot actual = createService(ruleInspectionService, algorithmRecommendationService, mock(MaskAlgorithmPropertyTemplateService.class),
217                 mock(MaskRuleDistSQLPlanningService.class)).plan(new TestWorkflowSessionContext(), createMetadataQueryFacade(), createQueryFacade(),
218                         createRequest("create"));
219         assertThat(actual.getStatus(), is("clarifying"));
220         assertThat(actual.getClarifiedIntent().getClarificationMessages().getFirst(), is("Please use a mask algorithm visible in the current Proxy."));

310     private MaskWorkflowPlanningService createService(final MaskRuleInspectionService ruleInspectionService,
311                                                       final MaskAlgorithmRecommendationService algorithmRecommendationService,
312                                                       final MaskAlgorithmPropertyTemplateService algorithmPropertyTemplateService,
313                                                       final MaskRuleDistSQLPlanningService ruleDistSQLPlanningService) {
314         try (
315                 MockedConstruction<MaskRuleInspectionService> ignoredInspectionService = mockConstruction(
316                         MaskRuleInspectionService.class, withSettings().defaultAnswer(AdditionalAnswers.delegatesTo(ruleInspectionService)));
317                 MockedConstruction<MaskAlgorithmRecommendationService> ignoredRecommendationService = mockConstruction(
318                         MaskAlgorithmRecommendationService.class, withSettings().defaultAnswer(AdditionalAnswers.delegatesTo(algorithmRecommendationService)));
319                 MockedConstruction<MaskAlgorithmPropertyTemplateService> ignoredPropertyTemplateService = mockConstruction(
320                         MaskAlgorithmPropertyTemplateService.class, withSettings().defaultAnswer(AdditionalAnswers.delegatesTo(algorithmPropertyTemplateService)));
321                 MockedConstruction<MaskRuleDistSQLPlanningService> ignoredDistSQLPlanningService = mockConstruction(
322                         MaskRuleDistSQLPlanningService.class, withSettings().defaultAnswer(AdditionalAnswers.delegatesTo(ruleDistSQLPlanningService)))) {
323             return new MaskWorkflowPlanningService();
324         }
325     }

327     private void assertRuleDistSQLOnlyPayloadDoesNotExpose(final WorkflowContextSnapshot snapshot, final String term) {
328         Map<String, Object> actualPayload = WorkflowPlanPayloadBuilder.buildWithArtifacts(snapshot, snapshot.getRequest());
329         assertFalse(String.valueOf(actualPayload).toLowerCase(Locale.ENGLISH).contains(term));
330     }
