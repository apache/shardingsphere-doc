18  package org.apache.shardingsphere.mcp.support.descriptor;
19  
20  import lombok.AccessLevel;
21  import lombok.AllArgsConstructor;
22  import org.apache.shardingsphere.mcp.api.capability.completion.MCPCompletionTargetDescriptor;
23  import org.apache.shardingsphere.mcp.support.protocol.MCPResponseMode;
24  
25  import java.util.Collection;
26  import java.util.LinkedHashMap;
27  import java.util.List;
28  import java.util.Map;
29  
30  @AllArgsConstructor(access = AccessLevel.PRIVATE)
31  final class MCPDescriptorCatalogPayloadBuilder {
32      
33      private final MCPDescriptorCatalog catalog;
34      
35      static Map<String, Object> build(final MCPDescriptorCatalog catalog, final Collection<?> supportedStatements) {
36          return new MCPDescriptorCatalogPayloadBuilder(catalog).build(supportedStatements);
37      }
38      
39      private Map<String, Object> build(final Collection<?> supportedStatements) {
40          Map<String, Object> result = new LinkedHashMap<>(4, 1F);
41          List<Map<String, Object>> completionTargets = catalog.getShardingSphereDescriptors().getCompletionTargetDescriptors().stream().map(this::toCompletionTargetPayload).toList();
42          List<Map<String, Object>> resourceNavigation = catalog.getShardingSphereDescriptors().getResourceNavigationDescriptors().stream().map(this::toResourceNavigationPayload).toList();
43          result.put("response_mode", MCPResponseMode.CATALOG);
44          result.put("supportedStatementClasses", supportedStatements);
45          result.put("completionTargets", completionTargets);
46          result.put("resourceNavigation", resourceNavigation);
47          return result;
48      }
49      
50      private Map<String, Object> toCompletionTargetPayload(final MCPCompletionTargetDescriptor descriptor) {
51          Map<String, Object> result = new LinkedHashMap<>(5, 1F);
52          result.put("referenceType", descriptor.getReferenceType());
53          result.put("reference", descriptor.getReference());
54          result.put("arguments", descriptor.getArguments());
55          result.put("maxValues", descriptor.getMaxValues());
56          if (!descriptor.getMeta().isEmpty()) {
57              result.put("meta", descriptor.getMeta());
58          }
59          return result;
60      }
61      
62      private Map<String, Object> toResourceNavigationPayload(final MCPResourceNavigationDescriptor descriptor) {
63          Map<String, Object> result = new LinkedHashMap<>(7, 1F);
64          result.put("from", descriptor.getFrom());
65          result.put("from_type", resolveReferenceType(descriptor.getFrom()));
66          result.put("to", descriptor.getTo());
67          result.put("to_type", resolveReferenceType(descriptor.getTo()));
68          result.put("requiredArguments", descriptor.getRequiredArguments());
69          result.put("carriedArguments", descriptor.getCarriedArguments());
70          result.put("description", descriptor.getDescription());
71          return result;
72      }
73      
74      private String resolveReferenceType(final String reference) {
75          if (catalog.getProtocolDescriptors().getAllResourceDescriptors().stream().anyMatch(each -> each.getUriTemplate().equals(reference))) {
76              return reference.contains("{") ? "resource_template" : "resource";
77          }
78          if (catalog.getProtocolDescriptors().getToolDescriptors().stream().anyMatch(each -> each.getName().equals(reference))) {
79              return "tool";
80          }
81          if (catalog.getProtocolDescriptors().getPromptDescriptors().stream().anyMatch(each -> each.getName().equals(reference))) {
82              return "prompt";
83          }
84          return "unknown";
85      }
86      
87  }
