101     @Test
102     void assertExtractUnaryExpressionTableOwner() {
103         SimpleTableSegment table = new SimpleTableSegment(new TableNameSegment(30, 38, new IdentifierValue("foo_table")));
104         ColumnSegment column = new ColumnSegment(14, 15, new IdentifierValue("id"));
105         column.setOwner(new OwnerSegment(4, 12, new IdentifierValue("foo_table")));
106         ProjectionsSegment projections = new ProjectionsSegment(0, 15);
107         projections.getProjections().add(new ExpressionProjectionSegment(3, 15, "~foo_table.id", new UnaryOperationExpression(3, 15, column, "~", "~foo_table.id")));
108         tableExtractor.extractTablesFromSelect(SelectStatement.builder().projections(projections).from(table).build());
109         assertThat(tableExtractor.getRewriteTables().size(), is(2));
110         Iterator<SimpleTableSegment> actualTables = tableExtractor.getRewriteTables().iterator();
111         assertThat(actualTables.next(), is(table));
112         SimpleTableSegment actualOwner = actualTables.next();
113         assertThat(actualOwner.getStartIndex(), is(4));
114         assertThat(actualOwner.getStopIndex(), is(12));
115         assertThat(actualOwner.getTableName().getIdentifier().getValue(), is("foo_table"));
116     }
