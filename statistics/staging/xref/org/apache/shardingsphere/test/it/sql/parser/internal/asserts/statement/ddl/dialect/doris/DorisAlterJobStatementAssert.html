31  import static org.hamcrest.MatcherAssert.assertThat;
32  import static org.hamcrest.Matchers.is;
33  import static org.junit.jupiter.api.Assertions.assertNotNull;
34  import static org.junit.jupiter.api.Assertions.assertTrue;
35  
36  /**
37   * Alter job statement assert for Doris.
38   */
39  @NoArgsConstructor(access = AccessLevel.PRIVATE)
40  public final class DorisAlterJobStatementAssert {
41      
42      /**
43       * Assert alter job statement is correct with expected parser result.
44       *
45       * @param assertContext assert context
46       * @param actual actual alter job statement
47       * @param expected expected alter job statement test case
48       */
49      public static void assertIs(final SQLCaseAssertContext assertContext, final DorisAlterJobStatement actual, final DorisAlterJobStatementTestCase expected) {
50          assertJobName(assertContext, actual, expected);
51          assertProperties(assertContext, actual, expected);
52          assertInsertStatement(assertContext, actual, expected);
53      }
54      
55      private static void assertJobName(final SQLCaseAssertContext assertContext, final DorisAlterJobStatement actual, final DorisAlterJobStatementTestCase expected) {
56          assertNotNull(actual.getJobName(), assertContext.getText("Job name should exist."));
57          assertThat(assertContext.getText("Job name does not match: "), actual.getJobName().getIdentifier().getValue(), is(expected.getJobName().getName()));
58          SQLSegmentAssert.assertIs(assertContext, actual.getJobName(), expected.getJobName());
59      }
60      
61      private static void assertProperties(final SQLCaseAssertContext assertContext, final DorisAlterJobStatement actual, final DorisAlterJobStatementTestCase expected) {
62          if (!expected.getProperties().isEmpty()) {
63              assertTrue(actual.getProperties().isPresent(), assertContext.getText("Properties should exist."));
64              assertThat(assertContext.getText("Properties size does not match: "), actual.getProperties().get().getProperties().size(), is(expected.getProperties().size()));
65              for (int i = 0; i < expected.getProperties().size(); i++) {
66                  assertProperty(assertContext, actual.getProperties().get().getProperties().get(i), expected.getProperties().get(i));
67              }
68          }
69      }
70      
71      private static void assertProperty(final SQLCaseAssertContext assertContext, final PropertySegment actual, final PropertyTestCase expected) {
72          assertThat(assertContext.getText(String.format("Property key '%s' assertion error: ", expected.getKey())), actual.getKey(), is(expected.getKey()));
73          assertThat(assertContext.getText(String.format("Property value for key '%s' assertion error: ", expected.getKey())), actual.getValue(), is(expected.getValue()));
74          SQLSegmentAssert.assertIs(assertContext, actual, expected);
75      }
76      
77      private static void assertInsertStatement(final SQLCaseAssertContext assertContext, final DorisAlterJobStatement actual, final DorisAlterJobStatementTestCase expected) {
78          if (null == expected.getInsertTable() && null == expected.getInsertSelectTable()) {
79              return;
80          }
81          assertTrue(actual.getInsertStatement().isPresent(), assertContext.getText("Insert statement should exist."));
82          if (null != expected.getInsertTable()) {
83              assertTrue(actual.getInsertStatement().get().getTable().isPresent(), assertContext.getText("Insert target table should exist."));
84              TableAssert.assertIs(assertContext, actual.getInsertStatement().get().getTable().get(), expected.getInsertTable());
85          }
86          if (null != expected.getInsertSelectTable()) {
87              assertTrue(actual.getInsertStatement().get().getInsertSelect().isPresent(), assertContext.getText("Insert select should exist."));
88              SimpleTableSegment fromTable = (SimpleTableSegment) actual.getInsertStatement().get().getInsertSelect().get().getSelect().getFrom().orElse(null);
89              assertNotNull(fromTable, assertContext.getText("Insert select from table should exist."));
90              TableAssert.assertIs(assertContext, fromTable, expected.getInsertSelectTable());
91          }
92      }
93  }
