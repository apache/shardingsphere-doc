52  @PipelineE2ESettings(database = @PipelineE2EDatabaseSettings(type = "MySQL"))
53  @Slf4j
54  class MySQLMigrationGeneralE2EIT extends AbstractMigrationE2EIT {
55      
56      private static final String SOURCE_TABLE_NAME = "t_order_copy";
57      
58      private static final String TARGET_TABLE_NAME = "t_order";
59      
60      @ParameterizedTest(name = "{0}")
61      @EnabledIf("isEnabled")
62      @ArgumentsSource(PipelineE2ETestCaseArgumentsProvider.class)
63      void assertMigrationSuccess(final PipelineTestParameter testParam) throws SQLException, InterruptedException {
64          try (PipelineContainerComposer containerComposer = new PipelineContainerComposer(testParam)) {
65              PipelineE2EDistSQLFacade distSQLFacade = new PipelineE2EDistSQLFacade(containerComposer, new MigrationJobType());
66              distSQLFacade.alterPipelineRule();
67              IntPkLargeOrderDAO orderDAO = new IntPkLargeOrderDAO(containerComposer.getSourceDataSource(), containerComposer.getDatabaseType(), SOURCE_TABLE_NAME);
68              orderDAO.createTable();
69              IntPkOrderItemDAO orderItemDAO = new IntPkOrderItemDAO(containerComposer.getSourceDataSource(), containerComposer.getDatabaseType(), null);
70              orderItemDAO.createTable();
71              addMigrationSourceResource(containerComposer);
72              addMigrationTargetResource(containerComposer);
73              createTargetOrderTableRule(containerComposer);
74              createTargetOrderTableEncryptRule(containerComposer);
75              createTargetOrderItemTableRule(containerComposer);
76              log.info("init data begin: {}", LocalDateTime.now());
77              orderDAO.batchInsert(PipelineContainerComposer.TABLE_INIT_ROW_COUNT);
78              orderItemDAO.batchInsert(PipelineContainerComposer.TABLE_INIT_ROW_COUNT);
79              log.info("init data end: {}", LocalDateTime.now());
80              startMigration(containerComposer, SOURCE_TABLE_NAME, TARGET_TABLE_NAME);
81              startMigration(containerComposer, "t_order_item", "t_order_item");
82              String orderJobId = distSQLFacade.getJobIdByTableName("ds_0." + SOURCE_TABLE_NAME);
83              distSQLFacade.waitJobPreparingStageFinished(orderJobId);
84              containerComposer.startIncrementTask(
85                      new E2EIncrementalTask(containerComposer.getSourceDataSource(), SOURCE_TABLE_NAME, new SnowflakeKeyGenerateAlgorithm(), containerComposer.getDatabaseType(), 30));
86              TimeUnit.SECONDS.timedJoin(containerComposer.getIncreaseTaskThread(), 30L);
87              containerComposer.sourceExecuteWithLog(String.format("""
88                      INSERT INTO %s (order_id, user_id, status, t_json) VALUES
89                      (10000, 1, 'OK', '{"backspace":"a\\\\bb","form_feed":"a\\\\fb","newline":"a\\\\nb","carriage_return":"a\\\\rb","tab":"a\\\\tb","unicode":"a\\\\u0001b","tab\\\\tkey":"value"}')
90                      """, SOURCE_TABLE_NAME));
91              orderItemDAO.insert(10000L, 10000L, 1, "OK");
92              distSQLFacade.pauseJob(orderJobId);
93              distSQLFacade.resumeJob(orderJobId);
94              DataSource jdbcDataSource = containerComposer.generateShardingSphereDataSourceFromProxy();
95              containerComposer.assertRecordExists(jdbcDataSource, "t_order", 10000);
96              containerComposer.assertRecordExists(jdbcDataSource, "t_order_item", 10000);
97              assertMigrationSuccessById(distSQLFacade, orderJobId, "DATA_MATCH", ImmutableMap.of("chunk-size", "300", "streaming-range-type", "SMALL"));
98              String orderItemJobId = distSQLFacade.getJobIdByTableName("ds_0.t_order_item");
99              assertMigrationSuccessById(distSQLFacade, orderItemJobId, "DATA_MATCH", ImmutableMap.of("chunk-size", "300", "streaming-range-type", "LARGE"));
100             containerComposer.sleepSeconds(2);
101             assertMigrationSuccessById(distSQLFacade, orderItemJobId, "CRC32_MATCH", Collections.emptyMap());
102             for (String each : distSQLFacade.listJobIds()) {
103                 distSQLFacade.commit(each);
104             }
105             assertTrue(distSQLFacade.listJobIds().isEmpty());
106             containerComposer.assertGreaterThanOrderTableInitRows(jdbcDataSource, PipelineContainerComposer.TABLE_INIT_ROW_COUNT, "");
107         }
108     }
109     
110     private void assertMigrationSuccessById(final PipelineE2EDistSQLFacade distSQLFacade, final String jobId,
111                                             final String algorithmType, final Map<String, String> algorithmProps) throws SQLException {
112         List<Map<String, Object>> jobStatus = distSQLFacade.waitJobIncrementalStageFinished(jobId);
113         for (Map<String, Object> each : jobStatus) {
114             assertTrue(Integer.parseInt(each.get("processed_records_count").toString()) > 0);
115             assertThat(Integer.parseInt(each.get("inventory_finished_percentage").toString()), is(100));
116         }
117         distSQLFacade.startCheck(jobId, algorithmType, algorithmProps);
118         distSQLFacade.verifyCheck(jobId);
119     }
120     
121     private static boolean isEnabled(final ExtensionContext context) {
122         return PipelineE2ECondition.isEnabled(context);
123     }
