389         Collection<OrderByItemSegment> orderByItems = new LinkedList<>();
390         orderByItems.add(new ColumnOrderByItemSegment(createColumnSegment("foo_" + name + "_order_column"), OrderDirection.ASC, NullsOrderType.FIRST));
391         orderByItems.add(new ExpressionOrderByItemSegment(0, 0, "order_by_expr", OrderDirection.ASC, NullsOrderType.FIRST,
392                 createBinaryOperation("foo_" + name + "_order_expr_left", "bar_" + name + "_order_expr_right")));
393         return new OrderBySegment(0, 0, orderByItems);
394     }
