46  /**
47   * Select statement binder.
48   */
49  @RequiredArgsConstructor
50  public final class SelectStatementBinder implements SQLStatementBinder<SelectStatement> {
51      
52      private final Multimap<CaseInsensitiveString, TableSegmentBinderContext> outerTableBinderContexts;
53      
54      public SelectStatementBinder() {
55          outerTableBinderContexts = LinkedHashMultimap.create();
56      }
57      
58      @Override
59      public SelectStatement bind(final SelectStatement sqlStatement, final SQLStatementBinderContext binderContext) {
60          SelectStatement result = copy(sqlStatement);
61          Multimap<CaseInsensitiveString, TableSegmentBinderContext> tableBinderContexts = LinkedHashMultimap.create();
62          Optional<TableSegment> boundTableSegment = sqlStatement.getFrom().map(optional -> TableSegmentBinder.bind(optional, binderContext, tableBinderContexts, outerTableBinderContexts));
63          boundTableSegment.ifPresent(result::setFrom);
64          result.setProjections(ProjectionsSegmentBinder.bind(sqlStatement.getProjections(), binderContext, boundTableSegment.orElse(null), tableBinderContexts, outerTableBinderContexts));
65          sqlStatement.getWhere().ifPresent(optional -> result.setWhere(WhereSegmentBinder.bind(optional, binderContext, tableBinderContexts, outerTableBinderContexts)));
66          sqlStatement.getCombine().ifPresent(optional -> result.setCombine(CombineSegmentBinder.bind(optional, binderContext, outerTableBinderContexts)));
67          sqlStatement.getLock().ifPresent(optional -> result.setLock(LockSegmentBinder.bind(optional, binderContext, tableBinderContexts, outerTableBinderContexts)));
68          Multimap<CaseInsensitiveString, TableSegmentBinderContext> currentTableBinderContexts = createCurrentTableBinderContexts(binderContext, result);
69          sqlStatement.getGroupBy().ifPresent(optional -> result.setGroupBy(
70                  GroupBySegmentBinder.bind(optional, binderContext, currentTableBinderContexts, tableBinderContexts, outerTableBinderContexts)));
71          sqlStatement.getOrderBy().ifPresent(optional -> result.setOrderBy(
72                  OrderBySegmentBinder.bind(optional, binderContext, currentTableBinderContexts, tableBinderContexts, outerTableBinderContexts)));
73          sqlStatement.getHaving().ifPresent(optional -> result.setHaving(HavingSegmentBinder.bind(optional, binderContext, currentTableBinderContexts, outerTableBinderContexts)));
74          // TODO support other segment bind in select statement
75          return result;
76      }
77      
78      private Multimap<CaseInsensitiveString, TableSegmentBinderContext> createCurrentTableBinderContexts(final SQLStatementBinderContext binderContext, final SelectStatement selectStatement) {
79          Multimap<CaseInsensitiveString, TableSegmentBinderContext> result = LinkedHashMultimap.create();
80          Collection<ProjectionSegment> subqueryProjections = SubqueryTableBindUtils.createSubqueryProjections(
81                  selectStatement.getProjections().getProjections(), new IdentifierValue(""), binderContext.getSqlStatement().getDatabaseType());
82          result.put(new CaseInsensitiveString(""), new SimpleTableSegmentBinderContext(subqueryProjections));
83          return result;
84      }
85      
86      @SneakyThrows(ReflectiveOperationException.class)
87      private SelectStatement copy(final SelectStatement sqlStatement) {
88          SelectStatement result = sqlStatement.getClass().getDeclaredConstructor().newInstance();
89          sqlStatement.getLimit().ifPresent(result::setLimit);
90          sqlStatement.getWindow().ifPresent(result::setWindow);
91          sqlStatement.getModelSegment().ifPresent(result::setModelSegment);
92          sqlStatement.getSubqueryType().ifPresent(result::setSubqueryType);
93          sqlStatement.getWithSegment().ifPresent(result::setWithSegment);
94          result.addParameterMarkerSegments(sqlStatement.getParameterMarkerSegments());
95          result.getCommentSegments().addAll(sqlStatement.getCommentSegments());
96          result.getVariableNames().addAll(sqlStatement.getVariableNames());
97          return result;
98      }
99  }
