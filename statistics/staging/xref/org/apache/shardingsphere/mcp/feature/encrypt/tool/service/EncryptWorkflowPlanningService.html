48      private static final List<String> SUPPORTED_OPERATION_TYPES = List.of(WorkflowLifecycle.OPERATION_CREATE, WorkflowLifecycle.OPERATION_DROP);
49      
50      private static final List<String> INTERACTION_STEPS = List.of(
51              "Confirm database, table, column and target operation",
52              "Inspect existing encrypt rules and algorithm metadata",
53              "Clarify missing encrypt requirements and choose algorithms",
54              "Collect explicit rule column names and algorithm properties",
55              "Generate rule DistSQL artifacts",
56              "Review artifacts and choose execution mode",
57              "Execute or export rule DistSQL artifacts",
58              "Validate rule state and summarize");
59      
60      private static final List<String> VALIDATION_LAYERS = List.of("rules");

62      private final WorkflowPlanningSupport planningSupport = new WorkflowPlanningSupport();

72      /**
73       * Plan encrypt workflow.
74       *
75       * @param workflowSessionContext workflow session context
76       * @param metadataQueryFacade metadata query facade
77       * @param queryFacade query facade
78       * @param request workflow request
79       * @return workflow snapshot
80       */
81      public WorkflowContextSnapshot plan(final WorkflowSessionContext workflowSessionContext, final MCPMetadataQueryFacade metadataQueryFacade, final MCPFeatureQueryFacade queryFacade,
82                                          final EncryptWorkflowRequest request) {
83          WorkflowContextSnapshot result = workflowSessionContext.getOrCreate(request.getPlanId());
84          EncryptWorkflowRequest mergedRequest = prepareSnapshot(result, request);
85          ClarifiedIntent clarifiedIntent = result.getClarifiedIntent();
86          planningSupport.applyResolvedIntent(mergedRequest, clarifiedIntent);
87          if (!planningSupport.ensureSupportedOperationType(clarifiedIntent, SUPPORTED_OPERATION_TYPES, result)) {
88              return planningSupport.persistPlanningInterruption(workflowSessionContext, result);
89          }
90          if (!planningSupport.ensurePlanningContext(metadataQueryFacade, queryFacade, mergedRequest, clarifiedIntent, result)) {
91              return planningSupport.persistPlanningInterruption(workflowSessionContext, result);
92          }
93          queryFacade.checkDatabaseCapability(mergedRequest.getDatabase());
94          List<Map<String, Object>> existingRules = ruleInspectionService.queryEncryptRules(queryFacade, mergedRequest.getDatabase(), mergedRequest.getTable());
95          if (!ensureLifecycleState(clarifiedIntent, mergedRequest, existingRules, result, queryFacade)) {
96              return workflowSessionContext.persist(result, WorkflowLifecycle.STEP_FAILED, WorkflowLifecycle.STATUS_FAILED);
97          }
98          if (isDropWorkflow(clarifiedIntent)) {
99              if (!ensureSupportedRuleRewrite(queryFacade, mergedRequest, existingRules, result)) {
100                 return workflowSessionContext.persist(result, WorkflowLifecycle.STEP_CLARIFYING, WorkflowLifecycle.STATUS_CLARIFYING);
101             }
102             planDrop(mergedRequest, result);
103             return workflowSessionContext.persist(result, WorkflowLifecycle.STEP_REVIEW, WorkflowLifecycle.STATUS_PLANNED);
104         }
105         if (!planNonDrop(queryFacade, clarifiedIntent, mergedRequest, existingRules, result)) {
106             return workflowSessionContext.persist(result, WorkflowLifecycle.STEP_CLARIFYING, WorkflowLifecycle.STATUS_CLARIFYING);
107         }
108         return workflowSessionContext.persist(result, WorkflowLifecycle.STEP_REVIEW, WorkflowLifecycle.STATUS_PLANNED);
109     }
110     
111     private EncryptWorkflowRequest prepareSnapshot(final WorkflowContextSnapshot snapshot, final EncryptWorkflowRequest request) {
112         EncryptWorkflowRequest result = EncryptWorkflowRequest.merge(snapshot.getRequest(), request);
113         RuleWorkflowFeatureData workflowState = getWorkflowState(snapshot);
114         return planningSupport.prepareSnapshot(snapshot, EncryptFeatureDefinition.WORKFLOW_KIND, result, workflowState,
115                 resolveIntent(result), "Encrypt workflow plan.", INTERACTION_STEPS, VALIDATION_LAYERS);
116     }
117     
118     private ClarifiedIntent resolveIntent(final EncryptWorkflowRequest request) {
119         ClarifiedIntent result = planningSupport.createOperationIntent(request, WorkflowLifecycle.OPERATION_CREATE);
120         if (WorkflowLifecycle.OPERATION_CREATE.equalsIgnoreCase(result.getOperationType())) {
121             addMissingRequirement(result, WorkflowFieldNames.REQUIRES_DECRYPT, request.getOptions().getRequiresDecrypt(), "Do you need reversible decryption?");
122             addMissingRequirement(result, WorkflowFieldNames.REQUIRES_EQUALITY_FILTER, request.getOptions().getRequiresEqualityFilter(), "Do you need equality query?");
123             addMissingRequirement(result, WorkflowFieldNames.REQUIRES_LIKE_QUERY, request.getOptions().getRequiresLikeQuery(), "Do you need LIKE query?");
124         }
125         return result;
126     }
127     
128     private void addMissingRequirement(final ClarifiedIntent clarifiedIntent, final String fieldName, final Boolean value, final String message) {
129         if (null == value) {
130             clarifiedIntent.getUnresolvedFields().add(fieldName);
131             clarifiedIntent.getClarificationMessages().add(message);
132         }
133     }
134     
135     private RuleWorkflowFeatureData getWorkflowState(final WorkflowContextSnapshot snapshot) {
136         return null == snapshot.getFeatureData() ? new RuleWorkflowFeatureData() : snapshot.getFeatureData();
137     }
138     
139     private boolean ensureLifecycleState(final ClarifiedIntent clarifiedIntent, final EncryptWorkflowRequest request,
140                                          final List<Map<String, Object>> encryptRules, final WorkflowContextSnapshot snapshot, final MCPFeatureQueryFacade queryFacade) {
141         boolean ruleExists = encryptRules.stream().anyMatch(each -> queryFacade.isSameIdentifier(
142                 request.getDatabase(), IdentifierScope.COLUMN, request.getColumn(), WorkflowRuleValueUtils.getRuleValue(each, "logic_column")));
143         return planningSupport.ensureLifecycleState("Encrypt rule", clarifiedIntent, ruleExists, snapshot);
144     }
145     
146     private boolean isDropWorkflow(final ClarifiedIntent clarifiedIntent) {
147         return WorkflowLifecycle.OPERATION_DROP.equalsIgnoreCase(clarifiedIntent.getOperationType());
148     }
149     
150     private void planDrop(final EncryptWorkflowRequest request, final WorkflowContextSnapshot snapshot) {
151         addDropLifecycleWarnings(snapshot);
152         snapshot.getRuleArtifacts().addAll(ruleDistSQLPlanningService.planEncryptDropRule(request));
153         snapshot.setFeatureData(new RuleWorkflowFeatureData(List.of()));
154     }
155     
156     private boolean planNonDrop(final MCPFeatureQueryFacade queryFacade, final ClarifiedIntent clarifiedIntent, final EncryptWorkflowRequest request,
157                                 final List<Map<String, Object>> existingRules, final WorkflowContextSnapshot snapshot) {
158         planAlgorithms(queryFacade, request, snapshot);
159         if (!planningSupport.isReadyForArtifactPlanning(request, clarifiedIntent, snapshot, findPropertyRequirements(request),
160                 "Please use an encrypt algorithm that is visible in the current Proxy and satisfies the requirements.")) {
161             return false;
162         }
163         if (!ensureSupportedRuleRewrite(queryFacade, request, existingRules, snapshot)) {
164             return false;
165         }
166         if (!ensureRequiredRuleInputs(request, clarifiedIntent, snapshot)) {
167             return false;
168         }
169         planEncryptArtifacts(request, snapshot);
170         snapshot.setFeatureData(new RuleWorkflowFeatureData(List.of(createExpectedTargetRule(request))));
171         return true;
172     }
173     
174     private boolean ensureSupportedRuleRewrite(final MCPFeatureQueryFacade queryFacade, final EncryptWorkflowRequest request,
175                                                final List<Map<String, Object>> encryptRules, final WorkflowContextSnapshot snapshot) {
176         long remainingRuleCount = encryptRules.stream()
177                 .filter(each -> !queryFacade.isSameIdentifier(
178                         request.getDatabase(), IdentifierScope.COLUMN, request.getColumn(), WorkflowRuleValueUtils.getRuleValue(each, "logic_column")))
179                 .count();
180         if (0L == remainingRuleCount) {
181             return true;
182         }
183         snapshot.getClarifiedIntent().getClarificationMessages().add(
184                 "Current Proxy DistSQL cannot automatically rewrite an existing encrypt table rule with a partial column set. "
185                         + "Recreate the rule manually during a maintenance window.");
186         snapshot.getIssues().add(new WorkflowIssue(WorkflowIssueCode.ENCRYPT_RULE_REWRITE_LIMITED, "error", WorkflowLifecycle.STEP_PLANNING_ARTIFACTS,
187                 "Encrypt planning cannot automatically rewrite an existing table rule while preserving other columns.",
188                 "Manually recreate the encrypt rule with the complete column set after reviewing data impact.", true, Map.of("requires_table_rule_rewrite", true)));
189         return false;
190     }
191     
192     private void planEncryptArtifacts(final EncryptWorkflowRequest request, final WorkflowContextSnapshot snapshot) {
193         snapshot.getRuleArtifacts().addAll(ruleDistSQLPlanningService.planEncryptRule(request));
194     }
195     
196     private void addDropLifecycleWarnings(final WorkflowContextSnapshot snapshot) {
197         snapshot.getIssues().add(new WorkflowIssue(WorkflowIssueCode.ENCRYPT_DROP_SCOPE_LIMITED, "warning", WorkflowLifecycle.STEP_PLANNING_ARTIFACTS,
198                 "Encrypt drop only removes the rule. MCP will not restore historical plaintext data.", "Review business impact before execution.", true, Map.of()));
199     }
200     
201     private Map<String, Object> createExpectedTargetRule(final EncryptWorkflowRequest request) {
202         Map<String, Object> result = new LinkedHashMap<>(10, 1F);
203         result.put("logic_column", request.getColumn());
204         result.put("cipher_column", request.getOptions().getCipherColumnName());
205         result.put("assisted_query_column", Boolean.TRUE.equals(request.getOptions().getRequiresEqualityFilter()) ? request.getOptions().getAssistedQueryColumnName() : "");
206         result.put("like_query_column", Boolean.TRUE.equals(request.getOptions().getRequiresLikeQuery()) ? request.getOptions().getLikeQueryColumnName() : "");
207         result.put("encryptor_type", request.getAlgorithmType());
208         result.put("encryptor_props", request.getPrimaryAlgorithmProperties());
209         result.put("assisted_query_type", Boolean.TRUE.equals(request.getOptions().getRequiresEqualityFilter()) ? request.getOptions().getAssistedQueryAlgorithmType() : "");
210         result.put("assisted_query_props", Boolean.TRUE.equals(request.getOptions().getRequiresEqualityFilter()) ? request.getOptions().getAssistedQueryAlgorithmProperties() : Map.of());
211         result.put("like_query_type", Boolean.TRUE.equals(request.getOptions().getRequiresLikeQuery()) ? request.getOptions().getLikeQueryAlgorithmType() : "");
212         result.put("like_query_props", Boolean.TRUE.equals(request.getOptions().getRequiresLikeQuery()) ? request.getOptions().getLikeQueryAlgorithmProperties() : Map.of());
213         return result;
214     }
215     
216     private void planAlgorithms(final MCPFeatureQueryFacade queryFacade, final EncryptWorkflowRequest request, final WorkflowContextSnapshot snapshot) {
217         List<Map<String, Object>> encryptAlgorithms = ruleInspectionService.queryEncryptAlgorithms(queryFacade);
218         List<AlgorithmCandidate> algorithmCandidates = algorithmRecommendationService.recommendEncryptAlgorithms(request, encryptAlgorithms, snapshot.getIssues());
219         snapshot.getAlgorithmCandidates().addAll(algorithmCandidates);
220         applyRecommendedAlgorithms(request, algorithmCandidates);
221     }
222     
223     private List<AlgorithmPropertyRequirement> findPropertyRequirements(final EncryptWorkflowRequest request) {
224         return algorithmPropertyTemplateService.findRequirements(request.getAlgorithmType(), request.getOptions().getAssistedQueryAlgorithmType(), request.getOptions().getLikeQueryAlgorithmType());
225     }
226     
227     private void applyRecommendedAlgorithms(final EncryptWorkflowRequest request, final List<AlgorithmCandidate> algorithmCandidates) {
228         for (AlgorithmCandidate each : algorithmCandidates) {
229             if (EncryptFeatureDefinition.ALGORITHM_ROLE_PRIMARY.equals(each.getAlgorithmRole())) {
230                 request.setAlgorithmType(each.getAlgorithmType());
231                 continue;
232             }
233             if (EncryptFeatureDefinition.ALGORITHM_ROLE_ASSISTED_QUERY.equals(each.getAlgorithmRole())) {
234                 request.getOptions().setAssistedQueryAlgorithmType(each.getAlgorithmType());
235                 continue;
236             }
237             if (EncryptFeatureDefinition.ALGORITHM_ROLE_LIKE_QUERY.equals(each.getAlgorithmRole())) {
238                 request.getOptions().setLikeQueryAlgorithmType(each.getAlgorithmType());
239             }
240         }
241     }
242     
243     private boolean ensureRequiredRuleInputs(final EncryptWorkflowRequest request, final ClarifiedIntent clarifiedIntent, final WorkflowContextSnapshot snapshot) {
244         List<String> missingInputs = new LinkedList<>();
245         addMissingInput(missingInputs, WorkflowFieldNames.CIPHER_COLUMN_NAME, request.getOptions().getCipherColumnName());
246         if (Boolean.TRUE.equals(request.getOptions().getRequiresEqualityFilter())) {
247             addMissingInput(missingInputs, WorkflowFieldNames.ASSISTED_QUERY_COLUMN_NAME, request.getOptions().getAssistedQueryColumnName());
248             addMissingInput(missingInputs, WorkflowFieldNames.ASSISTED_QUERY_ALGORITHM_TYPE, request.getOptions().getAssistedQueryAlgorithmType());
249         }
250         if (Boolean.TRUE.equals(request.getOptions().getRequiresLikeQuery())) {
251             addMissingInput(missingInputs, WorkflowFieldNames.LIKE_QUERY_COLUMN_NAME, request.getOptions().getLikeQueryColumnName());
252             addMissingInput(missingInputs, WorkflowFieldNames.LIKE_QUERY_ALGORITHM_TYPE, request.getOptions().getLikeQueryAlgorithmType());
253         }
254         if (missingInputs.isEmpty()) {
255             return true;
256         }
257         for (String each : missingInputs) {
258             clarifiedIntent.getClarificationMessages().add(String.format("Please provide `%s` for encrypt rule DistSQL.", each));
259         }
260         snapshot.getIssues().add(new WorkflowIssue(WorkflowIssueCode.RULE_INPUT_REQUIRED, "error", WorkflowLifecycle.STEP_COLLECTING_RULE_INPUTS,
261                 "Encrypt rule DistSQL requires explicit rule column and query algorithm inputs.", "Provide the missing rule inputs and retry planning.", true,
262                 Map.of("missing_inputs", missingInputs)));
263         return false;
264     }
265     
266     private void addMissingInput(final List<String> missingInputs, final String fieldName, final String value) {
267         if (value.isEmpty()) {
268             missingInputs.add(fieldName);
269         }
270     }
