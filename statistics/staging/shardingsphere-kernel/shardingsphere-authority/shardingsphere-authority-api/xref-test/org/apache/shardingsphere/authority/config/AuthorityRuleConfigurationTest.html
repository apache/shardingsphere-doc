20  import org.apache.shardingsphere.infra.algorithm.core.config.AlgorithmConfiguration;
21  import org.apache.shardingsphere.infra.config.rule.validator.RuleConfigurationValidator;
22  import org.apache.shardingsphere.infra.exception.kernel.metadata.rule.InvalidRuleConfigurationException;
23  import org.junit.jupiter.params.ParameterizedTest;
24  import org.junit.jupiter.params.provider.Arguments;
25  import org.junit.jupiter.params.provider.MethodSource;
26  
27  import java.util.Collections;
28  import java.util.Properties;
29  import java.util.stream.Stream;
30  
31  import static org.hamcrest.MatcherAssert.assertThat;
32  import static org.hamcrest.Matchers.is;
33  import static org.junit.jupiter.api.Assertions.assertDoesNotThrow;
34  import static org.junit.jupiter.api.Assertions.assertThrows;
35  
36  class AuthorityRuleConfigurationTest {
37      
38      @ParameterizedTest(name = "{0}")
39      @MethodSource("validRuleConfigurationArguments")
40      void assertValidateValidRuleConfiguration(final String name, final AuthorityRuleConfiguration ruleConfig) {
41          assertDoesNotThrow(() -> RuleConfigurationValidator.validate(ruleConfig));
42      }
43      
44      private static Stream<Arguments> validRuleConfigurationArguments() {
45          AlgorithmConfiguration privilegeProvider = createAlgorithmConfiguration("FIXTURE");
46          UserConfiguration userConfig = new UserConfiguration("foo_user", null, "%", null, false);
47          return Stream.of(
48                  Arguments.of("Empty users and authenticators", new AuthorityRuleConfiguration(
49                          Collections.emptyList(), privilegeProvider, Collections.emptyMap(), null)),
50                  Arguments.of("User with nullable optional values", new AuthorityRuleConfiguration(
51                          Collections.singleton(userConfig), privilegeProvider, Collections.emptyMap(), null)),
52                  Arguments.of("User with null hostname", createRuleConfiguration(new UserConfiguration("foo_user", null, null, null, false))),
53                  Arguments.of("User with empty hostname", createRuleConfiguration(new UserConfiguration("foo_user", null, "", null, false))),
54                  Arguments.of("Whitespace authenticator type remains nonempty", new AuthorityRuleConfiguration(Collections.emptyList(), privilegeProvider,
55                          Collections.singletonMap("foo_authenticator", createAlgorithmConfiguration(" ")), null)),
56                  Arguments.of("Complete configuration", new AuthorityRuleConfiguration(Collections.singleton(
57                          new UserConfiguration("foo_admin", "foo_password", "localhost", "foo_authenticator", true)), privilegeProvider,
58                          Collections.singletonMap("foo_authenticator", createAlgorithmConfiguration("MD5")), "foo_authenticator")));
59      }
60      
61      private static AlgorithmConfiguration createAlgorithmConfiguration(final String type) {
62          return new AlgorithmConfiguration(type, new Properties());
63      }
64      
65      @ParameterizedTest(name = "{0}")
66      @MethodSource("algorithmTypeArguments")
67      void assertValidateAlgorithmType(final String name, final AuthorityRuleConfiguration ruleConfig, final String expectedMessage) {
68          InvalidRuleConfigurationException actual = assertThrows(InvalidRuleConfigurationException.class, () -> RuleConfigurationValidator.validate(ruleConfig));
69          assertThat(actual.getMessage(), is("Invalid 'AuthorityRuleConfiguration' rule, error message is: " + expectedMessage));
70      }
71      
72      private static Stream<Arguments> algorithmTypeArguments() {
73          AlgorithmConfiguration privilegeProvider = createAlgorithmConfiguration("FIXTURE");
74          return Stream.of(
75                  Arguments.of("Null privilege provider type before SPI", new AuthorityRuleConfiguration(Collections.emptyList(), createAlgorithmConfiguration(null), Collections.emptyMap(), null),
76                          "Property `privilegeProvider.type` Type is required."),
77                  Arguments.of("Empty privilege provider type before SPI", new AuthorityRuleConfiguration(Collections.emptyList(), createAlgorithmConfiguration(""), Collections.emptyMap(), null),
78                          "Property `privilegeProvider.type` Type is required."),
79                  Arguments.of("Whitespace privilege provider type reaches SPI", new AuthorityRuleConfiguration(Collections.emptyList(), createAlgorithmConfiguration(" "), Collections.emptyMap(), null),
80                          "Property `privilegeProvider` does not match an available SPI implementation."),
81                  Arguments.of("Null authenticator type", new AuthorityRuleConfiguration(Collections.emptyList(), privilegeProvider,
82                          Collections.singletonMap("foo_authenticator", createAlgorithmConfiguration(null)), null), "Property `authenticators[foo_authenticator].type` Type is required."),
83                  Arguments.of("Empty authenticator type", new AuthorityRuleConfiguration(Collections.emptyList(), privilegeProvider,
84                          Collections.singletonMap("foo_authenticator", createAlgorithmConfiguration("")), null), "Property `authenticators[foo_authenticator].type` Type is required."));
85      }
86      
87      @ParameterizedTest(name = "{0}")
88      @MethodSource("invalidRuleConfigurationArguments")
89      void assertValidateInvalidRuleConfiguration(final String name, final AuthorityRuleConfiguration ruleConfig) {
90          assertThrows(InvalidRuleConfigurationException.class, () -> RuleConfigurationValidator.validate(ruleConfig));
91      }
92      
93      private static Stream<Arguments> invalidRuleConfigurationArguments() {
94          AlgorithmConfiguration privilegeProvider = createAlgorithmConfiguration("FIXTURE");
95          AlgorithmConfiguration authenticator = createAlgorithmConfiguration("MD5");
96          return Stream.of(
97                  Arguments.of("Null users", new AuthorityRuleConfiguration(null, privilegeProvider, Collections.emptyMap(), null)),
98                  Arguments.of("Null user", new AuthorityRuleConfiguration(Collections.singleton(null), privilegeProvider, Collections.emptyMap(), null)),
99                  Arguments.of("Null username", createRuleConfiguration(new UserConfiguration(null, null, "%", null, false))),
100                 Arguments.of("Empty username", createRuleConfiguration(new UserConfiguration("", null, "%", null, false))),
101                 Arguments.of("Blank username", createRuleConfiguration(new UserConfiguration(" ", null, "%", null, false))),
102                 Arguments.of("Null privilege provider", new AuthorityRuleConfiguration(Collections.emptyList(), null, Collections.emptyMap(), null)),
103                 Arguments.of("Missing privilege provider", new AuthorityRuleConfiguration(
104                         Collections.emptyList(), createAlgorithmConfiguration("MISSING"), Collections.emptyMap(), null)),
105                 Arguments.of("Null authenticators", new AuthorityRuleConfiguration(Collections.emptyList(), privilegeProvider, null, null)),
106                 Arguments.of("Null authenticator name", new AuthorityRuleConfiguration(
107                         Collections.emptyList(), privilegeProvider, Collections.singletonMap(null, authenticator), null)),
108                 Arguments.of("Empty authenticator name", new AuthorityRuleConfiguration(
109                         Collections.emptyList(), privilegeProvider, Collections.singletonMap("", authenticator), null)),
110                 Arguments.of("Blank authenticator name", new AuthorityRuleConfiguration(
111                         Collections.emptyList(), privilegeProvider, Collections.singletonMap(" ", authenticator), null)),
112                 Arguments.of("Null authenticator", new AuthorityRuleConfiguration(
113                         Collections.emptyList(), privilegeProvider, Collections.singletonMap("foo_authenticator", null), null)));
114     }
115     
116     private static AuthorityRuleConfiguration createRuleConfiguration(final UserConfiguration userConfig) {
117         return new AuthorityRuleConfiguration(Collections.singleton(userConfig), createAlgorithmConfiguration("FIXTURE"), Collections.emptyMap(), null);
118     }
