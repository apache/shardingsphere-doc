111         when(metaDataContextManager.getMetaDataPersistFacade()).thenReturn(metaDataPersistFacade);

121     private MetaDataContexts createMetaDataContexts(final Collection<ShardingSphereDatabase> databases) {
122         return new MetaDataContexts(new ShardingSphereMetaData(databases, new ResourceMetaData(Collections.emptyMap(), Collections.emptyMap()),
123                 new RuleMetaData(Collections.emptyList()), props, protocolType), new ShardingSphereStatistics());
124     }

126     @Test
127     void assertImportDatabaseConfigurationNormally() {
128         ReadwriteSplittingRuleConfiguration ruleConfig = createRuleConfiguration();
129         doAnswer(invocation -> {
130             metaDataContexts.update(createRegisteredMetaDataContexts(dataSourceProps));
131             return null;
132         }).when(dataSourceUnitService).persist("foo_db", dataSourceProps);
133         new YamlDatabaseConfigurationImportExecutor(contextManager).importDatabaseConfiguration("foo_db", dataSourceProps, Collections.singleton(ruleConfig));
134         ShardingSphereDatabase actual = metaDataContexts.getMetaData().getDatabase("foo_db");
135         assertThat(actual.getResourceMetaData().getStorageUnits().keySet(), containsInAnyOrder("foo_ds_0", "foo_ds_1"));
136         assertThat(actual.getRuleMetaData().getSingleRule(ReadwriteSplittingRule.class).getConfiguration(), is(ruleConfig));
137         verify(metaDataPersistFacade.getDatabaseRuleService()).persist(eq("foo_db"), argThat(configs -> 1 == configs.size() && configs.contains(ruleConfig)));
138     }

140     private ReadwriteSplittingRuleConfiguration createRuleConfiguration() {
141         return new ReadwriteSplittingRuleConfiguration(Collections.singleton(
142                 new ReadwriteSplittingDataSourceGroupRuleConfiguration("foo_group", "foo_ds_0", Collections.singletonList("foo_ds_1"), "")), Collections.emptyMap());
143     }

145     private MetaDataContexts createRegisteredMetaDataContexts(final Map<String, DataSourcePoolProperties> registeredProps) {
146         Map<String, DataSource> dataSources = new LinkedHashMap<>(registeredProps.size(), 1F);
147         for (Entry<String, DataSourcePoolProperties> entry : registeredProps.entrySet()) {
148             MockedDataSource dataSource = new MockedDataSource();
149             dataSource.setUrl(entry.getValue().getConnectionPropertySynonyms().getStandardProperties().get("url").toString());
150             dataSources.put(entry.getKey(), dataSource);
151         }
152         ShardingSphereDatabase database = new ShardingSphereDatabase("foo_db", protocolType, new ResourceMetaData(dataSources),
153                 new RuleMetaData(Collections.emptyList()), Collections.emptyList(), props);
154         return createMetaDataContexts(Collections.singleton(database));
155     }
156     
157     @Test
158     void assertImportDatabaseConfigurationWithDelayedStorageUnitNotification() {
159         ReadwriteSplittingRuleConfiguration ruleConfig = createRuleConfiguration();
160         MetaDataContexts pendingMetaDataContexts = createRegisteredMetaDataContexts(Collections.singletonMap("foo_ds_0", dataSourceProps.get("foo_ds_0")));
161         MetaDataContexts completeMetaDataContexts = createRegisteredMetaDataContexts(dataSourceProps);
162         AtomicBoolean registrationStarted = new AtomicBoolean();
163         AtomicBoolean notificationsDelivered = new AtomicBoolean();
164         doAnswer(invocation -> {
165             registrationStarted.set(true);
166             return null;
167         }).when(dataSourceUnitService).persist("foo_db", dataSourceProps);
168         doAnswer(invocation -> {
169             if (registrationStarted.get() && !notificationsDelivered.get()
170                     && metaDataContexts.getMetaData().getDatabase("foo_db").getResourceMetaData().getStorageUnits().keySet().containsAll(dataSourceProps.keySet())) {
171                 metaDataContexts.update(pendingMetaDataContexts);
172                 notificationsDelivered.set(true);
173             }
174             return metaDataContexts;
175         }).when(contextManager).getMetaDataContexts();
176         try (MockedConstruction<RetryExecutor> ignored = mockConstruction(RetryExecutor.class, (retry, context) -> when(retry.execute(any(), any())).thenAnswer(invocation -> {
177             Predicate<Object> predicate = invocation.getArgument(0);
178             Object arg = invocation.getArgument(1);
179             if (predicate.test(arg)) {
180                 return true;
181             }
182             metaDataContexts.update(pendingMetaDataContexts);
183             metaDataContexts.update(completeMetaDataContexts);
184             notificationsDelivered.set(true);
185             return predicate.test(arg);
186         }))) {
187             new YamlDatabaseConfigurationImportExecutor(contextManager).importDatabaseConfiguration("foo_db", dataSourceProps, Collections.singleton(ruleConfig));
188         }
189         ShardingSphereDatabase actual = metaDataContexts.getMetaData().getDatabase("foo_db");
190         assertTrue(notificationsDelivered.get());
191         assertThat(actual.getResourceMetaData().getStorageUnits().keySet(), containsInAnyOrder("foo_ds_0", "foo_ds_1"));
192         assertThat(actual.getRuleMetaData().getSingleRule(ReadwriteSplittingRule.class).getConfiguration(), is(ruleConfig));
193         verify(metaDataPersistFacade.getDatabaseRuleService()).persist(eq("foo_db"), argThat(configs -> 1 == configs.size() && configs.contains(ruleConfig)));
194     }

196     private Map<String, DataSourcePoolProperties> createDataSourceProperties() {
197         Map<String, DataSourcePoolProperties> result = new LinkedHashMap<>(2, 1F);
198         for (String each : Arrays.asList("foo_ds_0", "foo_ds_1")) {
199             Map<String, Object> poolProps = new LinkedHashMap<>(3, 1F);
200             poolProps.put("url", "jdbc:mock://127.0.0.1/" + each);
201             poolProps.put("username", "root");
202             poolProps.put("password", "root");
203             result.put(each, new DataSourcePoolProperties(MockedDataSource.class.getName(), poolProps));
204         }
205         return result;
206     }
