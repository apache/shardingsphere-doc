76  /**
77   * Data consistency checker for migration job.
78   */
79  @Slf4j
80  public final class MigrationDataConsistencyChecker implements PipelineDataConsistencyChecker {
81      
82      private final MigrationJobConfiguration jobConfig;
83      
84      private final JobRateLimitAlgorithm readRateLimitAlgorithm;
85      
86      private final ConsistencyCheckJobItemProgressContext progressContext;
87      
88      private final AtomicReference<TableInventoryChecker> currentTableInventoryChecker = new AtomicReference<>();
89      
90      private final AtomicBoolean canceling = new AtomicBoolean(false);
91      
92      public MigrationDataConsistencyChecker(final MigrationJobConfiguration jobConfig, final TransmissionProcessContext processContext, final ConsistencyCheckJobItemProgressContext progressContext) {
93          this.jobConfig = jobConfig;
94          readRateLimitAlgorithm = processContext.getReadRateLimitAlgorithm();
95          this.progressContext = progressContext;
96      }
97      
98      @Override
99      public Map<String, TableDataConsistencyCheckResult> check(final String algorithmType, final Properties algorithmProps) {
100         List<String> sourceTableNames = new LinkedList<>();
101         jobConfig.getJobShardingDataNodes().forEach(each -> each.getEntries().forEach(entry -> entry.getDataNodes()
102                 .forEach(dataNode -> sourceTableNames.add(new QualifiedTable(dataNode.getSchemaName(), dataNode.getTableName()).format()))));
103         progressContext.setRecordsCount(getRecordsCount());
104         progressContext.getTableNames().addAll(sourceTableNames);
105         progressContext.setStatus(JobStatus.PREPARING);
106         progressContext.onProgressUpdated(new PipelineJobUpdateProgress(0));
107         Map<QualifiedTable, TableDataConsistencyCheckResult> checkResultMap = new LinkedHashMap<>();
108         try (
109                 PipelineDataSourceManager dataSourceManager = new PipelineDataSourceManager();
110                 TableDataConsistencyChecker tableChecker = TableDataConsistencyCheckerFactory.newInstance(algorithmType, algorithmProps)) {
111             Map<String, StorageUnit> storageUnits = PipelineContextManager.getProxyContext().getStorageUnits(jobConfig.getTargetDatabaseName());
112             transformDataSourceConfigurations(storageUnits);
113             if (progressContext.getTableCheckRangePositions().isEmpty()) {
114                 progressContext.getTableCheckRangePositions().addAll(splitCrossTables());
115             }
116             progressContext.setStatus(JobStatus.EXECUTE_INVENTORY_TASK);
117             progressContext.onProgressUpdated(new PipelineJobUpdateProgress(0));
118             for (TableCheckRangePosition each : progressContext.getTableCheckRangePositions()) {
119                 TableDataConsistencyCheckResult checkResult = checkSingleTableInventoryData(each, tableChecker, dataSourceManager);
120                 log.info("checkResult: {}, table: {}, checkRangePosition: {}", checkResult, each.getSourceDataNode(), each);
121                 DataNode dataNode = new DataNode(each.getSourceDataNode());
122                 QualifiedTable sourceTable = new QualifiedTable(dataNode.getSchemaName(), dataNode.getTableName());
123                 checkResultMap.put(sourceTable, checkResult);
124                 if (checkResult.isIgnored()) {
125                     progressContext.getIgnoredTableNames().add(sourceTable.format());
126                     log.info("Table '{}' is ignored, ignore type: {}", each.getSourceDataNode(), checkResult.getIgnoredType());
127                     continue;
128                 }
129                 if (!checkResult.isMatched()) {
130                     log.info("Unmatched on table '{}', ignore left tables", each.getSourceDataNode());
131                     cancel();
132                     return checkResultMap.entrySet().stream().collect(Collectors.toMap(entry -> entry.getKey().toString(), Entry::getValue));
133                 }
134             }
135         }
136         log.info("check done, jobId={}", jobConfig.getJobId());
137         return checkResultMap.entrySet().stream().collect(Collectors.toMap(entry -> entry.getKey().format(), Entry::getValue));
138     }
139     
140     private void transformDataSourceConfigurations(final Map<String, StorageUnit> targetStorageUnits) {
141         Map<String, DataSourcePoolProperties> sourceDataSourcePoolProps =
142                 new PipelineDataSourcePersistService().load(PipelineJobIdUtils.parseContextKey(jobConfig.getJobId()), new MigrationJobType().getType());
143         for (Entry<String, PipelineDataSourceConfiguration> entry : jobConfig.getSources().entrySet()) {
144             entry.setValue(sourceDataSourcePoolProps.containsKey(entry.getKey())
145                     ? PipelineDataSourceConfigurationUtils.transformPipelineDataSourceConfiguration(jobConfig.getJobId(), entry.getKey(), entry.getValue(),
146                             sourceDataSourcePoolProps.get(entry.getKey()))
147                     : entry.getValue());
148         }
149         PipelineDataSourceConfigurationUtils.transformPipelineDataSourceConfiguration(jobConfig.getJobId(), (ShardingSpherePipelineDataSourceConfiguration) jobConfig.getTarget(), targetStorageUnits);
150     }
151     
152     private long getRecordsCount() {
153         Map<Integer, TransmissionJobItemProgress> jobProgress = new TransmissionJobManager(new MigrationJobType()).getJobProgress(jobConfig);
154         return jobProgress.values().stream().filter(Objects::nonNull).mapToLong(TransmissionJobItemProgress::getInventoryRecordsCount).sum();
155     }
156     
157     private List<TableCheckRangePosition> splitCrossTables() {
158         List<TableCheckRangePosition> result = new LinkedList<>();
159         int splittingItem = 0;
160         for (JobDataNodeLine each : jobConfig.getJobShardingDataNodes()) {
161             for (JobDataNodeEntry entry : each.getEntries()) {
162                 for (DataNode dataNode : entry.getDataNodes()) {
163                     result.add(new TableCheckRangePosition(splittingItem++, dataNode.format(), entry.getLogicTableName(),
164                             UniqueKeyIngestPosition.ofUnsplit(), UniqueKeyIngestPosition.ofUnsplit(), null));
165                 }
166             }
167         }
168         return result;
169     }
170     
171     private TableDataConsistencyCheckResult checkSingleTableInventoryData(final TableCheckRangePosition checkRangePosition,
172                                                                           final TableDataConsistencyChecker tableChecker, final PipelineDataSourceManager dataSourceManager) {
173         log.info("checkSingleTableInventoryData, jobId: {}, checkRangePosition: {}", jobConfig.getJobId(), checkRangePosition);
174         DataNode dataNode = new DataNode(checkRangePosition.getSourceDataNode());
175         QualifiedTable sourceTable = new QualifiedTable(dataNode.getSchemaName(), dataNode.getTableName());
176         PipelineDataSource sourceDataSource = dataSourceManager.getDataSource(jobConfig.getSources().get(dataNode.getDataSourceName()));
177         PipelineTableMetaDataLoader metaDataLoader = new StandardPipelineTableMetaDataLoader(sourceDataSource);
178         PipelineTableMetaData tableMetaData = metaDataLoader.getTableMetaData(dataNode.getSchemaName(), dataNode.getTableName());
179         ShardingSpherePreconditions.checkNotNull(tableMetaData,
180                 () -> new PipelineTableDataConsistencyCheckLoadingFailedException(new QualifiedTable(dataNode.getSchemaName(), dataNode.getTableName())));
181         String targetTableName = checkRangePosition.getLogicTableName();
182         List<String> columnNames = tableMetaData.getColumnNames();
183         List<PipelineColumnMetaData> uniqueKeys = PipelineTableMetaDataUtils.getUniqueKeyColumns(sourceTable.getSchemaName(), sourceTable.getTableName(), metaDataLoader);
184         PipelineDataSource targetDataSource = dataSourceManager.getDataSource(jobConfig.getTarget());
185         IdentifierCasePolicy targetSchemaIdentifierPolicy = IdentifierNormalizeEngine.resolvePolicy(jobConfig.getTargetDatabaseType(), targetDataSource, IdentifierScope.SCHEMA);
186         IdentifierCasePolicy targetTableIdentifierPolicy = IdentifierNormalizeEngine.resolvePolicy(jobConfig.getTargetDatabaseType(), targetDataSource, IdentifierScope.TABLE);
187         IdentifierCasePolicy targetColumnIdentifierPolicy = IdentifierNormalizeEngine.resolvePolicy(jobConfig.getTargetDatabaseType(), targetDataSource, IdentifierScope.COLUMN);
188         String targetSchemaName = null != dataNode.getSchemaName() && new DatabaseTypeRegistry(jobConfig.getTargetDatabaseType()).getDialectDatabaseMetaData().getSchemaOption().isSchemaAvailable()
189                 ? IdentifierNormalizeEngine.normalize(targetSchemaIdentifierPolicy, dataNode.getSchemaName())
190                 : null;
191         QualifiedTable targetTable = new QualifiedTable(targetSchemaName, IdentifierNormalizeEngine.normalize(targetTableIdentifierPolicy, targetTableName));
192         List<String> targetColumnNames = columnNames.stream().map(each -> IdentifierNormalizeEngine.normalize(targetColumnIdentifierPolicy, each)).collect(Collectors.toList());
193         List<PipelineColumnMetaData> targetUniqueKeys = uniqueKeys.stream().map(each -> getTargetColumnMetaData(targetColumnIdentifierPolicy, each)).collect(Collectors.toList());
194         TableInventoryCheckParameter param = new TableInventoryCheckParameter(
195                 jobConfig.getJobId(), checkRangePosition.getSplittingItem(), sourceDataSource, targetDataSource, sourceTable, targetTable,
196                 columnNames, uniqueKeys, targetColumnNames, targetUniqueKeys, readRateLimitAlgorithm, progressContext, checkRangePosition.getQueryCondition());
197         TableInventoryChecker tableInventoryChecker = tableChecker.buildTableInventoryChecker(param);
198         currentTableInventoryChecker.set(tableInventoryChecker);
199         Optional<TableDataConsistencyCheckResult> preCheckResult = tableInventoryChecker.preCheck();
200         TableDataConsistencyCheckResult result = preCheckResult.orElseGet(tableInventoryChecker::checkSingleTableInventoryData);
201         tableInventoryChecker.cancel();
202         currentTableInventoryChecker.set(null);
203         return result;
204     }
205     
206     private PipelineColumnMetaData getTargetColumnMetaData(final IdentifierCasePolicy targetColumnIdentifierPolicy, final PipelineColumnMetaData sourceColumnMetaData) {
207         return new PipelineColumnMetaData(sourceColumnMetaData.getOrdinalPosition(), IdentifierNormalizeEngine.normalize(targetColumnIdentifierPolicy, sourceColumnMetaData.getName()),
208                 sourceColumnMetaData.getDataType(), sourceColumnMetaData.getDataTypeName(), sourceColumnMetaData.isNullable(),
209                 sourceColumnMetaData.isPrimaryKey(), sourceColumnMetaData.isUniqueKey());
210     }
211     
212     @Override
213     public void cancel() {
214         canceling.set(true);
215         Optional.ofNullable(currentTableInventoryChecker.get()).ifPresent(TableInventoryChecker::cancel);
216         currentTableInventoryChecker.set(null);
217     }
218     
219     @Override
220     public boolean isCanceling() {
221         return canceling.get();
222     }
