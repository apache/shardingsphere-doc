63  /**
64   * Command executor factory for Firebird.
65   */
66  @NoArgsConstructor(access = AccessLevel.PRIVATE)
67  public final class FirebirdCommandExecutorFactory {
68      
69      /**
70       * Create new instance of command executor.
71       *
72       * @param commandPacketType command packet type for Firebird
73       * @param commandPacket command packet for Firebird
74       * @param connectionSession connection session
75       * @return created instance
76       * @throws SQLException SQL exception
77       */
78      public static CommandExecutor newInstance(final FirebirdCommandPacketType commandPacketType, final CommandPacket commandPacket,
79                                                final ConnectionSession connectionSession) throws SQLException {
80          switch (commandPacketType) {
81              case INFO_DATABASE:
82                  return new FirebirdDatabaseInfoExecutor((FirebirdInfoPacket) commandPacket, connectionSession);
83              case TRANSACTION:
84                  return new FirebirdStartTransactionCommandExecutor((FirebirdStartTransactionPacket) commandPacket, connectionSession);
85              case CREATE_BLOB:
86              case CREATE_BLOB2:
87                  return new FirebirdCreateBlobCommandExecutor((FirebirdCreateBlobCommandPacket) commandPacket, connectionSession);
88              case OPEN_BLOB:
89              case OPEN_BLOB2:
90                  return new FirebirdOpenBlobCommandExecutor((FirebirdOpenBlobCommandPacket) commandPacket, connectionSession);
91              case GET_SEGMENT:
92                  return new FirebirdGetBlobSegmentCommandExecutor((FirebirdGetBlobSegmentCommandPacket) commandPacket, connectionSession);
93              case PUT_SEGMENT:
94                  return new FirebirdPutBlobSegmentCommandExecutor((FirebirdPutBlobSegmentCommandPacket) commandPacket, connectionSession);
95              case CANCEL_BLOB:
96                  return new FirebirdCancelBlobCommandExecutor((FirebirdCancelBlobCommandPacket) commandPacket, connectionSession);
97              case CLOSE_BLOB:
98                  return new FirebirdCloseBlobCommandExecutor((FirebirdCloseBlobCommandPacket) commandPacket, connectionSession);
99              case SEEK_BLOB:
100                 return new FirebirdSeekBlobCommandExecutor((FirebirdSeekBlobCommandPacket) commandPacket, connectionSession);
101             case ALLOCATE_STATEMENT:
102                 return new FirebirdAllocateStatementCommandExecutor((FirebirdAllocateStatementPacket) commandPacket, connectionSession);
103             case PREPARE_STATEMENT:
104                 return new FirebirdPrepareStatementCommandExecutor((FirebirdPrepareStatementPacket) commandPacket, connectionSession);
105             case EXECUTE:
106             case EXECUTE2:
107                 return new FirebirdExecuteStatementCommandExecutor((FirebirdExecuteStatementPacket) commandPacket, connectionSession);
108             case FETCH:
109                 return new FirebirdFetchStatementCommandExecutor((FirebirdFetchStatementPacket) commandPacket, connectionSession);
110             case INFO_SQL:
111                 return new FirebirdSQLInfoExecutor((FirebirdInfoPacket) commandPacket, connectionSession);
112             case COMMIT:
113                 return new FirebirdCommitTransactionCommandExecutor((FirebirdCommitTransactionPacket) commandPacket, connectionSession);
114             case ROLLBACK:
115                 return new FirebirdRollbackTransactionCommandExecutor((FirebirdRollbackTransactionPacket) commandPacket, connectionSession);
116             case FREE_STATEMENT:
117                 return new FirebirdFreeStatementCommandExecutor((FirebirdFreeStatementPacket) commandPacket, connectionSession);
118             default:
119                 return new FirebirdUnsupportedCommandExecutor();
120         }
121     }
122 }
