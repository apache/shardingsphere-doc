117     @SuppressWarnings("unchecked")
118     @Test
119     void assertInitTasksSuccess() throws SQLException {
120         when(InventoryChannelCreator.create(any(AlgorithmConfiguration.class), anyInt(), any())).thenReturn(mock(PipelineChannel.class));
121         when(IncrementalChannelCreator.create(any(AlgorithmConfiguration.class), any())).thenReturn(mock(PipelineChannel.class));
122         when(IncrementalDumperCreator.create(any(CreateIncrementalDumperParameter.class))).thenReturn(mock(IncrementalDumper.class));
123         DialectIncrementalPositionManager dialectPositionManager = mock(DialectIncrementalPositionManager.class);
124         IngestPosition ingestPosition = new IngestPlaceholderPosition();
125         when(dialectPositionManager.init(any(DataSource.class), any())).thenReturn(ingestPosition);
126         when(DatabaseTypedSPILoader.getService(any(), any())).thenReturn(dialectPositionManager);
127         PipelineProcessConfiguration processConfig = new PipelineProcessConfiguration(
128                 new PipelineReadConfiguration(1, 1, 1, null), new PipelineWriteConfiguration(1, 1, null), new AlgorithmConfiguration("MEMORY", new Properties()));
129         TransmissionProcessContext processContext = mock(TransmissionProcessContext.class, RETURNS_DEEP_STUBS);
130         when(processContext.getProcessConfiguration()).thenReturn(processConfig);
131         when(jobItemManager.getProgress(any(), anyInt())).thenReturn(Optional.empty(), Optional.empty(), Optional.of(mock(TransmissionJobItemProgress.class)));
132         IncrementalTaskProgress incrementalTaskProgress = new IncrementalTaskProgress(new IngestPlaceholderPosition());
133         TransmissionJobItemProgress initProgress = mock(TransmissionJobItemProgress.class);
134         when(initProgress.getIncremental()).thenReturn(new JobItemIncrementalTasksProgress(incrementalTaskProgress));
135         IngestPosition incrementalPositionFromProgress = new IngestPlaceholderPosition();
136         incrementalTaskProgress.setPosition(incrementalPositionFromProgress);
137         PipelineDataSourceConfiguration dataSourceConfig = mock(PipelineDataSourceConfiguration.class);
138         CDCJobItemContext incrementalOnlyContext = createJobItemContext("job_incremental_only", false, true, initProgress, processContext, dataSourceConfig);
139         CDCJobItemContext stoppingContext = createJobItemContext("job_stopping", true, false, null, processContext, dataSourceConfig);
140         stoppingContext.setStopping(true);
141         CDCJobItemContext fullContext = createJobItemContext("job_full", true, false, null, processContext, dataSourceConfig);
142         InventoryDumperContext uniqueKeyContext = createInventoryDumperContext(
143                 fullContext.getTaskConfig().getDumperContext().getCommonContext(), new IngestPlaceholderPosition(), "actual_table_0", 0, true);
144         InventoryDumperContext finishedContext = createInventoryDumperContext(
145                 fullContext.getTaskConfig().getDumperContext().getCommonContext(), new IngestFinishedPosition(), "actual_table_1", 1, false);
146         List<InventoryDataRecordPositionCreator> positionCreators = new LinkedList<>();
147         try (
148                 MockedConstruction<InventoryDumperContextSplitter> ignoredSplitter = mockConstruction(InventoryDumperContextSplitter.class,
149                         (mock, context) -> when(mock.split(any())).thenReturn(Arrays.asList(uniqueKeyContext, finishedContext)));
150                 MockedConstruction<InventoryDumper> ignoredInventoryDumperConstruction = mockConstruction(InventoryDumper.class,
151                         (mock, context) -> positionCreators.add((InventoryDataRecordPositionCreator) context.arguments().get(3)))) {
152             preparer.initTasks(Arrays.asList(fullContext, incrementalOnlyContext, stoppingContext));
153             assertThat(fullContext.getInventoryTasks().size(), is(2));
154             assertThat(fullContext.getInventoryTasks().iterator().next().start().size(), is(2));
155             assertTrue(((PipelineTask) fullContext.getInventoryTasks().toArray()[1]).start().isEmpty());
156             assertThat(positionCreators.get(0), isA(UniqueKeyInventoryDataRecordPositionCreator.class));
157             assertThat(positionCreators.get(1), isA(PlaceholderInventoryDataRecordPositionCreator.class));
158             assertThat(fullContext.getIncrementalTasks().size(), is(1));
159             assertThat(fullContext.getIncrementalTasks().iterator().next().start().size(), is(2));
160             assertTrue(incrementalOnlyContext.getInventoryTasks().isEmpty());
161             assertThat(incrementalOnlyContext.getIncrementalTasks().iterator().next().start().size(), is(1));
162         }
163         assertThat(fullContext.getTaskConfig().getDumperContext().getCommonContext().getPosition(), is(ingestPosition));
164         assertThat(incrementalOnlyContext.getTaskConfig().getDumperContext().getCommonContext().getPosition(), is(incrementalPositionFromProgress));
165         verify(jobItemManager, times(2)).persistProgress(any(CDCJobItemContext.class));
166     }

168     private CDCJobItemContext createJobItemContext(final String jobId, final boolean full, final boolean decodeWithTX, final TransmissionJobItemProgress initProgress,
169                                                    final TransmissionProcessContext processContext, final PipelineDataSourceConfiguration dataSourceConfig) {
170         CDCJobConfiguration jobConfig = mock(CDCJobConfiguration.class);
171         when(jobConfig.getJobId()).thenReturn(jobId);
172         lenient().when(jobConfig.isFull()).thenReturn(full);
173         DumperCommonContext dumperCommonContext = createDumperCommonContext(dataSourceConfig);
174         IncrementalDumperContext incrementalDumperContext = new IncrementalDumperContext(dumperCommonContext, jobId, decodeWithTX);
175         ImporterConfiguration importerConfig = new ImporterConfiguration(dataSourceConfig, Collections.emptyMap(), new TableAndSchemaNameMapper(Collections.emptyMap()), 1, null, 0, 1);
176         CDCTaskConfiguration taskConfig = new CDCTaskConfiguration(incrementalDumperContext, importerConfig);
177         return new CDCJobItemContext(jobConfig, 0, initProgress, processContext, taskConfig, dataSourceManager, sink);
178     }
179     
180     private DumperCommonContext createDumperCommonContext(final PipelineDataSourceConfiguration dataSourceConfig) {
181         Map<ShardingSphereIdentifier, ShardingSphereIdentifier> tableNameMap = Collections.singletonMap(new ShardingSphereIdentifier("actual_table_0"),
182                 new ShardingSphereIdentifier("logic_table_0"));
183         return new DumperCommonContext("ds_0", dataSourceConfig, new ActualAndLogicTableNameMapper(tableNameMap),
184                 new TableAndSchemaNameMapper(Collections.singletonMap("logic_table_0", "logic_schema")));
185     }
186     
187     private InventoryDumperContext createInventoryDumperContext(final DumperCommonContext commonContext,
188                                                                 final IngestPosition position, final String actualTableName, final int shardingItem, final boolean hasUniqueKey) {
189         InventoryDumperContext result = new InventoryDumperContext(commonContext);
190         result.getCommonContext().setPosition(position);
191         result.setActualTableName(actualTableName);
192         result.setLogicTableName("logic_" + actualTableName);
193         result.setShardingItem(shardingItem);
194         result.setUniqueKeyColumns(hasUniqueKey ? Collections.singletonList(new PipelineColumnMetaData(1, "id", Types.BIGINT, "bigint", false, true, true)) : Collections.emptyList());
195         return result;
196     }

198     @Test
199     void assertInitTasksFailure() {
200         when(DatabaseTypedSPILoader.getService(any(), any())).thenAnswer(invocationOnMock -> {
201             DialectIncrementalPositionManager result = mock(DialectIncrementalPositionManager.class);
202             when(result.init(any(DataSource.class), any())).thenThrow(SQLException.class);
203             return result;
204         });
205         TransmissionProcessContext processContext = mock(TransmissionProcessContext.class);
206         when(jobItemManager.getProgress(any(), anyInt())).thenReturn(Optional.empty());
207         CDCJobItemContext jobItemContext = createJobItemContext("failure_job", false, false, null, processContext, mock());
208         assertThrows(PrepareJobWithGetBinlogPositionException.class, () -> preparer.initTasks(Collections.singleton(jobItemContext)));
209         verify(jobItemManager).persistProgress(jobItemContext);
210     }
