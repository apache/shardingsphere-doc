18  package org.apache.shardingsphere.mcp.support.database.exception;
19  
20  import org.junit.jupiter.api.Test;
21  import org.junit.jupiter.params.ParameterizedTest;
22  import org.junit.jupiter.params.provider.Arguments;
23  import org.junit.jupiter.params.provider.MethodSource;
24  
25  import java.sql.SQLException;
26  import java.sql.SQLFeatureNotSupportedException;
27  import java.sql.SQLInvalidAuthorizationSpecException;
28  import java.sql.SQLNonTransientConnectionException;
29  import java.sql.SQLSyntaxErrorException;
30  import java.sql.SQLTimeoutException;
31  import java.util.stream.Stream;
32  
33  import static org.hamcrest.MatcherAssert.assertThat;
34  import static org.hamcrest.Matchers.is;
35  
36  class MCPJDBCExceptionClassifierTest {
37      
38      @ParameterizedTest(name = "{0}")
39      @MethodSource("assertClassifyCases")
40      void assertClassify(final String name, final SQLException cause, final MCPJDBCErrorCategory expected) {
41          assertThat(MCPJDBCExceptionClassifier.classify(cause), is(expected));
42      }
43      
44      @Test
45      void assertClassifyCauseChain() {
46          assertThat(MCPJDBCExceptionClassifier.classify(new IllegalStateException(new SQLException("missing table", "42P01"))),
47                  is(MCPJDBCErrorCategory.OBJECT_NOT_VISIBLE));
48      }
49      
50      @Test
51      void assertClassifyNextExceptionChain() {
52          SQLException cause = new SQLException("batch failed");
53          cause.setNextException(new SQLException("syntax error", "42601"));
54          assertThat(MCPJDBCExceptionClassifier.classify(cause), is(MCPJDBCErrorCategory.SYNTAX));
55      }
56      
57      @Test
58      void assertClassifyNextExceptionAfterDialectFallback() {
59          SQLException cause = new SQLException("batch failed", "42000", 1055);
60          cause.setNextException(new SQLException("syntax error", "42000", 1064));
61          assertThat(MCPJDBCExceptionClassifier.classify("MySQL", cause), is(MCPJDBCErrorCategory.SYNTAX));
62      }
63      
64      @Test
65      void assertClassifySpecificNextExceptionAfterSyntaxCandidate() {
66          SQLException cause = new SQLSyntaxErrorException("statement failed");
67          cause.setNextException(new SQLException("missing table", "42P01"));
68          assertThat(MCPJDBCExceptionClassifier.classify(cause), is(MCPJDBCErrorCategory.OBJECT_NOT_VISIBLE));
69      }
70      
71      @Test
72      void assertClassifyIgnoresSuppressedException() {
73          SQLException cause = new SQLException("query failed");
74          cause.addSuppressed(new SQLTimeoutException("timeout"));
75          assertThat(MCPJDBCExceptionClassifier.classify(cause), is(MCPJDBCErrorCategory.QUERY_FAILED));
76      }
77      
78      @Test
79      void assertClassifyClassifiedQueryFailure() {
80          MCPDatabaseQueryFailedException cause = new MCPDatabaseQueryFailedException(
81                  MCPJDBCErrorCategory.AUTHORIZATION, new SQLSyntaxErrorException("access denied", "42000", 1044));
82          assertThat(MCPJDBCExceptionClassifier.classify(cause), is(MCPJDBCErrorCategory.AUTHORIZATION));
83      }
84      
85      @Test
86      void assertClassifyDatabaseSyntaxFailure() {
87          assertThat(MCPJDBCExceptionClassifier.classify(new MCPDatabaseSQLSyntaxException(new SQLException("syntax error", "42000", 1064))),
88                  is(MCPJDBCErrorCategory.SYNTAX));
89      }
90      
91      @Test
92      void assertClassifyUnambiguousJDBCTypeBeforeDialect() {
93          assertThat(MCPJDBCExceptionClassifier.classify("MySQL", new SQLTimeoutException("timeout", "42000", 1064)), is(MCPJDBCErrorCategory.TIMEOUT));
94      }
95      
96      @Test
97      void assertClassifyDialectBeforePortableAuthentication() {
98          assertThat(MCPJDBCExceptionClassifier.classify("Firebird", new SQLInvalidAuthorizationSpecException("permission denied", "28000", 335544352)),
99                  is(MCPJDBCErrorCategory.AUTHORIZATION));
100     }
101     
102     @Test
103     void assertClassifyAmbiguousDialectError() {
104         assertThat(MCPJDBCExceptionClassifier.classify("MySQL", new SQLSyntaxErrorException("ambiguous", "42000", 1055)),
105                 is(MCPJDBCErrorCategory.QUERY_FAILED));
106     }
107     
108     @ParameterizedTest(name = "{0}")
109     @MethodSource("assertClassifyDialectCases")
110     void assertClassifyDialect(final String databaseType, final SQLException cause, final MCPJDBCErrorCategory expected) {
111         assertThat(MCPJDBCExceptionClassifier.classify(databaseType, cause), is(expected));
112     }
113     
114     @Test
115     void assertClassifyMariaDBAlias() {
116         assertThat(MCPJDBCExceptionClassifier.classify("MariaDB", new SQLException("syntax error", "42000", 1064)), is(MCPJDBCErrorCategory.SYNTAX));
117     }
118     
119     private static Stream<Arguments> assertClassifyCases() {
120         return Stream.of(
121                 Arguments.of("timeout", new SQLTimeoutException(), MCPJDBCErrorCategory.TIMEOUT),
122                 Arguments.of("query timeout SQLState", new SQLException("", "HYT00"), MCPJDBCErrorCategory.TIMEOUT),
123                 Arguments.of("connection timeout SQLState", new SQLException("", "HYT01"), MCPJDBCErrorCategory.TIMEOUT),
124                 Arguments.of("feature not supported", new SQLFeatureNotSupportedException(), MCPJDBCErrorCategory.FEATURE_NOT_SUPPORTED),
125                 Arguments.of("connection subtype", new SQLNonTransientConnectionException(), MCPJDBCErrorCategory.CONNECTION),
126                 Arguments.of("connection SQLState", new SQLException("", "08006"), MCPJDBCErrorCategory.CONNECTION),
127                 Arguments.of("authentication", new SQLException("", "28000"), MCPJDBCErrorCategory.AUTHENTICATION),
128                 Arguments.of("authentication subtype", new SQLInvalidAuthorizationSpecException(), MCPJDBCErrorCategory.AUTHENTICATION),
129                 Arguments.of("authorization", new SQLException("", "42501"), MCPJDBCErrorCategory.AUTHORIZATION),
130                 Arguments.of("database not visible", new SQLException("", "3D000"), MCPJDBCErrorCategory.OBJECT_NOT_VISIBLE),
131                 Arguments.of("object not visible", new SQLSyntaxErrorException("", "42P01"), MCPJDBCErrorCategory.OBJECT_NOT_VISIBLE),
132                 Arguments.of("routine not visible", new SQLException("", "42883"), MCPJDBCErrorCategory.OBJECT_NOT_VISIBLE),
133                 Arguments.of("syntax subtype", new SQLSyntaxErrorException(), MCPJDBCErrorCategory.SYNTAX),
134                 Arguments.of("syntax SQLState", new SQLException("", "42601"), MCPJDBCErrorCategory.SYNTAX),
135                 Arguments.of("ambiguous class 42", new SQLException("", "42000"), MCPJDBCErrorCategory.QUERY_FAILED),
136                 Arguments.of("unknown", new SQLException(), MCPJDBCErrorCategory.QUERY_FAILED));
137     }
138     
139     private static Stream<Arguments> assertClassifyDialectCases() {
140         return Stream.of(
141                 Arguments.of("ClickHouse", new SQLException("unknown table", "07000", 60), MCPJDBCErrorCategory.OBJECT_NOT_VISIBLE),
142                 Arguments.of("Firebird", new SQLException("permission denied", "28000", 335544352), MCPJDBCErrorCategory.AUTHORIZATION),
143                 Arguments.of("Hive", new SQLException("invalid table", "42000", 10001), MCPJDBCErrorCategory.OBJECT_NOT_VISIBLE),
144                 Arguments.of("Oracle", new SQLException("insufficient privileges", "42000", 1031), MCPJDBCErrorCategory.AUTHORIZATION),
145                 Arguments.of("Presto", new SQLException("not supported", null, 13), MCPJDBCErrorCategory.FEATURE_NOT_SUPPORTED),
146                 Arguments.of("SQLServer", new SQLException("login failed", "S0001", 18456), MCPJDBCErrorCategory.AUTHENTICATION));
147     }
