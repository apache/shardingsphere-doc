18  package org.apache.shardingsphere.database.connector.oracle.jdbcurl;
19  
20  import com.google.common.base.Strings;
21  import org.apache.shardingsphere.database.connector.core.exception.UnrecognizedDatabaseURLException;
22  import org.apache.shardingsphere.database.connector.core.jdbcurl.parser.ConnectionProperties;
23  import org.apache.shardingsphere.database.connector.core.jdbcurl.parser.ConnectionPropertiesParser;
24  import org.apache.shardingsphere.database.connector.core.jdbcurl.parser.StandardJdbcUrlParser;
25  
26  import java.util.Arrays;
27  import java.util.List;
28  import java.util.Properties;
29  import java.util.regex.Matcher;
30  import java.util.regex.Pattern;
31  
32  /**
33   * Connection properties parser of Oracle.
34   */
35  public final class OracleConnectionPropertiesParser implements ConnectionPropertiesParser {
36      
37      private static final int DEFAULT_PORT = 1521;
38      
39      private static final int THIN_MATCH_GROUP_COUNT = 6;
40      
41      private static final String QUERY_DELIMITER = "?";
42      
43      private static final Pattern THIN_URL_PATTERN = Pattern.compile(
44              "jdbc:oracle:(thin|oci|kprb):@((?:(?:tcp|tcps):)?//)?(\\[[^]]+\\]|[\\w\\-\\.]+)(?::(\\d+))?"
45                      + "(?:/([\\w\\-\\.]+)(?::(?:DEDICATED|SHARED|POOLED))?(?:/[\\w\\-\\.]+)?|:([\\w\\-\\.]+)|//[\\w\\-\\.]+|/:(?:DEDICATED|SHARED|POOLED)(?:/[\\w\\-\\.]+)?)?",
46              Pattern.CASE_INSENSITIVE);
47      
48      private static final Pattern CONNECT_DESCRIPTOR_URL_PATTERN = Pattern.compile(
49              "jdbc:oracle:(thin|oci|kprb):@[(\\w\\s=)]+HOST\\s*=\\s*(\\[[^]]+\\]|[\\w\\-\\.]+).*PORT\\s*=\\s*(\\d+).*SERVICE_NAME\\s*=\\s*([\\w\\.]+)\\).*\\)");
50      
51      @Override
52      public ConnectionProperties parse(final String url, final String username, final String catalog) {
53          String urlWithoutQuery = url.contains(QUERY_DELIMITER) ? url.substring(0, url.indexOf(QUERY_DELIMITER)) : url;
54          Properties queryProps = url.contains(QUERY_DELIMITER) ? new StandardJdbcUrlParser().parseQueryProperties(url.substring(url.indexOf(QUERY_DELIMITER) + 1)) : new Properties();
55          List<Matcher> matchers = Arrays.asList(THIN_URL_PATTERN.matcher(urlWithoutQuery), CONNECT_DESCRIPTOR_URL_PATTERN.matcher(urlWithoutQuery));
56          Matcher matcher = matchers.stream().filter(Matcher::matches).filter(each -> THIN_MATCH_GROUP_COUNT == each.groupCount() || hasNoDescriptorTail(urlWithoutQuery))
57                  .findAny().orElseThrow(() -> new UnrecognizedDatabaseURLException(url, THIN_URL_PATTERN.pattern()));
58          int groupCount = matcher.groupCount();
59          return THIN_MATCH_GROUP_COUNT == groupCount ? getThinConnectionProperties(username, matcher, queryProps) : getStandardConnectionProperties(username, matcher, queryProps);
60      }
61      
62      private boolean hasNoDescriptorTail(final String url) {
63          int depth = 0;
64          for (int index = url.indexOf('@') + 1; index < url.length(); index++) {
65              char current = url.charAt(index);
66              if ('(' == current) {
67                  depth++;
68                  continue;
69              }
70              if (')' != current || --depth > 0) {
71                  continue;
72              }
73              int nextIndex = index + 1;
74              while (nextIndex < url.length() && Character.isWhitespace(url.charAt(nextIndex))) {
75                  nextIndex++;
76              }
77              if (nextIndex < url.length() && '(' != url.charAt(nextIndex) && ')' != url.charAt(nextIndex)) {
78                  return false;
79              }
80          }
81          return true;
82      }
83      
84      private ConnectionProperties getThinConnectionProperties(final String username, final Matcher matcher, final Properties queryProps) {
85          String serviceName = null == matcher.group(5) ? matcher.group(6) : matcher.group(5);
86          String hostname = matcher.group(3);
87          return new ConnectionProperties(hostname.startsWith("[") ? hostname.substring(1, hostname.length() - 1) : hostname,
88                  Strings.isNullOrEmpty(matcher.group(4)) ? DEFAULT_PORT : Integer.parseInt(matcher.group(4)), serviceName, username, queryProps);
89      }
90      
91      private ConnectionProperties getStandardConnectionProperties(final String username, final Matcher matcher, final Properties queryProps) {
92          String hostname = matcher.group(2);
93          return new ConnectionProperties(hostname.startsWith("[") ? hostname.substring(1, hostname.length() - 1) : hostname,
94                  Integer.parseInt(matcher.group(3)), matcher.group(4), username, queryProps);
95      }
96      
97      @Override
98      public String getDatabaseType() {
99          return "Oracle";
100     }
101 }
