73          Map<String, Object> actual = validate(createService(inspectionService), createRuleSnapshot(), queryFacade);
74          assertThat(actual.get("overall_status"), is(WorkflowLifecycle.STATUS_PASSED));
75      }

241     void assertValidateApplyArtifactsRejectsUnavailableRuleAlgorithm() {
242         WorkflowContextSnapshot snapshot = createRuleSnapshot();
243         ShadowRuleWorkflowRequest request = (ShadowRuleWorkflowRequest) snapshot.getRequest();
244         try (MockedStatic<TypedSPILoader> mockedStatic = mockStatic(TypedSPILoader.class)) {
245             mockedStatic.when(() -> TypedSPILoader.checkService(ShadowAlgorithm.class, "VALUE_MATCH",
246                     WorkflowAlgorithmUtils.createProperties(request.getAlgorithmProperties()))).thenThrow(new IllegalArgumentException("unavailable"));
247             List<Map<String, Object>> actual = new ShadowWorkflowValidationService().validate(snapshot, List.of(createRuleDistSQLArtifact("CREATE SHADOW RULE `shadow_rule`(...)")));
248             assertThat(actual.size(), is(1));
249             assertThat(actual.getFirst().get("code"), is(WorkflowIssueCode.SQL_EXECUTABILITY_FAILED));
250         }
251     }

316     private WorkflowContextSnapshot createRuleSnapshot() {
317         ShadowRuleWorkflowRequest request = new ShadowRuleWorkflowRequest();
318         request.setDatabase("logic_db");
319         request.setRuleName("shadow_rule");
320         request.setSourceStorageUnit("demo_ds");
321         request.setShadowStorageUnit("demo_ds_shadow");
322         request.setTableName("t_order");
323         request.setAlgorithmType("VALUE_MATCH");
324         request.putAlgorithmProperties(Map.of("operation", "INSERT", "column", "shadow", "value", "true"));
325         return createSnapshot(request, "create");
326     }

355     private WorkflowContextSnapshot createCleanupSnapshot() {
356         ShadowAlgorithmCleanupWorkflowRequest request = new ShadowAlgorithmCleanupWorkflowRequest();
357         request.setDatabase("logic_db");
358         request.setAlgorithmName("unused_algorithm");
359         return createSnapshot(request, WorkflowLifecycle.OPERATION_DROP);

362     private WorkflowContextSnapshot createSnapshot(final WorkflowRequest request, final String operationType) {
363         WorkflowContextSnapshot result = new WorkflowContextSnapshot();
364         result.setPlanId("plan-1");
365         result.setSessionId("session-1");
366         result.setStatus(WorkflowLifecycle.STATUS_EXECUTED);
367         result.setRequest(request);
368         ClarifiedIntent clarifiedIntent = new ClarifiedIntent();
369         clarifiedIntent.setOperationType(operationType);
370         result.setClarifiedIntent(clarifiedIntent);
371         return result;
372     }
