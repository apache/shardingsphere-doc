45  /**
46   * Interval sharding algorithm.
47   */
48  public final class IntervalShardingAlgorithm implements StandardShardingAlgorithm<Comparable<?>> {
49      
50      private static final String DATE_TIME_PATTERN_KEY = "datetime-pattern";
51      
52      private static final String DATE_TIME_LOWER_KEY = "datetime-lower";
53      
54      private static final String DATE_TIME_UPPER_KEY = "datetime-upper";
55      
56      private static final String SHARDING_SUFFIX_FORMAT_KEY = "sharding-suffix-pattern";
57      
58      private static final String INTERVAL_AMOUNT_KEY = "datetime-interval-amount";
59      
60      private static final String INTERVAL_UNIT_KEY = "datetime-interval-unit";
61      
62      private String dateTimePatternString;
63      
64      private DateTimeFormatter dateTimeFormatter;
65      
66      private int dateTimePatternLength;
67      
68      private TemporalAccessor dateTimeLower;
69      
70      private TemporalAccessor dateTimeUpper;
71      
72      private DateTimeFormatter tableSuffixPattern;
73      
74      private int stepAmount;
75      
76      private ChronoUnit stepUnit;
77      
78      @Override
79      public void init(final Properties props) {
80          dateTimePatternString = getDateTimePattern(props);
81          dateTimeFormatter = DateTimeFormatter.ofPattern(dateTimePatternString);
82          dateTimePatternLength = dateTimePatternString.length();
83          dateTimeLower = getDateTimeLower(props, dateTimePatternString);
84          dateTimeUpper = getDateTimeUpper(props, dateTimePatternString);
85          tableSuffixPattern = getTableSuffixPattern(props);
86          stepAmount = Integer.parseInt(props.getOrDefault(INTERVAL_AMOUNT_KEY, 1).toString());
87          stepUnit = props.containsKey(INTERVAL_UNIT_KEY) ? getStepUnit(props.getProperty(INTERVAL_UNIT_KEY)) : ChronoUnit.DAYS;
88      }
89      
90      private String getDateTimePattern(final Properties props) {
91          ShardingSpherePreconditions.checkContainsKey(props, DATE_TIME_PATTERN_KEY, () -> new AlgorithmInitializationException(this, String.format("%s can not be null", DATE_TIME_PATTERN_KEY)));
92          return props.getProperty(DATE_TIME_PATTERN_KEY);
93      }
94      
95      private TemporalAccessor getDateTimeLower(final Properties props, final String dateTimePattern) {
96          ShardingSpherePreconditions.checkContainsKey(props, DATE_TIME_LOWER_KEY, () -> new AlgorithmInitializationException(this, String.format("%s can not be null.", DATE_TIME_LOWER_KEY)));
97          return getDateTime(DATE_TIME_LOWER_KEY, props.getProperty(DATE_TIME_LOWER_KEY), dateTimePattern);
98      }
99      
100     private TemporalAccessor getDateTimeUpper(final Properties props, final String dateTimePattern) {
101         return props.containsKey(DATE_TIME_UPPER_KEY) ? getDateTime(DATE_TIME_UPPER_KEY, props.getProperty(DATE_TIME_UPPER_KEY), dateTimePattern) : LocalDateTime.now();
102     }
103     
104     private TemporalAccessor getDateTime(final String dateTimeKey, final String dateTimeValue, final String dateTimePattern) {
105         try {
106             return dateTimeFormatter.parse(dateTimeValue);
107         } catch (final DateTimeParseException ignored) {
108             throw new InvalidDatetimeFormatException(dateTimeKey, dateTimeValue, dateTimePattern);
109         }
110     }
111     
112     private DateTimeFormatter getTableSuffixPattern(final Properties props) {
113         String suffix = props.getProperty(SHARDING_SUFFIX_FORMAT_KEY);
114         ShardingSpherePreconditions.checkNotEmpty(suffix, () -> new AlgorithmInitializationException(this, String.format("%s can not be null or empty.", SHARDING_SUFFIX_FORMAT_KEY)));
115         return DateTimeFormatter.ofPattern(suffix);
116     }
117     
118     private ChronoUnit getStepUnit(final String stepUnit) {
119         for (ChronoUnit each : ChronoUnit.values()) {
120             if (each.toString().equalsIgnoreCase(stepUnit)) {
121                 return each;
122             }
123         }
124         throw new UnsupportedSQLOperationException(String.format("Cannot find step unit for specified %s property: `%s`", INTERVAL_UNIT_KEY, stepUnit));
125     }
126     
127     @Override
128     public String doSharding(final Collection<String> availableTargetNames, final PreciseShardingValue<Comparable<?>> shardingValue) {
129         ShardingSpherePreconditions.checkNotNull(shardingValue.getValue(), NullShardingValueException::new);
130         Range<Comparable<?>> range = Range.singleton(shardingValue.getValue());
131         return getMatchedTables(availableTargetNames, range).stream().findFirst().orElse(null);
132     }
133     
134     @Override
135     public Collection<String> doSharding(final Collection<String> availableTargetNames, final RangeShardingValue<Comparable<?>> shardingValue) {
136         return getMatchedTables(availableTargetNames, shardingValue.getValueRange());
137     }
138     
139     @SuppressWarnings("unchecked")
140     private <T extends TemporalAccessor & Comparable<?>> Collection<String> getMatchedTables(final Collection<String> availableTargetNames, final Range<Comparable<?>> range) {
141         Collection<String> result = new HashSet<>();
142         TemporalHandler<T> temporalHandler = TemporalHandlerFactory.newInstance(dateTimeLower);
143         T dateTimeUpper = temporalHandler.convertTo(this.dateTimeUpper);
144         T dateTimeLower = temporalHandler.convertTo(this.dateTimeLower);
145         T calculateTimeAsView = temporalHandler.convertTo(this.dateTimeLower);
146         while (!temporalHandler.isAfter(calculateTimeAsView, dateTimeUpper, stepAmount)) {
147             if (hasIntersection(Range.closedOpen(calculateTimeAsView, temporalHandler.add(calculateTimeAsView, stepAmount, stepUnit)), range, dateTimeLower, dateTimeUpper, temporalHandler)) {
148                 result.addAll(getMatchedTables(calculateTimeAsView, availableTargetNames, temporalHandler));
149             }
150             calculateTimeAsView = temporalHandler.add(calculateTimeAsView, stepAmount, stepUnit);
151         }
152         return result;
153     }
154     
155     private <T extends TemporalAccessor & Comparable<?>> Collection<String> getMatchedTables(final TemporalAccessor calculateTimeAsView,
156                                                                                              final Collection<String> availableTargetNames, final TemporalHandler<T> temporalHandler) {
157         return availableTargetNames.parallelStream().filter(each -> each.endsWith(tableSuffixPattern.format(temporalHandler.convertTo(calculateTimeAsView)))).collect(Collectors.toSet());
158     }
159     
160     private <T extends TemporalAccessor & Comparable<?>> boolean hasIntersection(final Range<T> calculateRange, final Range<Comparable<?>> range,
161                                                                                  final T temporalLower, final T temporalUpper, final TemporalHandler<T> temporalHandler) {
162         T lower = range.hasLowerBound() ? parseTemporal(range.lowerEndpoint(), temporalHandler) : temporalLower;
163         T upper = range.hasUpperBound() ? parseTemporal(range.upperEndpoint(), temporalHandler) : temporalUpper;
164         BoundType lowerBoundType = range.hasLowerBound() ? range.lowerBoundType() : BoundType.CLOSED;
165         BoundType upperBoundType = range.hasUpperBound() ? range.upperBoundType() : BoundType.CLOSED;
166         Range<T> dateTimeRange = Range.range(lower, lowerBoundType, upper, upperBoundType);
167         return calculateRange.isConnected(dateTimeRange) && !calculateRange.intersection(dateTimeRange).isEmpty();
168     }
169     
170     private <T extends TemporalAccessor> T parseTemporal(final Comparable<?> endpoint, final TemporalHandler<T> temporalHandler) {
171         String dateTimeText = getDateTimeText(endpoint);
172         return dateTimeText.length() >= dateTimePatternLength
173                 ? temporalHandler.parse(dateTimeText.substring(0, dateTimePatternLength), dateTimeFormatter)
174                 : temporalHandler.parse(dateTimeText, createRelaxedDateTimeFormatter(dateTimeText));
175     }
176     
177     private String getDateTimeText(final Comparable<?> endpoint) {
178         if (endpoint instanceof Instant) {
179             return dateTimeFormatter.format(((Instant) endpoint).atZone(ZoneId.systemDefault()));
180         }
181         if (endpoint instanceof TemporalAccessor) {
182             return dateTimeFormatter.format((TemporalAccessor) endpoint);
183         }
184         if (endpoint instanceof java.sql.Date) {
185             return dateTimeFormatter.format(((java.sql.Date) endpoint).toLocalDate());
186         }
187         if (endpoint instanceof java.util.Date) {
188             return dateTimeFormatter.format(((java.util.Date) endpoint).toInstant().atZone(ZoneId.systemDefault()).toLocalDateTime());
189         }
190         return endpoint.toString();
191     }
192     
193     /*
194      * When the sharding key is a {@link String} and the length of this {@link String} is less than the `datetime-pattern` set by the algorithm, ShardingSphere will try to use a substring of
195      * `datetime-pattern` to parse the sharding key. This is to be compatible with the behavior of ORM libraries such as <a href="https://github.com/go-gorm/gorm">go-gorm/gorm</a>.
196      *
197      * @param dateTimeText Sharding key with class name {@link String}
198      *
199      * @return Child `datetime-pattern`, the pattern length is consistent with the shard key.
200      */
201     private DateTimeFormatter createRelaxedDateTimeFormatter(final String dateTimeText) {
202         return DateTimeFormatter.ofPattern(dateTimePatternString.substring(0, dateTimeText.length()));
203     }
204     
205     @Override
206     public String getType() {
207         return "INTERVAL";
208     }
209 }
