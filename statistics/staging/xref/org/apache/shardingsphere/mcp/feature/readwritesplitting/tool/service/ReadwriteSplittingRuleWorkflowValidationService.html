44  /**
45   * Readwrite-splitting rule workflow validation service.
46   */
47  public final class ReadwriteSplittingRuleWorkflowValidationService implements MCPWorkflowRuntimeHandler, MCPWorkflowApplyArtifactValidator {
48      
49      private final WorkflowValidationSupport validationSupport = new WorkflowValidationSupport();
50      
51      private final ReadwriteSplittingInspectionService inspectionService = new ReadwriteSplittingInspectionService();
52      
53      @Override
54      public ValidationReport validate(final WorkflowContextSnapshot snapshot, final MCPFeatureQueryFacade queryFacade) {
55          ValidationReport result = new ValidationReport();
56          queryFacade.checkDatabaseCapability(snapshot.getRequest().getDatabase());
57          List<Map<String, Object>> rules = inspectionService.queryRules(queryFacade, snapshot.getRequest().getDatabase());
58          result.setRuleValidation(validateRules(snapshot, rules, result, queryFacade));
59          result.setOverallStatus(validationSupport.resolveOverallStatus(result.getRuleValidation()));
60          return result;
61      }
62      
63      @Override
64      public List<Map<String, Object>> validate(final WorkflowContextSnapshot snapshot, final Collection<ExecutableWorkflowArtifact> artifacts) {
65          List<Map<String, Object>> result = new LinkedList<>();
66          for (ExecutableWorkflowArtifact each : artifacts) {
67              addRuleDistSQLIssues(result, snapshot, each.sql(), each.displaySql());
68          }
69          return result;
70      }
71      
72      private void addRuleDistSQLIssues(final List<Map<String, Object>> issues, final WorkflowContextSnapshot snapshot, final String sql, final String displaySql) {
73          if (!isReadwriteSplittingRuleDistSQL(sql) || !(snapshot.getRequest() instanceof final ReadwriteSplittingRuleWorkflowRequest request)) {
74              return;
75          }
76          addLoadBalancerIssue(issues, request, displaySql);
77          addWeightIssues(issues, request, displaySql);
78      }
79      
80      private boolean isReadwriteSplittingRuleDistSQL(final String sql) {
81          String actualSQL = sql.trim().toUpperCase(Locale.ENGLISH);
82          return actualSQL.startsWith("CREATE READWRITE_SPLITTING RULE") || actualSQL.startsWith("ALTER READWRITE_SPLITTING RULE");
83      }
84      
85      private void addLoadBalancerIssue(final List<Map<String, Object>> issues, final ReadwriteSplittingRuleWorkflowRequest request, final String displaySql) {
86          if (request.getLoadBalancerType().isEmpty()) {
87              return;
88          }
89          if (!WorkflowAlgorithmUtils.isAlgorithmServiceAvailable(LoadBalanceAlgorithm.class, request.getLoadBalancerType(), request.getLoadBalancerProperties())) {
90              issues.add(validationSupport.createSQLExecutabilityIssue(String.format("Generated readwrite-splitting DistSQL references load balancer algorithm `%s`, "
91                      + "but it cannot be loaded or initialized by LoadBalanceAlgorithm SPI.", request.getLoadBalancerType()), displaySql));
92          }
93      }
94      
95      private void addWeightIssues(final List<Map<String, Object>> issues, final ReadwriteSplittingRuleWorkflowRequest request, final String displaySql) {
96          if (!"WEIGHT".equalsIgnoreCase(request.getLoadBalancerType())) {
97              return;
98          }
99          if (request.getLoadBalancerProperties().isEmpty()) {
100             issues.add(validationSupport.createSQLExecutabilityIssue("Generated readwrite-splitting DistSQL uses WEIGHT load balancer without weight properties.", displaySql));
101             return;
102         }
103         for (String each : request.getLoadBalancerProperties().keySet()) {
104             if (!request.getReadStorageUnits().contains(each)) {
105                 issues.add(validationSupport.createSQLExecutabilityIssue(
106                         String.format("Generated readwrite-splitting DistSQL defines weight for unknown read storage unit `%s`.", each), displaySql));
107             }
108         }
109         for (String each : request.getReadStorageUnits()) {
110             if (!request.getLoadBalancerProperties().containsKey(each)) {
111                 issues.add(validationSupport.createSQLExecutabilityIssue(
112                         String.format("Generated readwrite-splitting DistSQL is missing weight for read storage unit `%s`.", each), displaySql));
113             }
114         }
115     }
116     
117     private ValidationSection validateRules(final WorkflowContextSnapshot snapshot, final List<Map<String, Object>> rules,
118                                             final ValidationReport validationReport, final MCPFeatureQueryFacade queryFacade) {
119         ReadwriteSplittingRuleWorkflowRequest request = (ReadwriteSplittingRuleWorkflowRequest) snapshot.getRequest();
120         boolean ruleExists = containsRule(rules, queryFacade, request.getDatabase(), request.getRuleName());
121         boolean dropWorkflow = WorkflowLifecycleUtils.isDropWorkflow(snapshot);
122         boolean expectedRuleExists = !dropWorkflow;
123         if (expectedRuleExists != ruleExists) {
124             addRuleMismatch(validationReport, request.getRuleName(), dropWorkflow);
125             return new ValidationSection(WorkflowLifecycle.STATUS_FAILED, rules, "Readwrite-splitting rule state does not match the planned DistSQL artifact.");
126         }
127         if (!dropWorkflow && !matchesRuleShape(rules, queryFacade, request)) {
128             addRuleShapeMismatch(validationReport, request.getRuleName());
129             return new ValidationSection(WorkflowLifecycle.STATUS_FAILED, rules, "Readwrite-splitting rule fields do not match the planned DistSQL artifact.");
130         }
131         return new ValidationSection(WorkflowLifecycle.STATUS_PASSED, rules, "Readwrite-splitting rule state matches the planned DistSQL artifact.");
132     }
133     
134     private boolean containsRule(final List<Map<String, Object>> rules, final MCPFeatureQueryFacade queryFacade, final String databaseName, final String ruleName) {
135         return rules.stream().anyMatch(each -> queryFacade.isSameIdentifier(databaseName, IdentifierScope.TABLE, ruleName, WorkflowRuleValueUtils.getRuleValue(each, "name")));
136     }
137     
138     private boolean matchesRuleShape(final List<Map<String, Object>> rules, final MCPFeatureQueryFacade queryFacade, final ReadwriteSplittingRuleWorkflowRequest request) {
139         List<Map<String, Object>> targetRules = rules.stream().filter(each -> queryFacade.isSameIdentifier(
140                 request.getDatabase(), IdentifierScope.TABLE, request.getRuleName(), WorkflowRuleValueUtils.getRuleValue(each, "name")))
141                 .toList();
142         return 1 == targetRules.size() && matchesRuleShape(targetRules.getFirst(), queryFacade, request);
143     }
144     
145     private boolean matchesRuleShape(final Map<String, Object> rule, final MCPFeatureQueryFacade queryFacade, final ReadwriteSplittingRuleWorkflowRequest request) {
146         return queryFacade.isSameIdentifier(
147                 request.getDatabase(), IdentifierScope.TABLE, request.getWriteStorageUnit(), WorkflowRuleValueUtils.getRuleValue(rule, "write_storage_unit_name"))
148                 && matchesReadStorageUnits(rule, queryFacade, request)
149                 && WorkflowRuleValueUtils.getRuleValue(rule, "transactional_read_query_strategy").equalsIgnoreCase(request.getTransactionalReadQueryStrategy())
150                 && WorkflowRuleValueUtils.getRuleValue(rule, "load_balancer_type").equalsIgnoreCase(request.getLoadBalancerType())
151                 && WorkflowAlgorithmUtils.createPropertyMap(rule.get("load_balancer_props")).equals(request.getLoadBalancerProperties());
152     }
153     
154     private boolean matchesReadStorageUnits(final Map<String, Object> rule, final MCPFeatureQueryFacade queryFacade, final ReadwriteSplittingRuleWorkflowRequest request) {
155         String readStorageUnits = WorkflowRuleValueUtils.getRuleValue(rule, "read_storage_unit_names");
156         List<String> actualReadStorageUnits = Arrays.stream(readStorageUnits.split(",")).map(String::trim).filter(each -> !each.isEmpty()).toList();
157         return actualReadStorageUnits.size() == request.getReadStorageUnits().size() && request.getReadStorageUnits().stream().allMatch(each -> actualReadStorageUnits.stream()
158                 .anyMatch(actual -> queryFacade.isSameIdentifier(request.getDatabase(), IdentifierScope.TABLE, each, actual)));
159     }
160     
161     private void addRuleMismatch(final ValidationReport validationReport, final String ruleName, final boolean dropWorkflow) {
162         validationReport.getMismatches().add(validationSupport.createMismatch(WorkflowIssueCode.RULE_STATE_MISMATCH, "rule",
163                 dropWorkflow ? "rule absent" : "rule present", ruleName,
164                 dropWorkflow ? "Readwrite-splitting rule still exists after drop." : "Readwrite-splitting rule is missing after create or alter.",
165                 dropWorkflow ? "Drop the readwrite-splitting rule again or investigate the failure." : "Apply the planned readwrite-splitting rule DistSQL again."));
166     }
167     
168     private void addRuleShapeMismatch(final ValidationReport validationReport, final String ruleName) {
169         validationReport.getMismatches().add(validationSupport.createMismatch(WorkflowIssueCode.RULE_STATE_MISMATCH, "rule", "matching fields", ruleName,
170                 "Readwrite-splitting rule exists but its storage units, transactional strategy or load balancer differ from the planned artifact.",
171                 "Review the current rule state and re-apply or re-plan the readwrite-splitting rule DistSQL."));
172     }
