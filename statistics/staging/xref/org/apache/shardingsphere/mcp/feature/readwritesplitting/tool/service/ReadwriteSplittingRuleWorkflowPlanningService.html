46      private static final List<String> SUPPORTED_OPERATION_TYPES = List.of(
47              WorkflowLifecycle.OPERATION_CREATE, WorkflowLifecycle.OPERATION_ALTER, WorkflowLifecycle.OPERATION_DROP);
48      
49      private static final List<String> INTERACTION_STEPS = List.of(
50              "Confirm database, rule name, storage units, strategy and lifecycle",
51              "Inspect DistSQL-visible readwrite-splitting rules",
52              "Generate readwrite-splitting rule DistSQL artifacts",
53              "Review artifacts and choose execution mode",
54              "Execute or export artifacts",
55              "Validate and summarize");
56      
57      private static final List<String> VALIDATION_LAYERS = List.of("rules");
58      
59      private final WorkflowPlanningSupport planningSupport = new WorkflowPlanningSupport();
60      
61      private final ReadwriteSplittingInspectionService inspectionService = new ReadwriteSplittingInspectionService();
62      
63      private final ReadwriteSplittingAlgorithmRecommendationService algorithmRecommendationService = new ReadwriteSplittingAlgorithmRecommendationService();
64      
65      private final ReadwriteSplittingAlgorithmPropertyTemplateService algorithmPropertyTemplateService = new ReadwriteSplittingAlgorithmPropertyTemplateService();
66      
67      private final ReadwriteSplittingRuleDistSQLPlanningService distSQLPlanningService = new ReadwriteSplittingRuleDistSQLPlanningService();

69      /**
70       * Plan readwrite-splitting rule workflow.
71       *
72       * @param workflowSessionContext workflow session context
73       * @param queryFacade query facade
74       * @param request workflow request
75       * @return workflow snapshot
76       */
77      public WorkflowContextSnapshot plan(final WorkflowSessionContext workflowSessionContext, final MCPFeatureQueryFacade queryFacade,
78                                          final ReadwriteSplittingRuleWorkflowRequest request) {
79          WorkflowContextSnapshot result = workflowSessionContext.getOrCreate(request.getPlanId());
80          ReadwriteSplittingRuleWorkflowRequest mergedRequest = prepareSnapshot(result, request);
81          ClarifiedIntent clarifiedIntent = result.getClarifiedIntent();
82          planningSupport.applyResolvedIntent(mergedRequest, clarifiedIntent);
83          if (!planningSupport.ensureSupportedOperationType(clarifiedIntent, SUPPORTED_OPERATION_TYPES, result)) {
84              return planningSupport.persistPlanningInterruption(workflowSessionContext, result);
85          }
86          if (!ensurePlanningContext(mergedRequest, clarifiedIntent, result)) {
87              return planningSupport.persistPlanningInterruption(workflowSessionContext, result);
88          }
89          queryFacade.checkDatabaseCapability(mergedRequest.getDatabase());
90          List<Map<String, Object>> existingRules = inspectionService.queryRules(queryFacade, mergedRequest.getDatabase());
91          if (!ensureLifecycleState(clarifiedIntent, mergedRequest, existingRules, result, queryFacade)) {
92              return workflowSessionContext.persist(result, WorkflowLifecycle.STEP_FAILED, WorkflowLifecycle.STATUS_FAILED);
93          }
94          if (!WorkflowLifecycle.OPERATION_DROP.equalsIgnoreCase(clarifiedIntent.getOperationType())) {
95              planAlgorithms(queryFacade, mergedRequest, result);
96              if (!mergedRequest.getLoadBalancerType().isEmpty() && !planningSupport.isReadyForArtifactPlanning(mergedRequest, clarifiedIntent, result, findPropertyRequirements(mergedRequest),
97                      "Please use a load-balance algorithm visible in the current Proxy and provide required properties.")) {
98                  return workflowSessionContext.persist(result, WorkflowLifecycle.STEP_CLARIFYING, WorkflowLifecycle.STATUS_CLARIFYING);
99              }
100         }
101         addRuleArtifact(result, mergedRequest, clarifiedIntent.getOperationType());
102         return workflowSessionContext.persist(result, WorkflowLifecycle.STEP_REVIEW, WorkflowLifecycle.STATUS_PLANNED);
103     }
104     
105     private ReadwriteSplittingRuleWorkflowRequest prepareSnapshot(final WorkflowContextSnapshot snapshot, final ReadwriteSplittingRuleWorkflowRequest request) {
106         ReadwriteSplittingRuleWorkflowRequest result = ReadwriteSplittingRuleWorkflowRequest.merge(snapshot.getRequest(), request);
107         planningSupport.prepareSnapshot(snapshot, ReadwriteSplittingFeatureDefinition.RULE_WORKFLOW_KIND, result, null,
108                 planningSupport.createOperationIntent(result, WorkflowLifecycle.OPERATION_CREATE),
109                 "Readwrite-splitting rule workflow plan.", INTERACTION_STEPS, VALIDATION_LAYERS);
110         snapshot.getResourceUriTemplates().add(ReadwriteSplittingFeatureDefinition.STORAGE_UNITS_RESOURCE_URI);
111         return result;
112     }
113     
114     private boolean ensurePlanningContext(final ReadwriteSplittingRuleWorkflowRequest request, final ClarifiedIntent clarifiedIntent, final WorkflowContextSnapshot snapshot) {
115         if (request.getDatabase().isEmpty()) {
116             clarifiedIntent.getClarificationMessages().add("Please provide logical database first.");
117             snapshot.getIssues().add(new WorkflowIssue(WorkflowIssueCode.DATABASE_REQUIRED, "error", WorkflowLifecycle.STEP_INTAKING,
118                     "Database is required before planning readwrite-splitting rule DistSQL.", "Provide the logical database name.", true, Map.of()));
119             snapshot.setStatus(WorkflowLifecycle.STATUS_CLARIFYING);
120             return false;
121         }
122         if (!planningSupport.ensureOptionalSupportedIdentifiers("database", List.of(request.getDatabase()), snapshot, WorkflowLifecycle.STEP_INTAKING)
123                 || !planningSupport.ensureOptionalSupportedIdentifiers(ReadwriteSplittingFeatureDefinition.RULE_FIELD, List.of(request.getRuleName()),
124                         snapshot, WorkflowLifecycle.STEP_INTAKING)
125                 || !planningSupport.ensureOptionalSupportedIdentifiers(ReadwriteSplittingFeatureDefinition.WRITE_STORAGE_UNIT_FIELD, List.of(request.getWriteStorageUnit()),
126                         snapshot, WorkflowLifecycle.STEP_INTAKING)
127                 || !planningSupport.ensureOptionalSupportedIdentifiers(ReadwriteSplittingFeatureDefinition.READ_STORAGE_UNITS_FIELD, request.getReadStorageUnits(),
128                         snapshot, WorkflowLifecycle.STEP_INTAKING)) {
129             snapshot.setStatus(WorkflowLifecycle.STATUS_FAILED);
130             return false;
131         }
132         addMissingInputs(request, clarifiedIntent, snapshot);
133         if (WorkflowLifecycle.OPERATION_DROP.equalsIgnoreCase(request.getOperationType())) {
134             return !request.getRuleName().isEmpty();
135         }
136         return !request.getRuleName().isEmpty() && !request.getWriteStorageUnit().isEmpty() && !request.getReadStorageUnits().isEmpty()
137                 && !request.getTransactionalReadQueryStrategy().isEmpty();
138     }
139     
140     private void addMissingInputs(final ReadwriteSplittingRuleWorkflowRequest request, final ClarifiedIntent clarifiedIntent, final WorkflowContextSnapshot snapshot) {
141         List<String> missingInputs = new LinkedList<>();
142         addMissingInput(missingInputs, request.getRuleName(), ReadwriteSplittingFeatureDefinition.RULE_FIELD);
143         if (!WorkflowLifecycle.OPERATION_DROP.equalsIgnoreCase(request.getOperationType())) {
144             addMissingInput(missingInputs, request.getWriteStorageUnit(), ReadwriteSplittingFeatureDefinition.WRITE_STORAGE_UNIT_FIELD);
145             if (request.getReadStorageUnits().isEmpty()) {
146                 missingInputs.add(ReadwriteSplittingFeatureDefinition.READ_STORAGE_UNITS_FIELD);
147             }
148             addMissingInput(missingInputs, request.getTransactionalReadQueryStrategy(), ReadwriteSplittingFeatureDefinition.TRANSACTIONAL_READ_QUERY_STRATEGY_FIELD);
149         }
150         if (!missingInputs.isEmpty()) {
151             clarifiedIntent.getClarificationMessages().add("Please provide readwrite-splitting rule name, storage units and strategy fields required by DistSQL.");
152             snapshot.getIssues().add(new WorkflowIssue(WorkflowIssueCode.RULE_INPUT_REQUIRED, "error", WorkflowLifecycle.STEP_INTAKING,
153                     "Readwrite-splitting rule DistSQL requires explicit structured inputs.", "Provide the missing fields instead of relying on inferred storage units.", true,
154                     Map.of("missing_inputs", missingInputs)));
155             snapshot.setStatus(WorkflowLifecycle.STATUS_CLARIFYING);
156         }
157     }
158     
159     private void addMissingInput(final Collection<String> missingInputs, final String value, final String fieldName) {
160         if (value.isEmpty()) {
161             missingInputs.add(fieldName);
162         }
163     }
164     
165     private boolean ensureLifecycleState(final ClarifiedIntent clarifiedIntent, final ReadwriteSplittingRuleWorkflowRequest request, final List<Map<String, Object>> rules,
166                                          final WorkflowContextSnapshot snapshot, final MCPFeatureQueryFacade queryFacade) {
167         boolean ruleExists = containsRule(rules, queryFacade, request.getDatabase(), request.getRuleName());
168         String operationType = clarifiedIntent.getOperationType().toLowerCase(Locale.ENGLISH);
169         if (WorkflowLifecycle.OPERATION_CREATE.equals(operationType) && ruleExists) {
170             snapshot.getIssues().add(new WorkflowIssue(WorkflowIssueCode.RULE_STATE_MISMATCH, "error", WorkflowLifecycle.STEP_DISCOVERING,
171                     String.format("Readwrite-splitting rule `%s` already exists.", request.getRuleName()), "Use alter instead of create.", false, Map.of("rule", request.getRuleName())));
172             return false;
173         }
174         if (WorkflowLifecycle.OPERATION_ALTER.equals(operationType) && !ruleExists) {
175             snapshot.getIssues().add(new WorkflowIssue(WorkflowIssueCode.RULE_STATE_MISMATCH, "error", WorkflowLifecycle.STEP_DISCOVERING,
176                     String.format("Readwrite-splitting rule `%s` does not exist.", request.getRuleName()), "Use create instead of alter.", false, Map.of("rule", request.getRuleName())));
177             return false;
178         }
179         if (WorkflowLifecycle.OPERATION_DROP.equals(operationType) && !ruleExists) {
180             snapshot.getIssues().add(new WorkflowIssue(WorkflowIssueCode.DROP_TARGET_RULE_NOT_FOUND, "error", WorkflowLifecycle.STEP_DISCOVERING,
181                     String.format("Readwrite-splitting rule `%s` does not exist.", request.getRuleName()), "Confirm target rule or skip the drop request.", false,
182                     Map.of("rule", request.getRuleName())));
183             return false;
184         }
185         return true;
186     }
187     
188     private boolean containsRule(final List<Map<String, Object>> rules, final MCPFeatureQueryFacade queryFacade, final String databaseName, final String ruleName) {
189         return rules.stream().anyMatch(each -> queryFacade.isSameIdentifier(databaseName, IdentifierScope.TABLE, ruleName, WorkflowRuleValueUtils.getRuleValue(each, "name")));
190     }
191     
192     private void planAlgorithms(final MCPFeatureQueryFacade queryFacade, final ReadwriteSplittingRuleWorkflowRequest request, final WorkflowContextSnapshot snapshot) {
193         List<AlgorithmCandidate> algorithmCandidates = algorithmRecommendationService.recommendLoadBalanceAlgorithms(
194                 request, inspectionService.queryLoadBalanceAlgorithmPlugins(queryFacade), snapshot.getIssues());
195         snapshot.getAlgorithmCandidates().addAll(algorithmCandidates);
196         if (!request.getLoadBalancerType().isEmpty() && !algorithmCandidates.isEmpty()) {
197             request.setLoadBalancerType(algorithmCandidates.getFirst().getAlgorithmType());
198         }
199     }
200     
201     private List<AlgorithmPropertyRequirement> findPropertyRequirements(final ReadwriteSplittingRuleWorkflowRequest request) {
202         return algorithmPropertyTemplateService.findRequirements(request.getLoadBalancerType(), request.getReadStorageUnits());
203     }
204     
205     private void addRuleArtifact(final WorkflowContextSnapshot snapshot, final ReadwriteSplittingRuleWorkflowRequest request, final String operationType) {
206         if (WorkflowLifecycle.OPERATION_DROP.equalsIgnoreCase(operationType)) {
207             snapshot.getRuleArtifacts().add(distSQLPlanningService.planDropRule(request.getRuleName()));
208         } else if (WorkflowLifecycle.OPERATION_ALTER.equalsIgnoreCase(operationType)) {
209             snapshot.getRuleArtifacts().add(distSQLPlanningService.planAlterRule(request));
210         } else {
211             snapshot.getRuleArtifacts().add(distSQLPlanningService.planCreateRule(request));
212         }
213     }
