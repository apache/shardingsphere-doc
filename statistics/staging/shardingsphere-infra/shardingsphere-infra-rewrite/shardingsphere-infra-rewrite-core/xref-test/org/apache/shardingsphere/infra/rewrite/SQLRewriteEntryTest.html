57  class SQLRewriteEntryTest {
58      
59      private final DatabaseType databaseType = TypedSPILoader.getService(DatabaseType.class, "FIXTURE");
60      
61      @Test
62      void assertRewriteForGenericSQLRewriteResult() {
63          ShardingSphereDatabase database = new ShardingSphereDatabase("foo_db", databaseType, mockResourceMetaData(),
64                  mock(RuleMetaData.class), Collections.singleton(new ShardingSphereSchema("test")));
65          SQLRewriteEntry sqlRewriteEntry = new SQLRewriteEntry(
66                  database, new RuleMetaData(Collections.singleton(new SQLTranslatorRule(new DefaultSQLTranslatorRuleConfigurationBuilder().build()))), new ConfigurationProperties(new Properties()));
67          RouteContext routeContext = new RouteContext();
68          GenericSQLRewriteResult sqlRewriteResult = (GenericSQLRewriteResult) sqlRewriteEntry.rewrite(createQueryContext(), routeContext);
69          assertThat(sqlRewriteResult.getSqlRewriteUnit().getSql(), is("SELECT ?"));
70          assertThat(sqlRewriteResult.getSqlRewriteUnit().getParameters(), is(Collections.singletonList(1)));
71      }
72      
73      private QueryContext createQueryContext() {
74          QueryContext result = mock(QueryContext.class, RETURNS_DEEP_STUBS);
75          when(result.getSql()).thenReturn("SELECT ?");
76          when(result.getParameters()).thenReturn(Collections.singletonList(1));
77          SQLStatementContext sqlStatementContext = mock(SQLStatementContext.class, RETURNS_DEEP_STUBS);
78          when(sqlStatementContext.getTablesContext().getDatabaseNames()).thenReturn(Collections.emptyList());
79          SQLStatement sqlStatement = mock(SQLStatement.class);
80          when(sqlStatement.getAttributes()).thenReturn(new SQLStatementAttributes());
81          when(sqlStatementContext.getSqlStatement()).thenReturn(sqlStatement);
82          when(sqlStatementContext.getSqlStatement().getDatabaseType()).thenReturn(databaseType);
83          when(result.getSqlStatementContext()).thenReturn(sqlStatementContext);
84          when(result.getHintValueContext()).thenReturn(new HintValueContext());
85          when(result.getMetaData().getProps()).thenReturn(new ConfigurationProperties(new Properties()));
86          return result;
87      }
88      
89      @Test
90      void assertRewriteForRouteSQLRewriteResult() {
91          ShardingSphereDatabase database = new ShardingSphereDatabase(
92                  "foo_db", databaseType, mockResourceMetaData(), mock(RuleMetaData.class), Collections.singleton(new ShardingSphereSchema("test")));
93          SQLTranslatorRule sqlTranslatorRule = mock(SQLTranslatorRule.class);
94          when(sqlTranslatorRule.translate(any(), any(), any(), any(), any(), any())).thenReturn(Optional.of(new SQLTranslatorContext("", Collections.emptyList())));
95          SQLRewriteEntry sqlRewriteEntry = new SQLRewriteEntry(database, new RuleMetaData(Collections.singleton(sqlTranslatorRule)), new ConfigurationProperties(new Properties()));
96          RouteContext routeContext = new RouteContext();
97          RouteUnit firstRouteUnit = mock(RouteUnit.class);
98          when(firstRouteUnit.getDataSourceMapper()).thenReturn(new RouteMapper("ds", "ds_0"));
99          RouteUnit secondRouteUnit = mock(RouteUnit.class);
100         when(secondRouteUnit.getDataSourceMapper()).thenReturn(new RouteMapper("ds", "ds_1"));
101         routeContext.getRouteUnits().addAll(Arrays.asList(firstRouteUnit, secondRouteUnit));
102         RouteSQLRewriteResult sqlRewriteResult = (RouteSQLRewriteResult) sqlRewriteEntry.rewrite(createQueryContext(), routeContext);
103         assertThat(sqlRewriteResult.getSqlRewriteUnits().size(), is(2));
104     }
105     
106     private ResourceMetaData mockResourceMetaData() {
107         Map<String, StorageUnit> storageUnits = new LinkedHashMap<>(2, 1F);
108         StorageUnit storageUnit1 = mock(StorageUnit.class);
109         when(storageUnit1.getStorageType()).thenReturn(databaseType);
110         StorageUnit storageUnit2 = mock(StorageUnit.class);
111         when(storageUnit2.getStorageType()).thenReturn(mock());
112         storageUnits.put("ds_0", storageUnit1);
113         storageUnits.put("ds_1", storageUnit2);
114         ResourceMetaData result = mock(ResourceMetaData.class);
115         when(result.getStorageUnits()).thenReturn(storageUnits);
116         return result;
117     }
