46  @ExtendWith(AutoMockExtension.class)
47  @StaticMockSettings({ColumnExtractor.class, ShadowExtractor.class})
48  class ShadowSelectStatementDataSourceMappingsRetrieverTest {
49      
50      @Test
51      void assertRetrieveWithColumnOwner() {
52          SelectStatementContext sqlStatementContext = mock(SelectStatementContext.class, RETURNS_DEEP_STUBS);
53          WhereSegment whereSegment = mock(WhereSegment.class);
54          ExpressionSegment expressionSegment = mock(ExpressionSegment.class);
55          when(whereSegment.getExpr()).thenReturn(expressionSegment);
56          when(sqlStatementContext.getWhereSegments()).thenReturn(Arrays.asList(whereSegment, mock(WhereSegment.class, RETURNS_DEEP_STUBS)));
57          ColumnSegment columnSegment = mock(ColumnSegment.class, RETURNS_DEEP_STUBS);
58          when(columnSegment.getColumnBoundInfo().getOriginalTable().getValue()).thenReturn("foo_tbl");
59          OwnerSegment ownerSegment = new OwnerSegment(0, 0, new IdentifierValue("foo"));
60          ownerSegment.setTableBoundInfo(new TableSegmentBoundInfo(new IdentifierValue("foo_db"), new IdentifierValue("foo_schema")));
61          when(columnSegment.getOwner()).thenReturn(Optional.of(ownerSegment));
62          when(ColumnExtractor.extract(expressionSegment)).thenReturn(Collections.singleton(columnSegment));
63          when(ShadowExtractor.extractValues(expressionSegment, Collections.singletonList("foo"))).thenReturn(Optional.of(Collections.singleton("foo")));
64          when(sqlStatementContext.getTablesContext().getTableNames()).thenReturn(Collections.singleton("foo_tbl"));
65          ShadowSelectStatementDataSourceMappingsRetriever retriever = new ShadowSelectStatementDataSourceMappingsRetriever(sqlStatementContext, Collections.singletonList("foo"));
66          Collection<ShadowColumnCondition> actual = retriever.getShadowColumnConditions("foo_col");
67          assertThat(actual.size(), is(1));
68          ShadowColumnCondition actualCondition = actual.iterator().next();
69          assertThat(actualCondition.getTable(), is("foo_tbl"));
70          assertThat(actualCondition.getColumn(), is("foo_col"));
71          assertThat(actualCondition.getValues(), is(Collections.singleton("foo")));
72          
73      }
74      
75      @Test
76      void assertRetrieveWithoutColumnOwner() {
77          SelectStatementContext sqlStatementContext = mock(SelectStatementContext.class, RETURNS_DEEP_STUBS);
78          WhereSegment whereSegment = mock(WhereSegment.class);
79          ExpressionSegment expressionSegment = mock(ExpressionSegment.class);
80          when(whereSegment.getExpr()).thenReturn(expressionSegment);
81          when(sqlStatementContext.getWhereSegments()).thenReturn(Arrays.asList(whereSegment, mock(WhereSegment.class, RETURNS_DEEP_STUBS)));
82          ColumnSegment columnSegment = mock(ColumnSegment.class, RETURNS_DEEP_STUBS);
83          when(columnSegment.getColumnBoundInfo().getOriginalTable().getValue()).thenReturn("foo_tbl");
84          when(columnSegment.getOwner()).thenReturn(Optional.empty());
85          when(ColumnExtractor.extract(expressionSegment)).thenReturn(Collections.singleton(columnSegment));
86          when(ShadowExtractor.extractValues(expressionSegment, Collections.singletonList("foo"))).thenReturn(Optional.of(Collections.singleton("foo")));
87          when(sqlStatementContext.getTablesContext().getTableNames()).thenReturn(Collections.singleton("foo_tbl"));
88          ShadowSelectStatementDataSourceMappingsRetriever retriever = new ShadowSelectStatementDataSourceMappingsRetriever(sqlStatementContext, Collections.singletonList("foo"));
89          Collection<ShadowColumnCondition> actual = retriever.getShadowColumnConditions("foo_col");
90          assertThat(actual.size(), is(1));
91          ShadowColumnCondition actualCondition = actual.iterator().next();
92          assertThat(actualCondition.getTable(), is("foo_tbl"));
93          assertThat(actualCondition.getColumn(), is("foo_col"));
94          assertThat(actualCondition.getValues(), is(Collections.singleton("foo")));
95      }
