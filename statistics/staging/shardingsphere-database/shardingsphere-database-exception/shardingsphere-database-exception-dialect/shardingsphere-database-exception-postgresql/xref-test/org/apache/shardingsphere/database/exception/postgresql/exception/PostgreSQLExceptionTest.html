21  import org.apache.shardingsphere.database.exception.postgresql.vendor.PostgreSQLVendorError;
22  import org.apache.shardingsphere.infra.exception.external.sql.sqlstate.SQLState;
23  import org.apache.shardingsphere.infra.exception.external.sql.vendor.VendorError;
24  import org.junit.jupiter.api.Test;
25  import org.junit.jupiter.params.ParameterizedTest;
26  import org.junit.jupiter.params.provider.Arguments;
27  import org.junit.jupiter.params.provider.MethodSource;
28  import org.mockito.internal.configuration.plugins.Plugins;
29  
30  import java.util.stream.Stream;
31  
32  import static org.hamcrest.MatcherAssert.assertThat;
33  import static org.hamcrest.Matchers.is;
34  import static org.junit.jupiter.api.Assertions.assertNull;
35  import static org.mockito.Mockito.mock;
36  import static org.mockito.Mockito.when;
37  
38  class PostgreSQLExceptionTest {
39      
40      @Test
41      void assertConstructWithMessageAndState() {
42          PostgreSQLException actual = new PostgreSQLException("unexpected", "XX000");
43          assertThat(actual.getMessage(), is("unexpected"));
44          assertThat(actual.getSQLState(), is("XX000"));
45          assertNull(actual.getServerErrorMessage());
46      }
47      
48      @Test
49      void assertConstructWithServerErrorMessage() {
50          ServerErrorMessage serverErrorMessage = new ServerErrorMessage("FATAL", PostgreSQLVendorError.INVALID_CATALOG_NAME, "logic_db");
51          PostgreSQLException actual = new PostgreSQLException(serverErrorMessage);
52          assertThat(actual.getMessage(), is(serverErrorMessage.toString()));
53          assertThat(actual.getSQLState(), is(serverErrorMessage.getSqlState()));
54          assertThat(actual.getServerErrorMessage(), is(serverErrorMessage));
55      }
56      
57      @Test
58      void assertConstructServerErrorMessage() {
59          ServerErrorMessage actual = new ServerErrorMessage("FATAL", PostgreSQLVendorError.INVALID_CATALOG_NAME, "logic_db");
60          assertThat(actual.getSeverity(), is("FATAL"));
61          assertThat(actual.getSqlState(), is(PostgreSQLVendorError.INVALID_CATALOG_NAME.getSqlState().getValue()));
62          assertThat(actual.getMessage(), is(String.format(PostgreSQLVendorError.INVALID_CATALOG_NAME.getReason(), "logic_db")));
63      }
64      
65      @ParameterizedTest(name = "{0}")
66      @MethodSource("toStringArguments")
67      void assertToString(final String name, final ServerErrorMessage serverErrorMessage, final String expected) {
68          assertThat(serverErrorMessage.toString(), is(expected));
69      }
70      
71      private static Stream<Arguments> toStringArguments() throws ReflectiveOperationException {
72          return Stream.of(
73                  Arguments.of("all_fields_present", createServerErrorMessage("FATAL", "3D000", "logic_db"),
74                          "FATAL: database \"logic_db\" does not exist\n  Server SQLState: 3D000"),
75                  Arguments.of("null_severity", createServerErrorMessage(null, "3D000", "logic_db"),
76                          "database \"logic_db\" does not exist\n  Server SQLState: 3D000"),
77                  Arguments.of("null_message", createServerErrorMessageWithNullMessage(), "FATAL: \n  Server SQLState: 3D000"),
78                  Arguments.of("null_sql_state", createServerErrorMessage("FATAL", null, "logic_db"),
79                          "FATAL: database \"logic_db\" does not exist"));
80      }
81      
82      private static ServerErrorMessage createServerErrorMessage(final String severity, final String sqlStateValue, final Object... reasonArgs) {
83          SQLState sqlState = mock(SQLState.class);
84          when(sqlState.getValue()).thenReturn(sqlStateValue);
85          VendorError vendorError = mock(VendorError.class);
86          when(vendorError.getSqlState()).thenReturn(sqlState);
87          when(vendorError.getReason()).thenReturn("database \"%s\" does not exist");
88          return new ServerErrorMessage(severity, vendorError, reasonArgs);
89      }
90      
91      private static ServerErrorMessage createServerErrorMessageWithNullMessage() throws ReflectiveOperationException {
92          ServerErrorMessage result = createServerErrorMessage("FATAL", "3D000", "logic_db");
93          Plugins.getMemberAccessor().set(ServerErrorMessage.class.getDeclaredField("message"), result, null);
94          return result;
95      }
96  }
