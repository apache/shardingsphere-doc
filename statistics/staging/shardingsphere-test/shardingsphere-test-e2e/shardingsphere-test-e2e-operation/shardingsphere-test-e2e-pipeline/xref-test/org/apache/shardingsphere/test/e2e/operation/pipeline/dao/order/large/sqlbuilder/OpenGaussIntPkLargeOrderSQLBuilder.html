111     @Override
112     public List<Object[]> generateInsertData(final KeyGenerateAlgorithm keyGenerateAlgorithm, final int recordCount) {
113         List<Object[]> result = new ArrayList<>(recordCount);
114         for (int i = 0; i < recordCount; i++) {
115             Object orderId = keyGenerateAlgorithm.generateKeys(mock(AlgorithmSQLContext.class), 1).iterator().next();
116             byte[] bytesValue = {Byte.MIN_VALUE, -1, 0, 1, Byte.MAX_VALUE};
117             Object[] params = new Object[]{
118                     orderId, PipelineCaseHelper.generateInt(0, 1000), "'status'" + i,
119                     PipelineCaseHelper.generateInt(-1000, 9999), PipelineCaseHelper.generateInt(0, 100),
120                     PipelineCaseHelper.generateFloat(), PipelineCaseHelper.generateDouble(), BigDecimal.valueOf(PipelineCaseHelper.generateDouble()), false,
121                     PipelineCaseHelper.generateString(6), "texts", bytesValue, bytesValue,
122                     LocalDate.now(), LocalTime.now(), "2001-10-01", Timestamp.valueOf(LocalDateTime.now()), OffsetDateTime.now(), "0 years 0 mons 1 days 2 hours 3 mins 4 secs", "{1, 2, 3}",
123                     PipelineCaseHelper.generateJsonString(8, false), PipelineCaseHelper.generateJsonString(8, true),
124                     UUID.randomUUID().toString(), DigestUtils.md5Hex(orderId.toString()),
125                     "'rat' 'sat'", "tsquery", "0000",
126                     "[1,1000)", "[2020-01-02,2021-01-01)", "[2020-01-01 00:00:00,2021-01-01 00:00:00)",
127                     "1 years 1 mons 10 days -06:00:00", "2000-01-02 00:00:00+00",
128                     "(1.0,1.0)", "[(0.0,0.0),(2.0,2.0)]", "(3.0,3.0),(1.0,1.0)", "<(5.0,5.0),5.0>", "1111",
129                     "192.168.0.0/16", "192.168.1.1", "08:00:2b:01:02:03",
130                     "\\x484c4c00000000002b05000000000000000000000000000000000000", 999
131             };
132             result.add(params);
133         }
134         return result;
135     }
