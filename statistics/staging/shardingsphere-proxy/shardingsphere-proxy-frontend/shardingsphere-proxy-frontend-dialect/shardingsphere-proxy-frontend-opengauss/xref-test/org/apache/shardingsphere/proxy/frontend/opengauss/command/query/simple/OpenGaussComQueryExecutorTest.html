75  @ExtendWith(AutoMockExtension.class)
76  @StaticMockSettings({ProxySQLComQueryParser.class, ProxyBackendHandlerFactory.class})
77  class OpenGaussComQueryExecutorTest {
78      
79      @Mock
80      private PortalContext portalContext;
81      
82      @Mock
83      private ProxyBackendHandler proxyBackendHandler;
84      
85      @Mock(answer = Answers.RETURNS_DEEP_STUBS)
86      private PostgreSQLComQueryPacket queryPacket;
87      
88      private DatabaseType databaseType;
89      
90      private OpenGaussComQueryExecutor queryExecutor;
91      
92      @BeforeEach
93      void setUp() throws SQLException {
94          databaseType = TypedSPILoader.getService(DatabaseType.class, "openGauss");
95          ConnectionSession connectionSession = mock(ConnectionSession.class);
96          when(queryPacket.getSQL()).thenReturn("sql");
97          when(queryPacket.getHintValueContext()).thenReturn(new HintValueContext());
98          SQLStatement sqlStatement = mock(SQLStatement.class);
99          when(ProxySQLComQueryParser.parse("sql", databaseType, connectionSession)).thenReturn(sqlStatement);
100         when(ProxyBackendHandlerFactory.newInstance(databaseType, "sql", sqlStatement, connectionSession, queryPacket.getHintValueContext())).thenReturn(proxyBackendHandler);
101         queryExecutor = new OpenGaussComQueryExecutor(portalContext, queryPacket, connectionSession);
102     }
103     
104     @Test
105     void assertExecuteQueryReturnsRowDescription() throws SQLException, ReflectiveOperationException {
106         QueryHeader queryHeader = new QueryHeader("schema", "table", "label", "column", 1, "type", 2, 3, true, true, true, true);
107         QueryHeader compositeQueryHeader = new QueryHeader("schema", "table", "composite_label", "composite_column", Types.STRUCT, "record_type", 4, 5, true, true, true, true);
108         compositeQueryHeader.getProtocolAttributes().put(PostgreSQLQueryHeaderBuilder.TYPE_OID, 2249);
109         QueryResponseHeader queryResponseHeader = mock(QueryResponseHeader.class);
110         when(queryResponseHeader.getQueryHeaders()).thenReturn(Arrays.asList(queryHeader, compositeQueryHeader));
111         when(proxyBackendHandler.execute()).thenReturn(queryResponseHeader);
112         Collection<DatabasePacket> actualPackets = queryExecutor.execute();
113         List<DatabasePacket> actualPacketList = new LinkedList<>(actualPackets);
114         assertThat(actualPacketList.size(), is(1));
115         PostgreSQLRowDescriptionPacket actualPacket = (PostgreSQLRowDescriptionPacket) actualPacketList.iterator().next();
116         assertThat(queryExecutor.getResponseType(), is(ResponseType.QUERY));
117         Collection<?> columnDescriptions = (Collection<?>) Plugins.getMemberAccessor()
118                 .get(PostgreSQLRowDescriptionPacket.class.getDeclaredField("columnDescriptions"), actualPacket);
119         List<?> actualColumns = new LinkedList<>(columnDescriptions);
120         PostgreSQLColumnDescription actualColumn = (PostgreSQLColumnDescription) actualColumns.get(0);
121         assertThat(actualColumn.getColumnName(), is("label"));
122         assertThat(actualColumn.getColumnIndex(), is(1));
123         PostgreSQLColumnDescription actualCompositeColumn = (PostgreSQLColumnDescription) actualColumns.get(1);
124         assertThat(actualCompositeColumn.getColumnName(), is("composite_label"));
125         assertThat(actualCompositeColumn.getColumnIndex(), is(2));
126         assertThat(actualCompositeColumn.getTypeOID(), is(2249));
127     }
128     
129     @Test
130     void assertExecuteUpdateWithCommitClosesPortals() throws SQLException {
131         UpdateResponseHeader updateResponseHeader = new UpdateResponseHeader(new CommitStatement(databaseType));
132         when(proxyBackendHandler.execute()).thenReturn(updateResponseHeader);
133         Collection<DatabasePacket> actualPackets = queryExecutor.execute();
134         assertThat(new LinkedList<>(actualPackets).getFirst(), is(isA(PostgreSQLCommandCompletePacket.class)));
135         assertThat(queryExecutor.getResponseType(), is(ResponseType.UPDATE));
136         verify(portalContext).closeAll();
137     }
138

142         when(proxyBackendHandler.execute()).thenReturn(updateResponseHeader);
143         Collection<DatabasePacket> actualPackets = queryExecutor.execute();
144         assertThat(new LinkedList<>(actualPackets).getFirst(), is(isA(PostgreSQLCommandCompletePacket.class)));
145         verify(portalContext).closeAll();

148     @Test
149     void assertExecuteUpdateWithSetStatementReturnsParameterStatus() throws SQLException, ReflectiveOperationException {
150         VariableAssignSegment assignWithValue = new VariableAssignSegment(0, 0, new VariableSegment(0, 0, "timezone"), "'UTC'");
151         VariableAssignSegment assignWithNull = new VariableAssignSegment(0, 0, new VariableSegment(0, 0, "work_mem"), null);
152         SetStatement setStatement = new SetStatement(databaseType, Arrays.asList(assignWithValue, assignWithNull));
153         when(proxyBackendHandler.execute()).thenReturn(new UpdateResponseHeader(setStatement));
154         Collection<DatabasePacket> actualPackets = queryExecutor.execute();
155         List<DatabasePacket> actualPacketList = new LinkedList<>(actualPackets);
156         PostgreSQLCommandCompletePacket commandCompletePacket = (PostgreSQLCommandCompletePacket) actualPacketList.get(0);
157         PostgreSQLParameterStatusPacket parameterStatusPacket = (PostgreSQLParameterStatusPacket) actualPacketList.get(1);
158         PostgreSQLParameterStatusPacket parameterStatusPacketWithNull = (PostgreSQLParameterStatusPacket) actualPacketList.get(2);
159         assertThat(commandCompletePacket, is(isA(PostgreSQLCommandCompletePacket.class)));
160         Object actualValue = Plugins.getMemberAccessor().get(PostgreSQLParameterStatusPacket.class.getDeclaredField("value"), parameterStatusPacket);
161         assertThat(actualValue, is("UTC"));
162         Object actualNullValue = Plugins.getMemberAccessor().get(PostgreSQLParameterStatusPacket.class.getDeclaredField("value"), parameterStatusPacketWithNull);
163         assertNull(actualNullValue);
164     }

170         assertThat(new LinkedList<>(actualPackets).getFirst(), is(isA(PostgreSQLEmptyQueryResponsePacket.class)));
171         assertThat(queryExecutor.getResponseType(), is(ResponseType.UPDATE));
