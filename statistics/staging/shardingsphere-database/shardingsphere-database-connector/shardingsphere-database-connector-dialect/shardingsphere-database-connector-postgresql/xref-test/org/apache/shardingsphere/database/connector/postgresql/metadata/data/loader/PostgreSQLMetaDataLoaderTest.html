75              + " JOIN information_schema.key_column_usage kc ON kc.table_schema = tc.table_schema AND kc.table_name = tc.table_name AND kc.constraint_name = tc.constraint_name"
76              + " WHERE tc.constraint_type = 'PRIMARY KEY' AND kc.ordinal_position IS NOT NULL AND kc.table_schema IN ('public')";

80      private static final String ADVANCE_INDEX_META_DATA_SQL =
81              "SELECT idx.relname as index_name, insp.nspname as index_schema, tbl.relname as table_name, att.attname AS column_name, pgi.indisunique as is_unique"
82                      + " FROM pg_index pgi JOIN pg_class idx ON idx.oid = pgi.indexrelid JOIN pg_namespace insp ON insp.oid = idx.relnamespace JOIN pg_class tbl ON tbl.oid = pgi.indrelid"
83                      + " JOIN pg_namespace tnsp ON tnsp.oid = tbl.relnamespace JOIN generate_subscripts(pgi.indkey, 1) AS index_position(position) ON TRUE"
84                      + " JOIN pg_attribute att ON att.attrelid = tbl.oid AND att.attnum = pgi.indkey[index_position.position] WHERE tnsp.nspname IN ('public')"
85                      + " ORDER BY insp.nspname, idx.relname, index_position.position";

229     private void assertTableMetaDataMap(final Collection<SchemaMetaData> schemaMetaDataList) {
230         assertThat(schemaMetaDataList.size(), is(1));
231         TableMetaData actualTableMetaData = schemaMetaDataList.iterator().next().getTables().iterator().next();

237         Iterator<IndexMetaData> indexesIterator = actualTableMetaData.getIndexes().iterator();
