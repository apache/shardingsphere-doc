18  package org.apache.shardingsphere.database.connector.core.metadata.identifier;
19  
20  import org.apache.shardingsphere.database.connector.core.metadata.database.enums.QuoteCharacter;
21  import org.junit.jupiter.api.Test;
22  import org.junit.jupiter.params.ParameterizedTest;
23  import org.junit.jupiter.params.provider.Arguments;
24  import org.junit.jupiter.params.provider.MethodSource;
25  
26  import java.util.Locale;
27  import java.util.stream.Stream;
28  
29  import static org.hamcrest.MatcherAssert.assertThat;
30  import static org.hamcrest.Matchers.is;
31  import static org.junit.jupiter.api.Assertions.assertTrue;
32  
33  class IdentifierCasePolicyTest {
34      
35      private final IdentifierCasePolicy policy = new IdentifierCasePolicy(LookupMode.EXACT, LookupMode.NORMALIZED,
36              each -> each, each -> each.toLowerCase(Locale.ENGLISH), each -> each.toUpperCase(Locale.ENGLISH), each -> each.equals(each.toLowerCase(Locale.ENGLISH)));
37      
38      @Test
39      void assertGetLookupModeWithQuotedIdentifier() {
40          assertThat(policy.getLookupMode(QuoteCharacter.QUOTE), is(LookupMode.EXACT));
41      }
42      
43      @Test
44      void assertGetLookupModeWithUnquotedIdentifier() {
45          assertThat(policy.getLookupMode(QuoteCharacter.NONE), is(LookupMode.NORMALIZED));
46      }
47      
48      @Test
49      void assertNormalizeDefinitionWithQuotedIdentifier() {
50          assertThat(policy.normalizeForDefinition("Foo", QuoteCharacter.QUOTE), is("Foo"));
51      }
52      
53      @Test
54      void assertNormalizeDefinitionWithUnquotedIdentifier() {
55          assertThat(policy.normalizeForDefinition("Foo", QuoteCharacter.NONE), is("foo"));
56      }
57      
58      @Test
59      void assertNormalizeLookup() {
60          assertThat(policy.normalizeForLookup("Foo"), is("FOO"));
61      }
62      
63      @Test
64      void assertMatchesWithQuotedNormalizedLookup() {
65          IdentifierCasePolicy actual = new IdentifierCasePolicy(LookupMode.NORMALIZED, LookupMode.NORMALIZED,
66                  each -> each, each -> each.toLowerCase(Locale.ENGLISH), each -> each.toLowerCase(Locale.ENGLISH), each -> each.equals(each.toLowerCase(Locale.ENGLISH)));
67          assertTrue(actual.matches("t_mask", "T_MASK", QuoteCharacter.BACK_QUOTE));
68      }
69      
70      @ParameterizedTest(name = "{0}")
71      @MethodSource("matchesArguments")
72      void assertMatches(final String name, final String storedName, final String actualIdentifier, final QuoteCharacter quoteCharacter, final boolean expected) {
73          assertThat(policy.matches(storedName, actualIdentifier, quoteCharacter), is(expected));
74      }
75      
76      private static Stream<Arguments> matchesArguments() {
77          return Stream.of(
78                  Arguments.of("quoted_match", "Foo", "Foo", QuoteCharacter.QUOTE, true),
79                  Arguments.of("quoted_mismatch", "Foo", "foo", QuoteCharacter.QUOTE, false),
80                  Arguments.of("unquoted_match", "foo", "FOO", QuoteCharacter.NONE, true),
81                  Arguments.of("unquoted_non_default_stored_name", "Foo", "FOO", QuoteCharacter.NONE, false),
82                  Arguments.of("unquoted_different_actual_identifier", "foo", "bar", QuoteCharacter.NONE, false));
83      }
84  }
