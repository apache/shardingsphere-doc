107     @BeforeEach
108     void setUp() throws SQLException {
109         when(queryPacket.getSQL()).thenReturn("select 1");
110         when(queryPacket.getHintValueContext()).thenReturn(new HintValueContext());
111         SQLStatement sqlStatement = new SQLStatement(DATABASE_TYPE);
112         when(ProxySQLComQueryParser.parse(queryPacket.getSQL(), DATABASE_TYPE, connectionSession)).thenReturn(sqlStatement);
113         when(ProxyBackendHandlerFactory.newInstance(DATABASE_TYPE, queryPacket.getSQL(), sqlStatement, connectionSession, queryPacket.getHintValueContext())).thenReturn(proxyBackendHandler);
114         queryExecutor = new PostgreSQLComQueryExecutor(portalContext, queryPacket, connectionSession);
115     }

117     @Test
118     void assertExecuteQueryWithColumnDescription() throws SQLException, ReflectiveOperationException {
119         QueryResponseHeader queryResponseHeader = mock(QueryResponseHeader.class);
120         QueryHeader queryHeader = new QueryHeader("schema", "table", "label", "column", Types.STRUCT, "record_type", 2, 3, true, true, true, true);
121         queryHeader.getProtocolAttributes().put(PostgreSQLQueryHeaderBuilder.TYPE_OID, 2249);
122         when(queryResponseHeader.getQueryHeaders()).thenReturn(Collections.singletonList(queryHeader));
123         when(proxyBackendHandler.execute()).thenReturn(queryResponseHeader);
124         Collection<DatabasePacket> actual = queryExecutor.execute();
125         PostgreSQLRowDescriptionPacket rowDescriptionPacket = (PostgreSQLRowDescriptionPacket) actual.iterator().next();
126         Collection<PostgreSQLColumnDescription> columnDescriptions = getColumnDescriptions(rowDescriptionPacket);
127         PostgreSQLColumnDescription columnDescription = columnDescriptions.iterator().next();
128         assertThat(actual.size(), is(1));
129         assertThat(columnDescriptions.size(), is(1));
130         assertThat(columnDescription.getColumnName(), is("label"));
131         assertThat(columnDescription.getColumnIndex(), is(1));
132         assertThat(columnDescription.getColumnLength(), is(2));
133         assertThat(columnDescription.getTypeOID(), is(2249));
134         assertThat(queryExecutor.getResponseType(), is(ResponseType.QUERY));
135     }

149     @ParameterizedTest(name = "{0}")
150     @MethodSource("updateScenarios")
151     void assertExecuteUpdate(final String testName, final SQLStatement sqlStatement, final Class<? extends DatabasePacket> expectedPacketType,
152                              final String expectedTag, final boolean expectCloseAll) throws SQLException, ReflectiveOperationException {
153         when(proxyBackendHandler.execute()).thenReturn(new UpdateResponseHeader(sqlStatement));
154         Collection<DatabasePacket> actual = queryExecutor.execute();
155         DatabasePacket packet = actual.iterator().next();
156         assertThat(actual.size(), is(1));
157         assertThat(packet, isA(expectedPacketType));
158         if (packet instanceof PostgreSQLCommandCompletePacket && null != expectedTag) {
159             assertThat(getSqlCommand((PostgreSQLCommandCompletePacket) packet), is(expectedTag));
160         }
161         assertThat(queryExecutor.getResponseType(), is(ResponseType.UPDATE));
162         if (expectCloseAll) {
163             verify(portalContext).closeAll();
164         } else {
165             verify(portalContext, never()).closeAll();
166         }
167     }

169     @Test
170     void assertExecuteUpdateWithSetStatement() throws SQLException, ReflectiveOperationException {
171         VariableSegment searchPathVariable = new VariableSegment(0, 0, "search_path");
172         VariableSegment workMemVariable = new VariableSegment(1, 1, "work_mem");
173         List<VariableAssignSegment> variableAssigns = new ArrayList<>(2);
174         variableAssigns.add(new VariableAssignSegment(0, 0, searchPathVariable, null));
175         variableAssigns.add(new VariableAssignSegment(1, 1, workMemVariable, "'64MB'"));
176         UpdateResponseHeader updateResponseHeader = new UpdateResponseHeader(new SetStatement(DATABASE_TYPE, variableAssigns));
177         when(proxyBackendHandler.execute()).thenReturn(updateResponseHeader);
178         Collection<DatabasePacket> actual = queryExecutor.execute();
179         Iterator<DatabasePacket> iterator = actual.iterator();
180         PostgreSQLCommandCompletePacket commandCompletePacket = (PostgreSQLCommandCompletePacket) iterator.next();
181         PostgreSQLParameterStatusPacket firstStatusPacket = (PostgreSQLParameterStatusPacket) iterator.next();
182         assertThat(actual.size(), is(3));
183         assertThat(getSqlCommand(commandCompletePacket), is("SET"));
184         assertThat(getParameterStatusKey(firstStatusPacket), is("search_path"));
185         assertNull(getParameterStatusValue(firstStatusPacket));
186         PostgreSQLParameterStatusPacket secondStatusPacket = (PostgreSQLParameterStatusPacket) iterator.next();
187         assertThat(getParameterStatusKey(secondStatusPacket), is("work_mem"));
188         assertThat(getParameterStatusValue(secondStatusPacket), is("64MB"));
189         assertThat(queryExecutor.getResponseType(), is(ResponseType.UPDATE));
190         verify(portalContext, never()).closeAll();
191     }
