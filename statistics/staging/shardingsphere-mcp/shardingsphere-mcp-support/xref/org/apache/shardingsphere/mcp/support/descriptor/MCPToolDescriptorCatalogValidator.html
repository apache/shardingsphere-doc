38  /**
39   * Validator for MCP tool descriptors and their runtime bindings.
40   */
41  @NoArgsConstructor(access = AccessLevel.PRIVATE)
42  public final class MCPToolDescriptorCatalogValidator {
43      
44      private static final Collection<String> SECRET_WORKFLOW_OUTPUT_FIELDS = List.of("masked_property_preview", "secret_reference_summary");
45      
46      /**
47       * Validate tool descriptors in one descriptor catalog.
48       *
49       * @param catalog descriptor catalog
50       */
51      public static void validate(final MCPDescriptorCatalog catalog) {
52          Collection<MCPToolDescriptor> descriptors = catalog.getProtocolDescriptors().getToolDescriptors();
53          Collection<MCPToolRuntimeDescriptor> runtimeDescriptors = catalog.getShardingSphereDescriptors().getToolRuntimeDescriptors();
54          validateToolRuntimeDescriptors(runtimeDescriptors);
55          Map<String, MCPToolDescriptor> registered = new LinkedHashMap<>(descriptors.size(), 1F);
56          Map<String, MCPToolRuntimeDescriptor> runtimes = runtimeDescriptors.stream()
57                  .collect(Collectors.toMap(MCPToolRuntimeDescriptor::getToolName, each -> each));
58          Set<String> resourceIdentifiers = catalog.getProtocolDescriptors().getAllResourceDescriptors().stream().map(MCPResourceDescriptor::getUriTemplate).collect(Collectors.toSet());
59          Set<String> shardingSphereResourceIdentifiers = catalog.getShardingSphereDescriptors().getResourceMetadata().stream()
60                  .map(ShardingSphereMCPResourceMetadata::getUriTemplate).collect(Collectors.toSet());
61          for (MCPToolDescriptor each : descriptors) {
62              ShardingSpherePreconditions.checkState(null == registered.putIfAbsent(each.getName(), each),
63                      () -> new IllegalStateException(String.format("Duplicate MCP tool descriptor `%s`.", each.getName())));
64              MCPToolInputSchemaValidator.validate(each);
65              MCPToolOutputSchemaValidator.validate(each);
66              validateToolDescriptorContract(each, runtimes.get(each.getName()));
67              validateDestructiveToolDescriptor(each);
68              validateNonDestructiveExecutionMode(each);
69              validateRelatedResourceUris(each, resourceIdentifiers, shardingSphereResourceIdentifiers);
70          }
71          validatePlanningToolRuntimeDescriptors(registered, runtimeDescriptors);
72      }
73      
74      private static void validateToolRuntimeDescriptors(final Collection<MCPToolRuntimeDescriptor> runtimeDescriptors) {
75          for (MCPToolRuntimeDescriptor each : runtimeDescriptors) {
76              ShardingSpherePreconditions.checkState(each.getWorkflowRole().isBlank() || "plan".equals(each.getWorkflowRole()),
77                      () -> new IllegalStateException(String.format("Tool `%s` runtime workflowRole `%s` is unsupported.", each.getToolName(), each.getWorkflowRole())));
78          }
79      }
80      
81      private static void validateToolDescriptorContract(final MCPToolDescriptor descriptor, final MCPToolRuntimeDescriptor runtimeDescriptor) {
82          if (CoreToolNames.SEARCH_METADATA.equals(descriptor.getName())) {
83              validateSearchMetadataDescriptor(descriptor);
84              return;
85          }
86          if (CoreToolNames.EXECUTE_QUERY.equals(descriptor.getName())) {
87              validateExecuteQueryDescriptor(descriptor);
88              return;
89          }
90          if (CoreToolNames.EXECUTE_EXPLAIN_QUERY.equals(descriptor.getName())) {
91              validateExecuteExplainDescriptor(descriptor);
92              return;
93          }
94          if (CoreToolNames.EXECUTE_UPDATE.equals(descriptor.getName())) {
95              validateExecuteUpdateDescriptor(descriptor);
96              return;
97          }
98          if (CoreToolNames.VALIDATE_RUNTIME_DATABASE.equals(descriptor.getName())) {
99              validateRuntimeDatabaseDescriptor(descriptor);
100             return;
101         }
102         if (WorkflowToolDescriptors.APPLY_TOOL_NAME.equals(descriptor.getName())) {
103             validateApplyWorkflowDescriptor(descriptor);
104             return;
105         }
106         if (WorkflowToolDescriptors.VALIDATE_TOOL_NAME.equals(descriptor.getName())) {
107             validateValidateWorkflowDescriptor(descriptor);
108             return;
109         }
110         if (null != runtimeDescriptor && "plan".equals(runtimeDescriptor.getWorkflowRole())) {
111             validatePlanningWorkflowDescriptor(descriptor);
112             validatePlanningToolAnnotations(descriptor);
113         }
114     }

116     private static void validateSearchMetadataDescriptor(final MCPToolDescriptor descriptor) {
117         MCPToolDescriptorValidationUtils.validateRequiredOutputFields(descriptor,
118                 List.of("response_mode", MCPPayloadFieldNames.ITEMS, "count", "search_context", "total_match_count"));
119         Map<?, ?> properties = (Map<?, ?>) descriptor.getOutputSchema().get("properties");
120         Object items = properties.get(MCPPayloadFieldNames.ITEMS);
121         ShardingSpherePreconditions.checkState(items instanceof Map,
122                 () -> new IllegalStateException("Tool `database_gateway_search_metadata` outputSchema property `items` must be an object."));
123         Object itemSchema = ((Map<?, ?>) items).get("items");
124         ShardingSpherePreconditions.checkState(itemSchema instanceof Map,
125                 () -> new IllegalStateException("Tool `database_gateway_search_metadata` outputSchema property `items.items` must be an object."));
126         Object itemProperties = ((Map<?, ?>) itemSchema).get("properties");
127         ShardingSpherePreconditions.checkState(itemProperties instanceof Map && !((Map<?, ?>) itemProperties).isEmpty(),
128                 () -> new IllegalStateException("Tool `database_gateway_search_metadata` outputSchema property `items.items.properties` must declare properties."));
129         validateSearchMetadataItemFields((Map<?, ?>) itemProperties);
130     }
131     
132     private static void validateSearchMetadataItemFields(final Map<?, ?> properties) {
133         for (String each : List.of("database", "schema", "objectType", "table", "view", "name", MCPPayloadFieldNames.RESOURCE, MCPPayloadFieldNames.PARENT_RESOURCE,
134                 MCPPayloadFieldNames.NEXT_RESOURCES, "derivation_status", "match_kind", "matched_fields", "matched_value")) {
135             ShardingSpherePreconditions.checkState(properties.containsKey(each),
136                     () -> new IllegalStateException(String.format("Tool `database_gateway_search_metadata` outputSchema item must declare `%s`.", each)));
137             Object property = properties.get(each);
138             ShardingSpherePreconditions.checkState(property instanceof Map,
139                     () -> new IllegalStateException(String.format("Tool `database_gateway_search_metadata` outputSchema item property `%s` must be an object.", each)));
140             Object description = ((Map<?, ?>) property).get("description");
141             MCPToolDescriptorValidationUtils.checkDescription(null == description ? "" : description.toString(),
142                     String.format("Tool output item field `database_gateway_search_metadata.%s` description", each));
143         }
144     }

146     private static void validateExecuteQueryDescriptor(final MCPToolDescriptor descriptor) {
147         MCPToolDescriptorValidationUtils.validateRequiredOutputFields(descriptor, List.of("response_mode", "result_kind", "statement_class", "statement_type", "status", "returned_row_count",
148                 "applied_max_rows", "applied_timeout_ms", "truncated", MCPPayloadFieldNames.NEXT_ACTIONS));
149     }

151     private static void validateExecuteExplainDescriptor(final MCPToolDescriptor descriptor) {
152         validateRequiredInput(descriptor, "database");
153         validateRequiredInput(descriptor, "sql");
154         validateRequiredInput(descriptor, "explain_sql");
155         validateExecuteQueryDescriptor(descriptor);
156     }

158     private static void validateRequiredInput(final MCPToolDescriptor descriptor, final String fieldName) {
159         ShardingSpherePreconditions.checkState(MCPToolDescriptorValidationUtils.findToolInputProperty(descriptor, fieldName).isPresent(),
160                 () -> new IllegalStateException(String.format("Tool `%s` must declare `%s`.", descriptor.getName(), fieldName)));
161         ShardingSpherePreconditions.checkState(MCPToolDescriptorValidationUtils.isRequiredToolInput(descriptor, fieldName),
162                 () -> new IllegalStateException(String.format("Tool `%s` `%s` must be required.", descriptor.getName(), fieldName)));
163     }
164     
165     private static void validateExecuteUpdateDescriptor(final MCPToolDescriptor descriptor) {
166         MCPToolDescriptorValidationUtils.validateRequiredOutputFields(descriptor, List.of("response_mode", "result_kind", "statement_class", "statement_type", "status", "columns", "rows",
167                 "returned_row_count", "applied_max_rows", "applied_timeout_ms", "suggested_arguments", MCPPayloadFieldNames.NEXT_ACTIONS));
168         Map<?, ?> executionMode = MCPToolDescriptorValidationUtils.findToolInputProperty(descriptor, MCPPayloadFieldNames.EXECUTION_MODE).orElseThrow(
169                 () -> new IllegalStateException("Tool `database_gateway_execute_update` must declare execution_mode."));
170         ShardingSpherePreconditions.checkState(MCPToolDescriptorValidationUtils.isRequiredToolInput(descriptor, MCPPayloadFieldNames.EXECUTION_MODE),
171                 () -> new IllegalStateException("Tool `database_gateway_execute_update` execution_mode must be required."));
172         Object executionModes = executionMode.get("enum");
173         ShardingSpherePreconditions.checkState(executionModes instanceof Collection && ((Collection<?>) executionModes).containsAll(List.of("execute", "preview")),
174                 () -> new IllegalStateException("Tool `database_gateway_execute_update` execution_mode must allow execute and preview."));
175     }

177     private static void validateRuntimeDatabaseDescriptor(final MCPToolDescriptor descriptor) {
178         MCPToolDescriptorValidationUtils.validateRequiredOutputFields(descriptor, List.of("response_mode", "status", "database", "checks", "category", MCPPayloadFieldNames.RECOVERY));
179         Map<?, ?> responseMode = MCPToolDescriptorValidationUtils.findToolOutputProperty(descriptor, "response_mode").orElseThrow(
180                 () -> new IllegalStateException("Tool `database_gateway_validate_runtime_database` must declare response_mode."));
181         Object responseModes = responseMode.get("enum");
182         ShardingSpherePreconditions.checkState(responseModes instanceof Collection && ((Collection<?>) responseModes).contains("validation"),
183                 () -> new IllegalStateException("Tool `database_gateway_validate_runtime_database` response_mode must allow validation."));
184         ShardingSpherePreconditions.checkState(MCPToolDescriptorValidationUtils.findToolInputProperty(descriptor, "database").isPresent(),
185                 () -> new IllegalStateException("Tool `database_gateway_validate_runtime_database` must declare `database`."));
186         ShardingSpherePreconditions.checkState(MCPToolDescriptorValidationUtils.isRequiredToolInput(descriptor, "database"),
187                 () -> new IllegalStateException("Tool `database_gateway_validate_runtime_database` database must be required."));
188         for (String each : List.of("databaseType", "jdbcUrl", "username", "password", "driverClassName")) {
189             ShardingSpherePreconditions.checkState(MCPToolDescriptorValidationUtils.findToolInputProperty(descriptor, each).isEmpty(),
190                     () -> new IllegalStateException(String.format("Tool `database_gateway_validate_runtime_database` must not expose `%s`.", each)));
191         }
192         validateRuntimeDatabaseChecks(descriptor);
193     }
194     
195     private static void validateRuntimeDatabaseChecks(final MCPToolDescriptor descriptor) {
196         Map<?, ?> properties = (Map<?, ?>) descriptor.getOutputSchema().get("properties");
197         Object checks = properties.get("checks");
198         ShardingSpherePreconditions.checkState(checks instanceof Map,
199                 () -> new IllegalStateException("Tool `database_gateway_validate_runtime_database` outputSchema property `checks` must be an object."));
200         Object checkItemSchema = ((Map<?, ?>) checks).get("items");
201         ShardingSpherePreconditions.checkState(checkItemSchema instanceof Map,
202                 () -> new IllegalStateException("Tool `database_gateway_validate_runtime_database` outputSchema property `checks.items` must be an object."));
203         Object checkItemProperties = ((Map<?, ?>) checkItemSchema).get("properties");
204         ShardingSpherePreconditions.checkState(checkItemProperties instanceof Map && !((Map<?, ?>) checkItemProperties).isEmpty(),
205                 () -> new IllegalStateException("Tool `database_gateway_validate_runtime_database` outputSchema property `checks.items.properties` must declare properties."));
206         for (String each : List.of("name", "status", "category", "message")) {
207             ShardingSpherePreconditions.checkState(((Map<?, ?>) checkItemProperties).containsKey(each),
208                     () -> new IllegalStateException(String.format("Tool `database_gateway_validate_runtime_database` outputSchema check item must declare `%s`.", each)));
209         }
210     }

220                 List.of("response_mode", MCPPayloadFieldNames.SUMMARY, WorkflowFieldNames.PLAN_ID, "status", "overall_status", "issues", MCPPayloadFieldNames.NEXT_ACTIONS));

223     private static void validatePlanningWorkflowDescriptor(final MCPToolDescriptor descriptor) {
224         Map<?, ?> properties = (Map<?, ?>) descriptor.getOutputSchema().get("properties");
225         if (properties.containsKey("masked_property_preview") || properties.containsKey("secret_reference_summary")) {
226             MCPToolDescriptorValidationUtils.validateRequiredWorkflowPlanOutputFields(descriptor, SECRET_WORKFLOW_OUTPUT_FIELDS);
227         } else {
228             MCPToolDescriptorValidationUtils.validateRequiredWorkflowPlanOutputFields(descriptor);
229         }
230         MCPToolDescriptorValidationUtils.validateRequiredWorkflowPlanMetaFields(descriptor);
231     }

233     private static void validateRelatedResourceUris(final MCPToolDescriptor descriptor, final Set<String> resourceIdentifiers, final Set<String> shardingSphereResourceIdentifiers) {
234         Object value = descriptor.getMeta().get(MCPShardingSphereMetadataKeys.RELATED_RESOURCE_URIS);
235         if (null == value) {
236             return;
237         }
238         ShardingSpherePreconditions.checkState(value instanceof Collection,
239                 () -> new IllegalStateException(String.format("Tool `%s` metadata `%s` must be a list.", descriptor.getName(), MCPShardingSphereMetadataKeys.RELATED_RESOURCE_URIS)));
240         for (Object each : (Collection<?>) value) {
241             String uri = String.valueOf(each);
242             ShardingSpherePreconditions.checkState(resourceIdentifiers.contains(uri),
243                     () -> new IllegalStateException(String.format("Tool `%s` metadata `%s` references unknown resource `%s`.",
244                             descriptor.getName(), MCPShardingSphereMetadataKeys.RELATED_RESOURCE_URIS, uri)));
245             ShardingSpherePreconditions.checkState(shardingSphereResourceIdentifiers.contains(uri),
246                     () -> new IllegalStateException(String.format("Tool `%s` metadata `%s` references resource `%s` without ShardingSphere metadata.",
247                             descriptor.getName(), MCPShardingSphereMetadataKeys.RELATED_RESOURCE_URIS, uri)));
248         }
249     }
250     
251     private static void validatePlanningToolRuntimeDescriptors(final Map<String, MCPToolDescriptor> descriptors, final Collection<MCPToolRuntimeDescriptor> runtimeDescriptors) {
252         Map<String, String> workflowKinds = new LinkedHashMap<>(runtimeDescriptors.size(), 1F);
253         for (MCPToolRuntimeDescriptor each : runtimeDescriptors) {
254             if (!"plan".equals(each.getWorkflowRole())) {
255                 continue;
256             }
257             MCPToolDescriptor descriptor = descriptors.get(each.getToolName());
258             ShardingSpherePreconditions.checkState(null != descriptor,
259                     () -> new IllegalStateException(String.format("Planning runtime tool `%s` must reference a registered tool descriptor.", each.getToolName())));
260             Object workflowKind = descriptor.getMeta().get(MCPShardingSphereMetadataKeys.WORKFLOW_KIND);
261             ShardingSpherePreconditions.checkState(null != workflowKind && !workflowKind.toString().isBlank(),
262                     () -> new IllegalStateException(String.format("Planning tool `%s` metadata must declare `%s`.", each.getToolName(), MCPShardingSphereMetadataKeys.WORKFLOW_KIND)));
263             String previousToolName = workflowKinds.putIfAbsent(workflowKind.toString(), each.getToolName());
264             ShardingSpherePreconditions.checkState(null == previousToolName, () -> new IllegalStateException(
265                     String.format("Planning workflow kind `%s` is used by both `%s` and `%s`.", workflowKind, previousToolName, each.getToolName())));
266         }
267     }

269     private static void validatePlanningToolAnnotations(final MCPToolDescriptor descriptor) {
270         MCPToolAnnotations annotations = descriptor.getAnnotations();
271         ShardingSpherePreconditions.checkState(!annotations.isReadOnlyHint(),
272                 () -> new IllegalStateException(String.format("Planning tool `%s` annotations.readOnlyHint must be false.", descriptor.getName())));
273         ShardingSpherePreconditions.checkState(!annotations.isDestructiveHint(),
274                 () -> new IllegalStateException(String.format("Planning tool `%s` annotations.destructiveHint must be false.", descriptor.getName())));
275         ShardingSpherePreconditions.checkState(!annotations.isIdempotentHint(),
276                 () -> new IllegalStateException(String.format("Planning tool `%s` annotations.idempotentHint must be false.", descriptor.getName())));
277     }
278     
279     private static void validateDestructiveToolDescriptor(final MCPToolDescriptor descriptor) {
280         if (!descriptor.getAnnotations().isDestructiveHint()) {
281             return;
282         }
283         Map<?, ?> executionMode = MCPToolDescriptorValidationUtils.findToolInputProperty(descriptor, MCPPayloadFieldNames.EXECUTION_MODE).orElseThrow(
284                 () -> new IllegalStateException(String.format("Destructive tool `%s` must declare execution_mode.", descriptor.getName())));
285         ShardingSpherePreconditions.checkState(MCPToolDescriptorValidationUtils.isRequiredToolInput(descriptor, MCPPayloadFieldNames.EXECUTION_MODE),
286                 () -> new IllegalStateException(String.format("Destructive tool `%s` execution_mode must be required.", descriptor.getName())));
287         Collection<?> executionModes = executionMode.get("enum") instanceof Collection ? (Collection<?>) executionMode.get("enum") : List.of();
288         ShardingSpherePreconditions.checkState(executionModes.contains("preview"),
289                 () -> new IllegalStateException(String.format("Destructive tool `%s` execution_mode must allow preview.", descriptor.getName())));
290         ShardingSpherePreconditions.checkState(!executionModes.contains("auto-execute"),
291                 () -> new IllegalStateException(String.format("Destructive tool `%s` execution_mode must not expose auto-execute.", descriptor.getName())));
292     }
293     
294     private static void validateNonDestructiveExecutionMode(final MCPToolDescriptor descriptor) {
295         if (descriptor.getAnnotations().isDestructiveHint()) {
296             return;
297         }
298         Optional<Map<?, ?>> executionMode = MCPToolDescriptorValidationUtils.findToolInputProperty(descriptor, MCPPayloadFieldNames.EXECUTION_MODE);
299         if (executionMode.isEmpty()) {
300             return;
301         }
302         Collection<?> executionModes = executionMode.get().get("enum") instanceof Collection ? (Collection<?>) executionMode.get().get("enum") : List.of();
303         ShardingSpherePreconditions.checkState(!executionModes.contains("preview"),
304                 () -> new IllegalStateException(String.format("Non-destructive tool `%s` execution_mode must not expose preview.", descriptor.getName())));
305         ShardingSpherePreconditions.checkState(!executionModes.contains("auto-execute"),
306                 () -> new IllegalStateException(String.format("Non-destructive tool `%s` execution_mode must not expose auto-execute.", descriptor.getName())));
307     }
