183     private static ShardingRuleConfiguration createValidRuleConfiguration() {
184         ShardingRuleConfiguration result = new ShardingRuleConfiguration();
185         ShardingTableRuleConfiguration tableConfig = new ShardingTableRuleConfiguration("foo_tbl", "foo_ds.foo_tbl_${0..1}");
186         tableConfig.setDatabaseShardingStrategy(new StandardShardingStrategyConfiguration("user_id", "foo_standard"));
187         tableConfig.setTableShardingStrategy(new ComplexShardingStrategyConfiguration("order_id,user_id", "foo_complex"));
188         tableConfig.setAuditStrategy(new ShardingAuditStrategyConfiguration(Collections.singleton("foo_auditor"), false));
189         result.getTables().add(tableConfig);
190         ShardingAutoTableRuleConfiguration autoTableConfig = new ShardingAutoTableRuleConfiguration("foo_auto_tbl", "foo_ds_${0..1}");
191         autoTableConfig.setShardingStrategy(new HintShardingStrategyConfiguration("foo_hint"));
192         autoTableConfig.setAuditStrategy(new ShardingAuditStrategyConfiguration(Collections.singleton("foo_auditor"), true));
193         result.getAutoTables().add(autoTableConfig);
194         result.getBindingTableGroups().add(new ShardingTableReferenceRuleConfiguration("foo_group", "foo_tbl,foo_item"));
195         result.setDefaultDatabaseShardingStrategy(new StandardShardingStrategyConfiguration("user_id", "foo_standard"));
196         result.setDefaultTableShardingStrategy(new ComplexShardingStrategyConfiguration("order_id,user_id", "foo_complex"));
197         result.setDefaultKeyGenerateStrategy(new KeyGenerateStrategyConfiguration("order_id", "foo_key_generator"));
198         result.setDefaultAuditStrategy(new ShardingAuditStrategyConfiguration(Collections.singleton("foo_auditor"), false));
199         result.setDefaultShardingColumn("user_id");
200         result.getKeyGenerateStrategies().put("foo_column", new ColumnKeyGenerateStrategiesRuleConfiguration("foo_key_generator", "foo_tbl", "order_id"));
201         result.getKeyGenerateStrategies().put("foo_sequence", new SequenceKeyGenerateStrategiesRuleConfiguration("foo_key_generator", "foo_sequence"));
202         AlgorithmConfiguration shardingAlgorithmConfig = new AlgorithmConfiguration("FIXTURE", new Properties());
203         result.getShardingAlgorithms().put("foo_standard", shardingAlgorithmConfig);
204         result.getShardingAlgorithms().put("foo_complex", shardingAlgorithmConfig);
205         result.getShardingAlgorithms().put("foo_hint", shardingAlgorithmConfig);
206         result.getKeyGenerators().put("foo_key_generator", new AlgorithmConfiguration("FIXTURE", new Properties()));
207         result.getAuditors().put("foo_auditor", new AlgorithmConfiguration("FIXTURE_AUDIT", new Properties()));
208         return result;
209     }
