43  /**
44   * Encrypt show create table merged result.
45   */
46  public final class EncryptShowCreateViewMergedResult extends DecoratorMergedResult {
47      
48      private static final String COMMA = ", ";
49      
50      private static final int CREATE_TABLE_DEFINITION_INDEX = 2;
51      
52      private final MergedResult mergedResult;
53      
54      private final String viewName;
55      
56      private final EncryptRule rule;
57      
58      private final ShardingSphereMetaData metaData;
59      
60      private final SQLParserEngine sqlParserEngine;
61      
62      private final String currentDatabaseName;
63      
64      private final DatabaseType databaseType;
65      
66      public EncryptShowCreateViewMergedResult(final ShardingSphereMetaData metaData, final MergedResult mergedResult, final String viewName, final EncryptRule rule, final String currentDatabaseName) {
67          super(mergedResult);
68          this.mergedResult = mergedResult;
69          this.viewName = viewName;
70          this.rule = rule;
71          this.metaData = metaData;
72          this.currentDatabaseName = currentDatabaseName;
73          databaseType = metaData.getDatabase(this.currentDatabaseName).getProtocolType();
74          sqlParserEngine = metaData.getGlobalRuleMetaData().getSingleRule(SQLParserRule.class).getSQLParserEngine(databaseType);
75      }
76      
77      @Override
78      public Object getValue(final int columnIndex, final Class<?> type) throws SQLException {
79          if (CREATE_TABLE_DEFINITION_INDEX != columnIndex) {
80              return mergedResult.getValue(columnIndex, type);
81          }
82          String createViewSQL = mergedResult.getValue(CREATE_TABLE_DEFINITION_INDEX, type).toString();
83          Optional<EncryptTable> encryptView = rule.findEncryptTable(viewName);
84          if (!encryptView.isPresent()) {
85              return createViewSQL;
86          }
87          CreateViewStatement createViewStatement = (CreateViewStatement) sqlParserEngine.parse(createViewSQL, false);
88          Optional<TableSegment> from = createViewStatement.getSelect().getFrom();
89          if (!from.isPresent() || !(from.get() instanceof SimpleTableSegment)) {
90              return createViewSQL;
91          }
92          String tableName = ((SimpleTableSegment) from.get()).getTableName().getIdentifier().getValue();
93          Optional<EncryptTable> encryptTable =
94                  metaData.getDatabase(getDatabaseName((SimpleTableSegment) from.get(), databaseType, currentDatabaseName).getValue()).getRuleMetaData().findSingleRule(EncryptRule.class)
95                          .flatMap(encryptRule -> encryptRule.findEncryptTable(tableName));
96          if (!encryptTable.isPresent()) {
97              return createViewSQL;
98          }
99          List<ProjectionSegment> projections = createViewStatement.getSelect().getProjections().getProjections();
100         StringBuilder result = new StringBuilder(createViewSQL.substring(0, projections.get(0).getStartIndex()));
101         for (ProjectionSegment each : projections) {
102             findLogicColumnDefinition(each, encryptView.get(), encryptTable.get(), createViewSQL).ifPresent(optional -> result.append(optional).append(COMMA));
103         }
104         result.delete(result.length() - COMMA.length(), result.length()).append(createViewSQL.substring(projections.get(projections.size() - 1).getStopIndex() + 1));
105         return result.toString();
106     }
107     
108     private static IdentifierValue getDatabaseName(final SimpleTableSegment segment, final DatabaseType databaseType, final String currentDatabaseName) {
109         DialectDatabaseMetaData dialectDatabaseMetaData = new DatabaseTypeRegistry(databaseType).getDialectDatabaseMetaData();
110         Optional<OwnerSegment> owner = dialectDatabaseMetaData.getSchemaOption().getDefaultSchema().isPresent() ? segment.getOwner().flatMap(OwnerSegment::getOwner) : segment.getOwner();
111         return new IdentifierValue(owner.map(optional -> optional.getIdentifier().getValue()).orElse(currentDatabaseName));
112     }
113     
114     private Optional<String> findLogicColumnDefinition(final ProjectionSegment projectionSegment, final EncryptTable encryptView, final EncryptTable encryptTable, final String createTableSQL) {
115         if (!(projectionSegment instanceof ColumnProjectionSegment)) {
116             return Optional.of(createTableSQL.substring(projectionSegment.getStartIndex(), projectionSegment.getStopIndex() + 1));
117         }
118         ColumnSegment columnSegment = ((ColumnProjectionSegment) projectionSegment).getColumn();
119         String columnName = ((ColumnProjectionSegment) projectionSegment).getAliasName().orElse(columnSegment.getIdentifier().getValue());
120         if (encryptView.isCipherColumn(columnName)) {
121             String logicColumn = encryptView.getLogicColumnByCipherColumn(columnName);
122             String actualColumn = encryptTable.getLogicColumnByCipherColumn(columnSegment.getIdentifier().getValue());
123             StringBuilder result = new StringBuilder(actualColumn);
124             Optional<IdentifierValue> alias = ((ColumnProjectionSegment) projectionSegment).getAlias();
125             alias.ifPresent(identifierValue -> result.append(" AS ").append(identifierValue.getQuoteCharacter().wrap(logicColumn)));
126             return Optional.of(result.toString());
127         }
128         if (isDerivedColumn(encryptView, columnName)) {
129             return Optional.empty();
130         }
131         return Optional.of(createTableSQL.substring(projectionSegment.getStartIndex(), projectionSegment.getStopIndex() + 1));
132     }
133     
134     private boolean isDerivedColumn(final EncryptTable encryptTable, final String columnName) {
135         return encryptTable.isAssistedQueryColumn(columnName) || encryptTable.isLikeQueryColumn(columnName);
136     }
