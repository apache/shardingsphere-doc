36  class PipelineProcessConfigurationUtilsTest {
37      
38      @Test
39      void assertFillInDefaultValueWithNullConfiguration() {
40          PipelineProcessConfiguration actual = PipelineProcessConfigurationUtils.fillInDefaultValue(null);
41          assertThat(actual.getRead().getWorkerThread(), is(20));
42          assertThat(actual.getRead().getBatchSize(), is(1000));
43          assertThat(actual.getRead().getShardingSize(), is(10000000));
44          assertNull(actual.getRead().getRateLimiter());
45          assertThat(actual.getWrite().getWorkerThread(), is(20));
46          assertThat(actual.getWrite().getBatchSize(), is(1000));
47          assertNull(actual.getWrite().getRateLimiter());
48          assertThat(actual.getStreamChannel().getType(), is("MEMORY"));
49          assertThat(actual.getStreamChannel().getProps(), is(PropertiesBuilder.build(new Property("block-queue-size", "2000"))));
50      }
51      
52      @ParameterizedTest(name = "Missing {0}")
53      @MethodSource("getPartialConfigurations")
54      void assertFillInDefaultValueWithMissingSection(final String section, final PipelineProcessConfiguration originalConfig, final PipelineProcessConfiguration expected) {
55          PipelineProcessConfiguration actual = PipelineProcessConfigurationUtils.fillInDefaultValue(originalConfig);
56          assertThat(actual.getRead().getWorkerThread(), is(expected.getRead().getWorkerThread()));
57          assertThat(actual.getRead().getBatchSize(), is(expected.getRead().getBatchSize()));
58          assertThat(actual.getRead().getShardingSize(), is(expected.getRead().getShardingSize()));
59          assertNull(actual.getRead().getRateLimiter());
60          assertThat(actual.getWrite().getWorkerThread(), is(expected.getWrite().getWorkerThread()));
61          assertThat(actual.getWrite().getBatchSize(), is(expected.getWrite().getBatchSize()));
62          assertNull(actual.getWrite().getRateLimiter());
63          assertThat(actual.getStreamChannel(), is(expected.getStreamChannel()));
64      }
65      
66      private static Stream<Arguments> getPartialConfigurations() {
67          PipelineReadConfiguration readConfig = new PipelineReadConfiguration(40, 500, 6000000, null);
68          PipelineWriteConfiguration writeConfig = new PipelineWriteConfiguration(10, 250, null);
69          AlgorithmConfiguration streamChannel = new AlgorithmConfiguration("CUSTOM", PropertiesBuilder.build(new Property("foo_key", "foo_value")));
70          PipelineReadConfiguration defaultReadConfig = new PipelineReadConfiguration(20, 1000, 10000000, null);
71          PipelineWriteConfiguration defaultWriteConfig = new PipelineWriteConfiguration(20, 1000, null);
72          AlgorithmConfiguration expectedStreamChannel = new AlgorithmConfiguration("CUSTOM", PropertiesBuilder.build(new Property("foo_key", "foo_value")));
73          AlgorithmConfiguration defaultStreamChannel = new AlgorithmConfiguration("MEMORY", PropertiesBuilder.build(new Property("block-queue-size", "2000")));
74          return Stream.of(
75                  Arguments.of("read", new PipelineProcessConfiguration(null, writeConfig, streamChannel), new PipelineProcessConfiguration(defaultReadConfig, writeConfig, expectedStreamChannel)),
76                  Arguments.of("write", new PipelineProcessConfiguration(readConfig, null, streamChannel), new PipelineProcessConfiguration(readConfig, defaultWriteConfig, expectedStreamChannel)),
77                  Arguments.of("streamChannel", new PipelineProcessConfiguration(readConfig, writeConfig, null), new PipelineProcessConfiguration(readConfig, writeConfig, defaultStreamChannel)));
78      }

80      @Test
81      void assertFillInDefaultValueWithCustomConfiguration() {
82          PipelineReadConfiguration readConfig = new PipelineReadConfiguration(40, 500, 6000000, new AlgorithmConfiguration("INPUT", PropertiesBuilder.build(new Property("qps", "50"))));
83          PipelineWriteConfiguration writeConfig = new PipelineWriteConfiguration(10, 250, new AlgorithmConfiguration("OUTPUT", PropertiesBuilder.build(new Property("tps", "75"))));
84          AlgorithmConfiguration streamChannel = new AlgorithmConfiguration("CUSTOM", PropertiesBuilder.build(new Property("topic", "foo_topic")));
85          PipelineProcessConfiguration originalConfig = new PipelineProcessConfiguration(readConfig, writeConfig, streamChannel);
86          PipelineProcessConfiguration actual = PipelineProcessConfigurationUtils.fillInDefaultValue(originalConfig);
87          assertThat(actual.getRead().getWorkerThread(), is(40));
88          assertThat(actual.getRead().getBatchSize(), is(500));
89          assertThat(actual.getRead().getShardingSize(), is(6000000));
90          assertThat(actual.getRead().getRateLimiter().getType(), is("INPUT"));
91          assertThat(actual.getRead().getRateLimiter().getProps(), is(PropertiesBuilder.build(new Property("qps", "50"))));
92          assertThat(actual.getWrite().getWorkerThread(), is(10));
93          assertThat(actual.getWrite().getBatchSize(), is(250));
94          assertThat(actual.getWrite().getRateLimiter().getType(), is("OUTPUT"));
95          assertThat(actual.getWrite().getRateLimiter().getProps(), is(PropertiesBuilder.build(new Property("tps", "75"))));
96          assertThat(actual.getStreamChannel().getType(), is("CUSTOM"));
97          assertThat(actual.getStreamChannel().getProps(), is(PropertiesBuilder.build(new Property("topic", "foo_topic"))));
98          assertThat(originalConfig.getRead().getRateLimiter().getProps().getProperty("qps"), is("50"));
99          assertThat(originalConfig.getWrite().getRateLimiter().getProps().getProperty("tps"), is("75"));
100         assertThat(originalConfig.getStreamChannel().getProps().getProperty("topic"), is("foo_topic"));
101     }

103     @Test
104     void assertFillInDefaultValueWithZeroValues() {
105         PipelineProcessConfiguration originalConfig = new PipelineProcessConfiguration(new PipelineReadConfiguration(0, 0, 0, null), new PipelineWriteConfiguration(0, 0, null),
106                 new AlgorithmConfiguration("MEMORY", PropertiesBuilder.build()));
107         PipelineProcessConfiguration actual = PipelineProcessConfigurationUtils.fillInDefaultValue(originalConfig);
108         assertThat(actual.getRead().getWorkerThread(), is(0));
109         assertThat(actual.getRead().getBatchSize(), is(0));
110         assertThat(actual.getRead().getShardingSize(), is(0));
111         assertThat(actual.getWrite().getWorkerThread(), is(0));
112         assertThat(actual.getWrite().getBatchSize(), is(0));
113         assertTrue(actual.getStreamChannel().getProps().isEmpty());
114     }
115     
116     @Test
117     void assertFillInDefaultValueWithUnchangedInput() {
118         PipelineProcessConfiguration originalConfig = new PipelineProcessConfiguration(null, null, null);
119         PipelineProcessConfiguration actual = PipelineProcessConfigurationUtils.fillInDefaultValue(originalConfig);
120         assertNotNull(actual.getRead());
121         assertNotNull(actual.getWrite());
122         assertNotNull(actual.getStreamChannel());
123         assertNull(originalConfig.getRead());
124         assertNull(originalConfig.getWrite());
125         assertNull(originalConfig.getStreamChannel());
126     }
127     
128     @Test
129     void assertFillInDefaultValueWithIndependentDefaults() {
130         PipelineProcessConfiguration actualFirst = PipelineProcessConfigurationUtils.fillInDefaultValue(null);
131         PipelineProcessConfiguration actualSecond = PipelineProcessConfigurationUtils.fillInDefaultValue(null);
132         actualFirst.getStreamChannel().getProps().setProperty("block-queue-size", "10");
133         assertThat(actualSecond.getStreamChannel().getProps().getProperty("block-queue-size"), is("2000"));
134     }
