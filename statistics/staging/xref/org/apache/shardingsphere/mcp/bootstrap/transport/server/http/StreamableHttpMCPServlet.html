88      StreamableHttpMCPServlet(final MCPSessionManager sessionManager, final McpJsonMapper jsonMapper, final HttpTransportConfiguration config) {
89          sessionAttributionResolver = new SessionAttributionResolver(config.getSessionAttributionSource());
90          securityValidator = new ShardingSphereServerTransportSecurityValidator(sessionManager,
91                  new OriginHeaderConstraint(HttpTransportHostUtils.isLoopbackHost(config.getBindHost())),
92                  new ProtocolVersionHeaderConstraint(), sessionAttributionResolver);
93          delegate = createDelegate(jsonMapper, config.getEndpointPath());
94          this.sessionManager = sessionManager;
95          sessionExecutionCoordinator = new MCPSessionExecutionCoordinator(sessionManager);
96          closed = new AtomicBoolean();
97          lifecycleLock = new StampedLock();
98      }

100     private static HttpServletStreamableServerTransportProvider createDelegate(final McpJsonMapper jsonMapper, final String endpointPath) {
101         return HttpServletStreamableServerTransportProvider.builder().jsonMapper(jsonMapper).mcpEndpoint(endpointPath).build();
102     }

104     @Override
105     public List<String> protocolVersions() {
106         return MCPTransportConstants.SUPPORTED_PROTOCOL_VERSIONS;
107     }

109     @Override
110     public void setSessionFactory(final McpStreamableServerSession.Factory sessionFactory) {
111         delegate.setSessionFactory(initializeRequest -> {
112             McpSchema.InitializeRequest actualInitializeRequest = negotiateInitializeRequest(initializeRequest);
113             return sessionFactory.startSession(actualInitializeRequest);
114         });
115     }

117     private McpSchema.InitializeRequest negotiateInitializeRequest(final McpSchema.InitializeRequest initializeRequest) {
118         String requestedProtocolVersion = Objects.toString(initializeRequest.protocolVersion(), "").trim();
119         ShardingSpherePreconditions.checkNotEmpty(requestedProtocolVersion, () -> new IllegalArgumentException("protocolVersion is required."));
120         String negotiatedProtocolVersion = protocolVersions().contains(requestedProtocolVersion) ? requestedProtocolVersion : MCPTransportConstants.PROTOCOL_VERSION;
121         return negotiatedProtocolVersion.equals(initializeRequest.protocolVersion())
122                 ? initializeRequest
123                 : new McpSchema.InitializeRequest(negotiatedProtocolVersion, initializeRequest.capabilities(), initializeRequest.clientInfo(), initializeRequest.meta());
124     }

136     @Override
137     protected void service(final HttpServletRequest request, final HttpServletResponse response) throws ServletException, IOException {
138         long lifecycleStamp = enterRequest(response);
139         if (0L == lifecycleStamp) {
140             return;
141         }
142         try {
143             super.service(request, response);
144         } catch (final IOException | ServletException ex) {
145             if (response.isCommitted()) {
146                 throw ex;
147             }
148             log.error("Unexpected MCP HTTP request failure.", ex);
149             response.reset();
150             response.setStatus(HttpServletResponse.SC_INTERNAL_SERVER_ERROR);
151         } finally {
152             lifecycleLock.unlockRead(lifecycleStamp);
153         }
154     }
155     
156     @Override
157     protected void doGet(final HttpServletRequest request, final HttpServletResponse response) throws IOException {
158         setUtf8Encoding(request, response);
159         if (!validateTransportSecurity(request, response)) {
160             return;
161         }
162         String sessionId = Objects.toString(request.getHeader(SESSION_HEADER), "");
163         if (!sessionId.isBlank() && !sessionManager.hasSession(sessionId)) {
164             response.sendError(HttpServletResponse.SC_NOT_FOUND);
165             return;
166         }
167         if (!isEventStreamAccepted(request)) {
168             response.sendError(HttpServletResponse.SC_BAD_REQUEST, "Accept must include text/event-stream.");
169             return;
170         }
171         response.sendError(HttpServletResponse.SC_METHOD_NOT_ALLOWED, "HTTP GET event streams are not supported.");
172     }
173     
174     private boolean isEventStreamAccepted(final HttpServletRequest request) {
175         String acceptHeader = Objects.toString(request.getHeader(HttpHeaders.ACCEPT), "");
176         for (String each : acceptHeader.split(",")) {
177             if (EVENT_STREAM_CONTENT_TYPE.equalsIgnoreCase(each.split(";", 2)[0].trim())) {
178                 return true;
179             }
180         }
181         return false;
182     }
183     
184     private void serviceWithApplicationClassLoader(final HttpServletRequest request, final HttpServletResponse response) throws IOException, ServletException {
185         ClassLoader originalClassLoader = Thread.currentThread().getContextClassLoader();
186         try {
187             Thread.currentThread().setContextClassLoader(getClass().getClassLoader());
188             delegate.service(request, response);
189         } finally {
190             Thread.currentThread().setContextClassLoader(originalClassLoader);
191         }
192     }
193     
194     @Override
195     protected void doPost(final HttpServletRequest request, final HttpServletResponse response) throws IOException, ServletException {
196         setUtf8Encoding(request, response);
197         if (!validateTransportSecurity(request, response)) {
198             return;
199         }
200         if (!isJsonContentType(request)) {
201             response.sendError(HttpServletResponse.SC_UNSUPPORTED_MEDIA_TYPE, "Content-Type must be application/json.");
202             return;
203         }
204         SessionAwareHttpServletResponse actualResponse = withInitializeSessionHeaders(request, response);
205         boolean completed = false;
206         try {
207             serviceWithApplicationClassLoader(request, actualResponse);
208             completed = true;
209         } finally {
210             if (!actualResponse.getSessionId().isEmpty() && (!completed || HttpServletResponse.SC_OK != response.getStatus())) {
211                 sessionExecutionCoordinator.closeSession(actualResponse.getSessionId());
212             }
213         }
214     }
215     
216     private boolean isJsonContentType(final HttpServletRequest request) {
217         String contentType = Objects.toString(request.getContentType(), "").trim();
218         return contentType.isEmpty() || JSON_CONTENT_TYPE.equalsIgnoreCase(contentType.split(";", 2)[0].trim());
219     }
220     
221     private boolean validateTransportSecurity(final HttpServletRequest request, final HttpServletResponse response) throws IOException {
222         try {
223             securityValidator.validateHeaders(extractHeaders(request));
224             return true;
225         } catch (final MCPTransportSecurityException ex) {
226             writeTransportSecurityError(response, ex);
227             return false;
228         } catch (final ServerTransportSecurityException ex) {
229             response.sendError(ex.getStatusCode(), ex.getMessage());
230             return false;
231         }
232     }
233     
234     private void writeTransportSecurityError(final HttpServletResponse response, final MCPTransportSecurityException cause) throws IOException {
235         McpError error = MCPTransportErrorFactory.createError(cause);
236         response.setStatus(cause.getStatusCode());
237         response.setContentType(JSON_CONTENT_TYPE);
238         response.getWriter().write(JsonEngine.marshal(Map.of("jsonrpc", McpSchema.JSONRPC_VERSION, "error", error.getJsonRpcError())));
239     }
240     
241     private Map<String, List<String>> extractHeaders(final HttpServletRequest request) {
242         Enumeration<String> headerNames = request.getHeaderNames();
243         if (null == headerNames) {
244             return Map.of();
245         }
246         Map<String, List<String>> result = new LinkedHashMap<>();
247         for (String each : Collections.list(headerNames)) {
248             Enumeration<String> headerValues = request.getHeaders(each);
249             result.put(each, null == headerValues ? List.of() : Collections.list(headerValues));
250         }
251         return result;
252     }
253     
254     @Override
255     protected void doDelete(final HttpServletRequest request, final HttpServletResponse response) throws IOException, ServletException {
256         setUtf8Encoding(request, response);
257         if (!validateTransportSecurity(request, response)) {
258             return;
259         }
260         String sessionId = Objects.toString(request.getHeader(SESSION_HEADER), "");
261         serviceWithApplicationClassLoader(request, response);
262         if (200 == response.getStatus()) {
263             sessionExecutionCoordinator.closeSession(sessionId);
264         }
265     }
266     
267     private void setUtf8Encoding(final HttpServletRequest request, final HttpServletResponse response) throws IOException {
268         request.setCharacterEncoding(StandardCharsets.UTF_8.name());
269         response.setCharacterEncoding(StandardCharsets.UTF_8.name());
270     }
271     
272     @Override
273     public Mono<Void> closeGracefully() {
274         return closed.compareAndSet(false, true)
275                 ? Mono.using(lifecycleLock::writeLock, ignored -> delegate.closeGracefully(), lifecycleStamp -> {
276                     try {
277                         sessionExecutionCoordinator.closeAllSessions();
278                     } finally {
279                         lifecycleLock.unlockWrite(lifecycleStamp);
280                     }
281                 })
282                 : Mono.empty();
283     }
284     
285     @Override
286     public void destroy() {
287         try {
288             closeGracefully().block();
289         } finally {
290             super.destroy();
291         }
292     }
293     
294     private long enterRequest(final HttpServletResponse response) throws IOException {
295         long result = lifecycleLock.readLock();
296         if (!closed.get()) {
297             return result;
298         }
299         lifecycleLock.unlockRead(result);
300         response.sendError(HttpServletResponse.SC_SERVICE_UNAVAILABLE, "Server is shutting down");
301         return 0L;
302     }

304     private SessionAwareHttpServletResponse withInitializeSessionHeaders(final HttpServletRequest request, final HttpServletResponse response) {
305         return new SessionAwareHttpServletResponse(response) {
306             
307             @Override
308             public void setHeader(final String name, final String value) {
309                 registerSession(request, this, name, value);
310                 super.setHeader(name, value);
311                 addNegotiatedProtocolHeader(this, name);
312             }
313             
314             @Override
315             public void addHeader(final String name, final String value) {
316                 registerSession(request, this, name, value);
317                 super.addHeader(name, value);
318                 addNegotiatedProtocolHeader(this, name);
319             }
320         };
321     }
322     
323     private void registerSession(final HttpServletRequest request, final SessionAwareHttpServletResponse response, final String name, final String sessionId) {
324         if (!SESSION_HEADER.equalsIgnoreCase(name)) {
325             return;
326         }
327         ShardingSpherePreconditions.checkState(!response.isCommitted(), () -> new IllegalStateException("Cannot register an MCP session after the response is committed."));
328         String existingSessionId = response.getSessionId();
329         if (!existingSessionId.isEmpty()) {
330             ShardingSpherePreconditions.checkState(existingSessionId.equals(sessionId), () -> new IllegalStateException("MCP session response header changed during initialization."));
331             return;
332         }
333         MCPSessionIdentity sessionIdentity = sessionAttributionResolver.resolve(request, sessionId);
334         sessionManager.createSession(sessionIdentity);
335         response.setSessionId(sessionId);
336     }
337     
338     private void addNegotiatedProtocolHeader(final SessionAwareHttpServletResponse response, final String name) {
339         if (SESSION_HEADER.equalsIgnoreCase(name)) {
340             response.setHeader(PROTOCOL_HEADER, MCPTransportConstants.PROTOCOL_VERSION);
341         }
342     }
343     
344     private abstract static class SessionAwareHttpServletResponse extends HttpServletResponseWrapper {
345         
346         private String sessionId = "";
347         
348         SessionAwareHttpServletResponse(final HttpServletResponse response) {
349             super(response);
350         }
351         
352         protected final String getSessionId() {
353             return sessionId;
354         }
355         
356         protected final void setSessionId(final String sessionId) {
357             this.sessionId = Objects.toString(sessionId, "");
358         }
359     }
