115     public SelectStatementBaseContext(final SelectStatement sqlStatement, final ShardingSphereMetaData metaData, final String currentDatabaseName, final Collection<TableSegment> inheritedTables) {
116         this.sqlStatement = sqlStatement;
117         whereSegments = createWhereSegments(sqlStatement);
118         columnSegments = ColumnExtractor.extractColumnSegments(whereSegments);
119         Collection<TableSegment> tableSegments = getAllTableSegments(inheritedTables);
120         ExpressionExtractor.extractJoinConditions(joinConditions, whereSegments);
121         subqueryContexts = createSubqueryContexts(metaData, currentDatabaseName, tableSegments);
122         tablesContext = new TablesContext(tableSegments);
123         groupByContext = new GroupByContextEngine().createGroupByContext(sqlStatement);
124         orderByContext = new OrderByContextEngine(sqlStatement.getDatabaseType()).createOrderBy(sqlStatement, groupByContext);
125         projectionsContext = new ProjectionsContextEngine(sqlStatement.getDatabaseType()).createProjectionsContext(sqlStatement.getProjections(), groupByContext, orderByContext);
126         containsEnhancedTable = isContainsEnhancedTable(metaData, tablesContext.getDatabaseNames(), currentDatabaseName);
127     }
128     
129     private Collection<WhereSegment> createWhereSegments(final SelectStatement selectStatement) {
130         Collection<WhereSegment> result = new LinkedList<>();
131         selectStatement.getWhere().ifPresent(result::add);
132         selectStatement.getHaving().ifPresent(optional -> result.add(new WhereSegment(optional.getExpr().getStartIndex(), optional.getExpr().getStopIndex(), optional.getExpr())));
133         result.addAll(WhereExtractor.extractHierarchicalQueryWhereSegments(selectStatement));
134         result.addAll(WhereExtractor.extractSubqueryWhereSegments(selectStatement));
135         result.addAll(WhereExtractor.extractJoinWhereSegments(selectStatement));
136         return result;
137     }
138     
139     private Collection<TableSegment> getAllTableSegments(final Collection<TableSegment> inheritedTables) {
140         TableExtractor tableExtractor = new TableExtractor();
141         appendInheritedSimpleTables(inheritedTables, tableExtractor);
142         tableExtractor.extractTablesFromSelect(sqlStatement);
143         Collection<TableSegment> result = new LinkedList<>(tableExtractor.getRewriteTables());
144         for (TableSegment each : tableExtractor.getTableContext()) {
145             if (each instanceof SubqueryTableSegment) {
146                 result.add(each);
147             }
148         }
149         return result;
150     }
151     
152     private void appendInheritedSimpleTables(final Collection<TableSegment> inheritedTables, final TableExtractor tableExtractor) {
153         for (TableSegment each : inheritedTables) {
154             if (each instanceof SimpleTableSegment) {
155                 tableExtractor.getTableContext().add(each);
156             }
157         }
158     }

160     private boolean isContainsEnhancedTable(final ShardingSphereMetaData metaData, final Collection<String> databaseNames, final String currentDatabaseName) {
161         for (String each : databaseNames) {
162             if (isContainsEnhancedTable(metaData, each, getTablesContext().getTableNames())) {
163                 return true;
164             }
165         }
166         return null != currentDatabaseName && isContainsEnhancedTable(metaData, currentDatabaseName, getTablesContext().getTableNames());
167     }
168     
169     private boolean isContainsEnhancedTable(final ShardingSphereMetaData metaData, final String databaseName, final Collection<String> tableNames) {
170         for (TableMapperRuleAttribute each : getTableMapperRuleAttributes(metaData, databaseName)) {
171             for (String tableName : tableNames) {
172                 if (each.getEnhancedTableNames().contains(tableName)) {
173                     return true;
174                 }
175             }
176         }
177         return false;
178     }
179     
180     private Collection<TableMapperRuleAttribute> getTableMapperRuleAttributes(final ShardingSphereMetaData metaData, final String databaseName) {
181         if (null == databaseName) {
182             ShardingSpherePreconditions.checkMustEmpty(tablesContext.getSimpleTables(), NoDatabaseSelectedException::new);
183             return Collections.emptyList();
184         }
185         ShardingSphereDatabase database = metaData.getDatabase(databaseName);
186         ShardingSpherePreconditions.checkNotNull(database, () -> new UnknownDatabaseException(databaseName));
187         return database.getRuleMetaData().getAttributes(TableMapperRuleAttribute.class);
188     }

190     private Map<Integer, SelectStatementContext> createSubqueryContexts(final ShardingSphereMetaData metaData, final String currentDatabaseName, final Collection<TableSegment> tableSegments) {
191         Collection<SubquerySegment> subquerySegments = SubqueryExtractor.extractSubquerySegments(sqlStatement, false);
192         Map<Integer, SelectStatementContext> result = new HashMap<>(subquerySegments.size(), 1F);
193         for (SubquerySegment each : subquerySegments) {
194             SelectStatementContext subqueryContext = new SelectStatementContext(each.getSelect(), metaData, currentDatabaseName, tableSegments);
195             each.getSelect().getSubqueryType().ifPresent(subqueryContext::setSubqueryType);
196             result.put(each.getStartIndex(), subqueryContext);
197         }
198         return result;
199     }

286      * Set indexes.
287      *
288      * @param columnLabelIndexMap map for column label and index
289      */
290     public void setIndexes(final Map<String, Integer> columnLabelIndexMap) {
291         setIndexForAggregationProjection(columnLabelIndexMap);
292         setIndexForOrderItem(columnLabelIndexMap, orderByContext.getItems());
293         setIndexForOrderItem(columnLabelIndexMap, groupByContext.getItems());
294     }

296     private void setIndexForAggregationProjection(final Map<String, Integer> columnLabelIndexMap) {
297         for (AggregationProjection each : projectionsContext.getAggregationProjections()) {
298             String columnLabel = SQLUtils.getExactlyValue(each.getAlias().map(IdentifierValue::getValue).orElse(each.getColumnName()));
299             Preconditions.checkState(columnLabelIndexMap.containsKey(columnLabel), "Can't find index: %s, please add alias for aggregate selections", each);
300             each.setIndex(columnLabelIndexMap.get(columnLabel));
301             for (AggregationProjection derived : each.getDerivedAggregationProjections()) {
302                 String derivedColumnLabel = SQLUtils.getExactlyValue(derived.getAlias().map(IdentifierValue::getValue).orElse(each.getColumnName()));
303                 Preconditions.checkState(columnLabelIndexMap.containsKey(derivedColumnLabel), "Can't find index: %s", derived);
304                 derived.setIndex(columnLabelIndexMap.get(derivedColumnLabel));
305             }
306         }
307     }
308     
309     private void setIndexForOrderItem(final Map<String, Integer> columnLabelIndexMap, final Collection<OrderByItem> orderByItems) {
310         for (OrderByItem each : orderByItems) {
311             if (each.getSegment() instanceof IndexOrderByItemSegment) {
312                 each.setIndex(((IndexOrderByItemSegment) each.getSegment()).getColumnIndex());
313                 continue;
314             }
315             if (each.getSegment() instanceof ColumnOrderByItemSegment && ((ColumnOrderByItemSegment) each.getSegment()).getColumn().getOwner().isPresent()) {
316                 Optional<Integer> itemIndex = projectionsContext.findProjectionIndex(((ColumnOrderByItemSegment) each.getSegment()).getText());
317                 if (itemIndex.isPresent()) {
318                     each.setIndex(itemIndex.get());
319                     continue;
320                 }
321             }
322             String columnLabel = getAlias(each.getSegment()).orElseGet(() -> getOrderItemText((TextOrderByItemSegment) each.getSegment()));
323             Preconditions.checkState(columnLabelIndexMap.containsKey(columnLabel), "Can't find index: %s", each);
324             if (columnLabelIndexMap.containsKey(columnLabel)) {
325                 each.setIndex(columnLabelIndexMap.get(columnLabel));
326             }
327         }
328     }

330     private Optional<String> getAlias(final OrderByItemSegment orderByItem) {
331         if (projectionsContext.isUnqualifiedShorthandProjection()) {
332             return Optional.empty();
333         }
334         String rawName = SQLUtils.getExactlyValue(((TextOrderByItemSegment) orderByItem).getText());
335         for (Projection each : projectionsContext.getProjections()) {
336             Optional<String> result = each.getAlias().map(IdentifierValue::getValue);
337             if (SQLUtils.getExactlyExpression(rawName).equalsIgnoreCase(SQLUtils.getExactlyExpression(SQLUtils.getExactlyValue(each.getExpression())))) {
338                 return result;
339             }
340             if (rawName.equalsIgnoreCase(result.orElse(null))) {
341                 return Optional.of(rawName);
342             }
343             if (isSameColumnName(each, rawName)) {
344                 return result;
345             }
346         }
347         return Optional.empty();
348     }

352     }

354     private String getOrderItemText(final TextOrderByItemSegment orderByItemSegment) {
355         if (orderByItemSegment instanceof ColumnOrderByItemSegment) {

361     /**
362      * Judge whether group by items and order by items can be proven to refer to the same select items in the same sequence, ignoring order direction.
363      *
364      * <p>Either order direction keeps rows of the same group by keys adjacent in the merged result, so the direction-insensitive judgment
365      * keeps pagination rewrite and group by stream merge available when the order by keys are exactly the group by keys.</p>
366      *
367      * @return group by and order by items are the same or not
368      */
369     public boolean isSameGroupByAndOrderByItems() {
370         if (groupByContext.getItems().isEmpty() || groupByContext.getItems().size() != orderByContext.getItems().size()) {
371             return false;
372         }
373         if (orderByContext.isGenerated()) {
374             return true;
375         }
376         Iterator<OrderByItem> groupByItems = groupByContext.getItems().iterator();
377         Iterator<OrderByItem> orderByItems = orderByContext.getItems().iterator();
378         while (groupByItems.hasNext()) {
379             if (!isSameGroupByAndOrderByItem(groupByItems.next(), orderByItems.next())) {
380                 return false;
381             }
382         }
383         return true;
384     }
385     
386     private boolean isSameGroupByAndOrderByItem(final OrderByItem groupByItem, final OrderByItem orderByItem) {
387         if (0 != groupByItem.getIndex() && 0 != orderByItem.getIndex() && groupByItem.getIndex() != orderByItem.getIndex()) {
388             return false;
389         }
390         OrderByItemSegment groupBySegment = groupByItem.getSegment();
391         OrderByItemSegment orderBySegment = orderByItem.getSegment();
392         if (groupBySegment.getClass() != orderBySegment.getClass()) {
393             return false;
394         }
395         if (groupBySegment instanceof IndexOrderByItemSegment) {
396             return ((IndexOrderByItemSegment) groupBySegment).getColumnIndex() == ((IndexOrderByItemSegment) orderBySegment).getColumnIndex();
397         }
398         if (groupBySegment instanceof ColumnOrderByItemSegment) {
399             ColumnSegment groupByColumn = ((ColumnOrderByItemSegment) groupBySegment).getColumn();
400             ColumnSegment orderByColumn = ((ColumnOrderByItemSegment) orderBySegment).getColumn();
401             return isSameColumn(groupByColumn, orderByColumn);
402         }
403         return false;
404     }
405     
406     private boolean isSameColumn(final ColumnSegment groupByColumn, final ColumnSegment orderByColumn) {
407         ColumnSegmentBoundInfo groupByBoundInfo = groupByColumn.getColumnBoundInfo();
408         ColumnSegmentBoundInfo orderByBoundInfo = orderByColumn.getColumnBoundInfo();
409         ColumnSegmentBoundInfo groupByOtherUsingBoundInfo = groupByColumn.getOtherUsingColumnBoundInfo();
410         ColumnSegmentBoundInfo orderByOtherUsingBoundInfo = orderByColumn.getOtherUsingColumnBoundInfo();
411         if (!groupByColumn.getQualifiedName().equals(orderByColumn.getQualifiedName()) || null == groupByBoundInfo || null == orderByBoundInfo
412                 || !isSameColumnBoundInfo(groupByBoundInfo, orderByBoundInfo)) {
413             return false;
414         }
415         if (null == groupByOtherUsingBoundInfo && null != orderByOtherUsingBoundInfo) {
416             return false;
417         }
418         if (null != groupByOtherUsingBoundInfo && (null == orderByOtherUsingBoundInfo || !isSameColumnBoundInfo(groupByOtherUsingBoundInfo, orderByOtherUsingBoundInfo))) {
419             return false;
420         }
421         return !isAmbiguousProjectionReference(groupByColumn);
422     }
423     
424     private boolean isAmbiguousProjectionReference(final ColumnSegment column) {
425         if (null != column.getNestedObjectAttributes() && !column.getNestedObjectAttributes().isEmpty()) {
426             return true;
427         }
428         if (column.getOwner().isPresent()) {
429             return false;
430         }
431         String identifier = new ProjectionIdentifierExtractEngine(sqlStatement.getDatabaseType()).getIdentifierValue(column.getIdentifier());
432         for (Projection each : projectionsContext.getExpandProjections()) {
433             if (!identifier.equalsIgnoreCase(each.getColumnLabel())) {
434                 continue;
435             }
436             if (each.getAlias().isPresent() || !(each instanceof ColumnProjection) || null == ((ColumnProjection) each).getColumnBoundInfo()
437                     || !isSameColumnBoundInfo(column.getColumnBoundInfo(), ((ColumnProjection) each).getColumnBoundInfo())) {
438                 return true;
439             }
440         }
441         return false;
442     }
443     
444     private boolean isSameColumnBoundInfo(final ColumnSegmentBoundInfo left, final ColumnSegmentBoundInfo right) {
445         if (null == left || null == right || left.getTableSourceType() != right.getTableSourceType()) {
446             return false;
447         }
448         if (!Objects.equals(left.getOriginalDatabase(), right.getOriginalDatabase()) || !Objects.equals(left.getOriginalSchema(), right.getOriginalSchema())) {
449             return false;
450         }
451         return Objects.equals(left.getOriginalTable(), right.getOriginalTable()) && Objects.equals(left.getOriginalColumn(), right.getOriginalColumn());
452     }

454     /**
455      * Find column bound info.
456      *
457      * @param columnIndex column index
458      * @return column bound info
459      */
460     public Optional<ColumnSegmentBoundInfo> findColumnBoundInfo(final int columnIndex) {
461         List<Projection> expandProjections = projectionsContext.getExpandProjections();
462         if (expandProjections.size() < columnIndex) {
463             return Optional.empty();
464         }
465         Projection projection = expandProjections.get(columnIndex - 1);
466         if (projection instanceof ColumnProjection) {
467             return Optional.of(((ColumnProjection) projection).getColumnBoundInfo());
468         }
469         if (projection instanceof SubqueryProjection && ((SubqueryProjection) projection).getProjection() instanceof ColumnProjection) {
470             return Optional.of(((ColumnProjection) ((SubqueryProjection) projection).getProjection()).getColumnBoundInfo());
471         }
472         return Optional.empty();
473     }

493     private boolean containsPivotOutputShorthandProjection() {
494         for (ProjectionSegment each : sqlStatement.getProjections().getProjections()) {
495             if (each instanceof ShorthandProjectionSegment && isPivotOutputShorthandProjection((ShorthandProjectionSegment) each)) {
496                 return true;
497             }
498         }
499         return false;
500     }
501     
502     private boolean isPivotOutputShorthandProjection(final ShorthandProjectionSegment projectionSegment) {
503         if (!sqlStatement.getFrom().isPresent()) {
504             return false;
505         }
506         TableSegment from = sqlStatement.getFrom().get();
507         return projectionSegment.getOwner().isPresent() ? isOwnerPivotTable(projectionSegment.getOwner().get().getIdentifier().getValue(), from) : containsPivotTable(from);
508     }
509     
510     private boolean isOwnerPivotTable(final String owner, final TableSegment tableSegment) {
511         if (tableSegment instanceof JoinTableSegment) {
512             JoinTableSegment joinTableSegment = (JoinTableSegment) tableSegment;
513             return isOwnerPivotTable(owner, joinTableSegment.getLeft()) || isOwnerPivotTable(owner, joinTableSegment.getRight());
514         }
515         return hasPivot(tableSegment) && isOwnerOfTable(owner, tableSegment);
516     }
517     
518     private boolean isOwnerOfTable(final String owner, final TableSegment tableSegment) {
519         return tableSegment.getAliasName().map(optional -> optional.equalsIgnoreCase(owner)).orElse(false)
520                 || tableSegment instanceof SimpleTableSegment && ((SimpleTableSegment) tableSegment).getTableName().getIdentifier().getValue().equalsIgnoreCase(owner);
521     }

523     private boolean containsPivotTable(final TableSegment tableSegment) {
524         if (tableSegment instanceof JoinTableSegment) {
525             JoinTableSegment joinTableSegment = (JoinTableSegment) tableSegment;
526             return containsPivotTable(joinTableSegment.getLeft()) || containsPivotTable(joinTableSegment.getRight());
527         }
528         return hasPivot(tableSegment);
529     }
530     
531     private boolean hasPivot(final TableSegment tableSegment) {
532         return tableSegment instanceof SimpleTableSegment && ((SimpleTableSegment) tableSegment).getPivot().isPresent()
533                 || tableSegment instanceof SubqueryTableSegment && ((SubqueryTableSegment) tableSegment).getPivot().isPresent();
534     }
