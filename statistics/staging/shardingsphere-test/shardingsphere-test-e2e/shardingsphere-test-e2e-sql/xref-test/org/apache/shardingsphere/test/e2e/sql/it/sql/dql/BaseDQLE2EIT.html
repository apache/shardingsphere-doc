82      /**
83       * Execute DQL with concurrent execution control.
84       *
85       * @param context SQL E2E IT context
86       * @param callback DQL execution callback
87       * @throws SQLException SQL exception
88       * @throws IOException IO exception
89       * @throws JAXBException JAXB exception
90       */
91      protected final void executeDQL(final SQLE2EITContext context, final DQLExecutionCallback callback) throws SQLException, IOException, JAXBException {
92          Lock lock = DQLExclusiveExecutionDetector.requiresExclusiveExecution(context.getSQL()) || null != context.getAssertion().getInitialSQL() || null != context.getAssertion().getDestroySQL()
93                  ? DQL_EXECUTION_LOCK.writeLock()
94                  : DQL_EXECUTION_LOCK.readLock();
95          lock.lock();
96          try {
97              callback.execute();
98          } finally {
99              lock.unlock();
100         }
101     }

103     protected final void init(final AssertionTestParameter testParam, final SQLE2EITContext context) throws IOException, JAXBException {
104         fillDataOnlyOnce(testParam);
105         expectedDataSource = getExpectedDataSource(testParam, context);
106         useXMLAsExpectedDataset = null != context.getAssertion().getExpectedDataFile();
107         if (0 != testParam.getTestCaseContext().getTestCase().getDelayAssertionSeconds()) {
108             Awaitility.await().atMost(Duration.ofMinutes(5L)).pollDelay(testParam.getTestCaseContext().getTestCase().getDelayAssertionSeconds(), TimeUnit.SECONDS).until(() -> true);
109         }
110     }

112     private DataSource getExpectedDataSource(final AssertionTestParameter testParam, final SQLE2EITContext context) {
113         Map<String, DataSource> expectedDataSourceMap = getEnvironmentEngine().getExpectedDataSourceMap();
114         if (null != context.getAssertion().getExpectedDataSourceName() && 1 != expectedDataSourceMap.size()) {
115             return expectedDataSourceMap.get(context.getAssertion().getExpectedDataSourceName());
116         }
117         DataSource result = expectedDataSourceMap.get(testParam.getScenario());
118         if (null == result) {
119             result = expectedDataSourceMap.get(testParam.getScenario());
120         }
121         return null == result ? getFirstExpectedDataSource(expectedDataSourceMap.values()) : result;
122     }
123     
124     private void fillDataOnlyOnce(final AssertionTestParameter testParam) {
125         String cacheKey = testParam.getKey() + "-" + System.identityHashCode(getEnvironmentEngine().getActualDataSourceMap());
126         if (FILLED_SUITES.contains(cacheKey)) {
127             return;
128         }
129         synchronized (FILL_DATA_LOCKS.computeIfAbsent(cacheKey, unused -> new Object())) {
130             if (FILLED_SUITES.contains(cacheKey)) {
131                 return;
132             }
133             new DataSetEnvironmentManager(
134                     new ScenarioDataPath(testParam.getScenario(), Type.ACTUAL).getDataSetFile(), getEnvironmentEngine().getActualDataSourceMap(),
135                     testParam.getDatabaseType()).fillData(Collections.emptyList());
136             new DataSetEnvironmentManager(new ScenarioDataPath(testParam.getScenario(), Type.EXPECTED).getDataSetFile(),
137                     getEnvironmentEngine().getExpectedDataSourceMap(), testParam.getDatabaseType()).fillData(Collections.emptyList());
138             FILLED_SUITES.add(cacheKey);
139         }
140     }

142     private DataSource getFirstExpectedDataSource(final Collection<DataSource> dataSources) {
143         return dataSources.isEmpty() ? null : dataSources.iterator().next();
144     }

146     protected final void assertResultSet(final ResultSet actualResultSet, final ResultSet expectedResultSet, final AssertionTestParameter testParam) throws SQLException {
147         assertMetaData(actualResultSet.getMetaData(), expectedResultSet.getMetaData(), testParam);
148         assertRows(actualResultSet, expectedResultSet);
149     }

151     protected final void assertResultSet(final SQLE2EITContext context, final ResultSet resultSet) throws SQLException {
152         assertMetaData(resultSet.getMetaData(), getExpectedColumns(context));
153         assertRows(resultSet, getIgnoreAssertColumns(context), context.getDataSet().getRows());
154     }

156     private Collection<DataSetColumn> getExpectedColumns(final SQLE2EITContext context) {
157         Collection<DataSetColumn> result = new LinkedList<>();
158         for (DataSetMetaData each : context.getDataSet().getMetaDataList()) {
159             result.addAll(each.getColumns());
160         }
161         return result;
162     }
163     
164     private Collection<String> getIgnoreAssertColumns(final SQLE2EITContext context) {
165         Collection<String> result = new LinkedList<>();
166         for (DataSetMetaData each : context.getDataSet().getMetaDataList()) {
167             result.addAll(each.getColumns().stream().filter(DataSetColumn::isIgnoreAssertData).map(DataSetColumn::getName).collect(Collectors.toList()));
168         }
169         return result;
170     }

172     private void assertMetaData(final ResultSetMetaData actualResultSetMetaData, final ResultSetMetaData expectedResultSetMetaData, final AssertionTestParameter testParam) throws SQLException {
173         assertThat(actualResultSetMetaData.getColumnCount(), is(expectedResultSetMetaData.getColumnCount()));
174         for (int i = 0; i < actualResultSetMetaData.getColumnCount(); i++) {
175             assertThat(actualResultSetMetaData.getColumnLabel(i + 1), is(expectedResultSetMetaData.getColumnLabel(i + 1)));
176             assertThat(actualResultSetMetaData.getColumnName(i + 1), is(expectedResultSetMetaData.getColumnName(i + 1)));
177             if ("db_tbl_sql_federation".equals(testParam.getScenario())) {
178                 continue;
179             }
180             if (isCompositeColumn(expectedResultSetMetaData, i + 1)) {
181                 assertThat(actualResultSetMetaData.getColumnType(i + 1), is(expectedResultSetMetaData.getColumnType(i + 1)));
182                 assertThat(actualResultSetMetaData.getColumnTypeName(i + 1), is(expectedResultSetMetaData.getColumnTypeName(i + 1)));
183             } else if ("jdbc".equals(testParam.getAdapter()) && Mode.CLUSTER == testParam.getMode() && "encrypt".equals(testParam.getScenario())
184                     || "MySQL".equals(testParam.getDatabaseType().getType()) && "passthrough".equals(testParam.getScenario())) {
185                 // FIXME correct columnType with proxy adapter and other jdbc scenario
186                 assertThat(actualResultSetMetaData.getColumnType(i + 1), is(expectedResultSetMetaData.getColumnType(i + 1)));
187             }
188         }
189     }

191     private void assertMetaData(final ResultSetMetaData actual, final Collection<DataSetColumn> expected) throws SQLException {
192         assertThat(actual.getColumnCount(), is(expected.size()));
193         int index = 1;
194         for (DataSetColumn each : expected) {
195             assertThat(actual.getColumnLabel(index++).toLowerCase(), is(each.getName().toLowerCase()));
196         }
197     }

199     private boolean isCompositeColumn(final ResultSetMetaData resultSetMetaData, final int columnIndex) throws SQLException {
200         return Types.STRUCT == resultSetMetaData.getColumnType(columnIndex);
201     }
202     
203     private void assertRows(final ResultSet actualResultSet, final ResultSet expectedResultSet) throws SQLException {
204         ResultSetMetaData actualMetaData = actualResultSet.getMetaData();
205         ResultSetMetaData expectedMetaData = expectedResultSet.getMetaData();
206         while (actualResultSet.next()) {
207             assertTrue(expectedResultSet.next(), "Size of actual result set is different with size of expected result set.");
208             assertRow(actualResultSet, actualMetaData, expectedResultSet, expectedMetaData);
209         }
210         assertFalse(expectedResultSet.next(), "Size of actual result set is different with size of expected result set.");
211     }
212     
213     private void assertRows(final ResultSet actual, final Collection<String> notAssertionColumns, final List<DataSetRow> expected) throws SQLException {
214         int rowCount = 0;
215         ResultSetMetaData actualMetaData = actual.getMetaData();
216         while (actual.next()) {
217             assertTrue(rowCount < expected.size(), "Size of actual result set is different with size of expected data set rows.");
218             assertRow(actual, notAssertionColumns, actualMetaData, expected.get(rowCount));
219             rowCount++;
220         }
221         assertThat("Size of actual result set is different with size of expected data set rows.", rowCount, is(expected.size()));
222     }
223     
224     private void assertRow(final ResultSet actualResultSet, final ResultSetMetaData actualMetaData, final ResultSet expectedResultSet, final ResultSetMetaData expectedMetaData) throws SQLException {
225         for (int i = 0; i < actualMetaData.getColumnCount(); i++) {
226             try {
227                 assertThat(actualResultSet.getObject(i + 1), is(expectedResultSet.getObject(i + 1)));
228                 assertThat(actualResultSet.getObject(actualMetaData.getColumnLabel(i + 1)), is(expectedResultSet.getObject(expectedMetaData.getColumnLabel(i + 1))));
229             } catch (final AssertionError ex) {
230                 // FIXME #15593 verify accurate data types
231                 Object actualValue = actualResultSet.getObject(i + 1);
232                 Object expectedValue = expectedResultSet.getObject(i + 1);
233                 if (actualValue instanceof Double || actualValue instanceof Float || actualValue instanceof BigDecimal) {
234                     assertThat(Math.floor(Double.parseDouble(actualValue.toString())), is(Math.floor(Double.parseDouble(expectedValue.toString()))));
235                 } else if (actualValue instanceof Timestamp && expectedValue instanceof LocalDateTime) {
236                     // TODO Since mysql 8.0.23, for the DATETIME type, the mysql driver returns the LocalDateTime type, but the proxy returns the Timestamp type.
237                     assertThat(((Timestamp) actualValue).toLocalDateTime(), is(expectedValue));
238                 } else if (Types.TIMESTAMP == actualMetaData.getColumnType(i + 1) || Types.TIMESTAMP == expectedMetaData.getColumnType(i + 1)) {
239                     Object convertedActualValue = Types.TIMESTAMP == actualMetaData.getColumnType(i + 1) ? formatTimestamp(actualResultSet.getTimestamp(i + 1)) : actualValue;
240                     Object convertedExpectedValue = Types.TIMESTAMP == expectedMetaData.getColumnType(i + 1) ? formatTimestamp(expectedResultSet.getTimestamp(i + 1)) : expectedValue;
241                     assertThat(String.valueOf(convertedActualValue), is(String.valueOf(convertedExpectedValue)));
242                 } else if (expectedValue instanceof Clob) {
243                     assertThat(String.valueOf(actualValue), is(((Clob) expectedValue).getSubString(1, (int) ((Clob) expectedValue).length())));
244                 } else if (actualValue instanceof String && expectedValue instanceof byte[]) {
245                     assertThat(actualValue, is(new String((byte[]) expectedValue)));
246                 } else {
247                     assertThat(String.valueOf(actualValue), is(String.valueOf(expectedValue)));
248                 }
249             }
250         }
251     }
252     
253     private void assertRow(final ResultSet actual, final Collection<String> notAssertionColumns, final ResultSetMetaData actualMetaData, final DataSetRow expected) throws SQLException {
254         int columnIndex = 1;
255         for (String each : expected.splitValues("|")) {
256             String columnLabel = actualMetaData.getColumnLabel(columnIndex);
257             if (!notAssertionColumns.contains(columnLabel)) {
258                 assertObjectValue(actual, columnIndex, columnLabel, each);
259             }
260             columnIndex++;
261         }
262     }
263     
264     private String formatTimestamp(final Timestamp timestamp) {
265         return null == timestamp ? null : timestamp.toLocalDateTime().format(DateTimeFormatterFactory.getDatetimeFormatter());
266     }
267     
268     private void assertObjectValue(final ResultSet actual, final int columnIndex, final String columnLabel, final String expected) throws SQLException {
269         assertThat(String.valueOf(actual.getObject(columnIndex)).trim(), is(expected));
270         assertThat(String.valueOf(actual.getObject(columnLabel)).trim(), is(expected));
271     }
