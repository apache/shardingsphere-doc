66      /**
67       * Register storage unit.
68       *
69       * @param storageUnitName storage unit name
70       * @throws SQLException SQL exception
71       */
72      public void registerStorageUnit(final String storageUnitName) throws SQLException {
73          String username = ProxyDatabaseTypeUtils.isOracleBranch(containerComposer.getDatabaseType()) ? storageUnitName : containerComposer.getUsername();
74          String registerStorageUnitSQL = ("REGISTER STORAGE UNIT ${ds} ( "
75                  + "URL='${url}', "
76                  + "USER='${user}', "
77                  + "PASSWORD='${password}', "
78                  + "PROPERTIES('driverClassName'='${driverClassName}'))")
79                  .replace("${ds}", storageUnitName)
80                  .replace("${user}", username)
81                  .replace("${password}", containerComposer.getPassword())
82                  .replace("${driverClassName}", containerComposer.getDriverClassName())
83                  .replace("${url}", containerComposer.getActualJdbcUrlTemplate(storageUnitName,
84                          RunEnvironment.Type.DOCKER == E2ETestEnvironment.getInstance().getRunEnvironment().getType()));
85          containerComposer.proxyExecuteOneSQLWithLog(registerStorageUnitSQL, 0);
86          int timeout = containerComposer.getDatabaseType() instanceof OpenGaussDatabaseType ? 60 : 10;
87          Awaitility.waitAtMost(timeout, TimeUnit.SECONDS).ignoreExceptions().pollInterval(3L, TimeUnit.SECONDS).until(() -> containerComposer.showStorageUnitsName().contains(storageUnitName));
88      }

200     /**
201      * Wait job preparing stage finished.
202      *
203      * @param jobId job id
204      */
205     public void waitJobPreparingStageFinished(final String jobId) {
206         String sql = buildShowJobStatusDistSQL(jobId);
207         for (int i = 0; i < 5; i++) {
208             List<Map<String, Object>> jobStatusRecords = containerComposer.queryForListWithLog(sql);
209             log.info("Wait job preparing stage finished, job status records: {}", jobStatusRecords);
210             Set<String> statusSet = jobStatusRecords.stream().map(each -> String.valueOf(each.get("status"))).collect(Collectors.toSet());
211             if (statusSet.contains(JobStatus.PREPARING.name()) || statusSet.contains(JobStatus.RUNNING.name())) {
212                 containerComposer.sleepSeconds(2);
213                 continue;
214             }
215             break;
216         }
217     }

219     /**
220      * Wait job incremental stage started.
221      *
222      * @param jobId job id
223      * @throws IllegalStateException if job incremental stage not started
224      */
225     public void waitJobIncrementalStageStarted(final String jobId) {
226         String sql = buildShowJobStatusDistSQL(jobId);
227         JobStatus jobStatus = JobStatus.EXECUTE_INCREMENTAL_TASK;
228         for (int i = 0; i < 10; i++) {
229             List<Map<String, Object>> jobStatusRecords = containerComposer.queryForListWithLog(sql);
230             log.info("Wait job Incremental stage started, job status records: {}", jobStatusRecords);
231             List<String> statusList = jobStatusRecords.stream().map(each -> String.valueOf(each.get("status"))).toList();
232             if (statusList.stream().allMatch(each -> each.equals(jobStatus.name()))) {
233                 return;
234             }
235             containerComposer.sleepSeconds(2);
236         }
237         throw new IllegalStateException("Job status not reached: " + jobStatus);
238     }

240     /**
241      * Wait job incremental stage finished.
242      *
243      * @param jobId job id
244      * @return result
245      */
246     public List<Map<String, Object>> waitJobIncrementalStageFinished(final String jobId) {
247         String sql = buildShowJobStatusDistSQL(jobId);
248         for (int i = 0; i < 10; i++) {
249             List<Map<String, Object>> jobStatusRecords = containerComposer.queryForListWithLog(sql);
250             log.info("Wait job incremental stage finished, job status records: {}", jobStatusRecords);
251             Set<String> statusSet = new HashSet<>(jobStatusRecords.size(), 1F);
252             List<Integer> incrementalIdleSecondsList = new LinkedList<>();
253             for (Map<String, Object> each : jobStatusRecords) {
254                 assertTrue(Strings.isNullOrEmpty((String) each.get("error_message")), "error_message: `" + each.get("error_message") + "`");
255                 statusSet.add(each.get("status").toString());
256                 String incrementalIdleSeconds = (String) each.get("incremental_idle_seconds");
257                 incrementalIdleSecondsList.add(Strings.isNullOrEmpty(incrementalIdleSeconds) ? 0 : Integer.parseInt(incrementalIdleSeconds));
258             }
259             if (Collections.min(incrementalIdleSecondsList) <= 5) {
260                 containerComposer.sleepSeconds(3);
261                 continue;
262             }
263             if (1 == statusSet.size() && statusSet.contains(JobStatus.EXECUTE_INCREMENTAL_TASK.name())) {
264                 return jobStatusRecords;
265             }
266         }
267         return Collections.emptyList();
268     }

286     /**
287      * Start check.
288      *
289      * @param jobId job id
290      * @param algorithmType algorithm type
291      * @param algorithmProps algorithm properties
292      * @throws SQLException SQL exception
293      */
294     public void startCheck(final String jobId, final String algorithmType, final Map<String, String> algorithmProps) throws SQLException {
295         containerComposer.proxyExecuteWithLog(buildConsistencyCheckDistSQL(jobId, algorithmType, algorithmProps), 0);
296     }
297     
298     private String buildConsistencyCheckDistSQL(final String jobId, final String algorithmType, final Map<String, String> algorithmProps) {
299         if (null == algorithmProps || algorithmProps.isEmpty()) {
300             return String.format("CHECK %s %s BY TYPE (NAME='%s')", jobTypeName, jobId, algorithmType);
301         }
302         String sqlTemplate = "CHECK %s %s BY TYPE (NAME='%s', PROPERTIES("
303                 + algorithmProps.entrySet().stream().map(entry -> String.format("'%s'='%s'", entry.getKey(), entry.getValue())).collect(Collectors.joining(","))
304                 + "))";
305         return String.format(sqlTemplate, jobTypeName, jobId, algorithmType);
306     }

308     /**
309      * Verify check.
310      *
311      * @param jobId job id
312      */
313     public void verifyCheck(final String jobId) {
314         List<Map<String, Object>> checkStatusRecords = Collections.emptyList();
315         for (int i = 0; i < 10; i++) {
316             checkStatusRecords = queryCheckJobStatus(jobId);
317             if (checkStatusRecords.isEmpty()) {
318                 containerComposer.sleepSeconds(3);
319                 continue;
320             }
321             List<String> checkEndTimeList = checkStatusRecords.stream().map(map -> map.get("check_end_time").toString()).filter(each -> !Strings.isNullOrEmpty(each)).toList();
322             if (checkEndTimeList.size() == checkStatusRecords.size()) {
323                 break;
324             } else {
325                 containerComposer.sleepSeconds(3);
326             }
327         }
328         log.info("Verify check, results: {}", checkStatusRecords);
329         assertFalse(checkStatusRecords.isEmpty());
330         for (Map<String, Object> entry : checkStatusRecords) {
331             assertThat(entry.get("inventory_finished_percentage").toString(), is("100"));
332             assertTrue(Boolean.parseBoolean(entry.get("result").toString()));
333         }
334     }
