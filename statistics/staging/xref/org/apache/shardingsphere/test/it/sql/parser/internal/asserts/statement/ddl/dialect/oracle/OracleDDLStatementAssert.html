53  /**
54   * DDL statement assert for Oracle.
55   */
56  @NoArgsConstructor(access = AccessLevel.PRIVATE)
57  public final class OracleDDLStatementAssert {
58      
59      /**
60       * Assert DDL statement is correct with expected parser result.
61       *
62       * @param assertContext assert context
63       * @param actual actual DDL statement
64       * @param expected expected parser result
65       */
66      public static void assertIs(final SQLCaseAssertContext assertContext, final DDLStatement actual, final SQLParserTestCase expected) {
67          if (actual instanceof OracleAlterSessionStatement) {
68              OracleAlterSessionStatementAssert.assertIs(assertContext, (OracleAlterSessionStatement) actual, (OracleAlterSessionStatementTestCase) expected);
69          } else if (actual instanceof OracleAlterSystemStatement) {
70              OracleAlterSystemStatementAssert.assertIs(assertContext, (OracleAlterSystemStatement) actual, (OracleAlterSystemStatementTestCase) expected);
71          } else if (actual instanceof OracleAnalyzeStatement) {
72              OracleAnalyzeStatementAssert.assertIs(assertContext, (OracleAnalyzeStatement) actual, (OracleAnalyzeStatementTestCase) expected);
73          } else if (actual instanceof OracleAssociateStatisticsStatement) {
74              OracleAssociateStatisticsStatementAssert.assertIs(assertContext, (OracleAssociateStatisticsStatement) actual, (OracleAssociateStatisticsStatementTestCase) expected);
75          } else if (actual instanceof OracleDisassociateStatisticsStatement) {
76              OracleDisassociateStatisticsStatementAssert.assertIs(assertContext, (OracleDisassociateStatisticsStatement) actual, (OracleDisassociateStatisticsStatementTestCase) expected);
77          } else if (actual instanceof OracleAuditStatement) {
78              OracleAuditStatementAssert.assertIs(assertContext, (OracleAuditStatement) actual, (OracleAuditStatementTestCase) expected);
79          } else if (actual instanceof OracleNoAuditStatement) {
80              OracleNoAuditStatementAssert.assertIs(assertContext, (OracleNoAuditStatement) actual, (OracleNoAuditStatementTestCase) expected);
81          } else if (actual instanceof OracleFlashbackTableStatement) {
82              OracleFlashbackTableStatementAssert.assertIs(assertContext, (OracleFlashbackTableStatement) actual, (OracleFlashbackTableStatementTestCase) expected);
83          } else if (actual instanceof OraclePurgeStatement) {
84              OraclePurgeStatementAssert.assertIs(assertContext, (OraclePurgeStatement) actual, (OraclePurgeStatementTestCase) expected);
85          }
86      }
87  }
