25  import org.apache.shardingsphere.sql.parser.statement.core.segment.dml.expr.FunctionSegment;

27  import org.apache.shardingsphere.sql.parser.statement.core.statement.SQLStatement;

35  import org.apache.shardingsphere.sql.parser.statement.core.statement.type.dml.CallStatement;
36  import org.apache.shardingsphere.sql.parser.statement.core.statement.type.dml.SelectStatement;

38  import java.util.List;
39  import java.util.Locale;
40  import java.util.regex.Pattern;
41  
42  final class SQLStatementSafetyValidator {
43      
44      private static final Pattern WHITESPACE_PATTERN = Pattern.compile("\\s+");
45      
46      private static final List<String> SIDE_EFFECTING_FUNCTION_NAMES = List.of("NEXTVAL", "NEXT VALUE FOR", "SETVAL", "GET_LOCK", "RELEASE_LOCK", "RELEASE_ALL_LOCKS",
47              "PG_ADVISORY_LOCK", "PG_ADVISORY_XACT_LOCK", "PG_TRY_ADVISORY_LOCK", "PG_TRY_ADVISORY_XACT_LOCK", "PG_ADVISORY_UNLOCK", "PG_ADVISORY_UNLOCK_ALL",
48              "SET_CONFIG", "PG_REPLICATION_SLOT_ADVANCE", "PG_LOGICAL_SLOT_GET_CHANGES", "PG_LOGICAL_SLOT_GET_BINARY_CHANGES", "PG_LOGICAL_EMIT_MESSAGE", "PG_SWITCH_WAL",
49              "PG_RELOAD_CONF", "PG_CANCEL_BACKEND", "PG_TERMINATE_BACKEND");
50      
51      private static final List<String> METADATA_LOOKUP_FUNCTION_NAMES = List.of("TO_REGCLASS", "TO_REGTYPE", "TO_REGPROC", "TO_REGPROCEDURE", "TO_REGOPER",
52              "TO_REGOPERATOR", "TO_REGNAMESPACE", "TO_REGROLE", "OBJECT_ID");
53      
54      void checkLeadingStatement(final String upperSql, final boolean executableComment) {
55          if (executableComment || upperSql.startsWith("USE ") || upperSql.startsWith("SET ") || upperSql.startsWith("COPY ") || upperSql.startsWith("LOAD ")
56                  || upperSql.startsWith("CALL ") || isAlterSystemStatement(upperSql)) {
57              throw new MCPBannedSQLStatementException();
58          }
59          if (isMetadataIntrospectionStatement(upperSql)) {
60              throw new MetadataIntrospectionSQLStatementException(extractStatementType(upperSql));
61          }
62      }
63      
64      void checkParsedStatement(final SQLStatement sqlStatement) {
65          new SQLStatementTreeWalker(this::checkStatement, this::checkExpression).walk(sqlStatement);
66      }
67      
68      private boolean isBannedStatementType(final SQLStatement sqlStatement) {
69          return switch (sqlStatement) {
70              case SetStatement ignored -> true;
71              case CallStatement ignored -> true;
72              case CreateUserStatement ignored -> true;
73              case AlterUserStatement ignored -> true;
74              case DropUserStatement ignored -> true;
75              case CreateRoleStatement ignored -> true;
76              case AlterRoleStatement ignored -> true;
77              case DropRoleStatement ignored -> true;
78              case null, default -> false;
79          };
80      }
81      
82      private boolean containsExecutableComment(final SQLStatement sqlStatement) {
83          for (CommentSegment each : sqlStatement.getComments()) {
84              String text = each.getText().trim();
85              if (text.startsWith("/*!") || text.toUpperCase(Locale.ENGLISH).startsWith("/*M!")) {
86                  return true;
87              }
88          }
89          return false;
90      }
91      
92      private void checkStatement(final SQLStatement sqlStatement) {
93          if (isBannedStatementType(sqlStatement) || containsExecutableComment(sqlStatement)) {
94              throw new MCPBannedSQLStatementException();
95          }
96          if (sqlStatement instanceof SelectStatement select) {
97              if (select.getInto().isPresent() || select.getOutfile().isPresent()) {
98                  throw new MCPBannedSQLStatementException();
99              }
100             if (select.getLock().isPresent()) {
101                 throw new MCPLockingReadStatementException();
102             }
103         }
104     }
105     
106     private void checkExpression(final ExpressionSegment expression) {
107         if (expression instanceof FunctionSegment function) {
108             checkFunction(function);
109         } else if (expression instanceof BinaryOperationExpression binary && ":=".equals(binary.getOperator())) {
110             throw new MCPBannedSQLStatementException();
111         } else if (expression instanceof ColumnSegment column) {
112             if (column.getOwner().isPresent() && "NEXTVAL".equalsIgnoreCase(column.getIdentifier().getValue())) {
113                 throw new MCPBannedSQLStatementException();
114             }
115         }
116     }
117     
118     private void checkFunction(final FunctionSegment function) {
119         String functionName = function.getFunctionName().toUpperCase(Locale.ENGLISH);
120         if (SIDE_EFFECTING_FUNCTION_NAMES.contains(functionName) || METADATA_LOOKUP_FUNCTION_NAMES.contains(functionName)
121                 || "LAST_INSERT_ID".equals(functionName) && !function.getParameters().isEmpty()) {
122             throw new MCPBannedSQLStatementException();
123         }
124     }
125     
126     private boolean isMetadataIntrospectionStatement(final String upperSql) {
127         return "SHOW".equals(upperSql) || upperSql.startsWith("SHOW ") || "DESCRIBE".equals(upperSql) || upperSql.startsWith("DESCRIBE ")
128                 || "DESC".equals(upperSql) || upperSql.startsWith("DESC ");
129     }
130     
131     private boolean isAlterSystemStatement(final String upperSql) {
132         String[] words = WHITESPACE_PATTERN.split(upperSql, 3);
133         return 2 <= words.length && "ALTER".equals(words[0]) && "SYSTEM".equals(words[1]);
134     }
135     
136     private String extractStatementType(final String upperSql) {
137         if (upperSql.startsWith("SHOW RULES USED STORAGE UNIT")) {
138             return "SHOW RULES USED STORAGE UNIT";
139         }
140         if (upperSql.startsWith("SHOW STORAGE UNITS")) {
141             return "SHOW STORAGE UNITS";
142         }
143         if (upperSql.startsWith("SHOW DEFAULT SINGLE TABLE STORAGE UNIT")) {
144             return "SHOW DEFAULT SINGLE TABLE STORAGE UNIT";
145         }
146         if (upperSql.startsWith("SHOW SINGLE TABLES")) {
147             return "SHOW SINGLE TABLES";
148         }
149         if (upperSql.startsWith("SHOW SINGLE TABLE")) {
150             return "SHOW SINGLE TABLE";
151         }
152         return WHITESPACE_PATTERN.split(upperSql)[0];
153     }
154 }
