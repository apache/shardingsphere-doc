81      @Test
82      void assertAdviceWhenCallableSucceeded() throws ReflectiveOperationException {
83          List<String> queue = new LinkedList<>();
84          Map<String, Collection<StaticMethodAdvice>> advices = new LinkedHashMap<>(2, 1F);
85          advices.put("foo", Collections.singletonList(new RecordingStaticMethodAdvice(queue, "plain")));
86          advices.put("bar", Collections.singletonList(new ConfigurableStaticMethodAdvice(queue, "config", false, false, false, false)));
87          StaticMethodAdviceExecutor executor = new StaticMethodAdviceExecutor(advices);
88          Method method = TargetObjectFixture.class.getMethod("staticCall", List.class);
89          Callable<Object> callable = () -> {
90              queue.add("origin call");
91              return "result";
92          };
93          Object actualResult = executor.advice(TargetObjectFixture.class, method, new Object[]{queue}, callable);
94          assertThat(actualResult, is("result"));
95          assertThat(queue, is(Arrays.asList("plain before foo", "origin call", "plain after foo")));
96      }

98      @Test
99      void assertAdviceWhenCallableThrows() throws ReflectiveOperationException {
100         List<String> queue = new LinkedList<>();
101         Map<String, Collection<StaticMethodAdvice>> advices = new LinkedHashMap<>(2, 1F);
102         advices.put("foo", Collections.singletonList(new RecordingStaticMethodAdvice(queue, "plain")));
103         advices.put("bar", Collections.singletonList(new ConfigurableStaticMethodAdvice(queue, "config", true, false, false, true)));
104         StaticMethodAdviceExecutor executor = new StaticMethodAdviceExecutor(advices);
105         Method method = TargetObjectFixture.class.getMethod("staticCallWhenExceptionThrown", List.class);
106         Callable<Object> callable = () -> {
107             throw new IllegalStateException("callable error");
108         };
109         assertThrows(IllegalStateException.class, () -> executor.advice(TargetObjectFixture.class, method, new Object[]{queue}, callable));
110         assertThat(queue, is(Arrays.asList("plain before foo", "config before bar", "plain throw foo", "plain after foo", "config after bar")));
111         assertLog("Failed to execute the error handler of method `{0}` in class `{1}`, {2}.", method.getName(), "callable error");
112     }

114     private void assertLog(final String expectedMessage, final String methodName, final String errorMessage) {
115         assertThat(records.size(), is(1));
116         LogRecord actual = records.get(0);
117         assertThat(actual.getLevel(), is(Level.SEVERE));
118         assertThat(actual.getMessage(), is(expectedMessage));
119         assertThat(actual.getParameters(), is(new String[]{methodName, TargetObjectFixture.class.getName(), errorMessage}));
120         records.remove(0);
121     }

236         }

243     @RequiredArgsConstructor
244     private static final class RecordingStaticMethodAdvice implements StaticMethodAdvice {
245         
246         private final List<String> queue;
247         
248         private final String label;
249         
250         @Override
251         public void beforeMethod(final Class<?> clazz, final TargetAdviceMethod method, final Object[] args, final String pluginType) {
252             queue.add(label + " before " + pluginType);
253         }
254         
255         @Override
256         public void afterMethod(final Class<?> clazz, final TargetAdviceMethod method, final Object[] args, final Object result, final String pluginType) {
257             queue.add(label + " after " + pluginType);
258         }
259         
260         @Override
261         public void onThrowing(final Class<?> clazz, final TargetAdviceMethod method, final Object[] args, final Throwable throwable, final String pluginType) {
262             queue.add(label + " throw " + pluginType);
263         }
264     }
265     
266     @RequiredArgsConstructor
267     private static final class ConfigurableStaticMethodAdvice implements StaticMethodAdvice, AgentPluginEnable {
268         
269         private final List<String> queue;
270         
271         private final String label;
272         
273         private final boolean pluginEnabled;
274         
275         private final boolean throwBefore;
276         
277         private final boolean throwAfter;
278         
279         private final boolean throwOnThrow;
280         
281         @Override
282         public boolean isPluginEnabled() {
283             return pluginEnabled;
284         }
285         
286         @Override
287         public void beforeMethod(final Class<?> clazz, final TargetAdviceMethod method, final Object[] args, final String pluginType) {
288             if (throwBefore) {
289                 throw new IllegalStateException("before");
290             }
291             queue.add(label + " before " + pluginType);
292         }
293         
294         @Override
295         public void afterMethod(final Class<?> clazz, final TargetAdviceMethod method, final Object[] args, final Object result, final String pluginType) {
296             if (throwAfter) {
297                 throw new IllegalStateException("after");
298             }
299             queue.add(label + " after " + pluginType);
300         }
301         
302         @Override
303         public void onThrowing(final Class<?> clazz, final TargetAdviceMethod method, final Object[] args, final Throwable throwable, final String pluginType) {
304             if (throwOnThrow) {
305                 throw new IllegalStateException("throw");
306             }
307             queue.add(label + " throw " + pluginType);
308         }
