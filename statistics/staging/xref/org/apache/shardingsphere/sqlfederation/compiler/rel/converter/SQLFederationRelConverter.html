70  public final class SQLFederationRelConverter {

72      private final SqlToRelConverter sqlToRelConverter;
73      
74      public SQLFederationRelConverter(final CompilerContext compilerContext, final List<String> schemaPath, final DatabaseType databaseType, final Convention convention) {
75          RelDataTypeFactory typeFactory = SQLFederationDataTypeFactory.getInstance();
76          CalciteConnectionConfig connectionConfig = compilerContext.getConnectionConfig();
77          CalciteCatalogReader catalogReader = new SQLFederationCatalogReader(compilerContext.getCalciteSchema(), schemaPath, typeFactory, connectionConfig);
78          SqlValidator validator = createSqlValidator(catalogReader, typeFactory, connectionConfig, compilerContext.getOperatorTables());
79          RelOptCluster relOptCluster = createRelOptCluster(typeFactory, convention);
80          sqlToRelConverter = createSqlToRelConverter(catalogReader, validator, relOptCluster, compilerContext.getSqlParserRule(), databaseType, true);
81      }
82      
83      private SqlValidator createSqlValidator(final CalciteCatalogReader catalogReader, final RelDataTypeFactory typeFactory,
84                                              final CalciteConnectionConfig connectionConfig, final Collection<SqlOperatorTable> operatorTables) {
85          SqlValidator.Config validatorConfig = SqlValidator.Config.DEFAULT.withLenientOperatorLookup(connectionConfig.lenientOperatorLookup()).withConformance(connectionConfig.conformance())
86                  .withDefaultNullCollation(connectionConfig.defaultNullCollation()).withIdentifierExpansion(true);
87          SqlOperatorTable sqlOperatorTable = getSQLOperatorTable(operatorTables, catalogReader);
88          return SqlValidatorUtil.newValidator(sqlOperatorTable, catalogReader, typeFactory, validatorConfig);
89      }
90      
91      private static SqlOperatorTable getSQLOperatorTable(final Collection<SqlOperatorTable> operatorTables, final CalciteCatalogReader catalogReader) {
92          Collection<SqlOperatorTable> allOperatorTables = new LinkedList<>(operatorTables);
93          allOperatorTables.add(catalogReader);
94          return SqlOperatorTables.chain(allOperatorTables);
95      }
96      
97      private SqlToRelConverter createSqlToRelConverter(final CalciteCatalogReader catalogReader, final SqlValidator validator, final RelOptCluster cluster, final SQLParserRule sqlParserRule,
98                                                        final DatabaseType databaseType, final boolean needsViewExpand) {
99          ViewExpander expander = needsViewExpand
100                 ? new ShardingSphereViewExpander(sqlParserRule, databaseType, createSqlToRelConverter(catalogReader, validator, cluster, sqlParserRule, databaseType, false))
101                 : (rowType, queryString, schemaPath, viewPath) -> null;
102         // TODO remove withRemoveSortInSubQuery when calcite can expand view which contains order by correctly
103         Config converterConfig = SqlToRelConverter.config().withTrimUnusedFields(true).withRemoveSortInSubQuery(false);
104         return new SqlToRelConverter(expander, validator, catalogReader, cluster, SQLFederationRelConverter::getConvertlet, converterConfig);
105     }
106     
107     private static @Nullable SqlRexConvertlet getConvertlet(final SqlCall call) {
108         if (call.getKind().belongsTo(SqlKind.ORDER_COMPARISON)) {
109             return SQLFederationRelConverter::convertOrderComparison;
110         }
111         return call.getOperator() instanceof SqlBetweenOperator ? SQLFederationRelConverter::convertBetween : StandardConvertletTable.INSTANCE.get(call);
112     }
113     
114     private static RexNode convertOrderComparison(final SqlRexContext context, final SqlCall call) {
115         RexBuilder rexBuilder = context.getRexBuilder();
116         List<RexNode> operands = convertOperands(context, call);
117         return rexBuilder.makeCall(call.getParserPosition(), rexBuilder.deriveReturnType(call.getOperator(), operands), call.getOperator(), operands);
118     }
119     
120     private static RexNode convertBetween(final SqlRexContext context, final SqlCall call) {
121         List<RexNode> operands = convertOperands(context, call);
122         RexNode value = operands.get(SqlBetweenOperator.VALUE_OPERAND);
123         RexNode lower = operands.get(SqlBetweenOperator.LOWER_OPERAND);
124         RexNode upper = operands.get(SqlBetweenOperator.UPPER_OPERAND);
125         RexBuilder rexBuilder = context.getRexBuilder();
126         RexNode firstRange = rexBuilder.makeCall(call.getParserPosition(), SqlStdOperatorTable.AND,
127                 rexBuilder.makeCall(call.getParserPosition(), SqlStdOperatorTable.GREATER_THAN_OR_EQUAL, value, lower),
128                 rexBuilder.makeCall(call.getParserPosition(), SqlStdOperatorTable.LESS_THAN_OR_EQUAL, value, upper));
129         SqlBetweenOperator operator = (SqlBetweenOperator) call.getOperator();
130         RexNode result;
131         switch (operator.flag) {
132             case ASYMMETRIC:
133                 result = firstRange;
134                 break;
135             case SYMMETRIC:
136                 RexNode secondRange = rexBuilder.makeCall(call.getParserPosition(), SqlStdOperatorTable.AND,
137                         rexBuilder.makeCall(call.getParserPosition(), SqlStdOperatorTable.GREATER_THAN_OR_EQUAL, value, upper),
138                         rexBuilder.makeCall(call.getParserPosition(), SqlStdOperatorTable.LESS_THAN_OR_EQUAL, value, lower));
139                 result = rexBuilder.makeCall(call.getParserPosition(), SqlStdOperatorTable.OR, firstRange, secondRange);
140                 break;
141             default:
142                 throw Util.unexpected(operator.flag);
143         }
144         return operator.isNegated() ? rexBuilder.makeCall(call.getParserPosition(), SqlStdOperatorTable.NOT, result) : result;
145     }
146     
147     private static List<RexNode> convertOperands(final SqlRexContext context, final SqlCall call) {
148         RexBuilder rexBuilder = context.getRexBuilder();
149         List<RexNode> result = new ArrayList<>(call.operandCount());
150         for (SqlNode each : call.getOperandList()) {
151             result.add(context.convertExpression(each));
152         }
153         List<RelDataType> operandTypes = context.getValidator().getValidatedOperandTypes(call);
154         if (null != operandTypes) {
155             for (int i = 0; i < result.size(); i++) {
156                 result.set(i, rexBuilder.ensureType(call.getParserPosition(), operandTypes.get(i), result.get(i), true));
157             }
158         }
159         RelDataType consistentType = getConsistentType(context, RexUtil.types(result));
160         if (null != consistentType) {
161             result.replaceAll(each -> rexBuilder.ensureType(call.getParserPosition(), consistentType, each, true));
162         }
163         return result;
164     }
165     
166     private static @Nullable RelDataType getConsistentType(final SqlRexContext context, final List<RelDataType> types) {
167         if (SqlTypeUtil.areSameFamily(types)) {
168             return null;
169         }
170         List<RelDataType> nonCharacterTypes = new ArrayList<>(types.size());
171         for (RelDataType each : types) {
172             if (SqlTypeFamily.CHARACTER != each.getFamily()) {
173                 nonCharacterTypes.add(each);
174             }
175         }
176         if (nonCharacterTypes.size() < types.size()) {
177             RelDataTypeFamily family = nonCharacterTypes.get(0).getFamily();
178             if (SqlTypeFamily.INTEGER == family || SqlTypeFamily.NUMERIC == family) {
179                 nonCharacterTypes.add(context.getTypeFactory().createSqlType(SqlTypeName.BIGINT));
180             }
181         }
182         return context.getTypeFactory().leastRestrictive(nonCharacterTypes);
183     }
184     
185     private RelOptCluster createRelOptCluster(final RelDataTypeFactory typeFactory, final Convention convention) {
186         RelOptPlanner volcanoPlanner = SQLFederationPlannerBuilder.buildVolcanoPlanner(convention);
187         return RelOptCluster.create(volcanoPlanner, new RexBuilder(typeFactory));
188     }

207     public RelRoot convertQuery(final SqlNode sqlNode, final boolean needsValidation, final boolean top) {
208         return sqlToRelConverter.convertQuery(sqlNode, needsValidation, top);
