49  class ShardingValueTypeConvertUtilsTest {
50      
51      @ParameterizedTest(name = "{0}")
52      @MethodSource("convertToTargetTypeArguments")
53      void assertConvertToTargetType(final String name, final Comparable<?> value, final Class<?> targetType, final Object expected) {
54          assertThat(ShardingValueTypeConvertUtils.convertToTargetType(value, targetType), is(expected));
55      }
56      
57      @ParameterizedTest(name = "{0}")
58      @ValueSource(classes = {Instant.class, LocalDate.class, LocalDateTime.class, Time.class})
59      void assertConvertToTargetTypeWithInvalidTemporalValue(final Class<?> targetType) {
60          assertThrows(DateTimeParseException.class, () -> ShardingValueTypeConvertUtils.convertToTargetType("2024-08-05T25:30:45", targetType));
61      }
62      
63      @ParameterizedTest(name = "{0}")
64      @MethodSource("convertCollectionTypeArguments")
65      void assertConvertCollectionType(final String name, final Collection<Comparable<?>> source, final Class<?> targetType, final int expectedSize, final Class<?> expectedElementType) {
66          Collection<Comparable<?>> actual = ShardingValueTypeConvertUtils.convertCollectionType(source, targetType);
67          assertThat(actual.size(), is(expectedSize));
68          for (Comparable<?> each : actual) {
69              assertThat(each, isA(expectedElementType));
70          }
71      }
