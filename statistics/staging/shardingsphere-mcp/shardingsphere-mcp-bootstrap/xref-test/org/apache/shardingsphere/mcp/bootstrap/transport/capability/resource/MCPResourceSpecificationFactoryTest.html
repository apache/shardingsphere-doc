94      @Test
95      void assertCreateResourceSpecificationsHandleReadResource() {
96          SyncResourceSpecification actualSpecification = findResourceSpecification(
97                  new MCPResourceSpecificationFactory(createRuntimeContext()).createResourceSpecifications(), "shardingsphere://capabilities");
98          ReadResourceResult actual = actualSpecification.readHandler().apply(createExchange(), new ReadResourceRequest("shardingsphere://capabilities"));

103         assertTrue(actualContents.text().contains("\"response_mode\":\"catalog\""));
104         assertTrue(actualContents.text().contains("\"resourceNavigation\""));
105         assertTrue(actualContents.text().contains("\"to\":\"shardingsphere://guidance\""));
106     }

108     @Test
109     void assertReadResourceUsesExchangeSession() {
110         MCPResourceDescriptor descriptor = new MCPResourceDescriptor("shardingsphere://session", "session", "Session", "Read session.", "application/json",
111                 MCPResourceAnnotations.EMPTY, Map.of());
112         McpSyncServerExchange exchange = createExchange();
113         ArgumentCaptor<MCPFeatureRuntimeRequestContext> requestContextCaptor = ArgumentCaptor.forClass(MCPFeatureRuntimeRequestContext.class);
114         try (MockedStatic<ResourceDefinitionRegistry> mocked = mockStatic(ResourceDefinitionRegistry.class)) {
115             mocked.when(ResourceDefinitionRegistry::getSupportedResourceDescriptors).thenReturn(List.of(descriptor));
116             mocked.when(() -> ResourceDefinitionRegistry.dispatch(any(MCPFeatureRuntimeRequestContext.class), eq("shardingsphere://session")))
117                     .thenReturn(Optional.of(new MCPMapPayload(Map.of())));
118             SyncResourceSpecification actualSpecification = findResourceSpecification(
119                     new MCPResourceSpecificationFactory(createRuntimeContext()).createResourceSpecifications(), "shardingsphere://session");
120             actualSpecification.readHandler().apply(exchange, new ReadResourceRequest("shardingsphere://session"));
121             mocked.verify(() -> ResourceDefinitionRegistry.dispatch(requestContextCaptor.capture(), eq("shardingsphere://session")));
122         }
123         assertThat(requestContextCaptor.getValue().getSessionIdentity().getSessionId(), is("session-1"));
124     }

166     @Test
167     void assertReadResourceSanitizesRuntimeFailure(final LogCaptureAssertion logCaptureAssertion) {
168         MCPResourceDescriptor descriptor = new MCPResourceDescriptor("shardingsphere://runtime-error", "runtime-error", "Runtime Error", "Read runtime error.", "application/json",
169                 MCPResourceAnnotations.EMPTY, Map.of());
170         try (MockedStatic<ResourceDefinitionRegistry> mocked = mockStatic(ResourceDefinitionRegistry.class)) {
171             mocked.when(ResourceDefinitionRegistry::getSupportedResourceDescriptors).thenReturn(List.of(descriptor));
172             RuntimeException expectedException = new RuntimeException("runtime failure");
173             mocked.when(() -> ResourceDefinitionRegistry.dispatch(any(MCPFeatureRuntimeRequestContext.class), eq("shardingsphere://runtime-error"))).thenThrow(expectedException);
174             SyncResourceSpecification actualSpecification = findResourceSpecification(
175                     new MCPResourceSpecificationFactory(createRuntimeContext()).createResourceSpecifications(), "shardingsphere://runtime-error");
176             McpError actual = assertThrows(McpError.class,
177                     () -> actualSpecification.readHandler().apply(createExchange(), new ReadResourceRequest("shardingsphere://runtime-error")));
178             assertThat(actual.getJsonRpcError().code(), is(McpSchema.ErrorCodes.INTERNAL_ERROR));
179             assertThat(actual.getJsonRpcError().message(), is("Service is temporarily unavailable."));
180             assertFalse(String.valueOf(actual.getJsonRpcError().data()).contains("runtime failure"));
181             logCaptureAssertion.assertErrorLog(actualException -> assertThat(((ThrowableProxy) actualException).getThrowable(), sameInstance(expectedException)));
182         }
183     }

185     @Test
186     void assertReadResourceSanitizesSerializationFailure(final LogCaptureAssertion logCaptureAssertion) {
187         MCPResourceDescriptor descriptor = new MCPResourceDescriptor("shardingsphere://foo", "foo", "Foo", "Read foo.", "application/json", MCPResourceAnnotations.EMPTY, Map.of());
188         Map<String, Object> payload = Map.of("foo_field", "foo_value");
189         IllegalStateException expectedCause = new IllegalStateException("foo_sensitive_detail");
190         JsonException expectedException = mock(JsonException.class);
191         when(expectedException.getMessage()).thenReturn("foo_sensitive_detail");
192         when(expectedException.getCause()).thenReturn(expectedCause);
193         try (
194                 MockedStatic<ResourceDefinitionRegistry> mockedRegistry = mockStatic(ResourceDefinitionRegistry.class);
195                 MockedStatic<JsonEngine> mockedJsonEngine = mockStatic(JsonEngine.class)) {
196             mockedRegistry.when(ResourceDefinitionRegistry::getSupportedResourceDescriptors).thenReturn(List.of(descriptor));
197             mockedRegistry.when(() -> ResourceDefinitionRegistry.dispatch(any(MCPFeatureRuntimeRequestContext.class), eq("shardingsphere://foo")))
198                     .thenReturn(Optional.of(new MCPMapPayload(payload)));
199             mockedJsonEngine.when(() -> JsonEngine.marshal(payload)).thenThrow(expectedException);
200             SyncResourceSpecification actualSpecification = new MCPResourceSpecificationFactory(createRuntimeContext()).createResourceSpecifications().getFirst();
201             McpError actual = assertThrows(McpError.class, () -> actualSpecification.readHandler().apply(createExchange(), new ReadResourceRequest("shardingsphere://foo")));
202             assertThat(actual.getJsonRpcError().code(), is(McpSchema.ErrorCodes.INTERNAL_ERROR));
203             assertThat(actual.getJsonRpcError().message(), is("Service is temporarily unavailable."));
204             Map<?, ?> actualData = (Map<?, ?>) actual.getJsonRpcError().data();
205             assertThat(actualData.get("error_id"), isA(String.class));
206             assertThat(actualData, is(Map.of("response_mode", "recovery", "summary", "Service is temporarily unavailable.", "error_id", actualData.get("error_id"))));
207             logCaptureAssertion.assertErrorLog(actualException -> {
208                 assertThat(((ThrowableProxy) actualException).getThrowable(), sameInstance(expectedException));
209                 assertThat(((ThrowableProxy) actualException).getThrowable().getCause(), sameInstance(expectedCause));
210             });
211         }
212     }

226     @Test
227     void assertCreateResourceTemplateSpecificationsWithAnnotations() {
228         MCPResourceDescriptor descriptor = new MCPResourceDescriptor("shardingsphere://foo/{database}", "foo_resource", "Foo Resource", "Foo description.", "application/json",
229                 new MCPResourceAnnotations(List.of("user"), 0.6D, "2025-01-01T00:00:00Z"), Map.of("foo_key", "foo_value"));
230         try (MockedStatic<ResourceDefinitionRegistry> mocked = mockStatic(ResourceDefinitionRegistry.class)) {
231             mocked.when(ResourceDefinitionRegistry::getSupportedResourceDescriptors).thenReturn(List.of(descriptor));
232             mocked.when(() -> ResourceDefinitionRegistry.dispatch(any(MCPFeatureRuntimeRequestContext.class), eq("shardingsphere://foo/foo_db")))
233                     .thenReturn(Optional.of(new MCPMapPayload(Map.of("foo_field", "foo_value"))));
234             SyncResourceTemplateSpecification actualSpecification = new MCPResourceSpecificationFactory(createRuntimeContext()).createResourceTemplateSpecifications().getFirst();
235             assertThat(actualSpecification.resourceTemplate().annotations().priority(), is(0.6D));
236             assertThat(actualSpecification.resourceTemplate().meta(), is(Map.of("foo_key", "foo_value")));
237             ReadResourceResult actual = actualSpecification.readHandler().apply(createExchange(), new ReadResourceRequest("shardingsphere://foo/foo_db"));
238             assertThat(actual.contents().getFirst().uri(), is("shardingsphere://foo/foo_db"));
239             ArgumentCaptor<MCPFeatureRuntimeRequestContext> requestContextCaptor = ArgumentCaptor.forClass(MCPFeatureRuntimeRequestContext.class);
240             mocked.verify(() -> ResourceDefinitionRegistry.dispatch(requestContextCaptor.capture(), eq("shardingsphere://foo/foo_db")));
241             assertThat(requestContextCaptor.getValue().getSessionIdentity().getSessionId(), is("session-1"));
242         }

245     private SyncResourceSpecification findResourceSpecification(final Collection<SyncResourceSpecification> specifications, final String uri) {
246         return specifications.stream().filter(each -> uri.equals(each.resource().uri())).findFirst().orElseThrow();
247     }
248     
249     private MCPRuntimeContext createRuntimeContext() {
250         MCPSessionManager sessionManager = new MCPSessionManager(Collections.emptyMap());
251         sessionManager.createSession(new MCPSessionIdentity("session-1", "", "", Map.of()));
252         return new MCPRuntimeContext(sessionManager, mock(MCPDatabaseCapabilityProvider.class), MCPTransportType.HTTP);
253     }
254     
255     private McpSyncServerExchange createExchange() {
256         McpSyncServerExchange result = mock(McpSyncServerExchange.class);
257         when(result.sessionId()).thenReturn("session-1");
258         return result;
259     }
260     
261     private SyncResourceTemplateSpecification findResourceTemplateSpecification(final Collection<SyncResourceTemplateSpecification> specifications, final String uriTemplate) {
262         return specifications.stream().filter(each -> uriTemplate.equals(each.resourceTemplate().uriTemplate())).findFirst().orElseThrow();
263     }
