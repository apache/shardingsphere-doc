36  /**
37   * MCP basic recovery payload factory.
38   */
39  @NoArgsConstructor(access = AccessLevel.PRIVATE)
40  final class MCPBasicRecoveryPayloadFactory {
41      
42      static Map<String, Object> createUnsupportedToolRecovery(final String toolName) {
43          Map<String, Object> result = MCPRecoveryPayloadSupport.createBaseRecovery("unsupported_tool", "Call one of the supported tools returned by tools/list.");
44          result.put("tool_name", toolName);
45          result.put("discovery_method", "tools/list");
46          result.put(MCPPayloadFieldNames.RESOURCES_TO_READ, MCPRecoveryPayloadSupport.createResourceHintList(
47                  "shardingsphere://capabilities", "capability", "Read ShardingSphere catalog details after checking tools/list."));
48          result.put(MCPPayloadFieldNames.NEXT_ACTIONS, List.of(MCPNextActionUtils.readResource("shardingsphere://capabilities",
49                  "Read the ShardingSphere domain catalog only when tools/list does not provide enough context.")));
50          return result;
51      }
52      
53      static Map<String, Object> createUnsupportedResourceRecovery(final String resourceUri) {
54          Map<String, Object> result = MCPRecoveryPayloadSupport.createBaseRecovery(
55                  "unsupported_resource", "Read one of the supported resources or templates returned by resources/list and resources/templates/list.");
56          result.put(MCPPayloadFieldNames.RESOURCE, MCPResourceHintUtils.create(resourceUri, "resource", "inspect_detail", "Unsupported resource URI requested.", MCPPayloadFieldNames.RECOVERY));
57          result.put("discovery_methods", List.of("resources/list", "resources/templates/list"));
58          result.put(MCPPayloadFieldNames.RESOURCES_TO_READ, MCPRecoveryPayloadSupport.createResourceHintList(
59                  "shardingsphere://capabilities", "capability", "Read ShardingSphere catalog details after checking resources/list and resources/templates/list."));
60          result.put(MCPPayloadFieldNames.NEXT_ACTIONS, List.of(MCPNextActionUtils.readResource("shardingsphere://capabilities",
61                  "Read the ShardingSphere domain catalog only when official resource discovery does not provide enough context.")));
62          return result;
63      }
64      
65      static Map<String, Object> createToolCallLimitRecovery(final MCPToolCallLimitExceededException cause) {
66          Map<String, Object> result = MCPRecoveryPayloadSupport.createBaseRecovery(
67                  "tool_call_limit_exceeded", "Stop the current loop, summarize progress, or start a new MCP session before retrying.");
68          result.put("identity_scope", "mcp_session");
69          result.put("session_id", cause.getSessionId());
70          result.put("tool_name", cause.getToolName());
71          result.put("max_tool_calls_per_session", cause.getMaxToolCallsPerSession());
72          result.put(MCPPayloadFieldNames.RESOURCES_TO_READ, MCPRecoveryPayloadSupport.createResourceHintList(
73                  "shardingsphere://guidance", "guidance", "Read the current MCP safety policy before retrying."));
74          result.put(MCPPayloadFieldNames.NEXT_ACTIONS, List.of(MCPNextActionUtils.readResource("shardingsphere://guidance", "Read current MCP safety policy and tool call limits.")));
75          return result;
76      }
77      
78      static Map<String, Object> createInvalidToolArgumentRecovery(final MCPInvalidToolArgumentException cause) {
79          return createInvalidIntegerArgumentRecovery(cause.getArgumentPath(), cause.getSuggestedValue(), cause.getSourceTool(), cause.getTargetTool(), cause.getMinimumValue(),
80                  cause.getMaximumValue());
81      }
82      
83      static Map<String, Object> createToolArgumentContractViolationRecovery(final MCPToolArgumentContractViolationException cause) {
84          Map<String, Object> result = MCPRecoveryPayloadSupport.createBaseRecovery(cause.getCategory(), createToolArgumentContractModelAction(cause));
85          result.put(MCPPayloadFieldNames.FIELD, cause.getArgumentPath());
86          result.put("tool_name", cause.getToolName());
87          if (!cause.getExpectedType().isEmpty()) {
88              result.put("expected_type", cause.getExpectedType());
89          }
90          if (!cause.getAllowedValues().isEmpty()) {
91              result.put(MCPPayloadFieldNames.ALLOWED_VALUES, cause.getAllowedValues());
92          }
93          if (!cause.getSuggestedArguments().isEmpty()) {
94              result.put("suggested_arguments", cause.getSuggestedArguments());
95          }
96          result.put(MCPPayloadFieldNames.NEXT_ACTIONS, createToolArgumentContractNextActions(cause));
97          return result;
98      }
99      
100     private static String createToolArgumentContractModelAction(final MCPToolArgumentContractViolationException cause) {
101         if ("unknown_argument".equals(cause.getCategory())) {
102             return "Remove arguments that are not declared by the tool inputSchema before retrying.";
103         }
104         if ("invalid_enum_value".equals(cause.getCategory())) {
105             return "Retry with one of the allowed enum values, or omit the optional argument when safe.";
106         }
107         return "Retry with a value that matches the declared inputSchema type.";
108     }
109     
110     private static List<Map<String, Object>> createToolArgumentContractNextActions(final MCPToolArgumentContractViolationException cause) {
111         if (cause.getSuggestedArguments().isEmpty()) {
112             return List.of(MCPNextActionUtils.askUser("Ask the user for a value that matches the tool inputSchema.", List.of(cause.getArgumentPath())));
113         }
114         return List.of(MCPNextActionUtils.retryTool(cause.getToolName(), "Retry with arguments that match the tool inputSchema.", cause.getSuggestedArguments()));
115     }
116     
117     static Map<String, Object> createMissingArgumentRecovery(final String argumentName) {
118         boolean missingDatabase = WorkflowFieldNames.DATABASE.equals(argumentName);
119         String category = missingDatabase ? "missing_database" : "missing_argument";
120         String modelAction = missingDatabase
121                 ? "Read shardingsphere://databases, infer a single safe database when possible, or ask the user."
122                 : "Provide the missing argument, infer it from resources when safe, or ask the user.";
123         Map<String, Object> result = MCPRecoveryPayloadSupport.createBaseRecovery(category, modelAction);
124         result.put("missing_fields", List.of(argumentName));
125         result.put(MCPPayloadFieldNames.RESOURCES_TO_READ, createMissingArgumentResourcesToRead(argumentName));
126         result.put(MCPPayloadFieldNames.NEXT_ACTIONS, createMissingArgumentNextActions(argumentName));
127         return result;
128     }
129     
130     static Map<String, Object> createInvalidObjectTypesRecovery(final MCPInvalidMetadataObjectTypesException cause) {
131         Map<String, Object> result = MCPRecoveryPayloadSupport.createBaseRecovery("invalid_enum_value", "Retry with one or more allowed object_types values.");
132         result.put(MCPPayloadFieldNames.FIELD, "object_types");
133         result.put(MCPPayloadFieldNames.ALLOWED_VALUES, cause.getAllowedValues());
134         result.put("suggested_arguments", Map.of("object_types", cause.getAllowedValues()));
135         result.put(MCPPayloadFieldNames.NEXT_ACTIONS, List.of(MCPNextActionUtils.retryTool(CoreToolNames.SEARCH_METADATA,
136                 "Retry database_gateway_search_metadata with allowed object_types values, or omit object_types to search every supported metadata type.",
137                 Map.of("object_types", cause.getAllowedValues()))));
138         return result;
139     }
140     
141     private static Map<String, Object> createInvalidIntegerArgumentRecovery(final String fieldName, final int suggestedValue, final String sourceTool, final String targetTool,
142                                                                             final int minimumValue, final int maximumValue) {
143         Map<String, Object> result = MCPRecoveryPayloadSupport.createBaseRecovery("invalid_integer_argument", "Retry with an integer value inside the documented bounds.");
144         result.put(MCPPayloadFieldNames.FIELD, fieldName);
145         if (!sourceTool.isEmpty()) {
146             result.put("source_tool", sourceTool);
147         }
148         if (!targetTool.isEmpty()) {
149             result.put("tool_name", targetTool);
150         }
151         result.put("minimum_value", minimumValue);
152         result.put("maximum_value", maximumValue);
153         result.put("suggested_value", suggestedValue);
154         Map<String, Object> suggestedArguments = Map.of(fieldName, suggestedValue);
155         result.put("suggested_arguments", suggestedArguments);
156         result.put(MCPPayloadFieldNames.NEXT_ACTIONS, createInvalidIntegerArgumentNextActions(fieldName, targetTool, suggestedArguments));
157         return result;
158     }
159     
160     private static List<Map<String, Object>> createInvalidIntegerArgumentNextActions(final String fieldName, final String targetTool, final Map<String, Object> suggestedArguments) {
161         if (targetTool.isEmpty()) {
162             return List.of(MCPNextActionUtils.askUser("Ask the user for a bounded integer value before retrying.", List.of(fieldName)));
163         }
164         return List.of(MCPNextActionUtils.retryTool(targetTool, "Retry with a bounded integer argument.", suggestedArguments));
165     }
166     
167     private static List<Map<String, Object>> createMissingArgumentResourcesToRead(final String argumentName) {
168         if (WorkflowFieldNames.DATABASE.equals(argumentName)) {
169             return MCPRecoveryPayloadSupport.createResourceHintList(
170                     "shardingsphere://databases", "logical-database", "Read logical databases before retrying with a database argument.");
171         }
172         if (WorkflowFieldNames.PLAN_ID.equals(argumentName)) {
173             return MCPRecoveryPayloadSupport.createResourceHintList(
174                     "shardingsphere://capabilities", "capability", "Read workflow-capable MCP tools before retrying with a plan_id argument.");
175         }
176         return List.of();
177     }
178     
179     private static List<Map<String, Object>> createMissingArgumentNextActions(final String argumentName) {
180         List<Map<String, Object>> resources = createMissingArgumentResourcesToRead(argumentName);
181         if (resources.isEmpty()) {
182             return List.of(MCPNextActionUtils.askUser("Ask the user for the missing argument.", List.of(argumentName)));
183         }
184         return List.of(
185                 MCPNextActionUtils.readResource(Objects.toString(resources.iterator().next().get(MCPPayloadFieldNames.URI), ""), "Read a safe resource before retrying with the missing argument."));
186     }
