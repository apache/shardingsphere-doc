154     void assertGetRowsWithDatabaseResolution(final String sql, final String currentDatabaseName) {
155         connectionContext = new ConnectionContext(Collections::emptyList);
156         connectionContext.setCurrentDatabaseName(currentDatabaseName);
157         executor.setConnectionContext(mockConnectionContext(new HintValueContext(), connectionContext, mock(DatabaseConnectionManager.class)));
158         ShardingSphereDatabase database = contextManager.getMetaDataContexts().getMetaData().getDatabase("foo_db");
159         ExecutionContext executionContext = new ExecutionContext(mock(QueryContext.class), Collections.singletonList(createExecutionUnit("foo_ds", "SELECT 1")), mock(RouteContext.class));
160         try (
161                 MockedConstruction<SQLFederationEngine> ignoredFederationEngine = mockConstruction(SQLFederationEngine.class, (mock, context) -> {
162                     assertThat(context.arguments().get(0), is("foo_db"));
163                     when(mock.decide(any(QueryContext.class), any(RuleMetaData.class))).thenReturn(false);
164                 });
165                 MockedConstruction<KernelProcessor> ignoredKernelProcessor = mockConstruction(KernelProcessor.class,
166                         (mock, context) -> when(mock.generateExecutionContext(any(QueryContext.class), any(RuleMetaData.class), any(ConfigurationProperties.class)))
167                                 .thenAnswer(invocation -> {
168                                     QueryContext actualQueryContext = invocation.getArgument(0);
169                                     assertThat(actualQueryContext.getUsedDatabase(), sameInstance(database));
170                                     return executionContext;
171                                 }))) {
172             assertThat(executor.getRows(new PreviewStatement(sql), contextManager).iterator().next().getCell(1), is("foo_ds"));
173             assertThat(connectionContext.getCurrentDatabaseName(), is(Optional.ofNullable(currentDatabaseName)));
174         }

340     private JDBCExecutionUnit createJDBCExecutionUnit(final String dataSourceName, final String sql, final boolean isThrowSQLException) throws SQLException {
341         Statement statement = mock(Statement.class);
342         if (isThrowSQLException) {
343             when(statement.executeQuery(sql)).thenThrow(new SQLException("mock exception"));
344         } else {
345             when(statement.executeQuery(sql)).thenReturn(mock(ResultSet.class));
346         }
347         return new JDBCExecutionUnit(createExecutionUnit(dataSourceName, sql), ConnectionMode.MEMORY_STRICTLY, statement);
348     }

350     private void configureSQLFederationEngine(final SQLFederationEngine federationEngine, final MockedConstruction.Context context, final AtomicReference<String> actualSchemaName) {
351         assertThat(context.arguments().get(0), is("foo_db"));
352         actualSchemaName.set(context.arguments().get(1).toString());
353         when(federationEngine.decide(any(QueryContext.class), any(RuleMetaData.class))).thenReturn(true);
354         doAnswer(this::executePreviewCallback).when(federationEngine).executeQuery(any(), any(), any(SQLFederationContext.class));
355     }
356     
357     private Object executePreviewCallback(final InvocationOnMock invocation) throws SQLException {
358         SQLFederationContext sqlFederationContext = invocation.getArgument(2);
359         assertThat(sqlFederationContext.getProcessId(), is("process_id"));
360         JDBCExecutorCallback<? extends ExecuteResult> callback = invocation.getArgument(1);
361         callback.execute(Collections.singletonList(createJDBCExecutionUnit("foo_ds", "SELECT 2", false)), true, "process_id");
362         try {
363             callback.execute(Collections.singletonList(createJDBCExecutionUnit("bar_ds", "SELECT 3", true)), true, "process_id");
364         } catch (final SQLException ignored) {
365         }
366         sqlFederationContext.getPreviewExecutionUnits().add(createExecutionUnit("bar_ds", "SELECT 2"));
367         return null;
368     }
