27  import org.junit.jupiter.params.provider.MethodSource;

29  import java.sql.SQLException;

32  import java.util.Map;
33  import java.util.Properties;
34  import java.util.stream.Stream;

36  import static org.hamcrest.MatcherAssert.assertThat;
37  import static org.hamcrest.Matchers.anyOf;
38  import static org.hamcrest.Matchers.is;
39  import static org.junit.jupiter.api.Assertions.assertDoesNotThrow;
40  import static org.junit.jupiter.api.Assertions.assertThrows;
41  import static org.mockito.Mockito.mock;
42  
43  class ProxyConfigurationLoadResultValidatorTest {
44      
45      @Test
46      void assertValidateWithoutDuplicates() {
47          ProxyConfigurationLoadResult loadResult = createConfig("foo_db", "foo_db", Collections.singletonMap("global_ds", mock(DataSourceConfiguration.class)),
48                  Collections.singletonMap("db_ds", mock(DataSourceConfiguration.class)));
49          assertDoesNotThrow(() -> ProxyConfigurationLoadResultValidator.validate(loadResult));
50      }
51      
52      private ProxyConfigurationLoadResult createConfig(final String databaseKey, final String databaseName, final Map<String, DataSourceConfiguration> globalDataSources,
53                                                        final Map<String, DataSourceConfiguration> databaseDataSources) {
54          return new ProxyConfigurationLoadResult(null, globalDataSources, Collections.emptyList(), new Properties(),
55                  Collections.singletonMap(databaseKey, new ProxyDatabaseConfiguration(databaseName, databaseDataSources, Collections.emptyList())));
56      }
57      
58      @ParameterizedTest(name = "{0}")
59      @MethodSource("duplicateDataSourceArguments")
60      void assertValidateWithDuplicateDataSource(final String name, final String databaseKey, final String databaseName, final String dataSourceName, final String expectedMessage) {
61          Map<String, DataSourceConfiguration> dataSources = Collections.singletonMap(dataSourceName, mock(DataSourceConfiguration.class));
62          ProxyConfigurationLoadResult loadResult = createConfig(databaseKey, databaseName, dataSources, dataSources);
63          DuplicateStorageUnitException actual = assertThrows(DuplicateStorageUnitException.class, () -> ProxyConfigurationLoadResultValidator.validate(loadResult));
64          assertThat(actual.getMessage(), is(expectedMessage));
65          SQLException actualSQLException = actual.toSQLException();
66          assertThat(actualSQLException.getMessage(), is(expectedMessage));
67          assertThat(actualSQLException.getSQLState(), is("42S01"));
68          assertThat(actualSQLException.getErrorCode(), is(10104));
69      }
70      
71      private static Stream<Arguments> duplicateDataSourceArguments() {
72          return Stream.of(
73                  Arguments.of("same database key and name", "foo_db", "foo_db", "ds_0", "Duplicate storage unit names 'ds_0' on database 'foo_db'."),
74                  Arguments.of("different database key and name", "map_key", "configured_db", "ds_0", "Duplicate storage unit names 'ds_0' on database 'configured_db'."),
75                  Arguments.of("special characters in map keys", "db.[key]{value}${suffix}\\key", "foo_db", "ds.[0]{value}${suffix}\\name",
76                          "Duplicate storage unit names 'ds.[0]{value}${suffix}\\name' on database 'foo_db'."));
77      }
78      
79      @Test
80      void assertValidateWithMultipleDuplicateDataSources() {
81          Map<String, DataSourceConfiguration> dataSources = new HashMap<>(2, 1F);
82          dataSources.put("ds_0", mock(DataSourceConfiguration.class));
83          dataSources.put("ds_1", mock(DataSourceConfiguration.class));
84          ProxyConfigurationLoadResult loadResult = createConfig("foo_db", "foo_db", dataSources, dataSources);
85          DuplicateStorageUnitException actual = assertThrows(DuplicateStorageUnitException.class, () -> ProxyConfigurationLoadResultValidator.validate(loadResult));
86          assertThat(actual.getMessage(), anyOf(is("Duplicate storage unit names 'ds_0, ds_1' on database 'foo_db'."), is("Duplicate storage unit names 'ds_1, ds_0' on database 'foo_db'.")));
87          SQLException actualSQLException = actual.toSQLException();
88          assertThat(actualSQLException.getMessage(), is(actual.getMessage()));
89          assertThat(actualSQLException.getSQLState(), is("42S01"));
90          assertThat(actualSQLException.getErrorCode(), is(10104));
91      }
92  }
