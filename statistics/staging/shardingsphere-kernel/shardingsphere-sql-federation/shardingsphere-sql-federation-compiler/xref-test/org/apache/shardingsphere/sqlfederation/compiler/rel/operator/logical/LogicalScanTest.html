67  class LogicalScanTest {
68      
69      private RelDataType rowType;
70      
71      private RelOptCluster cluster;
72      
73      private LogicalTableScan tableScan;
74      
75      private RexBuilder rexBuilder;
76      
77      @BeforeEach
78      void setUp() {
79          RelDataTypeFactory typeFactory = new JavaTypeFactoryImpl();
80          rexBuilder = new RexBuilder(typeFactory);
81          cluster = RelOptCluster.create(new VolcanoPlanner(), rexBuilder);
82          rowType = typeFactory.builder().add("col", SqlTypeName.INTEGER).build();
83          TranslatableTable federationTable = mock(TranslatableTable.class, withSettings().extraInterfaces(ModifiableTable.class));
84          when(federationTable.getRowType(typeFactory)).thenReturn(rowType);
85          when(federationTable.getStatistic()).thenReturn(Statistics.of(1D, ImmutableList.of()));
86          when(federationTable.getJdbcTableType()).thenReturn(Schema.TableType.TABLE);
87          when(federationTable.toRel(any(RelOptTable.ToRelContext.class), any(RelOptTable.class))).thenAnswer(invocation -> LogicalTableScan.create(
88                  invocation.getArgument(0, RelOptTable.ToRelContext.class).getCluster(), invocation.getArgument(1, RelOptTable.class), ImmutableList.of()));
89          RelOptSchema relOptSchema = mock(RelOptSchema.class);
90          when(relOptSchema.getTypeFactory()).thenReturn(typeFactory);
91          RelOptTable relOptTable = RelOptTableImpl.create(relOptSchema, rowType, ImmutableList.of("test", "tbl"), federationTable, each -> Expressions.constant(federationTable));
92          when(relOptSchema.getTableForMember(anyList())).thenReturn(relOptTable);
93          tableScan = LogicalTableScan.create(cluster, relOptTable, Collections.emptyList());
94      }
95      
96      @Test
97      void assertPushDownFilter() {
98          LogicalFilter logicalFilter = LogicalFilter.create(tableScan, rexBuilder.makeLiteral(true));
99          LogicalScan logicalScan = new LogicalScan(tableScan, "MySQL");
100         logicalScan.pushDown(logicalFilter);
101         assertThat(logicalScan.getRowType(), is(logicalFilter.getRowType()));
102     }
103     
104     @Test
105     void assertPushDownProject() {
106         LogicalProject logicalProject = LogicalProject.create(
107                 tableScan, Collections.emptyList(), Collections.singletonList(rexBuilder.makeInputRef(rowType, 0)), Collections.singletonList("col"), Collections.emptySet());
108         LogicalScan logicalScan = new LogicalScan(tableScan, "MySQL");
109         logicalScan.pushDown(logicalProject);
110         assertThat(logicalScan.getRowType(), is(logicalProject.getRowType()));
111     }
112     
113     @Test
114     void assertPeek() {
115         assertNotNull(new LogicalScan(tableScan, "MySQL").peek());
116     }
117     
118     @Test
119     void assertExplainTerms() {
120         LogicalScan logicalScan = new LogicalScan(tableScan, "MySQL");
121         StringWriter writer = new StringWriter();
122         RelWriter relWriter = new RelWriterImpl(new PrintWriter(writer), SqlExplainLevel.EXPPLAN_ATTRIBUTES, false);
123         logicalScan.explainTerms(relWriter);
124         relWriter.done(logicalScan);
125         assertTrue(writer.toString().contains("pushDownRelBuilder"));
126     }

129     void assertDeepEqualsSelf() {
130         LogicalScan logicalScan = new LogicalScan(tableScan, "MySQL");
131         assertTrue(logicalScan.deepEquals(logicalScan));
132     }
133     
134     @Test
135     void assertDeepEqualsWithNull() {
136         assertFalse(new LogicalScan(tableScan, "MySQL").deepEquals(null));
137     }
138     
139     @Test
140     void assertDeepEqualsWithDifferentInstanceWithoutSharedBuilder() {
141         assertFalse(new LogicalScan(tableScan, "MySQL").deepEquals(new LogicalScan(tableScan, "MySQL")));
142     }
143     
144     @Test
145     void assertDeepEqualsWithSharedBuilder() throws ReflectiveOperationException {
146         LogicalScan logicalScan = new LogicalScan(tableScan, "MySQL");
147         LogicalScan sharedBuilder = new LogicalScan(tableScan, "MySQL");
148         MemberAccessor memberAccessor = Plugins.getMemberAccessor();
149         Field builderField = LogicalScan.class.getDeclaredField("pushDownRelBuilder");
150         Object pushDownRelBuilder = memberAccessor.get(builderField, logicalScan);
151         memberAccessor.set(builderField, sharedBuilder, pushDownRelBuilder);
152         assertTrue(logicalScan.deepEquals(sharedBuilder));
153     }
154     
155     @Test
156     void assertDeepEqualsWithDifferentClass() {
157         assertFalse(new LogicalScan(tableScan, "MySQL").deepEquals(new Object()));
158     }
159     
160     @Test
161     void assertDeepEqualsWithDifferentDatabase() throws ReflectiveOperationException {
162         LogicalScan logicalScan = new LogicalScan(tableScan, "MySQL");
163         LogicalScan differentDatabase = new LogicalScan(tableScan, "PostgreSQL");
164         MemberAccessor memberAccessor = Plugins.getMemberAccessor();
165         Field builderField = LogicalScan.class.getDeclaredField("pushDownRelBuilder");
166         memberAccessor.set(builderField, differentDatabase, memberAccessor.get(builderField, logicalScan));
167         assertFalse(logicalScan.deepEquals(differentDatabase));
168     }
169     
170     @Test
171     void assertDeepEqualsWithDifferentHints() throws ReflectiveOperationException {
172         LogicalScan logicalScan = new LogicalScan(tableScan, "MySQL");
173         LogicalScan differentHints = new LogicalScan(tableScan, "MySQL");
174         MemberAccessor memberAccessor = Plugins.getMemberAccessor();
175         Field builderField = LogicalScan.class.getDeclaredField("pushDownRelBuilder");
176         memberAccessor.set(builderField, differentHints, memberAccessor.get(builderField, logicalScan));
177         memberAccessor.set(TableScan.class.getDeclaredField("hints"), differentHints, ImmutableList.of(RelHint.builder("h").build()));
178         assertFalse(logicalScan.deepEquals(differentHints));
179     }
180     
181     @Test
182     void assertDeepEqualsWithDifferentRowType() throws ReflectiveOperationException {
183         LogicalScan logicalScan = new LogicalScan(tableScan, "MySQL");
184         LogicalScan differentRowType = new LogicalScan(tableScan, "MySQL");
185         MemberAccessor memberAccessor = Plugins.getMemberAccessor();
186         Field builderField = LogicalScan.class.getDeclaredField("pushDownRelBuilder");
187         memberAccessor.set(builderField, differentRowType, memberAccessor.get(builderField, logicalScan));
188         RelDataTypeFactory typeFactory = cluster.getTypeFactory();
189         RelDataType anotherRowType = typeFactory.builder().add("col", SqlTypeName.BIGINT).build();
190         memberAccessor.set(AbstractRelNode.class.getDeclaredField("rowType"), differentRowType, anotherRowType);
191         assertFalse(logicalScan.deepEquals(differentRowType));
192     }
193     
194     @Test
195     void assertDeepHashCode() {
196         LogicalScan logicalScan = new LogicalScan(tableScan, "MySQL");
197         int expectedHashCode = Objects.hash(logicalScan.getTraitSet(), "MySQL", logicalScan.getPushDownRelBuilder(), logicalScan.getHints(), logicalScan.getRowType());
198         assertThat(logicalScan.deepHashCode(), is(expectedHashCode));
199     }
