69      @Test
70      void assertPlanApplyAndValidateEncryptWorkflow() throws IOException, InterruptedException {
71          try (MCPInteractionClient interactionClient = createOpenedInteractionClient()) {
72              Map<String, Object> clarifyingResponse = interactionClient.call(PLAN_TOOL_NAME,
73                      createEncryptPlanArguments(Map.of("table", "orders", "column", "status"), false));
74              assertThat(String.valueOf(clarifyingResponse.get("status")), is("clarifying"));
75              assertModelFacingPayloadContract(clarifyingResponse);
76              assertThat(getClarificationMessages(clarifyingResponse), is(List.of("Please provide logical database first.")));
77              String planId = String.valueOf(clarifyingResponse.get("plan_id"));
78              Map<String, Object> plannedResponse = interactionClient.call(PLAN_TOOL_NAME,
79                      Map.of("plan_id", planId, "database", getLogicalDatabaseName(), "algorithm_type", "AES",
80                              "cipher_column_name", "status_cipher", "primary_algorithm_properties", Map.of("aes-key-value", TEMPLATE_SECRET_VALUE)));
81              assertThat(String.valueOf(plannedResponse.get("status")), is("planned"));
82              assertModelFacingPayloadContract(plannedResponse);
83              assertSecretRedacted(plannedResponse, TEMPLATE_SECRET_VALUE);
84              assertThat(String.valueOf(plannedResponse.get("current_step")), is("review"));
85              assertThat(getStringListOrEmpty(plannedResponse.get("global_steps")).size(), is(8));
86              Map<String, Object> maskedPropertyPreview = getObjectOrEmpty(plannedResponse.get("masked_property_preview"));
87              assertThat(String.valueOf(getObjectOrEmpty(maskedPropertyPreview.get("primary")).get("aes-key-value")), is("******"));
88              List<Map<String, Object>> distSqlArtifacts = getObjectListOrEmpty(plannedResponse.get("distsql_artifacts"));
89              assertThat(distSqlArtifacts.size(), is(1));
90              assertThat(String.valueOf(distSqlArtifacts.getFirst().get("sql")), containsString("'aes-key-value'='******'"));
91              assertThat(String.valueOf(distSqlArtifacts.getFirst().get("sql")), containsString("CIPHER=`status_cipher`"));
92              assertSecretRedacted(interactionClient.readResource(String.format(WORKFLOW_RESOURCE_URI, planId)), TEMPLATE_SECRET_VALUE);
93              Map<String, Object> applyResponse = applyReviewedWorkflow(interactionClient, planId, TEMPLATE_SECRET_VALUE);
94              assertApplyCompleted(applyResponse);
95              assertThat(getObjectListOrEmpty(applyResponse.get("step_results")).size(), is(1));
96              Map<String, Object> validationResponse = interactionClient.call(VALIDATE_TOOL_NAME, Map.of("plan_id", planId));
97              assertValidationPassed(validationResponse);
98              assertSecretRedacted(validationResponse, TEMPLATE_SECRET_VALUE);
99              assertTrue(getValidationSection(validationResponse, "logical_metadata").isEmpty());
100             assertTrue(getValidationSection(validationResponse, "sql_executability").isEmpty());
101             Map<String, Object> ruleValidation = getValidationSection(validationResponse, "rule");
102             assertThat(String.valueOf(ruleValidation.get("status")), is("passed"));
103             Map<String, Object> ruleEvidence = getObjectListOrEmpty(ruleValidation.get("evidence")).getFirst();
104             assertThat(String.valueOf(ruleEvidence.get("logic_column")), is("status"));
105             assertThat(String.valueOf(ruleEvidence.get("cipher_column")), is("status_cipher"));
106             assertThat(String.valueOf(ruleEvidence.get("encryptor_type")).toUpperCase(Locale.ENGLISH), is("AES"));
107         }
108     }

111     void assertPlanRecommendsAssistedQueryEncryptWorkflow() throws IOException, InterruptedException {
112         try (MCPInteractionClient interactionClient = createOpenedInteractionClient()) {
113             Map<String, Object> actualClarifyingResponse = interactionClient.call(PLAN_TOOL_NAME,
114                     createEncryptPlanArguments(Map.of("database", getLogicalDatabaseName(), "table", "orders", "column", "status"), true));
115             assertThat(String.valueOf(actualClarifyingResponse.get("status")), is("clarifying"));
116             assertThat(getIssueCodes(actualClarifyingResponse), hasItem(WorkflowIssueCode.REQUIRED_PROPERTY_MISSING));
117             Map<String, Object> actualClarificationQuestion = getObjectListOrEmpty(actualClarifyingResponse.get("clarification_questions")).getFirst();
118             assertThat(String.valueOf(actualClarificationQuestion.get("field")), is("primary_algorithm_properties.aes-key-value"));
119             assertThat(String.valueOf(actualClarificationQuestion.get("input_type")), is("secret"));
120             assertTrue((Boolean) actualClarificationQuestion.get("secret"));
121             assertThat(String.valueOf(actualClarificationQuestion.get("message")),
122                     is("Sensitive input must be provided through configured secure channels before continuing the same planner."));
123             assertFalse(actualClarificationQuestion.containsKey("display_message"));
124             List<Map<String, Object>> actualRecommendations = getObjectListOrEmpty(actualClarifyingResponse.get("algorithm_recommendations"));
125             assertThat(actualRecommendations.size(), is(2));
126             assertThat(String.valueOf(actualRecommendations.getFirst().get("algorithm_role")), is("primary"));
127             assertThat(String.valueOf(actualRecommendations.getFirst().get("algorithm_type")).toUpperCase(Locale.ENGLISH), is("AES"));
128             assertThat(String.valueOf(actualRecommendations.get(1).get("algorithm_role")), is("assisted_query"));
129             assertThat(String.valueOf(actualRecommendations.get(1).get("algorithm_type")).toUpperCase(Locale.ENGLISH), is("MD5"));
130             String planId = String.valueOf(actualClarifyingResponse.get("plan_id"));
131             Map<String, Object> actualPlannedResponse = interactionClient.call(PLAN_TOOL_NAME,
132                     Map.of("plan_id", planId, "cipher_column_name", "status_cipher", "assisted_query_column_name", "status_assisted_query",
133                             "assisted_query_algorithm_type", "MD5", "primary_algorithm_properties", Map.of("aes-key-value", "assisted-secret")));
134             assertThat(String.valueOf(actualPlannedResponse.get("status")), is("planned"));
135             List<Map<String, Object>> actualDistSqlArtifacts = getObjectListOrEmpty(actualPlannedResponse.get("distsql_artifacts"));
136             assertThat(actualDistSqlArtifacts.size(), is(1));
137             assertThat(String.valueOf(actualDistSqlArtifacts.getFirst().get("sql")), containsString("ASSISTED_QUERY_COLUMN=`status_assisted_query`"));
138             Map<String, Object> actualApplyResponse = applyReviewedWorkflow(interactionClient, planId);
139             assertApplyCompleted(actualApplyResponse);
140             assertThat(getObjectListOrEmpty(actualApplyResponse.get("step_results")).size(), is(1));
141             Map<String, Object> actualValidationResponse = interactionClient.call(VALIDATE_TOOL_NAME, Map.of("plan_id", planId));
142             assertValidationPassed(actualValidationResponse);
143             assertTrue(getValidationSection(actualValidationResponse, "sql_executability").isEmpty());
144         }

212         try (MCPInteractionClient interactionClient = createOpenedInteractionClient()) {
213             Map<String, Object> actualPlanResponse = interactionClient.call(PLAN_TOOL_NAME,
214                     createEncryptPlanArguments(Map.of("database", getLogicalDatabaseName(), "table", "orders", "column", "status",

217             String planId = String.valueOf(actualPlanResponse.get("plan_id"));

219             assertValidationPassed(interactionClient.call(VALIDATE_TOOL_NAME, Map.of("plan_id", planId)));

223             assertFalse((Boolean) actualCustomAlgorithm.get("capability_confirmed"));

225                     interactionClient.readResource(String.format(RULES_RESOURCE_URI, getLogicalDatabaseName())));
226             Map<String, Object> actualEncryptRule = findItemByField(actualEncryptRules, "logic_column", "status");

228             List<Map<String, Object>> actualSingleRuleItems = getPayloadItems(
229                     interactionClient.readResource(String.format(TABLE_RULES_RESOURCE_URI, getLogicalDatabaseName(), "orders")));
230             assertThat(actualSingleRuleItems.size(), is(1));
231             assertThat(String.valueOf(actualSingleRuleItems.getFirst().get("logic_column")), is("status"));
232         }

235     private void createEncryptRuleWithoutEquality(final MCPInteractionClient interactionClient) throws IOException, InterruptedException {
236         createEncryptRuleWithoutEquality(interactionClient, "status", "base-secret");
237     }

239     private void createEncryptRuleWithoutEquality(final MCPInteractionClient interactionClient, final String columnName, final String secret) throws IOException, InterruptedException {
240         Map<String, Object> actualCreatePlanResponse = interactionClient.call(PLAN_TOOL_NAME,
241                 createEncryptPlanArguments(Map.of("database", getLogicalDatabaseName(), "table", "orders", "column", columnName, "algorithm_type", "AES",
242                         "cipher_column_name", columnName + "_cipher", "primary_algorithm_properties", Map.of("aes-key-value", secret)), false));
243         assertThat(String.valueOf(actualCreatePlanResponse.get("status")), is("planned"));
244         String planId = String.valueOf(actualCreatePlanResponse.get("plan_id"));
245         assertApplyCompleted(applyReviewedWorkflow(interactionClient, planId));
246         assertValidationPassed(interactionClient.call(VALIDATE_TOOL_NAME, Map.of("plan_id", planId)));
247     }
248     
249     private Map<String, Object> createEncryptPlanArguments(final Map<String, Object> arguments, final boolean requiresEqualityFilter) {
250         Map<String, Object> result = new LinkedHashMap<>(arguments);
251         result.put("operation_type", "create");
252         result.put("requires_decrypt", true);
253         result.put("requires_equality_filter", requiresEqualityFilter);
254         result.put("requires_like_query", false);
255         return result;
256     }
257     
258     private Map<String, Object> applyReviewedWorkflow(final MCPInteractionClient interactionClient, final String planId, final String secretValue) throws IOException, InterruptedException {
259         Map<String, Object> previewResponse = previewWorkflow(interactionClient, planId);
260         assertSecretRedacted(previewResponse, secretValue);
261         Map<String, Object> result = interactionClient.call(APPLY_TOOL_NAME, createReviewThenExecuteArguments(planId, getApprovedSteps(previewResponse)));
262         assertSecretRedacted(result, secretValue);
263         return result;
264     }
265     
266     private void assertSecretRedacted(final Map<String, Object> actual, final String secretValue) {
267         if (!secretValue.isEmpty()) {
268             assertFalse(String.valueOf(actual).contains(secretValue));
269         }
270     }
