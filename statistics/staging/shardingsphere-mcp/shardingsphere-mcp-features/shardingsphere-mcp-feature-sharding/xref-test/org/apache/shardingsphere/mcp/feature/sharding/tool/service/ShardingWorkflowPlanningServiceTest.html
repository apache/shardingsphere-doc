308     void assertPlanKeyGenerateStrategyContinuesAfterMissingKeyGeneratorTypeClarification() {
309         TestWorkflowSessionContext sessionContext = new TestWorkflowSessionContext();
310         ShardingWorkflowRequest initialRequest = createDatabaseRequest();
311         initialRequest.setKeyGenerateStrategyName("order_sequence_strategy");
312         initialRequest.setSequenceName("order_seq");
313         initialRequest.putKeyGeneratorProperties(Map.of("worker-id", "1"));
314         WorkflowContextSnapshot initial = planningService.planKeyGenerateStrategy(sessionContext, mock(MCPFeatureQueryFacade.class), initialRequest);
315         assertThat(initial.getStatus(), is(WorkflowLifecycle.STATUS_CLARIFYING));
316         assertThat(initial.getClarifiedIntent().getUnresolvedFields(), is(List.of("key_generator_type")));
317         assertThat(initial.getClarifiedIntent().getClarificationMessages(), is(List.of("Please provide key generator type.")));
318         Map<String, Object> initialPayload = WorkflowPlanPayloadBuilder.build(initial);
319         assertThat(initialPayload.get("missing_required_inputs"), is(List.of("key_generator_type")));
320         Map<?, ?> actualQuestion = (Map<?, ?>) ((List<?>) initialPayload.get("clarification_questions")).getFirst();
321         assertThat(actualQuestion.get("field"), is("key_generator_type"));
322         assertThat(actualQuestion.get("display_message"), is("Please provide key generator type."));
323         Map<?, ?> actualNextAction = (Map<?, ?>) ((List<?>) initialPayload.get("next_actions")).getFirst();
324         assertThat(actualNextAction.get("required_inputs"), is(List.of("key_generator_type")));
325         ShardingWorkflowRequest continuation = new ShardingWorkflowRequest();
326         continuation.setPlanId(initial.getPlanId());
327         continuation.setKeyGeneratorType("SNOWFLAKE");
328         WorkflowContextSnapshot actual = planningService.planKeyGenerateStrategy(sessionContext, createQueryFacade(List.of()), continuation);
329         assertThat(actual.getStatus(), is(WorkflowLifecycle.STATUS_PLANNED));
330         assertThat(((ShardingWorkflowRequest) actual.getRequest()).getKeyGeneratorProperties(), is(Map.of("worker-id", "1")));
331     }
