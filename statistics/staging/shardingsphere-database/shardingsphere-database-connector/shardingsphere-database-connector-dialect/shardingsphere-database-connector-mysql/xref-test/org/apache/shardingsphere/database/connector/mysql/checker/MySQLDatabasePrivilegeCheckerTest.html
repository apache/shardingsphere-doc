54      private final DatabaseType databaseType = TypedSPILoader.getService(DatabaseType.class, "MySQL");
55      
56      private final DialectDatabasePrivilegeChecker checker = DatabaseTypedSPILoader.getService(DialectDatabasePrivilegeChecker.class, databaseType);
57      
58      @Mock
59      private DataSource dataSource;
60      
61      @Mock
62      private Connection connection;
63      
64      @Mock
65      private DatabaseMetaData databaseMetaData;
66      
67      @Mock
68      private PreparedStatement preparedStatement;
69      
70      @Mock
71      private ResultSet resultSet;
72      
73      @ParameterizedTest(name = "{0}")
74      @MethodSource("checkSuccessArguments")
75      void assertCheckSuccess(final String name, final PrivilegeCheckType privilegeCheckType, final int majorVersion, final String catalog, final String grantedPrivilege) throws SQLException {
76          mockShowGrantsQuery();
77          if (PrivilegeCheckType.XA == privilegeCheckType) {
78              mockXAMetaData(majorVersion);
79          }
80          if (PrivilegeCheckType.SELECT == privilegeCheckType) {
81              when(connection.getCatalog()).thenReturn(catalog);
82          }
83          when(resultSet.next()).thenReturn(true);
84          when(resultSet.getString(1)).thenReturn(grantedPrivilege);
85          checker.check(dataSource, privilegeCheckType);
86          verify(preparedStatement).executeQuery();
87      }
88      
89      @ParameterizedTest(name = "{0}")
90      @MethodSource("checkMissingPrivilegeArguments")
91      void assertCheckWithMissingPrivilege(final String name,
92                                           final PrivilegeCheckType privilegeCheckType, final int majorVersion, final String catalog, final boolean containsGrantRow) throws SQLException {
93          mockShowGrantsQuery();
94          if (PrivilegeCheckType.XA == privilegeCheckType) {
95              mockXAMetaData(majorVersion);
96          }
97          if (PrivilegeCheckType.SELECT == privilegeCheckType) {
98              when(connection.getCatalog()).thenReturn(catalog);
99          }
100         if (containsGrantRow) {
101             when(resultSet.next()).thenReturn(true, false);
102             when(resultSet.getString(1)).thenReturn("GRANT INSERT ON *.* TO '%'@'%'");
103         }
104         assertThrows(MissingRequiredPrivilegeException.class, () -> checker.check(dataSource, privilegeCheckType));
105     }

194     @Test
195     void assertCheckWithMissingPrivilegeAndCloseConnectionFailure() throws SQLException {
196         mockShowGrantsQuery();
197         SQLException expected = new SQLException("mocked close connection failure");
198         doThrow(expected).when(connection).close();
199         MissingRequiredPrivilegeException actual = assertThrows(MissingRequiredPrivilegeException.class, () -> checker.check(dataSource, PrivilegeCheckType.PIPELINE));
200         assertThat(actual.getSuppressed().length, is(1));
201         assertThat(actual.getSuppressed()[0], is(expected));
202     }

206         when(dataSource.getConnection()).thenReturn(connection);
207         when(connection.getMetaData()).thenReturn(databaseMetaData);
208         when(databaseMetaData.getDatabaseMajorVersion()).thenReturn(5);
209         checker.check(dataSource, PrivilegeCheckType.XA);
210         verify(preparedStatement, never()).executeQuery();
211     }

213     private void mockShowGrantsQuery() throws SQLException {
214         mockConnectionAndPreparedStatement();
215         when(preparedStatement.executeQuery()).thenReturn(resultSet);
216     }
217     
218     private void mockConnectionAndPreparedStatement() throws SQLException {
219         when(dataSource.getConnection()).thenReturn(connection);
220         when(connection.prepareStatement("SHOW GRANTS")).thenReturn(preparedStatement);
221     }
222     
223     private void mockXAMetaData(final int majorVersion) throws SQLException {
224         when(connection.getMetaData()).thenReturn(databaseMetaData);
225         when(databaseMetaData.getDatabaseMajorVersion()).thenReturn(majorVersion);
226     }

228     private static Stream<Arguments> checkSuccessArguments() {
229         return Stream.of(
230                 Arguments.of("pipeline with replication privilege", PrivilegeCheckType.PIPELINE, 0, null, "GRANT REPLICATION SLAVE, REPLICATION CLIENT ON *.* TO '%'@'%'"),
231                 Arguments.of("pipeline with all privileges", PrivilegeCheckType.PIPELINE, 0, null, "GRANT ALL PRIVILEGES ON *.* TO '%'@'%'"),
232                 Arguments.of("xa with xa recover admin on mysql8", PrivilegeCheckType.XA, 8, null, "GRANT XA_RECOVER_ADMIN ON *.* TO '%'@'%'"),
233                 Arguments.of("xa with all privileges on mysql8", PrivilegeCheckType.XA, 8, null, "GRANT ALL PRIVILEGES ON *.* TO '%'@'%'"),
234                 Arguments.of("select with select privilege on all databases", PrivilegeCheckType.SELECT, 0, "foo_db", "GRANT SELECT ON *.* TO '%'@'%'"),
235                 Arguments.of("select with select privilege on target database", PrivilegeCheckType.SELECT, 0, "foo_db", "GRANT SELECT ON `FOO_DB`.* TO '%'@'%'"),
236                 Arguments.of("select with escaped target database", PrivilegeCheckType.SELECT, 0, "account_db", "GRANT ALL PRIVILEGES ON `ACCOUNT\\_DB`.* TO '%'@'%'"),
237                 Arguments.of("select with all privileges on all databases", PrivilegeCheckType.SELECT, 0, "foo_db", "GRANT ALL PRIVILEGES ON *.* TO '%'@'%'"),
238                 Arguments.of("select with all privileges on target database", PrivilegeCheckType.SELECT, 0, "foo_db", "GRANT ALL PRIVILEGES ON `FOO_DB`.* TO '%'@'%'"));
239     }
240     
241     private static Stream<Arguments> checkMissingPrivilegeArguments() {
242         return Stream.of(
243                 Arguments.of("pipeline without grant rows", PrivilegeCheckType.PIPELINE, 0, null, false),
244                 Arguments.of("pipeline with unrelated grant row", PrivilegeCheckType.PIPELINE, 0, null, true),
245                 Arguments.of("xa without grant rows on mysql8", PrivilegeCheckType.XA, 8, null, false),
246                 Arguments.of("select with unrelated grant row", PrivilegeCheckType.SELECT, 0, "foo_db", true),
247                 Arguments.of("none with unrelated grant row", PrivilegeCheckType.NONE, 0, null, true));
248     }
