76      /**
77       * Assert actual table segments is correct with expected tables.
78       *
79       * @param assertContext assert context
80       * @param actual actual tables
81       * @param expected expected tables
82       * @throws UnsupportedOperationException unsupported operation exception
83       */
84      public static void assertIs(final SQLCaseAssertContext assertContext, final TableSegment actual, final ExpectedTable expected) {
85          if (actual instanceof JoinTableSegment) {
86              assertIs(assertContext, (JoinTableSegment) actual, expected.getJoinTable());
87          } else if (actual instanceof SimpleTableSegment) {
88              assertIs(assertContext, (SimpleTableSegment) actual, expected.getSimpleTable());
89          } else if (actual instanceof SubqueryTableSegment) {
90              assertIs(assertContext, (SubqueryTableSegment) actual, expected.getSubqueryTable());
91          } else if (actual instanceof FunctionTableSegment) {
92              assertIs(assertContext, (FunctionTableSegment) actual, expected.getFunctionTable());
93          } else if (actual instanceof CollectionTableSegment) {
94              assertIs(assertContext, (CollectionTableSegment) actual, expected.getCollectionTable());
95          } else {
96              throw new UnsupportedOperationException(String.format("Unsupported table segment type `%s`.", actual.getClass()));
97          }
98      }

128     /**
129      * Assert actual table segment is correct with expected table.
130      *
131      * @param assertContext assert context
132      * @param actual actual table
133      * @param expected expected table
134      */
135     public static void assertIs(final SQLCaseAssertContext assertContext, final SimpleTableSegment actual, final ExpectedSimpleTable expected) {
136         assertNotNull(actual, assertContext.getText("Actual table should exist."));
137         assertTableNameSegment(assertContext, actual, expected);
138         assertThat(assertContext.getText("Table alias assertion error: "), actual.getAliasName().orElse(null), is(expected.getAlias()));
139         if (null != expected.getDbLink()) {
140             assertThat(assertContext.getText("Database link assertion error: "), actual.getDbLink().map(each -> each.getValue()).orElse(null), is(expected.getDbLink()));
141         }
142         if (null != expected.getAt()) {
143             assertThat(assertContext.getText("Database link separator assertion error: "), actual.getAt().map(each -> each.getValue()).orElse(null), is(expected.getAt()));
144         }
145         if (null != expected.getTableNameStartIndex()) {
146             assertThat(assertContext.getText("Table name start index assertion error: "), actual.getTableName().getStartIndex(), is(expected.getTableNameStartIndex()));
147         }
148         if (null != expected.getTableNameStopIndex()) {
149             assertThat(assertContext.getText("Table name stop index assertion error: "), actual.getTableName().getStopIndex(), is(expected.getTableNameStopIndex()));
150         }
151         if (null == expected.getOwner()) {
152             assertFalse(actual.getOwner().isPresent(), assertContext.getText("Actual owner should not exist."));
153         } else {
154             assertTrue(actual.getOwner().isPresent(), assertContext.getText("Actual owner should exist."));
155             OwnerAssert.assertIs(assertContext, actual.getOwner().get(), expected.getOwner());
156         }
157         if (!expected.getIndexHints().isEmpty()) {
158             assertFalse(actual.getIndexHintSegments().isEmpty());
159             assertIs(assertContext, actual.getIndexHintSegments(), expected.getIndexHints());
160         }
161         assertPivot(assertContext, actual.getPivot().orElse(null), expected.getPivot());
162         SQLSegmentAssert.assertIs(assertContext, actual, expected);
163     }

165     private static void assertIs(final SQLCaseAssertContext assertContext, final Collection<IndexHintSegment> actual, final Collection<ExpectedIndexHint> expected) {
166         assertThat(expected.size(), is(actual.size()));
167         Iterator<ExpectedIndexHint> expectedIndexHintIterator = expected.iterator();
168         Iterator<IndexHintSegment> actualIndexHintIterator = actual.iterator();
169         while (expectedIndexHintIterator.hasNext()) {
170             ExpectedIndexHint expectedIndexHint = expectedIndexHintIterator.next();
171             IndexHintSegment actualIndexHint = actualIndexHintIterator.next();
172             assertIs(assertContext, actualIndexHint, expectedIndexHint);
173         }
174     }

176     private static void assertIs(final SQLCaseAssertContext assertContext, final IndexHintSegment actual, final ExpectedIndexHint expected) {
177         assertThat(assertContext.getText("Index hint start index assertion error: "), actual.getStartIndex(), is(expected.getStartIndex()));
178         assertThat(assertContext.getText("Index hint stop index assertion error: "), actual.getStopIndex(), is(expected.getStopIndex()));
179         assertThat(expected.getHintIndexNames().size(), is(actual.getIndexNames().size()));
180         Iterator<ExpectedHintIndexName> expectedIndexNameIterator = expected.getHintIndexNames().iterator();
181         Iterator<String> actualIndexNameIterator = actual.getIndexNames().iterator();
182         int indexPosition = 0;
183         while (expectedIndexNameIterator.hasNext()) {
184             ExpectedHintIndexName expectedIndexName = expectedIndexNameIterator.next();
185             String actualIndexName = actualIndexNameIterator.next();
186             assertThat(assertContext.getText("Index hint name assertion error: "), actualIndexName, is(expectedIndexName.getName()));
187             if (indexPosition < actual.getIndexStartIndices().size() && indexPosition < actual.getIndexStopIndices().size()) {
188                 assertThat(assertContext.getText("Index hint name start index assertion error: "),
189                         actual.getIndexStartIndices().get(indexPosition), is(expectedIndexName.getStartIndex()));
190                 assertThat(assertContext.getText("Index hint name stop index assertion error: "),
191                         actual.getIndexStopIndices().get(indexPosition), is(expectedIndexName.getStopIndex()));
192             }
193             indexPosition++;
194         }
195         assertThat(assertContext.getText("Index hint origin text assertion error: "), actual.getOriginText(), is(expected.getOriginText()));
196     }

198     /**
199      * Assert subquery expression.
200      *
201      * @param assertContext assert context
202      * @param actual actual subquery segment
203      * @param expected expected subquery expression
204      */
205     public static void assertIs(final SQLCaseAssertContext assertContext, final SubqueryTableSegment actual, final ExpectedSubqueryTable expected) {
206         SQLSegmentAssert.assertIs(assertContext, actual, expected);
207         if (null != actual.getSubquery().getSelect()) {
208             SelectStatementAssert.assertIs(assertContext, actual.getSubquery().getSelect(), expected.getSubquery().getSelectTestCases());
209         }
210         if (null != actual.getSubquery().getMerge()) {
211             MergeStatementAssert.assertIs(assertContext, actual.getSubquery().getMerge(), expected.getSubquery().getMergeTestCases());
212         }
213         assertThat(assertContext.getText("Table alias assertion error: "), actual.getAliasName().orElse(null), is(expected.getAlias()));
214         if (!expected.getColumns().isEmpty()) {
215             assertThat(assertContext.getText("Subquery table columns size assertion error: "), actual.getColumns().size(), is(expected.getColumns().size()));
216             int count = 0;
217             for (ColumnSegment each : actual.getColumns()) {
218                 ColumnAssert.assertIs(assertContext, each, expected.getColumns().get(count));
219                 count++;
220             }
221         }
222         if (!expected.getColumnAliases().isEmpty()) {
223             assertTrue(actual.getAliasSegment().isPresent(), assertContext.getText("Subquery table alias should exist."));
224             assertThat(assertContext.getText("Subquery table column aliases size assertion error: "),
225                     actual.getAliasSegment().get().getColumnAliases().size(), is(expected.getColumnAliases().size()));
226             Iterator<String> expectedIterator = expected.getColumnAliases().iterator();
227             actual.getAliasSegment().get().getColumnAliases()
228                     .forEach(each -> assertThat(assertContext.getText("Subquery table column alias assertion error: "), each.getValue(), is(expectedIterator.next())));
229         }
230         assertPivot(assertContext, actual.getPivot().orElse(null), expected.getPivot());
231     }

265      * Assert join table.
266      *
267      * @param assertContext assert context
268      * @param actual actual JoinTableSegment
269      * @param expected expected ExpectedJoinTable
270      */
271     public static void assertIs(final SQLCaseAssertContext assertContext, final JoinTableSegment actual, final ExpectedJoinTable expected) {
272         assertIs(assertContext, actual.getLeft(), expected.getLeft());
273         assertIs(assertContext, actual.getRight(), expected.getRight());
274         ExpressionAssert.assertExpression(assertContext, actual.getCondition(), expected.getOnCondition());
275         assertJoinType(assertContext, actual.getJoinType(), expected.getJoinType());
276         assertThat(assertContext.getText("Natural should be the same."), actual.isNatural(), is(expected.isNatural()));
277         assertThat(assertContext.getText("Column size assertion error: "), actual.getUsing().size(), is(expected.getUsingColumns().size()));
278         int count = 0;
279         for (ExpectedColumn each : expected.getUsingColumns()) {
280             ColumnAssert.assertIs(assertContext, actual.getUsing().get(count), each);
281             count++;
282         }
283         assertUsingAlias(assertContext, actual, expected);
284         assertColumns(assertContext, actual.getLeftQueryPartitionListSegments(), expected.getLeftQueryPartitionColumns(), "Left query partition");
285         assertColumns(assertContext, actual.getRightQueryPartitionListSegments(), expected.getRightQueryPartitionColumns(), "Right query partition");
286     }

288     private static void assertUsingAlias(final SQLCaseAssertContext assertContext, final JoinTableSegment actual, final ExpectedJoinTable expected) {
289         if (null == expected.getUsingAlias()) {
290             assertFalse(actual.getUsingAliasSegment().isPresent(), assertContext.getText("USING alias should not exist."));
291             return;
292         }
293         assertTrue(actual.getUsingAliasSegment().isPresent(), assertContext.getText("USING alias should exist."));
294         assertThat(actual.getUsingAliasSegment().get().getIdentifier().getValue(), is(expected.getUsingAlias()));
295         assertThat(actual.getUsingAliasSegment().get().getStartIndex(), is(expected.getUsingAliasStartIndex()));
296         assertThat(actual.getUsingAliasSegment().get().getStopIndex(), is(expected.getUsingAliasStopIndex()));
297     }

299     private static void assertPivot(final SQLCaseAssertContext assertContext, final PivotSegment actual, final ExpectedPivot expected) {
300         if (null == expected) {
301             return;
302         }
303         assertNotNull(actual, assertContext.getText("Actual pivot should exist."));
304         SQLSegmentAssert.assertIs(assertContext, actual, expected);
305         assertThat(assertContext.getText("Pivot unpivot assertion error: "), actual.isUnPivot(), is(expected.isUnpivot()));
306         assertThat(assertContext.getText("Pivot XML assertion error: "), actual.isXml(), is(expected.isXml()));
307         assertColumns(assertContext, actual.getPivotAggregationColumns(), expected.getAggregationColumns(), "Pivot aggregation");
308         assertColumns(assertContext, actual.getPivotForColumns(), expected.getForColumns(), "Pivot for");
309         assertColumns(assertContext, actual.getPivotInColumns(), expected.getInColumns(), "Pivot in");
310         if (null != actual.getUnpivotColumns()) {
311             assertColumns(assertContext, actual.getUnpivotColumns(), expected.getUnpivotColumns(), "Unpivot");
312         }
313     }

315     private static void assertColumns(final SQLCaseAssertContext assertContext, final Collection<? extends ExpressionSegment> actual, final Collection<ExpectedColumn> expected,
316                                       final String segmentName) {
317         if (expected.isEmpty()) {
318             return;
319         }
320         assertThat(assertContext.getText(segmentName + " columns size assertion error: "), actual.size(), is(expected.size()));
321         Iterator<? extends ExpressionSegment> actualIterator = actual.iterator();
322         for (ExpectedColumn each : expected) {
323             ExpressionSegment actualExpression = actualIterator.next();
324             assertThat(assertContext.getText(segmentName + " column expression assertion error: "), actualExpression, isA(ColumnSegment.class));
325             ColumnAssert.assertIs(assertContext, (ColumnSegment) actualExpression, each);
326         }
327     }

329     private static void assertJoinType(final SQLCaseAssertContext assertContext, final String actual, final String expected) {
330         if (Strings.isNullOrEmpty(expected)) {
331             assertTrue(Strings.isNullOrEmpty(actual), assertContext.getText("Actual join-type should not exist."));
332         } else {
333             assertThat(assertContext.getText("Actual join-type should exist."), actual, is(expected));
334         }
335     }

337     private static void assertTableFunction(final SQLCaseAssertContext assertContext, final ExpressionSegment actual, final ExpectedTableFunction expected) {
338         if (actual instanceof XmlTableFunctionSegment) {
339             XmlTableFunctionSegment actualXmlTableFunction = (XmlTableFunctionSegment) actual;
340             assertThat(assertContext.getText("Function name assertion error"), actualXmlTableFunction.getFunctionName(), is(expected.getFunctionName()));
341             assertThat(assertContext.getText("Function text assert error"), actual.getText(), is(expected.getText()));
342             assertFunctionParameters(assertContext, actualXmlTableFunction.getXmlTableOption().getParameters(), expected.getParameters());
343             assertXmlTableColumns(assertContext, actualXmlTableFunction.getXmlTableOption().getXmlTableColumnSegments(), expected.getXmlColumns());
344         } else if (actual instanceof FunctionSegment) {
345             FunctionSegment actualTableFunction = (FunctionSegment) actual;
346             assertThat(assertContext.getText("Function name assertion error"), actualTableFunction.getFunctionName(), is(expected.getFunctionName()));
347             assertThat(assertContext.getText("Function text assert error"), actual.getText(), is(expected.getText()));
348             assertFunctionParameters(assertContext, actualTableFunction.getParameters(), expected.getParameters());
349         }
350     }

352     private static void assertFunctionParameters(final SQLCaseAssertContext assertContext, final Collection<ExpressionSegment> actual, final List<ExpectedExpression> expected) {
353         if (expected.isEmpty()) {
354             return;
355         }
356         assertThat(assertContext.getText("Table function parameter count assertion error"), actual.size(), is(expected.size()));
357         int count = 0;
358         for (ExpressionSegment each : actual) {
359             ExpressionAssert.assertExpression(assertContext, each, expected.get(count++));
360         }
361     }

363     private static void assertXmlTableColumns(final SQLCaseAssertContext assertContext, final Collection<XmlTableColumnSegment> actual, final List<ExpectedXmlTableColumn> expected) {
364         if (expected.isEmpty()) {
365             return;
366         }
367         assertThat(assertContext.getText("XMLTABLE column count assertion error"), actual.size(), is(expected.size()));
368         int count = 0;
369         for (XmlTableColumnSegment each : actual) {
370             ExpectedXmlTableColumn expectedColumn = expected.get(count++);
371             assertThat(each.getColumnName(), is(expectedColumn.getName()));
372             assertThat(each.getDataType(), is(expectedColumn.getDataType()));
373             assertThat(each.getPath(), is(expectedColumn.getPath()));
374             ExpressionAssert.assertExpression(assertContext, each.getDefaultExpr(), expectedColumn.getDefaultExpr());
375             SQLSegmentAssert.assertIs(assertContext, each, expectedColumn);
376         }
377     }
