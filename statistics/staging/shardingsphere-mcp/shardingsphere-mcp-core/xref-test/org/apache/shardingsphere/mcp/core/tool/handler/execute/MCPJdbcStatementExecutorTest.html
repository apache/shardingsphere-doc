92      @ParameterizedTest(name = "{0}")
93      @MethodSource("readOnlyFailureCases")
94      void assertExecuteReadOnlyWithFailure(final String name, final boolean executionFailure, final boolean rollbackFailure, final boolean autoCommitFailure, final boolean readOnlyFailure,
95                                            final String expectedMessage, final Collection<String> expectedSuppressed) throws SQLException {
96          Statement statement = mock(Statement.class);
97          Connection connection = createStatementConnection(statement);
98          when(connection.getAutoCommit()).thenReturn(true);
99          if (executionFailure) {
100             when(statement.execute("SELECT 1")).thenThrow(new SQLException("execution failed"));
101         } else {
102             when(statement.execute("SELECT 1")).thenReturn(true);
103             ResultSet resultSet = createResultSet(createColumns(), List.of(List.of(1, "foo_status")));
104             when(statement.getResultSet()).thenReturn(resultSet);
105         }
106         if (rollbackFailure) {
107             doThrow(new SQLException("rollback failed")).when(connection).rollback();
108         }
109         if (autoCommitFailure) {
110             doThrow(new SQLException("autoCommit failed")).when(connection).setAutoCommit(true);
111         }
112         if (readOnlyFailure) {
113             doThrow(new SQLException("readOnly failed")).when(connection).setReadOnly(false);
114         }
115         RuntimeDatabaseConfiguration databaseConfig = mock(RuntimeDatabaseConfiguration.class);
116         when(databaseConfig.openConnection("foo_db")).thenReturn(connection);
117         MCPJdbcStatementExecutor executor = new MCPJdbcStatementExecutor(Map.of("foo_db", databaseConfig), mock(MCPJdbcTransactionResourceManager.class));
118         SQLExecutionRequest request = new SQLExecutionRequest("foo_session", "foo_db", "", "SELECT 1", 0, 0, true);
119         ClassificationResult classification = new ClassificationResult(SupportedMCPStatement.QUERY, "SELECT", "SELECT 1", "", List.of(), false);
120         MCPDatabaseQueryFailedException actual = assertThrows(MCPDatabaseQueryFailedException.class,
121                 () -> executor.execute(request, classification, createDatabaseCapability(SchemaExecutionSemantics.FIXED_TO_DATABASE)));
122         assertThat(actual.getCause().getMessage(), is(expectedMessage));
123         assertThat(Arrays.stream(actual.getCause().getSuppressed()).map(Throwable::getMessage).toList(), is(expectedSuppressed));
124         verify(connection).rollback();
125         verify(connection).setAutoCommit(true);
126         verify(connection).setReadOnly(false);
127         verify(connection).close();
128     }

586     private static ResultSet createResultSet(final List<SQLExecutionColumnDefinition> columns, final List<List<Object>> rows) throws SQLException {
587         ResultSet result = mock(ResultSet.class);
588         ResultSetMetaData resultSetMetaData = mock(ResultSetMetaData.class);
589         AtomicInteger rowIndex = new AtomicInteger(-1);
590         when(result.getMetaData()).thenReturn(resultSetMetaData);
591         when(result.next()).thenAnswer(invocation -> rowIndex.incrementAndGet() < rows.size());
592         when(result.getObject(anyInt())).thenAnswer(invocation -> rows.get(rowIndex.get()).get(invocation.getArgument(0, Integer.class) - 1));
593         when(resultSetMetaData.getColumnCount()).thenReturn(columns.size());
594         for (int index = 0; index < columns.size(); index++) {
595             SQLExecutionColumnDefinition column = columns.get(index);
596             int columnIndex = index + 1;
597             when(resultSetMetaData.getColumnLabel(columnIndex)).thenReturn(column.getColumnName());
598             when(resultSetMetaData.getColumnTypeName(columnIndex)).thenReturn(column.getNativeType());
599             when(resultSetMetaData.isNullable(columnIndex)).thenReturn(column.isNullable() ? ResultSetMetaData.columnNullable : ResultSetMetaData.columnNoNulls);
600         }
601         return result;
602     }
