53  @ExtendWith(AutoMockExtension.class)
54  @StaticMockSettings(ProxyContext.class)
55  class FirebirdStatementResourceCleanerTest {
56      
57      private static final int CONNECTION_ID = 1;
58      
59      private static final int STATEMENT_ID = 2;
60      
61      @Mock(answer = Answers.RETURNS_DEEP_STUBS)
62      private ConnectionSession connectionSession;
63      
64      @Mock
65      private ProxyDatabaseConnectionManager connectionManager;
66      
67      @Mock
68      private ConnectionContext connectionContext;
69      
70      @Mock
71      private ProxyBackendHandler proxyBackendHandler;
72      
73      @AfterEach
74      void tearDown() {
75          FirebirdFetchStatementCache.getInstance().unregisterConnection(CONNECTION_ID);
76      }
77      
78      @Test
79      void assertCreatePreparedStatementCacheKey() {
80          assertThat(FirebirdStatementResourceCleaner.createPreparedStatementCacheKey(STATEMENT_ID).getToken(), is("firebird:2"));
81      }
82      
83      @Test
84      void assertCleanWithPreparedStatementCacheInvalidation() throws SQLException {
85          FirebirdFetchStatementCache.getInstance().registerConnection(CONNECTION_ID);
86          FirebirdFetchStatementCache.getInstance().registerStatement(CONNECTION_ID, STATEMENT_ID, proxyBackendHandler);
87          when(connectionSession.getConnectionId()).thenReturn(CONNECTION_ID);
88          when(connectionSession.getDatabaseConnectionManager()).thenReturn(connectionManager);
89          when(connectionSession.getConnectionContext()).thenReturn(connectionContext);
90          FirebirdStatementResourceCleaner.clean(connectionSession, STATEMENT_ID, true);
91          InOrder inOrder = inOrder(connectionManager, connectionContext, proxyBackendHandler, connectionSession);
92          inOrder.verify(connectionManager).removeResource(proxyBackendHandler);
93          inOrder.verify(connectionContext).clearCursorContext();
94          inOrder.verify(proxyBackendHandler).close();
95          inOrder.verify(connectionSession).invalidatePreparedStatementCache(FirebirdStatementResourceCleaner.createPreparedStatementCacheKey(STATEMENT_ID));
96          assertNull(FirebirdFetchStatementCache.getInstance().getFetchBackendHandler(CONNECTION_ID, STATEMENT_ID));
97      }
98      
99      @Test
100     void assertCleanWithPreparedStatementCacheInvalidationWithoutFetchHandler() throws SQLException {
101         when(connectionSession.getConnectionId()).thenReturn(CONNECTION_ID);
102         when(connectionSession.getConnectionContext()).thenReturn(connectionContext);
103         FirebirdStatementResourceCleaner.clean(connectionSession, STATEMENT_ID, true);
104         verify(connectionContext).clearCursorContext();
105         verify(connectionSession).invalidatePreparedStatementCache(FirebirdStatementResourceCleaner.createPreparedStatementCacheKey(STATEMENT_ID));
106         verifyNoInteractions(connectionManager, proxyBackendHandler);
107     }
108     
109     @Test
110     void assertCleanWithoutPreparedStatementCacheInvalidation() throws SQLException {
111         FirebirdFetchStatementCache.getInstance().registerConnection(CONNECTION_ID);
112         FirebirdFetchStatementCache.getInstance().registerStatement(CONNECTION_ID, STATEMENT_ID, proxyBackendHandler);
113         when(connectionSession.getConnectionId()).thenReturn(CONNECTION_ID);
114         when(connectionSession.getDatabaseConnectionManager()).thenReturn(connectionManager);
115         when(connectionSession.getConnectionContext()).thenReturn(connectionContext);
116         FirebirdStatementResourceCleaner.clean(connectionSession, STATEMENT_ID, false);
117         verify(connectionManager).removeResource(proxyBackendHandler);
118         verify(proxyBackendHandler).close();
119         verify(connectionContext).clearCursorContext();
120         verify(connectionSession, never()).invalidatePreparedStatementCache(any());
121         assertNull(FirebirdFetchStatementCache.getInstance().getFetchBackendHandler(CONNECTION_ID, STATEMENT_ID));
122     }
123     
124     @Test
125     void assertCleanBeforeCloseExecutionResources() throws SQLException, BackendConnectionException {
126         ConnectionSession actualConnectionSession = mock(ConnectionSession.class, Answers.RETURNS_DEEP_STUBS);
127         ProxyBackendHandler actualProxyBackendHandler = mock(ProxyBackendHandler.class);
128         when(ProxyContext.getInstance().getContextManager().getMetaDataContexts().getMetaData().getGlobalRuleMetaData().getRules()).thenReturn(Collections.emptyList());
129         when(actualConnectionSession.getConnectionId()).thenReturn(CONNECTION_ID);
130         when(actualConnectionSession.getConnectionContext().getTransactionContext().getTransactionType()).thenReturn(Optional.of("XA"));
131         when(actualConnectionSession.getTransactionStatus().isInConnectionHeldTransaction(any())).thenReturn(true);
132         ProxyDatabaseConnectionManager actualConnectionManager = new ProxyDatabaseConnectionManager(actualConnectionSession);
133         when(actualConnectionSession.getDatabaseConnectionManager()).thenReturn(actualConnectionManager);
134         FirebirdFetchStatementCache.getInstance().registerConnection(CONNECTION_ID);
135         FirebirdFetchStatementCache.getInstance().registerStatement(CONNECTION_ID, STATEMENT_ID, actualProxyBackendHandler);
136         actualConnectionManager.add(actualProxyBackendHandler);
137         actualConnectionManager.markResourceInUse(actualProxyBackendHandler);
138         AtomicBoolean closed = new AtomicBoolean(false);
139         doAnswer(invocation -> {
140             if (closed.getAndSet(true)) {
141                 throw new SQLException("close twice");
142             }
143             return null;
144         }).when(actualProxyBackendHandler).close();
145         FirebirdStatementResourceCleaner.clean(actualConnectionSession, STATEMENT_ID, true);
146         actualConnectionManager.closeExecutionResources();
147         verify(actualProxyBackendHandler).close();
148         verify(actualConnectionSession).invalidatePreparedStatementCache(FirebirdStatementResourceCleaner.createPreparedStatementCacheKey(STATEMENT_ID));
149         assertNull(FirebirdFetchStatementCache.getInstance().getFetchBackendHandler(CONNECTION_ID, STATEMENT_ID));
150     }
