50      @Test
51      void assertExecuteSearchAcrossDatabases() {
52          MetadataSearchResult actual = execute(createDatabaseMetadata(), createRequest("", "", "order",
53                  Set.of(SupportedMCPMetadataObjectType.TABLE, SupportedMCPMetadataObjectType.VIEW, SupportedMCPMetadataObjectType.INDEX)));
54          Set<String> actualNames = actual.getItems().stream().map(MetadataSearchHit::getName).collect(Collectors.toSet());
55          assertTrue(actualNames.contains("orders"));
56          assertTrue(actualNames.contains("order_items"));
57          assertTrue(actualNames.contains("active_orders"));
58          assertTrue(actualNames.contains("idx_orders_status"));
59          assertThat(extractResourceHintUri(findHit(actual, "idx_orders_status")), is("shardingsphere://databases/logic_db/schemas/public/tables/orders/indexes/idx_orders_status"));
60          assertFalse(actualNames.contains("mv_orders"));
61          assertFalse(actualNames.contains("order_seq"));
62      }

125     @Test
126     void assertExecuteSearchWithCompleteResult() {
127         Set<SupportedMCPMetadataObjectType> objectTypes = Set.of(SupportedMCPMetadataObjectType.TABLE, SupportedMCPMetadataObjectType.VIEW);
128         MetadataSearchResult actual = execute(createDatabaseMetadata(),
129                 createRequest("logic_db", "", "order", objectTypes));
130         assertThat(actual.getItems().size(), is(4));
131         assertThat(actual.getItems().getFirst().getName(), is("order_items"));
132         assertThat(actual.getItems().getFirst().getMatchKind(), is("prefix"));
133         assertThat(actual.getItems().getFirst().getMatchedFields(), is(List.of("name", "table")));
134         assertThat(actual.getItems().getFirst().getMatchedValue(), is("order_items"));
135         assertThat(actual.getItems().get(1).getName(), is("orders"));
136         assertThat(actual.getItems().get(2).getName(), is("active_orders"));
137         assertThat(actual.getItems().get(3).getName(), is("archived_orders"));
138         Map<?, ?> actualSearchContext = actual.getSearchContext();
139         assertThat(actualSearchContext.get("query"), is("order"));
140         assertThat(actualSearchContext.get("database"), is("logic_db"));
141         assertThat(actualSearchContext.get("database_scope"), is("single_database"));
142         assertThat(actualSearchContext.get("schema"), is(""));
143         assertThat(actualSearchContext.get("object_types"), is(List.of("table", "view")));
144         assertThat(actualSearchContext.keySet(), is(Set.of("query", "database", "database_scope", "schema", "object_types", "limit", "offset")));
145         assertThat(actual.getTotalMatchCount(), is(4));
146     }

328     private MetadataSearchRequest createRequest(final String database, final String schema, final String query,
329                                                 final Set<SupportedMCPMetadataObjectType> objectTypes) {
330         return new MetadataSearchRequest(database, schema, query, objectTypes, 100, 0);
331     }

333     private MetadataSearchHit findHit(final MetadataSearchResult searchResult, final String name) {
334         return searchResult.getItems().stream().filter(each -> name.equals(each.getName())).findFirst().orElseThrow();
335     }
336     
337     private String extractResourceHintUri(final MetadataSearchHit hit) {
338         return String.valueOf(hit.getResource().getOrDefault("uri", ""));
339     }
340     
341     private String extractParentResourceHintUri(final MetadataSearchHit hit) {
342         return String.valueOf(hit.getParentResource().getOrDefault("uri", ""));
343     }
344     
345     private List<String> extractNextResourceHintUris(final MetadataSearchHit hit) {
346         return hit.getNextResources().stream().map(each -> String.valueOf(each.get("uri"))).toList();
347     }
