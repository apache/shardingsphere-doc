61  @ExtendWith(AutoMockExtension.class)
62  @StaticMockSettings(TableMetaDataLoader.class)
63  class HiveMetaDataLoaderTest {
64      
65      private final DatabaseType databaseType = TypedSPILoader.getService(DatabaseType.class, "Hive");
66      
67      private final DialectMetaDataLoader loader = DatabaseTypedSPILoader.getService(DialectMetaDataLoader.class, databaseType);
68      
69      private Map<String, Map<String, Integer>> dataTypes;
70      
71      @BeforeEach
72      void setUp() {
73          dataTypes = getDataTypes();
74          Map<String, Integer> hiveDataTypes = new CaseInsensitiveMap<>();
75          hiveDataTypes.put("string", Types.VARCHAR);
76          hiveDataTypes.put("int", Types.INTEGER);
77          dataTypes.put("Hive", hiveDataTypes);
78      }
79      
80      @SuppressWarnings("unchecked")
81      @SneakyThrows(ReflectiveOperationException.class)
82      private Map<String, Map<String, Integer>> getDataTypes() {
83          return (Map<String, Map<String, Integer>>) Plugins.getMemberAccessor().get(DataTypeRegistry.class.getDeclaredField("DATA_TYPES"), null);
84      }
85      
86      @AfterEach
87      void tearDown() {
88          dataTypes.remove("Hive");
89      }
90      
91      @Test
92      void assertLoadWithInformationSchemaAndFilter() throws SQLException {
93          DataSource dataSource = mockDataSource(mockInformationSchemaConnection(true), mockColumnMetaDataConnectionWithInformationSchemaAndFilter());
94          Collection<SchemaMetaData> filteredSchemas = loader.load(new MetaDataLoaderMaterial(Collections.singleton("target_table"), "ds_0", dataSource, databaseType, "def_schema"));
95          SchemaMetaData filteredSchema = filteredSchemas.iterator().next();
96          TableMetaData filteredTable = filteredSchema.getTables().iterator().next();
97          Iterator<ColumnMetaData> filteredColumns = filteredTable.getColumns().iterator();
98          ColumnMetaData firstColumn = filteredColumns.next();
99          assertThat(firstColumn.getName(), is("c1"));
100         assertThat(firstColumn.getDataType(), is(Types.VARCHAR));
101         assertTrue(firstColumn.isNullable());
102         ColumnMetaData secondColumn = filteredColumns.next();
103         assertThat(secondColumn.getDataType(), is(Types.OTHER));
104         assertFalse(secondColumn.isNullable());
105     }
106     
107     @Test
108     void assertLoadWithInformationSchemaWithoutFilter() throws SQLException {
109         DataSource dataSource = mockDataSource(mockInformationSchemaConnection(true), mockColumnMetaDataConnectionWithInformationSchemaWithoutFilter());
110         Collection<SchemaMetaData> fullSchemas = loader.load(new MetaDataLoaderMaterial(Collections.emptyList(), "ds_1", dataSource, databaseType, "default_db"));
111         SchemaMetaData fullSchema = fullSchemas.iterator().next();
112         assertThat(fullSchema.getTables().iterator().next().getName(), is("full_table"));
113     }
114     
115     @Test
116     void assertLoadWithoutInformationSchemaFallbackToDefaultLoader() throws SQLException {
117         DataSource dataSource = mockDataSource(mockInformationSchemaConnection(false));
118         TableMetaData tableMetaData = new TableMetaData("present_table", Collections.emptyList(), Collections.emptyList(), Collections.emptyList());
119         when(TableMetaDataLoader.load(dataSource, "missing_table", databaseType)).thenReturn(Optional.empty());
120         when(TableMetaDataLoader.load(dataSource, "present_table", databaseType)).thenReturn(Optional.of(tableMetaData));
121         Collection<SchemaMetaData> fallbackSchemas = loader.load(new MetaDataLoaderMaterial(Arrays.asList("missing_table", "present_table"), "ds_2", dataSource, databaseType, "fallback_schema"));
122         TableMetaData defaultLoadedTable = fallbackSchemas.iterator().next().getTables().iterator().next();
123         assertThat(defaultLoadedTable.getName(), is("present_table"));
124     }
125     
126     private DataSource mockDataSource(final Connection... connections) throws SQLException {
127         DataSource result = mock(DataSource.class);
128         when(result.getConnection()).thenReturn(connections[0], Arrays.copyOfRange(connections, 1, connections.length));
129         return result;
130     }
131     
132     private Connection mockInformationSchemaConnection(final boolean hasResult) throws SQLException {
133         ResultSet resultSet = mock(ResultSet.class);
134         when(resultSet.next()).thenReturn(hasResult);
135         Statement statement = mock(Statement.class);
136         String sql = "SHOW DATABASES LIKE 'INFORMATION_SCHEMA'";
137         when(statement.executeQuery(sql)).thenReturn(resultSet);
138         Connection result = mock(Connection.class);
139         when(result.createStatement()).thenReturn(statement);
140         return result;
141     }
142     
143     private Connection mockColumnMetaDataConnectionWithInformationSchemaAndFilter() throws SQLException {
144         ResultSet resultSet = mock(ResultSet.class);
145         when(resultSet.next()).thenReturn(true, true, false);
146         when(resultSet.getString("TABLE_NAME")).thenReturn("target_table", "target_table");
147         when(resultSet.getString("COLUMN_NAME")).thenReturn("c1", "c2");
148         when(resultSet.getString("DATA_TYPE")).thenReturn("string", "unknown");
149         when(resultSet.getString("IS_NULLABLE")).thenReturn("YES", "NO");
150         PreparedStatement preparedStatement = mock(PreparedStatement.class);
151         when(preparedStatement.executeQuery()).thenReturn(resultSet);
152         Connection connection = mock(Connection.class);
153         String sql = "SELECT TABLE_CATALOG,TABLE_NAME,COLUMN_NAME,DATA_TYPE,ORDINAL_POSITION,IS_NULLABLE FROM INFORMATION_SCHEMA.COLUMNS "
154                 + "WHERE TABLE_CATALOG=? AND UPPER(TABLE_NAME) IN ('TARGET_TABLE') ORDER BY ORDINAL_POSITION";
155         when(connection.prepareStatement(sql)).thenReturn(preparedStatement);
156         return connection;
157     }
158     
159     private Connection mockColumnMetaDataConnectionWithInformationSchemaWithoutFilter() throws SQLException {
160         ResultSet resultSet = mock(ResultSet.class);
161         when(resultSet.next()).thenReturn(true, false);
162         when(resultSet.getString("TABLE_NAME")).thenReturn("full_table");
163         when(resultSet.getString("COLUMN_NAME")).thenReturn("id");
164         when(resultSet.getString("DATA_TYPE")).thenReturn("int");
165         when(resultSet.getString("IS_NULLABLE")).thenReturn("NO");
166         PreparedStatement preparedStatement = mock(PreparedStatement.class);
167         when(preparedStatement.executeQuery()).thenReturn(resultSet);
168         Connection result = mock(Connection.class);
169         String sql = "SELECT TABLE_CATALOG,TABLE_NAME,COLUMN_NAME,DATA_TYPE,ORDINAL_POSITION,IS_NULLABLE FROM INFORMATION_SCHEMA.COLUMNS WHERE TABLE_CATALOG=? ORDER BY ORDINAL_POSITION";
170         when(result.prepareStatement(sql)).thenReturn(preparedStatement);
171         return result;
172     }
