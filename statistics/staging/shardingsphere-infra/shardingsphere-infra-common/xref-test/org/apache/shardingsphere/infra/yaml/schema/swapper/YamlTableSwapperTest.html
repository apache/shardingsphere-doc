38  import static org.apache.shardingsphere.test.infra.framework.matcher.ShardingSphereAssertionMatchers.deepEqual;
39  import static org.hamcrest.MatcherAssert.assertThat;
40  import static org.hamcrest.Matchers.hasKey;
41  import static org.hamcrest.Matchers.is;
42  import static org.hamcrest.Matchers.not;
43  
44  class YamlTableSwapperTest {
45      
46      private static final String YAML_FILE = "yaml/schema/table.yaml";
47      
48      private static final String EMPTY_YAML_FILE = "yaml/schema/empty-table.yaml";
49      
50      private final YamlTableSwapper swapper = new YamlTableSwapper();
51      
52      @Test
53      void assertSwapToYamlConfiguration() {
54          ShardingSphereTable table = createShardingSphereTable();
55          YamlShardingSphereTable actual = swapper.swapToYamlConfiguration(table);
56          YamlShardingSphereTable expected = YamlEngine.unmarshal(SystemResourceFileUtils.readFile(YAML_FILE), YamlShardingSphereTable.class);
57          assertThat(actual, deepEqual(expected));
58      }
59      
60      @Test
61      void assertSwapToObject() {
62          ShardingSphereTable actual = swapper.swapToObject(YamlEngine.unmarshal(SystemResourceFileUtils.readFile(YAML_FILE), YamlShardingSphereTable.class));
63          ShardingSphereTable expected = createShardingSphereTable();
64          assertThat(actual, deepEqual(expected));
65      }
66      
67      @Test
68      void assertSwapToObjectWithEmptySchema() {
69          ShardingSphereTable actual = swapper.swapToObject(YamlEngine.unmarshal(SystemResourceFileUtils.readFile(EMPTY_YAML_FILE), YamlShardingSphereTable.class));
70          ShardingSphereTable expected = new ShardingSphereTable("", Collections.emptyList(), Collections.emptyList(), Collections.emptyList(), TableType.TABLE);
71          assertThat(actual, deepEqual(expected));
72      }
73      
74      @Test
75      void assertSwapToYamlConfigurationWithCaseSensitiveKeys() {
76          ShardingSphereTable table = new ShardingSphereTable("foo_tbl",
77                  Arrays.asList(new ShardingSphereColumn("Foo_Col", 0, true, false, false, true, false, false),
78                          new ShardingSphereColumn("foo_col", 1, false, false, false, true, false, false)),
79                  Arrays.asList(new ShardingSphereIndex("Foo_Idx", Collections.emptyList(), false), new ShardingSphereIndex("foo_idx", Collections.emptyList(), false)),
80                  Arrays.asList(new ShardingSphereConstraint("Foo_Constraint", "foo_tbl"), new ShardingSphereConstraint("foo_constraint", "foo_tbl")), TableType.TABLE);
81          YamlShardingSphereTable actual = swapper.swapToYamlConfiguration(table);
82          assertThat(actual.getColumns().size(), is(2));
83          assertThat(actual.getColumns(), hasKey("Foo_Col"));
84          assertThat(actual.getColumns(), hasKey("foo_col"));
85          assertThat(actual.getIndexes().size(), is(2));
86          assertThat(actual.getIndexes(), hasKey("Foo_Idx"));
87          assertThat(actual.getIndexes(), hasKey("foo_idx"));
88          assertThat(actual.getConstraints().size(), is(2));
89          assertThat(actual.getConstraints(), hasKey("Foo_Constraint"));
90          assertThat(actual.getConstraints(), hasKey("foo_constraint"));
91          assertThat(actual.getColumns(), not(hasKey("FOO_COL")));
92      }
93      
94      @Test
95      void assertSwapToObjectWithCaseSensitiveYamlKeys() {
96          YamlShardingSphereTable yamlTable = new YamlShardingSphereTable();
97          yamlTable.setName("foo_tbl");
98          yamlTable.setColumns(createYamlColumns());
99          yamlTable.setIndexes(createYamlIndexes());
100         yamlTable.setConstraints(createYamlConstraints());
101         yamlTable.setType(TableType.TABLE);
102         ShardingSphereTable actual = swapper.swapToObject(yamlTable);
103         assertThat(actual.getAllColumns().size(), is(2));
104         assertThat(actual.getAllIndexes().size(), is(2));
105         assertThat(actual.getAllConstraints().size(), is(2));
106     }
107     
108     private ShardingSphereTable createShardingSphereTable() {
109         return new ShardingSphereTable("foo_tbl",
110                 Collections.singleton(new ShardingSphereColumn("foo_col", 0, true, false, false, true, false, false)),
111                 Collections.singleton(new ShardingSphereIndex("PRIMARY", Collections.emptyList(), false)),
112                 Collections.singleton(new ShardingSphereConstraint("foo_constraint", "foo_tbl")), null);
113     }
114     
115     private Map<String, YamlShardingSphereColumn> createYamlColumns() {
116         Map<String, YamlShardingSphereColumn> result = new LinkedHashMap<>(2, 1F);
117         result.put("Foo_Col", createYamlColumn("Foo_Col", 0, true));
118         result.put("foo_col", createYamlColumn("foo_col", 1, false));
119         return result;
120     }
121     
122     private YamlShardingSphereColumn createYamlColumn(final String name, final int dataType, final boolean primaryKey) {
123         YamlShardingSphereColumn result = new YamlShardingSphereColumn();
124         result.setName(name);
125         result.setDataType(dataType);
126         result.setPrimaryKey(primaryKey);
127         result.setGenerated(false);
128         result.setCaseSensitive(false);
129         result.setVisible(true);
130         result.setUnsigned(false);
131         result.setNullable(false);
132         return result;
133     }
134     
135     private Map<String, YamlShardingSphereIndex> createYamlIndexes() {
136         Map<String, YamlShardingSphereIndex> result = new LinkedHashMap<>(2, 1F);
137         result.put("Foo_Idx", createYamlIndex("Foo_Idx"));
138         result.put("foo_idx", createYamlIndex("foo_idx"));
139         return result;
140     }
141     
142     private YamlShardingSphereIndex createYamlIndex(final String name) {
143         YamlShardingSphereIndex result = new YamlShardingSphereIndex();
144         result.setName(name);
145         result.setColumns(Collections.emptyList());
146         result.setUnique(false);
147         return result;
148     }
149     
150     private Map<String, YamlShardingSphereConstraint> createYamlConstraints() {
151         Map<String, YamlShardingSphereConstraint> result = new LinkedHashMap<>(2, 1F);
152         result.put("Foo_Constraint", createYamlConstraint("Foo_Constraint"));
153         result.put("foo_constraint", createYamlConstraint("foo_constraint"));
154         return result;
155     }
156     
157     private YamlShardingSphereConstraint createYamlConstraint(final String name) {
158         YamlShardingSphereConstraint result = new YamlShardingSphereConstraint();
159         result.setName(name);
160         result.setReferencedTableName("foo_tbl");
161         return result;
162     }
163 }
