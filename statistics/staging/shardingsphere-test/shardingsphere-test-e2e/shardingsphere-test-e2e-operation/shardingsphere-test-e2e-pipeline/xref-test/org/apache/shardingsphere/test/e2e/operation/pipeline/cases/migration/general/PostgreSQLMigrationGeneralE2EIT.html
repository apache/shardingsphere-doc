53  @PipelineE2ESettings(database = {
54          @PipelineE2EDatabaseSettings(type = "PostgreSQL"),
55          @PipelineE2EDatabaseSettings(type = "openGauss")})
56  @Slf4j
57  class PostgreSQLMigrationGeneralE2EIT extends AbstractMigrationE2EIT {
58      
59      private static final String SOURCE_TABLE_NAME = "t_order";
60      
61      private static final long CROSS_SCHEMA_SENTINEL_ORDER_ID = 20000L;
62      
63      @ParameterizedTest(name = "{0}")
64      @EnabledIf("isEnabled")
65      @ArgumentsSource(PipelineE2ETestCaseArgumentsProvider.class)
66      void assertMigrationSuccess(final PipelineTestParameter testParam) throws SQLException, InterruptedException {
67          try (PipelineContainerComposer containerComposer = new PipelineContainerComposer(testParam)) {
68              PipelineE2EDistSQLFacade distSQLFacade = new PipelineE2EDistSQLFacade(containerComposer, new MigrationJobType());
69              distSQLFacade.alterPipelineRule();
70              createSourceSchema(containerComposer, PipelineContainerComposer.SCHEMA_NAME);
71              IntPkLargeOrderDAO orderDAO = new IntPkLargeOrderDAO(containerComposer.getSourceDataSource(),
72                      containerComposer.getDatabaseType(), containerComposer.createQualifiedTableWithSchema(SOURCE_TABLE_NAME).format());
73              orderDAO.createTable();
74              IntPkLargeOrderDAO publicOrderDAO = new IntPkLargeOrderDAO(containerComposer.getSourceDataSource(), containerComposer.getDatabaseType(), "public." + SOURCE_TABLE_NAME);
75              publicOrderDAO.createTable();
76              IntPkOrderItemDAO orderItemDAO = new IntPkOrderItemDAO(containerComposer.getSourceDataSource(), containerComposer.getDatabaseType(), PipelineContainerComposer.SCHEMA_NAME);
77              orderItemDAO.createTable();
78              containerComposer.createSourceTableIndexList(PipelineContainerComposer.SCHEMA_NAME, SOURCE_TABLE_NAME);
79              containerComposer.createSourceCommentOnList(PipelineContainerComposer.SCHEMA_NAME, SOURCE_TABLE_NAME);
80              addMigrationSourceResource(containerComposer);
81              addMigrationTargetResource(containerComposer);
82              createTargetOrderTableRule(containerComposer);
83              createTargetOrderItemTableRule(containerComposer);
84              log.info("init data begin: {}", LocalDateTime.now());
85              orderDAO.batchInsert(PipelineContainerComposer.TABLE_INIT_ROW_COUNT);
86              orderItemDAO.batchInsert(PipelineContainerComposer.TABLE_INIT_ROW_COUNT);
87              int replicationSlotsCount = getReplicationSlotsCount(containerComposer);
88              log.info("init data end: {}, replication slots count: {}", LocalDateTime.now(), replicationSlotsCount);
89              startMigrationWithSchema(containerComposer, SOURCE_TABLE_NAME, "t_order");
90              Awaitility.waitAtMost(10L, TimeUnit.SECONDS).pollInterval(1L, TimeUnit.SECONDS).until(() -> !distSQLFacade.listJobIds().isEmpty());
91              String jobId = distSQLFacade.getJobIdByTableName("ds_0.test." + SOURCE_TABLE_NAME);
92              distSQLFacade.waitJobPreparingStageFinished(jobId);
93              distSQLFacade.waitJobIncrementalStageStarted(jobId);
94              publicOrderDAO.insert(CROSS_SCHEMA_SENTINEL_ORDER_ID, 1, "OK");
95              String qualifiedTableName = String.join(".", PipelineContainerComposer.SCHEMA_NAME, SOURCE_TABLE_NAME);
96              containerComposer.startIncrementTask(new E2EIncrementalTask(containerComposer.getSourceDataSource(), qualifiedTableName, new SnowflakeKeyGenerateAlgorithm(),
97                      containerComposer.getDatabaseType(), 20));
98              TimeUnit.SECONDS.timedJoin(containerComposer.getIncreaseTaskThread(), 30L);
99              orderDAO.insert(10000L, 1, "OK");
100             // TODO Insert new record in t_order_item
101             DataSource jdbcDataSource = containerComposer.generateShardingSphereDataSourceFromProxy();
102             containerComposer.assertRecordExists(jdbcDataSource, qualifiedTableName, 10000);
103             assertRecordAbsent(jdbcDataSource, qualifiedTableName, CROSS_SCHEMA_SENTINEL_ORDER_ID);
104             checkOrderMigration(distSQLFacade, jobId);
105             startMigrationWithSchema(containerComposer, "t_order_item", "t_order_item");
106             checkOrderItemMigration(distSQLFacade);
107             for (String each : distSQLFacade.listJobIds()) {
108                 distSQLFacade.commit(each);
109             }
110             assertTrue(distSQLFacade.listJobIds().isEmpty());
111             containerComposer.assertGreaterThanOrderTableInitRows(jdbcDataSource, PipelineContainerComposer.TABLE_INIT_ROW_COUNT + 1, PipelineContainerComposer.SCHEMA_NAME);
112             assertThat("Replication slots count doesn't match, it might be not cleaned, run `SELECT * FROM pg_replication_slots;` in PostgreSQL to verify",
113                     getReplicationSlotsCount(containerComposer), is(replicationSlotsCount));
114         }
115     }
116     
117     private void assertRecordAbsent(final DataSource dataSource, final String tableName, final long orderId) throws SQLException {
118         try (
119                 Connection connection = dataSource.getConnection();
120                 Statement statement = connection.createStatement();
121                 ResultSet resultSet = statement.executeQuery(String.format("SELECT 1 FROM %s WHERE order_id = %d", tableName, orderId))) {
122             assertFalse(resultSet.next(), "Cross-schema sentinel must not be migrated");
123         }
124     }
125     
126     private void checkOrderMigration(final PipelineE2EDistSQLFacade distSQLFacade, final String jobId) throws SQLException {
127         distSQLFacade.waitJobIncrementalStageFinished(jobId);
128         distSQLFacade.pauseJob(jobId);
129         distSQLFacade.resumeJob(jobId);
130         distSQLFacade.startCheckAndVerify(jobId, "DATA_MATCH");
131     }
132     
133     private void checkOrderItemMigration(final PipelineE2EDistSQLFacade distSQLFacade) throws SQLException {
134         String jobId = distSQLFacade.getJobIdByTableName("ds_0.test.t_order_item");
135         distSQLFacade.waitJobIncrementalStageStarted(jobId);
136         distSQLFacade.startCheckAndVerify(jobId, "DATA_MATCH");
137     }
138     
139     private int getReplicationSlotsCount(final PipelineContainerComposer containerComposer) throws SQLException {
140         try (
141                 Connection connection = containerComposer.getSourceDataSource().getConnection();
142                 Statement statement = connection.createStatement();
143                 ResultSet resultSet = statement.executeQuery("SELECT COUNT(1) FROM pg_replication_slots")) {
144             if (!resultSet.next()) {
145                 return 0;
146             }
147             return resultSet.getInt(1);
148         }
149     }
150     
151     private static boolean isEnabled(final ExtensionContext context) {
152         return PipelineE2ECondition.isEnabled(context);
153     }
