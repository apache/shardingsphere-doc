38  /**
39   * Seek blob command executor for Firebird.
40   *
41   * <p>Firebird defines seek for stream BLOBs only and returns the resulting position in the BLOB id of the response.
42   * A position outside the BLOB is not an error, it is limited to the BLOB content.</p>
43   *
44   * @see <a href="https://github.com/FirebirdSQL/firebird/blob/v5.0.3/src/jrd/blb.cpp">Firebird blb::BLB_lseek</a>
45   */
46  @RequiredArgsConstructor
47  public final class FirebirdSeekBlobCommandExecutor implements CommandExecutor {
48      
49      private final FirebirdSeekBlobCommandPacket packet;
50      
51      private final ConnectionSession connectionSession;
52      
53      @Override
54      @HighFrequencyInvocation
55      public Collection<DatabasePacket> execute() {
56          int connectionId = connectionSession.getConnectionId();
57          int blobHandle = FirebirdBlobHandleGenerator.getInstance().resolveBlobHandle(connectionId, packet.getBlobHandle());
58          Optional<Boolean> streamBlob = FirebirdBlobReadCache.getInstance().isStreamBlob(connectionId, blobHandle);
59          ShardingSpherePreconditions.checkState(streamBlob.isPresent(), () -> new InvalidSegstrHandleException(blobHandle));
60          ShardingSpherePreconditions.checkState(streamBlob.get(), () -> new InvalidSegstrTypeException(blobHandle));
61          OptionalInt position = FirebirdBlobReadCache.getInstance().seek(connectionId, blobHandle, packet.getSeekMode(), packet.getOffset());
62          return Collections.singleton(new FirebirdGenericResponsePacket().setId(position.getAsInt()));
63      }
64  }
