46      private final WorkflowValidationSupport validationSupport = new WorkflowValidationSupport();

48      private final ShardingInspectionService inspectionService = new ShardingInspectionService();

50      @Override
51      public ValidationReport validate(final WorkflowContextSnapshot snapshot, final MCPFeatureQueryFacade queryFacade) {
52          ValidationReport result = new ValidationReport();
53          result.setRuleValidation(validateByWorkflowKind(snapshot, queryFacade, result));
54          result.setOverallStatus(validationSupport.resolveOverallStatus(result.getRuleValidation()));
55          return result;
56      }
57      
58      private String normalizeStrategyType(final ShardingWorkflowRequest request) {
59          return request.getStrategyType().isEmpty() ? "standard" : request.getStrategyType().toLowerCase(Locale.ENGLISH);
60      }
61      
62      private ValidationSection validateByWorkflowKind(final WorkflowContextSnapshot snapshot, final MCPFeatureQueryFacade queryFacade, final ValidationReport validationReport) {
63          ShardingWorkflowRequest request = (ShardingWorkflowRequest) snapshot.getRequest();
64          queryFacade.checkDatabaseCapability(request.getDatabase());
65          if (ShardingFeatureDefinition.TABLE_RULE_WORKFLOW_KIND.equals(snapshot.getWorkflowKind())) {
66              List<Map<String, Object>> rows = inspectionService.queryTableRule(queryFacade, request.getDatabase(), request.getTable());
67              return validateNamedState(snapshot, validationReport, rows, queryFacade, "table", request.getTable(), matchesTableRule(rows, queryFacade, request));
68          }
69          if (ShardingFeatureDefinition.TABLE_REFERENCE_WORKFLOW_KIND.equals(snapshot.getWorkflowKind())) {
70              List<Map<String, Object>> rows = inspectionService.queryTableReferenceRule(queryFacade, request.getDatabase(), request.getRuleName());
71              return validateNamedState(snapshot, validationReport, rows, queryFacade, "name", request.getRuleName(), matchesTableReferenceRule(rows, queryFacade, request));
72          }
73          if (ShardingFeatureDefinition.KEY_GENERATOR_WORKFLOW_KIND.equals(snapshot.getWorkflowKind())) {
74              List<Map<String, Object>> rows = inspectionService.queryKeyGenerator(queryFacade, request.getDatabase(), request.getKeyGeneratorName());
75              return validateNamedState(snapshot, validationReport, rows, queryFacade, "name", request.getKeyGeneratorName(), matchesKeyGenerator(rows, queryFacade, request));
76          }
77          if (ShardingFeatureDefinition.KEY_GENERATE_STRATEGY_WORKFLOW_KIND.equals(snapshot.getWorkflowKind())) {
78              List<Map<String, Object>> rows = inspectionService.queryKeyGenerateStrategy(queryFacade, request.getDatabase(), request.getKeyGenerateStrategyName());
79              return validateNamedState(snapshot, validationReport, rows, queryFacade, "name", request.getKeyGenerateStrategyName(), matchesKeyGenerateStrategy(rows, queryFacade, request));
80          }
81          if (ShardingFeatureDefinition.DEFAULT_STRATEGY_WORKFLOW_KIND.equals(snapshot.getWorkflowKind())) {
82              return validateDefaultStrategy(snapshot, validationReport, inspectionService.queryDefaultStrategy(queryFacade, request.getDatabase()), queryFacade, request);
83          }
84          return validateCleanup(validationReport, queryFacade, request);
85      }
86      
87      private ValidationSection validateNamedState(final WorkflowContextSnapshot snapshot, final ValidationReport validationReport, final List<Map<String, Object>> rows,
88                                                   final MCPFeatureQueryFacade queryFacade, final String fieldName, final String expected, final boolean shapeMatches) {
89          boolean ruleExists = containsNamedRow(rows, queryFacade, snapshot.getRequest().getDatabase(), fieldName, expected);
90          boolean dropWorkflow = WorkflowLifecycleUtils.isDropWorkflow(snapshot);
91          boolean expectedRuleExists = !dropWorkflow;
92          if (expectedRuleExists != ruleExists) {
93              addMismatch(validationReport, fieldName, expected);
94              return new ValidationSection(WorkflowLifecycle.STATUS_FAILED, rows, "Sharding rule state does not match the planned DistSQL artifact.");
95          }
96          if (!dropWorkflow && !shapeMatches) {
97              addMismatch(validationReport, fieldName, expected);
98              return new ValidationSection(WorkflowLifecycle.STATUS_FAILED, rows, "Sharding rule fields do not match the planned DistSQL artifact.");
99          }
100         return new ValidationSection(WorkflowLifecycle.STATUS_PASSED, rows, "Sharding rule state matches the planned DistSQL artifact.");
101     }

103     private boolean matchesTableRule(final List<Map<String, Object>> rows, final MCPFeatureQueryFacade queryFacade, final ShardingWorkflowRequest request) {
104         if (1 != rows.size() || !containsNamedRow(rows, queryFacade, request.getDatabase(), "table", request.getTable())) {
105             return false;
106         }
107         Map<String, Object> row = rows.getFirst();
108         boolean dataSourceShapeMatches = request.getStorageUnits().isEmpty()
109                 ? WorkflowRuleValueUtils.getRuleValue(row, "actual_data_nodes").equals(request.getDataNodes())
110                 : matchesIdentifiers(WorkflowRuleValueUtils.getRuleValue(row, "actual_data_sources"), splitValues(request.getStorageUnits()), IdentifierScope.TABLE,
111                         queryFacade, request.getDatabase());
112         return dataSourceShapeMatches && matchesTableStrategy(row, queryFacade, request) && matchesTableKeyGenerator(row, queryFacade, request)
113                 && matchesTableAuditors(row, request);
114     }
115     
116     private boolean matchesTableStrategy(final Map<String, Object> row, final MCPFeatureQueryFacade queryFacade, final ShardingWorkflowRequest request) {
117         String strategyType = request.getStorageUnits().isEmpty() ? normalizeStrategyType(request) : "standard";
118         if ("none".equals(strategyType)) {
119             return true;
120         }
121         if (!WorkflowRuleValueUtils.getRuleValue(row, "table_strategy_type").equalsIgnoreCase(strategyType)
122                 || !WorkflowRuleValueUtils.getRuleValue(row, "table_sharding_algorithm_type").equalsIgnoreCase(request.getAlgorithmType())
123                 || !WorkflowAlgorithmUtils.createPropertyMap(row.get("table_sharding_algorithm_props")).equals(request.getPrimaryAlgorithmProperties())) {
124             return false;
125         }
126         if ("hint".equals(strategyType)) {
127             return true;
128         }
129         String actualColumns = WorkflowRuleValueUtils.getRuleValue(row, "table_sharding_column");
130         return "complex".equals(strategyType)
131                 ? matchesIdentifiers(actualColumns, splitValues(request.getShardingColumns()), IdentifierScope.COLUMN, queryFacade, request.getDatabase())
132                 : queryFacade.isSameIdentifier(request.getDatabase(), IdentifierScope.COLUMN, request.getColumn(), actualColumns);
133     }
134     
135     private boolean matchesTableKeyGenerator(final Map<String, Object> row, final MCPFeatureQueryFacade queryFacade, final ShardingWorkflowRequest request) {
136         if (request.getKeyGenerateColumn().isEmpty()) {
137             return true;
138         }
139         if (!queryFacade.isSameIdentifier(request.getDatabase(), IdentifierScope.COLUMN, request.getKeyGenerateColumn(),
140                 WorkflowRuleValueUtils.getRuleValue(row, "key_generate_column"))) {
141             return false;
142         }
143         if (!request.getKeyGeneratorName().isEmpty()) {
144             return containsNamedRow(inspectionService.queryTableRulesUsedKeyGenerator(queryFacade, request.getDatabase(), request.getKeyGeneratorName()),
145                     queryFacade, request.getDatabase(), "name", request.getTable());
146         }
147         return WorkflowRuleValueUtils.getRuleValue(row, "key_generator_type").equalsIgnoreCase(request.getKeyGeneratorType())
148                 && WorkflowAlgorithmUtils.createPropertyMap(row.get("key_generator_props")).equals(request.getKeyGeneratorProperties());
149     }
150     
151     private boolean matchesTableAuditors(final Map<String, Object> row, final ShardingWorkflowRequest request) {
152         if (request.getAuditorNames().isEmpty()) {
153             return true;
154         }
155         String allowHintDisable = request.getAllowHintDisable().isEmpty() ? "false" : request.getAllowHintDisable();
156         return matchesValues(WorkflowRuleValueUtils.getRuleValue(row, "auditor_types"), request.getAuditorNames())
157                 && WorkflowRuleValueUtils.getRuleValue(row, "allow_hint_disable").equalsIgnoreCase(allowHintDisable);
158     }
159     
160     private boolean matchesTableReferenceRule(final List<Map<String, Object>> rows, final MCPFeatureQueryFacade queryFacade, final ShardingWorkflowRequest request) {
161         return 1 == rows.size() && containsNamedRow(rows, queryFacade, request.getDatabase(), "name", request.getRuleName())
162                 && matchesIdentifiers(WorkflowRuleValueUtils.getRuleValue(rows.getFirst(), "sharding_table_reference"), request.getReferenceTables(), IdentifierScope.TABLE,
163                         queryFacade, request.getDatabase());
164     }
165     
166     private boolean matchesKeyGenerator(final List<Map<String, Object>> rows, final MCPFeatureQueryFacade queryFacade, final ShardingWorkflowRequest request) {
167         return 1 == rows.size() && containsNamedRow(rows, queryFacade, request.getDatabase(), "name", request.getKeyGeneratorName())
168                 && WorkflowRuleValueUtils.getRuleValue(rows.getFirst(), "type").equalsIgnoreCase(request.getKeyGeneratorType())
169                 && WorkflowAlgorithmUtils.createPropertyMap(rows.getFirst().get("props")).equals(request.getKeyGeneratorProperties());
170     }
171     
172     private boolean matchesKeyGenerateStrategy(final List<Map<String, Object>> rows, final MCPFeatureQueryFacade queryFacade, final ShardingWorkflowRequest request) {
173         if (1 != rows.size() || !containsNamedRow(rows, queryFacade, request.getDatabase(), "name", request.getKeyGenerateStrategyName())) {
174             return false;
175         }
176         Map<String, Object> row = rows.getFirst();
177         boolean sequenceStrategy = !request.getSequenceName().isEmpty();
178         if (!WorkflowRuleValueUtils.getRuleValue(row, "type").equalsIgnoreCase(sequenceStrategy ? "sequence" : "column")) {
179             return false;
180         }
181         boolean targetMatches = sequenceStrategy
182                 ? request.getSequenceName().equals(WorkflowRuleValueUtils.getRuleValue(row, "sequence"))
183                 : queryFacade.isSameIdentifier(request.getDatabase(), IdentifierScope.TABLE, request.getTable(), WorkflowRuleValueUtils.getRuleValue(row, "table"))
184                         && queryFacade.isSameIdentifier(request.getDatabase(), IdentifierScope.COLUMN, request.getColumn(), WorkflowRuleValueUtils.getRuleValue(row, "column"));
185         if (!targetMatches) {
186             return false;
187         }
188         return request.getKeyGeneratorName().isEmpty()
189                 ? WorkflowRuleValueUtils.getRuleValue(row, "generator_type").equalsIgnoreCase(request.getKeyGeneratorType())
190                         && WorkflowAlgorithmUtils.createPropertyMap(row.get("generator_props")).equals(request.getKeyGeneratorProperties())
191                 : queryFacade.isSameIdentifier(request.getDatabase(), IdentifierScope.TABLE, request.getKeyGeneratorName(),
192                         WorkflowRuleValueUtils.getRuleValue(row, "generator_name"));
193     }

195     private ValidationSection validateDefaultStrategy(final WorkflowContextSnapshot snapshot, final ValidationReport validationReport,
196                                                       final List<Map<String, Object>> rows, final MCPFeatureQueryFacade queryFacade, final ShardingWorkflowRequest request) {
197         boolean strategyExists = rows.stream().anyMatch(each -> queryFacade.isSameIdentifier(request.getDatabase(), IdentifierScope.TABLE, request.getDefaultStrategyType(),
198                 WorkflowRuleValueUtils.getRuleValue(each, "name")) && !WorkflowRuleValueUtils.getRuleValue(each, "type").isEmpty());
199         boolean dropWorkflow = WorkflowLifecycleUtils.isDropWorkflow(snapshot);
200         boolean expectedStrategyExists = !dropWorkflow;
201         if (expectedStrategyExists != strategyExists) {
202             addMismatch(validationReport, "name", request.getDefaultStrategyType());
203             return new ValidationSection(WorkflowLifecycle.STATUS_FAILED, rows, "Default sharding strategy state does not match the planned DistSQL artifact.");
204         }
205         if (!dropWorkflow && !matchesDefaultStrategy(rows, queryFacade, request)) {
206             addMismatch(validationReport, "name", request.getDefaultStrategyType());
207             return new ValidationSection(WorkflowLifecycle.STATUS_FAILED, rows, "Default sharding strategy fields do not match the planned DistSQL artifact.");
208         }
209         return new ValidationSection(WorkflowLifecycle.STATUS_PASSED, rows, "Default sharding strategy state matches the planned DistSQL artifact.");
210     }
211     
212     private boolean matchesDefaultStrategy(final List<Map<String, Object>> rows, final MCPFeatureQueryFacade queryFacade, final ShardingWorkflowRequest request) {
213         List<Map<String, Object>> matchingRows = rows.stream().filter(each -> queryFacade.isSameIdentifier(request.getDatabase(), IdentifierScope.TABLE,
214                 request.getDefaultStrategyType(), WorkflowRuleValueUtils.getRuleValue(each, "name"))).toList();
215         if (1 != matchingRows.size()) {
216             return false;
217         }
218         Map<String, Object> row = matchingRows.getFirst();
219         String strategyType = normalizeStrategyType(request);
220         if (!WorkflowRuleValueUtils.getRuleValue(row, "type").equalsIgnoreCase(strategyType)) {
221             return false;
222         }
223         if ("none".equals(strategyType)) {
224             return true;
225         }
226         if (!"hint".equals(strategyType)) {
227             String actualColumns = WorkflowRuleValueUtils.getRuleValue(row, "sharding_column");
228             boolean columnsMatch = "complex".equals(strategyType)
229                     ? matchesIdentifiers(actualColumns, splitValues(request.getShardingColumns()), IdentifierScope.COLUMN, queryFacade, request.getDatabase())
230                     : queryFacade.isSameIdentifier(request.getDatabase(), IdentifierScope.COLUMN, request.getColumn(), actualColumns);
231             if (!columnsMatch) {
232                 return false;
233             }
234         }
235         return WorkflowRuleValueUtils.getRuleValue(row, "sharding_algorithm_type").equalsIgnoreCase(request.getAlgorithmType())
236                 && WorkflowAlgorithmUtils.createPropertyMap(row.get("sharding_algorithm_props")).equals(request.getPrimaryAlgorithmProperties());
237     }
238     
239     private ValidationSection validateCleanup(final ValidationReport validationReport, final MCPFeatureQueryFacade queryFacade, final ShardingWorkflowRequest request) {
240         List<Map<String, Object>> rows = queryCleanupRows(queryFacade, request);
241         if (containsNamedRow(rows, queryFacade, request.getDatabase(), "name", request.getComponentName())) {
242             addMismatch(validationReport, "name", request.getComponentName());
243             return new ValidationSection(WorkflowLifecycle.STATUS_FAILED, rows, "Sharding component still exists after cleanup.");
244         }
245         return new ValidationSection(WorkflowLifecycle.STATUS_PASSED, rows, "Sharding component cleanup state matches the planned DistSQL artifact.");
246     }
247     
248     private List<Map<String, Object>> queryCleanupRows(final MCPFeatureQueryFacade queryFacade, final ShardingWorkflowRequest request) {
249         switch (request.getComponentType().toLowerCase(Locale.ENGLISH).replace('_', '-')) {
250             case "algorithm":
251                 return inspectionService.queryAlgorithms(queryFacade, request.getDatabase());
252             case "key-generator":
253                 return inspectionService.queryKeyGenerators(queryFacade, request.getDatabase());
254             case "auditor":
255                 return inspectionService.queryAuditors(queryFacade, request.getDatabase());
256             default:
257                 return List.of(Map.of("unsupported_component_type", request.getComponentType()));
258         }
259     }
260     
261     private boolean containsNamedRow(final List<Map<String, Object>> rows, final MCPFeatureQueryFacade queryFacade, final String databaseName,
262                                      final String fieldName, final String expected) {
263         return rows.stream().anyMatch(each -> queryFacade.isSameIdentifier(databaseName, IdentifierScope.TABLE, expected, WorkflowRuleValueUtils.getRuleValue(each, fieldName)));
264     }
265     
266     private boolean matchesIdentifiers(final String actualValues, final Collection<String> expectedValues, final IdentifierScope identifierScope,
267                                        final MCPFeatureQueryFacade queryFacade, final String databaseName) {
268         List<String> actual = splitValues(actualValues);
269         return actual.size() == expectedValues.size() && expectedValues.stream().allMatch(expected -> actual.stream()
270                 .anyMatch(each -> queryFacade.isSameIdentifier(databaseName, identifierScope, expected, each)));
271     }
272     
273     private boolean matchesValues(final String actualValues, final Collection<String> expectedValues) {
274         List<String> actual = splitValues(actualValues);
275         return actual.size() == expectedValues.size() && expectedValues.stream().allMatch(expected -> actual.stream().anyMatch(expected::equalsIgnoreCase));
276     }
277     
278     private List<String> splitValues(final String values) {
279         return Arrays.stream(values.split(",")).map(String::trim).filter(each -> !each.isEmpty()).toList();
280     }
281     
282     private void addMismatch(final ValidationReport validationReport, final String field, final String expected) {
283         validationReport.getMismatches().add(validationSupport.createMismatch(WorkflowIssueCode.RULE_STATE_MISMATCH, field, expected, "",
284                 "Sharding DistSQL-visible state differs from the planned artifact.", "Inspect current sharding state and re-apply or re-plan the workflow."));
285     }
