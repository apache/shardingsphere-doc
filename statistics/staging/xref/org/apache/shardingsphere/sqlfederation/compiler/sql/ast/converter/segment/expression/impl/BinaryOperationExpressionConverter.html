44  /**
45   * Binary operation expression converter.
46   */
47  @NoArgsConstructor(access = AccessLevel.PRIVATE)
48  public final class BinaryOperationExpressionConverter {
49      
50      private static final Map<String, SqlOperator> REGISTRY = new CaseInsensitiveMap<>();
51      
52      static {
53          register();
54          registerAlias();
55      }
56      
57      private static void register() {
58          register(SqlStdOperatorTable.EQUALS);
59          register(SqlStdOperatorTable.NOT_EQUALS);
60          register(SqlStdOperatorTable.GREATER_THAN);
61          register(SqlStdOperatorTable.GREATER_THAN_OR_EQUAL);
62          register(SqlStdOperatorTable.LESS_THAN);
63          register(SqlStdOperatorTable.LESS_THAN_OR_EQUAL);
64          register(SqlStdOperatorTable.OR);
65          register(SqlStdOperatorTable.AND);
66          register(SqlStdOperatorTable.PLUS);
67          register(SqlStdOperatorTable.MINUS);
68          register(SqlStdOperatorTable.MULTIPLY);
69          register(SqlStdOperatorTable.DIVIDE);
70          register(SqlStdOperatorTable.LIKE);
71          register(SqlStdOperatorTable.NOT_LIKE);
72          register(SqlStdOperatorTable.PERCENT_REMAINDER);
73          register(SqlStdOperatorTable.IS_NULL);
74          register(SqlStdOperatorTable.IS_NOT_NULL);
75          register(SqlStdOperatorTable.ALL_GT);
76          register(SqlStdOperatorTable.IS_FALSE);
77          register(SqlStdOperatorTable.IS_NOT_FALSE);
78          register(SqlStdOperatorTable.IS_TRUE);
79          register(SqlStdOperatorTable.IS_NOT_TRUE);
80          register(SqlStdOperatorTable.CONCAT);
81          register(SqlStdOperatorTable.PATTERN_ALTER);
82          register(SqlStdOperatorTable.MOD);
83          SqlStdOperatorTable.QUANTIFY_OPERATORS.forEach(BinaryOperationExpressionConverter::register);
84          register(SQLExtensionOperatorTable.DIV);
85          register(SQLExtensionOperatorTable.CARET);
86          register(SQLExtensionOperatorTable.AMPERSAND);
87          register(SQLExtensionOperatorTable.SIGNED_RIGHT_SHIFT);
88          register(SQLExtensionOperatorTable.SIGNED_LEFT_SHIFT);
89          register(SQLExtensionOperatorTable.XOR);
90          register(SQLExtensionOperatorTable.LOGICAL_AND);
91          register(SQLExtensionOperatorTable.REGEXP);
92          register(SQLExtensionOperatorTable.NOT_REGEXP);
93          register(SQLExtensionOperatorTable.SOUNDS_LIKE);
94          register(SQLExtensionOperatorTable.NULL_SAFE);
95          register(SQLExtensionOperatorTable.ASSIGNMENT);
96          register(SQLExtensionOperatorTable.RLIKE);
97      }
98      
99      private static void register(final SqlOperator sqlOperator) {
100         REGISTRY.put(sqlOperator.getName(), sqlOperator);
101     }
102     
103     private static void registerAlias() {
104         REGISTRY.put("!=", SqlStdOperatorTable.NOT_EQUALS);
105         REGISTRY.put("~", SqlStdOperatorTable.POSIX_REGEX_CASE_SENSITIVE);
106         REGISTRY.put("~*", SqlStdOperatorTable.NEGATED_POSIX_REGEX_CASE_SENSITIVE);
107         REGISTRY.put("!~", SqlStdOperatorTable.NEGATED_POSIX_REGEX_CASE_SENSITIVE);
108         REGISTRY.put("!~*", SqlStdOperatorTable.NEGATED_POSIX_REGEX_CASE_INSENSITIVE);
109     }
110     
111     /**
112      * Convert binary operation expression to SQL node.
113      *
114      * @param segment binary operation expression
115      * @return SQL node
116      */
117     public static SqlBasicCall convert(final BinaryOperationExpression segment) {
118         SqlOperator operator = convertOperator(segment);
119         List<SqlNode> sqlNodes = convertSqlNodes(segment, operator);
120         return new SqlBasicCall(operator, sqlNodes, SqlParserPos.ZERO);
121     }
122     
123     private static SqlOperator convertOperator(final BinaryOperationExpression segment) {
124         String operator = getOperator(segment);
125         Preconditions.checkState(REGISTRY.containsKey(operator), "Unsupported SQL operator: `%s`", operator);
126         return REGISTRY.get(operator);
127     }
128     
129     private static String getOperator(final BinaryOperationExpression segment) {
130         String result = null;
131         if ("IS".equalsIgnoreCase(segment.getOperator())) {
132             result = findIsOperator(segment).orElse(null);
133         }
134         if (segment.getRight() instanceof QuantifySubqueryExpression) {
135             result = findQuantifyOperator(segment.getOperator(), (QuantifySubqueryExpression) segment.getRight()).orElse(null);
136         }
137         return null == result ? segment.getOperator() : result;
138     }
139     
140     private static Optional<String> findIsOperator(final BinaryOperationExpression segment) {
141         String literals = String.valueOf(((LiteralExpressionSegment) segment.getRight()).getLiterals());
142         if ("NULL".equalsIgnoreCase(literals)) {
143             return Optional.of(SqlStdOperatorTable.IS_NULL.getName());
144         } else if ("NOT NULL".equalsIgnoreCase(literals)) {
145             return Optional.of(SqlStdOperatorTable.IS_NOT_NULL.getName());
146         } else if ("FALSE".equalsIgnoreCase(literals)) {
147             return Optional.of(SqlStdOperatorTable.IS_FALSE.getName());
148         } else if ("NOT FALSE".equalsIgnoreCase(literals)) {
149             return Optional.of(SqlStdOperatorTable.IS_NOT_FALSE.getName());
150         } else if ("TRUE".equalsIgnoreCase(literals)) {
151             return Optional.of(SqlStdOperatorTable.IS_TRUE.getName());
152         } else if ("NOT TRUE".equalsIgnoreCase(literals)) {
153             return Optional.of(SqlStdOperatorTable.IS_NOT_TRUE.getName());
154         }
155         return Optional.empty();
156     }
157     
158     private static Optional<String> findQuantifyOperator(final String operator, final QuantifySubqueryExpression quantifySubquery) {
159         if ("=".equals(operator)) {
160             return "ALL".equalsIgnoreCase(quantifySubquery.getQuantifyOperator()) ? Optional.of(SqlStdOperatorTable.ALL_EQ.getName()) : Optional.of(SqlStdOperatorTable.SOME_EQ.getName());
161         } else if (">".equals(operator)) {
162             return "ALL".equalsIgnoreCase(quantifySubquery.getQuantifyOperator()) ? Optional.of(SqlStdOperatorTable.ALL_GT.getName()) : Optional.of(SqlStdOperatorTable.SOME_GT.getName());
163         } else if (">=".equals(operator)) {
164             return "ALL".equalsIgnoreCase(quantifySubquery.getQuantifyOperator()) ? Optional.of(SqlStdOperatorTable.ALL_GE.getName()) : Optional.of(SqlStdOperatorTable.SOME_GE.getName());
165         } else if ("<".equals(operator)) {
166             return "ALL".equalsIgnoreCase(quantifySubquery.getQuantifyOperator()) ? Optional.of(SqlStdOperatorTable.ALL_LT.getName()) : Optional.of(SqlStdOperatorTable.SOME_LT.getName());
167         } else if ("<=".equals(operator)) {
168             return "ALL".equalsIgnoreCase(quantifySubquery.getQuantifyOperator()) ? Optional.of(SqlStdOperatorTable.ALL_LE.getName()) : Optional.of(SqlStdOperatorTable.SOME_LE.getName());
169         } else if ("!=".equals(operator) || "<>".equals(operator)) {
170             return "ALL".equalsIgnoreCase(quantifySubquery.getQuantifyOperator()) ? Optional.of(SqlStdOperatorTable.ALL_NE.getName()) : Optional.of(SqlStdOperatorTable.SOME_NE.getName());
171         }
172         return Optional.empty();
173     }
174     
175     private static List<SqlNode> convertSqlNodes(final BinaryOperationExpression segment, final SqlOperator operator) {
176         SqlNode left = ExpressionConverter.convert(segment.getLeft()).orElseThrow(IllegalStateException::new);
177         List<SqlNode> result = new LinkedList<>();
178         result.add(left);
179         if (SqlStdOperatorTable.IS_FALSE.equals(operator) || SqlStdOperatorTable.IS_NOT_FALSE.equals(operator) || SqlStdOperatorTable.IS_TRUE.equals(operator)
180                 || SqlStdOperatorTable.IS_NOT_TRUE.equals(operator)) {
181             if (left instanceof SqlNumericLiteral) {
182                 result.remove(0);
183                 Long value = ((SqlNumericLiteral) left).getValueAs(Long.class);
184                 result.add(value == null || value == 0L ? SqlLiteral.createBoolean(false, left.getParserPosition()) : SqlLiteral.createBoolean(true, left.getParserPosition()));
185             }
186         } else if (!SqlStdOperatorTable.IS_NULL.equals(operator) && !SqlStdOperatorTable.IS_NOT_NULL.equals(operator)) {
187             SqlNode right = ExpressionConverter.convert(segment.getRight()).orElseThrow(IllegalStateException::new);
188             result.addAll(right instanceof SqlNodeList ? ((SqlNodeList) right).getList() : Collections.singletonList(right));
189         }
190         return result;
191     }
