70      @Test
71      void assertBind() {
72          SimpleTableSegment simpleTableSegment = new SimpleTableSegment(new TableNameSegment(0, 0, new IdentifierValue("t_order")));
73          ErrorLoggingSegment errorLogging = new ErrorLoggingSegment(0, 0, null, null, "UNLIMITED");
74          ReturningSegment returning = new ReturningSegment(0, 0, new ProjectionsSegment(0, 0));
75          UpdateStatement updateStatement = UpdateStatement.builder()
76                  .databaseType(databaseType)
77                  .table(simpleTableSegment)
78                  .where(new WhereSegment(0, 0, new BinaryOperationExpression(0, 0, new ColumnSegment(0, 0, new IdentifierValue("status")),
79                          new LiteralExpressionSegment(0, 0, 0), "=", "status = 1")))
80                  .errorLogging(errorLogging)
81                  .returning(returning)
82                  .build();
83          UpdateStatement actual = new UpdateStatementBinder().bind(updateStatement, new SQLStatementBinderContext(createMetaData(), "foo_db", new HintValueContext(), updateStatement));
84          assertThat(actual, not(updateStatement));
85          assertThat(actual.getErrorLogging().get(), is(errorLogging));
86          assertThat(actual.getReturning().get(), is(returning));
87          assertThat(actual.getTable(), not(updateStatement.getTable()));
88          assertThat(actual.getTable(), isA(SimpleTableSegment.class));
89          assertTrue(actual.getWhere().isPresent());
90          assertThat(actual.getWhere().get(), not(updateStatement.getWhere()));
91          assertThat(actual.getWhere().get(), isA(WhereSegment.class));
92          assertTrue(updateStatement.getWhere().isPresent());
93          assertThat(actual.getWhere().get().getExpr(), not(updateStatement.getWhere().get().getExpr()));
94          assertThat(actual.getWhere().get().getExpr(), isA(BinaryOperationExpression.class));
95          assertThat(((BinaryOperationExpression) actual.getWhere().get().getExpr()).getLeft(), isA(ColumnSegment.class));
96          assertThat(((ColumnSegment) ((BinaryOperationExpression) actual.getWhere().get().getExpr()).getLeft()).getColumnBoundInfo().getOriginalTable().getValue(), is("t_order"));
97      }

116     @Test
117     void assertBindUpdateTargetTableAlias() {
118         SimpleTableSegment targetTable = new SimpleTableSegment(new TableNameSegment(0, 0, new IdentifierValue("o")));
119         SimpleTableSegment fromTable = new SimpleTableSegment(new TableNameSegment(0, 0, new IdentifierValue("t_order")));
120         fromTable.setAlias(new AliasSegment(0, 0, new IdentifierValue("o")));
121         ColumnSegment columnSegment = new ColumnSegment(0, 0, new IdentifierValue("status"));
122         UpdateStatement updateStatement = UpdateStatement.builder()
123                 .databaseType(databaseType)
124                 .table(targetTable)
125                 .from(fromTable)
126                 .setAssignment(new SetAssignmentSegment(0, 0, Collections.singletonList(
127                         new ColumnAssignmentSegment(0, 0, Collections.singletonList(columnSegment), new LiteralExpressionSegment(0, 0, 1)))))
128                 .targetTableIsFromAlias(true)
129                 .build();
130         UpdateStatement actual = new UpdateStatementBinder().bind(updateStatement,
131                 new SQLStatementBinderContext(createMetaData(), "foo_db", new HintValueContext(), updateStatement));
132         ColumnSegment actualColumn = actual.getAssignment().get().getAssignments().iterator().next()
133                 .getColumns().iterator().next();
134         assertThat(((SimpleTableSegment) actual.getTable()).getTableName().getIdentifier().getValue(), is("t_order"));
135         assertThat(((SimpleTableSegment) actual.getTable()).getAliasName().get(), is("o"));
136         assertThat(actualColumn.getColumnBoundInfo().getOriginalTable().getValue(), is("t_order"));
137         assertTrue(actual.isTargetTableIsFromAlias());
138     }

167         fromTable.setAlias(new AliasSegment(0, 0, new IdentifierValue("o")));
168         UpdateStatement updateStatement = UpdateStatement.builder()
169                 .databaseType(databaseType)
170                 .table(targetTable)
171                 .from(fromTable)
172                 .setAssignment(new SetAssignmentSegment(0, 0, Collections.singletonList(
173                         new ColumnAssignmentSegment(0, 0, Collections.singletonList(new ColumnSegment(0, 0, new IdentifierValue("status"))), new LiteralExpressionSegment(0, 0, 1)))))

186         return new WithSegment(0, 0, new LinkedList<>(Collections.singletonList(
187                 new CommonTableExpressionSegment(0, 0, new AliasSegment(0, 0, new IdentifierValue("combined_users")),
188                         new SubquerySegment(0, 0, createWithSelectStatement(), ""), null))),
189                 false);

192     private SelectStatement createWithSelectStatement() {
193         ProjectionsSegment projections = new ProjectionsSegment(0, 0);
194         projections.getProjections().add(new ColumnProjectionSegment(new ColumnSegment(0, 0, new IdentifierValue("user_id"))));
195         return SelectStatement.builder().databaseType(databaseType).projections(projections)
196                 .from(new SimpleTableSegment(new TableNameSegment(0, 0, new IdentifierValue("t_user")))).build();
197     }

199     private ShardingSphereMetaData createMetaData() {
200         ShardingSphereSchema schema = mock(ShardingSphereSchema.class, RETURNS_DEEP_STUBS);
201         IdentifierValue fooDatabase = new IdentifierValue("foo_db");
202         IdentifierValue tOrder = new IdentifierValue("t_order");
203         IdentifierValue tUser = new IdentifierValue("t_user");
204         when(schema.getTable("t_order").getAllColumns()).thenReturn(Arrays.asList(
205                 new ShardingSphereColumn("order_id", Types.INTEGER, true, false, false, true, false, false),
206                 new ShardingSphereColumn("user_id", Types.INTEGER, false, false, false, true, false, false),
207                 new ShardingSphereColumn("status", Types.INTEGER, false, false, false, true, false, false)));
208         when(schema.getTable("t_user").getAllColumns()).thenReturn(Collections.singletonList(
209                 new ShardingSphereColumn("user_id", Types.INTEGER, false, false, false, true, false, false)));
210         when(schema.getTable(tOrder).getAllColumns()).thenReturn(Arrays.asList(
211                 new ShardingSphereColumn("order_id", Types.INTEGER, true, false, false, true, false, false),
212                 new ShardingSphereColumn("user_id", Types.INTEGER, false, false, false, true, false, false),
213                 new ShardingSphereColumn("status", Types.INTEGER, false, false, false, true, false, false)));
214         when(schema.getTable(tUser).getAllColumns()).thenReturn(Collections.singletonList(
215                 new ShardingSphereColumn("user_id", Types.INTEGER, false, false, false, true, false, false)));
216         ShardingSphereMetaData result = mock(ShardingSphereMetaData.class, RETURNS_DEEP_STUBS);
217         when(result.getDatabase("foo_db").getSchema("foo_db")).thenReturn(schema);
218         when(result.getDatabase(fooDatabase).getSchema(fooDatabase)).thenReturn(schema);
219         when(result.containsDatabase("foo_db")).thenReturn(true);
220         when(result.containsDatabase(fooDatabase)).thenReturn(true);
221         when(result.getDatabase("foo_db").getDefaultSchemaName()).thenReturn("foo_db");
222         when(result.getDatabase(fooDatabase).getDefaultSchemaName()).thenReturn("foo_db");
223         when(result.getDatabase("foo_db").containsSchema("foo_db")).thenReturn(true);
224         when(result.getDatabase(fooDatabase).containsSchema(fooDatabase)).thenReturn(true);
225         when(result.getDatabase("foo_db").getSchema("foo_db").containsTable("t_order")).thenReturn(true);
226         when(result.getDatabase("foo_db").getSchema("foo_db").containsTable("t_user")).thenReturn(true);
227         when(result.getDatabase(fooDatabase).getSchema(fooDatabase).containsTable(tOrder)).thenReturn(true);
228         when(result.getDatabase(fooDatabase).getSchema(fooDatabase).containsTable(tUser)).thenReturn(true);
229         return result;
230     }
