104             assertThat(((Map<?, ?>) actualPayload.get("search_context")).get("object_types"), is(List.of("index")));

121     @Test
122     void assertHandleSearchMetadataWithStorageUnit() {
123         MCPFeatureQueryFacade queryFacade = mock(MCPFeatureQueryFacade.class);
124         when(queryFacade.query("logic_db", "SHOW STORAGE UNITS FROM logic_db")).thenReturn(List.of(Map.of("name", "write_ds")));
125         MCPFeatureRequestContext requestContext = mock(MCPFeatureRequestContext.class);
126         when(requestContext.getMetadataQueryFacade()).thenReturn(mock(MCPMetadataQueryFacade.class));
127         when(requestContext.getQueryFacade()).thenReturn(queryFacade);
128         MCPSuccessPayload actual = new SearchMetadataToolHandler().handle(requestContext, Map.of("database", "logic_db", "query", "write", "object_types", List.of("storage_unit")));
129         Map<String, Object> actualPayload = actual.toPayload();
130         assertThat(actual, isA(MCPItemsPayload.class));
131         MetadataSearchHit actualHit = (MetadataSearchHit) ((List<?>) actualPayload.get("items")).getFirst();
132         assertThat(actualHit.getObjectType(), is("storage_unit"));
133         assertThat(actualHit.getName(), is("write_ds"));
134         assertThat(actualHit.getResource().get("uri"), is("shardingsphere://databases/logic_db/storage-units/write_ds"));
135         assertThat(((Map<?, ?>) actualPayload.get("search_context")).get("object_types"), is(List.of("storage_unit")));
136     }

170     void assertHandleSearchMetadataWithLargeResultGuidance() {
171         try (RequestContextFixture requestContextFixture = createSearchRequestContext(createLargeDatabaseMetadata())) {
172             MCPFeatureRuntimeRequestContext requestContext = requestContextFixture.getRequestContext();
173             MCPSuccessPayload actual = new SearchMetadataToolHandler().handle(requestContext, Map.of("database", "large_db", "object_types", List.of(SupportedMCPMetadataObjectType.TABLE.name())));
174             Map<String, Object> actualPayload = actual.toPayload();

176             assertThat(actualPayload.get("summary"), is("Metadata search returned 100 of 101 matches."));
177             assertThat(actualPayload.get("total_match_count"), is(101));
178             assertThat(actualPayload.get("count"), is(100));
179             assertTrue((Boolean) actualPayload.get("truncated"));
180             assertThat(actualPayload.get("continuation_mode"), is("pagination"));
181             assertThat(actualPayload.get("next_offset"), is(100));
182             Map<?, ?> actualLargeResultGuidance = (Map<?, ?>) actualPayload.get("large_result_guidance");
183             assertThat(actualLargeResultGuidance.get("state"), is("metadata_search_result_truncated"));

185             assertThat(actualLargeResultGuidance.get("narrowing_arguments"), is(List.of("database", "schema", "query", "object_types")));
186             List<?> actualNextActions = (List<?>) actualPayload.get("next_actions");
187             assertThat(actualNextActions.size(), is(1));
188             Map<?, ?> actualNextAction = (Map<?, ?>) actualNextActions.getFirst();
189             assertThat(actualNextAction.get("type"), is("tool_call"));
190             assertThat(actualNextAction.get("tool_name"), is("database_gateway_search_metadata"));
191             assertThat(((Map<?, ?>) actualNextAction.get("arguments")).get("offset"), is(100));
192         }
