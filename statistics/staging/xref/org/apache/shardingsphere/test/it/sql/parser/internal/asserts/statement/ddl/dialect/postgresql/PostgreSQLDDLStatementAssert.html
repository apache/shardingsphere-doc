126 /**
127  * DDL statement assert for PostgreSQL.
128  */
129 @NoArgsConstructor(access = AccessLevel.PRIVATE)
130 public final class PostgreSQLDDLStatementAssert {
131     
132     /**
133      * Assert DDL statement is correct with expected parser result.
134      *
135      * @param assertContext assert context
136      * @param actual actual DDL statement
137      * @param expected expected parser result
138      */
139     public static void assertIs(final SQLCaseAssertContext assertContext, final DDLStatement actual, final SQLParserTestCase expected) {
140         if ("PostgreSQL".equals(actual.getDatabaseType().getType())) {
141             assertStatementType(assertContext, actual, expected);
142             if (actual instanceof CreateTableStatement) {
143                 int count = 0;
144                 for (ConstraintDefinitionSegment each : ((CreateTableStatement) actual).getConstraintDefinitions()) {
145                     assertThat(assertContext.getText("Constraint definition primary key column size assertion error: "), each.getPrimaryKeyColumns().size(),
146                             is(((CreateTableStatementTestCase) expected).getConstraintDefinitions().get(count++).getPrimaryKeyColumns().size()));
147                 }
148             }
149         }
150         if (actual instanceof PostgreSQLDeclareStatement) {
151             PostgreSQLDeclareStatementAssert.assertIs(assertContext, (PostgreSQLDeclareStatement) actual, (PostgreSQLDeclareStatementTestCase) expected);
152         } else if (actual instanceof PostgreSQLClusterStatement) {
153             PostgreSQLClusterStatementAssert.assertIs(assertContext, (PostgreSQLClusterStatement) actual, (PostgreSQLClusterStatementTestCase) expected);
154         } else if (actual instanceof PostgreSQLListenStatement) {
155             PostgreSQLListenStatementAssert.assertIs(assertContext, (PostgreSQLListenStatement) actual, (PostgreSQLListenStatementTestCase) expected);
156         } else if (actual instanceof PostgreSQLUnlistenStatement) {
157             PostgreSQLUnlistenStatementAssert.assertIs(assertContext, (PostgreSQLUnlistenStatement) actual, (PostgreSQLUnlistenStatementTestCase) expected);
158         } else if (actual instanceof PostgreSQLNotifyStmtStatement) {
159             PostgreSQLNotifyStmtStatementAssert.assertIs(assertContext, (PostgreSQLNotifyStmtStatement) actual, (PostgreSQLNotifyStmtStatementTestCase) expected);
160         } else if (actual instanceof PostgreSQLReindexStatement) {
161             PostgreSQLReindexStatementAssert.assertIs(assertContext, (PostgreSQLReindexStatement) actual, (PostgreSQLReindexStatementTestCase) expected);
162         } else if (actual instanceof PostgreSQLSecurityLabelStmtStatement) {
163             PostgreSQLSecurityLabelStmtStatementAssert.assertIs(assertContext, (PostgreSQLSecurityLabelStmtStatement) actual, (PostgreSQLSecurityLabelStmtStatementTestCase) expected);
164         } else if (actual instanceof PostgreSQLOpenStatement) {
165             PostgreSQLOpenStatementAssert.assertIs(assertContext, (PostgreSQLOpenStatement) actual, (PostgreSQLOpenStatementTestCase) expected);
166         }
167     }
