18  package org.apache.shardingsphere.mcp.core.tool.handler.metadata;
19  
20  import org.apache.shardingsphere.infra.exception.ShardingSpherePreconditions;
21  import org.apache.shardingsphere.mcp.api.exception.MCPInvalidRequestException;
22  import org.apache.shardingsphere.mcp.core.metadata.GovernanceMetadataQueryService;
23  import org.apache.shardingsphere.mcp.core.tool.payload.MetadataSearchHit;
24  import org.apache.shardingsphere.mcp.core.tool.payload.MetadataSearchResult;
25  import org.apache.shardingsphere.mcp.core.tool.request.MetadataSearchRequest;
26  import org.apache.shardingsphere.mcp.support.database.capability.SupportedMCPMetadataObjectType;
27  import org.apache.shardingsphere.mcp.support.database.spi.MCPFeatureQueryFacade;
28  import org.apache.shardingsphere.mcp.support.database.spi.MCPMetadataQueryFacade;
29  
30  import java.util.EnumSet;
31  import java.util.LinkedHashMap;
32  import java.util.List;
33  import java.util.Locale;
34  import java.util.Map;
35  import java.util.Set;
36  
37  /**
38   * Search metadata tool service.
39   */
40  public final class SearchMetadataToolService {
41      
42      private static final int LARGE_RESULT_THRESHOLD = 100;
43      
44      private static final Map<String, Integer> OBJECT_TYPE_ORDERS = Map.of(
45              "database", 0, "schema", 1, "storage_unit", 2, "table", 3, "view", 4, "column", 5, "index", 6, "sequence", 7);
46      
47      private final MetadataSearchCollector collector;
48      
49      private final MetadataSearchMatcher matcher = new MetadataSearchMatcher();
50      
51      public SearchMetadataToolService(final MCPMetadataQueryFacade metadataQueryFacade, final MCPFeatureQueryFacade queryFacade) {
52          collector = new MetadataSearchCollector(metadataQueryFacade, queryFacade, new GovernanceMetadataQueryService(), new MetadataSearchResourceUriFactory());
53      }
54      
55      /**
56       * Search metadata.
57       *
58       * @param request search request
59       * @return search result
60       */
61      public MetadataSearchResult execute(final MetadataSearchRequest request) {
62          ShardingSpherePreconditions.checkState(request.getSchema().isEmpty() || !request.getDatabase().isEmpty(), () -> new MCPInvalidRequestException("Schema cannot be provided without database."));
63          if (isBlankAllDatabaseSearch(request)) {
64              return executeBlankAllDatabaseSearch(request);
65          }
66          Set<SupportedMCPMetadataObjectType> searchObjectTypes = getSearchObjectTypes(request.getObjectTypes());
67          return createSearchResult(collector.collect(request, searchObjectTypes), request, searchObjectTypes, false);
68      }
69      
70      private boolean isBlankAllDatabaseSearch(final MetadataSearchRequest request) {
71          return request.getDatabase().isEmpty() && request.getQuery().isEmpty() && request.getObjectTypes().isEmpty();
72      }
73      
74      private MetadataSearchResult executeBlankAllDatabaseSearch(final MetadataSearchRequest request) {
75          Set<SupportedMCPMetadataObjectType> searchObjectTypes = Set.of(SupportedMCPMetadataObjectType.DATABASE);
76          return createSearchResult(collector.collectDatabases(), request, searchObjectTypes, true);
77      }
78      
79      private MetadataSearchResult createSearchResult(final List<MetadataSearchHit> metadataItems, final MetadataSearchRequest request,
80                                                      final Set<SupportedMCPMetadataObjectType> searchObjectTypes, final boolean broadSearchGuarded) {
81          List<MetadataSearchHit> filteredItems = matcher.filterByQuery(metadataItems, request.getQuery());
82          filteredItems.sort(this::compareSearchHits);
83          List<MetadataSearchHit> returnedItems = pageSearchResult(filteredItems, request);
84          return new MetadataSearchResult(returnedItems, createSearchContext(request, searchObjectTypes, broadSearchGuarded), filteredItems.size(), returnedItems.size(),
85                  request.getOffset() + returnedItems.size() < filteredItems.size(), LARGE_RESULT_THRESHOLD);
86      }
87      
88      private List<MetadataSearchHit> pageSearchResult(final List<MetadataSearchHit> items, final MetadataSearchRequest request) {
89          int fromIndex = Math.min(request.getOffset(), items.size());
90          int toIndex = (int) Math.min((long) fromIndex + request.getLimit(), items.size());
91          return items.subList(fromIndex, toIndex);
92      }
93      
94      private Map<String, Object> createSearchContext(final MetadataSearchRequest request, final Set<SupportedMCPMetadataObjectType> searchObjectTypes, final boolean broadSearchGuarded) {
95          Map<String, Object> result = new LinkedHashMap<>(8, 1F);
96          result.put("query", request.getQuery());
97          result.put("database", request.getDatabase());
98          result.put("database_scope", request.getDatabase().isEmpty() ? "all_query_databases" : "single_database");
99          result.put("schema", request.getSchema());
100         result.put("object_types", createObjectTypeNames(searchObjectTypes));
101         result.put("limit", request.getLimit());
102         result.put("offset", request.getOffset());
103         if (broadSearchGuarded) {
104             result.put("broad_search_guarded", true);
105             result.put("guard_reason", "Blank cross-database metadata search lists databases only instead of expanding every object type.");
106             result.put("recommended_narrowing_arguments", List.of("database", "query", "object_types"));
107         }
108         return result;
109     }
110     
111     private List<String> createObjectTypeNames(final Set<SupportedMCPMetadataObjectType> searchObjectTypes) {
112         return searchObjectTypes.stream().map(each -> each.name().toLowerCase(Locale.ENGLISH)).sorted(this::compareObjectTypeNames).toList();
113     }
114     
115     private int compareObjectTypeNames(final String left, final String right) {
116         return Integer.compare(getObjectTypeOrder(left), getObjectTypeOrder(right));
117     }
118     
119     private int compareSearchHits(final MetadataSearchHit left, final MetadataSearchHit right) {
120         int result = left.getDatabase().compareTo(right.getDatabase());
121         if (0 != result) {
122             return result;
123         }
124         result = left.getSchema().compareTo(right.getSchema());
125         if (0 != result) {
126             return result;
127         }
128         result = Integer.compare(getObjectTypeOrder(left.getObjectType()), getObjectTypeOrder(right.getObjectType()));
129         if (0 != result) {
130             return result;
131         }
132         result = left.getTable().compareTo(right.getTable());
133         if (0 != result) {
134             return result;
135         }
136         result = left.getView().compareTo(right.getView());
137         if (0 != result) {
138             return result;
139         }
140         return left.getName().compareTo(right.getName());
141     }
142     
143     private int getObjectTypeOrder(final String objectType) {
144         return OBJECT_TYPE_ORDERS.getOrDefault(objectType, Integer.MAX_VALUE);
145     }
146     
147     private Set<SupportedMCPMetadataObjectType> getSearchObjectTypes(final Set<SupportedMCPMetadataObjectType> objectTypes) {
148         if (!objectTypes.isEmpty()) {
149             return objectTypes;
150         }
151         return EnumSet.of(SupportedMCPMetadataObjectType.DATABASE, SupportedMCPMetadataObjectType.SCHEMA, SupportedMCPMetadataObjectType.TABLE, SupportedMCPMetadataObjectType.VIEW,
152                 SupportedMCPMetadataObjectType.COLUMN, SupportedMCPMetadataObjectType.INDEX, SupportedMCPMetadataObjectType.SEQUENCE);
153     }
