20  import org.apache.shardingsphere.mcp.api.capability.tool.MCPToolHandler;
21  import org.apache.shardingsphere.mcp.api.payload.MCPSuccessPayload;
22  import org.apache.shardingsphere.mcp.core.tool.payload.SQLExecutionPayload;
23  import org.apache.shardingsphere.mcp.core.tool.request.MCPToolArguments;
24  import org.apache.shardingsphere.mcp.support.MCPFeatureRequestContext;
25  import org.apache.shardingsphere.mcp.support.descriptor.CoreToolNames;
26  import org.apache.shardingsphere.mcp.support.protocol.MCPPayloadFieldNames;
27  
28  import java.util.LinkedHashMap;
29  import java.util.Map;
30  
31  /**
32   * Execute read-only SQL query tool handler.
33   */
34  public final class ExecuteQueryToolHandler implements MCPToolHandler<MCPFeatureRequestContext> {
35      
36      @Override
37      public Class<MCPFeatureRequestContext> getContextType() {
38          return MCPFeatureRequestContext.class;
39      }
40      
41      @Override
42      public String getToolName() {
43          return CoreToolNames.EXECUTE_QUERY;
44      }
45      
46      @Override
47      public MCPSuccessPayload handle(final MCPFeatureRequestContext requestContext, final Map<String, Object> arguments) {
48          MCPToolArguments toolArguments = new MCPToolArguments(arguments);
49          String sql = toolArguments.getStringArgument("sql");
50          checkReadOnlyQuery(requestContext, toolArguments, sql);
51          SQLExecutionToolHandlerSupport.checkExecutionArguments(toolArguments, CoreToolNames.EXECUTE_QUERY);
52          return SQLExecutionPayload.query(requestContext.getExecutionFacade().execute(SQLExecutionToolHandlerSupport.createReadOnlyExecutionRequest(
53                  requestContext.getSessionIdentity().getSessionId(), toolArguments,
54                  SQLExecutionToolHandlerSupport.resolveSchema(requestContext, toolArguments), sql, CoreToolNames.EXECUTE_QUERY)));
55      }
56      
57      private void checkReadOnlyQuery(final MCPFeatureRequestContext requestContext, final MCPToolArguments toolArguments, final String sql) {
58          ClassificationResult classificationResult = SQLExecutionToolHandlerSupport.analyze(requestContext, toolArguments, sql);
59          if (!SQLExecutionToolHandlerSupport.isQueryStatement(classificationResult)) {
60              throw new SQLToolMismatchException(
61                      "database_gateway_execute_query only supports parser-approved QUERY statements. "
62                              + "Use database_gateway_execute_explain_query for EXPLAIN diagnostics or database_gateway_execute_update for side-effecting SQL.",
63                      CoreToolNames.EXECUTE_QUERY, CoreToolNames.EXECUTE_UPDATE, classificationResult,
64                      createSuggestedArguments(toolArguments, classificationResult));
65          }
66      }
67      
68      private Map<String, Object> createSuggestedArguments(final MCPToolArguments toolArguments, final ClassificationResult classificationResult) {
69          Map<String, Object> result = new LinkedHashMap<>(4, 1F);
70          SQLExecutionToolHandlerSupport.putIfNotEmpty(result, "database", toolArguments.getStringArgument("database"));
71          SQLExecutionToolHandlerSupport.putIfNotEmpty(result, "schema", toolArguments.getStringArgument("schema"));
72          result.put("sql", classificationResult.getNormalizedSql());
73          result.put(MCPPayloadFieldNames.EXECUTION_MODE, "preview");
74          return result;
75      }
76  }
