18  package org.apache.shardingsphere.database.protocol.firebird.packet.command.query.statement.execute.protocol;

20  import io.netty.buffer.ByteBuf;
21  import io.netty.buffer.Unpooled;
22  import org.apache.shardingsphere.database.protocol.firebird.payload.FirebirdPacketPayload;
23  import org.junit.jupiter.api.Test;
24  import org.junit.jupiter.api.extension.ExtendWith;
25  import org.junit.jupiter.params.ParameterizedTest;
26  import org.junit.jupiter.params.provider.CsvSource;
27  import org.mockito.InOrder;
28  import org.mockito.Mock;
29  import org.mockito.junit.jupiter.MockitoExtension;
30  
31  import java.nio.charset.StandardCharsets;
32  import java.time.OffsetDateTime;
33  
34  import static org.hamcrest.MatcherAssert.assertThat;
35  import static org.hamcrest.Matchers.is;
36  import static org.mockito.Mockito.inOrder;
37  
38  @ExtendWith(MockitoExtension.class)
39  class FirebirdTimestampTZBinaryProtocolValueTest {
40      
41      @Mock
42      private FirebirdPacketPayload payload;
43      
44      @ParameterizedTest(name = "{2}: {3}")
45      @CsvSource({
46              "60310, 801301234, 1559, 2024-01-02T00:15:30.123400+02:00",
47              "60617, 234000000, 65361, 2024-11-03T01:30:00-05:00",
48              "60617, 234000000, -175, 2024-11-03T01:30:00-05:00"
49      })
50      void assertRead(final int encodedDate, final int encodedTime, final int encodedTimeZone, final String expected) {
51          ByteBuf buffer = Unpooled.buffer();
52          try {
53              buffer.writeInt(encodedDate).writeInt(encodedTime).writeInt(encodedTimeZone).writeInt(0x12345678);
54              FirebirdPacketPayload actualPayload = new FirebirdPacketPayload(buffer, StandardCharsets.UTF_8);
55              Object actual = new FirebirdTimestampTZBinaryProtocolValue().read(actualPayload);
56              assertThat(actual, is(OffsetDateTime.parse(expected)));
57              assertThat(buffer.readerIndex(), is(12));
58              assertThat(actualPayload.readInt4(), is(0x12345678));
59          } finally {
60              buffer.release();
61          }
62      }
63      
64      @ParameterizedTest(name = "{0}")
65      @CsvSource({
66              "2024-01-01T12:00:00Z, 60310, 432000000, 1439",
67              "2024-01-01T12:00:00+02:00, 60310, 360000000, 1559",
68              "2024-01-01T12:00:00-05:00, 60310, 612000000, 1139",
69              "2024-01-01T12:00:00+05:30, 60310, 234000000, 1769",
70              "2024-01-01T01:00:00+08:00, 60309, 612000000, 1919",
71              "2024-01-01T23:30:00-05:00, 60311, 162000000, 1139",
72              "2024-01-02T00:15:30.123400+02:00, 60310, 801301234, 1559",
73              "2024-03-10T02:30:00Z, 60379, 90000000, 1439",
74              "2024-11-03T01:30:00-05:00, 60617, 234000000, 1139"
75      })
76      void assertWrite(final String value, final int encodedDate, final int encodedTime, final int encodedTimeZone) {
77          new FirebirdTimestampTZBinaryProtocolValue().write(payload, OffsetDateTime.parse(value));
78          InOrder ordered = inOrder(payload);
79          ordered.verify(payload).writeInt4(encodedDate);
80          ordered.verify(payload).writeInt4(encodedTime);
81          ordered.verify(payload).writeInt4(encodedTimeZone);
82      }
83      
84      @Test
85      void assertGetLength() {
86          assertThat(new FirebirdTimestampTZBinaryProtocolValue().getLength(payload), is(12));
87      }
88  }
