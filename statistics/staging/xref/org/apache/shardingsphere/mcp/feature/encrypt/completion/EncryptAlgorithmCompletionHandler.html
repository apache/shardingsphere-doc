18  package org.apache.shardingsphere.mcp.feature.encrypt.completion;
19  
20  import org.apache.shardingsphere.mcp.api.capability.completion.MCPCompletionCandidate;
21  import org.apache.shardingsphere.mcp.api.capability.completion.MCPCompletionHandler;
22  import org.apache.shardingsphere.mcp.api.capability.completion.MCPCompletionHandlerResult;
23  import org.apache.shardingsphere.mcp.api.capability.completion.MCPCompletionRequest;
24  import org.apache.shardingsphere.mcp.feature.encrypt.EncryptFeatureDefinition;
25  import org.apache.shardingsphere.mcp.feature.encrypt.tool.service.EncryptRuleInspectionService;
26  import org.apache.shardingsphere.mcp.support.MCPFeatureRequestContext;
27  
28  import java.util.Map;
29  import java.util.Objects;
30  import java.util.Set;
31  
32  /**
33   * Encrypt algorithm completion handler.
34   */
35  public final class EncryptAlgorithmCompletionHandler implements MCPCompletionHandler<MCPFeatureRequestContext> {
36      
37      private static final Set<String> SUPPORTED_ARGUMENTS = Set.of("algorithm_type", "assisted_query_algorithm_type", "like_query_algorithm_type");
38      
39      private final EncryptRuleInspectionService ruleInspectionService = new EncryptRuleInspectionService();
40      
41      @Override
42      public Class<MCPFeatureRequestContext> getContextType() {
43          return MCPFeatureRequestContext.class;
44      }
45      
46      @Override
47      public boolean supports(final MCPCompletionRequest request) {
48          return SUPPORTED_ARGUMENTS.contains(request.getArgumentName()) && isEncryptReference(request);
49      }
50      
51      private boolean isEncryptReference(final MCPCompletionRequest request) {
52          String reference = request.getDescriptor().getReference();
53          return EncryptFeatureDefinition.PLAN_PROMPT_NAME.equals(reference) || EncryptFeatureDefinition.ALGORITHMS_RESOURCE_URI.equals(reference);
54      }
55      
56      @Override
57      public MCPCompletionHandlerResult complete(final MCPFeatureRequestContext handlerContext, final MCPCompletionRequest request) {
58          return new MCPCompletionHandlerResult(ruleInspectionService.queryEncryptAlgorithms(handlerContext.getQueryFacade()).stream()
59                  .map(this::createAlgorithmCandidate).filter(each -> !each.getValue().isEmpty()).toList());
60      }
61      
62      private MCPCompletionCandidate createAlgorithmCandidate(final Map<String, Object> row) {
63          String value = Objects.toString(row.get("type"), "").trim();
64          String label = Objects.toString(row.getOrDefault("description", "algorithm"), "algorithm");
65          return new MCPCompletionCandidate(value, label, "encrypt-algorithm");
66      }
67  }
