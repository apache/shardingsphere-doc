42  @SQLBinderITSettings("Oracle")
43  class OracleBinderIT extends SQLBinderIT {
44      
45      @Test
46      void assertBindRemovedUnparenthesizedFunctionNamesAsDerivedColumns() {
47          String sql = "SELECT derived.day, derived.rownum_, derived.row_number "
48                  + "FROM (SELECT order_id AS day, user_id AS rownum_, status AS row_number FROM t_order) derived";
49          SelectStatement actual = (SelectStatement) bindSQLStatement("Oracle", sql);
50          List<ProjectionSegment> actualProjections = actual.getProjections().getProjections();
51          assertColumnBound(actualProjections.get(0), "t_order", "order_id");
52          assertColumnBound(actualProjections.get(1), "t_order", "user_id");
53          assertColumnBound(actualProjections.get(2), "t_order", "status");
54      }
55      
56      @Test
57      void assertBindQuotedUnparenthesizedFunctionAsDerivedColumn() {
58          String sql = "SELECT derived.\"SYSDATE\" FROM (SELECT creation_date AS \"SYSDATE\" FROM t_order) derived";
59          SelectStatement actual = (SelectStatement) bindSQLStatement("Oracle", sql);
60          assertColumnBound(actual.getProjections().getProjections().get(0), "t_order", "creation_date");
61      }
62      
63      @Test
64      void assertBindOraclePaginationAlias() {
65          String sql = "SELECT tt.rownum_ FROM (SELECT ROWNUM rownum_ FROM t_order) tt WHERE tt.rownum_ > 1";
66          SelectStatement actual = (SelectStatement) bindSQLStatement("Oracle", sql);
67          assertColumnBound(actual.getProjections().getProjections().get(0), "", "ROWNUM");
68      }
69      
70      private void assertColumnBound(final ProjectionSegment actualProjection, final String expectedTable, final String expectedColumn) {
71          assertThat(actualProjection, isA(ColumnProjectionSegment.class));
72          ColumnSegmentBoundInfo actual = ((ColumnProjectionSegment) actualProjection).getColumn().getColumnBoundInfo();
73          assertThat(actual.getOriginalTable().getValue(), is(expectedTable));
74          assertThat(actual.getOriginalColumn().getValue(), is(expectedColumn));
75      }
76      
77      @Test
78      void assertBindDottedDatabaseLink() {
79          SelectStatement actual = (SelectStatement) bindSQLStatement("Oracle", "SELECT 1 FROM foo_table@foo_link.example.com");
80          SimpleTableSegment actualTable = (SimpleTableSegment) actual.getFrom().get();
81          assertThat(actualTable.getDbLink().get().getValue(), is("foo_link.example.com"));
82          assertThat(actualTable.getAt().get().getValue(), is("@"));
83      }
84      
85      @ParameterizedTest(name = "{0}")
86      @MethodSource("getTypeAttributeStatements")
87      void assertBindWithTypeAttribute(final String name, final String sql) {
88          SQLStatement actual = bindSQLStatement("Oracle", sql);
89          assertTrue(new CommonSQLStatementContext(actual).getTablesContext().getTableNames().isEmpty());
90      }

93          return Stream.of(
94                  Arguments.of("BlockWithLocalRecordFieldTypeAttribute", "DECLARE TYPE foo_rec IS RECORD (order_id NUMBER); t_order foo_rec; foo_copy t_order.order_id%TYPE; BEGIN NULL; END;"),

96                  Arguments.of("PackageWithLocalRecordFieldTypeAttribute",
97                          "CREATE PACKAGE foo_pkg AS TYPE foo_rec IS RECORD (foo_id NUMBER); foo_record foo_rec; foo_copy foo_pkg.foo_record.foo_id%TYPE; END;"),
98                  Arguments.of("PackageWithSchemaQualifiedCursorRowtypeAttribute",
99                          "CREATE PACKAGE foo_schema.foo_pkg AS TYPE foo_rec IS RECORD (foo_id NUMBER); CURSOR foo_cursor RETURN foo_rec;"
100                                 + " TYPE foo_ref IS REF CURSOR RETURN foo_schema.foo_pkg.foo_cursor%ROWTYPE; END;"),
101                 Arguments.of("PackageWithNestedLocalCursorRowtypeAttribute",
102                         "CREATE PACKAGE BODY foo_pkg AS PROCEDURE foo_proc IS CURSOR foo_cursor IS SELECT 1 foo_id FROM dual;"
103                                 + " TYPE foo_ref IS REF CURSOR RETURN foo_pkg.foo_proc.foo_cursor%ROWTYPE; BEGIN NULL; END; END;"),
104                 Arguments.of("BlockWithNestedLocalCursorRowtypeAttribute",
105                         "<<foo_block>> DECLARE PROCEDURE foo_proc IS CURSOR foo_cursor IS SELECT 1 foo_id FROM dual;"
106                                 + " TYPE foo_ref IS REF CURSOR RETURN foo_block.foo_proc.foo_cursor%ROWTYPE; BEGIN NULL; END; BEGIN NULL; END;"),
107                 Arguments.of("PackageWithSchemaQualifiedScalarTypeAttribute", "CREATE PACKAGE foo_schema.foo_pkg AS foo_var NUMBER; foo_copy foo_schema.foo_pkg.foo_var%TYPE; END;"),
108                 Arguments.of("PackageWithNestedScalarTypeAttribute",
109                         "CREATE PACKAGE BODY foo_pkg AS PROCEDURE foo_proc IS foo_var NUMBER; foo_copy foo_pkg.foo_proc.foo_var%TYPE; BEGIN NULL; END; END;"),
110                 Arguments.of("FunctionWithSchemaQualifiedCursorRowtypeAttribute",
111                         "CREATE FUNCTION foo_schema.foo_func RETURN NUMBER AS CURSOR foo_cursor IS SELECT 1 foo_id FROM dual;"
112                                 + " TYPE foo_ref IS REF CURSOR RETURN foo_schema.foo_func.foo_cursor%ROWTYPE; BEGIN RETURN 1; END;"));
113     }
