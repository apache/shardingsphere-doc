50  /**
51   * Insert SQL statement binding context.
52   */
53  public final class InsertStatementBindingContext implements SQLStatementContext {
54      
55      private final InsertStatementBaseContext baseContext;
56      
57      @Getter
58      private final List<InsertValueContext> insertValueContexts;
59      
60      @Getter
61      private final InsertSelectContext insertSelectContext;
62      
63      @Getter
64      private final OnDuplicateUpdateContext onDuplicateKeyUpdateValueContext;
65      
66      private final GeneratedKeyContext generatedKeyContext;
67      
68      public InsertStatementBindingContext(final InsertStatementBaseContext baseContext, final List<Object> params, final ShardingSphereMetaData metaData, final String currentDatabaseName) {
69          this.baseContext = baseContext;
70          AtomicInteger parametersOffset = new AtomicInteger(getInsertValuesParametersOffset(params));
71          insertValueContexts = getInsertValueContexts(params, parametersOffset, baseContext.getValueExpressions());
72          insertSelectContext = getInsertSelectContext(params, parametersOffset, metaData, currentDatabaseName).orElse(null);
73          parametersOffset.addAndGet(getOnConflictTargetParameterCount());
74          onDuplicateKeyUpdateValueContext = getOnDuplicateKeyUpdateValueContext(params, parametersOffset).orElse(null);
75          generatedKeyContext = new GeneratedKeyContextEngine(baseContext.getSqlStatement(), baseContext.getSchema())
76                  .createGenerateKeyContext(baseContext.getInsertColumnNamesAndIndexes(), insertValueContexts, params).orElse(null);
77      }
78      
79      private int getInsertValuesParametersOffset(final List<Object> params) {
80          if (params.size() < baseContext.getSqlStatement().getParameterCount()) {
81              return 0;
82          }
83          ExpressionSegment firstValue = null;
84          for (List<ExpressionSegment> each : baseContext.getValueExpressions()) {
85              if (!each.isEmpty()) {
86                  firstValue = each.get(0);
87                  break;
88              }
89          }
90          int parameterFrameStartIndex;
91          if (null != firstValue) {
92              parameterFrameStartIndex = firstValue.getStartIndex();
93          } else if (baseContext.getSqlStatement().getInsertSelect().isPresent()) {
94              return 0;
95          } else if (baseContext.getSqlStatement().getOnDuplicateKeyColumns().isPresent()) {
96              parameterFrameStartIndex = baseContext.getSqlStatement().getOnDuplicateKeyColumns().get().getStartIndex();
97          } else if (baseContext.getSqlStatement().getReturning().isPresent()) {
98              parameterFrameStartIndex = baseContext.getSqlStatement().getReturning().get().getStartIndex();
99          } else {
100             return 0;
101         }
102         int result = 0;
103         int firstParameterIndex = Integer.MAX_VALUE;
104         for (ParameterMarkerSegment each : baseContext.getSqlStatement().getParameterMarkers()) {
105             if (each.getStopIndex() < parameterFrameStartIndex) {
106                 result = Math.max(result, each.getParameterIndex() + 1);
107             } else {
108                 firstParameterIndex = Math.min(firstParameterIndex, each.getParameterIndex());
109             }
110         }
111         return Integer.MAX_VALUE == firstParameterIndex ? result : firstParameterIndex;
112     }
113     
114     private List<InsertValueContext> getInsertValueContexts(final List<Object> params, final AtomicInteger paramsOffset, final List<List<ExpressionSegment>> valueExpressions) {
115         List<InsertValueContext> result = new ArrayList<>(valueExpressions.size());
116         for (Collection<ExpressionSegment> each : valueExpressions) {
117             InsertValueContext insertValueContext = new InsertValueContext(each, params, paramsOffset.get());
118             result.add(insertValueContext);
119             paramsOffset.addAndGet(insertValueContext.getParameterCount());
120         }
121         return result;
122     }
123     
124     private Optional<InsertSelectContext> getInsertSelectContext(final List<Object> params, final AtomicInteger paramsOffset,
125                                                                  final ShardingSphereMetaData metaData, final String currentDatabaseName) {
126         if (!baseContext.getSqlStatement().getInsertSelect().isPresent()) {
127             return Optional.empty();
128         }
129         SubquerySegment insertSelectSegment = baseContext.getSqlStatement().getInsertSelect().get();
130         SelectStatementContext selectStatementContext = new SelectStatementContext(createInsertSelectStatement(insertSelectSegment), metaData, currentDatabaseName, Collections.emptyList());
131         selectStatementContext.bindParameters(params);
132         selectStatementContext.setSubqueryType(SubqueryType.INSERT_SELECT);
133         setCombineSelectSubqueryType(selectStatementContext);
134         setProjectionSelectSubqueryType(selectStatementContext);
135         InsertSelectContext insertSelectContext = new InsertSelectContext(selectStatementContext, params, paramsOffset.get());
136         paramsOffset.addAndGet(insertSelectContext.getSelectStatementContext().getSqlStatement().getParameterCount());
137         return Optional.of(insertSelectContext);
138     }
139     
140     private SelectStatement createInsertSelectStatement(final SubquerySegment insertSelectSegment) {
141         SelectStatement selectStatement = insertSelectSegment.getSelect();
142         Optional<WithSegment> withSegment = baseContext.getSqlStatement().getWith();
143         if (!withSegment.isPresent()) {
144             return selectStatement;
145         }
146         SelectStatement result = SelectStatement.builder().databaseType(selectStatement.getDatabaseType()).projections(selectStatement.getProjections())
147                 .from(selectStatement.getFrom().orElse(null)).where(selectStatement.getWhere().orElse(null)).hierarchicalQuery(selectStatement.getHierarchicalQuery().orElse(null))
148                 .groupBy(selectStatement.getGroupBy().orElse(null)).having(selectStatement.getHaving().orElse(null)).orderBy(selectStatement.getOrderBy().orElse(null))
149                 .combine(selectStatement.getCombine().orElse(null)).with(selectStatement.getWith().orElse(withSegment.get())).subqueryType(selectStatement.getSubqueryType().orElse(null))
150                 .limit(selectStatement.getLimit().orElse(null)).lock(selectStatement.getLock().orElse(null)).window(selectStatement.getWindow().orElse(null))
151                 .into(selectStatement.getInto().orElse(null)).model(selectStatement.getModel().orElse(null)).outfile(selectStatement.getOutfile().orElse(null))
152                 .intoVariables(selectStatement.isIntoVariables()).withTableHint(selectStatement.getWithTableHint().orElse(null)).values(selectStatement.getValues()).build();
153         result.getTransformSegments().addAll(selectStatement.getTransformSegments());
154         result.addParameterMarkers(getInsertSelectParameterMarkers(insertSelectSegment, withSegment.get()));
155         result.getVariableNames().addAll(selectStatement.getVariableNames());
156         result.getComments().addAll(selectStatement.getComments());
157         return result;
158     }
159     
160     private Collection<ParameterMarkerSegment> getInsertSelectParameterMarkers(final SubquerySegment insertSelectSegment, final WithSegment withSegment) {
161         Collection<ParameterMarkerSegment> result = new LinkedList<>();
162         for (ParameterMarkerSegment each : baseContext.getSqlStatement().getParameterMarkers()) {
163             if (withSegment.getStartIndex() <= each.getStartIndex() && each.getStopIndex() <= withSegment.getStopIndex()
164                     || insertSelectSegment.getStartIndex() <= each.getStartIndex() && each.getStopIndex() <= insertSelectSegment.getStopIndex()) {
165                 result.add(each);
166             }
167         }
168         return result;
169     }
170     
171     private void setCombineSelectSubqueryType(final SelectStatementContext selectStatementContext) {
172         if (selectStatementContext.getSqlStatement().getCombine().isPresent()) {
173             CombineSegment combineSegment = selectStatementContext.getSqlStatement().getCombine().get();
174             Optional.ofNullable(selectStatementContext.getSubqueryContexts().get(combineSegment.getLeft().getStartIndex()))
175                     .ifPresent(optional -> optional.setSubqueryType(SubqueryType.INSERT_SELECT));
176             Optional.ofNullable(selectStatementContext.getSubqueryContexts().get(combineSegment.getRight().getStartIndex()))
177                     .ifPresent(optional -> optional.setSubqueryType(SubqueryType.INSERT_SELECT));
178         }
179     }
180     
181     private void setProjectionSelectSubqueryType(final SelectStatementContext selectStatementContext) {
182         for (Entry<Integer, SelectStatementContext> entry : selectStatementContext.getSubqueryContexts().entrySet()) {
183             if (entry.getKey() >= selectStatementContext.getProjectionsContext().getStartIndex() && entry.getKey() <= selectStatementContext.getProjectionsContext().getStopIndex()) {
184                 entry.getValue().setSubqueryType(SubqueryType.INSERT_SELECT);
185             }
186         }
187     }
188     
189     private Optional<OnDuplicateUpdateContext> getOnDuplicateKeyUpdateValueContext(final List<Object> params, final AtomicInteger parametersOffset) {
190         Optional<OnDuplicateKeyColumnsSegment> onDuplicateKeyColumnsSegment = baseContext.getSqlStatement().getOnDuplicateKeyColumns();
191         if (!onDuplicateKeyColumnsSegment.isPresent()) {
192             return Optional.empty();
193         }
194         Collection<ColumnAssignmentSegment> onDuplicateKeyColumns = onDuplicateKeyColumnsSegment.get().getColumns();
195         OnDuplicateUpdateContext onDuplicateUpdateContext = new OnDuplicateUpdateContext(onDuplicateKeyColumns, params, parametersOffset.get(),
196                 baseContext.getSqlStatement().getParameterMarkers());
197         parametersOffset.addAndGet(onDuplicateUpdateContext.getParameterCount());
198         return Optional.of(onDuplicateUpdateContext);
199     }
200     
201     private int getOnConflictTargetParameterCount() {
202         Optional<OnDuplicateKeyColumnsSegment> conflict = baseContext.getSqlStatement().getOnDuplicateKeyColumns();
203         if (!conflict.isPresent()) {
204             return 0;
205         }
206         int actionStartIndex = baseContext.getSqlStatement().getWhere().map(WhereSegment::getStartIndex).orElse(Integer.MAX_VALUE);
207         for (ColumnAssignmentSegment each : conflict.get().getColumns()) {
208             actionStartIndex = Math.min(actionStartIndex, each.getStartIndex());
209         }
210         int result = 0;
211         for (ParameterMarkerSegment each : baseContext.getSqlStatement().getParameterMarkers()) {
212             if (each.getStartIndex() >= conflict.get().getStartIndex() && each.getStopIndex() <= conflict.get().getStopIndex() && each.getStartIndex() < actionStartIndex) {
213                 result++;
214             }
215         }
216         return result;
217     }

219     /**
220      * Get grouped parameters.
221      *
222      * @return grouped parameters
223      */
224     public List<List<Object>> getGroupedParameters() {
225         List<List<Object>> result = new LinkedList<>();
226         for (InsertValueContext each : insertValueContexts) {
227             result.add(each.getParameters());
228         }
229         return result;
230     }

232     /**
233      * Get on duplicate key update parameters.
234      *
235      * @return on duplicate key update parameters
236      */
237     @SuppressWarnings("CollectionWithoutInitialCapacity")
238     public List<Object> getOnDuplicateKeyUpdateParameters() {
239         return null == onDuplicateKeyUpdateValueContext ? new ArrayList<>() : onDuplicateKeyUpdateValueContext.getParameters();
240     }
241     
242     /**
243      * Get generated key context.
244      *
245      * @return generated key context
246      */
247     public Optional<GeneratedKeyContext> getGeneratedKeyContext() {
248         return Optional.ofNullable(generatedKeyContext);
249     }

253         return baseContext.getSqlStatement();
254     }

258         return baseContext.getTablesContext();
259     }
