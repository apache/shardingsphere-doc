137     private ByteBuf mockJsonObjectByteBuf(final List<Object[]> jsonEntries, final boolean isSmall) {
138         ByteBuf result = Unpooled.buffer();
139         result.writeByte(isSmall ? JsonValueTypes.SMALL_JSON_OBJECT : JsonValueTypes.LARGE_JSON_OBJECT);
140         result.writeBytes(mockJsonObjectByteBufValue(jsonEntries, isSmall));
141         return result;
142     }

144     private ByteBuf mockJsonObjectByteBufValue(final List<Object[]> jsonEntries, final boolean isSmall) {
145         ByteBuf result = Unpooled.buffer();
146         writeInt(result, jsonEntries.size(), isSmall);
147         writeInt(result, 0, isSmall);
148         int startOffset = isSmall
149                 ? 1 + SMALL_JSON_INT_LENGTH + SMALL_JSON_INT_LENGTH + jsonEntries.size() * SMALL_JSON_KEY_META_DATA_LENGTH + jsonEntries.size() * SMALL_JSON_VALUE_META_DATA_LENGTH - 1
150                 : 1 + LARGE_JSON_INT_LENGTH + LARGE_JSON_INT_LENGTH + jsonEntries.size() * LARGE_JSON_KEY_META_DATA_LENGTH + jsonEntries.size() * LARGE_JSON_VALUE_META_DATA_LENGTH - 1;
151         ByteBuf keyByteBuf = writeKeys(result, jsonEntries, startOffset, isSmall);
152         startOffset += keyByteBuf.readableBytes();
153         ByteBuf valueByteBuf = writeValues(result, jsonEntries, startOffset, isSmall);
154         result.writeBytes(keyByteBuf);
155         result.writeBytes(valueByteBuf);
156         return result;
157     }

166     private ByteBuf mockJsonArrayByteBufValue(final List<Object[]> jsonEntries, final boolean isSmall) {
167         ByteBuf result = Unpooled.buffer();
168         writeInt(result, jsonEntries.size(), isSmall);
169         writeInt(result, 0, isSmall);
170         int startOffset = isSmall
171                 ? 1 + SMALL_JSON_INT_LENGTH + SMALL_JSON_INT_LENGTH + jsonEntries.size() * SMALL_JSON_VALUE_META_DATA_LENGTH - 1
172                 : 1 + LARGE_JSON_INT_LENGTH + LARGE_JSON_INT_LENGTH + jsonEntries.size() * LARGE_JSON_VALUE_META_DATA_LENGTH - 1;
173         ByteBuf valueByteBuf = writeValues(result, jsonEntries, startOffset, isSmall);
174         result.writeBytes(valueByteBuf);
175         return result;
176     }
177     
178     private void writeInt(final ByteBuf jsonByteBuf, final int value, final boolean isSmall) {
179         if (isSmall) {
180             jsonByteBuf.writeShortLE(value);
181         } else {
182             jsonByteBuf.writeIntLE(value);
183         }
184     }
185     
186     private ByteBuf writeKeys(final ByteBuf jsonByteBuf, final List<Object[]> jsonEntries, final int startOffset, final boolean isSmall) {
187         ByteBuf result = Unpooled.buffer();
188         for (Object[] each : jsonEntries) {
189             writeInt(jsonByteBuf, startOffset + result.readableBytes(), isSmall);
190             byte[] keyBytes = ((String) each[1]).getBytes(StandardCharsets.UTF_8);
191             jsonByteBuf.writeShortLE(keyBytes.length);
192             result.writeBytes(keyBytes);
193         }
194         return result;
195     }
196     
197     private ByteBuf writeValues(final ByteBuf jsonByteBuf, final List<Object[]> jsonEntries, final int startOffset, final boolean isSmall) {
198         ByteBuf result = Unpooled.buffer();
199         for (Object[] each : jsonEntries) {
200             jsonByteBuf.writeByte((byte) each[0]);
201             int offsetOrInlineValue = getOffsetOrInlineValue(startOffset, result.readableBytes(), each, isSmall);
202             writeInt(jsonByteBuf, offsetOrInlineValue, isSmall);
203             writeValueToByteBuf(each, result, isSmall);
204         }
205         return result;
206     }
207     
208     private int getOffsetOrInlineValue(final int startOffset, final int readableBytes, final Object[] jsonEntry, final boolean isSmall) {
209         switch ((byte) jsonEntry[0]) {
210             case JsonValueTypes.INT16:
211             case JsonValueTypes.UINT16:
212                 return (int) jsonEntry[2];
213             case JsonValueTypes.LITERAL:
214                 return (byte) jsonEntry[2];
215             case JsonValueTypes.INT32:
216             case JsonValueTypes.UINT32:
217                 return isSmall ? startOffset + readableBytes : (int) jsonEntry[2];
218             default:
219                 return startOffset + readableBytes;
220         }
221     }
222     
223     @SuppressWarnings("unchecked")
224     private void writeValueToByteBuf(final Object[] jsonEntry, final ByteBuf byteBuf, final boolean isSmall) {
225         switch ((byte) jsonEntry[0]) {
226             case JsonValueTypes.SMALL_JSON_OBJECT:
227                 byteBuf.writeBytes(mockJsonObjectByteBufValue((List<Object[]>) jsonEntry[2], true));
228                 break;
229             case JsonValueTypes.LARGE_JSON_OBJECT:
230                 byteBuf.writeBytes(mockJsonObjectByteBufValue((List<Object[]>) jsonEntry[2], false));
231                 break;
232             case JsonValueTypes.SMALL_JSON_ARRAY:
233                 byteBuf.writeBytes(mockJsonArrayByteBufValue((List<Object[]>) jsonEntry[2], true));
234                 break;
235             case JsonValueTypes.LARGE_JSON_ARRAY:
236                 byteBuf.writeBytes(mockJsonArrayByteBufValue((List<Object[]>) jsonEntry[2], false));
237                 break;
238             case JsonValueTypes.INT32:
239             case JsonValueTypes.UINT32:
240                 if (isSmall) {
241                     byteBuf.writeIntLE((int) jsonEntry[2]);
242                 }
243                 break;
244             case JsonValueTypes.INT64:
245             case JsonValueTypes.UINT64:
246                 byteBuf.writeLongLE((long) jsonEntry[2]);
247                 break;
248             case JsonValueTypes.DOUBLE:
249                 byteBuf.writeDoubleLE((double) jsonEntry[2]);
250                 break;
251             case JsonValueTypes.STRING:
252                 writeString(byteBuf, (String) jsonEntry[2]);
253                 break;
254             default:
255         }
256     }

258     private void writeString(final ByteBuf jsonByteBuf, final String value) {
259         byte[] valueBytes = value.getBytes(StandardCharsets.UTF_8);
260         byte[] result = codecDataLength(valueBytes.length);
261         jsonByteBuf.writeBytes(result, 0, result.length);
262         jsonByteBuf.writeBytes(valueBytes);
263     }
264     
265     private byte[] codecDataLength(final int length) {
266         byte[] lengthData = new byte[32 / 7 + (0 == 32 % 7 ? 0 : 1)];
267         for (int i = 0; i < lengthData.length; i++) {
268             lengthData[i] = (byte) ((length >> (7 * i)) & 0x7f);
269         }
270         int index = lengthData.length - 1;
271         while (index > 0) {
272             if (0 != lengthData[index]) {
273                 break;
274             }
275             index--;
276         }
277         for (int i = 0; i < index; i++) {
278             lengthData[i] |= 0x80;
279         }
280         byte[] result = new byte[index + 1];
281         System.arraycopy(lengthData, 0, result, 0, index + 1);
282         return result;
283     }

329         String value2 = Strings.repeat("1", (int) (Math.pow(2D, 7D) - 1D));
330         String value3 = Strings.repeat("1", (int) (Math.pow(2D, 7D) - 1D + 1D));
