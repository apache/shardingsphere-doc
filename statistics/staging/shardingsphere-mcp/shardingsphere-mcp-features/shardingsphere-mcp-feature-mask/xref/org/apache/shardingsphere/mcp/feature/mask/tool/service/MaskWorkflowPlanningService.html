42  /**
43   * Mask workflow planning service.
44   */
45  public final class MaskWorkflowPlanningService {
46      
47      private static final List<String> SUPPORTED_OPERATION_TYPES = List.of(WorkflowLifecycle.OPERATION_CREATE, WorkflowLifecycle.OPERATION_DROP);
48      
49      private static final List<String> INTERACTION_STEPS = List.of(
50              "Confirm database, table, column and target lifecycle",
51              "Inspect mask algorithm plugins and DistSQL-visible rules",
52              "Clarify masking intent and choose algorithm",
53              "Collect algorithm properties",
54              "Generate mask rule DistSQL artifacts",
55              "Review artifacts and choose execution mode",
56              "Execute or export artifacts",
57              "Validate and summarize");
58      
59      private static final List<String> VALIDATION_LAYERS = List.of("rules");
60      
61      private final WorkflowPlanningSupport planningSupport = new WorkflowPlanningSupport();
62      
63      private final MaskRuleInspectionService ruleInspectionService = new MaskRuleInspectionService();
64      
65      private final MaskAlgorithmRecommendationService algorithmRecommendationService = new MaskAlgorithmRecommendationService();
66      
67      private final MaskAlgorithmPropertyTemplateService algorithmPropertyTemplateService = new MaskAlgorithmPropertyTemplateService();
68      
69      private final MaskRuleDistSQLPlanningService ruleDistSQLPlanningService = new MaskRuleDistSQLPlanningService();
70      
71      /**
72       * Plan mask workflow.
73       *
74       * @param workflowSessionContext workflow session context
75       * @param metadataQueryFacade metadata query facade
76       * @param queryFacade query facade
77       * @param request workflow request
78       * @return workflow snapshot
79       */
80      public WorkflowContextSnapshot plan(final WorkflowSessionContext workflowSessionContext, final MCPMetadataQueryFacade metadataQueryFacade, final MCPFeatureQueryFacade queryFacade,
81                                          final WorkflowRequest request) {
82          WorkflowContextSnapshot result = workflowSessionContext.getOrCreate(request.getPlanId());
83          WorkflowRequest mergedRequest = prepareSnapshot(result, request);
84          ClarifiedIntent clarifiedIntent = result.getClarifiedIntent();
85          planningSupport.applyResolvedIntent(mergedRequest, clarifiedIntent);
86          if (!planningSupport.ensureSupportedOperationType(clarifiedIntent, SUPPORTED_OPERATION_TYPES, result)) {
87              return planningSupport.persistPlanningInterruption(workflowSessionContext, result);
88          }
89          if (!planningSupport.ensurePlanningContext(metadataQueryFacade, queryFacade, mergedRequest, clarifiedIntent, result)) {
90              return planningSupport.persistPlanningInterruption(workflowSessionContext, result);
91          }
92          queryFacade.checkDatabaseCapability(mergedRequest.getDatabase());
93          List<Map<String, Object>> existingRules = ruleInspectionService.queryMaskRules(queryFacade, mergedRequest.getDatabase(), mergedRequest.getTable());
94          if (!ensureLifecycleState(clarifiedIntent, mergedRequest, existingRules, result, queryFacade)) {
95              return workflowSessionContext.persist(result, WorkflowLifecycle.STEP_FAILED, WorkflowLifecycle.STATUS_FAILED);
96          }
97          if (!ensureSupportedRuleMutation(clarifiedIntent, mergedRequest, existingRules, result, queryFacade)) {
98              return workflowSessionContext.persist(result, WorkflowLifecycle.STEP_CLARIFYING, WorkflowLifecycle.STATUS_CLARIFYING);
99          }
100         if (isDropWorkflow(clarifiedIntent)) {
101             planArtifacts(clarifiedIntent, mergedRequest, result);
102             return workflowSessionContext.persist(result, WorkflowLifecycle.STEP_REVIEW, WorkflowLifecycle.STATUS_PLANNED);
103         }
104         if (!planNonDrop(queryFacade, clarifiedIntent, mergedRequest, result)) {
105             return workflowSessionContext.persist(result, WorkflowLifecycle.STEP_CLARIFYING, WorkflowLifecycle.STATUS_CLARIFYING);
106         }
107         return workflowSessionContext.persist(result, WorkflowLifecycle.STEP_REVIEW, WorkflowLifecycle.STATUS_PLANNED);
108     }
109     
110     private WorkflowRequest prepareSnapshot(final WorkflowContextSnapshot snapshot, final WorkflowRequest request) {
111         WorkflowRequest result = WorkflowRequest.merge(snapshot.getRequest(), request);
112         return planningSupport.prepareSnapshot(snapshot, MaskFeatureDefinition.WORKFLOW_KIND, result, null,
113                 planningSupport.createOperationIntent(result, WorkflowLifecycle.OPERATION_CREATE),
114                 "Mask workflow plan.", INTERACTION_STEPS, VALIDATION_LAYERS);
115     }
116     
117     private boolean ensureLifecycleState(final ClarifiedIntent clarifiedIntent, final WorkflowRequest request,
118                                          final List<Map<String, Object>> maskRules, final WorkflowContextSnapshot snapshot, final MCPFeatureQueryFacade queryFacade) {
119         boolean ruleExists = maskRules.stream().anyMatch(each -> queryFacade.isSameIdentifier(
120                 request.getDatabase(), IdentifierScope.COLUMN, request.getColumn(), WorkflowRuleValueUtils.getRuleValue(each, "column")));
121         return planningSupport.ensureLifecycleState("Mask rule", clarifiedIntent, ruleExists, snapshot);
122     }
123     
124     private boolean ensureSupportedRuleMutation(final ClarifiedIntent clarifiedIntent, final WorkflowRequest request,
125                                                 final List<Map<String, Object>> maskRules, final WorkflowContextSnapshot snapshot, final MCPFeatureQueryFacade queryFacade) {
126         if (isDropWorkflow(clarifiedIntent)) {
127             return !hasRemainingRulesAfterDrop(queryFacade, request, maskRules) || rejectExistingTableRuleMutation(clarifiedIntent, request, maskRules, snapshot);
128         }
129         return maskRules.isEmpty() || rejectExistingTableRuleMutation(clarifiedIntent, request, maskRules, snapshot);
130     }
131     
132     private boolean hasRemainingRulesAfterDrop(final MCPFeatureQueryFacade queryFacade, final WorkflowRequest request, final List<Map<String, Object>> maskRules) {
133         return maskRules.stream().anyMatch(each -> !queryFacade.isSameIdentifier(
134                 request.getDatabase(), IdentifierScope.COLUMN, request.getColumn(), WorkflowRuleValueUtils.getRuleValue(each, "column")));
135     }
136     
137     private boolean rejectExistingTableRuleMutation(final ClarifiedIntent clarifiedIntent, final WorkflowRequest request,
138                                                     final List<Map<String, Object>> maskRules, final WorkflowContextSnapshot snapshot) {
139         snapshot.getClarifiedIntent().getClarificationMessages().add(
140                 "Current Proxy DistSQL cannot automatically mutate an existing mask table rule. Recreate the mask rule manually with the complete column set during a maintenance window.");
141         snapshot.getIssues().add(new WorkflowIssue(WorkflowIssueCode.MASK_RULE_REWRITE_LIMITED, "error", WorkflowLifecycle.STEP_PLANNING_ARTIFACTS,
142                 "Mask planning cannot automatically rewrite an existing table rule or shrink it while preserving remaining columns.",
143                 "Manually recreate the mask rule with the complete column set after reviewing data impact.", true,
144                 Map.of("operation_type", clarifiedIntent.getOperationType(), "target_column", request.getColumn(), "existing_columns", createExistingRuleColumns(maskRules))));
145         return false;
146     }
147     
148     private List<String> createExistingRuleColumns(final List<Map<String, Object>> maskRules) {
149         List<String> result = new LinkedList<>();
150         for (Map<String, Object> each : maskRules) {
151             result.add(WorkflowRuleValueUtils.getRuleValue(each, "column"));
152         }
153         return result;
154     }
155     
156     private boolean isDropWorkflow(final ClarifiedIntent clarifiedIntent) {
157         return WorkflowLifecycle.OPERATION_DROP.equalsIgnoreCase(clarifiedIntent.getOperationType());
158     }
159     
160     private boolean planNonDrop(final MCPFeatureQueryFacade queryFacade, final ClarifiedIntent clarifiedIntent, final WorkflowRequest request, final WorkflowContextSnapshot snapshot) {
161         planAlgorithms(queryFacade, request, snapshot);
162         if (!planningSupport.isReadyForArtifactPlanning(request, clarifiedIntent, snapshot, findPropertyRequirements(request), "Please use a mask algorithm visible in the current Proxy.")) {
163             return false;
164         }
165         planArtifacts(clarifiedIntent, request, snapshot);
166         return true;
167     }
168     
169     private void planAlgorithms(final MCPFeatureQueryFacade queryFacade, final WorkflowRequest request, final WorkflowContextSnapshot snapshot) {
170         List<Map<String, Object>> maskAlgorithms = ruleInspectionService.queryMaskAlgorithms(queryFacade);
171         List<AlgorithmCandidate> algorithmCandidates = algorithmRecommendationService.recommendMaskAlgorithms(request, maskAlgorithms, snapshot.getIssues());
172         snapshot.getAlgorithmCandidates().addAll(algorithmCandidates);
173         if (!algorithmCandidates.isEmpty()) {
174             request.setAlgorithmType(algorithmCandidates.getFirst().getAlgorithmType());
175         }
176     }
177     
178     private List<AlgorithmPropertyRequirement> findPropertyRequirements(final WorkflowRequest request) {
179         return algorithmPropertyTemplateService.findRequirements(request.getAlgorithmType());
180     }
181     
182     private void planArtifacts(final ClarifiedIntent clarifiedIntent, final WorkflowRequest request, final WorkflowContextSnapshot snapshot) {
183         snapshot.getRuleArtifacts().add(isDropWorkflow(clarifiedIntent)
184                 ? ruleDistSQLPlanningService.planMaskDropRule(request)
185                 : ruleDistSQLPlanningService.planMaskRule(request));
186         snapshot.setFeatureData(new RuleWorkflowFeatureData(isDropWorkflow(clarifiedIntent) ? List.of() : List.of(createExpectedTargetRule(request))));
187     }
188     
189     private Map<String, Object> createExpectedTargetRule(final WorkflowRequest request) {
190         Map<String, Object> result = new LinkedHashMap<>(3, 1F);
191         result.put("column", request.getColumn());
192         result.put("algorithm_type", request.getAlgorithmType());
193         result.put("algorithm_props", request.getPrimaryAlgorithmProperties());
194         return result;
195     }
