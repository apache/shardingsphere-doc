328     private MCPToolDescriptor createExecutionModeTool(final boolean destructive, final boolean executionModePresent, final List<String> executionModes,
329                                                       final boolean executionModeRequired) {
330         Map<String, Object> properties = executionModePresent
331                 ? Map.of(MCPPayloadFieldNames.EXECUTION_MODE, Map.of("type", "string", "description", "Execution mode.", "enum", executionModes))
332                 : Map.of();
333         List<String> required = executionModeRequired ? List.of(MCPPayloadFieldNames.EXECUTION_MODE) : List.of();
334         MCPToolAnnotations annotations = MCPToolAnnotations.builder().title("Test Tool").readOnlyHint(false).destructiveHint(destructive)
335                 .idempotentHint(false).openWorldHint(true).build();
336         return new MCPToolDescriptor("database_gateway_test_tool", "Test Tool", "Run a test tool.", createInputSchema(properties, required),
337                 createOutputSchema(), annotations, Map.of());
338     }

391     private MCPToolDescriptor createExecuteUpdateDescriptorWithout(final String fieldName) {
392         Map<String, Object> outputProperties = new LinkedHashMap<>();
393         for (String each : List.of("response_mode", "result_kind", "statement_class", "statement_type", "status")) {
394             outputProperties.put(each, Map.of("type", "string", "description", "Execution result field."));
395         }
396         for (String each : List.of("columns", "rows")) {
397             outputProperties.put(each, Map.of("type", "array", "description", "Result-set field."));
398         }
399         for (String each : List.of("returned_row_count", "applied_max_rows", "applied_timeout_ms")) {
400             outputProperties.put(each, Map.of("type", "integer", "description", "Execution limit field."));
401         }
402         outputProperties.put("suggested_arguments", Map.of("type", "object", "description", "Suggested execution arguments."));
403         outputProperties.put(MCPPayloadFieldNames.NEXT_ACTIONS, createNextActionsSchema());
404         outputProperties.remove(fieldName);
405         Map<String, Object> inputSchema = createInputSchema(Map.of(MCPPayloadFieldNames.EXECUTION_MODE,
406                 Map.of("type", "string", "description", "Execution mode.", "enum", List.of("execute", "preview"))), List.of(MCPPayloadFieldNames.EXECUTION_MODE));
407         return new MCPToolDescriptor("database_gateway_execute_update", "Execute Update", "Execute a side-effecting statement.", inputSchema,
408                 createOutputSchema(outputProperties, List.of(Map.of("response_mode", "preview"))), createReadOnlyToolAnnotations("Execute Update"), Map.of());
409     }

447     private Map<String, Object> createWorkflowPlanOutputSchema() {
448         Map<String, Object> properties = new LinkedHashMap<>();
449         for (String each : List.of(
450                 "response_mode", MCPPayloadFieldNames.SUMMARY, WorkflowFieldNames.PLAN_ID, "workflow_kind", "status", "missing_required_inputs", "clarification_questions",
451                 "elicitation_support", "fallback_reason", "issues", "global_steps", "current_step", "algorithm_recommendations", "property_requirements", "validation_strategy",
452                 "delivery_mode", "execution_mode", "intent_inference", "argument_provenance", "review_focus", "proxy_topology_hint", "distsql_artifacts",
453                 MCPPayloadFieldNames.RESOURCES_TO_READ)) {
454             properties.put(each, Map.of("type", "object", "description", "Workflow plan field."));
455         }
456         properties.put(MCPPayloadFieldNames.NEXT_ACTIONS, createNextActionsSchema());
457         Map<String, Object> result = new LinkedHashMap<>(createOutputSchema(properties, List.of(Map.of("response_mode", "planning"))));

459                 "response_mode", WorkflowFieldNames.PLAN_ID, "workflow_kind", "status", "missing_required_inputs",
460                 MCPPayloadFieldNames.RESOURCES_TO_READ, MCPPayloadFieldNames.NEXT_ACTIONS));
461         return result;
462     }

483     private Map<String, Object> createNextActionsSchema(final String additionalFieldName) {
484         Map<String, Object> actionProperties = new LinkedHashMap<>();
485         actionProperties.put("order", Map.of("type", "integer", "description", "1-based action order."));
486         actionProperties.put("type", Map.of("type", "string", "description", "Canonical action type."));
487         actionProperties.put("title", Map.of("type", "string", "description", "Action title."));
488         actionProperties.put("tool_name", Map.of("type", "string", "description", "Canonical tool name."));
489         actionProperties.put("arguments", Map.of("type", "object", "description", "Canonical tool arguments."));
490         if (!additionalFieldName.isEmpty()) {
491             actionProperties.put(additionalFieldName, Map.of("type", "string", "description", "Unsupported field."));
492         }
493         return Map.of("type", "array", "description", "Structured follow-up actions.", "items", Map.of("type", "object", "properties", actionProperties));
494     }
