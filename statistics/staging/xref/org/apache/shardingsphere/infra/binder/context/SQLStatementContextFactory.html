71  /**
72   * SQL statement context factory.
73   */
74  @NoArgsConstructor(access = AccessLevel.PRIVATE)
75  public final class SQLStatementContextFactory {
76      
77      /**
78       * Create SQL statement context.
79       *
80       * @param metaData metadata
81       * @param databaseType database type
82       * @param sqlStatement SQL statement
83       * @param params SQL parameters
84       * @param currentDatabaseName current database name
85       * @return SQL statement context
86       */
87      public static SQLStatementContext newInstance(final ShardingSphereMetaData metaData,
88                                                    final DatabaseType databaseType, final SQLStatement sqlStatement, final List<Object> params, final String currentDatabaseName) {
89          if (sqlStatement.getAttributes().findAttribute(TableSQLStatementAttribute.class).isPresent()
90                  && DatabaseTypedSPILoader.findService(DialectCommonSQLStatementContextWarpProvider.class, databaseType)
91                          .map(optional -> optional.getNeedToWarpSQLStatementTypes().contains(sqlStatement.getClass())).orElse(false)) {
92              return new CommonSQLStatementContext(databaseType, sqlStatement);
93          }
94          if (sqlStatement instanceof DMLStatement) {
95              return getDMLStatementContext(metaData, databaseType, (DMLStatement) sqlStatement, params, currentDatabaseName);
96          }
97          if (sqlStatement instanceof DDLStatement) {
98              return getDDLStatementContext(metaData, databaseType, (DDLStatement) sqlStatement, params, currentDatabaseName);
99          }
100         if (sqlStatement instanceof DCLStatement) {
101             return getDCLStatementContext(databaseType, (DCLStatement) sqlStatement);
102         }
103         if (sqlStatement instanceof DALStatement) {
104             return getDALStatementContext(metaData, databaseType, (DALStatement) sqlStatement, params, currentDatabaseName);
105         }
106         return new CommonSQLStatementContext(databaseType, sqlStatement);
107     }
108     
109     private static SQLStatementContext getDMLStatementContext(final ShardingSphereMetaData metaData, final DatabaseType databaseType,
110                                                               final DMLStatement sqlStatement, final List<Object> params, final String currentDatabaseName) {
111         if (sqlStatement instanceof SelectStatement) {
112             return new SelectStatementContext(databaseType, (SelectStatement) sqlStatement, params, metaData, currentDatabaseName, Collections.emptyList());
113         }
114         if (sqlStatement instanceof UpdateStatement) {
115             return new UpdateStatementContext(databaseType, (UpdateStatement) sqlStatement);
116         }
117         if (sqlStatement instanceof DeleteStatement) {
118             return new DeleteStatementContext(databaseType, (DeleteStatement) sqlStatement);
119         }
120         if (sqlStatement instanceof InsertStatement) {
121             return new InsertStatementContext(databaseType, (InsertStatement) sqlStatement, params, metaData, currentDatabaseName);
122         }
123         return new CommonSQLStatementContext(databaseType, sqlStatement);
124     }
125     
126     private static SQLStatementContext getDDLStatementContext(final ShardingSphereMetaData metaData, final DatabaseType databaseType,
127                                                               final DDLStatement sqlStatement, final List<Object> params, final String currentDatabaseName) {
128         if (sqlStatement instanceof CreateTableStatement) {
129             return new CreateTableStatementContext(databaseType, (CreateTableStatement) sqlStatement);
130         }
131         if (sqlStatement instanceof AlterTableStatement) {
132             return new AlterTableStatementContext(databaseType, (AlterTableStatement) sqlStatement);
133         }
134         if (sqlStatement instanceof CreateIndexStatement) {
135             return new CreateIndexStatementContext(databaseType, (CreateIndexStatement) sqlStatement);
136         }
137         if (sqlStatement instanceof AlterIndexStatement) {
138             return new AlterIndexStatementContext(databaseType, (AlterIndexStatement) sqlStatement);
139         }
140         if (sqlStatement instanceof DropIndexStatement) {
141             return new DropIndexStatementContext(databaseType, (DropIndexStatement) sqlStatement);
142         }
143         if (sqlStatement instanceof CreateProcedureStatement) {
144             return new CreateProcedureStatementContext(databaseType, (CreateProcedureStatement) sqlStatement);
145         }
146         if (sqlStatement instanceof CreateViewStatement) {
147             return new CreateViewStatementContext(metaData, databaseType, params, (CreateViewStatement) sqlStatement, currentDatabaseName);
148         }
149         if (sqlStatement instanceof AlterViewStatement) {
150             return new AlterViewStatementContext(metaData, databaseType, params, (AlterViewStatement) sqlStatement, currentDatabaseName);
151         }
152         if (sqlStatement instanceof CursorStatement) {
153             return new CursorStatementContext(metaData, databaseType, params, (CursorStatement) sqlStatement, currentDatabaseName);
154         }
155         if (sqlStatement instanceof CloseStatement) {
156             return new CloseStatementContext(databaseType, (CloseStatement) sqlStatement);
157         }
158         if (sqlStatement instanceof MoveStatement) {
159             return new MoveStatementContext(databaseType, (MoveStatement) sqlStatement);
160         }
161         if (sqlStatement instanceof FetchStatement) {
162             return new FetchStatementContext(databaseType, (FetchStatement) sqlStatement);
163         }
164         return new CommonSQLStatementContext(databaseType, sqlStatement);
165     }
166     
167     private static SQLStatementContext getDCLStatementContext(final DatabaseType databaseType, final DCLStatement sqlStatement) {
168         return new CommonSQLStatementContext(databaseType, sqlStatement);
169     }
170     
171     private static SQLStatementContext getDALStatementContext(final ShardingSphereMetaData metaData, final DatabaseType databaseType,
172                                                               final DALStatement sqlStatement, final List<Object> params, final String currentDatabaseName) {
173         if (sqlStatement instanceof ExplainStatement) {
174             return new ExplainStatementContext(metaData, databaseType, (ExplainStatement) sqlStatement, params, currentDatabaseName);
175         }
176         return new CommonSQLStatementContext(databaseType, sqlStatement);
177     }
