18  package org.apache.shardingsphere.database.protocol.opengauss.packet.authentication;
19  
20  import org.apache.shardingsphere.infra.util.string.HexStringUtils;
21  import org.junit.jupiter.api.Test;
22  import org.mockito.internal.configuration.plugins.Plugins;
23  
24  import static org.hamcrest.MatcherAssert.assertThat;
25  import static org.hamcrest.Matchers.is;
26  import static org.junit.jupiter.api.Assertions.assertThrows;
27  
28  class OpenGaussMacCalculatorTest {
29      
30      @Test
31      void assertRequestServerMac() throws ReflectiveOperationException {
32          OpenGaussAuthenticationHexData authHexData = new OpenGaussAuthenticationHexData();
33          Plugins.getMemberAccessor().set(OpenGaussAuthenticationHexData.class.getDeclaredField("salt"), authHexData, "73616c74");
34          Plugins.getMemberAccessor().set(OpenGaussAuthenticationHexData.class.getDeclaredField("nonce"), authHexData, "6e6f6e6365");
35          String actual = OpenGaussMacCalculator.requestServerMac("password", authHexData, 4096);
36          assertThat(actual, is("788471142739dac2d5f7d6de58af1425aec57db7a4461e6ba78d2f93af132445"));
37      }
38      
39      @Test
40      void assertRequestClientMac() {
41          byte[] actual = OpenGaussMacCalculator.requestClientMac("password", "73616c74", 4096);
42          assertThat(HexStringUtils.toHexString(actual), is("8e795a7388652dea59019ec44c5dcc39e7c6a90e1fa1c260cbde4a2e57f1c933"));
43      }
44      
45      @Test
46      void assertRequestClientMacWithEmptySalt() {
47          assertThrows(IllegalArgumentException.class, () -> OpenGaussMacCalculator.requestClientMac("password", "", 4096));
48      }
49      
50      @Test
51      void assertCalculateClientMac() {
52          byte[] storedKey = OpenGaussMacCalculator.requestClientMac("password", "73616c74", 4096);
53          byte[] actual = OpenGaussMacCalculator.calculateClientMac("00112233445566778899aabbccddeeff00112233445566778899aabbccddeeff", "6e6f6e6365", storedKey);
54          assertThat(HexStringUtils.toHexString(actual), is("8503aa194430c80f5c156328ff53a90b1c3ddb9190ffae48eaba46e7e377aba7"));
55      }
56      
57      @Test
58      void assertCalculateClientMacWithDifferentLength() {
59          byte[] storedKey = OpenGaussMacCalculator.requestClientMac("password", "73616c74", 4096);
60          assertThrows(IllegalArgumentException.class, () -> OpenGaussMacCalculator.calculateClientMac("00", "6e6f6e6365", storedKey));
61      }
62  }
