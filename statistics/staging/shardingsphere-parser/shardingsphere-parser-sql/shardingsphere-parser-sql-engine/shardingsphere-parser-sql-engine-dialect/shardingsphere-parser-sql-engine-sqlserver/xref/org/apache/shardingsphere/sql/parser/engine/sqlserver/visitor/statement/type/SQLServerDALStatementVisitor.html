34  import java.util.Collections;
35  
36  /**
37   * DAL statement visitor for SQLServer.
38   */
39  public final class SQLServerDALStatementVisitor extends SQLServerStatementVisitor implements DALStatementVisitor {
40      
41      public SQLServerDALStatementVisitor(final DatabaseType databaseType) {
42          super(databaseType);
43      }
44      
45      @Override
46      public ASTNode visitExplain(final ExplainContext ctx) {
47          return new ExplainStatement(getDatabaseType(), (SQLStatement) visit(ctx.explainableStatement()));
48      }
49      
50      @Override
51      public ASTNode visitSet(final SetContext ctx) {
52          return new SetStatement(getDatabaseType(), Collections.singletonList((VariableAssignSegment) visit(ctx.setParameter())));
53      }
54      
55      @Override
56      public ASTNode visitSetParameter(final SetParameterContext ctx) {
57          int startIndex = ctx.start.getStartIndex();
58          int stopIndex = ctx.stop.getStopIndex();
59          if (null != ctx.QUOTED_IDENTIFIER()) {
60              VariableSegment variable = new VariableSegment(ctx.QUOTED_IDENTIFIER().getSymbol().getStartIndex(), ctx.QUOTED_IDENTIFIER().getSymbol().getStopIndex(), "QUOTED_IDENTIFIER");
61              return new VariableAssignSegment(startIndex, stopIndex, variable, null == ctx.ON() ? "OFF" : "ON");
62          }
63          VariableSegment variable = new VariableSegment(ctx.TEXTSIZE().getSymbol().getStartIndex(), ctx.TEXTSIZE().getSymbol().getStopIndex(), "TEXTSIZE");
64          return new VariableAssignSegment(startIndex, stopIndex, variable, ctx.numberLiterals().getText());
65      }
66      
67      @Override
68      public ASTNode visitExplainableStatement(final ExplainableStatementContext ctx) {
69          if (null != ctx.select()) {
70              return visit(ctx.select());
71          }
72          if (null != ctx.insert()) {
73              return visit(ctx.insert());
74          }
75          if (null != ctx.update()) {
76              return visit(ctx.update());
77          }
78          if (null != ctx.delete()) {
79              return visit(ctx.delete());
80          }
81          return visit(ctx.createTableAsSelectClause());
82      }
83  }
