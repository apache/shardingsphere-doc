122     private BinaryOperationExpression createBinaryOperationExpression(final ColumnSegment statusColumn, final ParameterMarkerExpressionSegment statusParameterExpression) {
123         ExpressionSegment leftExpression = new BinaryOperationExpression(0, 0, statusColumn, statusParameterExpression, "=", "status=?");
124         ColumnSegment countColumn = new ColumnSegment(0, 0, new IdentifierValue("count"));
125         BinaryOperationExpression rightExpression = createBinaryOperationExpression(statusColumn, statusParameterExpression, countColumn);
126         return new BinaryOperationExpression(0, 0, leftExpression, rightExpression, "OR", "status=? OR status=? AND count=?");
127     }
128     
129     private BinaryOperationExpression createBinaryOperationExpression(final ColumnSegment statusColumn,
130                                                                       final ParameterMarkerExpressionSegment statusParameterExpression, final ColumnSegment countColumn) {
131         ParameterMarkerExpressionSegment countParameterExpression = new ParameterMarkerExpressionSegment(0, 0, 1);
132         ExpressionSegment subLeftExpression = new BinaryOperationExpression(0, 0, statusColumn, statusParameterExpression, "=", "status=?");
133         ExpressionSegment subRightExpression = new BinaryOperationExpression(0, 0, countColumn, countParameterExpression, "=", "count=?");
134         return new BinaryOperationExpression(0, 0, subLeftExpression, subRightExpression, "AND", "status=? AND count=?");
135     }

229     private static Stream<Arguments> provideParameterMarkerExpressions() {
230         ParameterMarkerExpressionSegment direct = new ParameterMarkerExpressionSegment(0, 0, 1, ParameterMarkerType.DOLLAR);
231         ParameterMarkerExpressionSegment question = new ParameterMarkerExpressionSegment(1, 1, 0, ParameterMarkerType.QUESTION);
232         TypeCastExpression cast = new TypeCastExpression(1, 8, "?::int[]", question, "int[]");
233         CommonExpressionSegment wrappedCast = new CommonExpressionSegment(0, 12, "(?::int[])[1]", Arrays.asList(cast, new LiteralExpressionSegment(11, 11, 1)));
234         ParameterMarkerExpressionSegment firstDollar = new ParameterMarkerExpressionSegment(2, 3, 1, ParameterMarkerType.DOLLAR);
235         ParameterMarkerExpressionSegment secondDollar = new ParameterMarkerExpressionSegment(18, 19, 0, ParameterMarkerType.DOLLAR);
236         TypeCastExpression dollarCast = new TypeCastExpression(2, 10, "$2::int[]", firstDollar, "int[]");
237         CommonExpressionSegment inner = new CommonExpressionSegment(1, 14, "($2::int[])[1]", Arrays.asList(dollarCast, new LiteralExpressionSegment(13, 13, 1)));
238         CommonExpressionSegment nested = new CommonExpressionSegment(0, 15, "(($2::int[])[1])", Collections.singleton(inner));
239         RowExpression innerRow = new RowExpression(1, 19, "ROW($2, $1)");
240         innerRow.getItems().addAll(Arrays.asList(firstDollar, secondDollar));
241         RowExpression row = new RowExpression(0, 20, "ROW(?, ROW($2, $1))");
242         row.getItems().addAll(Arrays.asList(question, innerRow));
243         return Stream.of(
244                 Arguments.of("Direct type cast", Collections.singleton(new TypeCastExpression(0, 0, "$2::varchar", direct, "varchar")), Collections.singletonList(direct)),
245                 Arguments.of("Empty common expression", Collections.singleton(new CommonExpressionSegment(0, 0, "legacy", Collections.emptyList())), Collections.emptyList()),
246                 Arguments.of("Wrapped type cast", Collections.singleton(wrappedCast), Collections.singletonList(question)),
247                 Arguments.of("Nested common and sibling", Arrays.asList(nested, secondDollar), Arrays.asList(firstDollar, secondDollar)),
248                 Arguments.of("Nested row marker order", Collections.singleton(row), Arrays.asList(question, firstDollar, secondDollar)),
249                 Arguments.of("Unary marker", Collections.singleton(new UnaryOperationExpression(0, 3, firstDollar, "~", "~$2")), Collections.singletonList(firstDollar)),
250                 Arguments.of("Collate expression marker", Collections.singleton(new CollateExpression(0, 20, new LiteralExpressionSegment(18, 20, "C"), dollarCast)),
251                         Collections.singletonList(firstDollar)));
252     }

296     private static Stream<Arguments> provideColumnExpressions() {
297         ColumnSegment first = new ColumnSegment(1, 7, new IdentifierValue("foo_col"));
298         ColumnSegment second = new ColumnSegment(9, 15, new IdentifierValue("bar_col"));
299         RowExpression row = new RowExpression(0, 25, "ROW(foo_col, bar_col, foo_col)");
300         row.getItems().addAll(Arrays.asList(first, second, first));
301         UnaryOperationExpression unary = new UnaryOperationExpression(0, 7, first, "~", "~foo_col");
302         CollateExpression collate = new CollateExpression(0, 20, new LiteralExpressionSegment(10, 20, "foo_collation"), second);
303         CommonExpressionSegment common = new CommonExpressionSegment(0, 25, "foo_col[bar_col][foo_col]", Arrays.asList(first, second, first));
304         CommonExpressionSegment nested = new CommonExpressionSegment(0, 50, "foo_nested", Arrays.asList(unary, row, collate));
305         SelectStatement select = SelectStatement.builder().projections(new ProjectionsSegment(0, 20)).build();
306         select.getProjections().getProjections().add(new ColumnProjectionSegment(first));
307         SubqueryExpressionSegment subquery = new SubqueryExpressionSegment(new SubquerySegment(0, 20, select, "(SELECT foo_col)"));
308         CommonExpressionSegment commonSubquery = new CommonExpressionSegment(0, 30, "(SELECT foo_col)[bar_col]", Arrays.asList(subquery, second));
309         UnaryOperationExpression unarySubquery = new UnaryOperationExpression(0, 21, subquery, "~", "~(SELECT foo_col)");
310         CollateExpression collateSubquery = new CollateExpression(0, 40, new LiteralExpressionSegment(22, 34, "foo_collation"), subquery);
311         RowExpression rowSubquery = new RowExpression(0, 40, "ROW((SELECT foo_col), bar_col)");
312         rowSubquery.getItems().addAll(Arrays.asList(subquery, second));
313         return Stream.of(
314                 Arguments.of("Plain column", first, false, Collections.singletonList(first)),
315                 Arguments.of("Literal without columns", new LiteralExpressionSegment(0, 0, 1), false, Collections.emptyList()),
316                 Arguments.of("Common column order and duplicates", common, false, Arrays.asList(first, second, first)),
317                 Arguments.of("Empty common expression", new CommonExpressionSegment(0, 0, "foo_empty", Collections.emptyList()), false, Collections.emptyList()),
318                 Arguments.of("Nested expression columns", nested, false, Arrays.asList(first, first, second, first, second)),
319                 Arguments.of("Row column order and duplicates", row, false, Arrays.asList(first, second, first)),
320                 Arguments.of("Empty row", new RowExpression(0, 4, "ROW()"), false, Collections.emptyList()),
321                 Arguments.of("Unary column", unary, false, Collections.singletonList(first)),
322                 Arguments.of("Unary literal", new UnaryOperationExpression(0, 1, new LiteralExpressionSegment(1, 1, 1), "~", "~1"), false, Collections.emptyList()),
323                 Arguments.of("Collate operand column", collate, false, Collections.singletonList(second)),
324                 Arguments.of("Absent collate operand", new CollateExpression(0, 12, new LiteralExpressionSegment(0, 12, "foo_collation"), null), false, Collections.emptyList()),
325                 Arguments.of("Common excludes subquery columns", commonSubquery, false, Collections.singletonList(second)),
326                 Arguments.of("Common includes subquery columns", commonSubquery, true, Arrays.asList(first, second)),
327                 Arguments.of("Unary excludes subquery columns", unarySubquery, false, Collections.emptyList()),
328                 Arguments.of("Unary includes subquery columns", unarySubquery, true, Collections.singletonList(first)),
329                 Arguments.of("Collate excludes subquery columns", collateSubquery, false, Collections.emptyList()),
330                 Arguments.of("Collate includes subquery columns", collateSubquery, true, Collections.singletonList(first)),
331                 Arguments.of("Row excludes subquery columns", rowSubquery, false, Collections.singletonList(second)),
332                 Arguments.of("Row includes subquery columns", rowSubquery, true, Arrays.asList(first, second)));
333     }
