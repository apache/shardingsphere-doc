21  import lombok.NoArgsConstructor;

24  import org.apache.shardingsphere.infra.exception.kernel.metadata.DuplicateColumnException;
25  import org.apache.shardingsphere.infra.exception.kernel.metadata.DuplicateIndexException;
26  import org.apache.shardingsphere.infra.exception.kernel.metadata.IndexNotFoundException;
27  import org.apache.shardingsphere.infra.metadata.database.schema.model.ShardingSphereTable;
28  import org.apache.shardingsphere.sql.parser.statement.core.segment.ddl.column.ColumnDefinitionSegment;

40  import java.util.Collection;

42  /**
43   * Alter table metadata checker.
44   */
45  @NoArgsConstructor(access = AccessLevel.PRIVATE)
46  public final class AlterTableMetadataCheckUtils {
47      
48      /**
49       * Check alter table.
50       *
51       * @param alterTableStatement alter table statement
52       * @param table ShardingSphere table
53       */
54      public static void checkAlterTable(final AlterTableStatement alterTableStatement, final ShardingSphereTable table) {
55          validateAddColumns(table, alterTableStatement.getAddColumnDefinitions());
56          validateModifyColumns(table, alterTableStatement.getModifyColumnDefinitions());
57          validateChangeColumns(table, alterTableStatement.getChangeColumnDefinitions());
58          validateRenameColumns(table, alterTableStatement.getRenameColumnDefinitions());
59          validateDropColumns(table, alterTableStatement.getDropColumnDefinitions());
60          validateAddIndexes(table, alterTableStatement.getAddConstraintDefinitions());
61          validateDropIndexes(table, alterTableStatement.getDropIndexDefinitions());
62          validateRenameIndexes(table, alterTableStatement.getRenameIndexDefinitions());
63      }
64      
65      private static void validateAddColumns(final ShardingSphereTable table, final Collection<AddColumnDefinitionSegment> addColumns) {
66          for (AddColumnDefinitionSegment each : addColumns) {
67              for (ColumnDefinitionSegment columnDefinition : each.getColumnDefinitions()) {
68                  String addColumnName = columnDefinition.getColumnName().getIdentifier().getValue();
69                  ShardingSpherePreconditions.checkState(!containsColumn(table, addColumnName), () -> new DuplicateColumnException(addColumnName));
70              }
71          }
72      }
73      
74      private static void validateModifyColumns(final ShardingSphereTable table, final Collection<ModifyColumnDefinitionSegment> modifyColumns) {
75          for (ModifyColumnDefinitionSegment each : modifyColumns) {
76              String columnName = each.getColumnDefinition().getColumnName().getIdentifier().getValue();
77              ShardingSpherePreconditions.checkState(containsColumn(table, columnName), () -> new ColumnNotFoundException(columnName, table.getName()));
78          }
79      }
80      
81      private static void validateChangeColumns(final ShardingSphereTable table, final Collection<ChangeColumnDefinitionSegment> changeColumnDefinitions) {
82          for (ChangeColumnDefinitionSegment each : changeColumnDefinitions) {
83              String newColumnName = each.getColumnDefinition().getColumnName().getIdentifier().getValue();
84              ShardingSpherePreconditions.checkState(!containsColumn(table, newColumnName), () -> new DuplicateColumnException(newColumnName));
85              String oldColumnName = each.getPreviousColumn().getIdentifier().getValue();
86              ShardingSpherePreconditions.checkState(containsColumn(table, oldColumnName), () -> new ColumnNotFoundException(oldColumnName, table.getName()));
87          }
88      }
89      
90      private static void validateRenameColumns(final ShardingSphereTable table, final Collection<RenameColumnSegment> renameColumns) {
91          for (RenameColumnSegment each : renameColumns) {
92              String newColumnName = each.getColumnName().getIdentifier().getValue();
93              ShardingSpherePreconditions.checkState(!containsColumn(table, newColumnName), () -> new DuplicateColumnException(newColumnName));
94              String oldColumnName = each.getOldColumnName().getIdentifier().getValue();
95              ShardingSpherePreconditions.checkState(containsColumn(table, oldColumnName), () -> new ColumnNotFoundException(oldColumnName, table.getName()));
96          }
97      }
98      
99      private static void validateDropColumns(final ShardingSphereTable table, final Collection<DropColumnDefinitionSegment> dropColumns) {
100         for (DropColumnDefinitionSegment each : dropColumns) {
101             for (ColumnSegment column : each.getColumns()) {
102                 String columnName = column.getIdentifier().getValue();
103                 ShardingSpherePreconditions.checkState(containsColumn(table, columnName), () -> new ColumnNotFoundException(columnName, table.getName()));
104             }
105         }
106     }
107     
108     private static void validateAddIndexes(final ShardingSphereTable table, final Collection<AddConstraintDefinitionSegment> addConstraints) {
109         for (AddConstraintDefinitionSegment each : addConstraints) {
110             String indexName = each.getConstraintDefinition().getIndexName().map(optional -> optional.getIndexName().getIdentifier().getValue()).orElse("");
111             if (indexName.isEmpty()) {
112                 continue;
113             }
114             ShardingSpherePreconditions.checkState(!containsIndex(table, indexName), () -> new DuplicateIndexException(indexName));
115         }
116     }
117     
118     private static void validateDropIndexes(final ShardingSphereTable table, final Collection<DropIndexDefinitionSegment> dropIndexes) {
119         for (DropIndexDefinitionSegment each : dropIndexes) {
120             String indexName = each.getIndexSegment().getIndexName().getIdentifier().getValue();
121             ShardingSpherePreconditions.checkState(containsIndex(table, indexName), () -> new IndexNotFoundException(indexName));
122         }
123     }
124     
125     private static void validateRenameIndexes(final ShardingSphereTable table, final Collection<RenameIndexDefinitionSegment> renameIndexes) {
126         for (RenameIndexDefinitionSegment each : renameIndexes) {
127             String oldIndexName = each.getIndexSegment().getIndexName().getIdentifier().getValue();
128             ShardingSpherePreconditions.checkState(containsIndex(table, oldIndexName), () -> new IndexNotFoundException(oldIndexName));
129             String newIndexName = each.getRenameIndexSegment().getIndexName().getIdentifier().getValue();
130             ShardingSpherePreconditions.checkState(!containsIndex(table, newIndexName), () -> new DuplicateIndexException(newIndexName));
131         }
132     }
133     
134     private static boolean containsColumn(final ShardingSphereTable table, final String columnName) {
135         return table.containsColumn(columnName);
136     }
137     
138     private static boolean containsIndex(final ShardingSphereTable table, final String indexName) {
139         return table.containsIndex(indexName);
140     }
141 }
