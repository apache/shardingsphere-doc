52  /**
53   * Push down meta data refresh engine.
54   */
55  @RequiredArgsConstructor
56  public final class PushDownMetaDataRefreshEngine {
57      
58      private static final Collection<Class<?>> SUPPORTED_REFRESH_TYPES = new HashSet<>(Arrays.asList(CreateViewStatement.class, AlterViewStatement.class, DropViewStatement.class,
59              CreateIndexStatement.class, AlterIndexStatement.class, DropIndexStatement.class,
60              CreateSchemaStatement.class, AlterSchemaStatement.class, DropSchemaStatement.class,
61              CreateTableStatement.class, AlterTableStatement.class, DropTableStatement.class, RenameTableStatement.class));
62      
63      private final SQLStatementContext sqlStatementContext;
64      
65      /**
66       * Whether to need refresh meta data.
67       *
68       * @return is need refresh meta data or not
69       */
70      public boolean isNeedRefresh() {
71          return SUPPORTED_REFRESH_TYPES.contains(sqlStatementContext.getSqlStatement().getClass());
72      }
73      
74      /**
75       * Refresh push down meta data.
76       *
77       * @param metaDataManagerPersistService meta data manager persist service
78       * @param database database
79       * @param props configuration properties
80       * @param routeUnits route units
81       * @throws SQLException SQL exception
82       */
83      @SuppressWarnings("unchecked")
84      public void refresh(final MetaDataManagerPersistService metaDataManagerPersistService,
85                          final ShardingSphereDatabase database, final ConfigurationProperties props, final Collection<RouteUnit> routeUnits) throws SQLException {
86          Optional<PushDownMetaDataRefresher> refresher = findPushDownMetaDataRefresher(sqlStatementContext);
87          if (!refresher.isPresent()) {
88              return;
89          }
90          Collection<String> logicDataSourceNames = routeUnits.stream().map(each -> each.getDataSourceMapper().getLogicName()).collect(Collectors.toList());
91          String schemaName = SchemaRefreshUtils.getActualSchemaName(database, sqlStatementContext);
92          DatabaseType databaseType = routeUnits.stream().map(each -> database.getResourceMetaData().getStorageUnits().get(each.getDataSourceMapper().getActualName()))
93                  .filter(Objects::nonNull).findFirst().map(StorageUnit::getStorageType).orElseGet(() -> sqlStatementContext.getSqlStatement().getDatabaseType());
94          refresher.get().refresh(metaDataManagerPersistService, database,
95                  logicDataSourceNames.isEmpty() ? null : logicDataSourceNames.iterator().next(), schemaName, databaseType, sqlStatementContext.getSqlStatement(), props);
96      }
97      
98      @SuppressWarnings("rawtypes")
99      private Optional<PushDownMetaDataRefresher> findPushDownMetaDataRefresher(final SQLStatementContext sqlStatementContext) {
100         Optional<PushDownMetaDataRefresher> refresher = TypedSPILoader.findService(PushDownMetaDataRefresher.class, sqlStatementContext.getSqlStatement().getClass());
101         return refresher.isPresent() ? refresher : TypedSPILoader.findService(PushDownMetaDataRefresher.class, sqlStatementContext.getSqlStatement().getClass().getSuperclass());
102     }
103 }
