66  /**
67   * DDL statement assert for Doris.
68   */
69  @NoArgsConstructor(access = AccessLevel.PRIVATE)
70  public final class DorisDDLStatementAssert {
71      
72      /**
73       * Assert DDL statement is correct with expected parser result.
74       *
75       * @param assertContext assert context
76       * @param actual actual DDL statement
77       * @param expected expected DDL statement test case
78       */
79      public static void assertIs(final SQLCaseAssertContext assertContext, final DDLStatement actual, final SQLParserTestCase expected) {
80          if (actual instanceof DorisResumeJobStatement) {
81              DorisResumeJobStatementAssert.assertIs(assertContext, (DorisResumeJobStatement) actual, (DorisResumeJobStatementTestCase) expected);
82          } else if (actual instanceof DorisCreateFunctionStatement) {
83              DorisCreateFunctionStatementAssert.assertIs(assertContext, (DorisCreateFunctionStatement) actual, (CreateFunctionStatementTestCase) expected);
84          } else if (actual instanceof DorisAlterStoragePolicyStatement) {
85              DorisAlterStoragePolicyStatementAssert.assertIs(assertContext, (DorisAlterStoragePolicyStatement) actual, (DorisAlterStoragePolicyStatementTestCase) expected);
86          } else if (actual instanceof DorisDropFunctionStatement) {
87              DorisDropFunctionStatementAssert.assertIs(assertContext, (DorisDropFunctionStatement) actual, (DorisDropFunctionStatementTestCase) expected);
88          } else if (actual instanceof DorisResumeSyncJobStatement) {
89              DorisResumeSyncJobStatementAssert.assertIs(assertContext, (DorisResumeSyncJobStatement) actual, (DorisResumeSyncJobStatementTestCase) expected);
90          } else if (actual instanceof DorisPauseSyncJobStatement) {
91              DorisPauseSyncJobStatementAssert.assertIs(assertContext, (DorisPauseSyncJobStatement) actual, (DorisPauseSyncJobStatementTestCase) expected);
92          } else if (actual instanceof DorisCreateSyncJobStatement) {
93              DorisCreateSyncJobStatementAssert.assertIs(assertContext, (DorisCreateSyncJobStatement) actual, (DorisCreateSyncJobStatementTestCase) expected);
94          } else if (actual instanceof DorisStopSyncJobStatement) {
95              DorisStopSyncJobStatementAssert.assertIs(assertContext, (DorisStopSyncJobStatement) actual, (DorisStopSyncJobStatementTestCase) expected);
96          } else if (actual instanceof DorisCreateJobStatement) {
97              DorisCreateJobStatementAssert.assertIs(assertContext, (DorisCreateJobStatement) actual, (DorisCreateJobStatementTestCase) expected);
98          } else if (actual instanceof DorisAlterColocateGroupStatement) {
99              DorisAlterColocateGroupStatementAssert.assertIs(assertContext, (DorisAlterColocateGroupStatement) actual, (DorisAlterColocateGroupStatementTestCase) expected);
100         } else if (actual instanceof DorisAlterWorkloadGroupStatement) {
101             DorisAlterWorkloadGroupStatementAssert.assertIs(assertContext, (DorisAlterWorkloadGroupStatement) actual, (DorisAlterWorkloadGroupStatementTestCase) expected);
102         } else if (actual instanceof DorisPauseJobStatement) {
103             DorisPauseJobStatementAssert.assertIs(assertContext, (DorisPauseJobStatement) actual, (DorisPauseJobStatementTestCase) expected);
104         } else if (actual instanceof DorisDropJobStatement) {
105             DorisDropJobStatementAssert.assertIs(assertContext, (DorisDropJobStatement) actual, (DorisDropJobStatementTestCase) expected);
106         } else if (actual instanceof DorisAlterJobStatement) {
107             DorisAlterJobStatementAssert.assertIs(assertContext, (DorisAlterJobStatement) actual, (DorisAlterJobStatementTestCase) expected);
108         } else if (actual instanceof DorisCreateStreamingJobStatement) {
109             DorisCreateStreamingJobStatementAssert.assertIs(assertContext, (DorisCreateStreamingJobStatement) actual, (DorisCreateStreamingJobStatementTestCase) expected);
110         } else if (actual instanceof DorisCancelTaskStatement) {
111             DorisCancelTaskStatementAssert.assertIs(assertContext, (DorisCancelTaskStatement) actual, (DorisCancelTaskStatementTestCase) expected);
112         } else if (actual instanceof DorisCancelMaterializedViewTaskStatement) {
113             DorisCancelMaterializedViewTaskStatementAssert.assertIs(assertContext, (DorisCancelMaterializedViewTaskStatement) actual, (DorisCancelMaterializedViewTaskStatementTestCase) expected);
114         } else if (actual instanceof DorisRefreshMaterializedViewStatement) {
115             DorisRefreshMaterializedViewStatementAssert.assertIs(assertContext, (DorisRefreshMaterializedViewStatement) actual, (DorisRefreshMaterializedViewStatementTestCase) expected);
116         }
117     }
