44  /**
45   * Create Batch command executor for Firebird.
46   */
47  @RequiredArgsConstructor
48  public final class FirebirdBatchCreateCommandExecutor implements CommandExecutor {
49      
50      private static final long DEFAULT_BUFFER_SIZE = 16L * 1024 * 1024;
51      
52      private static final long MAX_BUFFER_SIZE = 256L * 1024 * 1024;
53      
54      private static final int BATCH_VERSION_1 = 1;
55      
56      private static final int TAG_MULTIERROR = 1;
57      
58      private static final int TAG_RECORD_COUNTS = 2;
59      
60      private static final int TAG_BUFFER_BYTES_SIZE = 3;
61      
62      private static final int TAG_BLOB_POLICY = 4;
63      
64      private static final int WIDE_CLUMPLET_LENGTH_SIZE = 4;
65      
66      private static final int INTEGER_VALUE_LENGTH = 4;
67      
68      private final FirebirdBatchCreateCommandPacket packet;
69      
70      private final ConnectionSession connectionSession;
71      
72      @Override
73      public Collection<DatabasePacket> execute() {
74          int statementId = packet.getStatementHandle();
75          FirebirdServerPreparedStatement preparedStatement = connectionSession.getServerPreparedStatementRegistry().getPreparedStatement(statementId);
76          ShardingSpherePreconditions.checkNotNull(preparedStatement, () -> new InvalidStatementHandleException(statementId));
77          int connectionId = connectionSession.getConnectionId();
78          ShardingSpherePreconditions.checkState(null == FirebirdBatchStatementManager.getInstance().getBatchStatement(connectionId, statementId), () -> new BatchAlreadyOpenedException(statementId));
79          if (preparedStatement.getSqlStatementContext().getSqlStatement() instanceof SelectStatement) {
80              ByteBuf batchBlr = packet.getBatchBlr();
81              validateMessageLength(FirebirdParseBatchBlr.parseForFraming(batchBlr, batchBlr.readableBytes()).getMessageLength());
82              FirebirdBatchRegistry.getInstance().registerRejectedBatchFormat(connectionId, statementId, batchBlr);
83              BatchParameters.validateWireParameters(packet.getBatchParametersBuffer());
84              throw new InvalidBatchStatementTypeException();
85          }
86          ByteBuf batchBlr = packet.getBatchBlr();
87          FirebirdParseBatchBlr messageFormat = FirebirdParseBatchBlr.parse(batchBlr, batchBlr.readableBytes());
88          ShardingSpherePreconditions.checkNotEmpty(messageFormat.getFields(), () -> new BatchParametersRequiredException(statementId));
89          validateMessageLength(messageFormat.getMessageLength());
90          BatchParameters batchParameters = BatchParameters.parse(packet.getBatchParametersBuffer());
91          FirebirdBatchStatementManager.getInstance().registerBatchStatement(
92                  connectionId, statementId, messageFormat.getFields(), batchParameters.getBufferSize(), batchParameters.isRecordCounts(), batchParameters.isMultiError());
93          return Collections.singleton(new FirebirdGenericResponsePacket().setHandle(statementId));
94      }
95      
96      private void validateMessageLength(final int expectedMessageLength) {
97          ShardingSpherePreconditions.checkState(expectedMessageLength == packet.getBatchMessageLength(),
98                  () -> new InvalidBatchMessageFormatException(String.format("invalid message length: computed %d from BLR but client sent %d", expectedMessageLength, packet.getBatchMessageLength())));
99      }
100     
101     @RequiredArgsConstructor
102     @Getter
103     static final class BatchParameters {
104         
105         private final int version;
106         
107         private final long bufferSize;
108         
109         private final boolean recordCounts;
110         
111         private final boolean multiError;
112         
113         static BatchParameters parse(final ByteBuf batchParametersBuffer) {
114             int version = validateVersion(batchParametersBuffer);
115             if (!batchParametersBuffer.isReadable()) {
116                 return new BatchParameters(version, DEFAULT_BUFFER_SIZE, false, false);
117             }
118             ByteBuf reader = batchParametersBuffer.duplicate();
119             reader.skipBytes(1);
120             long bufferSize = DEFAULT_BUFFER_SIZE;
121             boolean recordCounts = false;
122             boolean multiError = false;
123             while (reader.isReadable()) {
124                 ensureClumpletHeaderReadable(reader);
125                 int tag = reader.readUnsignedByte();
126                 int valueLength = reader.readIntLE();
127                 ensureClumpletValueReadable(reader, tag, valueLength);
128                 if (TAG_MULTIERROR == tag) {
129                     multiError = 0 != readIntegerValue(reader, tag, valueLength);
130                 } else if (TAG_RECORD_COUNTS == tag) {
131                     recordCounts = 0 != readIntegerValue(reader, tag, valueLength);
132                 } else if (TAG_BUFFER_BYTES_SIZE == tag) {
133                     bufferSize = getBufferSize(readIntegerValue(reader, tag, valueLength));
134                 } else if (TAG_BLOB_POLICY == tag) {
135                     // TODO Support BLOB policy after implementing the Firebird batch BLOB subprotocol.
136                     throw new DatabaseProtocolException("BLOB policy is not supported in Firebird batch operations");
137                 } else {
138                     reader.skipBytes(valueLength);
139                 }
140             }
141             return new BatchParameters(version, bufferSize, recordCounts, multiError);
142         }
143         
144         private static int validateVersion(final ByteBuf batchParamsBuffer) {
145             int result = batchParamsBuffer.isReadable() ? batchParamsBuffer.getUnsignedByte(batchParamsBuffer.readerIndex()) : BATCH_VERSION_1;
146             ShardingSpherePreconditions.checkState(BATCH_VERSION_1 == result, () -> new InvalidBatchParameterVersionException(result, BATCH_VERSION_1));
147             return result;
148         }
149         
150         private static long getBufferSize(final int requestedBufferSize) {
151             long result = Integer.toUnsignedLong(requestedBufferSize);
152             return 0L == result ? MAX_BUFFER_SIZE : Math.min(result, MAX_BUFFER_SIZE);
153         }
154         
155         private static void ensureClumpletHeaderReadable(final ByteBuf reader) {
156             ShardingSpherePreconditions.checkState(reader.readableBytes() >= 1 + WIDE_CLUMPLET_LENGTH_SIZE, () -> new DatabaseProtocolException("Invalid batch parameters buffer"));
157         }
158         
159         private static void ensureClumpletValueReadable(final ByteBuf reader, final int tag, final int valueLength) {
160             ShardingSpherePreconditions.checkState(valueLength >= 0 && valueLength <= reader.readableBytes(), () -> new DatabaseProtocolException("Invalid batch parameter length for tag %d: %d", tag,
161                     valueLength));
162         }
163         
164         private static int readIntegerValue(final ByteBuf reader, final int tag, final int valueLength) {
165             ShardingSpherePreconditions.checkState(INTEGER_VALUE_LENGTH == valueLength, () -> new DatabaseProtocolException("Invalid batch parameter integer length for tag %d: %d", tag, valueLength));
166             return reader.readIntLE();
167         }
168         
169         private static void validateWireParameters(final ByteBuf batchParamsBuffer) {
170             validateVersion(batchParamsBuffer);
171             validateWireParameter(batchParamsBuffer, TAG_RECORD_COUNTS);
172             validateWireParameter(batchParamsBuffer, TAG_BLOB_POLICY);
173         }
174         
175         private static void validateWireParameter(final ByteBuf batchParamsBuffer, final int expectedTag) {
176             if (!batchParamsBuffer.isReadable()) {
177                 return;
178             }
179             ByteBuf reader = batchParamsBuffer.duplicate();
180             reader.skipBytes(1);
181             while (reader.isReadable()) {
182                 ensureClumpletHeaderReadable(reader);
183                 int tag = reader.readUnsignedByte();
184                 int valueLength = reader.readIntLE();
185                 ensureClumpletValueReadable(reader, tag, valueLength);
186                 if (expectedTag == tag) {
187                     ShardingSpherePreconditions.checkState(valueLength <= INTEGER_VALUE_LENGTH, () -> new DatabaseProtocolException("Invalid batch parameter integer length for tag %d: %d", tag,
188                             valueLength));
189                     return;
190                 }
191                 reader.skipBytes(valueLength);
192             }
193         }
194     }
