34  import java.sql.Connection;
35  import java.sql.ResultSet;
36  import java.sql.ResultSetMetaData;
37  import java.sql.SQLException;
38  import java.sql.Statement;
39  import java.util.LinkedHashMap;
40  import java.util.LinkedList;
41  import java.util.List;
42  import java.util.Locale;
43  import java.util.Map;
44  import java.util.Optional;

46  /**
47   * Direct query service for Proxy-backed workflow reads.
48   */
49  @RequiredArgsConstructor
50  public final class WorkflowProxyQueryService implements MCPFeatureQueryFacade {
51      
52      private final MCPSessionManager sessionManager;
53      
54      private final MCPDatabaseCapabilityProvider databaseCapabilityProvider;
55      
56      @Override
57      public List<Map<String, Object>> query(final String databaseName, final String sql) {
58          String actualDatabaseName = WorkflowSQLUtils.normalizeIdentifier(databaseName);
59          try (
60                  Connection connection = openConnection(actualDatabaseName);
61                  Statement statement = connection.createStatement();
62                  ResultSet resultSet = statement.executeQuery(sql)) {
63              return extractRows(resultSet);
64          } catch (final SQLException ex) {
65              throw createQueryFailedException(actualDatabaseName, ex);
66          }
67      }
68      
69      @Override
70      public List<Map<String, Object>> queryWithAnyDatabase(final String sql) {
71          String databaseName = sessionManager.getTransactionResourceManager().getRuntimeDatabases().keySet().stream().findFirst()
72                  .orElseThrow(() -> new MCPUnavailableException("No runtime database is configured."));
73          return query(databaseName, sql);
74      }
75      
76      @Override
77      public void checkDatabaseCapability(final String databaseName) {
78          getDatabaseCapability(databaseName);
79      }
80      
81      @Override
82      public boolean isSameIdentifier(final String databaseName, final IdentifierScope identifierScope, final String identifier, final String existingIdentifier) {
83          return WorkflowSQLUtils.isSameIdentifier(getDatabaseCapability(databaseName).getIdentifierContext(), identifierScope, identifier, existingIdentifier);
84      }
85      
86      private MCPDatabaseQueryFailedException createQueryFailedException(final String databaseName, final SQLException cause) {
87          Optional<MCPDatabaseCapability> databaseCapability = databaseCapabilityProvider.provide(databaseName);
88          MCPJDBCErrorCategory category = databaseCapability.isPresent()
89                  ? MCPJDBCExceptionClassifier.classify(databaseCapability.get().getDatabaseType(), cause)
90                  : MCPJDBCExceptionClassifier.classify(cause);
91          return new MCPDatabaseQueryFailedException(category, cause);
92      }
93      
94      private Connection openConnection(final String databaseName) throws SQLException {
95          RuntimeDatabaseConfiguration runtimeDatabaseConfig = sessionManager.getTransactionResourceManager().getRuntimeDatabases().get(databaseName);
96          if (null == runtimeDatabaseConfig) {
97              throw new MCPUnavailableException(String.format("Database `%s` is not configured.", databaseName));
98          }
99          return runtimeDatabaseConfig.openConnection(databaseName);
100     }
101     
102     private MCPDatabaseCapability getDatabaseCapability(final String databaseName) {
103         return databaseCapabilityProvider.provide(WorkflowSQLUtils.normalizeIdentifier(databaseName)).orElseThrow(DatabaseCapabilityNotFoundException::new);
104     }
105     
106     private List<Map<String, Object>> extractRows(final ResultSet resultSet) throws SQLException {
107         ResultSetMetaData resultSetMetaData = resultSet.getMetaData();
108         List<Map<String, Object>> result = new LinkedList<>();
109         while (resultSet.next()) {
110             Map<String, Object> row = new LinkedHashMap<>(resultSetMetaData.getColumnCount(), 1F);
111             for (int index = 1; index <= resultSetMetaData.getColumnCount(); index++) {
112                 row.put(resultSetMetaData.getColumnLabel(index).toLowerCase(Locale.ENGLISH), resultSet.getObject(index));
113             }
114             result.add(row);
115         }
116         return result;
117     }
118     
119 }
