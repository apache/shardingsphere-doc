42  /**
43   * Handler for runtime status resource URI.
44   */
45  public final class RuntimeStatusHandler implements MCPResourceHandler<MCPFeatureRequestContext> {
46      
47      private static final String URI_PATTERN = "shardingsphere://runtime";
48      
49      @Override
50      public Class<MCPFeatureRequestContext> getContextType() {
51          return MCPFeatureRequestContext.class;
52      }
53      
54      @Override
55      public String getResourceUriTemplate() {
56          return URI_PATTERN;
57      }
58      
59      @Override
60      public MCPSuccessPayload handle(final MCPFeatureRequestContext handlerContext, final MCPResourceURIVariables uriVariables) {
61          List<RuntimeDatabaseProfile> databases = handlerContext.getMetadataQueryFacade().queryDatabases();
62          boolean hasConfiguredDatabase = !databases.isEmpty();
63          MCPTransportType activeTransport = handlerContext.getActiveTransport();
64          String activeTransportName = activeTransport.name().toLowerCase(Locale.ENGLISH);
65          Map<String, Object> result = new LinkedHashMap<>(12, 1F);
66          result.put("response_mode", MCPResponseMode.RUNTIME);
67          result.put(MCPPayloadFieldNames.SUMMARY, createSummary(hasConfiguredDatabase, databases.size()));
68          result.put("status", hasConfiguredDatabase ? "available" : "configuration_required");
69          result.put("active_transport", activeTransportName);
70          result.put("transport_security_summary", createTransportSecuritySummary(activeTransport));
71          result.put("configured_database_count", databases.size());
72          result.put("databases", databases.stream().map(each -> createDatabaseStatus(handlerContext, each)).toList());
73          result.put("readiness", createReadiness(hasConfiguredDatabase));
74          result.put("runtime_protection", MCPRuntimeProtectionPolicy.createRuntimeProtectionPayload());
75          result.put("diagnostics", createDiagnostics(hasConfiguredDatabase));
76          result.put(MCPPayloadFieldNames.RESOURCES_TO_READ, createResourcesToRead(hasConfiguredDatabase));
77          result.put(MCPPayloadFieldNames.NEXT_ACTIONS, createNextActions(hasConfiguredDatabase));
78          return new MCPMapPayload(result);
79      }
80      
81      private String createSummary(final boolean hasConfiguredDatabase, final int configuredDatabaseCount) {
82          return hasConfiguredDatabase
83                  ? String.format("Runtime is ready with %d configured logical database(s).", configuredDatabaseCount)
84                  : "Runtime requires at least one configured logical database before metadata discovery or SQL execution.";
85      }
86      
87      private Map<String, Object> createTransportSecuritySummary(final MCPTransportType activeTransport) {
88          Map<String, Object> result = new LinkedHashMap<>(3, 1F);
89          result.put("authentication", MCPTransportType.HTTP == activeTransport ? "not_enabled_by_mcp_transport" : "local_client_process");
90          result.put("recommended_exposure", MCPTransportType.HTTP == activeTransport ? "loopback_or_trusted_gateway" : "local_stdio_session");
91          result.put("model_action", "Do not request or echo JDBC URLs, credentials, raw environment variables, or stack traces.");
92          return result;
93      }
94      
95      private Map<String, Object> createReadiness(final boolean hasConfiguredDatabase) {
96          Map<String, Object> result = new LinkedHashMap<>(hasConfiguredDatabase ? 3 : 4, 1F);
97          result.put("ready", hasConfiguredDatabase);
98          result.put("token_safe", true);
99          result.put("resource_uri", URI_PATTERN);
100         if (hasConfiguredDatabase) {
101             return result;
102         }
103         result.put(MCPPayloadFieldNames.REASON, "No runtime databases are configured.");
104         return result;
105     }
106     
107     private Map<String, Object> createDiagnostics(final boolean hasConfiguredDatabase) {
108         Map<String, Object> result = new LinkedHashMap<>(4, 1F);
109         result.put("current_category", hasConfiguredDatabase ? "ready" : RuntimeDatabaseConnectionException.CATEGORY_INVALID_CONFIGURATION);
110         result.put("safe_categories", createSafeRuntimeCategories());
111         result.put("operator_next_actions", createDiagnosticOperatorActions());
112         result.put("secret_policy", "Expose only categories and operator actions; never expose JDBC URLs, credentials, raw environment variables, or stack traces.");
113         return result;
114     }
115     
116     private List<String> createSafeRuntimeCategories() {
117         return List.of(
118                 RuntimeDatabaseConnectionException.CATEGORY_MISSING_JDBC_DRIVER,
119                 RuntimeDatabaseConnectionException.CATEGORY_AUTHENTICATION_FAILED,
120                 RuntimeDatabaseConnectionException.CATEGORY_AUTHORIZATION_FAILED,
121                 RuntimeDatabaseConnectionException.CATEGORY_CONNECTION_TIMEOUT,
122                 RuntimeDatabaseConnectionException.CATEGORY_INVALID_CONFIGURATION,
123                 RuntimeDatabaseConnectionException.CATEGORY_DATABASE_UNAVAILABLE,
124                 RuntimeDatabaseConnectionException.CATEGORY_CONNECTION_FAILED,
125                 RuntimeDatabaseConnectionException.CATEGORY_DATABASE_NOT_VISIBLE);
126     }
127     
128     private List<Map<String, Object>> createDiagnosticOperatorActions() {
129         return List.of(
130                 createDiagnosticOperatorAction(RuntimeDatabaseConnectionException.CATEGORY_MISSING_JDBC_DRIVER, "Install the configured runtime database JDBC driver."),
131                 createDiagnosticOperatorAction(RuntimeDatabaseConnectionException.CATEGORY_AUTHENTICATION_FAILED, "Check runtime database credentials outside MCP."),
132                 createDiagnosticOperatorAction(RuntimeDatabaseConnectionException.CATEGORY_AUTHORIZATION_FAILED, "Check metadata and SQL privileges for the configured runtime database account."),
133                 createDiagnosticOperatorAction(RuntimeDatabaseConnectionException.CATEGORY_CONNECTION_TIMEOUT, "Check database reachability and timeout settings."),
134                 createDiagnosticOperatorAction(RuntimeDatabaseConnectionException.CATEGORY_INVALID_CONFIGURATION, "Fix runtimeDatabases JDBC URL, driver, or binding configuration."),
135                 createDiagnosticOperatorAction(RuntimeDatabaseConnectionException.CATEGORY_DATABASE_UNAVAILABLE, "Check database service availability and network access."),
136                 createDiagnosticOperatorAction(RuntimeDatabaseConnectionException.CATEGORY_CONNECTION_FAILED, "Inspect runtime database connection settings outside MCP."),
137                 createDiagnosticOperatorAction(RuntimeDatabaseConnectionException.CATEGORY_DATABASE_NOT_VISIBLE, "Check the configured logical database name and account visibility outside MCP."));
138     }
139     
140     private Map<String, Object> createDiagnosticOperatorAction(final String category, final String operatorAction) {
141         Map<String, Object> result = new LinkedHashMap<>(3, 1F);
142         result.put("category", category);
143         result.put("operator_action", operatorAction);
144         result.put("secret_safe", true);
145         return result;
146     }
147     
148     private List<Map<String, Object>> createResourcesToRead(final boolean hasConfiguredDatabase) {
149         Map<String, Object> capabilitiesResource = MCPResourceHintUtils.create("shardingsphere://capabilities", "capability", "read_first",
150                 "Read full MCP capabilities before choosing tools.", MCPPayloadFieldNames.RESOURCES_TO_READ);
151         if (!hasConfiguredDatabase) {
152             return List.of(capabilitiesResource);
153         }
154         return List.of(capabilitiesResource, MCPResourceHintUtils.create("shardingsphere://databases", "logical-database", "read_first",
155                 "Read logical databases before choosing a database scope.", MCPPayloadFieldNames.RESOURCES_TO_READ));
156     }
157     
158     private List<Map<String, Object>> createNextActions(final boolean hasConfiguredDatabase) {
159         if (hasConfiguredDatabase) {
160             return List.of(MCPNextActionUtils.readResource("shardingsphere://databases", "Read logical databases before choosing a database scope."));
161         }
162         Map<String, Object> capabilityAction = MCPNextActionUtils.readResource("shardingsphere://capabilities", "Read the full capability catalog before choosing tools.");
163         return MCPNextActionUtils.ordered(capabilityAction, MCPNextActionUtils.dependsOn(MCPNextActionUtils.askUser(
164                 "Ask the operator to configure at least one runtimeDatabases entry before metadata discovery or SQL execution.", List.of("runtimeDatabases")), 1));
165     }
166     
167     private Map<String, Object> createDatabaseStatus(final MCPFeatureRequestContext handlerContext, final RuntimeDatabaseProfile database) {
168         Optional<MCPDatabaseCapability> capability = handlerContext.getCapabilityFacade().provide(database.getDatabase());
169         Map<String, Object> result = new LinkedHashMap<>(4, 1F);
170         result.put("database", database.getDatabase());
171         result.put("database_type", database.getDatabaseType());
172         result.put("capabilities", capability.map(this::createCapabilityStatus).orElseGet(this::createUnavailableCapabilityStatus));
173         result.put(MCPPayloadFieldNames.RESOURCE, MCPResourceHintUtils.create(String.format("shardingsphere://databases/%s", MCPUriPathSegmentUtils.encodePathSegment(database.getDatabase())),
174                 "logical-database", "inspect_detail", "Read this logical database resource for metadata details.", "databases"));
175         return result;
176     }
177     
178     private Map<String, Object> createCapabilityStatus(final MCPDatabaseCapability capability) {
179         Map<String, Object> result = new LinkedHashMap<>(4, 1F);
180         result.put("available", true);
181         result.put("supports_explain", capability.supportsExplain());
182         result.put("supported_statement_classes", capability.getSupportedStatementClasses().stream().map(Enum::name).toList());
183         result.put("supported_metadata_object_types", capability.getSupportedMetadataObjectTypes().stream().map(Enum::name).toList());
184         return result;
185     }
186     
187     private Map<String, Object> createUnavailableCapabilityStatus() {
188         Map<String, Object> result = new LinkedHashMap<>(1, 1F);
189         result.put("available", false);
190         return result;
191     }
