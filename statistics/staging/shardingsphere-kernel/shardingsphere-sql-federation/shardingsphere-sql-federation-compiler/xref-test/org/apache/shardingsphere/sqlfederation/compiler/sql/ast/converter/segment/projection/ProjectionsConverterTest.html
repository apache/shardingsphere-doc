51  @ExtendWith(AutoMockExtension.class)
52  @StaticMockSettings({
53          ColumnProjectionConverter.class, ExpressionProjectionConverter.class, ShorthandProjectionConverter.class,
54          SubqueryProjectionConverter.class, AggregationProjectionConverter.class, ParameterMarkerExpressionConverter.class
55  })
56  class ProjectionsConverterTest {
57      
58      @Test
59      void assertConvertCollectsAllSupportedProjections() {
60          ProjectionsSegment projectionsSegment = new ProjectionsSegment(0, 0);
61          SqlNode expectedColumnNode = mock(SqlNode.class);
62          ColumnProjectionSegment columnProjectionSegment = mock(ColumnProjectionSegment.class);
63          when(ColumnProjectionConverter.convert(columnProjectionSegment)).thenReturn(expectedColumnNode);
64          projectionsSegment.getProjections().add(columnProjectionSegment);
65          SqlNode expectedExpressionNode = mock(SqlNode.class);
66          ExpressionProjectionSegment expressionProjectionSegment = mock(ExpressionProjectionSegment.class);
67          when(ExpressionProjectionConverter.convert(expressionProjectionSegment)).thenReturn(Optional.of(expectedExpressionNode));
68          projectionsSegment.getProjections().add(expressionProjectionSegment);
69          SqlNode expectedShorthandNode = mock(SqlNode.class);
70          ShorthandProjectionSegment shorthandProjectionSegment = mock(ShorthandProjectionSegment.class);
71          when(ShorthandProjectionConverter.convert(shorthandProjectionSegment)).thenReturn(Optional.of(expectedShorthandNode));
72          projectionsSegment.getProjections().add(shorthandProjectionSegment);
73          SqlNode expectedSubqueryNode = mock(SqlNode.class);
74          SubqueryProjectionSegment subqueryProjectionSegment = mock(SubqueryProjectionSegment.class);
75          when(SubqueryProjectionConverter.convert(subqueryProjectionSegment)).thenReturn(Optional.of(expectedSubqueryNode));
76          projectionsSegment.getProjections().add(subqueryProjectionSegment);
77          SqlNode expectedAggregationNode = mock(SqlNode.class);
78          AggregationProjectionSegment aggregationProjectionSegment = mock(AggregationProjectionSegment.class);
79          when(AggregationProjectionConverter.convert(aggregationProjectionSegment)).thenReturn(Optional.of(expectedAggregationNode));
80          projectionsSegment.getProjections().add(aggregationProjectionSegment);
81          SqlDynamicParam expectedParameterNode = mock(SqlDynamicParam.class);
82          ParameterMarkerExpressionSegment parameterMarkerExpressionSegment = new ParameterMarkerExpressionSegment(0, 0, 0);
83          when(ParameterMarkerExpressionConverter.convert(parameterMarkerExpressionSegment)).thenReturn(expectedParameterNode);
84          projectionsSegment.getProjections().add(parameterMarkerExpressionSegment);
85          SqlNodeList actual = ProjectionsConverter.convert(projectionsSegment).orElse(null);
86          assertNotNull(actual);
87          assertThat(actual.size(), is(6));
88          assertThat(actual.get(0), is(expectedColumnNode));
89          assertThat(actual.get(1), is(expectedExpressionNode));
90          assertThat(actual.get(2), is(expectedShorthandNode));
91          assertThat(actual.get(3), is(expectedSubqueryNode));
92          assertThat(actual.get(4), is(expectedAggregationNode));
93          assertThat(actual.get(5), is(expectedParameterNode));
94      }
95      
96      @Test
97      void assertConvertSkipsUnsupportedProjection() {
98          ProjectionsSegment projectionsSegment = new ProjectionsSegment(0, 0);
99          ProjectionSegment unsupportedProjection = mock(ProjectionSegment.class);
100         projectionsSegment.getProjections().add(unsupportedProjection);
101         SqlNodeList actual = ProjectionsConverter.convert(projectionsSegment).orElse(null);
102         assertNotNull(actual);
103         assertFalse(actual.iterator().hasNext());
104     }
