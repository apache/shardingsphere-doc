51  /**
52   * Schema meta data utility class.
53   */
54  @NoArgsConstructor(access = AccessLevel.PRIVATE)
55  public final class SchemaMetaDataUtils {
56      
57      /**
58       * Get meta data loader materials.
59       *
60       * @param tableNames table name collection
61       * @param protocolType protocol type
62       * @param material material
63       * @return meta data loader materials
64       */
65      public static Collection<MetaDataLoaderMaterial> getMetaDataLoaderMaterials(final Collection<String> tableNames, final DatabaseType protocolType,
66                                                                                  final GenericSchemaBuilderMaterial material) {
67          Map<String, Collection<String>> dataSourceTableGroups = new LinkedHashMap<>();
68          Collection<DatabaseType> unsupportedThreeTierStorageStructureDatabaseTypes = getUnsupportedThreeTierStorageStructureDatabaseTypes(material.getStorageUnits().values());
69          DataNodes dataNodes = new DataNodes(material.getRules());
70          boolean checkMetaDataEnable = material.getProps().getValue(ConfigurationPropertyKey.CHECK_TABLE_METADATA_ENABLED);
71          for (String each : tableNames) {
72              checkDataSourceTypeIncludeInstanceAndSetDatabaseTableMap(unsupportedThreeTierStorageStructureDatabaseTypes, dataNodes, each);
73              boolean tableNameLoadedFromStorage = dataNodes.isDataNodeTableNameLoadedFromStorage(each);
74              if (checkMetaDataEnable) {
75                  addAllActualTableDataNode(material, dataSourceTableGroups, dataNodes, each, protocolType, tableNameLoadedFromStorage);
76              } else {
77                  addOneActualTableDataNode(material, dataSourceTableGroups, dataNodes, each, protocolType, tableNameLoadedFromStorage);
78              }
79          }
80          Collection<MetaDataLoaderMaterial> result = new LinkedList<>();
81          int loadTableMetadataBatchSize = material.getProps().getValue(ConfigurationPropertyKey.LOAD_TABLE_METADATA_BATCH_SIZE);
82          for (Entry<String, Collection<String>> entry : dataSourceTableGroups.entrySet()) {
83              DatabaseType storageType = material.getStorageUnits().get(entry.getKey()).getStorageType();
84              String defaultSchemaName = getDefaultSchemaName(material, entry.getKey(), storageType);
85              result.addAll(buildMaterials(material, entry.getKey(), entry.getValue(), storageType, defaultSchemaName, loadTableMetadataBatchSize));
86          }
87          return result;
88      }
89      
90      private static String getDefaultSchemaName(final GenericSchemaBuilderMaterial material, final String dataSourceName, final DatabaseType storageType) {
91          DialectSchemaOption schemaOption = new DatabaseTypeRegistry(storageType).getDialectDatabaseMetaData().getSchemaOption();
92          return schemaOption.isSchemaAvailable() || DialectSchemaSemantics.DATABASE_AS_SCHEMA == schemaOption.getSchemaSemantics()
93                  ? DefaultSchemaNameResolver.resolveStorage(storageType, getDataSource(material, dataSourceName), material.getDefaultSchemaName())
94                  : material.getDefaultSchemaName();
95      }
96      
97      private static Collection<MetaDataLoaderMaterial> buildMaterials(final GenericSchemaBuilderMaterial material, final String dataSourceName, final Collection<String> actualTableNames,
98                                                                       final DatabaseType storageType, final String defaultSchemaName,
99                                                                       final int loadTableMetadataBatchSize) {
100         Collection<MetaDataLoaderMaterial> result = new LinkedList<>();
101         DataSource dataSource = getDataSource(material, dataSourceName);
102         for (List<String> each : Lists.partition(new ArrayList<>(actualTableNames), loadTableMetadataBatchSize)) {
103             result.add(new MetaDataLoaderMaterial(each, dataSourceName, dataSource, storageType, defaultSchemaName));
104         }
105         return result;
106     }
107     
108     private static DataSource getDataSource(final GenericSchemaBuilderMaterial material, final String dataSourceName) {
109         return material.getStorageUnits().get(dataSourceName.contains(".") ? dataSourceName.split("\\.")[0] : dataSourceName).getDataSource();
110     }
111     
112     private static void checkDataSourceTypeIncludeInstanceAndSetDatabaseTableMap(final Collection<DatabaseType> notSupportThreeTierStructureStorageTypes,
113                                                                                  final DataNodes dataNodes, final String tableName) {
114         for (DataNode dataNode : dataNodes.getDataNodes(tableName)) {
115             ShardingSpherePreconditions.checkState(notSupportThreeTierStructureStorageTypes.isEmpty() || !dataNode.getDataSourceName().contains("."),
116                     () -> new UnsupportedActualDataNodeStructureException(
117                             dataNode.getDataSourceName(), dataNode.getTableName(), notSupportThreeTierStructureStorageTypes.iterator().next().getJdbcUrlPrefixes()));
118             if (dataNode.getDataSourceName().contains(".")) {
119                 String database = dataNode.getDataSourceName().split("\\.")[1];
120                 GlobalDataSourceRegistry.getInstance().getCachedDatabaseTables().put(dataNode.getTableName(), database);
121             }
122         }
123     }
124     
125     private static Collection<DatabaseType> getUnsupportedThreeTierStorageStructureDatabaseTypes(final Collection<StorageUnit> storageUnits) {
126         return storageUnits.stream().map(StorageUnit::getStorageType)
127                 .filter(each -> !new DatabaseTypeRegistry(each).getDialectDatabaseMetaData().getConnectionOption().isSupportThreeTierStorageStructure()).collect(Collectors.toList());
128     }
129     
130     private static void addOneActualTableDataNode(final GenericSchemaBuilderMaterial material,
131                                                   final Map<String, Collection<String>> dataSourceTableGroups, final DataNodes dataNodes, final String table,
132                                                   final DatabaseType protocolType, final boolean tableNameLoadedFromStorage) {
133         Optional<DataNode> dataNode = dataNodes.getDataNodes(table).stream().filter(each -> isSameDataSourceNameSchemaName(material, each)).findFirst();
134         if (!dataNode.isPresent() && !material.getStorageUnits().keySet().iterator().hasNext()) {
135             return;
136         }
137         String dataSourceName = dataNode.map(DataNode::getDataSourceName).orElseGet(() -> material.getStorageUnits().keySet().iterator().next());
138         String tableName = dataNode.map(optional -> getTableName(material, protocolType, optional, tableNameLoadedFromStorage))
139                 .orElseGet(() -> normalizeTableName(material, protocolType, dataSourceName, table));
140         addDataSourceTableGroups(dataSourceName, tableName, dataSourceTableGroups);
141     }
142     
143     private static boolean isSameDataSourceNameSchemaName(final GenericSchemaBuilderMaterial material, final DataNode dataNode) {
144         String dataSourceName = dataNode.getDataSourceName().contains(".") ? dataNode.getDataSourceName().split("\\.")[0] : dataNode.getDataSourceName();
145         return material.getStorageUnits().containsKey(dataSourceName) && (null == dataNode.getSchemaName() || dataNode.getSchemaName().equalsIgnoreCase(material.getDefaultSchemaName()));
146     }
147     
148     private static void addAllActualTableDataNode(final GenericSchemaBuilderMaterial material,
149                                                   final Map<String, Collection<String>> dataSourceTableGroups, final DataNodes dataNodes, final String table,
150                                                   final DatabaseType protocolType, final boolean tableNameLoadedFromStorage) {
151         Collection<DataNode> tableDataNodes = dataNodes.getDataNodes(table);
152         if (tableDataNodes.isEmpty() && !material.getStorageUnits().isEmpty()) {
153             String dataSourceName = material.getStorageUnits().keySet().iterator().next();
154             addDataSourceTableGroups(dataSourceName, normalizeTableName(material, protocolType, dataSourceName, table), dataSourceTableGroups);
155         } else {
156             tableDataNodes.forEach(each -> addDataSourceTableGroups(each.getDataSourceName(), getTableName(material, protocolType, each, tableNameLoadedFromStorage), dataSourceTableGroups));
157         }
158     }
159     
160     private static String getTableName(final GenericSchemaBuilderMaterial material, final DatabaseType protocolType, final DataNode dataNode,
161                                        final boolean tableNameLoadedFromStorage) {
162         return tableNameLoadedFromStorage ? dataNode.getTableName() : normalizeTableName(material, protocolType, dataNode.getDataSourceName(), dataNode.getTableName());
163     }
164     
165     private static String normalizeTableName(final GenericSchemaBuilderMaterial material, final DatabaseType protocolType, final String dataSourceName, final String tableName) {
166         DatabaseType storageType = material.getStorageUnits().get(dataSourceName).getStorageType();
167         return protocolType.equals(storageType) ? tableName : material.getIdentifierContext().normalizeStorage(IdentifierScope.TABLE, new IdentifierValue(tableName));
168     }
169     
170     private static void addDataSourceTableGroups(final String dataSourceName, final String tableName, final Map<String, Collection<String>> dataSourceTableGroups) {
171         Collection<String> tables = dataSourceTableGroups.getOrDefault(dataSourceName, new LinkedList<>());
172         tables.add(tableName);
173         dataSourceTableGroups.putIfAbsent(dataSourceName, tables);
174     }
