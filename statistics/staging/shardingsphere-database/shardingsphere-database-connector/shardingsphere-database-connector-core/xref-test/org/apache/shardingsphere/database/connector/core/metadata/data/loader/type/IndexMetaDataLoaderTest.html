26  import java.sql.Connection;

35  import static org.hamcrest.MatcherAssert.assertThat;
36  import static org.hamcrest.Matchers.is;
37  import static org.junit.jupiter.api.Assertions.assertThrows;
38  import static org.junit.jupiter.api.Assertions.assertTrue;
39  import static org.mockito.Mockito.RETURNS_DEEP_STUBS;
40  import static org.mockito.Mockito.mock;
41  import static org.mockito.Mockito.when;
42  
43  class IndexMetaDataLoaderTest {
44      
45      @Test
46      void assertLoadWhenGetIndexInfoThrowsOracleViewNotAppropriateVendorCode() throws SQLException {
47          assertTrue(IndexMetaDataLoader.load(mockConnectionWithFailure(new SQLException("vendor", "42000", 1702)), "tbl").isEmpty());
48      }
49      
50      @Test
51      void assertLoadWhenGetIndexInfoThrowsSQLException() {
52          assertThrows(SQLException.class, () -> IndexMetaDataLoader.load(mockConnectionWithFailure(new SQLException("vendor", "42000", 1703)), "tbl"));
53      }
54      
55      private Connection mockConnectionWithFailure(final SQLException exception) throws SQLException {
56          Connection result = mock(Connection.class, RETURNS_DEEP_STUBS);
57          when(result.getCatalog()).thenReturn("catalog");
58          when(result.getMetaData().getIndexInfo("catalog", null, "tbl", false, false)).thenThrow(exception);
59          return result;
60      }
61      
62      @ParameterizedTest(name = "{0}")
63      @MethodSource("getLoadCases")
64      void assertLoad(final String name,
65                      final ResultSet resultSet, final int expectedSize, final String expectedName, final Collection<String> expectedColumns, final boolean expectedUnique) throws SQLException {
66          Collection<IndexMetaData> actual = IndexMetaDataLoader.load(mockConnection(resultSet), "tbl");
67          assertThat(actual.size(), is(expectedSize));
68          if (0 == expectedSize) {
69              return;
70          }
71          IndexMetaData actualIndexMetaData = actual.iterator().next();
72          assertThat(actualIndexMetaData.getName(), is(expectedName));
73          assertThat(actualIndexMetaData.getColumns(), is(expectedColumns));
74          assertThat(actualIndexMetaData.isUnique(), is(expectedUnique));
75      }
76      
77      private static Stream<Arguments> getLoadCases() throws SQLException {
78          return Stream.of(
79                  Arguments.of("skipNullIndexName", mockResultSet(new String[]{null}, new String[]{"foo_column"}, new boolean[]{false}),
80                          0, "", Collections.emptyList(), false),
81                  Arguments.of("loadUniqueIndex", mockResultSet(new String[]{"foo_index"}, new String[]{"foo_column"}, new boolean[]{false}),
82                          1, "foo_index", Collections.singletonList("foo_column"), true),
83                  Arguments.of("loadNonUniqueIndexWithMultipleColumns", mockResultSet(new String[]{"foo_index", "foo_index"},
84                          new String[]{"foo_column", "bar_column"}, new boolean[]{true, true}), 1, "foo_index", Arrays.asList("foo_column", "bar_column"), false));
85      }
86      
87      private static ResultSet mockResultSet(final String[] indexNames, final String[] columnNames, final boolean[] nonUniqueFlags) throws SQLException {
88          ResultSet result = mock(ResultSet.class);
89          AtomicInteger rowCursor = new AtomicInteger(-1);
90          when(result.next()).thenAnswer(invocation -> rowCursor.incrementAndGet() < indexNames.length);
91          when(result.getString("INDEX_NAME")).thenAnswer(invocation -> indexNames[rowCursor.get()]);
92          when(result.getString("COLUMN_NAME")).thenAnswer(invocation -> columnNames[rowCursor.get()]);
93          when(result.getBoolean("NON_UNIQUE")).thenAnswer(invocation -> nonUniqueFlags[rowCursor.get()]);
94          return result;
95      }
96      
97      private static Connection mockConnection(final ResultSet resultSet) throws SQLException {
98          Connection result = mock(Connection.class, RETURNS_DEEP_STUBS);
99          when(result.getCatalog()).thenReturn("catalog");
100         when(result.getMetaData().getIndexInfo("catalog", null, "tbl", false, false)).thenReturn(resultSet);
101         return result;
102     }
103 }
