52      /**
53       * Create workflow operation intent, leaving the operation unresolved when natural-language intent requires clarification.
54       *
55       * @param request workflow request
56       * @param defaultOperationType default operation type
57       * @return clarified workflow intent
58       */
59      public ClarifiedIntent createOperationIntent(final WorkflowRequest request, final String defaultOperationType) {
60          ClarifiedIntent result = new ClarifiedIntent();
61          if (!request.getOperationType().isEmpty()) {
62              result.setOperationType(request.getOperationType());
63          } else if (request.getNaturalLanguageIntent().isEmpty()) {
64              result.setOperationType(defaultOperationType);
65              result.getInferredValues().put(WorkflowFieldNames.OPERATION_TYPE, defaultOperationType);
66          }
67          return result;
68      }
69      
70      /**
71       * Create workflow operation intent for a workflow whose operation is fixed by its contract.
72       *
73       * @param request workflow request
74       * @param fixedOperationType fixed operation type
75       * @return clarified workflow intent
76       */
77      public ClarifiedIntent createFixedOperationIntent(final WorkflowRequest request, final String fixedOperationType) {
78          ClarifiedIntent result = new ClarifiedIntent();
79          if (request.getOperationType().isEmpty()) {
80              result.setOperationType(fixedOperationType);
81              result.getInferredValues().put(WorkflowFieldNames.OPERATION_TYPE, fixedOperationType);
82          } else {
83              result.setOperationType(request.getOperationType());
84          }
85          return result;
86      }

98      /**
99       * Prepare workflow snapshot for planning.
100      *
101      * @param snapshot workflow snapshot
102      * @param workflowKind workflow kind
103      * @param request merged request
104      * @param featureData feature-scoped workflow data
105      * @param clarifiedIntent clarified intent
106      * @param summary interaction summary
107      * @param interactionSteps interaction steps
108      * @param validationLayers validation layers
109      * @param <T> request type
110      * @return prepared request
111      */
112     public <T extends WorkflowRequest> T prepareSnapshot(final WorkflowContextSnapshot snapshot, final WorkflowKind workflowKind, final T request,
113                                                          final RuleWorkflowFeatureData featureData,
114                                                          final ClarifiedIntent clarifiedIntent, final String summary,
115                                                          final List<String> interactionSteps, final List<String> validationLayers) {
116         WorkflowKind existingWorkflowKind = snapshot.getWorkflowKind();
117         ShardingSpherePreconditions.checkState(null == existingWorkflowKind || existingWorkflowKind.equals(workflowKind),
118                 () -> new MCPInvalidRequestException(String.format("plan_id `%s` belongs to workflow kind `%s`; call the matching planning tool or omit plan_id to start `%s`.",
119                         snapshot.getPlanId(), existingWorkflowKind, workflowKind)));
120         snapshot.setWorkflowKind(workflowKind);
121         snapshot.setRequest(request);
122         snapshot.setFeatureData(featureData);
123         snapshot.setInteractionPlan(InteractionPlan.create(snapshot.getPlanId(), request, summary, interactionSteps, validationLayers));
124         snapshot.clearPlanningState();
125         snapshot.setClarifiedIntent(clarifiedIntent);
126         return request;
127     }

130      * Persist a workflow whose planning flow was interrupted by a clarification or terminal failure.
131      *
132      * @param workflowSessionContext workflow session context
133      * @param snapshot workflow snapshot
134      * @return persisted workflow snapshot
135      */
136     public WorkflowContextSnapshot persistPlanningInterruption(final WorkflowSessionContext workflowSessionContext, final WorkflowContextSnapshot snapshot) {
137         String currentStep = WorkflowLifecycle.STATUS_FAILED.equals(snapshot.getStatus()) ? WorkflowLifecycle.STEP_FAILED : WorkflowLifecycle.STEP_CLARIFYING;
138         return workflowSessionContext.persist(snapshot, currentStep, snapshot.getStatus());
139     }

141     /**
142      * Ensure lifecycle state matches the requested workflow operation.
143      *
144      * @param ruleLabel rule label for user-facing issues
145      * @param clarifiedIntent clarified intent
146      * @param ruleExists whether the target rule already exists
147      * @param snapshot workflow snapshot
148      * @return whether lifecycle state is valid
149      */
150     public boolean ensureLifecycleState(final String ruleLabel, final ClarifiedIntent clarifiedIntent,
151                                         final boolean ruleExists, final WorkflowContextSnapshot snapshot) {
152         String actualOperationType = clarifiedIntent.getOperationType().toLowerCase(Locale.ENGLISH);
153         if (WorkflowLifecycle.OPERATION_CREATE.equals(actualOperationType) && ruleExists) {
154             snapshot.getIssues().add(new WorkflowIssue(WorkflowIssueCode.RULE_STATE_MISMATCH, "error", WorkflowLifecycle.STEP_DISCOVERING,
155                     String.format("%s already exists for the target column.", ruleLabel), "Use a supported change path for the existing rule, or drop it before creating a replacement.", false,
156                     Map.of()));
157             return false;
158         }
159         if (WorkflowLifecycle.OPERATION_ALTER.equals(actualOperationType) && !ruleExists) {
160             snapshot.getIssues().add(new WorkflowIssue(WorkflowIssueCode.RULE_STATE_MISMATCH, "error", WorkflowLifecycle.STEP_DISCOVERING,
161                     String.format("%s does not exist for the target column.", ruleLabel), "Use create or confirm the target column.", false, Map.of()));
162             return false;
163         }
164         if (WorkflowLifecycle.OPERATION_DROP.equals(actualOperationType) && !ruleExists) {
165             snapshot.getIssues().add(new WorkflowIssue(WorkflowIssueCode.DROP_TARGET_RULE_NOT_FOUND, "error", WorkflowLifecycle.STEP_DISCOVERING,
166                     String.format("%s does not exist for the target column.", ruleLabel), "Confirm target table and column or skip the drop request.", false, Map.of()));
167             return false;
168         }
169         return true;
170     }

172     /**
173      * Ensure workflow operation type is exposed by the feature contract.
174      *
175      * @param clarifiedIntent clarified intent
176      * @param supportedOperationTypes supported operation types
177      * @param snapshot workflow snapshot
178      * @return whether the operation type is supported
179      */
180     public boolean ensureSupportedOperationType(final ClarifiedIntent clarifiedIntent, final Collection<String> supportedOperationTypes, final WorkflowContextSnapshot snapshot) {
181         if (containsOperationType(supportedOperationTypes, clarifiedIntent.getOperationType())) {
182             return true;
183         }
184         if (clarifiedIntent.getOperationType().isEmpty()) {
185             if (!clarifiedIntent.getUnresolvedFields().contains(WorkflowFieldNames.OPERATION_TYPE)) {
186                 clarifiedIntent.getUnresolvedFields().add(WorkflowFieldNames.OPERATION_TYPE);
187             }
188             clarifiedIntent.getClarificationMessages().add("Please provide operation_type.");
189             snapshot.getIssues().add(new WorkflowIssue(WorkflowIssueCode.RULE_INPUT_REQUIRED, "error", WorkflowLifecycle.STEP_INTAKING,
190                     "Workflow operation type is required when natural_language_intent is provided.",
191                     String.format("Provide operation_type as one of: %s.", String.join(", ", supportedOperationTypes)), true,
192                     Map.of("missing_inputs", List.of(WorkflowFieldNames.OPERATION_TYPE), "supported_operation_types", supportedOperationTypes)));
193             snapshot.setStatus(WorkflowLifecycle.STATUS_CLARIFYING);
194             return false;
195         }
196         snapshot.getIssues().add(new WorkflowIssue(WorkflowIssueCode.WORKFLOW_STATUS_INVALID, "error", WorkflowLifecycle.STEP_INTAKING,
197                 "Unsupported workflow operation type.", String.format("Use one of: %s.", String.join(", ", supportedOperationTypes)), false,
198                 Map.of("supported_operation_types", supportedOperationTypes)));
199         clarifiedIntent.getInferredValues().remove(WorkflowFieldNames.OPERATION_TYPE);
200         clarifiedIntent.setOperationType("");
201         if (null != snapshot.getRequest()) {
202             snapshot.getRequest().setOperationType("");
203         }
204         snapshot.setStatus(WorkflowLifecycle.STATUS_FAILED);
205         return false;
206     }
207     
208     private boolean containsOperationType(final Collection<String> supportedOperationTypes, final String actualOperationType) {
209         for (String each : supportedOperationTypes) {
210             if (each.equalsIgnoreCase(actualOperationType)) {
211                 return true;
212             }
213         }
214         return false;
215     }

217     /**
218      * Judge whether workflow planning can continue to artifact generation.
219      *
220      * @param request workflow request
221      * @param clarifiedIntent clarified intent
222      * @param snapshot workflow snapshot
223      * @param propertyRequirements property requirements
224      * @param fallbackQuestion fallback question
225      * @return whether artifact planning can continue
226      */
227     public boolean isReadyForArtifactPlanning(final WorkflowRequest request, final ClarifiedIntent clarifiedIntent, final WorkflowContextSnapshot snapshot,
228                                               final List<AlgorithmPropertyRequirement> propertyRequirements, final String fallbackQuestion) {
229         return requirementCollector.isReadyForArtifactPlanning(request, clarifiedIntent, snapshot, propertyRequirements, fallbackQuestion);
230     }
