87      /**
88       * Bind simple table segment.
89       *
90       * @param segment simple table segment
91       * @param binderContext SQL statement binder context
92       * @param tableBinderContexts table binder contexts
93       * @return bound simple table segment
94       */
95      public static SimpleTableSegment bind(final SimpleTableSegment segment, final SQLStatementBinderContext binderContext,
96                                            final Multimap<CaseInsensitiveString, TableSegmentBinderContext> tableBinderContexts) {
97          binderContext.getPivotColumnNames().addAll(segment.getPivot().map(PivotSegment::getPivotColumnNames).orElse(Collections.emptyList()));
98          IdentifierValue databaseName = getDatabaseName(segment, binderContext);
99          Optional<IdentifierValue> schemaName = getSchemaName(segment, binderContext, databaseName);
100         IdentifierValue tableName = segment.getTableName().getIdentifier();
101         Optional<ShardingSphereSchema> schema = schemaName.map(identifierValue -> binderContext.getMetaData().getDatabase(databaseName).getSchema(identifierValue));
102         if (isUpdateTargetTableAlias(binderContext, tableBinderContexts, schemaName, tableName.getValue(), segment)) {
103             return bindUpdateTargetTableAlias(segment, binderContext, tableBinderContexts, databaseName, schemaName, tableName);
104         }
105         checkTableExists(binderContext, schema.orElse(null), schemaName, tableName, segment, tableBinderContexts);
106         checkTableMetadata(binderContext, schema.orElse(null), schemaName.map(IdentifierValue::getValue).orElse(null), tableName);
107         String tableAliasOrName = segment.getAliasName().orElseGet(tableName::getValue);
108         Optional<SimpleTableSegmentBinderContext> tableBinderContext = createSimpleTableBinderContext(segment, schema.orElse(null), databaseName, schemaName.orElse(null), binderContext);
109         tableBinderContext.ifPresent(context -> tableBinderContexts.put(CaseInsensitiveString.of(tableAliasOrName), context));
110         TableNameSegment tableNameSegment = new TableNameSegment(segment.getTableName().getStartIndex(), segment.getTableName().getStopIndex(), tableName);
111         tableNameSegment.setTableBoundInfo(new TableSegmentBoundInfo(databaseName, schemaName.orElse(null)));
112         SimpleTableSegment result = new SimpleTableSegment(tableNameSegment);
113         segment.getOwner().ifPresent(result::setOwner);
114         segment.getAliasSegment().ifPresent(result::setAlias);
115         segment.getAt().ifPresent(result::setAt);
116         segment.getDbLink().ifPresent(result::setDbLink);
117         result.setTableSampled(segment.isTableSampled());
118         segment.getTableSampleExpression().map(optional -> ExpressionSegmentBinder.bind(optional, SegmentType.JOIN_ON, binderContext, tableBinderContexts, LinkedHashMultimap.create()))
119                 .ifPresent(result::setTableSampleExpression);
120         tableBinderContext.flatMap(context -> segment.getPivot()
121                 .map(optional -> PivotSegmentBinder.bind(optional, binderContext, createTableBinderContexts(tableAliasOrName, context), LinkedHashMultimap.create()))).ifPresent(result::setPivot);
122         return result;
123     }

125     private static Multimap<CaseInsensitiveString, TableSegmentBinderContext> createTableBinderContexts(final String tableAliasOrName, final TableSegmentBinderContext tableBinderContext) {
126         Multimap<CaseInsensitiveString, TableSegmentBinderContext> result = LinkedHashMultimap.create();
127         result.put(CaseInsensitiveString.of(tableAliasOrName), tableBinderContext);
128         return result;
129     }

131     private static IdentifierValue getDatabaseName(final SimpleTableSegment segment, final SQLStatementBinderContext binderContext) {
132         DialectDatabaseMetaData dialectDatabaseMetaData = new DatabaseTypeRegistry(binderContext.getSqlStatement().getDatabaseType()).getDialectDatabaseMetaData();
133         Optional<IdentifierValue> ownerDatabaseName = getOwnerDatabaseName(segment, binderContext, dialectDatabaseMetaData);
134         Optional<OwnerSegment> owner = dialectDatabaseMetaData.getSchemaOption().getDefaultSchema().isPresent() ? segment.getOwner().flatMap(OwnerSegment::getOwner) : segment.getOwner();
135         IdentifierValue result = ownerDatabaseName.orElseGet(() -> new IdentifierValue(owner.map(optional -> optional.getIdentifier().getValue()).orElse(binderContext.getCurrentDatabaseName())));
136         ShardingSpherePreconditions.checkNotNull(result.getValue(), NoDatabaseSelectedException::new);
137         if (binderContext.getHintValueContext().isSkipMetadataValidate() && null != binderContext.getCurrentDatabaseName() && !binderContext.getMetaData().containsDatabase(result)) {
138             return new IdentifierValue(binderContext.getCurrentDatabaseName());
139         }
140         ShardingSpherePreconditions.checkState(binderContext.getMetaData().containsDatabase(result), () -> new UnknownDatabaseException(result.getValue()));
141         return result;
142     }

144     private static Optional<IdentifierValue> getOwnerDatabaseName(final SimpleTableSegment segment, final SQLStatementBinderContext binderContext,
145                                                                   final DialectDatabaseMetaData dialectDatabaseMetaData) {
146         if (!dialectDatabaseMetaData.getSchemaOption().getDefaultSchema().isPresent() || !segment.getOwner().isPresent() || segment.getOwner().get().getOwner().isPresent()) {
147             return Optional.empty();
148         }
149         IdentifierValue owner = segment.getOwner().get().getIdentifier();
150         ShardingSphereDatabase currentDatabase = binderContext.getMetaData().getDatabase(binderContext.getCurrentDatabaseName());
151         if (null != currentDatabase && currentDatabase.containsSchema(owner)) {
152             return Optional.empty();
153         }
154         return binderContext.getMetaData().containsDatabase(owner) ? Optional.of(owner) : Optional.empty();
155     }

157     private static Optional<IdentifierValue> getSchemaName(final SimpleTableSegment segment, final SQLStatementBinderContext binderContext, final IdentifierValue databaseName) {
158         Optional<IdentifierValue> result = getSchemaNameValue(segment, binderContext, databaseName);
159         if (!binderContext.getHintValueContext().isSkipMetadataValidate()) {
160             result.ifPresent(identifierValue -> ShardingSpherePreconditions.checkState(binderContext.getMetaData().getDatabase(databaseName).containsSchema(identifierValue),
161                     () -> new SchemaNotFoundException(identifierValue.getValue())));
162         }
163         return result;
164     }

166     private static Optional<IdentifierValue> getSchemaNameValue(final SimpleTableSegment segment, final SQLStatementBinderContext binderContext, final IdentifierValue databaseName) {
167         DatabaseType databaseType = binderContext.getSqlStatement().getDatabaseType();
168         DatabaseTypeRegistry databaseTypeRegistry = new DatabaseTypeRegistry(databaseType);
169         DialectDatabaseMetaData dialectDatabaseMetaData = databaseTypeRegistry.getDialectDatabaseMetaData();
170         if (segment.getOwner().isPresent() && !isOwnerDatabaseName(segment, binderContext, dialectDatabaseMetaData)) {
171             return Optional.ofNullable(segment.getOwner().get().getIdentifier());
172         }
173         Optional<String> defaultSystemSchema = dialectDatabaseMetaData.getSchemaOption().getDefaultSystemSchema();
174         if (defaultSystemSchema.isPresent() && SystemSchemaManager.isSystemTable(databaseType.getType(), defaultSystemSchema.get(), segment.getTableName().getIdentifier().getValue())) {
175             return Optional.of(new IdentifierValue(defaultSystemSchema.get()));
176         }
177         if (dialectDatabaseMetaData.getSchemaOption().getDefaultSchema().isPresent()) {
178             ShardingSphereDatabase database = binderContext.getMetaData().getDatabase(databaseName);
179             if (binderContext.getSqlStatement() instanceof CreateTableStatement || binderContext.getSqlStatement() instanceof CreateViewStatement
180                     || binderContext.getSqlStatement() instanceof CreateIndexStatement) {
181                 return getCreateSchemaName(database, getDefaultSchemaName(database, binderContext.getCurrentSchema()));
182             }
183             Optional<IdentifierValue> result = findSchemaIdentifierByTableName(database, segment.getTableName().getIdentifier(), binderContext.getCurrentSchema());
184             if (result.isPresent()) {
185                 return result;
186             }
187             result = findUniqueSchemaIdentifierByTableName(database, segment.getTableName().getIdentifier());
188             return result.isPresent() ? result : Optional.ofNullable(database.getDefaultSchemaName()).map(IdentifierValue::new);
189         }
190         ShardingSphereDatabase database = binderContext.getMetaData().getDatabase(binderContext.getCurrentDatabaseName());
191         return Optional.ofNullable(database.getDefaultSchemaName()).map(IdentifierValue::new);
192     }

194     private static Optional<IdentifierValue> getDefaultSchemaName(final ShardingSphereDatabase database, final Collection<IdentifierValue> currentSchema) {
195         for (IdentifierValue each : currentSchema) {
196             if (database.containsSchema(each)) {
197                 return Optional.of(each);
198             }
199         }
200         return Optional.ofNullable(database.getDefaultSchemaName()).map(IdentifierValue::new);
201     }

203     private static Optional<IdentifierValue> findSchemaIdentifierByTableName(final ShardingSphereDatabase database, final IdentifierValue tableName,
204                                                                              final Collection<IdentifierValue> schemaSearchPath) {
205         for (IdentifierValue each : schemaSearchPath) {
206             ShardingSphereSchema schema = database.getSchema(each);
207             if (null != schema && schema.containsTable(tableName)) {
208                 return Optional.of(each);
209             }
210         }
211         return Optional.empty();
212     }

214     private static Optional<IdentifierValue> findUniqueSchemaIdentifierByTableName(final ShardingSphereDatabase database, final IdentifierValue tableName) {
215         IdentifierValue result = null;
216         for (ShardingSphereSchema each : database.getAllSchemas()) {
217             if (!each.containsTable(tableName)) {
218                 continue;
219             }
220             if (null != result) {
221                 return Optional.empty();
222             }
223             result = new IdentifierValue(each.getName());
224         }
225         return Optional.ofNullable(result);
226     }

228     private static Optional<IdentifierValue> getCreateSchemaName(final ShardingSphereDatabase database, final Optional<IdentifierValue> defaultSchemaName) {
229         if (defaultSchemaName.filter(database::containsSchema).isPresent()) {
230             return defaultSchemaName;
231         }
232         Collection<ShardingSphereSchema> schemas = database.getAllSchemas().stream().filter(each -> !isSystemSchema(database, each)).collect(Collectors.toList());
233         return 1 == schemas.size() ? Optional.of(new IdentifierValue(schemas.iterator().next().getName())) : defaultSchemaName;
234     }

237         for (String each : new SystemDatabase(database.getProtocolType()).getSystemSchemas()) {
238             if (each.equalsIgnoreCase(schema.getName())) {
239                 return true;

241         }
242         return false;

249     private static boolean isUpdateTargetTableAlias(final SQLStatementBinderContext binderContext, final Multimap<CaseInsensitiveString, TableSegmentBinderContext> tableBinderContexts,
250                                                     final Optional<IdentifierValue> schemaName, final String tableNameValue, final SimpleTableSegment segment) {
251         if (!(binderContext.getSqlStatement() instanceof UpdateStatement)) {
252             return false;
253         }
254         UpdateStatement updateStatement = (UpdateStatement) binderContext.getSqlStatement();
255         if (!updateStatement.getFrom().isPresent()) {
256             return false;
257         }
258         if (!updateStatement.isTargetTableIsFromAlias()) {
259             return false;
260         }
261         if (!(updateStatement.getTable() instanceof SimpleTableSegment)) {
262             return false;
263         }
264         if (!((SimpleTableSegment) updateStatement.getTable()).getTableName().getIdentifier().getValue().equalsIgnoreCase(tableNameValue)) {
265             return false;
266         }
267         if (segment.getAliasName().isPresent()) {
268             return false;
269         }
270         return !segment.getOwner().isPresent() && tableBinderContexts.containsKey(CaseInsensitiveString.of(tableNameValue))
271                 || tableBinderContexts.values().stream().anyMatch(each -> isSameUpdateTargetTableContext(segment, schemaName, tableNameValue, each));
272     }

274     private static SimpleTableSegment bindUpdateTargetTableAlias(final SimpleTableSegment segment, final SQLStatementBinderContext binderContext,
275                                                                  final Multimap<CaseInsensitiveString, TableSegmentBinderContext> tableBinderContexts, final IdentifierValue databaseName,
276                                                                  final Optional<IdentifierValue> schemaName, final IdentifierValue tableName) {
277         Collection<TableSegmentBinderContext> fromTableContexts = !segment.getOwner().isPresent() && tableBinderContexts.containsKey(CaseInsensitiveString.of(tableName.getValue()))
278                 ? tableBinderContexts.get(CaseInsensitiveString.of(tableName.getValue()))
279                 : tableBinderContexts.values().stream()
280                         .filter(each -> isSameUpdateTargetTableContext(segment, schemaName, tableName.getValue(), each)).collect(Collectors.toList());
281         IdentifierValue originalTableName = fromTableContexts.stream()
282                 .map(TableSegmentBinderContext::getOriginalTableName).filter(Optional::isPresent).map(Optional::get).findFirst().orElse(tableName);
283         Optional<OwnerSegment> fromTableOwner = fromTableContexts.stream()
284                 .map(TableSegmentBinderContext::getOriginalOwner).filter(Optional::isPresent).map(Optional::get).findFirst();
285         IdentifierValue resolvedSchemaName = fromTableOwner.map(OwnerSegment::getIdentifier).orElseGet(() -> schemaName.orElse(null));
286         TableNameSegment tableNameSegment = new TableNameSegment(segment.getTableName().getStartIndex(), segment.getTableName().getStopIndex(), originalTableName);
287         tableNameSegment.setTableBoundInfo(new TableSegmentBoundInfo(databaseName, resolvedSchemaName));
288         SimpleTableSegment result = new SimpleTableSegment(tableNameSegment);
289         fromTableOwner.ifPresent(result::setOwner);
290         result.setAlias(segment.getAliasSegment().orElseGet(() -> new AliasSegment(segment.getTableName().getStartIndex(), segment.getTableName().getStopIndex(), tableName)));
291         segment.getAt().ifPresent(result::setAt);
292         segment.getDbLink().ifPresent(result::setDbLink);
293         result.setTableSampled(segment.isTableSampled());
294         segment.getTableSampleExpression().map(optional -> ExpressionSegmentBinder.bind(optional, SegmentType.JOIN_ON, binderContext, tableBinderContexts, LinkedHashMultimap.create()))
295                 .ifPresent(result::setTableSampleExpression);
296         return result;
297     }

299     private static boolean isSameUpdateTargetTableContext(final SimpleTableSegment targetTable, final Optional<IdentifierValue> schemaName, final String tableName,
300                                                           final TableSegmentBinderContext tableBinderContext) {
301         return tableBinderContext.getOriginalTableName().map(each -> each.getValue().equalsIgnoreCase(tableName)).orElse(false)
302                 && isSameUpdateTargetOwner(targetTable, schemaName, tableBinderContext);
303     }
304     
305     private static boolean isSameUpdateTargetOwner(final SimpleTableSegment targetTable, final Optional<IdentifierValue> schemaName,
306                                                    final TableSegmentBinderContext tableBinderContext) {
307         Optional<OwnerSegment> originalOwner = tableBinderContext.getOriginalOwner();
308         if (!targetTable.getOwner().isPresent()) {
309             return !originalOwner.isPresent() || schemaName.map(optional -> originalOwner.get().getIdentifier().getValue().equalsIgnoreCase(optional.getValue())).orElse(false);
310         }
311         return originalOwner.isPresent() && isSameOwner(targetTable.getOwner().get(), originalOwner.get());
312     }
313     
314     private static boolean isSameOwner(final OwnerSegment targetOwner, final OwnerSegment originalOwner) {
315         return targetOwner.getIdentifier().getValue().equalsIgnoreCase(originalOwner.getIdentifier().getValue())
316                 && targetOwner.getOwner().isPresent() == originalOwner.getOwner().isPresent()
317                 && (!targetOwner.getOwner().isPresent() || isSameOwner(targetOwner.getOwner().get(), originalOwner.getOwner().get()));
318     }

320     private static void checkTableExists(final SQLStatementBinderContext binderContext, final ShardingSphereSchema schema, final Optional<IdentifierValue> schemaName,
321                                          final IdentifierValue tableName, final SimpleTableSegment segment, final Multimap<CaseInsensitiveString, TableSegmentBinderContext> tableBinderContexts) {
322         String tableNameValue = tableName.getValue();
323         if (isUpdateTargetTableAlias(binderContext, tableBinderContexts, schemaName, tableNameValue, segment)) {
324             return;
325         }
326         // TODO refactor table exists check with spi @duanzhengqiang
327         if (binderContext.getSqlStatement() instanceof CreateTableStatement && isCreateTable(((CreateTableStatement) binderContext.getSqlStatement()).getTable(), tableNameValue)) {
328             ShardingSpherePreconditions.checkState(binderContext.getHintValueContext().isSkipMetadataValidate()
329                     || ((CreateTableStatement) binderContext.getSqlStatement()).isIfNotExists() || null == schema || !schema.containsTable(tableName),
330                     () -> new TableExistsException(tableNameValue));
331             return;
332         }
333         if (binderContext.getSqlStatement() instanceof AlterTableStatement && isRenameTable((AlterTableStatement) binderContext.getSqlStatement(), tableNameValue)) {
334             ShardingSpherePreconditions.checkState(binderContext.getHintValueContext().isSkipMetadataValidate() || null == schema || !schema.containsTable(tableName),
335                     () -> new TableExistsException(tableNameValue));
336             return;
337         }
338         if (binderContext.getSqlStatement() instanceof DropTableStatement) {
339             ShardingSpherePreconditions.checkState(((DropTableStatement) binderContext.getSqlStatement()).isIfExists() || null != schema && schema.containsTable(tableName),
340                     () -> new TableNotFoundException(tableNameValue));
341             return;
342         }
343         if (binderContext.getSqlStatement() instanceof RenameTableStatement && isRenameTable((RenameTableStatement) binderContext.getSqlStatement(), tableNameValue)) {
344             ShardingSpherePreconditions.checkState(binderContext.getHintValueContext().isSkipMetadataValidate() || null == schema || !schema.containsTable(tableName),
345                     () -> new TableExistsException(tableNameValue));
346             return;
347         }
348         if (binderContext.getSqlStatement() instanceof CreateViewStatement && isCreateTable(((CreateViewStatement) binderContext.getSqlStatement()).getView(), tableNameValue)) {
349             ShardingSpherePreconditions.checkState(binderContext.getHintValueContext().isSkipMetadataValidate()
350                     || ((CreateViewStatement) binderContext.getSqlStatement()).isReplaceView() || null == schema || !schema.containsTable(tableName),
351                     () -> new TableExistsException(tableNameValue));
352             return;
353         }
354         if (binderContext.getSqlStatement() instanceof AlterViewStatement && isRenameView((AlterViewStatement) binderContext.getSqlStatement(), tableNameValue)) {
355             ShardingSpherePreconditions.checkState(binderContext.getHintValueContext().isSkipMetadataValidate() || null == schema || !schema.containsTable(tableName),
356                     () -> new TableExistsException(tableNameValue));
357             return;
358         }
359         if (binderContext.getSqlStatement() instanceof DropViewStatement) {
360             ShardingSpherePreconditions.checkState(((DropViewStatement) binderContext.getSqlStatement()).isIfExists() || null != schema && schema.containsTable(tableName),
361                     () -> new TableNotFoundException(tableNameValue));
362             return;
363         }
364         if ("DUAL".equalsIgnoreCase(tableNameValue)) {
365             return;
366         }
367         if (null != schema && SystemSchemaManager.isSystemTable(schema.getName(), tableNameValue)) {
368             return;
369         }
370         if (segment.getDbLink().isPresent()) {
371             return;
372         }
373         if (binderContext.getExternalTableBinderContexts().containsKey(CaseInsensitiveString.of(tableNameValue))) {
374             return;
375         }
376         if (binderContext.getCommonTableExpressionsSegmentsUniqueAliases().contains(tableNameValue)) {
377             return;
378         }
379         if (binderContext.getHintValueContext().isSkipMetadataValidate()) {
380             return;
381         }
382         ShardingSpherePreconditions.checkState(null != schema && schema.containsTable(tableName), () -> new TableNotFoundException(tableNameValue));
383     }

385     private static boolean isCreateTable(final SimpleTableSegment simpleTableSegment, final String tableName) {
386         return simpleTableSegment.getTableName().getIdentifier().getValue().equalsIgnoreCase(tableName);
387     }

389     private static boolean isRenameTable(final AlterTableStatement alterTableStatement, final String tableName) {
390         return alterTableStatement.getRenameTable().isPresent() && alterTableStatement.getRenameTable().get().getTableName().getIdentifier().getValue().equalsIgnoreCase(tableName);
391     }

393     private static boolean isRenameTable(final RenameTableStatement renameTableStatement, final String tableName) {
394         for (RenameTableDefinitionSegment each : renameTableStatement.getRenameTables()) {
395             if (each.getRenameTable().getTableName().getIdentifier().getValue().equalsIgnoreCase(tableName)) {
396                 return true;
397             }
398         }
399         return false;
400     }

403         return alterViewStatement.getRenameView().isPresent() && alterViewStatement.getRenameView().get().getTableName().getIdentifier().getValue().equalsIgnoreCase(tableName);
404     }

406     private static void checkTableMetadata(final SQLStatementBinderContext binderContext, final ShardingSphereSchema schema, final String schemaName, final IdentifierValue tableName) {
407         if (binderContext.getHintValueContext().isSkipMetadataValidate() || null == schema) {
408             return;
409         }
410         ShardingSphereTable shardingSphereTable = schema.getTable(tableName);
411         if (binderContext.getSqlStatement() instanceof AlterTableStatement) {
412             if (isRenameTable((AlterTableStatement) binderContext.getSqlStatement(), tableName.getValue())) {
413                 return;
414             }
415             ShardingSpherePreconditions.checkState(schema.containsTable(tableName), () -> new TableNotFoundException(tableName.getValue()));
416             AlterTableMetadataCheckUtils.checkAlterTable((AlterTableStatement) binderContext.getSqlStatement(), shardingSphereTable);
417             return;
418         }
419         if (binderContext.getSqlStatement() instanceof CreateIndexStatement) {
420             ShardingSpherePreconditions.checkState(schema.containsTable(tableName), () -> new TableNotFoundException(tableName.getValue()));
421             IdentifierValue indexName = ((CreateIndexStatement) binderContext.getSqlStatement()).getIndex().getIndexName().getIdentifier();
422             ShardingSpherePreconditions.checkState(!shardingSphereTable.containsIndex(indexName), () -> new DuplicateIndexException(indexName.getValue()));
423             return;
424         }
425         if (binderContext.getSqlStatement() instanceof DropIndexStatement) {
426             ShardingSpherePreconditions.checkState(schema.containsTable(tableName), () -> new TableNotFoundException(tableName.getValue()));
427             ((DropIndexStatement) binderContext.getSqlStatement()).getIndexes().forEach(each -> {
428                 IdentifierValue indexName = each.getIndexName().getIdentifier();
429                 ShardingSpherePreconditions.checkState(shardingSphereTable.containsIndex(indexName), () -> new IndexNotFoundException(schemaName, indexName.getValue()));
430             });
431         }
432     }

434     private static Optional<SimpleTableSegmentBinderContext> createSimpleTableBinderContext(final SimpleTableSegment segment, final ShardingSphereSchema schema, final IdentifierValue databaseName,
435                                                                                             final IdentifierValue schemaName, final SQLStatementBinderContext binderContext) {
436         IdentifierValue tableName = segment.getTableName().getIdentifier();
437         Optional<SimpleTableSegmentBinderContext> externalTableBinderContext = createExternalTableBinderContext(segment, tableName, binderContext);
438         if (externalTableBinderContext.isPresent()) {
439             return externalTableBinderContext;
440         }
441         if (null != schema && schema.containsTable(tableName)) {
442             return createSimpleTableSegmentBinderContextWithMetaData(segment, schema, databaseName, schemaName, binderContext, tableName);
443         }
444         if (binderContext.getSqlStatement() instanceof CreateTableStatement) {
445             Collection<ProjectionSegment> projectionSegments = createProjectionSegments((CreateTableStatement) binderContext.getSqlStatement(), databaseName, schemaName, tableName);
446             return Optional.of(new SimpleTableSegmentBinderContext(projectionSegments, TableSourceType.PHYSICAL_TABLE));
447         }
448         SimpleTableSegmentBinderContext result = new SimpleTableSegmentBinderContext(Collections.emptyList(), TableSourceType.TEMPORARY_TABLE);
449         result.setSkipColumnBind(binderContext.getHintValueContext().isSkipMetadataValidate() && segment.getOwner().isPresent());
450         segment.getDbLink().ifPresent(optional -> result.setContainsDBLink(true));
451         return Optional.of(result);
452     }
453     
454     private static Optional<SimpleTableSegmentBinderContext> createExternalTableBinderContext(final SimpleTableSegment segment, final IdentifierValue tableName,
455                                                                                               final SQLStatementBinderContext binderContext) {
456         if (segment.getOwner().isPresent()) {
457             return Optional.empty();
458         }
459         CaseInsensitiveString caseInsensitiveTableName = CaseInsensitiveString.of(tableName.getValue());
460         if (!binderContext.getExternalTableBinderContexts().containsKey(caseInsensitiveTableName)) {
461             return Optional.empty();
462         }
463         TableSegmentBinderContext tableSegmentBinderContext = binderContext.getExternalTableBinderContexts().get(caseInsensitiveTableName).iterator().next();
464         Collection<ProjectionSegment> subqueryProjections = SubqueryTableBindUtils.createSubqueryProjections(
465                 tableSegmentBinderContext.getProjectionSegments(), tableName, binderContext.getSqlStatement().getDatabaseType(), TableSourceType.TEMPORARY_TABLE);
466         SimpleTableSegmentBinderContext result = new SimpleTableSegmentBinderContext(subqueryProjections, TableSourceType.TEMPORARY_TABLE);
467         if (tableSegmentBinderContext instanceof SimpleTableSegmentBinderContext && ((SimpleTableSegmentBinderContext) tableSegmentBinderContext).isFromWithSegment()) {
468             result.setFromWithSegment(true);
469         }
470         return Optional.of(result);
471     }
472     
473     private static Collection<ProjectionSegment> createProjectionSegments(final CreateTableStatement sqlStatement, final IdentifierValue databaseName,
474                                                                           final IdentifierValue schemaName, final IdentifierValue tableName) {
475         Collection<ProjectionSegment> result = new LinkedList<>();
476         for (ColumnDefinitionSegment each : sqlStatement.getColumnDefinitions()) {
477             each.getColumnName().setColumnBoundInfo(
478                     new ColumnSegmentBoundInfo(new TableSegmentBoundInfo(databaseName, schemaName), tableName, each.getColumnName().getIdentifier(), TableSourceType.TEMPORARY_TABLE));
479             result.add(new ColumnProjectionSegment(each.getColumnName()));
480         }
481         return result;
482     }
483     
484     private static Optional<SimpleTableSegmentBinderContext> createSimpleTableSegmentBinderContextWithMetaData(final SimpleTableSegment segment, final ShardingSphereSchema schema,
485                                                                                                                final IdentifierValue databaseName, final IdentifierValue schemaName,
486                                                                                                                final SQLStatementBinderContext binderContext, final IdentifierValue tableName) {
487         Collection<ProjectionSegment> projectionSegments = new LinkedList<>();
488         QuoteCharacter quoteCharacter = new DatabaseTypeRegistry(binderContext.getSqlStatement().getDatabaseType()).getDialectDatabaseMetaData().getQuoteCharacter();
489         for (ShardingSphereColumn each : schema.getTable(tableName).getAllColumns()) {
490             ColumnProjectionSegment columnProjectionSegment = new ColumnProjectionSegment(createColumnSegment(segment, databaseName, schemaName, each, quoteCharacter, tableName));
491             columnProjectionSegment.setVisible(each.isVisible());
492             projectionSegments.add(columnProjectionSegment);
493         }
494         SimpleTableSegmentBinderContext result = new SimpleTableSegmentBinderContext(projectionSegments, TableSourceType.PHYSICAL_TABLE, tableName);
495         segment.getOwner().ifPresent(result::setOriginalOwner);
496         return Optional.of(result);
497     }
498     
499     private static ColumnSegment createColumnSegment(final SimpleTableSegment segment, final IdentifierValue databaseName, final IdentifierValue schemaName,
500                                                      final ShardingSphereColumn column, final QuoteCharacter quoteCharacter, final IdentifierValue tableName) {
501         ColumnSegment result = new ColumnSegment(0, 0, new IdentifierValue(column.getName(), quoteCharacter));
502         result.setOwner(new OwnerSegment(0, 0, segment.getAlias().orElse(tableName)));
503         result.setColumnBoundInfo(
504                 new ColumnSegmentBoundInfo(new TableSegmentBoundInfo(databaseName, schemaName), tableName, new IdentifierValue(column.getName(), quoteCharacter), TableSourceType.PHYSICAL_TABLE));
505         return result;
506     }
