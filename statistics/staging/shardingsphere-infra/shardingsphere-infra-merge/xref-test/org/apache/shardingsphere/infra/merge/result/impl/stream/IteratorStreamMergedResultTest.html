24  import java.io.ByteArrayInputStream;
25  import java.io.InputStreamReader;
26  import java.io.Reader;
27  import java.sql.SQLException;
28  import java.util.Arrays;
29  import java.util.Calendar;
30  import java.util.Collections;
31  
32  import static org.hamcrest.MatcherAssert.assertThat;
33  import static org.hamcrest.Matchers.is;
34  import static org.junit.jupiter.api.Assertions.assertFalse;
35  import static org.junit.jupiter.api.Assertions.assertNull;
36  import static org.junit.jupiter.api.Assertions.assertThrows;
37  import static org.junit.jupiter.api.Assertions.assertTrue;
38  import static org.mockito.Mockito.mock;
39  import static org.mockito.Mockito.when;
40  
41  class IteratorStreamMergedResultTest {
42      
43      @Test
44      void assertNextWhenCurrentHasRow() throws SQLException {
45          QueryResult queryResult = mock(QueryResult.class);
46          when(queryResult.next()).thenReturn(true);
47          assertTrue(new IteratorStreamMergedResult(Collections.singletonList(queryResult)).next());
48      }
49      
50      @Test
51      void assertNextWhenNoResults() throws SQLException {
52          assertFalse(new IteratorStreamMergedResult(Collections.singletonList(mock(QueryResult.class))).next());
53      }
54      
55      @Test
56      void assertNextWithSwitchToFollowingResult() throws SQLException {
57          QueryResult first = mock(QueryResult.class);
58          QueryResult second = mock(QueryResult.class);
59          when(second.next()).thenReturn(true);
60          assertTrue(new IteratorStreamMergedResult(Arrays.asList(first, second)).next());
61      }
62      
63      @Test
64      void assertNextUntilDataFound() throws SQLException {
65          QueryResult first = mock(QueryResult.class);
66          QueryResult second = mock(QueryResult.class);
67          QueryResult third = mock(QueryResult.class);
68          when(third.next()).thenReturn(true);
69          assertTrue(new IteratorStreamMergedResult(Arrays.asList(first, second, third)).next());
70      }
71      
72      @Test
73      void assertNextFalseWhenAllEmpty() throws SQLException {
74          assertFalse(new IteratorStreamMergedResult(Arrays.asList(mock(QueryResult.class), mock(QueryResult.class))).next());
75      }
76      
77      @Test
78      void assertGetValueAndTrackWasNull() throws SQLException {
79          QueryResult queryResult = mock(QueryResult.class);
80          when(queryResult.getValue(1, String.class)).thenReturn("v");
81          when(queryResult.wasNull()).thenReturn(true);
82          IteratorStreamMergedResult mergedResult = new IteratorStreamMergedResult(Collections.singletonList(queryResult));
83          Object actual = mergedResult.getValue(1, String.class);
84          assertThat(actual, is("v"));
85          assertTrue(mergedResult.wasNull());
86      }
87      
88      @Test
89      void assertGetCalendarValueAndTrackWasNull() throws SQLException {
90          QueryResult queryResult = mock(QueryResult.class);
91          Calendar calendar = Calendar.getInstance();
92          when(queryResult.getCalendarValue(1, String.class, calendar)).thenReturn("v");
93          IteratorStreamMergedResult mergedResult = new IteratorStreamMergedResult(Collections.singletonList(queryResult));
94          Object actual = mergedResult.getCalendarValue(1, String.class, calendar);
95          assertThat(actual, is("v"));
96          assertFalse(mergedResult.wasNull());
97      }
98      
99      @Test
100     void assertGetInputStreamAndTrackWasNull() throws SQLException {
101         QueryResult queryResult = mock(QueryResult.class);
102         ByteArrayInputStream inputStream = new ByteArrayInputStream(new byte[]{1});
103         when(queryResult.getInputStream(1, "stream")).thenReturn(inputStream);
104         IteratorStreamMergedResult mergedResult = new IteratorStreamMergedResult(Collections.singletonList(queryResult));
105         ByteArrayInputStream actual = (ByteArrayInputStream) mergedResult.getInputStream(1, "stream");
106         assertThat(actual, is(inputStream));
107         assertFalse(mergedResult.wasNull());
108     }
109     
110     @Test
111     void assertGetCharacterStreamAndTrackWasNull() throws SQLException {
112         QueryResult queryResult = mock(QueryResult.class);
113         Reader reader = new InputStreamReader(new ByteArrayInputStream(new byte[0]));
114         when(queryResult.getCharacterStream(1)).thenReturn(reader);
115         IteratorStreamMergedResult mergedResult = new IteratorStreamMergedResult(Collections.singletonList(queryResult));
116         assertThat(mergedResult.getCharacterStream(1), is(reader));
117         assertFalse(mergedResult.wasNull());
118     }
119     
120     @Test
121     void assertThrowWhenCurrentQueryResultIsNull() throws ReflectiveOperationException {
122         QueryResult queryResult = mock(QueryResult.class);
123         IteratorStreamMergedResult mergedResult = new IteratorStreamMergedResult(Collections.singletonList(queryResult));
124         Plugins.getMemberAccessor().set(StreamMergedResult.class.getDeclaredField("currentQueryResult"), mergedResult, null);
125         SQLException actual = assertThrows(SQLException.class, () -> mergedResult.getValue(1, Object.class));
126         assertThat(actual.getMessage(), is("Current ResultSet is null, ResultSet perhaps end of next"));
127     }
128     
129     @Test
130     void assertWasNull() throws SQLException {
131         QueryResult queryResult = mock(QueryResult.class);
132         when(queryResult.wasNull()).thenReturn(true);
133         IteratorStreamMergedResult mergedResult = new IteratorStreamMergedResult(Collections.singletonList(queryResult));
134         assertNull(mergedResult.getCharacterStream(1));
135         assertTrue(mergedResult.wasNull());
136     }
137 }
