54  /**
55   * Common table expression segment binder.
56   */
57  @NoArgsConstructor(access = AccessLevel.PRIVATE)
58  public final class CommonTableExpressionSegmentBinder {
59      
60      /**
61       * Bind common table expression segment.
62       *
63       * @param segment common table expression segment
64       * @param binderContext SQL statement binder context
65       * @param externalTableBinderContexts external table binder contexts
66       * @param recursive recursive
67       * @return bound common table expression segment
68       */
69      public static CommonTableExpressionSegment bind(final CommonTableExpressionSegment segment, final SQLStatementBinderContext binderContext,
70                                                      final Multimap<CaseInsensitiveString, TableSegmentBinderContext> externalTableBinderContexts, final boolean recursive) {
71          if (segment.getAliasName().isPresent()) {
72              ShardingSpherePreconditions.checkState(!binderContext.getCommonTableExpressionsSegmentsUniqueAliases().contains(segment.getAliasName().get()),
73                      () -> new DuplicateCommonTableExpressionAliasException(segment.getAliasName().get()));
74              binderContext.getCommonTableExpressionsSegmentsUniqueAliases().add(segment.getAliasName().get());
75          }
76          if (recursive && segment.getAliasName().isPresent()) {
77              binderContext.getExternalTableBinderContexts().put(new CaseInsensitiveString(segment.getAliasName().get()),
78                      new SimpleTableSegmentBinderContext(segment.getColumns().stream().map(ColumnProjectionSegment::new).collect(Collectors.toList()), TableSourceType.TEMPORARY_TABLE));
79          }
80          SubqueryTableSegment subqueryTableSegment = new SubqueryTableSegment(segment.getStartIndex(), segment.getStopIndex(), segment.getSubquery());
81          segment.getAliasSegment().ifPresent(subqueryTableSegment::setAlias);
82          SubqueryTableSegment boundSubquerySegment =
83                  SubqueryTableSegmentBinder.bind(subqueryTableSegment, binderContext, LinkedHashMultimap.create(), binderContext.getExternalTableBinderContexts(), true);
84          CommonTableExpressionSegment result = new CommonTableExpressionSegment(
85                  segment.getStartIndex(), segment.getStopIndex(), boundSubquerySegment.getAliasSegment().orElse(null), boundSubquerySegment.getSubquery());
86          Multimap<CaseInsensitiveString, TableSegmentBinderContext> currentTableBinderContexts =
87                  createCurrentTableBinderContexts(segment.getColumns(), binderContext, boundSubquerySegment.getSubquery().getSelect());
88          segment.getColumns()
89                  .forEach(each -> result.getColumns().add(ColumnSegmentBinder.bind(each, SegmentType.DEFINITION_COLUMNS, binderContext, currentTableBinderContexts, LinkedHashMultimap.create())));
90          putExternalTableBinderContext(segment, externalTableBinderContexts, recursive, currentTableBinderContexts);
91          return result;
92      }
93      
94      private static Multimap<CaseInsensitiveString, TableSegmentBinderContext> createCurrentTableBinderContexts(final Collection<ColumnSegment> definitionColumns,
95                                                                                                                 final SQLStatementBinderContext binderContext, final SelectStatement selectStatement) {
96          Collection<ProjectionSegment> subqueryProjections = SubqueryTableBindUtils.createSubqueryProjections(
97                  selectStatement.getProjections().getProjections(), new IdentifierValue(""), binderContext.getSqlStatement().getDatabaseType(), TableSourceType.TEMPORARY_TABLE);
98          Multimap<CaseInsensitiveString, TableSegmentBinderContext> result = LinkedHashMultimap.create();
99          Collection<ProjectionSegment> boundDefinitionColumns = createBoundDefinitionColumns(definitionColumns, subqueryProjections);
100         Collection<ProjectionSegment> boundProjectionSegments = definitionColumns.isEmpty() ? subqueryProjections : new LinkedList<>(boundDefinitionColumns);
101         SimpleTableSegmentBinderContext tableSegmentBinderContext = new SimpleTableSegmentBinderContext(boundProjectionSegments, TableSourceType.TEMPORARY_TABLE);
102         tableSegmentBinderContext.setFromWithSegment(true);
103         result.put(new CaseInsensitiveString(""), tableSegmentBinderContext);
104         return result;
105     }
106     
107     private static Collection<ProjectionSegment> createBoundDefinitionColumns(final Collection<ColumnSegment> definitionColumns, final Collection<ProjectionSegment> projectionSegments) {
108         Collection<ProjectionSegment> result = new LinkedList<>();
109         Collection<ColumnSegment> boundDefinitionColumns = bindDefinitionColumns(definitionColumns, new ArrayList<>(projectionSegments));
110         boundDefinitionColumns.forEach(each -> result.add(new ColumnProjectionSegment(each)));
111         return result;
112     }
113     
114     private static Collection<ColumnSegment> bindDefinitionColumns(final Collection<ColumnSegment> definitionColumns, final List<ProjectionSegment> projectionSegments) {
115         ShardingSpherePreconditions.checkState(definitionColumns.isEmpty() || definitionColumns.size() == projectionSegments.size(),
116                 DifferenceInColumnCountOfSelectListAndColumnNameListException::new);
117         int index = 0;
118         Collection<ColumnSegment> result = new LinkedList<>();
119         for (ColumnSegment each : definitionColumns) {
120             ColumnSegment boundColumnSegment = copy(each);
121             ProjectionSegment projectionSegment = projectionSegments.get(index);
122             if (projectionSegment instanceof ColumnProjectionSegment) {
123                 boundColumnSegment.setColumnBoundInfo(createColumnSegmentBoundInfo(each, ((ColumnProjectionSegment) projectionSegment).getColumn()));
124             }
125             result.add(boundColumnSegment);
126             index++;
127         }
128         return result;
129     }
130     
131     private static ColumnSegment copy(final ColumnSegment segment) {
132         ColumnSegment result = new ColumnSegment(segment.getStartIndex(), segment.getStopIndex(), segment.getIdentifier());
133         result.setNestedObjectAttributes(segment.getNestedObjectAttributes());
134         segment.getOwner().ifPresent(result::setOwner);
135         result.setVariable(segment.isVariable());
136         segment.getLeftParentheses().ifPresent(result::setLeftParentheses);
137         segment.getRightParentheses().ifPresent(result::setRightParentheses);
138         return result;
139     }
140     
141     private static ColumnSegmentBoundInfo createColumnSegmentBoundInfo(final ColumnSegment segment, final ColumnSegment inputColumnSegment) {
142         IdentifierValue originalDatabase = null == inputColumnSegment ? null : inputColumnSegment.getColumnBoundInfo().getOriginalDatabase();
143         IdentifierValue originalSchema = null == inputColumnSegment ? null : inputColumnSegment.getColumnBoundInfo().getOriginalSchema();
144         IdentifierValue segmentOriginalTable = segment.getColumnBoundInfo().getOriginalTable();
145         IdentifierValue originalTable = Strings.isNullOrEmpty(segmentOriginalTable.getValue())
146                 ? Optional.ofNullable(inputColumnSegment).map(optional -> optional.getColumnBoundInfo().getOriginalTable()).orElse(segmentOriginalTable)
147                 : segmentOriginalTable;
148         IdentifierValue segmentOriginalColumn = segment.getColumnBoundInfo().getOriginalColumn();
149         IdentifierValue originalColumn = Optional.ofNullable(inputColumnSegment).map(optional -> optional.getColumnBoundInfo().getOriginalColumn()).orElse(segmentOriginalColumn);
150         return new ColumnSegmentBoundInfo(new TableSegmentBoundInfo(originalDatabase, originalSchema), originalTable, originalColumn, TableSourceType.TEMPORARY_TABLE);
151     }
152     
153     private static void putExternalTableBinderContext(final CommonTableExpressionSegment segment, final Multimap<CaseInsensitiveString, TableSegmentBinderContext> externalTableBinderContexts,
154                                                       final boolean recursive, final Multimap<CaseInsensitiveString, TableSegmentBinderContext> currentTableBinderContexts) {
155         if (!segment.getAliasName().isPresent()) {
156             return;
157         }
158         if (recursive && segment.getAliasName().isPresent()) {
159             externalTableBinderContexts.removeAll(new CaseInsensitiveString(segment.getAliasName().get()));
160         }
161         externalTableBinderContexts.putAll(new CaseInsensitiveString(segment.getAliasName().get()), currentTableBinderContexts.values());
162     }
