18  package org.apache.shardingsphere.mcp.support.descriptor;
19  
20  import org.apache.shardingsphere.mcp.api.capability.prompt.MCPPromptDescriptor;
21  import org.apache.shardingsphere.mcp.api.capability.resource.MCPResourceAnnotations;
22  import org.apache.shardingsphere.mcp.api.capability.resource.MCPResourceDescriptor;
23  import org.apache.shardingsphere.mcp.api.capability.tool.MCPToolAnnotations;
24  import org.apache.shardingsphere.mcp.api.capability.tool.MCPToolDescriptor;
25  import org.junit.jupiter.api.Test;
26  
27  import java.util.Collection;
28  import java.util.List;
29  import java.util.Map;
30  
31  import static org.hamcrest.MatcherAssert.assertThat;
32  import static org.hamcrest.Matchers.is;
33  
34  class MCPProtocolDescriptorCatalogTest {
35      
36      @Test
37      void assertGetResourceDescriptors() {
38          Collection<MCPResourceDescriptor> resourceDescriptors = List.of(createResourceDescriptor("shardingsphere://test"));
39          MCPProtocolDescriptorCatalog actual = new MCPProtocolDescriptorCatalog(resourceDescriptors, List.of(), List.of(), List.of());
40          assertThat(actual.getResourceDescriptors(), is(resourceDescriptors));
41      }
42      
43      @Test
44      void assertGetResourceTemplateDescriptors() {
45          Collection<MCPResourceDescriptor> resourceTemplateDescriptors = List.of(createResourceDescriptor("shardingsphere://test/{id}"));
46          MCPProtocolDescriptorCatalog actual = new MCPProtocolDescriptorCatalog(List.of(), resourceTemplateDescriptors, List.of(), List.of());
47          assertThat(actual.getResourceTemplateDescriptors(), is(resourceTemplateDescriptors));
48      }
49      
50      @Test
51      void assertGetToolDescriptors() {
52          Collection<MCPToolDescriptor> toolDescriptors = List.of(createToolDescriptor());
53          MCPProtocolDescriptorCatalog actual = new MCPProtocolDescriptorCatalog(List.of(), List.of(), toolDescriptors, List.of());
54          assertThat(actual.getToolDescriptors(), is(toolDescriptors));
55      }
56      
57      @Test
58      void assertGetPromptDescriptors() {
59          Collection<MCPPromptDescriptor> promptDescriptors = List.of(createPromptDescriptor());
60          MCPProtocolDescriptorCatalog actual = new MCPProtocolDescriptorCatalog(List.of(), List.of(), List.of(), promptDescriptors);
61          assertThat(actual.getPromptDescriptors(), is(promptDescriptors));
62      }
63      
64      @Test
65      void assertGetAllResourceDescriptors() {
66          MCPResourceDescriptor resourceDescriptor = createResourceDescriptor("shardingsphere://test");
67          MCPResourceDescriptor resourceTemplateDescriptor = createResourceDescriptor("shardingsphere://test/{id}");
68          MCPProtocolDescriptorCatalog actual = new MCPProtocolDescriptorCatalog(List.of(resourceDescriptor), List.of(resourceTemplateDescriptor), List.of(), List.of());
69          assertThat(actual.getAllResourceDescriptors(), is(List.of(resourceDescriptor, resourceTemplateDescriptor)));
70      }
71      
72      private MCPResourceDescriptor createResourceDescriptor(final String uriTemplate) {
73          return new MCPResourceDescriptor(uriTemplate, "test-resource", "Test Resource", "Read a test resource.", "application/json", MCPResourceAnnotations.EMPTY, Map.of());
74      }
75      
76      private MCPToolDescriptor createToolDescriptor() {
77          return new MCPToolDescriptor("database_gateway_test_tool", "Test Tool", "Run a test tool.", Map.of(), Map.of(),
78                  MCPToolAnnotations.builder().title("Test Tool").readOnlyHint(true).destructiveHint(false).idempotentHint(true).openWorldHint(true).build(), Map.of());
79      }
80      
81      private MCPPromptDescriptor createPromptDescriptor() {
82          return new MCPPromptDescriptor("test_prompt", "Test Prompt", "Render a test prompt.", List.of(), Map.of());
83      }
84  }
