122     @Test
123     void assertCreateSanitizesSerializationFailure(final LogCaptureAssertion logCaptureAssertion) {
124         Map<String, Object> payload = Map.of("foo_field", "foo_value");
125         IllegalStateException expectedCause = new IllegalStateException("foo_sensitive_detail");
126         JsonException expectedException = mock(JsonException.class);
127         when(expectedException.getMessage()).thenReturn("foo_sensitive_detail");
128         when(expectedException.getCause()).thenReturn(expectedCause);
129         MCPToolDescriptor descriptor = createToolDescriptorWithoutOutputSchema("fixture_ping");
130         try (MockedStatic<JsonEngine> mockedJsonEngine = mockStatic(JsonEngine.class)) {
131             mockedJsonEngine.when(() -> JsonEngine.marshal(payload)).thenThrow(expectedException);
132             McpError actual = assertThrows(McpError.class, () -> new MCPCallToolResultFactory().create(descriptor, new MCPMapPayload(payload)));
133             assertThat(actual.getJsonRpcError().code(), is(McpSchema.ErrorCodes.INTERNAL_ERROR));
134             assertThat(actual.getJsonRpcError().message(), is("Service is temporarily unavailable."));
135             Map<?, ?> actualData = (Map<?, ?>) actual.getJsonRpcError().data();
136             assertThat(actualData.get("error_id"), isA(String.class));
137             assertThat(actualData, is(Map.of("response_mode", "recovery", "summary", "Service is temporarily unavailable.", "error_id", actualData.get("error_id"))));
138             logCaptureAssertion.assertErrorLog(actualException -> {
139                 assertThat(((ThrowableProxy) actualException).getThrowable(), sameInstance(expectedException));
140                 assertThat(((ThrowableProxy) actualException).getThrowable().getCause(), sameInstance(expectedCause));
141             });
142         }
143     }

170     @Test
171     void assertCreateWithExecutedResultSet() {
172         SQLExecutionColumnDefinition column = new SQLExecutionColumnDefinition("order_id", "BIGINT", "BIGINT", false);
173         MCPSuccessPayload response = SQLExecutionPayload.executed(SQLExecutionResult.resultSet(SupportedMCPStatement.DML, "SELECT",
174                 List.of(column), List.of(List.of(1L)), true, 1, 0, "WITH changed AS (...) SELECT order_id FROM changed"));
175         CallToolResult actual = createRealDescriptorCallToolResult("database_gateway_execute_update", response);
176         Map<String, Object> actualPayload = getStructuredContent(actual);
177         assertFalse(actual.isError(), () -> String.valueOf(actualPayload));
178         assertThat(actualPayload.get("columns"), is(List.of(column)));
179         assertThat(actualPayload.get("rows"), is(List.of(List.of(1L))));
180         assertThat(((Map<?, ?>) ((List<?>) actualPayload.get("next_actions")).getFirst()).get("reason"), is(
181                 "The side-effecting statement already executed and returned truncated rows; do not replay it automatically. "
182                         + "Use a separate read-only query if more data is needed."));
183     }

188         SQLExecutionColumnDefinition column = new SQLExecutionColumnDefinition("foo_observed_at", "TIMESTAMP_WITH_TIMEZONE", "TIMESTAMP WITH TIME ZONE", false);

199         assertThat(actualTextPayload.get("row_objects"), is(List.of(Map.of("foo_observed_at", -0.5D))));

324         for (MCPToolDescriptor each : ToolDefinitionRegistry.getSupportedToolDescriptors()) {

337     private MCPSuccessPayload createMaskPlanResponse() {
338         WorkflowContextSnapshot snapshot = createMaskSnapshot();
339         Map<String, Object> payload = WorkflowPlanPayloadBuilder.buildWithArtifacts(snapshot, snapshot.getRequest());
340         payload.put("masked_property_preview", Map.of("primary",
341                 WorkflowArtifactMaskUtils.maskPropertyMap(snapshot.getRequest().getPrimaryAlgorithmProperties(), snapshot.getPropertyRequirements())));
342         return new MCPMapPayload(payload);
343     }

345     private WorkflowContextSnapshot createMaskSnapshot() {
346         WorkflowRequest request = createWorkflowRequest();
347         request.getPrimaryAlgorithmProperties().put("first-n", "3");
348         request.getPrimaryAlgorithmProperties().put("last-m", "2");
349         request.getPrimaryAlgorithmProperties().put("replace-char", "*");
350         WorkflowContextSnapshot result = createWorkflowContextSnapshot(MaskFeatureDefinition.WORKFLOW_KIND, request);
351         result.getPropertyRequirements().add(new AlgorithmPropertyRequirement("primary", "first-n", true, false, "from", ""));
352         result.getRuleArtifacts().add(new RuleArtifact("create", "CREATE MASK RULE orders (TYPE(NAME='keep_first_n_last_m'))"));
353         return result;
354     }

364     private WorkflowContextSnapshot createEncryptSnapshot() {
365         EncryptWorkflowRequest request = new EncryptWorkflowRequest();
366         request.setDatabase("logic_db");
367         request.setSchema("public");
368         request.setTable("orders");
369         request.getPrimaryAlgorithmProperties().put("aes-key-value", "123456");
370         WorkflowContextSnapshot result = createWorkflowContextSnapshot(EncryptFeatureDefinition.WORKFLOW_KIND, request);
371         result.getPropertyRequirements().add(new AlgorithmPropertyRequirement("primary", "aes-key-value", true, true, "key", ""));
372         result.getRuleArtifacts().add(new RuleArtifact("create", "CREATE ENCRYPT RULE orders (PROPERTIES('aes-key-value'='123456'))"));
373         return result;
374     }

384     private WorkflowContextSnapshot createWorkflowContextSnapshot(final WorkflowKind workflowKind, final WorkflowRequest request) {
385         WorkflowContextSnapshot result = new WorkflowContextSnapshot();
386         result.setPlanId("plan-1");
387         result.setWorkflowKind(workflowKind);
388         result.setStatus(WorkflowLifecycle.STATUS_PLANNED);
389         result.setRequest(request);
390         result.setClarifiedIntent(new ClarifiedIntent());
391         result.setInteractionPlan(createInteractionPlan());
392         return result;
393     }

396         InteractionPlan result = new InteractionPlan();
397         result.setCurrentStep("review");
398         result.setDeliveryMode("interactive");
399         result.setExecutionMode("review-then-execute");
400         return result;
401     }
