54              .filter(CoreHandlerProvider.class::isInstance).map(CoreHandlerProvider.class::cast).findFirst().orElseThrow();

56      @ParameterizedTest(name = "{0}")
57      @MethodSource("metadataResourceCases")
58      void assertGetResourceHandlersWithMetadata(final String name, final String template, final String uri, final String nameField, final Collection<String> expectedNames) {
59          MetadataResourceHandler actualHandler = (MetadataResourceHandler) provider.getResourceHandlers().stream()
60                  .filter(handler -> template.equals(handler.getResourceUriTemplate())).findFirst().orElseThrow();
61          try (RequestContextFixture fixture = ResourceTestDataFactory.createRequestContextFixture(ResourceTestDataFactory.createDatabaseMetadata())) {
62              MCPResourceURIVariables variables = new MCPUriTemplate(template).parse(uri).orElseThrow();
63              Map<String, Object> actual = actualHandler.handle(fixture.getRequestContext(), variables).toPayload();
64              assertThat(actual.get("count"), is(expectedNames.size()));
65              assertThat(((Collection<?>) actual.get("items")).stream().map(item -> ((Map<?, ?>) item).get(nameField)).toList(), is(expectedNames));
66              assertThat(((Map<?, ?>) actual.get("self_resource")).get("uri"), is(uri));
67          }
68      }

101     @ParameterizedTest(name = "{0}")
102     @MethodSource("governanceResourceCases")
103     void assertGetResourceHandlersWithGovernanceMetadata(final String name, final String template, final String uri, final String sql, final Map<String, Object> expectedRow) {
104         MCPFeatureRequestContext requestContext = mock(MCPFeatureRequestContext.class);
105         MCPFeatureQueryFacade queryFacade = mock(MCPFeatureQueryFacade.class);
106         when(requestContext.getQueryFacade()).thenReturn(queryFacade);
107         when(queryFacade.query("foo_db", sql)).thenReturn(List.of(expectedRow));
108         MetadataResourceHandler actualHandler = (MetadataResourceHandler) provider.getResourceHandlers().stream()
109                 .filter(handler -> template.equals(handler.getResourceUriTemplate())).findFirst().orElseThrow();
110         Map<String, Object> actual = actualHandler.handle(requestContext, new MCPUriTemplate(template).parse(uri).orElseThrow()).toPayload();
111         assertThat(actual.get("items"), is(List.of(expectedRow)));
112         assertThat(((Map<?, ?>) actual.get("self_resource")).get("uri"), is(uri));
113         verify(queryFacade).query("foo_db", sql);
114     }

142     @Test
143     void assertGetResourceHandlers() {
144         Collection<MCPResourceHandler<?>> actual = new CoreHandlerProvider().getResourceHandlers();
145         assertThat(actual.size(), is(27));
146         List<String> actualUris = actual.stream().map(MCPResourceHandler::getResourceUriTemplate).toList();
147         assertTrue(actualUris.contains("shardingsphere://capabilities"));
148         assertTrue(actualUris.contains("shardingsphere://guidance"));
149         assertTrue(actualUris.contains("shardingsphere://runtime"));
150         assertTrue(actualUris.contains("shardingsphere://workflows/{plan_id}"));
151         assertTrue(actualUris.contains("shardingsphere://databases/{database}/storage-units/{storageUnit}"));
152         assertTrue(actualUris.contains("shardingsphere://databases/{database}/single-table/default-storage-unit"));
153         assertTrue(actualUris.contains("shardingsphere://databases/{database}/schemas/{schema}/tables/{table}/indexes/{index}"));
