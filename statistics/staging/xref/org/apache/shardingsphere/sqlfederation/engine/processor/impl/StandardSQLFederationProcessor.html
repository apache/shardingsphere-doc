60  /**
61   * Standard sql federation processor.
62   */
63  @RequiredArgsConstructor
64  public final class StandardSQLFederationProcessor implements SQLFederationProcessor {
65      
66      private final ShardingSphereMetaData metaData;
67      
68      private final ShardingSphereStatistics statistics;
69      
70      private final JDBCExecutor jdbcExecutor;
71      
72      @Override
73      public void registerExecutor(final DriverExecutionPrepareEngine<JDBCExecutionUnit, Connection> prepareEngine, final JDBCExecutorCallback<? extends ExecuteResult> callback,
74                                   final String databaseName, final String schemaName, final SQLFederationContext federationContext, final OptimizerContext optimizerContext,
75                                   final SchemaPlus schemaPlus) {
76          if (null == schemaPlus) {
77              return;
78          }
79          SQLFederationExecutorContext executorContext = new SQLFederationExecutorContext(databaseName, schemaName, metaData.getProps());
80          EnumerableScanExecutor scanExecutor =
81                  new EnumerableScanExecutor(prepareEngine, jdbcExecutor, callback, optimizerContext, executorContext, federationContext, metaData.getGlobalRuleMetaData(), statistics);
82          Collection<SimpleTableSegment> simpleTables = federationContext.getQueryContext().getSqlStatementContext() instanceof TableAvailable
83                  ? ((TableAvailable) federationContext.getQueryContext().getSqlStatementContext()).getTablesContext().getSimpleTables()
84                  : Collections.emptyList();
85          for (SimpleTableSegment each : simpleTables) {
86              Table table = schemaPlus.tables().get(each.getTableName().getIdentifier().getValue());
87              if (table instanceof SQLFederationTable) {
88                  ((SQLFederationTable) table).setScanExecutor(scanExecutor);
89              }
90          }
91      }
92      
93      @Override
94      public void unregisterExecutor(final QueryContext queryContext, final SchemaPlus schemaPlus) {
95          Collection<SimpleTableSegment> simpleTables = queryContext.getSqlStatementContext() instanceof TableAvailable
96                  ? ((TableAvailable) queryContext.getSqlStatementContext()).getTablesContext().getSimpleTables()
97                  : Collections.emptyList();
98          for (SimpleTableSegment each : simpleTables) {
99              Table table = schemaPlus.tables().get(each.getTableName().getIdentifier().getValue());
100             if (table instanceof SQLFederationTable) {
101                 ((SQLFederationTable) table).clearScanExecutor();
102             }
103         }
104     }
105     
106     @SuppressWarnings("unchecked")
107     @Override
108     public ResultSet executePlan(final DriverExecutionPrepareEngine<JDBCExecutionUnit, Connection> prepareEngine, final JDBCExecutorCallback<? extends ExecuteResult> callback,
109                                  final SQLFederationExecutionPlan executionPlan, final SqlToRelConverter converter, final SQLFederationContext federationContext,
110                                  final SchemaPlus schemaPlus) {
111         Bindable<Object> executablePlan = EnumerableInterpretable.toBindable(Collections.emptyMap(), null, (EnumerableRel) executionPlan.getPhysicalPlan(), Prefer.ARRAY);
112         Map<String, Object> params = createParameters(federationContext.getQueryContext().getParameters());
113         Enumerator<Object> enumerator = executablePlan.bind(new SQLFederationBindContext(converter, params)).enumerator();
114         return new SQLFederationResultSet(enumerator, schemaPlus, (SelectStatementContext) federationContext.getQueryContext().getSqlStatementContext(), executionPlan.getResultColumnType());
115     }
116     
117     private Map<String, Object> createParameters(final List<Object> params) {
118         Map<String, Object> result = new HashMap<>(params.size(), 1F);
119         int index = 0;
120         for (Object each : params) {
121             result.put("?" + index++, each);
122         }
123         return result;
124     }
125     
126     @Override
127     public Convention getConvention() {
128         return EnumerableConvention.INSTANCE;
129     }
130 }
