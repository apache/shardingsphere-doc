37  /**
38   * Workflow validation support.
39   */
40  public final class WorkflowValidationSupport {
41      
42      /**
43       * Check whether workflow validation can proceed.
44       *
45       * @param sessionId session identifier
46       * @param snapshot workflow snapshot
47       * @return rejection response or empty map
48       */
49      public Map<String, Object> checkValidatePreconditions(final String sessionId, final WorkflowContextSnapshot snapshot) {
50          if (!WorkflowLifecycleUtils.isOwnedBySession(sessionId, snapshot)) {
51              return createRejectedResponse(snapshot, WorkflowIssueCode.SESSION_OWNERSHIP_MISMATCH, "The workflow plan belongs to another MCP session.",
52                      "Continue the workflow from the same session that created the plan.");
53          }
54          if (!isValidatableStatus(snapshot)) {
55              return createRejectedResponse(snapshot, WorkflowIssueCode.WORKFLOW_STATUS_INVALID,
56                      String.format("Workflow status `%s` cannot enter validate in the current lifecycle.", snapshot.getStatus()),
57                      "Execute the workflow first or continue from a validatable status.");
58          }
59          return Map.of();
60      }
61      
62      /**
63       * Validate workflow snapshot and persist validation response.
64       *
65       * @param workflowSessionContext workflow session context
66       * @param sessionId session identifier
67       * @param snapshot workflow snapshot
68       * @param validationReportSupplier validation report supplier
69       * @return validation response
70       */
71      public Map<String, Object> validateAndFinalize(final WorkflowSessionContext workflowSessionContext, final String sessionId, final WorkflowContextSnapshot snapshot,
72                                                     final Supplier<ValidationReport> validationReportSupplier) {
73          Map<String, Object> rejectedResponse = checkValidatePreconditions(sessionId, snapshot);
74          if (!rejectedResponse.isEmpty()) {
75              return rejectedResponse;
76          }
77          ValidationReport validationReport = validationReportSupplier.get();
78          snapshot.setValidationReport(validationReport);
79          return finalizeValidation(workflowSessionContext, snapshot, validationReport);
80      }
81      
82      /**
83       * Resolve overall validation status from validation sections.
84       *
85       * @param validationSections validation sections
86       * @return overall validation status
87       */
88      public String resolveOverallStatus(final ValidationSection... validationSections) {
89          for (ValidationSection each : validationSections) {
90              if (null != each && WorkflowLifecycle.STATUS_FAILED.equals(each.getStatus())) {
91                  return WorkflowLifecycle.STATUS_FAILED;
92              }
93          }
94          return WorkflowLifecycle.STATUS_PASSED;
95      }
96      
97      /**
98       * Persist validation result and create response payload.
99       *
100      * @param workflowSessionContext workflow session context
101      * @param snapshot workflow snapshot
102      * @param validationReport validation report
103      * @return validation response
104      */
105     public Map<String, Object> finalizeValidation(final WorkflowSessionContext workflowSessionContext, final WorkflowContextSnapshot snapshot,
106                                                   final ValidationReport validationReport) {
107         String validationStatus = resolveValidationStatus(validationReport);
108         workflowSessionContext.persist(snapshot, WorkflowLifecycle.STEP_VALIDATED, validationStatus);
109         Map<String, Object> result = new LinkedHashMap<>(16, 1F);
110         result.put("response_mode", "validation");
111         result.put(MCPPayloadFieldNames.SUMMARY, createValidationSummary(snapshot, validationReport));
112         result.put(WorkflowFieldNames.PLAN_ID, snapshot.getPlanId());
113         result.put("status", validationStatus);
114         result.put("issues", createValidationIssues(validationReport));
115         result.putAll(validationReport.toMap());
116         WorkflowGuidancePayloadBuilder.appendValidationGuidance(result, snapshot, validationReport);
117         return result;
118     }
119     
120     /**
121      * Create one validation mismatch entry.
122      *
123      * @param code mismatch code
124      * @param layer mismatch layer
125      * @param expected expected value
126      * @param actual actual value
127      * @param impact mismatch impact
128      * @param remediation remediation guidance
129      * @return mismatch payload
130      */
131     public Map<String, Object> createMismatch(final String code, final String layer, final String expected,
132                                               final String actual, final String impact, final String remediation) {
133         Map<String, Object> result = new LinkedHashMap<>(6, 1F);
134         result.put("code", code);
135         result.put("layer", layer);
136         result.put("expected", expected);
137         result.put("actual", actual);
138         result.put("impact", impact);
139         result.put("remediation", remediation);
140         return result;
141     }
142     
143     /**
144      * Create one SQL executability issue.
145      *
146      * @param message issue message
147      * @param sql SQL artifact
148      * @return issue payload
149      */
150     public Map<String, Object> createSQLExecutabilityIssue(final String message, final String sql) {
151         return new WorkflowIssue(WorkflowIssueCode.SQL_EXECUTABILITY_FAILED, "error", WorkflowLifecycle.STEP_REVIEW,
152                 message, "Regenerate the workflow artifact through the feature planner before approval.", true, Map.of("sql", sql)).toMap();
153     }
154     
155     private List<Map<String, Object>> createValidationIssues(final ValidationReport validationReport) {
156         if (!WorkflowLifecycle.STATUS_FAILED.equals(validationReport.getOverallStatus())) {
157             return List.of();
158         }
159         return List.of(new WorkflowIssue(resolveValidationIssueCode(validationReport), "error", WorkflowLifecycle.STEP_VALIDATING,
160                 "Validation detected mismatches between the plan and the current state.", "Inspect mismatches and re-run the workflow after fixes.", true, Map.of()).toMap());
161     }
162     
163     private String resolveValidationStatus(final ValidationReport validationReport) {
164         return WorkflowLifecycle.STATUS_FAILED.equals(validationReport.getOverallStatus()) ? WorkflowLifecycle.STATUS_FAILED : WorkflowLifecycle.STATUS_VALIDATED;
165     }
166     
167     private String createValidationSummary(final WorkflowContextSnapshot snapshot, final ValidationReport validationReport) {
168         if (WorkflowLifecycle.STATUS_FAILED.equals(validationReport.getOverallStatus())) {
169             return String.format("Workflow validation failed for plan `%s` with %d mismatch(es).", snapshot.getPlanId(), validationReport.getMismatches().size());
170         }
171         return String.format("Workflow validation passed for plan `%s`.", snapshot.getPlanId());
172     }
173     
174     private boolean isValidatableStatus(final WorkflowContextSnapshot snapshot) {
175         String actualStatus = null == snapshot.getStatus() ? "" : snapshot.getStatus();
176         if (WorkflowLifecycle.STATUS_VALIDATED.equalsIgnoreCase(actualStatus)
177                 || WorkflowLifecycle.STATUS_EXECUTED.equalsIgnoreCase(actualStatus)
178                 || WorkflowLifecycle.STATUS_AWAITING_MANUAL_EXECUTION.equalsIgnoreCase(actualStatus)) {
179             return true;
180         }
181         if (!WorkflowLifecycle.STATUS_FAILED.equalsIgnoreCase(actualStatus)) {
182             return false;
183         }
184         String currentStep = WorkflowLifecycleUtils.resolveCurrentStep(snapshot);
185         return WorkflowLifecycle.STEP_VALIDATED.equalsIgnoreCase(currentStep)
186                 || WorkflowLifecycle.STEP_FAILED.equalsIgnoreCase(currentStep)
187                 || WorkflowLifecycle.STEP_EXECUTED.equalsIgnoreCase(currentStep);
188     }
189     
190     private Map<String, Object> createRejectedResponse(final WorkflowContextSnapshot snapshot, final String issueCode, final String message, final String userAction) {
191         Map<String, Object> result = new LinkedHashMap<>(10, 1F);
192         result.put("response_mode", "terminal");
193         result.put(MCPPayloadFieldNames.SUMMARY, String.format("Workflow validation cannot run for plan `%s`.", snapshot.getPlanId()));
194         result.put(WorkflowFieldNames.PLAN_ID, snapshot.getPlanId());
195         result.put("status", WorkflowLifecycle.STATUS_FAILED);
196         result.put("issues", List.of(new WorkflowIssue(issueCode, "error", WorkflowLifecycle.STEP_VALIDATING, message, userAction, false, Map.of()).toMap()));
197         result.put("overall_status", WorkflowLifecycle.STATUS_FAILED);
198         result.put("mismatches", List.of());
199         result.put("recovery_guidance", userAction);
200         result.put(MCPPayloadFieldNames.NEXT_ACTIONS, createRejectedNextActions(issueCode, userAction));
201         return result;
202     }
203     
204     private List<Map<String, Object>> createRejectedNextActions(final String issueCode, final String userAction) {
205         return MCPNextActionUtils.ordered(MCPNextActionUtils.askUser(userAction, List.of(resolveRejectedRequiredInput(issueCode))));
206     }
207     
208     private String resolveRejectedRequiredInput(final String issueCode) {
209         return WorkflowIssueCode.SESSION_OWNERSHIP_MISMATCH.equals(issueCode) ? "same_mcp_session" : "validatable_workflow_state";
210     }
211     
212     /**
213      * Resolve validation issue code from mismatches.
214      *
215      * @param validationReport validation report
216      * @return issue code
217      */
218     public String resolveValidationIssueCode(final ValidationReport validationReport) {
219         for (Map<String, Object> each : validationReport.getMismatches()) {
220             String actualCode = Objects.toString(each.get("code"), "").trim();
221             if (!actualCode.isEmpty()) {
222                 return actualCode;
223             }
224         }
225         return WorkflowIssueCode.RULE_STATE_MISMATCH;
226     }
