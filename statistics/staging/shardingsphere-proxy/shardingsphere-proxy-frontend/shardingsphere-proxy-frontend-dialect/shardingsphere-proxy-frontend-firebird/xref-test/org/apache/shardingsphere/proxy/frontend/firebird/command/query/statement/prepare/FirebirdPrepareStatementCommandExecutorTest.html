146         ShardingSphereTable table = new ShardingSphereTable("foo_tbl", Arrays.asList(column, blobColumn), Collections.emptyList(), Collections.emptyList());
147         ShardingSphereSchema schema = new ShardingSphereSchema("foo_db", databaseType, Collections.singleton(table), Collections.emptyList());

564     @ParameterizedTest(name = "{0}")
565     @MethodSource("describeWithSchemaReplacementArguments")
566     void assertDescribeWithSchemaReplacement(final String name, final String sql, final FirebirdSQLInfoPacketType describeType) throws Exception {
567         ShardingSphereDatabase database = ProxyContext.getInstance().getContextManager().getMetaDataContexts().getMetaData().getDatabase("foo_db");
568         ShardingSphereColumn idColumn = new ShardingSphereColumn("id", Types.INTEGER, false, false, true, true, false, true);
569         ShardingSphereColumn codeColumn = new ShardingSphereColumn("code", Types.BIGINT, false, false, true, true, false, true);
570         ShardingSphereTable originalTable = new ShardingSphereTable("foo_tbl", Arrays.asList(idColumn, codeColumn), Collections.emptyList(), Collections.emptyList());
571         database.addSchema(new ShardingSphereSchema("foo_db", databaseType, Collections.singleton(originalTable), Collections.emptyList()));
572         when(database.findDefaultSchema()).thenCallRealMethod();
573         ShardingSphereTable replacementTable = new ShardingSphereTable("foo_tbl", Collections.singleton(idColumn), Collections.emptyList(), Collections.emptyList());
574         ShardingSphereSchema replacementSchema = new ShardingSphereSchema("foo_db", databaseType, Collections.singleton(replacementTable), Collections.emptyList());
575         when(ProxyContext.getInstance().getContextManager().getDatabase("foo_db")).thenReturn(database).thenAnswer(invocation -> {
576             database.addSchema(replacementSchema);
577             return database;
578         }).thenReturn(database);
579         when(packet.getSQL()).thenReturn(sql);
580         when(packet.nextItem()).thenReturn(true, true, true, true, true, true, false);
581         when(packet.getCurrentItem()).thenReturn(FirebirdSQLInfoPacketType.STMT_TYPE, describeType, FirebirdSQLInfoPacketType.TYPE, FirebirdSQLInfoPacketType.TYPE,
582                 FirebirdSQLInfoPacketType.FIELD, FirebirdSQLInfoPacketType.FIELD, FirebirdSQLInfoPacketType.DESCRIBE_END, FirebirdSQLInfoPacketType.DESCRIBE_END);
583         FirebirdPrepareStatementReturnPacket actualReturnPacket = executeAndGetReturnPacket();
584         List<FirebirdReturnColumnPacket> actualColumns = FirebirdSQLInfoPacketType.BIND == describeType ? actualReturnPacket.getDescribeBind() : actualReturnPacket.getDescribeSelect();
585         assertThat(actualColumns.size(), is(2));
586         FirebirdPacketPayload idPayload = mock(FirebirdPacketPayload.class);
587         when(idPayload.getCharset()).thenReturn(StandardCharsets.UTF_8);
588         actualColumns.get(0).write(idPayload);
589         verify(idPayload).writeBytes("id".getBytes(StandardCharsets.UTF_8));
590         verify(idPayload).writeInt4LE(FirebirdBinaryColumnType.LONG.getValue() + 1);
591         FirebirdPacketPayload codePayload = mock(FirebirdPacketPayload.class);
592         when(codePayload.getCharset()).thenReturn(StandardCharsets.UTF_8);
593         actualColumns.get(1).write(codePayload);
594         verify(codePayload).writeBytes("code".getBytes(StandardCharsets.UTF_8));
595         verify(codePayload).writeInt4LE(FirebirdBinaryColumnType.INT64.getValue() + 1);
596     }
