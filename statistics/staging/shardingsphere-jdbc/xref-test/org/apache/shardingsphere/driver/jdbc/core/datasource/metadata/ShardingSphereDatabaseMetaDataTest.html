100         when(shardingSphereConnection.getDatabaseConnectionManager().getRandomPhysicalDataSourceName()).thenReturn(DATA_SOURCE_NAME);
101         when(shardingSphereConnection.getDatabaseConnectionManager().getRandomConnection()).thenReturn(connection);
102         when(shardingSphereConnection.getContextManager().getMetaDataContexts()).thenReturn(metaDataContexts);
103         when(shardingSphereConnection.getCurrentDatabaseName()).thenReturn("foo_db");
104         ShardingSphereDatabase database = mock(ShardingSphereDatabase.class, RETURNS_DEEP_STUBS);
105         when(metaDataContexts.getMetaData().getDatabase(shardingSphereConnection.getCurrentDatabaseName())).thenReturn(database);
106         when(shardingSphereConnection.getContextManager().getDatabase("foo_db")).thenReturn(database);
107         Map<String, StorageUnit> storageUnits = database.getResourceMetaData().getStorageUnits();
108         when(shardingSphereConnection.getContextManager().getStorageUnits("foo_db")).thenReturn(storageUnits);
109         ShardingRule shardingRule = mockShardingRule();
110         when(database.getRuleMetaData().getRules()).thenReturn(Collections.singleton(shardingRule));
111         when(connectionProps.getCatalog()).thenReturn("test");
112         when(connectionProps.getSchema()).thenReturn("test");
113         when(database.getResourceMetaData().getStorageUnits().get(DATA_SOURCE_NAME).getConnectionProperties()).thenReturn(connectionProps);
114         shardingSphereDatabaseMetaData = new ShardingSphereDatabaseMetaData(shardingSphereConnection);
115     }

117     private ShardingRule mockShardingRule() {
118         ShardingRuleConfiguration ruleConfig = new ShardingRuleConfiguration();
119         ShardingTableRuleConfiguration shardingTableRuleConfig = new ShardingTableRuleConfiguration(TABLE_NAME, DATA_SOURCE_NAME + "." + TABLE_NAME);
120         ruleConfig.setTables(Collections.singleton(shardingTableRuleConfig));
121         return new ShardingRule(ruleConfig, Collections.singletonMap(DATA_SOURCE_NAME, new MockedDataSource()), mock(ComputeNodeInstanceContext.class), Collections.emptyList());
122     }
