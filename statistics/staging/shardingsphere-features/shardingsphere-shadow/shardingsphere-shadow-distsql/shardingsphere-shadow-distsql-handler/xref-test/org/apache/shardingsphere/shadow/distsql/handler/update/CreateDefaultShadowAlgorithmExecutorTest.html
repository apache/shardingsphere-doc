54  @ExtendWith(MockitoExtension.class)
55  @MockitoSettings(strictness = Strictness.LENIENT)
56  class CreateDefaultShadowAlgorithmExecutorTest {
57      
58      private final CreateDefaultShadowAlgorithmExecutor executor = (CreateDefaultShadowAlgorithmExecutor) TypedSPILoader.getService(
59              DatabaseRuleDefinitionExecutor.class, CreateDefaultShadowAlgorithmStatement.class);
60      
61      @Mock(answer = Answers.RETURNS_DEEP_STUBS)
62      private ShardingSphereDatabase database;
63      
64      @BeforeEach
65      void setUp() {
66          when(database.getName()).thenReturn("shadow_db");
67          executor.setDatabase(database);
68      }
69      
70      @ParameterizedTest(name = "{0}")
71      @MethodSource("checkBeforeUpdateFailureArguments")
72      void assertCheckBeforeUpdateWithInvalidInput(final String name, final CreateDefaultShadowAlgorithmStatement sqlStatement,
73                                                   final ShadowRule rule, final Class<? extends Exception> expectedException) {
74          executor.setRule(rule);
75          assertThrows(expectedException, () -> executor.checkBeforeUpdate(sqlStatement));
76      }
77      
78      @Test
79      void assertCheckBeforeUpdate() {
80          executor.setRule(null);
81          assertDoesNotThrow(() -> executor.checkBeforeUpdate(createStatement(false, "SQL_HINT")));
82      }
83      
84      @Test
85      void assertCheckBeforeUpdateWithIfNotExists() {
86          ShadowRule rule = mock(ShadowRule.class);
87          executor.setRule(rule);
88          assertDoesNotThrow(() -> executor.checkBeforeUpdate(createStatement(true, "SQL_HINT")));
89      }
90      
91      @ParameterizedTest(name = "{0}")
92      @MethodSource("buildToBeCreatedRuleConfigurationArguments")
93      void assertBuildToBeCreatedRuleConfiguration(final String name, final ShadowRule rule, final int expectedAlgorithmSize, final String expectedDefaultAlgorithmName) {
94          executor.setRule(rule);
95          ShadowRuleConfiguration actual = executor.buildToBeCreatedRuleConfiguration(createStatement(false, "SQL_HINT"));
96          assertThat(actual.getShadowAlgorithms().size(), is(expectedAlgorithmSize));
97          assertThat(actual.getShadowAlgorithms().containsKey("default_shadow_algorithm"), is(expectedAlgorithmSize > 0));
98          assertThat(actual.getDefaultShadowAlgorithmName(), is(expectedDefaultAlgorithmName));
99      }
100     
101     @Test
102     void assertGetRuleClass() {
103         assertThat(executor.getRuleClass(), is(ShadowRule.class));
104     }
105     
106     private static Stream<Arguments> checkBeforeUpdateFailureArguments() {
107         ShadowRule duplicatedRule = mock(ShadowRule.class);
108         when(duplicatedRule.containsShadowAlgorithm("default_shadow_algorithm")).thenReturn(true);
109         return Stream.of(
110                 Arguments.of("missing algorithm type", createStatement(false, ""), null, MissingRequiredAlgorithmException.class),
111                 Arguments.of("invalid algorithm type", createStatement(false, "INVALID_TYPE"), null, ServiceProviderNotFoundException.class),
112                 Arguments.of("duplicate default algorithm", createStatement(false, "SQL_HINT"), duplicatedRule, InUsedAlgorithmException.class));
113     }
114     
115     private static CreateDefaultShadowAlgorithmStatement createStatement(final boolean ifNotExists, final String algorithmType) {
116         ShadowAlgorithmSegment shadowAlgorithmSegment = new ShadowAlgorithmSegment("algorithm_name", new AlgorithmSegment(algorithmType, PropertiesBuilder.build(new Property("type", "value"))));
117         return new CreateDefaultShadowAlgorithmStatement(ifNotExists, shadowAlgorithmSegment);
118     }
119     
120     private static Stream<Arguments> buildToBeCreatedRuleConfigurationArguments() {
121         ShadowRule duplicatedRule = mock(ShadowRule.class);
122         when(duplicatedRule.containsShadowAlgorithm("default_shadow_algorithm")).thenReturn(true);
123         return Stream.of(
124                 Arguments.of("rule is null", null, 1, "default_shadow_algorithm"),
125                 Arguments.of("rule without default algorithm", mock(ShadowRule.class), 1, "default_shadow_algorithm"),
126                 Arguments.of("rule already has default algorithm", duplicatedRule, 0, null));
127     }
