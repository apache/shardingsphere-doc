52  @ExtendWith(MockitoExtension.class)
53  class MySQLNegotiatePackageDecoderTest {
54      
55      @Mock(answer = Answers.RETURNS_DEEP_STUBS)
56      private ChannelHandlerContext channelHandlerContext;
57      
58      @BeforeEach
59      void setup() {
60          when(channelHandlerContext.channel().attr(CommonConstants.CHARSET_ATTRIBUTE_KEY).get()).thenReturn(StandardCharsets.UTF_8);
61      }
62      
63      @Test
64      void assertDecodeUnsupportedResponseHeader() throws ReflectiveOperationException {
65          MySQLNegotiatePackageDecoder negotiatePackageDecoder = new MySQLNegotiatePackageDecoder();
66          Plugins.getMemberAccessor().set(MySQLNegotiatePackageDecoder.class.getDeclaredField("handshakeReceived"), negotiatePackageDecoder, true);
67          List<Object> actual = new LinkedList<>();
68          assertThrows(UnsupportedSQLOperationException.class, () -> negotiatePackageDecoder.decode(channelHandlerContext, unsupportedHeaderPacket(), actual));
69          assertTrue(actual.isEmpty());
70      }
71      
72      private ByteBuf unsupportedHeaderPacket() {
73          ByteBuf result = Unpooled.buffer();
74          result.writeByte(0x10);
75          return result;
76      }
77      
78      @Test
79      void assertDecodeHandshakePacket() {
80          MySQLNegotiatePackageDecoder commandPacketDecoder = new MySQLNegotiatePackageDecoder();
81          List<Object> actual = new LinkedList<>();
82          commandPacketDecoder.decode(channelHandlerContext, handshakePacket(), actual);
83          assertHandshakePacket(actual);
84      }
85      
86      private ByteBuf handshakePacket() {
87          String handshakePacket = "0a352e372e32312d6c6f6700090000004a592a1f725a0d0900fff7210200ff8115000000000000000000001a437b30323a4d2b514b5870006d"
88                  + "7973716c5f6e61746976655f70617373776f72640000000002000000";
89          byte[] handshakePacketBytes = ByteBufUtil.decodeHexDump(handshakePacket);
90          ByteBuf result = Unpooled.buffer(handshakePacketBytes.length);
91          result.writeBytes(handshakePacketBytes);
92          return result;
93      }
94      
95      private void assertHandshakePacket(final List<Object> actual) {
96          assertThat(actual.size(), is(1));
97          assertThat(actual.get(0), isA(MySQLHandshakePacket.class));
98          MySQLHandshakePacket actualPacket = (MySQLHandshakePacket) actual.get(0);
99          assertThat(actualPacket.getProtocolVersion(), is(0x0a));
100         assertThat(actualPacket.getServerVersion(), is("5.7.21-log"));
101         assertThat(actualPacket.getConnectionId(), is(9));
102         assertThat(actualPacket.getCharacterSet(), is(33));
103         assertThat(actualPacket.getStatusFlag().getValue(), is(2));
104         assertThat(actualPacket.getCapabilityFlagsLower(), is(63487));
105         assertThat(actualPacket.getCapabilityFlagsUpper(), is(33279));
106         assertThat(actualPacket.getAuthPluginName(), is("mysql_native_password"));
107     }
108     
109     @Test
110     void assertDecodeAuthSwitchRequestPacket() throws ReflectiveOperationException {
111         MySQLNegotiatePackageDecoder negotiatePackageDecoder = new MySQLNegotiatePackageDecoder();
112         Plugins.getMemberAccessor().set(MySQLNegotiatePackageDecoder.class.getDeclaredField("handshakeReceived"), negotiatePackageDecoder, true);
113         List<Object> actual = new LinkedList<>();
114         negotiatePackageDecoder.decode(channelHandlerContext, authSwitchRequestPacket(), actual);
115         assertPacketByType(actual, MySQLAuthSwitchRequestPacket.class);
116     }
117     
118     private ByteBuf authSwitchRequestPacket() {
119         ByteBuf result = Unpooled.buffer();
120         result.writeByte(MySQLAuthSwitchRequestPacket.HEADER);
121         result.writeBytes("mysql_native_password".getBytes(StandardCharsets.UTF_8));
122         result.writeByte(0);
123         result.writeBytes("foo_auth_plugin_data_1234".getBytes(StandardCharsets.UTF_8));
124         result.writeByte(0);
125         return result;
126     }
127     
128     @Test
129     void assertDecodeAuthMoreDataPacket() throws ReflectiveOperationException {
130         MySQLNegotiatePackageDecoder negotiatePackageDecoder = new MySQLNegotiatePackageDecoder();
131         Plugins.getMemberAccessor().set(MySQLNegotiatePackageDecoder.class.getDeclaredField("handshakeReceived"), negotiatePackageDecoder, true);
132         List<Object> actual = new LinkedList<>();
133         negotiatePackageDecoder.decode(channelHandlerContext, authMoreDataPacket(), actual);
134         assertPacketByType(actual, MySQLAuthMoreDataPacket.class);
135     }
136     
137     private ByteBuf authMoreDataPacket() {
138         ByteBuf result = Unpooled.buffer();
139         result.writeByte(MySQLAuthMoreDataPacket.HEADER);
140         result.writeBytes(new byte[]{1, 2, 3});
141         return result;
142     }
143     
144     @Test
145     void assertDecodeErrPacket() throws ReflectiveOperationException {
146         MySQLNegotiatePackageDecoder negotiatePackageDecoder = new MySQLNegotiatePackageDecoder();
147         Plugins.getMemberAccessor().set(MySQLNegotiatePackageDecoder.class.getDeclaredField("handshakeReceived"), negotiatePackageDecoder, true);
148         List<Object> actual = new LinkedList<>();
149         negotiatePackageDecoder.decode(channelHandlerContext, errPacket(), actual);
150         assertPacketByType(actual, MySQLErrPacket.class);
151         MySQLErrPacket actualPacket = (MySQLErrPacket) actual.get(0);
152         assertThat(actualPacket.getErrorCode(), is(1234));
153         assertThat(actualPacket.getSqlState(), is("HY000"));
154         assertThat(actualPacket.getErrorMessage(), is("foo_error"));
155     }
156     
157     private ByteBuf errPacket() {
158         ByteBuf result = Unpooled.buffer();
159         result.writeByte(MySQLErrPacket.HEADER);
160         result.writeShortLE(1234);
161         result.writeByte('#');
162         result.writeBytes("HY000".getBytes(StandardCharsets.UTF_8));
163         result.writeBytes("foo_error".getBytes(StandardCharsets.UTF_8));
164         return result;
165     }
166     
167     @Test
168     void assertDecodeOkPacketAndRemoveDecoder() throws ReflectiveOperationException {
169         MySQLNegotiatePackageDecoder negotiatePackageDecoder = new MySQLNegotiatePackageDecoder();
170         Plugins.getMemberAccessor().set(MySQLNegotiatePackageDecoder.class.getDeclaredField("handshakeReceived"), negotiatePackageDecoder, true);
171         List<Object> actual = new LinkedList<>();
172         negotiatePackageDecoder.decode(channelHandlerContext, okPacket(), actual);
173         assertPacketByType(actual, MySQLOKPacket.class);
174         ChannelPipeline actualPipeline = channelHandlerContext.channel().pipeline();
175         verify(actualPipeline).remove(negotiatePackageDecoder);
176     }
177     
178     private ByteBuf okPacket() {
179         ByteBuf result = Unpooled.buffer();
180         result.writeByte(MySQLOKPacket.HEADER);
181         result.writeByte(0);
182         result.writeByte(0);
183         result.writeShortLE(2);
184         result.writeShortLE(0);
185         return result;
186     }
187     
188     private void assertPacketByType(final List<Object> actual, final Class<?> clazz) {
189         assertThat(actual.size(), is(1));
190         assertThat(actual.get(0), isA(clazz));
191     }
