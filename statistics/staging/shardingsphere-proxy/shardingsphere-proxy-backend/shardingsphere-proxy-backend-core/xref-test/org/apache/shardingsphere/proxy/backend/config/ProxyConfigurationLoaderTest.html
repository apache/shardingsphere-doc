98                  + "  - !GLOBAL_CLOCK\n"
99                  + "    type: REMOVED\n"
100                 + "    provider: REMOVED\n"
101                 + "authority:\n"
102                 + "  users:\n"
103                 + "    - user: root\n"
104                 + "      password: root\n"
105                 + "transaction:\n"
106                 + "  providerType: Atomikos\n"
107                 + "globalClock:\n"
108                 + "  type: FIXTURE\n"
109                 + "  provider: LOCAL\n"
110                 + "  enabled: true\n"
111                 + "sqlParser: {}\n"
112                 + "sqlTranslator:\n"
113                 + "  type: TRANSLATOR\n"
114                 + "sqlFederation:\n"
115                 + "  providerType: CALCITE\n"
116                 + "  allQueryUseSQLFederation: true\n");
117         writeConfigurationFile(tempDir, "config-compatible.yaml", "databaseName: compatible_db\n"
118                 + "dataSources:\n"
119                 + "  ds_0:\n"
120                 + "    url: jdbc:mock://127.0.0.1/compatible\n");
121         ProxyConfigurationLoadResult actual = ProxyConfigurationLoader.load(tempDir.toString());
122         assertThat(actual.getRules().size(), is(6));
123         Iterator<RuleConfiguration> rules = actual.getRules().iterator();
124         assertThat(rules.next(), isA(AuthorityRuleConfiguration.class));
125         assertThat(rules.next(), isA(TransactionRuleConfiguration.class));
126         assertThat(rules.next(), isA(SQLParserRuleConfiguration.class));
127         assertThat(rules.next(), isA(SQLTranslatorRuleConfiguration.class));
128         assertThat(rules.next(), isA(SQLFederationRuleConfiguration.class));
129         RuleConfiguration actualGlobalClockRule = rules.next();
130         assertThat(actualGlobalClockRule, isA(GlobalClockRuleConfiguration.class));
131         assertThat(((GlobalClockRuleConfiguration) actualGlobalClockRule).getProvider(), is("LOCAL"));
132         assertNotNull(actual.getDatabaseConfigurations().get("compatible_db").getDataSources());
133     }

136     @Test
137     void assertLoadWithUnknownRuleTagName(@TempDir final Path tempDir) throws IOException {
138         new YamlRuleConfigurationSwapperEngine().swapToRuleConfigurations(Collections.emptyList());
139         Collection<YamlRuleConfigurationSwapper> originalSwappers = ShardingSphereServiceLoader.getServiceInstances(YamlRuleConfigurationSwapper.class);
140         AtomicInteger yamlRuleConfigurationSwapperCallTimes = new AtomicInteger();
141         try (MockedStatic<ShardingSphereServiceLoader> mockedStatic = mockStatic(ShardingSphereServiceLoader.class, CALLS_REAL_METHODS)) {
142             mockedStatic.when(() -> ShardingSphereServiceLoader.getServiceInstances(YamlRuleConfigurationSwapper.class))
143                     .thenAnswer(invocation -> yamlRuleConfigurationSwapperCallTimes.getAndIncrement() < 2 ? originalSwappers : Collections.emptyList());
144             writeConfigurationFile(tempDir, "global.yaml", "");
145             writeConfigurationFile(tempDir, "database-unknown-rule-tag.yaml", "databaseName: unknown_rule_tag_db\n"
146                     + "dataSources:\n"
147                     + "  ds_0:\n"
148                     + "    url: jdbc:mock://127.0.0.1/unknown\n"
149                     + "rules:\n"
150                     + "  - !FIXTURE\n"
151                     + "    unique: duplicated\n"
152                     + "  - !FIXTURE\n"
153                     + "    unique: duplicated\n");
154             IllegalStateException actual = assertThrows(IllegalStateException.class, () -> ProxyConfigurationLoader.load(tempDir.toString()));
155             assertThat(actual.getMessage(), is("Not find rule tag name of class class " + MockedRuleConfiguration.class.getName()));
156         }
157     }

214         writeConfigurationFile(tempDir, "global.yaml", "dataSources:\n  global_ds:\n    url: jdbc:mock://127.0.0.1/global\n");

216                 + "rules:\n  - !SHARDING\n    defaultDatabaseStrategy:\n      standard:\n        shardingColumn: id\n        shardingAlgorithmName: fixture\n      none: {}\n");
217         try (
218                 MockedStatic<DataSourcePoolCreator> mockedPoolCreator = mockStatic(DataSourcePoolCreator.class);
219                 MockedConstruction<DataSourceGeneratedDatabaseConfiguration> mockedDatabaseConfigs = mockConstruction(DataSourceGeneratedDatabaseConfiguration.class)) {
220             assertThrows(InvalidShardingStrategyConfigCountException.class, () -> ProxyConfigurationLoader.load(tempDir.toString()));
221             mockedPoolCreator.verifyNoInteractions();
222             assertTrue(mockedDatabaseConfigs.constructed().isEmpty());
223         }

226     private void assertShardingRuleConfiguration(final ProxyDatabaseConfiguration actual) {
227         assertThat(actual.getDatabaseName(), is("sharding_db"));
228         assertThat(actual.getDataSources().size(), is(2));
229         assertDataSourceConfiguration(actual.getDataSources().get("ds_0"), "jdbc:mysql://127.0.0.1:3306/ds_0");
230         assertDataSourceConfiguration(actual.getDataSources().get("ds_1"), "jdbc:mysql://127.0.0.1:3306/ds_1");
231         Optional<ShardingRuleConfiguration> shardingRuleConfig = actual.getRules().stream().filter(ShardingRuleConfiguration.class::isInstance).findFirst().map(ShardingRuleConfiguration.class::cast);
232         assertTrue(shardingRuleConfig.isPresent());
233         assertShardingRuleConfiguration(shardingRuleConfig.get());
234         assertFalse(actual.getRules().stream().filter(EncryptRuleConfiguration.class::isInstance).findFirst().map(EncryptRuleConfiguration.class::cast).isPresent());
235     }
236     
237     private void assertShardingRuleConfiguration(final ShardingRuleConfiguration actual) {
238         assertThat(actual.getTables().size(), is(1));
239         assertThat(actual.getTables().iterator().next().getActualDataNodes(), is("ds_${0..1}.t_order_${0..1}"));
240         assertThat(((StandardShardingStrategyConfiguration) actual.getTables().iterator().next().getDatabaseShardingStrategy()).getShardingColumn(), is("user_id"));
241         assertThat(((StandardShardingStrategyConfiguration) actual.getTables().iterator().next().getDatabaseShardingStrategy()).getShardingAlgorithmName(), is("database_inline"));
242         assertThat(((StandardShardingStrategyConfiguration) actual.getTables().iterator().next().getTableShardingStrategy()).getShardingColumn(), is("order_id"));
243         assertThat(((StandardShardingStrategyConfiguration) actual.getTables().iterator().next().getTableShardingStrategy()).getShardingAlgorithmName(), is("table_inline"));
244         assertThat(actual.getDefaultDatabaseShardingStrategy(), isA(NoneShardingStrategyConfiguration.class));
245     }

247     private void assertReadwriteSplittingRuleConfiguration(final ProxyDatabaseConfiguration actual) {
248         assertThat(actual.getDatabaseName(), is("readwrite_splitting_db"));
249         assertThat(actual.getDataSources().size(), is(3));
250         assertDataSourceConfiguration(actual.getDataSources().get("write_ds"), "jdbc:mysql://127.0.0.1:3306/write_ds");
251         assertDataSourceConfiguration(actual.getDataSources().get("read_ds_0"), "jdbc:mysql://127.0.0.1:3306/read_ds_0");
252         assertDataSourceConfiguration(actual.getDataSources().get("read_ds_1"), "jdbc:mysql://127.0.0.1:3306/read_ds_1");
253         assertFalse(actual.getRules().stream().filter(ShardingRuleConfiguration.class::isInstance).findFirst().map(ShardingRuleConfiguration.class::cast).isPresent());
254         assertFalse(actual.getRules().stream().filter(EncryptRuleConfiguration.class::isInstance).findFirst().map(EncryptRuleConfiguration.class::cast).isPresent());
255         Optional<ReadwriteSplittingRuleConfiguration> ruleConfig = actual.getRules().stream()
256                 .filter(ReadwriteSplittingRuleConfiguration.class::isInstance).findFirst().map(ReadwriteSplittingRuleConfiguration.class::cast);
257         assertTrue(ruleConfig.isPresent());
258         for (ReadwriteSplittingDataSourceGroupRuleConfiguration each : ruleConfig.get().getDataSourceGroups()) {
259             assertReadwriteSplittingRuleConfiguration(each);
260         }
261     }

265         assertThat(actual.getReadDataSourceNames(), is(Arrays.asList("read_ds_0", "read_ds_1")));
266     }

268     private void assertEncryptRuleConfiguration(final ProxyDatabaseConfiguration actual) {
269         assertThat(actual.getDatabaseName(), is("encrypt_db"));
270         assertThat(actual.getDataSources().size(), is(1));
271         assertDataSourceConfiguration(actual.getDataSources().get("ds_0"), "jdbc:mysql://127.0.0.1:3306/encrypt_ds");
272         assertFalse(actual.getRules().stream().filter(ShardingRuleConfiguration.class::isInstance).findFirst().map(ShardingRuleConfiguration.class::cast).isPresent());
273         Optional<EncryptRuleConfiguration> encryptRuleConfig = actual.getRules().stream().filter(EncryptRuleConfiguration.class::isInstance).findFirst().map(EncryptRuleConfiguration.class::cast);
274         assertTrue(encryptRuleConfig.isPresent());
275         assertEncryptRuleConfiguration(encryptRuleConfig.get());
276     }

278     private void assertEncryptRuleConfiguration(final EncryptRuleConfiguration actual) {
279         assertThat(actual.getEncryptors().size(), is(1));
280         assertTrue(actual.getEncryptors().containsKey("aes_encryptor"));
281         AlgorithmConfiguration aesEncryptAlgorithmConfig = actual.getEncryptors().get("aes_encryptor");
282         assertThat(aesEncryptAlgorithmConfig.getType(), is("AES"));
283         assertThat(aesEncryptAlgorithmConfig.getProps().getProperty("aes-key-value"), is("123456abc"));
284         assertThat(aesEncryptAlgorithmConfig.getProps().getProperty("digest-algorithm-name"), is("SHA-1"));
285     }

287     private void assertDataSourceConfiguration(final DataSourceConfiguration actual, final String expectedURL) {
288         assertThat(actual.getConnection().getDriverClassName(), is("com.mysql.cj.jdbc.Driver"));
289         assertThat(actual.getConnection().getUrl(), is(expectedURL));
290         assertThat(actual.getConnection().getUsername(), is("root"));
291         assertNull(actual.getConnection().getPassword());
292         assertThat(actual.getPool().getConnectionTimeoutMilliseconds(), is(30000L));
293         assertThat(actual.getPool().getIdleTimeoutMilliseconds(), is(60000L));
294         assertThat(actual.getPool().getMaxLifetimeMilliseconds(), is(1800000L));
295         assertThat(actual.getPool().getMaxPoolSize(), is(50));
296     }
