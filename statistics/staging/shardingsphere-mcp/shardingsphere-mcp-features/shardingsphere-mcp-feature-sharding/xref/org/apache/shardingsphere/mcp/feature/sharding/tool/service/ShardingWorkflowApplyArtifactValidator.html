18  package org.apache.shardingsphere.mcp.feature.sharding.tool.service;
19  
20  import org.apache.shardingsphere.infra.algorithm.keygen.spi.KeyGenerateAlgorithm;
21  import org.apache.shardingsphere.mcp.feature.sharding.ShardingFeatureDefinition;
22  import org.apache.shardingsphere.mcp.feature.sharding.tool.model.ShardingWorkflowRequest;
23  import org.apache.shardingsphere.mcp.support.workflow.model.WorkflowContextSnapshot;
24  import org.apache.shardingsphere.mcp.support.workflow.service.WorkflowAlgorithmUtils;
25  import org.apache.shardingsphere.mcp.support.workflow.service.WorkflowArtifactBundle.ExecutableWorkflowArtifact;
26  import org.apache.shardingsphere.mcp.support.workflow.service.WorkflowValidationSupport;
27  import org.apache.shardingsphere.mcp.support.workflow.spi.MCPWorkflowApplyArtifactValidator;
28  import org.apache.shardingsphere.sharding.spi.ShardingAlgorithm;
29  import org.apache.shardingsphere.sharding.spi.ShardingAuditAlgorithm;
30  
31  import java.util.Collection;
32  import java.util.LinkedList;
33  import java.util.List;
34  import java.util.Locale;
35  import java.util.Map;
36  
37  /**
38   * Sharding workflow apply artifact validator.
39   */
40  public final class ShardingWorkflowApplyArtifactValidator implements MCPWorkflowApplyArtifactValidator {
41      
42      private final WorkflowValidationSupport validationSupport = new WorkflowValidationSupport();
43      
44      @Override
45      public List<Map<String, Object>> validate(final WorkflowContextSnapshot snapshot, final Collection<ExecutableWorkflowArtifact> artifacts) {
46          List<Map<String, Object>> result = new LinkedList<>();
47          for (ExecutableWorkflowArtifact each : artifacts) {
48              addRuleDistSQLIssues(result, snapshot, each.sql(), each.displaySql());
49          }
50          return result;
51      }
52      
53      private void addRuleDistSQLIssues(final List<Map<String, Object>> issues, final WorkflowContextSnapshot snapshot, final String sql, final String displaySql) {
54          if (!isRuleDefinitionDistSQL(sql) || !(snapshot.getRequest() instanceof ShardingWorkflowRequest)) {
55              return;
56          }
57          ShardingWorkflowRequest request = (ShardingWorkflowRequest) snapshot.getRequest();
58          if (ShardingFeatureDefinition.TABLE_RULE_WORKFLOW_KIND.equals(snapshot.getWorkflowKind())) {
59              addTableRuleAlgorithmIssues(issues, request, displaySql);
60              return;
61          }
62          if (ShardingFeatureDefinition.DEFAULT_STRATEGY_WORKFLOW_KIND.equals(snapshot.getWorkflowKind())) {
63              addShardingAlgorithmIssue(issues, request.getAlgorithmType(), request.getPrimaryAlgorithmProperties(), displaySql);
64              return;
65          }
66          if (ShardingFeatureDefinition.KEY_GENERATOR_WORKFLOW_KIND.equals(snapshot.getWorkflowKind())) {
67              addKeyGeneratorTypeIssue(issues, request, displaySql);
68              return;
69          }
70          if (ShardingFeatureDefinition.KEY_GENERATE_STRATEGY_WORKFLOW_KIND.equals(snapshot.getWorkflowKind())) {
71              addKeyGeneratorIssue(issues, request, displaySql);
72          }
73      }
74      
75      private boolean isRuleDefinitionDistSQL(final String sql) {
76          String actualSQL = sql.trim().toUpperCase(Locale.ENGLISH);
77          return actualSQL.startsWith("CREATE SHARDING ") || actualSQL.startsWith("CREATE DEFAULT SHARDING ");
78      }
79      
80      private void addTableRuleAlgorithmIssues(final List<Map<String, Object>> issues, final ShardingWorkflowRequest request, final String displaySql) {
81          if (!request.getStorageUnits().isEmpty() || !"none".equals(normalizeStrategyType(request))) {
82              addShardingAlgorithmIssue(issues, request.getAlgorithmType(), request.getPrimaryAlgorithmProperties(), displaySql);
83          }
84          addKeyGeneratorIssue(issues, request, displaySql);
85          for (String each : request.getAuditorNames()) {
86              addAuditorIssue(issues, each, displaySql);
87          }
88      }
89      
90      private void addShardingAlgorithmIssue(final List<Map<String, Object>> issues, final String algorithmType, final Map<String, String> properties, final String displaySql) {
91          if (algorithmType.isEmpty()) {
92              return;
93          }
94          if (!WorkflowAlgorithmUtils.isAlgorithmServiceAvailable(ShardingAlgorithm.class, algorithmType, properties)) {
95              issues.add(validationSupport.createSQLExecutabilityIssue(String.format("Generated sharding DistSQL references sharding algorithm `%s`, "
96                      + "but it cannot be loaded or initialized by ShardingAlgorithm SPI.", algorithmType), displaySql));
97          }
98      }
99      
100     private void addKeyGeneratorIssue(final List<Map<String, Object>> issues, final ShardingWorkflowRequest request, final String displaySql) {
101         if (!request.getKeyGeneratorName().isEmpty() || request.getKeyGeneratorType().isEmpty()) {
102             return;
103         }
104         addKeyGeneratorTypeIssue(issues, request, displaySql);
105     }
106     
107     private void addKeyGeneratorTypeIssue(final List<Map<String, Object>> issues, final ShardingWorkflowRequest request, final String displaySql) {
108         if (request.getKeyGeneratorType().isEmpty()) {
109             return;
110         }
111         if (!WorkflowAlgorithmUtils.isAlgorithmServiceAvailable(KeyGenerateAlgorithm.class, request.getKeyGeneratorType(), request.getKeyGeneratorProperties())) {
112             issues.add(validationSupport.createSQLExecutabilityIssue(String.format("Generated sharding DistSQL references key generator algorithm `%s`, "
113                     + "but it cannot be loaded or initialized by KeyGenerateAlgorithm SPI.", request.getKeyGeneratorType()), displaySql));
114         }
115     }
116     
117     private void addAuditorIssue(final List<Map<String, Object>> issues, final String algorithmType, final String displaySql) {
118         if (algorithmType.isEmpty()) {
119             return;
120         }
121         if (!WorkflowAlgorithmUtils.isAlgorithmServiceAvailable(ShardingAuditAlgorithm.class, algorithmType, Map.of())) {
122             issues.add(validationSupport.createSQLExecutabilityIssue(String.format("Generated sharding DistSQL references auditor algorithm `%s`, "
123                     + "but it cannot be loaded or initialized by ShardingAuditAlgorithm SPI.", algorithmType), displaySql));
124         }
125     }
126     
127     private String normalizeStrategyType(final ShardingWorkflowRequest request) {
128         return request.getStrategyType().isEmpty() ? "standard" : request.getStrategyType().toLowerCase(Locale.ENGLISH);
129     }
130     
131 }
