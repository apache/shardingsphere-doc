123     @Override
124     public Collection<DatabasePacket> execute() throws SQLException {
125         MetaDataContexts metaDataContexts = ProxyContext.getInstance().getContextManager().getMetaDataContexts();
126         SQLParserRule sqlParserRule = metaDataContexts.getMetaData().getGlobalRuleMetaData().getSingleRule(SQLParserRule.class);
127         DatabaseType databaseType = TypedSPILoader.getService(DatabaseType.class, "Firebird");
128         SQLStatement sqlStatement = resolveCurrentDatabase(sqlParserRule.getSQLParserEngine(databaseType).parse(packet.getSQL(), true));
129         SQLStatementContext sqlStatementContext = new SQLBindEngine(metaDataContexts.getMetaData(), connectionSession.getCurrentDatabaseName(), packet.getHintValueContext()).bind(sqlStatement);
130         int statementId = getStatementId();
131         if (packet.isValidStatementHandle()) {
132             FirebirdStatementResourceCleaner.clean(connectionSession, statementId, true);
133         }
134         FirebirdServerPreparedStatement serverPreparedStatement = new FirebirdServerPreparedStatement(packet.getSQL(), sqlStatementContext, packet.getHintValueContext());
135         connectionSession.getServerPreparedStatementRegistry().addPreparedStatement(statementId, serverPreparedStatement);
136         return createResponse(sqlStatementContext);
137     }

139     private SQLStatement resolveCurrentDatabase(final SQLStatement sqlStatement) {
140         if (!(sqlStatement instanceof DropDatabaseStatement) || null != ((DropDatabaseStatement) sqlStatement).getDatabaseName()) {
141             return sqlStatement;
142         }
143         String currentDatabaseName = connectionSession.getUsedDatabaseName();
144         ShardingSpherePreconditions.checkNotNull(currentDatabaseName, NoDatabaseSelectedException::new);
145         return new DropDatabaseStatement(sqlStatement.getDatabaseType(), currentDatabaseName, ((DropDatabaseStatement) sqlStatement).isIfExists());
146     }
147     
148     private int getStatementId() {
149         return packet.isValidStatementHandle() ? packet.getStatementId() : FirebirdStatementIdGenerator.getInstance().getStatementId(connectionSession.getConnectionId());
150     }

152     private Collection<DatabasePacket> createResponse(final SQLStatementContext sqlStatementContext) {
153         FirebirdSQLInfoReturnValue statementType = getFirebirdStatementType(sqlStatementContext.getSqlStatement());
154         FirebirdPrepareStatementReturnPacket returnPacket = new FirebirdPrepareStatementReturnPacket();
155         while (packet.nextItem()) {
156             switch (packet.getCurrentItem()) {
157                 case STMT_TYPE:
158                     returnPacket.setType(statementType);
159                     break;
160                 case SELECT:
161                     if (statementType.isSelectDescribable()) {
162                         processDescribe(sqlStatementContext, returnPacket.getDescribeSelect(), true);
163                     } else {
164                         skipDescribe();
165                     }
166                     break;
167                 case BIND:
168                     if (statementType.isBindDescribable()) {
169                         processDescribe(sqlStatementContext, returnPacket.getDescribeBind(), false);
170                     } else {
171                         skipDescribe();
172                     }
173                     break;
174                 default:
175                     throw new DatabaseProtocolException("Unknown statement info request type %d", packet.getCurrentItem());
176             }
177         }
178         return Collections.singleton(new FirebirdGenericResponsePacket().setData(returnPacket));
179     }
180     
181     private FirebirdSQLInfoReturnValue getFirebirdStatementType(final SQLStatement statement) {
182         if (statement instanceof SelectStatement) {
183             return FirebirdSQLInfoReturnValue.SELECT;
184         }
185         if (statement instanceof InsertStatement) {
186             returningSegment = ((InsertStatement) statement).getReturning().orElse(null);
187             return null == returningSegment ? FirebirdSQLInfoReturnValue.INSERT : FirebirdSQLInfoReturnValue.EXEC_PROCEDURE;
188         }
189         if (statement instanceof UpdateStatement) {
190             returningSegment = ((UpdateStatement) statement).getReturning().orElse(null);
191             return null == returningSegment ? FirebirdSQLInfoReturnValue.UPDATE : FirebirdSQLInfoReturnValue.EXEC_PROCEDURE;
192         }
193         if (statement instanceof DeleteStatement) {
194             returningSegment = ((DeleteStatement) statement).getReturning().orElse(null);
195             return null == returningSegment ? FirebirdSQLInfoReturnValue.DELETE : FirebirdSQLInfoReturnValue.EXEC_PROCEDURE;
196         }
197         if (statement instanceof DDLStatement) {
198             return FirebirdSQLInfoReturnValue.DDL;
199         }
200         if (statement instanceof BeginTransactionStatement) {
201             return FirebirdSQLInfoReturnValue.START_TRANS;
202         }
203         if (statement instanceof CommitStatement) {
204             return FirebirdSQLInfoReturnValue.COMMIT;
205         }
206         if (statement instanceof RollbackStatement) {
207             return FirebirdSQLInfoReturnValue.ROLLBACK;
208         }
209         if (statement instanceof SavepointStatement) {
210             return FirebirdSQLInfoReturnValue.SAVEPOINT;
211         }
212         return null;
213     }
214     
215     private void skipDescribe() {
216         while (packet.getCurrentItem() != FirebirdSQLInfoPacketType.DESCRIBE_END) {
217             packet.nextItem();
218         }
219     }
220     
221     private void processDescribe(final SQLStatementContext sqlStatementContext, final Collection<FirebirdReturnColumnPacket> describeColumns, final boolean returnAll) {
222         // TODO add exception if the first item is not DESCRIBE_VARS
223         packet.nextItem();
224         Collection<FirebirdSQLInfoPacketType> requestedItems = new LinkedList<>();
225         while (packet.nextItem()) {
226             requestedItems.add(packet.getCurrentItem());
227             if (packet.getCurrentItem() == FirebirdSQLInfoPacketType.DESCRIBE_END) {
228                 Supplier<ShardingSphereDatabase> database = Suppliers.memoize(() -> ProxyContext.getInstance().getContextManager().getDatabase(connectionSession.getCurrentDatabaseName()));
229                 Supplier<Optional<ShardingSphereSchema>> schema = Suppliers.memoize(() -> database.get().findDefaultSchema());
230                 if (returnAll) {
231                     processReturnValues(sqlStatementContext, database, schema, describeColumns, requestedItems);
232                 } else {
233                     processParameters(sqlStatementContext, database, schema, describeColumns, requestedItems);
234                 }
235                 return;
236             }
237         }
238     }
239     
240     private void processReturnValues(final SQLStatementContext sqlStatementContext, final Supplier<ShardingSphereDatabase> database,
241                                      final Supplier<Optional<ShardingSphereSchema>> schema, final Collection<FirebirdReturnColumnPacket> describeColumns,
242                                      final Collection<FirebirdSQLInfoPacketType> requestedItems) {
243         int columnCount = 0;
244         for (Projection each : getProjections(sqlStatementContext, database, schema)) {
245             if (each instanceof ColumnProjection) {
246                 String tableName = ((ColumnProjection) each).getOriginalTable().getValue();
247                 ShardingSphereTable table = getTable(tableName.isEmpty() ? getTableNames(sqlStatementContext).iterator().next() : tableName, database, schema);
248                 ShardingSphereColumn column = table.getColumn(((ColumnProjection) each).getOriginalColumn().getValue());
249                 processColumn(describeColumns, requestedItems, table, column, ((ColumnProjection) each).getOwner().orElse(null), each.getAlias().orElse(null), ++columnCount);
250             } else if (each instanceof ExpressionProjection) {
251                 processExpressionProjection((ExpressionProjection) each, describeColumns, requestedItems, ++columnCount);
252             } else if (each instanceof AggregationProjection) {
253                 String functionName = ((AggregationProjection) each).getType().name();
254                 processCustomColumn(null, functionName, each.getAlias().orElse(null), getFunctionType(functionName), describeColumns, requestedItems, ++columnCount);
255             } else if (each instanceof SubqueryProjection) {
256                 SubqueryProjection subquery = (SubqueryProjection) each;
257                 processCustomColumn(null, subquery.getColumnName(), subquery.getAlias().orElse(null), Types.INTEGER, describeColumns, requestedItems, ++columnCount);
258             }
259         }
260     }
261     
262     private Collection<Projection> getProjections(final SQLStatementContext sqlStatementContext, final Supplier<ShardingSphereDatabase> database,
263                                                   final Supplier<Optional<ShardingSphereSchema>> schema) {
264         if (sqlStatementContext instanceof SelectStatementContext) {
265             SelectStatementContext selectStatementContext = (SelectStatementContext) sqlStatementContext;
266             if (!(selectStatementContext.getSqlStatement().getFrom().orElse(null) instanceof SubqueryTableSegment)
267                     && selectStatementContext.getProjectionsContext().containsUnexpandedShorthandProjection()) {
268                 return resolveUnexpandedShorthandProjections(selectStatementContext, database, schema);
269             }
270             // workaround for SubqueryTableBindUtils transform Expression and Aggregation projections to Column projection
271             if (selectStatementContext.getSubqueryContexts().isEmpty() || !(selectStatementContext.getSqlStatement().getFrom().orElse(null) instanceof SubqueryTableSegment)) {
272                 return selectStatementContext.getProjectionsContext().getExpandProjections();
273             }
274             Collection<Projection> result = new LinkedList<>();
275             for (SelectStatementContext each : selectStatementContext.getSubqueryContexts().values()) {
276                 result.addAll(each.getProjectionsContext().getExpandProjections());
277             }
278             return result;
279         }
280         if (null == returningSegment) {
281             return Collections.emptyList();
282         }
283         ProjectionEngine projectionEngine = new ProjectionEngine(sqlStatementContext.getSqlStatement().getDatabaseType());
284         Collection<Projection> result = new LinkedList<>();
285         for (ProjectionSegment each : returningSegment.getProjections().getProjections()) {
286             Projection projection = projectionEngine.createProjection(each).orElse(null);
287             if (projection instanceof ShorthandProjection) {
288                 result.addAll(processShorthandProjection(sqlStatementContext, database, schema, (ShorthandProjection) projection));
289             } else if (!(projection instanceof DerivedProjection) && null != projection) {
290                 result.add(projection);
291             }
292         }
293         return result;
294     }
295     
296     private Collection<Projection> resolveUnexpandedShorthandProjections(final SelectStatementContext selectStatementContext, final Supplier<ShardingSphereDatabase> database,
297                                                                          final Supplier<Optional<ShardingSphereSchema>> schema) {
298         Collection<Projection> result = new LinkedList<>();
299         for (Projection each : selectStatementContext.getProjectionsContext().getProjections()) {
300             if (each instanceof ShorthandProjection) {
301                 result.addAll(resolveShorthandColumns(selectStatementContext, database, schema, (ShorthandProjection) each));
302             } else if (!(each instanceof DerivedProjection)) {
303                 result.add(each);
304             }
305         }
306         return result;
307     }
308     
309     private Collection<Projection> resolveShorthandColumns(final SQLStatementContext sqlStatementContext, final Supplier<ShardingSphereDatabase> database,
310                                                            final Supplier<Optional<ShardingSphereSchema>> schema, final ShorthandProjection projection) {
311         if (!projection.getActualColumns().isEmpty()) {
312             return projection.getActualColumns();
313         }
314         Collection<Projection> result = new LinkedList<>();
315         Optional<IdentifierValue> owner = projection.getOwner();
316         for (String tableName : getTableNames(sqlStatementContext)) {
317             if (sqlStatementContext instanceof SelectStatementContext && owner.isPresent() && !owner.get().getValue().equalsIgnoreCase(tableName)
318                     && !isAliasOf(owner.get().getValue(), tableName, sqlStatementContext)) {
319                 continue;
320             }
321             ShardingSphereTable table = getTable(tableName, database, schema);
322             table.getAllColumns().forEach(each -> {
323                 ColumnSegmentBoundInfo info = new ColumnSegmentBoundInfo(null, new IdentifierValue(table.getName()), new IdentifierValue(each.getName()), TableSourceType.PHYSICAL_TABLE);
324                 result.add(new ColumnProjection(null, new IdentifierValue(each.getName()), null, sqlStatementContext.getSqlStatement().getDatabaseType(), null, null, info));
325             });
326         }
327         return result;
328     }

330     private boolean isAliasOf(final String alias, final String tableName, final SQLStatementContext sqlStatementContext) {
331         for (SimpleTableSegment each : sqlStatementContext.getTablesContext().getSimpleTables()) {
332             if (each.getAlias().isPresent() && each.getAlias().get().getValue().equalsIgnoreCase(alias) && each.getTableName().getIdentifier().getValue().equalsIgnoreCase(tableName)) {
333                 return true;
334             }
335         }
336         return false;
337     }
338     
339     private Collection<Projection> processShorthandProjection(final SQLStatementContext sqlStatementContext, final Supplier<ShardingSphereDatabase> database,
340                                                               final Supplier<Optional<ShardingSphereSchema>> schema, final ShorthandProjection projection) {
341         return resolveShorthandColumns(sqlStatementContext, database, schema, projection);
342     }

344     @HighFrequencyInvocation
345     private ShardingSphereTable getTable(final String tableName, final Supplier<ShardingSphereDatabase> database, final Supplier<Optional<ShardingSphereSchema>> schema) {
346         ShardingSphereTable result = schema.get().map(optional -> optional.getTable(tableName)).orElse(null);
347         if (null == result) {
348             result = database.get().getSchema("system_tables").getTable(tableName);
349         }
350         ShardingSpherePreconditions.checkNotNull(result, () -> new TableNotFoundException(tableName));
351         return result;
352     }

354     private void processParameters(final SQLStatementContext sqlStatementContext, final Supplier<ShardingSphereDatabase> database,
355                                    final Supplier<Optional<ShardingSphereSchema>> schema, final Collection<FirebirdReturnColumnPacket> describeColumns,
356                                    final Collection<FirebirdSQLInfoPacketType> requestedItems) {
357         if (sqlStatementContext instanceof InsertStatementContext) {
358             processInsertStatement((InsertStatementContext) sqlStatementContext, database, schema, describeColumns, requestedItems);
359             return;
360         }
361         Collection<ColumnSegment> affectedColumns = findAffectedColumns(sqlStatementContext);
362         int parametersCount = sqlStatementContext.getSqlStatement().getParameterMarkers().size();
363         int columnCount = 0;
364         for (ColumnSegment columnSegment : affectedColumns) {
365             ShardingSphereTable table = getTable(columnSegment.getColumnBoundInfo().getOriginalTable().getValue(), database, schema);
366             ShardingSphereColumn column = table.getColumn(columnSegment.getColumnBoundInfo().getOriginalColumn().getValue());
367             processColumn(describeColumns, requestedItems, table, column, columnSegment.getOwner().map(OwnerSegment::getIdentifier).orElse(null), columnSegment.getIdentifier(), ++columnCount);
368         }
369         for (int i = 0; i < parametersCount - affectedColumns.size(); i++) {
370             processCustomColumn(null, null, null, 12, describeColumns, requestedItems, ++columnCount);
371         }
372     }
373     
374     private void processInsertStatement(final InsertStatementContext sqlStatementContext, final Supplier<ShardingSphereDatabase> database,
375                                         final Supplier<Optional<ShardingSphereSchema>> schema, final Collection<FirebirdReturnColumnPacket> describeColumns,
376                                         final Collection<FirebirdSQLInfoPacketType> requestedItems) {
377         Collection<String> tableNames = getTableNames(sqlStatementContext);
378         Collection<String> affectedColumns = new LinkedList<>();
379         for (InsertValueContext context : sqlStatementContext.getInsertValueContexts()) {
380             affectedColumns.addAll(processInsertValueContext(sqlStatementContext, context));
381         }
382         int columnCount = 0;
383         for (String tableName : tableNames) {
384             ShardingSphereTable table = getTable(tableName, database, schema);
385             for (String columnName : affectedColumns) {
386                 ShardingSphereColumn column = table.getColumn(columnName);
387                 processColumn(describeColumns, requestedItems, table, column, null, null, ++columnCount);
388             }
389         }
390     }
391     
392     private Collection<String> processInsertValueContext(final InsertStatementContext context, final InsertValueContext valueContext) {
393         Collection<String> result = new LinkedList<>();
394         for (int i = 0; i < valueContext.getValueExpressions().size(); i++) {
395             if (valueContext.getValueExpressions().get(i) instanceof ParameterMarkerExpressionSegment) {
396                 result.add(context.getColumnNames().get(i));
397             }
398         }
399         return result;
400     }
401     
402     private Collection<String> getTableNames(final SQLStatementContext sqlStatementContext) {
403         TablesContext tablesContext = sqlStatementContext.getTablesContext();
404         return null == tablesContext ? Collections.emptyList() : tablesContext.getTableNames();
405     }
406     
407     private Collection<ColumnSegment> findAffectedColumns(final SQLStatementContext sqlStatementContext) {
408         Collection<ColumnSegment> result = new LinkedList<>();
409         if (sqlStatementContext instanceof UpdateStatementContext) {
410             for (ColumnAssignmentSegment segment : ((UpdateStatementContext) sqlStatementContext).getSqlStatement().getSetAssignment().getAssignments()) {
411                 if (segment.getValue() instanceof ParameterMarkerExpressionSegment) {
412                     result.add(segment.getColumns().get(0));
413                 }
414             }
415         }
416         if (sqlStatementContext instanceof WhereContextAvailable) {
417             for (WhereSegment each : getWhereSegments((WhereContextAvailable) sqlStatementContext)) {
418                 processExpr(each.getExpr(), result);
419             }
420         }
421         return result;
422     }
423     
424     @HighFrequencyInvocation
425     private Collection<WhereSegment> getWhereSegments(final WhereContextAvailable sqlStatementContext) {
426         if (!(sqlStatementContext instanceof SelectStatementContext) || !((SelectStatementContext) sqlStatementContext).isContainsCombine()) {
427             return sqlStatementContext.getWhereSegments();
428         }
429         List<WhereSegment> result = new ArrayList<>(sqlStatementContext.getWhereSegments());
430         appendCombineWhereSegments(((SelectStatementContext) sqlStatementContext).getSqlStatement(), result);
431         result.sort(Comparator.comparingInt(WhereSegment::getStartIndex));
432         return result;
433     }
434     
435     private void appendCombineWhereSegments(final SelectStatement selectStatement, final Collection<WhereSegment> whereSegments) {
436         selectStatement.getCombine().ifPresent(combine -> {
437             appendBranchWhereSegments(combine.getLeft().getSelect(), whereSegments);
438             appendBranchWhereSegments(combine.getRight().getSelect(), whereSegments);
439         });
440     }
441     
442     private void appendBranchWhereSegments(final SelectStatement branch, final Collection<WhereSegment> whereSegments) {
443         branch.getWhere().filter(whereSegment -> !whereSegments.contains(whereSegment)).ifPresent(whereSegments::add);
444         appendCombineWhereSegments(branch, whereSegments);
445     }
446     
447     private boolean processExpr(final ExpressionSegment expr, final Collection<ColumnSegment> affectedColumns) {
448         if (!(expr instanceof BinaryOperationExpression)) {
449             return expr instanceof ParameterMarkerExpressionSegment;
450         }
451         BinaryOperationExpression binary = (BinaryOperationExpression) expr;
452         processExpr(binary.getLeft(), affectedColumns);
453         boolean rightIsParam = processExpr(binary.getRight(), affectedColumns);
454         if (rightIsParam && binary.getLeft() instanceof ColumnSegment) {
455             affectedColumns.add((ColumnSegment) binary.getLeft());
456         }
457         return false;
458     }
459     
460     private void processExpressionProjection(final ExpressionProjection expr, final Collection<FirebirdReturnColumnPacket> describeColumns, final Collection<FirebirdSQLInfoPacketType> requestedItems,
461                                              final int columnCount) {
462         ExpressionSegment exprSegment = expr.getExpressionSegment().getExpr();
463         if (exprSegment instanceof FunctionSegment) {
464             String functionName = ((FunctionSegment) exprSegment).getFunctionName();
465             processCustomColumn(null, functionName, expr.getAlias().orElse(null), getFunctionType(functionName), describeColumns, requestedItems, columnCount);
466         } else if (exprSegment instanceof BinaryOperationExpression) {
467             String operationName = getOperationName(((BinaryOperationExpression) exprSegment).getOperator());
468             int operationType = getOperationType(((BinaryOperationExpression) exprSegment).getOperator());
469             processCustomColumn(null, operationName, expr.getAlias().orElse(null), operationType, describeColumns, requestedItems, columnCount);
470         } else if (exprSegment instanceof LiteralExpressionSegment) {
471             Object value = ((LiteralExpressionSegment) exprSegment).getLiterals();
472             int type = Types.NULL;
473             if (value instanceof String) {
474                 type = Types.VARCHAR;
475             } else if (value instanceof Integer) {
476                 type = Types.INTEGER;
477             } else if (value instanceof Long) {
478                 type = Types.BIGINT;
479             } else if (value instanceof Number) {
480                 type = Types.NUMERIC;
481             }
482             processLiteralColumn(value, expr.getAlias().orElse(null), type, describeColumns, requestedItems, columnCount);
483         }
484     }
485     
486     private int getFunctionType(final String functionName) {
487         // TODO add proper coalesce and other conditional functions return types
488         switch (functionName.toLowerCase(Locale.ENGLISH)) {
489             case "substring":
490             case "current_role":
491             case "current_user":
492             case "coalesce":
493                 return Types.VARCHAR;
494             case "gen_id":
495             case "count":
496                 return Types.BIGINT;
497             case "current_timestamp":
498                 return Types.TIMESTAMP;
499             default:
500                 return Types.INTEGER;
501         }
502     }
503     
504     private String getOperationName(final String operationType) {
505         return "||".equals(operationType) ? "CONCATENATION" : operationType;
506     }
507     
508     private int getOperationType(final String operationType) {
509         return "||".equals(operationType) ? Types.VARCHAR : Types.INTEGER;
510     }

512     @HighFrequencyInvocation
513     private void processLiteralColumn(@Nullable final Object value, @Nullable final IdentifierValue columnAlias, final int dataType, final Collection<FirebirdReturnColumnPacket> describeColumns,
514                                       final Collection<FirebirdSQLInfoPacketType> requestedItems, final int columnCount) {
515         ShardingSphereColumn column = new ShardingSphereColumn("CONSTANT", dataType, false, false, true, true, false, false);
516         String columnAliasString = null == columnAlias ? column.getName() : columnAlias.getValue();
517         String owner = connectionSession.getConnectionContext().getGrantee().getUsername();
518         Integer columnLength = value instanceof String ? getStringLiteralLength((String) value) : null;
519         describeColumns.add(new FirebirdReturnColumnPacket(requestedItems, columnCount, LITERAL_TABLE, column, null, columnAliasString, owner, columnLength, false, null));
520     }
521     
522     private int getStringLiteralLength(final String value) {
523         String text = value.replace("''", "'");
524         return text.codePointCount(0, text.length());
525     }

527     private void processCustomColumn(final String tableName, final String columnName, final IdentifierValue columnAlias, final int dataType,
528                                      final Collection<FirebirdReturnColumnPacket> describeColumns, final Collection<FirebirdSQLInfoPacketType> requestedItems, final int columnCount) {
529         ShardingSphereTable table = new ShardingSphereTable(tableName, Collections.emptyList(), Collections.emptyList(), Collections.emptyList());
530         ShardingSphereColumn column = new ShardingSphereColumn(columnName, dataType, false, false, true, true, false, false);
531         processColumn(describeColumns, requestedItems, table, column, null, columnAlias, columnCount);
532     }
533     
534     private void processColumn(final Collection<FirebirdReturnColumnPacket> describeColumns, final Collection<FirebirdSQLInfoPacketType> requestedItems, final ShardingSphereTable table,
535                                final ShardingSphereColumn column, final IdentifierValue tableAlias, final IdentifierValue columnAlias, final int idx) {
536         String tableAliasString = null == tableAlias ? table.getName() : tableAlias.getValue();
537         String columnAliasString = null == columnAlias ? column.getName() : columnAlias.getValue();
538         String owner = connectionSession.getConnectionContext().getGrantee().getUsername();
539         boolean blobColumn = isBlobColumn(table, column);
540         Integer blobSubtype = blobColumn ? resolveBlobSubtype(table, column) : null;
541         Integer columnLength = blobColumn ? null : resolveColumnLength(table, column);
542         describeColumns.add(new FirebirdReturnColumnPacket(requestedItems, idx, table, column, tableAliasString, columnAliasString, owner, columnLength, blobColumn, blobSubtype));
543     }
544     
545     private boolean isBlobColumn(final ShardingSphereTable table, final ShardingSphereColumn column) {
546         if (null == table || null == column) {
547             return false;
548         }
549         if (null != table.getName() && FirebirdBlobInfoRegistry.isBlobColumn(connectionSession.getCurrentDatabaseName(), getActualTableName(table.getName()), column.getName())) {
550             return true;
551         }
552         return Types.BLOB == column.getDataType();
553     }
554     
555     @HighFrequencyInvocation
556     private String getActualTableName(final String logicTableName) {
557         ShardingSphereDatabase database = ProxyContext.getInstance().getContextManager().getDatabase(connectionSession.getCurrentDatabaseName());
558         for (DataNodeRuleAttribute each : database.getRuleMetaData().getAttributes(DataNodeRuleAttribute.class)) {
559             Optional<String> actualTableName = each.findFirstActualTable(logicTableName);
560             if (actualTableName.isPresent()) {
561                 return actualTableName.get();
562             }
563         }
564         return logicTableName;
565     }
566     
567     private Integer resolveBlobSubtype(final ShardingSphereTable table, final ShardingSphereColumn column) {
568         OptionalInt subtype = FirebirdBlobInfoRegistry.findBlobSubtype(connectionSession.getCurrentDatabaseName(), getActualTableName(table.getName()), column.getName());
569         return subtype.isPresent() ? subtype.getAsInt() : null;
570     }
571     
572     private Integer resolveColumnLength(final ShardingSphereTable table, final ShardingSphereColumn column) {
573         if (null == table || null == column || !isDynamicLengthType(column.getDataType())) {
574             return null;
575         }
576         OptionalInt columnSize = FirebirdNonFixedLengthColumnSizeRegistry.findColumnSize(connectionSession.getCurrentDatabaseName(), table.getName(), column.getName());
577         return columnSize.isPresent() ? columnSize.getAsInt() : null;
578     }

580     private boolean isDynamicLengthType(final int dataType) {
581         switch (dataType) {
582             case Types.CHAR:
583             case Types.NCHAR:
584             case Types.VARCHAR:
585             case Types.NVARCHAR:
586             case Types.LONGVARCHAR:
587             case Types.LONGNVARCHAR:
588                 return true;
589             default:
590                 return false;
591         }
592     }
