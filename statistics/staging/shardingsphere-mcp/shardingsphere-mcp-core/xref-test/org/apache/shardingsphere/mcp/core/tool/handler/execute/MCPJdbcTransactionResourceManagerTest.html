49      @ParameterizedTest(name = "{0}")
50      @MethodSource("commitCleanupFailureCases")
51      void assertCommitTransactionWithCleanupFailure(final String name, final boolean resetFailure, final boolean closeFailure, final String expectedMessage,
52                                                     final int expectedSuppressed) throws SQLException {
53          Connection connection = mock(Connection.class);
54          RuntimeDatabaseConfiguration databaseConfig = mock(RuntimeDatabaseConfiguration.class);
55          when(databaseConfig.openConnection("foo_db")).thenReturn(connection);
56          MCPJdbcTransactionResourceManager manager = new MCPJdbcTransactionResourceManager(Map.of("foo_db", databaseConfig));
57          manager.beginTransaction("foo_session", "foo_db");
58          if (resetFailure) {
59              doThrow(new SQLException("reset failed")).when(connection).setAutoCommit(true);
60          }
61          if (closeFailure) {
62              doThrow(new SQLException("close failed")).when(connection).close();
63          }
64          IllegalStateException actual = assertThrows(IllegalStateException.class, () -> manager.commitTransaction("foo_session"));
65          assertThat(actual.getMessage(), is(expectedMessage));
66          assertThat(actual.getCause().getSuppressed().length, is(expectedSuppressed));
67          assertTrue(manager.findTransactionConnection("foo_session", "foo_db").isEmpty());
68          verify(connection).commit();
69          verify(connection).close();
70      }

76                  Arguments.of("reset and close failure", true, true, "reset failed", 1));

101     @ParameterizedTest(name = "{0}")
102     @MethodSource("assertBeginTransactionCases")
103     void assertBeginTransaction(final String name, final String configuredDatabaseName, final String activeDatabaseName, final String databaseName,
104                                 final String openFailureMessage, final String expectedMessage) throws SQLException {
105         Connection connection = mock(Connection.class);
106         RuntimeDatabaseConfiguration runtimeDatabaseConfig = mock(RuntimeDatabaseConfiguration.class);
107         if (null != configuredDatabaseName && null == openFailureMessage) {
108             when(runtimeDatabaseConfig.openConnection(configuredDatabaseName)).thenReturn(connection);
109         }
110         if (null != configuredDatabaseName && null != openFailureMessage) {
111             when(runtimeDatabaseConfig.openConnection(configuredDatabaseName)).thenThrow(new SQLException(openFailureMessage));
112         }
113         Map<String, RuntimeDatabaseConfiguration> runtimeDatabases = null == configuredDatabaseName ? Collections.emptyMap() : Map.of(configuredDatabaseName, runtimeDatabaseConfig);
114         MCPJdbcTransactionResourceManager manager = new MCPJdbcTransactionResourceManager(runtimeDatabases);
115         if (null != activeDatabaseName) {
116             manager.beginTransaction("session-1", activeDatabaseName);
117         }
118         if (null != expectedMessage) {
119             assertThat(assertThrows(IllegalStateException.class, () -> manager.beginTransaction("session-1", databaseName)).getMessage(), is(expectedMessage));
120             return;
121         }
122         manager.beginTransaction("session-1", databaseName);
123         verify(runtimeDatabaseConfig).openConnection(databaseName);
124         verify(connection).setAutoCommit(false);
125     }

258     @Test
259     void assertCloseSession() throws SQLException {
260         TransactionState state = new TransactionState("NEW");
261         Connection connection = createTransactionalConnection(state);
262         MCPJdbcTransactionResourceManager manager = createResourceManager(connection);
263         manager.beginTransaction("session-1", "logic_db");
264         try (Statement statement = manager.findTransactionConnection("session-1", "logic_db").orElseThrow(IllegalStateException::new).createStatement()) {
265             statement.execute("UPDATE public.orders SET status = 'PROCESSING' WHERE order_id = 1");
266         }
267         assertThat(state.getCurrentStatus(), is("PROCESSING"));
268         manager.closeSession("session-1");
269         assertThat(state.getCurrentStatus(), is("NEW"));
270         verify(connection).rollback();
271         verify(connection).setAutoCommit(true);
272         verify(connection).close();
273     }

282         Connection connection = mock(Connection.class);

284         manager.beginTransaction("session-1", "logic_db");
285         doThrow(new SQLException("rollback failed")).when(connection).rollback();

293     private MCPJdbcTransactionResourceManager createResourceManager(final Connection connection) throws SQLException {
294         RuntimeDatabaseConfiguration runtimeDatabaseConfig = mock(RuntimeDatabaseConfiguration.class);
295         when(runtimeDatabaseConfig.openConnection("logic_db")).thenReturn(connection);
296         return new MCPJdbcTransactionResourceManager(Map.of("logic_db", runtimeDatabaseConfig));
297     }

299     private Connection createTransactionalConnection(final TransactionState state) throws SQLException {
300         Connection result = mock(Connection.class);
301         Statement statement = mock(Statement.class);
302         when(result.createStatement()).thenReturn(statement);
303         when(statement.execute("UPDATE public.orders SET status = 'PROCESSING' WHERE order_id = 1")).thenAnswer(invocation -> {
304             state.update("PROCESSING");
305             return true;
306         });
307         doAnswer(invocation -> {
308             state.rollback();
309             return null;
310         }).when(result).rollback();
311         return result;
312     }

365     private static final class TransactionState {
366         
367         private final String initialStatus;
368         
369         private String workingStatus;
370         
371         private TransactionState(final String initialStatus) {
372             this.initialStatus = initialStatus;
373             workingStatus = initialStatus;
374         }
375         
376         private void update(final String newStatus) {
377             workingStatus = newStatus;
378         }
379         
380         private void rollback() {
381             workingStatus = initialStatus;
382         }
383         
384         private String getCurrentStatus() {
385             return workingStatus;
386         }
