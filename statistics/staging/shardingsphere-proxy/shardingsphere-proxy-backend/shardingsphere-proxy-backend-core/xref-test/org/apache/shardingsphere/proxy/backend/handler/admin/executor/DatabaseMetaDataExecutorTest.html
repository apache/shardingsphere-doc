95                  new ResourceMetaData(Collections.singletonMap("foo_ds", new MockedDataSource(mockConnection(expectedResultSetMap)))), mock(), Collections.emptyList(),

99          assertThat(executor.getRows().get(0).get("sn"), is("foo_ds"));
100         assertThat(executor.getRows().get(0).get("DEFAULT_CHARACTER_SET_NAME"), is("utf8mb4"));
101     }

167     @ParameterizedTest(name = "{0}")
168     @ValueSource(strings = {"filtered_rows", "aligned_rows"})
169     void assertExecuteWithPreProcessedRows(final String scenario) throws SQLException {
170         if ("filtered_rows".equals(scenario)) {
171             Map<String, String> expectedResultSetMap = Collections.singletonMap("foo_column", "foo_value");
172             ShardingSphereDatabase database = new ShardingSphereDatabase("auth_db",
173                     databaseType, new ResourceMetaData(Collections.singletonMap("foo_ds", new MockedDataSource(mockConnection(expectedResultSetMap)))), mock(), Collections.emptyList(),
174                     new ConfigurationProperties(new Properties()));
175             DatabaseMetaDataExecutor executor = mock(DatabaseMetaDataExecutor.class, withSettings().useConstructor(
176                     "SELECT foo_column FROM foo_table", Collections.emptyList()).defaultAnswer(CALLS_REAL_METHODS));
177             doAnswer(invocation -> {
178                 Map<String, Object> rows = invocation.getArgument(1);
179                 rows.clear();
180                 return null;
181             }).when(executor).preProcess(any(ShardingSphereDatabase.class), anyMap(), anyMap());
182             executor.execute(connectionSession, mockMetaData(database, Collections.singleton("auth_db")));
183             assertTrue(executor.getRows().isEmpty());
184             assertThat(executor.getQueryResultMetaData().getColumnCount(), is(1));
185             assertFalse(executor.getMergedResult().next());
186             return;
187         }
188         Map<String, String> firstResultSetMap = new LinkedHashMap<>(2, 1F);
189         firstResultSetMap.put("foo_column", "foo_value");
190         firstResultSetMap.put("bar_column", "bar_value");
191         Map<String, String> secondResultSetMap = new LinkedHashMap<>(2, 1F);
192         secondResultSetMap.put("foo_column", "second_foo_value");
193         secondResultSetMap.put("bar_column", "second_bar_value");
194         ShardingSphereDatabase database = new ShardingSphereDatabase("auth_db",
195                 databaseType, new ResourceMetaData(Collections.singletonMap("foo_ds", new MockedDataSource(mockConnection(firstResultSetMap, secondResultSetMap)))), mock(),
196                 Collections.emptyList(), new ConfigurationProperties(new Properties()));
197         DatabaseMetaDataExecutor executor =
198                 mock(DatabaseMetaDataExecutor.class, withSettings().useConstructor("SELECT foo_column, bar_column FROM foo_table", Collections.emptyList()).defaultAnswer(CALLS_REAL_METHODS));
199         doAnswer(invocation -> {
200             Map<String, Object> rows = invocation.getArgument(1);
201             if ("foo_value".equals(rows.get("foo_column"))) {
202                 rows.remove("bar_column");
203                 rows.put("baz_column", "baz_value");
204             }
205             return null;
206         }).when(executor).preProcess(any(ShardingSphereDatabase.class), anyMap(), anyMap());
207         executor.execute(connectionSession, mockMetaData(database, Collections.singleton("auth_db")));
208         assertThat(executor.getQueryResultMetaData().getColumnCount(), is(3));
209         assertThat(executor.getQueryResultMetaData().getColumnLabel(1), is("foo_column"));
210         assertThat(executor.getQueryResultMetaData().getColumnLabel(2), is("baz_column"));
211         assertThat(executor.getQueryResultMetaData().getColumnLabel(3), is("bar_column"));
212         MergedResult actualMergedResult = executor.getMergedResult();
213         assertTrue(actualMergedResult.next());
214         assertThat(actualMergedResult.getValue(1, String.class), is("foo_value"));
215         assertThat(actualMergedResult.getValue(2, String.class), is("baz_value"));
216         assertThat(actualMergedResult.getValue(3, String.class), is(""));
217         assertTrue(actualMergedResult.next());
218         assertThat(actualMergedResult.getValue(1, String.class), is("second_foo_value"));
219         assertThat(actualMergedResult.getValue(2, String.class), is(""));
220         assertThat(actualMergedResult.getValue(3, String.class), is("second_bar_value"));
221         assertFalse(actualMergedResult.next());
222     }

224     private Connection mockConnection(final Map<String, String> expectedResultSetMap, final Map<String, String>... additionalExpectedResultSetMaps) throws SQLException {
225         Connection result = mock(Connection.class, RETURNS_DEEP_STUBS);
226         ResultSet resultSet = mockResultSet(expectedResultSetMap, additionalExpectedResultSetMaps);
227         when(result.prepareStatement(any(String.class)).executeQuery()).thenReturn(resultSet);
228         return result;
229     }
230     
231     private ResultSet mockResultSet(final Map<String, String> expectedResultSetMap, final Map<String, String>... additionalExpectedResultSetMaps) throws SQLException {
232         ResultSet result = mock(ResultSet.class, RETURNS_DEEP_STUBS);
233         List<String> keys = new ArrayList<>(expectedResultSetMap.keySet());
234         for (int i = 0; i < keys.size(); i++) {
235             when(result.getMetaData().getColumnName(i + 1)).thenReturn(keys.get(i));
236             when(result.getMetaData().getColumnLabel(i + 1)).thenReturn(keys.get(i));
237             List<String> columnValues = new ArrayList<>(additionalExpectedResultSetMaps.length + 1);
238             columnValues.add(expectedResultSetMap.get(keys.get(i)));
239             for (Map<String, String> each : additionalExpectedResultSetMaps) {
240                 columnValues.add(each.get(keys.get(i)));
241             }
242             when(result.getString(i + 1)).thenReturn(columnValues.get(0), columnValues.subList(1, columnValues.size()).toArray(new String[0]));
243         }
244         List<Boolean> nextResults = new ArrayList<>(additionalExpectedResultSetMaps.length + 2);
245         for (int i = 0; i < additionalExpectedResultSetMaps.length + 1; i++) {
246             nextResults.add(true);
247         }
248         nextResults.add(false);
249         when(result.next()).thenReturn(nextResults.get(0), nextResults.subList(1, nextResults.size()).toArray(new Boolean[0]));
250         when(result.getMetaData().getColumnCount()).thenReturn(expectedResultSetMap.size());
251         return result;
252     }

254     private ShardingSphereMetaData mockMetaData(final ShardingSphereDatabase database, final Collection<String> permittedDatabases) {
255         AuthorityRule authorityRule = mock(AuthorityRule.class);
256         when(authorityRule.findUser(grantee)).thenReturn(Optional.empty());
257         when(authorityRule.findPrivileges(grantee)).thenReturn(Optional.of(new DatabasePermittedPrivileges(permittedDatabases)));
258         return new ShardingSphereMetaData(Collections.singleton(database), mock(), new RuleMetaData(Collections.singleton(authorityRule)), mock());
259     }
