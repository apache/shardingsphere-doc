60  @ExtendWith(AutoMockExtension.class)
61  @StaticMockSettings({TypedSPILoader.class, DataSourcePoolPropertiesCreator.class, GenericSchemaManager.class})
62  class MetaDataPersistFacadeTest {
63      
64      @Mock
65      private DataSourceUnitPersistService dataSourceUnitService;
66      
67      @Mock(answer = Answers.RETURNS_DEEP_STUBS)
68      private DatabaseMetaDataPersistFacade databaseMetaDataFacade;
69      
70      @Mock
71      private DatabaseRulePersistService databaseRuleService;
72      
73      @Mock
74      private GlobalRulePersistService globalRuleService;
75      
76      @Mock
77      private PropertiesPersistService propsService;
78      
79      private MetaDataPersistFacade metaDataPersistFacade;
80      
81      @BeforeEach
82      void setUp() throws ReflectiveOperationException {
83          metaDataPersistFacade = new MetaDataPersistFacade(mock(PersistRepository.class));
84          Plugins.getMemberAccessor().set(MetaDataPersistFacade.class.getDeclaredField("dataSourceUnitService"), metaDataPersistFacade, dataSourceUnitService);
85          Plugins.getMemberAccessor().set(MetaDataPersistFacade.class.getDeclaredField("databaseMetaDataFacade"), metaDataPersistFacade, databaseMetaDataFacade);
86          Plugins.getMemberAccessor().set(MetaDataPersistFacade.class.getDeclaredField("databaseRuleService"), metaDataPersistFacade, databaseRuleService);
87          Plugins.getMemberAccessor().set(MetaDataPersistFacade.class.getDeclaredField("globalRuleService"), metaDataPersistFacade, globalRuleService);
88          Plugins.getMemberAccessor().set(MetaDataPersistFacade.class.getDeclaredField("propsService"), metaDataPersistFacade, propsService);
89      }
90      
91      @Test
92      void assertPersistGlobalRuleConfiguration() {
93          Collection<RuleConfiguration> globalRuleConfigs = Collections.singleton(mock(RuleConfiguration.class));
94          Properties props = new Properties();
95          metaDataPersistFacade.persistGlobalRuleConfiguration(globalRuleConfigs, props);
96          verify(globalRuleService).persist(globalRuleConfigs);
97          verify(propsService).persist(props);
98      }
99      
100     @Test
101     void assertPersistConfigurationsWithEmptyDatabase() {
102         metaDataPersistFacade.persistConfigurations("foo_db", mock(DatabaseConfiguration.class), Collections.emptyMap(), Collections.emptyList());
103         verify(databaseMetaDataFacade.getDatabase()).add("foo_db");
104     }
105     
106     @Test
107     void assertPersistConfigurationsWithDatabaseRuleConfigurations() {
108         DatabaseConfiguration databaseConfig = mock(DatabaseConfiguration.class, RETURNS_DEEP_STUBS);
109         when(databaseConfig.getStorageUnits()).thenReturn(Collections.emptyMap());
110         when(databaseConfig.getRuleConfigurations().isEmpty()).thenReturn(false);
111         ShardingSphereRule rule = mock(ShardingSphereRule.class);
112         RuleConfiguration ruleConfig = mock(RuleConfiguration.class);
113         when(rule.getConfiguration()).thenReturn(ruleConfig);
114         when(TypedSPILoader.findService(RuleConfigurationDecorator.class, ruleConfig.getClass())).thenReturn(Optional.of(mock(RuleConfigurationDecorator.class)));
115         metaDataPersistFacade.persistConfigurations("foo_db", databaseConfig, Collections.emptyMap(), Collections.singleton(rule));
116         verify(dataSourceUnitService).persist("foo_db", Collections.emptyMap());
117         verify(databaseRuleService).persist(eq("foo_db"), any());
118     }
119     
120     @Test
121     void assertPersistConfigurationsWithDataSourcePoolProperties() {
122         DatabaseConfiguration databaseConfig = mock(DatabaseConfiguration.class, RETURNS_DEEP_STUBS);
123         when(databaseConfig.getStorageUnits()).thenReturn(Collections.singletonMap("foo_ds", mock(StorageUnit.class, RETURNS_DEEP_STUBS)));
124         metaDataPersistFacade.persistConfigurations("foo_db", databaseConfig, Collections.emptyMap(), Collections.emptyList());
125         verify(dataSourceUnitService).persist(eq("foo_db"), any());
126         verify(databaseRuleService).persist("foo_db", Collections.emptyList());
127     }
128     
129     @Test
130     void assertLoadDataSourceConfigurations() {
131         DataSourcePoolProperties dataSourcePoolProps = mock(DataSourcePoolProperties.class);
132         when(dataSourceUnitService.load("foo_db")).thenReturn(Collections.singletonMap("foo_ds", dataSourcePoolProps));
133         DataSourceConfiguration dataSourceConfig = mock(DataSourceConfiguration.class);
134         when(DataSourcePoolPropertiesCreator.createConfiguration(dataSourcePoolProps)).thenReturn(dataSourceConfig);
135         Map<String, DataSourceConfiguration> actual = metaDataPersistFacade.loadDataSourceConfigurations("foo_db");
136         assertThat(actual.size(), is(1));
137         assertThat(actual.get("foo_ds"), is(dataSourceConfig));
138     }
