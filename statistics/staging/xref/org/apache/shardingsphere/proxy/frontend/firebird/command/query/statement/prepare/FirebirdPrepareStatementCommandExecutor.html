105     @Override
106     public Collection<DatabasePacket> execute() throws SQLException {
107         MetaDataContexts metaDataContexts = ProxyContext.getInstance().getContextManager().getMetaDataContexts();
108         SQLParserRule sqlParserRule = metaDataContexts.getMetaData().getGlobalRuleMetaData().getSingleRule(SQLParserRule.class);
109         DatabaseType databaseType = TypedSPILoader.getService(DatabaseType.class, "Firebird");
110         SQLStatement sqlStatement = sqlParserRule.getSQLParserEngine(databaseType).parse(packet.getSQL(), true);
111         SQLStatementContext sqlStatementContext = new SQLBindEngine(
112                 metaDataContexts.getMetaData(), connectionSession.getCurrentDatabaseName(), packet.getHintValueContext()).bind(sqlStatement, Collections.emptyList());
113         FirebirdServerPreparedStatement serverPreparedStatement = new FirebirdServerPreparedStatement(packet.getSQL(), sqlStatementContext, packet.getHintValueContext());
114         connectionSession.getServerPreparedStatementRegistry().addPreparedStatement(getStatementId(), serverPreparedStatement);
115         return createResponse(sqlStatementContext, metaDataContexts);
116     }
117     
118     private int getStatementId() {
119         return packet.isValidStatementHandle() ? packet.getStatementId() : FirebirdStatementIdGenerator.getInstance().getStatementId(connectionSession.getConnectionId());
120     }
121     
122     private Collection<DatabasePacket> createResponse(final SQLStatementContext sqlStatementContext, final MetaDataContexts metaDataContexts) {
123         FirebirdSQLInfoReturnValue statementType = getFirebirdStatementType(sqlStatementContext.getSqlStatement());
124         FirebirdPrepareStatementReturnPacket returnPacket = new FirebirdPrepareStatementReturnPacket();
125         while (packet.nextItem()) {
126             switch (packet.getCurrentItem()) {
127                 case STMT_TYPE:
128                     returnPacket.setType(statementType);
129                     break;
130                 case SELECT:
131                     if (FirebirdSQLInfoReturnValue.isSelectDescribable(statementType.getCode())) {
132                         processDescribe(sqlStatementContext, metaDataContexts, returnPacket.getDescribeSelect(), true);
133                     } else {
134                         skipDescribe();
135                     }
136                     break;
137                 case BIND:
138                     if (FirebirdSQLInfoReturnValue.isBindDescribable(statementType.getCode())) {
139                         processDescribe(sqlStatementContext, metaDataContexts, returnPacket.getDescribeBind(), false);
140                     } else {
141                         skipDescribe();
142                     }
143                     break;
144                 default:
145                     throw new FirebirdProtocolException("Unknown statement info request type %d", packet.getCurrentItem());
146             }
147         }
148         return Collections.singleton(new FirebirdGenericResponsePacket().setData(returnPacket));
149     }
150     
151     private FirebirdSQLInfoReturnValue getFirebirdStatementType(final SQLStatement statement) {
152         if (statement instanceof SelectStatement) {
153             return FirebirdSQLInfoReturnValue.SELECT;
154         }
155         if (statement instanceof InsertStatement) {
156             if (((InsertStatement) statement).getReturning().isPresent()) {
157                 returningSegment = ((InsertStatement) statement).getReturning().orElse(null);
158                 return FirebirdSQLInfoReturnValue.EXEC_PROCEDURE;
159             }
160             return FirebirdSQLInfoReturnValue.INSERT;
161         }
162         if (statement instanceof UpdateStatement) {
163             if (((UpdateStatement) statement).getReturning().isPresent()) {
164                 returningSegment = ((UpdateStatement) statement).getReturning().orElse(null);
165                 return FirebirdSQLInfoReturnValue.EXEC_PROCEDURE;
166             }
167             return FirebirdSQLInfoReturnValue.UPDATE;
168         }
169         if (statement instanceof DeleteStatement) {
170             if (((DeleteStatement) statement).getReturning().isPresent()) {
171                 returningSegment = ((DeleteStatement) statement).getReturning().orElse(null);
172                 return FirebirdSQLInfoReturnValue.EXEC_PROCEDURE;
173             }
174             return FirebirdSQLInfoReturnValue.DELETE;
175         }
176         if (statement instanceof DDLStatement) {
177             return FirebirdSQLInfoReturnValue.DDL;
178         }
179         if (statement instanceof BeginTransactionStatement) {
180             return FirebirdSQLInfoReturnValue.START_TRANS;
181         }
182         if (statement instanceof CommitStatement) {
183             return FirebirdSQLInfoReturnValue.COMMIT;
184         }
185         if (statement instanceof RollbackStatement) {
186             return FirebirdSQLInfoReturnValue.ROLLBACK;
187         }
188         if (statement instanceof SavepointStatement) {
189             return FirebirdSQLInfoReturnValue.SAVEPOINT;
190         }
191         return null;
192     }
193     
194     private void skipDescribe() {
195         while (packet.getCurrentItem() != FirebirdSQLInfoPacketType.DESCRIBE_END) {
196             packet.nextItem();
197         }
198     }
199     
200     private void processDescribe(final SQLStatementContext sqlStatementContext, final MetaDataContexts metaDataContexts, final Collection<FirebirdReturnColumnPacket> describeColumns,
201                                  final boolean returnAll) {
202         // TODO add exception if the first item is not DESCRIBE_VARS
203         packet.nextItem();
204         Collection<FirebirdSQLInfoPacketType> requestedItems = new LinkedList<>();
205         while (packet.nextItem()) {
206             requestedItems.add(packet.getCurrentItem());
207             if (packet.getCurrentItem() == FirebirdSQLInfoPacketType.DESCRIBE_END) {
208                 if (returnAll) {
209                     processReturnValues(sqlStatementContext, metaDataContexts, describeColumns, requestedItems);
210                 } else {
211                     processParameters(sqlStatementContext, metaDataContexts, describeColumns, requestedItems);
212                 }
213                 return;
214             }
215         }
216     }

218     private int processReturnValues(final SQLStatementContext sqlStatementContext, final MetaDataContexts metaDataContexts, final Collection<FirebirdReturnColumnPacket> describeColumns,
219                                     final Collection<FirebirdSQLInfoPacketType> requestedItems) {
220         String databaseName = connectionSession.getCurrentDatabaseName();
221         String schemaName = new DatabaseTypeRegistry(sqlStatementContext.getSqlStatement().getDatabaseType()).getDefaultSchemaName(databaseName);
222         ShardingSphereSchema schema = metaDataContexts.getMetaData().getDatabase(databaseName).getSchema(schemaName);
223         Collection<Projection> projections = getProjections(sqlStatementContext, schema);
224         int columnCount = 0;
225         for (Projection each : projections) {
226             if (each instanceof ColumnProjection) {
227                 String tableName = ((ColumnProjection) each).getOriginalTable().getValue();
228                 ShardingSphereTable table = schema.getTable(tableName.isEmpty() ? getTableNames(sqlStatementContext).iterator().next() : tableName);
229                 ShardingSphereColumn column = table.getColumn(((ColumnProjection) each).getOriginalColumn().getValue());
230                 processColumn(describeColumns, requestedItems, table, column, ((ColumnProjection) each).getOwner(), each.getAlias(), ++columnCount);
231             } else if (each instanceof ExpressionProjection) {
232                 processExpressionProjection((ExpressionProjection) each, describeColumns, requestedItems, ++columnCount);
233             } else if (each instanceof AggregationProjection) {
234                 String functionName = ((AggregationProjection) each).getType().name();
235                 processCustomColumn(null, functionName, each.getAlias(), getFunctionType(functionName), describeColumns, requestedItems, ++columnCount);
236             } else if (each instanceof SubqueryProjection) {
237                 SubqueryProjection subquery = (SubqueryProjection) each;
238                 processCustomColumn(null, subquery.getColumnName(), subquery.getAlias(), Types.INTEGER, describeColumns, requestedItems, ++columnCount);
239             }
240         }
241         return columnCount;
242     }
243     
244     private Collection<Projection> getProjections(final SQLStatementContext sqlStatementContext, final ShardingSphereSchema schema) {
245         if (sqlStatementContext instanceof SelectStatementContext) {
246             SelectStatementContext selectStatement = (SelectStatementContext) sqlStatementContext;
247             Collection<Projection> expandProjections = selectStatement.getProjectionsContext().getExpandProjections();
248             Map<Integer, SelectStatementContext> subquery = selectStatement.getSubqueryContexts();
249             if (subquery.isEmpty()) {
250                 return expandProjections;
251             }
252             // workaround for SubqueryTableBindUtils transform Expression and Aggregation projections to Column projection
253             TableSegment from = ((SelectStatement) sqlStatementContext.getSqlStatement()).getFrom().orElse(null);
254             if (!(from instanceof SubqueryTableSegment)) {
255                 return expandProjections;
256             }
257             Collection<Projection> subqueryProjections = new LinkedList<>();
258             for (SelectStatementContext value : subquery.values()) {
259                 subqueryProjections.addAll(value.getProjectionsContext().getExpandProjections());
260             }
261             return subqueryProjections;
262         }
263         if (returningSegment == null) {
264             return Collections.emptyList();
265         }
266         ProjectionEngine projectionEngine = new ProjectionEngine(sqlStatementContext.getSqlStatement().getDatabaseType());
267         Collection<Projection> result = new LinkedList<>();
268         Collection<ProjectionSegment> projections = returningSegment.getProjections().getProjections();
269         for (ProjectionSegment each : projections) {
270             Projection projection = projectionEngine.createProjection(each).orElse(null);
271             if (projection instanceof ShorthandProjection) {
272                 result.addAll(processShorthandProjection(sqlStatementContext, schema, (ShorthandProjection) projection));
273             } else if (!(projection instanceof DerivedProjection) && projection != null) {
274                 result.add(projection);
275             }
276         }
277         return result;
278     }
279     
280     private Collection<Projection> processShorthandProjection(final SQLStatementContext sqlStatementContext, final ShardingSphereSchema schema, final ShorthandProjection projection) {
281         if (!projection.getActualColumns().isEmpty()) {
282             return projection.getActualColumns();
283         }
284         Collection<Projection> result = new LinkedList<>();
285         TablesContext tablesContext = sqlStatementContext.getTablesContext();
286         if (tablesContext != null) {
287             for (String tableName : tablesContext.getTableNames()) {
288                 ShardingSphereTable table = schema.getTable(tableName);
289                 table.getAllColumns().forEach(each -> {
290                     ColumnSegmentBoundInfo info = new ColumnSegmentBoundInfo(null, new IdentifierValue(table.getName()), new IdentifierValue(each.getName()), TableSourceType.PHYSICAL_TABLE);
291                     result.add(new ColumnProjection(null, new IdentifierValue(each.getName()), null, sqlStatementContext.getSqlStatement().getDatabaseType(), null, null, info));
292                 });
293             }
294         }
295         return result;
296     }

298     private int processParameters(final SQLStatementContext sqlStatementContext, final MetaDataContexts metaDataContexts, final Collection<FirebirdReturnColumnPacket> describeColumns,
299                                   final Collection<FirebirdSQLInfoPacketType> requestedItems) {
300         if (sqlStatementContext instanceof InsertStatementContext) {
301             return processInsertStatement((InsertStatementContext) sqlStatementContext, metaDataContexts, describeColumns, requestedItems);
302         }
303         Collection<ColumnSegment> affectedColumns = findAffectedColumns(sqlStatementContext);
304         int parametersCount = sqlStatementContext.getSqlStatement().getParameterMarkers().size();
305         String databaseName = connectionSession.getCurrentDatabaseName();
306         String schemaName = new DatabaseTypeRegistry(sqlStatementContext.getSqlStatement().getDatabaseType()).getDefaultSchemaName(databaseName);
307         int columnCount = 0;
308         for (ColumnSegment columnSegment : affectedColumns) {
309             ShardingSphereTable table = metaDataContexts.getMetaData().getDatabase(databaseName).getSchema(schemaName).getTable(columnSegment.getColumnBoundInfo().getOriginalTable().getValue());
310             ShardingSphereColumn column = table.getColumn(columnSegment.getColumnBoundInfo().getOriginalColumn().getValue());
311             processColumn(describeColumns, requestedItems, table, column, columnSegment.getOwner().map(OwnerSegment::getIdentifier), Optional.ofNullable(columnSegment.getIdentifier()), ++columnCount);
312         }
313         for (int i = 0; i < parametersCount - affectedColumns.size(); i++) {
314             processCustomColumn(null, null, Optional.empty(), 12, describeColumns, requestedItems, ++columnCount);
315         }
316         return columnCount;
317     }

319     private int processInsertStatement(final InsertStatementContext sqlStatementContext, final MetaDataContexts metaDataContexts, final Collection<FirebirdReturnColumnPacket> describeColumns,
320                                        final Collection<FirebirdSQLInfoPacketType> requestedItems) {
321         Collection<String> tableNames = getTableNames(sqlStatementContext);
322         Collection<String> affectedColumns = new LinkedList<>();
323         for (InsertValueContext context : sqlStatementContext.getInsertValueContexts()) {
324             affectedColumns.addAll(processInsertValueContext(sqlStatementContext, context));
325         }
326         String databaseName = connectionSession.getCurrentDatabaseName();
327         String schemaName = new DatabaseTypeRegistry(sqlStatementContext.getSqlStatement().getDatabaseType()).getDefaultSchemaName(databaseName);
328         int columnCount = 0;
329         for (String tableName : tableNames) {
330             ShardingSphereTable table = metaDataContexts.getMetaData().getDatabase(databaseName).getSchema(schemaName).getTable(tableName);
331             for (String columnName : affectedColumns) {
332                 ShardingSphereColumn column = table.getColumn(columnName);
333                 processColumn(describeColumns, requestedItems, table, column, Optional.empty(), Optional.empty(), ++columnCount);
334             }
335         }
336         return columnCount;
337     }
338     
339     private Collection<String> processInsertValueContext(final InsertStatementContext context, final InsertValueContext valueContext) {
340         Collection<String> result = new LinkedList<>();
341         for (int i = 0; i < valueContext.getValueExpressions().size(); i++) {
342             ExpressionSegment expression = valueContext.getValueExpressions().get(i);
343             if (expression instanceof ParameterMarkerExpressionSegment) {
344                 result.add(context.getColumnNames().get(i));
345             }
346         }
347         return result;
348     }
349     
350     private Collection<String> getTableNames(final SQLStatementContext sqlStatementContext) {
351         TablesContext tablesContext = sqlStatementContext.getTablesContext();
352         return null == tablesContext ? Collections.emptyList() : tablesContext.getTableNames();
353     }
354     
355     private Collection<ColumnSegment> findAffectedColumns(final SQLStatementContext sqlStatementContext) {
356         Collection<ColumnSegment> result = new LinkedList<>();
357         if (sqlStatementContext instanceof UpdateStatementContext) {
358             for (ColumnAssignmentSegment segment : ((UpdateStatementContext) sqlStatementContext).getSqlStatement().getSetAssignment().getAssignments()) {
359                 if (segment.getValue() instanceof ParameterMarkerExpressionSegment) {
360                     result.add(segment.getColumns().get(0));
361                 }
362             }
363         }
364         if (sqlStatementContext instanceof WhereContextAvailable) {
365             Collection<WhereSegment> whereSegments = ((WhereContextAvailable) sqlStatementContext).getWhereSegments();
366             for (WhereSegment each : whereSegments) {
367                 processExpr(each.getExpr(), result);
368             }
369         }
370         return result;
371     }
372     
373     private boolean processExpr(final ExpressionSegment expr, final Collection<ColumnSegment> affectedColumns) {
374         if (!(expr instanceof BinaryOperationExpression)) {
375             return expr instanceof ParameterMarkerExpressionSegment;
376         }
377         BinaryOperationExpression binary = (BinaryOperationExpression) expr;
378         processExpr(binary.getLeft(), affectedColumns);
379         boolean rightIsParam = processExpr(binary.getRight(), affectedColumns);
380         if (rightIsParam && binary.getLeft() instanceof ColumnSegment) {
381             affectedColumns.add((ColumnSegment) binary.getLeft());
382         }
383         return false;
384     }
385     
386     private void processExpressionProjection(final ExpressionProjection expr, final Collection<FirebirdReturnColumnPacket> describeColumns, final Collection<FirebirdSQLInfoPacketType> requestedItems,
387                                              final int columnCount) {
388         final ExpressionSegment exprSegment = expr.getExpressionSegment().getExpr();
389         if (exprSegment instanceof FunctionSegment) {
390             String functionName = ((FunctionSegment) exprSegment).getFunctionName();
391             processCustomColumn(null, functionName, expr.getAlias(), getFunctionType(functionName), describeColumns, requestedItems, columnCount);
392         } else if (exprSegment instanceof BinaryOperationExpression) {
393             String operationName = getOperationName(((BinaryOperationExpression) exprSegment).getOperator());
394             int operationType = getOperationType(((BinaryOperationExpression) exprSegment).getOperator());
395             processCustomColumn(null, operationName, expr.getAlias(), operationType, describeColumns, requestedItems, columnCount);
396         } else if (exprSegment instanceof LiteralExpressionSegment) {
397             Object value = ((LiteralExpressionSegment) exprSegment).getLiterals();
398             int type = Types.NULL;
399             if (value instanceof String) {
400                 type = Types.VARCHAR;
401             } else if (value instanceof Integer) {
402                 type = Types.INTEGER;
403             } else if (value instanceof Long) {
404                 type = Types.BIGINT;
405             } else if (value instanceof Number) {
406                 type = Types.NUMERIC;
407             }
408             processCustomColumn(null, null, expr.getAlias(), type, describeColumns, requestedItems, columnCount);
409         }
410     }
411     
412     private int getFunctionType(final String functionName) {
413         // TODO add proper coalesce and other conditional functions return types
414         switch (functionName) {
415             case "substring":
416             case "current_role":
417             case "current_user":
418             case "coalesce":
419                 return Types.VARCHAR;
420             case "gen_id":
421                 return Types.BIGINT;
422             case "current_timestamp":
423                 return Types.TIMESTAMP;
424             default:
425                 return Types.INTEGER;
426         }
427     }
428     
429     private String getOperationName(final String operationType) {
430         switch (operationType) {
431             case "||":
432                 return "CONCATENATION";
433             default:
434                 return operationType;
435         }
436     }
437     
438     private int getOperationType(final String operationType) {
439         switch (operationType) {
440             case "||":
441                 return 12;
442             default:
443                 return 4;
444         }
445     }
446     
447     private void processCustomColumn(final String tableName, final String columnName, final Optional<IdentifierValue> columnAlias, final int dataType,
448                                      final Collection<FirebirdReturnColumnPacket> describeColumns, final Collection<FirebirdSQLInfoPacketType> requestedItems, final int columnCount) {
449         ShardingSphereTable table = new ShardingSphereTable(tableName, Collections.emptyList(), Collections.emptyList(), Collections.emptyList());
450         ShardingSphereColumn column = new ShardingSphereColumn(columnName, dataType, false, false, true, true, false, false);
451         processColumn(describeColumns, requestedItems, table, column, Optional.empty(), columnAlias, columnCount);
452     }
453     
454     private void processColumn(final Collection<FirebirdReturnColumnPacket> describeColumns, final Collection<FirebirdSQLInfoPacketType> requestedItems, final ShardingSphereTable table,
455                                final ShardingSphereColumn column, final Optional<IdentifierValue> tableAlias, final Optional<IdentifierValue> columnAlias, final int idx) {
456         String tableAliasString = tableAlias.map(IdentifierValue::getValue).orElse(table.getName());
457         String columnAliasString = columnAlias.map(IdentifierValue::getValue).orElse(column.getName());
458         String owner = connectionSession.getConnectionContext().getGrantee().getUsername();
459         describeColumns.add(new FirebirdReturnColumnPacket(requestedItems, idx, table, column, tableAliasString, columnAliasString, owner));
460     }
