82      @Test
83      void assertRunBlockingRetriesFiveTimesThenThrowsIngestException() {
84          AtomicInteger runningIndex = new AtomicInteger();
85          boolean[] runningSequence = {true, false, true, false, true, false, true, false, true, false};
86          OpenGaussIncrementalDumper dumper = mock(OpenGaussIncrementalDumper.class, withSettings()
87                  .useConstructor(createDumperContext(false), new WALPosition(new OpenGaussLogSequenceNumber(LogSequenceNumber.valueOf(1L))), new MemoryPipelineChannel(10, records -> {
88                  }), mock(PipelineTableMetaDataLoader.class))
89                  .defaultAnswer(invocation -> {
90                      if ("isRunning".equals(invocation.getMethod().getName())) {
91                          int index = runningIndex.getAndIncrement();
92                          return index < runningSequence.length && runningSequence[index];
93                      }
94                      return invocation.callRealMethod();
95                  }));
96          AtomicInteger attempts = new AtomicInteger();
97          try (MockedStatic<DriverManager> driverManagerMock = mockStatic(DriverManager.class)) {
98              driverManagerMock.when(() -> DriverManager.getConnection(anyString(), anyString(), anyString())).thenAnswer(invocation -> {
99                  attempts.incrementAndGet();
100                 throw new SQLException("failed");
101             });
102             assertThrows(IngestException.class, dumper::start);
103         }
104         assertThat(attempts.get(), is(5));
105     }

136     void assertDumpDecodeWithTXWhenMajorVersionAtLeastThree() throws ReflectiveOperationException, SQLException {
137         IngestPosition position = new WALPosition(new OpenGaussLogSequenceNumber(LogSequenceNumber.valueOf(1L)));
138         MemoryPipelineChannel channel = new MemoryPipelineChannel(10, records -> {
139         });
140         OpenGaussIncrementalDumper dumper = new OpenGaussIncrementalDumper(createDumperContext(true), position, channel, mock());
141         OpenGaussLogicalReplication logicalReplication = mock(OpenGaussLogicalReplication.class);
142         Plugins.getMemberAccessor().set(OpenGaussIncrementalDumper.class.getDeclaredField("logicalReplication"), dumper, logicalReplication);
143         PgConnection pgConnection = mock(PgConnection.class);
144         PGReplicationStream stream = mock(PGReplicationStream.class);
145         IncrementalDumperContext dumperContext = getDumperContext(dumper);
146         when(logicalReplication.createConnection((StandardPipelineDataSourceConfiguration) dumperContext.getCommonContext().getDataSourceConfig())).thenReturn(pgConnection);
147         when(pgConnection.unwrap(PgConnection.class)).thenReturn(pgConnection);
148         when(stream.getLastReceiveLSN()).thenReturn(LogSequenceNumber.valueOf(200L));
149         when(stream.readPending()).thenReturn(null, ByteBuffer.wrap("1".getBytes()), ByteBuffer.wrap("2".getBytes()),
150                 ByteBuffer.wrap("3".getBytes()), ByteBuffer.wrap("4".getBytes()), ByteBuffer.wrap("5".getBytes()));
151         Plugins.getMemberAccessor().set(OpenGaussIncrementalDumper.class.getDeclaredField("walEventConverter"), dumper, mock(WALEventConverter.class));
152         AtomicReference<Boolean> running = getRunningState(dumper);
153         try (
154                 MockedStatic<DriverManager> driverManagerMock = mockStatic(DriverManager.class);
155                 MockedStatic<PostgreSQLSlotNameGenerator> slotNameMock = mockStatic(PostgreSQLSlotNameGenerator.class);
156                 MockedConstruction<MppdbDecodingPlugin> ignoredConstruction = mockConstruction(MppdbDecodingPlugin.class, (mockedPlugin, context) -> {
157                     AtomicInteger counter = new AtomicInteger();
158                     when(mockedPlugin.decode(any(ByteBuffer.class), any(OpenGaussLogSequenceNumber.class)))
159                             .thenAnswer(invocation -> createWALEventWithDecodeSequence(running, counter.incrementAndGet()));
160                 })) {
161             slotNameMock.when(() -> PostgreSQLSlotNameGenerator.getUniqueSlotName(pgConnection, dumperContext.getJobId())).thenReturn("slot-1");
162             when(logicalReplication.createReplicationStream(eq(pgConnection), any(), eq("slot-1"), eq(3))).thenReturn(stream);
163             mockVersionQuery(driverManagerMock, "(openGauss 3.1.0 build )");
164             dumper.start();
165         }
166         assertThat(channel.fetch(10, 0L).size(), is(3));
167         AtomicReference<WALPosition> walPosition = (AtomicReference<WALPosition>) Plugins.getMemberAccessor().get(OpenGaussIncrementalDumper.class.getDeclaredField("walPosition"), dumper);
168         assertThat(walPosition.get().getLogSequenceNumber().asString(), is(LogSequenceNumber.valueOf(5L).asString()));
169         verify(stream).close();
170     }

172     private AbstractWALEvent createWALEventWithDecodeSequence(final AtomicReference<Boolean> running, final int index) {
173         AbstractWALEvent result;
174         switch (index) {
175             case 1:
176                 result = new BeginTXEvent(1L, 10L);
177                 break;
178             case 2:
179                 WriteRowEvent firstRowEvent = new WriteRowEvent();
180                 firstRowEvent.setTableName("t_order");
181                 firstRowEvent.setSchemaName("public");
182                 result = firstRowEvent;
183                 break;
184             case 3:
185                 result = new BeginTXEvent(2L, 20L);
186                 break;
187             case 4:
188                 WriteRowEvent secondRowEvent = new WriteRowEvent();
189                 secondRowEvent.setTableName("t_order");
190                 secondRowEvent.setSchemaName("public");
191                 result = secondRowEvent;
192                 break;
193             default:
194                 result = new CommitTXEvent(3L, 30L);
195                 running.set(false);
196         }
197         result.setLogSequenceNumber(new OpenGaussLogSequenceNumber(LogSequenceNumber.valueOf(index)));
198         return result;
199     }

203         IngestPosition position = new WALPosition(new OpenGaussLogSequenceNumber(LogSequenceNumber.valueOf(1L)));
204         MemoryPipelineChannel channel = new MemoryPipelineChannel(10, records -> {
205         });
206         OpenGaussIncrementalDumper dumper = new OpenGaussIncrementalDumper(createDumperContext(true), position, channel, mock());
207         OpenGaussLogicalReplication logicalReplication = mock(OpenGaussLogicalReplication.class);

210         PGReplicationStream stream = mock(PGReplicationStream.class);
211         IncrementalDumperContext dumperContext = getDumperContext(dumper);
212         when(logicalReplication.createConnection((StandardPipelineDataSourceConfiguration) dumperContext.getCommonContext().getDataSourceConfig())).thenReturn(pgConnection);
213         when(pgConnection.unwrap(PgConnection.class)).thenReturn(pgConnection);

216         doThrow(new SQLException("close error")).when(stream).close();
217         WALEventConverter walEventConverter = mock(WALEventConverter.class);
218         Plugins.getMemberAccessor().set(OpenGaussIncrementalDumper.class.getDeclaredField("walEventConverter"), dumper, walEventConverter);
219         ArgumentCaptor<AbstractWALEvent> eventCaptor = ArgumentCaptor.forClass(AbstractWALEvent.class);
220         AtomicReference<Boolean> running = getRunningState(dumper);
221         try (
222                 MockedStatic<DriverManager> driverManagerMock = mockStatic(DriverManager.class);
223                 MockedStatic<PostgreSQLSlotNameGenerator> slotNameMock = mockStatic(PostgreSQLSlotNameGenerator.class);
224                 MockedConstruction<MppdbDecodingPlugin> ignoredConstruction = mockConstruction(MppdbDecodingPlugin.class, (mockedPlugin, context) -> {
225                     AtomicInteger counter = new AtomicInteger();
226                     when(mockedPlugin.decode(any(ByteBuffer.class), any(OpenGaussLogSequenceNumber.class)))

228                 })) {

231             mockVersionQuery(driverManagerMock, "(openGauss 2.0 build )");
232             dumper.start();
233         }
234         verify(walEventConverter, times(2)).convert(eventCaptor.capture());
235         AbstractRowEvent rowEvent = (AbstractRowEvent) eventCaptor.getAllValues().get(0);
236         assertThat(rowEvent.getCsn(), is(40L));
237     }

307         poolProps.put("url", "jdbc:postgresql://localhost:5432/test_db");
308         poolProps.put("username", "root");
309         poolProps.put("password", "root");
310         DumperCommonContext commonContext = new DumperCommonContext(null, new StandardPipelineDataSourceConfiguration(poolProps),

326         Connection connection = mock(Connection.class);

328         ResultSet resultSet = mock(ResultSet.class);
