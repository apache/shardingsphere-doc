76      @Test
77      void assertExtractSubquerySegmentsInWhere() {
78          SelectStatement subquerySelectStatement = mockSelectStatement();
79          when(subquerySelectStatement.getFrom()).thenReturn(Optional.of(new SimpleTableSegment(new TableNameSegment(73, 99, new IdentifierValue("t_order")))));
80          ProjectionsSegment subqueryProjections = new ProjectionsSegment(59, 66);
81          when(subquerySelectStatement.getProjections()).thenReturn(subqueryProjections);
82          subqueryProjections.getProjections().add(new ColumnProjectionSegment(new ColumnSegment(59, 66, new IdentifierValue("order_id"))));
83          ColumnSegment subqueryWhereLeft = new ColumnSegment(87, 92, new IdentifierValue("status"));
84          LiteralExpressionSegment subqueryWhereRight = new LiteralExpressionSegment(96, 99, "OK");
85          WhereSegment subqueryWhereSegment = new WhereSegment(81, 99, new BinaryOperationExpression(87, 99, subqueryWhereLeft, subqueryWhereRight, "=", "status = 'OK'"));
86          when(subquerySelectStatement.getWhere()).thenReturn(Optional.of(subqueryWhereSegment));
87          SelectStatement selectStatement = mockSelectStatement();
88          when(selectStatement.getFrom()).thenReturn(Optional.of(new SimpleTableSegment(new TableNameSegment(21, 32, new IdentifierValue("t_order_item")))));
89          when(selectStatement.getProjections()).thenReturn(new ProjectionsSegment(7, 14));
90          selectStatement.getProjections().getProjections().add(new ColumnProjectionSegment(new ColumnSegment(7, 14, new IdentifierValue("order_id"))));
91          ColumnSegment left = new ColumnSegment(40, 47, new IdentifierValue("order_id"));
92          SubqueryExpressionSegment right = new SubqueryExpressionSegment(new SubquerySegment(51, 100, subquerySelectStatement, ""));
93          WhereSegment whereSegment = new WhereSegment(34, 100, new BinaryOperationExpression(40, 100, left, right, "=", "order_id = (SELECT order_id FROM t_order WHERE status = 'OK')"));
94          when(selectStatement.getWhere()).thenReturn(Optional.of(whereSegment));
95          Collection<SubquerySegment> actual = SubqueryExtractor.extractSubquerySegments(selectStatement, true);
96          assertThat(actual.size(), is(1));
97          assertThat(actual.iterator().next(), is(right.getSubquery()));
98          assertThat(SubqueryExtractor.extractSubquerySegments(selectStatement, false).size(), is(1));
99      }

232                                                      final List<SubquerySegment> expected, final List<SubqueryType> expectedTypes) {
233         Collection<SubquerySegment> actual = SubqueryExtractor.extractSubquerySegments(selectStatement, needRecursive);
234         assertThat(actual.size(), is(expected.size()));

236         for (int index = 0; index < expected.size(); index++) {
237             SubquerySegment actualSubquery = actualIterator.next();
238             assertThat(actualSubquery, sameInstance(expected.get(index)));
239             assertThat(actualSubquery.getSelect().getSubqueryType().orElse(null), is(expectedTypes.get(index)));
240         }
241     }

243     private static Stream<Arguments> provideExpressionSubqueries() {
244         SubquerySegment first = new SubquerySegment(1, 9, SelectStatement.builder().projections(new ProjectionsSegment(0, 0)).build(), "");
245         SubquerySegment second = new SubquerySegment(11, 19, SelectStatement.builder().projections(new ProjectionsSegment(0, 0)).build(), "");
246         CommonExpressionSegment inner = new CommonExpressionSegment(10, 20, "nested", Collections.singleton(new SubqueryExpressionSegment(second)));
247         CommonExpressionSegment siblings = new CommonExpressionSegment(0, 21, "indexed", Arrays.asList(new SubqueryExpressionSegment(first), inner));
248         SubquerySegment predicate = new SubquerySegment(1, 9, SelectStatement.builder().projections(new ProjectionsSegment(0, 0)).build(), "");
249         SelectStatement predicateStatement = SelectStatement.builder().projections(new ProjectionsSegment(0, 0))
250                 .where(new WhereSegment(0, 10, new CommonExpressionSegment(0, 10, "indexed", Collections.singleton(new SubqueryExpressionSegment(predicate))))).build();
251         SubquerySegment directNested = new SubquerySegment(2, 8, SelectStatement.builder().projections(new ProjectionsSegment(0, 0)).build(), "");
252         SubquerySegment directOuter = createSubqueryWithNestedPredicate(directNested);
253         SubquerySegment recursiveNested = new SubquerySegment(2, 8, SelectStatement.builder().projections(new ProjectionsSegment(0, 0)).build(), "");
254         SubquerySegment recursiveOuter = createSubqueryWithNestedPredicate(recursiveNested);
255         SubquerySegment firstRow = new SubquerySegment(1, 9, SelectStatement.builder().projections(new ProjectionsSegment(0, 0)).build(), "");
256         SubquerySegment secondRow = new SubquerySegment(11, 19, SelectStatement.builder().projections(new ProjectionsSegment(0, 0)).build(), "");
257         RowExpression innerRow = new RowExpression(10, 20, "ROW");
258         innerRow.getItems().add(new SubqueryExpressionSegment(secondRow));
259         RowExpression row = new RowExpression(0, 21, "ROW");
260         row.getItems().addAll(Arrays.asList(new SubqueryExpressionSegment(firstRow), innerRow));
261         SubquerySegment unaryNested = new SubquerySegment(2, 8, SelectStatement.builder().projections(new ProjectionsSegment(0, 0)).build(), "");
262         SubquerySegment unaryOuter = createSubqueryWithNestedPredicate(unaryNested);
263         SelectStatement unaryPredicate = SelectStatement.builder().projections(new ProjectionsSegment(0, 0))
264                 .where(new WhereSegment(0, 10, new UnaryOperationExpression(0, 10, new SubqueryExpressionSegment(unaryOuter), "~", "~(SELECT)"))).build();
265         SubquerySegment collateNested = new SubquerySegment(2, 8, SelectStatement.builder().projections(new ProjectionsSegment(0, 0)).build(), "");
266         SubquerySegment collateOuter = createSubqueryWithNestedPredicate(collateNested);
267         CollateExpression collate = new CollateExpression(0, 15, new LiteralExpressionSegment(12, 14, "C"), new SubqueryExpressionSegment(collateOuter));
268         SubquerySegment directNotNested = new SubquerySegment(2, 8, SelectStatement.builder().projections(new ProjectionsSegment(0, 0)).build(), "");
269         SubquerySegment directNotOuter = createSubqueryWithNestedPredicate(directNotNested);
270         SelectStatement directNotPredicate = SelectStatement.builder().projections(new ProjectionsSegment(0, 0))
271                 .where(new WhereSegment(0, 10, new NotExpression(0, 10, new SubqueryExpressionSegment(directNotOuter), true))).build();
272         SubquerySegment recursiveNotNested = new SubquerySegment(2, 8, SelectStatement.builder().projections(new ProjectionsSegment(0, 0)).build(), "");
273         SubquerySegment recursiveNotOuter = createSubqueryWithNestedPredicate(recursiveNotNested);
274         SelectStatement recursiveNotPredicate = SelectStatement.builder().projections(new ProjectionsSegment(0, 0))
275                 .where(new WhereSegment(0, 10, new NotExpression(0, 10, new SubqueryExpressionSegment(recursiveNotOuter), true))).build();
276         SubquerySegment nestedNotSubquery = new SubquerySegment(1, 9, SelectStatement.builder().projections(new ProjectionsSegment(0, 0)).build(), "");
277         NotExpression innerNot = new NotExpression(0, 10, new SubqueryExpressionSegment(nestedNotSubquery), true);
278         SelectStatement nestedNotPredicate = SelectStatement.builder().projections(new ProjectionsSegment(0, 0)).where(new WhereSegment(0, 10, new NotExpression(0, 10, innerNot, true))).build();
279         return Stream.of(
280                 Arguments.of("Empty common expression", createSelectWithExpressionProjection(new CommonExpressionSegment(0, 0, "legacy", Collections.emptyList())), false,
281                         Collections.emptyList(), Collections.emptyList()),
282                 Arguments.of("Nested common projection order", createSelectWithExpressionProjection(siblings), false,
283                         Arrays.asList(first, second), Arrays.asList(SubqueryType.PROJECTION, SubqueryType.PROJECTION)),
284                 Arguments.of("Common predicate type", predicateStatement, false, Collections.singletonList(predicate), Collections.singletonList(SubqueryType.PREDICATE)),
285                 Arguments.of("Common without SELECT recursion", createSelectWithExpressionProjection(
286                         new CommonExpressionSegment(0, 10, "indexed", Collections.singleton(new SubqueryExpressionSegment(directOuter)))), false,
287                         Collections.singletonList(directOuter), Collections.singletonList(SubqueryType.PROJECTION)),
288                 Arguments.of("Common with SELECT recursion", createSelectWithExpressionProjection(
289                         new CommonExpressionSegment(0, 10, "indexed", Collections.singleton(new SubqueryExpressionSegment(recursiveOuter)))), true,
290                         Arrays.asList(recursiveOuter, recursiveNested), Arrays.asList(SubqueryType.PROJECTION, SubqueryType.PREDICATE)),
291                 Arguments.of("Nested row projection order", createSelectWithExpressionProjection(row), false,
292                         Arrays.asList(firstRow, secondRow), Arrays.asList(SubqueryType.PROJECTION, SubqueryType.PROJECTION)),
293                 Arguments.of("Unary predicate without SELECT recursion", unaryPredicate, false, Collections.singletonList(unaryOuter), Collections.singletonList(SubqueryType.PREDICATE)),
294                 Arguments.of("Collate with SELECT recursion", createSelectWithExpressionProjection(collate), true,
295                         Arrays.asList(collateOuter, collateNested), Arrays.asList(SubqueryType.PROJECTION, SubqueryType.PREDICATE)),
296                 Arguments.of("Not predicate without SELECT recursion", directNotPredicate, false, Collections.singletonList(directNotOuter), Collections.singletonList(SubqueryType.PREDICATE)),
297                 Arguments.of("Not predicate with SELECT recursion", recursiveNotPredicate, true,
298                         Arrays.asList(recursiveNotOuter, recursiveNotNested), Arrays.asList(SubqueryType.PREDICATE, SubqueryType.PREDICATE)),
299                 Arguments.of("Nested not predicate", nestedNotPredicate, false, Collections.singletonList(nestedNotSubquery), Collections.singletonList(SubqueryType.PREDICATE)));
300     }

302     private static SubquerySegment createSubqueryWithNestedPredicate(final SubquerySegment nested) {
303         SelectStatement selectStatement = SelectStatement.builder().projections(new ProjectionsSegment(0, 0)).where(new WhereSegment(0, 10, new SubqueryExpressionSegment(nested))).build();
304         return new SubquerySegment(0, 10, selectStatement, "");
305     }

307     private static SelectStatement createSelectWithExpressionProjection(final ExpressionSegment expression) {
308         ProjectionsSegment projections = new ProjectionsSegment(expression.getStartIndex(), expression.getStopIndex());
309         projections.getProjections().add(new ExpressionProjectionSegment(expression.getStartIndex(), expression.getStopIndex(), expression.getText(), expression));
310         return SelectStatement.builder().projections(projections).build();
311     }
