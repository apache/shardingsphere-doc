35  /**
36   * Readwrite-splitting status workflow validation service.
37   */
38  public final class ReadwriteSplittingStatusWorkflowValidationService implements MCPWorkflowRuntimeHandler {
39      
40      private final WorkflowValidationSupport validationSupport = new WorkflowValidationSupport();
41      
42      private final ReadwriteSplittingInspectionService inspectionService = new ReadwriteSplittingInspectionService();
43      
44      private final ReadwriteSplittingStatusDistSQLPlanningService distSQLPlanningService = new ReadwriteSplittingStatusDistSQLPlanningService();
45      
46      @Override
47      public ValidationReport validate(final WorkflowContextSnapshot snapshot, final MCPFeatureQueryFacade queryFacade) {
48          ValidationReport result = new ValidationReport();
49          ReadwriteSplittingStatusWorkflowRequest request = (ReadwriteSplittingStatusWorkflowRequest) snapshot.getRequest();
50          queryFacade.checkDatabaseCapability(request.getDatabase());
51          List<Map<String, Object>> statuses = inspectionService.queryRuleStatus(queryFacade, request.getDatabase(), request.getRuleName());
52          result.setRuleValidation(validateStatus(request, statuses, result, queryFacade));
53          result.setOverallStatus(validationSupport.resolveOverallStatus(result.getRuleValidation()));
54          return result;
55      }
56      
57      private ValidationSection validateStatus(final ReadwriteSplittingStatusWorkflowRequest request, final List<Map<String, Object>> statuses,
58                                               final ValidationReport validationReport, final MCPFeatureQueryFacade queryFacade) {
59          String expectedStatus = "ENABLE".equals(distSQLPlanningService.resolveStatusOperation(request)) ? "ENABLED" : "DISABLED";
60          List<Map<String, Object>> targetStatuses = statuses.stream().filter(each -> matchesStatusTarget(request, each, queryFacade)).toList();
61          if (1 != targetStatuses.size() || !expectedStatus.equalsIgnoreCase(WorkflowRuleValueUtils.getRuleValue(targetStatuses.getFirst(), "status"))) {
62              validationReport.getMismatches().add(validationSupport.createMismatch(WorkflowIssueCode.RULE_STATE_MISMATCH, "status", expectedStatus, request.getStorageUnit(),
63                      "Readwrite-splitting storage-unit status does not match the planned artifact.",
64                      "Apply the readwrite-splitting status DistSQL again or inspect current status."));
65              return new ValidationSection(WorkflowLifecycle.STATUS_FAILED, statuses, "Readwrite-splitting status state does not match the planned DistSQL artifact.");
66          }
67          return new ValidationSection(WorkflowLifecycle.STATUS_PASSED, statuses, "Readwrite-splitting status state matches the planned DistSQL artifact.");
68      }
69      
70      private boolean matchesStatusTarget(final ReadwriteSplittingStatusWorkflowRequest request, final Map<String, Object> status, final MCPFeatureQueryFacade queryFacade) {
71          return queryFacade.isSameIdentifier(request.getDatabase(), IdentifierScope.TABLE, request.getRuleName(), WorkflowRuleValueUtils.getRuleValue(status, "name"))
72                  && queryFacade.isSameIdentifier(request.getDatabase(), IdentifierScope.TABLE, request.getStorageUnit(), WorkflowRuleValueUtils.getRuleValue(status, "storage_unit"));
73      }
74  }
