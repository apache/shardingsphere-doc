18  package org.apache.shardingsphere.test.natived.commons.util;
19  
20  import org.apache.shardingsphere.database.connector.core.DefaultDatabase;
21  import org.apache.shardingsphere.driver.jdbc.core.connection.ShardingSphereConnection;
22  import org.apache.shardingsphere.infra.metadata.database.resource.unit.StorageUnit;
23  import org.apache.shardingsphere.mode.manager.ContextManager;
24  
25  import javax.sql.DataSource;
26  import java.sql.Connection;
27  import java.sql.SQLException;
28  
29  /**
30   * When creating {@link com.zaxxer.hikari.HikariDataSource} through {@link org.apache.shardingsphere.driver.ShardingSphereDriver}
31   * or starting ShardingSphere Proxy through {@link org.apache.shardingsphere.proxy.Bootstrap},
32   * the internal real data source will not be closed directly.
33   * This causes the hooks for closing the data source of third-party dependencies such as Seata Client to not take effect.
34   * Refer to the changes in <a href="https://github.com/apache/incubator-seata/pull/7044">apache/incubator-seata#7044</a>.
35   *
36   * @see org.apache.shardingsphere.driver.jdbc.core.datasource.ShardingSphereDataSource#close()
37   */
38  public class ResourceUtils {
39      
40      /**
41       * Close JDBC dataSource.
42       *
43       * @param dataSource Usually {@link com.zaxxer.hikari.HikariDataSource}
44       * @throws SQLException SQL exception
45       */
46      public static void closeJdbcDataSource(final DataSource dataSource) throws SQLException {
47          try (Connection connection = dataSource.getConnection()) {
48              ContextManager contextManager = connection.unwrap(ShardingSphereConnection.class).getContextManager();
49              close(dataSource);
50              contextManager.getStorageUnits(DefaultDatabase.LOGIC_NAME).values().stream().map(StorageUnit::getDataSource).forEach(ResourceUtils::close);
51              contextManager.close();
52          }
53      }
54      
55      /**
56       * Close JDBC dataSource.
57       *
58       * @param dataSource        Usually {@link com.zaxxer.hikari.HikariDataSource}
59       * @param logicDataBaseName The logical database name used in the shardingsphere metadata, if not set,
60       *                          it is {@link DefaultDatabase#LOGIC_NAME}
61       * @throws SQLException SQL exception
62       */
63      public static void closeJdbcDataSource(final DataSource dataSource, final String logicDataBaseName) throws SQLException {
64          try (Connection connection = dataSource.getConnection()) {
65              ContextManager contextManager = connection.unwrap(ShardingSphereConnection.class).getContextManager();
66              close(dataSource);
67              contextManager.getStorageUnits(logicDataBaseName).values().stream().map(StorageUnit::getDataSource).forEach(ResourceUtils::close);
68              contextManager.close();
69          }
70      }
71      
72      private static void close(final DataSource dataSource) {
73          if (dataSource instanceof AutoCloseable) {
74              try {
75                  ((AutoCloseable) dataSource).close();
76                  // CHECKSTYLE:OFF
77              } catch (final Exception ex) {
78                  // CHECKSTYLE:ON
79                  throw new RuntimeException(ex);
80              }
81          }
82      }
83  }
