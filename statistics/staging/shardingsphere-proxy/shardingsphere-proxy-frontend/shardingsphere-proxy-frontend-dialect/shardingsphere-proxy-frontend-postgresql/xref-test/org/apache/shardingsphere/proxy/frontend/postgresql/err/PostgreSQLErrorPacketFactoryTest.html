18  package org.apache.shardingsphere.proxy.frontend.postgresql.err;
19  
20  import org.apache.shardingsphere.database.exception.core.exception.transaction.TableModifyInTransactionException;
21  import org.apache.shardingsphere.database.exception.postgresql.exception.PostgreSQLException;
22  import org.apache.shardingsphere.database.exception.postgresql.vendor.PostgreSQLVendorError;
23  import org.apache.shardingsphere.database.protocol.postgresql.constant.PostgreSQLMessageSeverityLevel;
24  import org.apache.shardingsphere.database.protocol.postgresql.packet.generic.PostgreSQLErrorResponsePacket;
25  import org.junit.jupiter.api.Test;
26  import org.mockito.internal.configuration.plugins.Plugins;
27  import org.postgresql.util.PSQLException;
28  import org.postgresql.util.PSQLState;
29  import org.postgresql.util.ServerErrorMessage;
30  
31  import java.util.Map;
32  
33  import static org.hamcrest.MatcherAssert.assertThat;
34  import static org.hamcrest.Matchers.is;
35  import static org.mockito.Mockito.mock;
36  import static org.mockito.Mockito.when;
37  
38  class PostgreSQLErrorPacketFactoryTest {
39      
40      @Test
41      void assertPSQLExceptionWithServerErrorMessageNotNull() throws ReflectiveOperationException {
42          ServerErrorMessage serverErrorMessage = mock(ServerErrorMessage.class);
43          when(serverErrorMessage.getSeverity()).thenReturn(PostgreSQLMessageSeverityLevel.FATAL);
44          when(serverErrorMessage.getSQLState()).thenReturn("sqlState");
45          when(serverErrorMessage.getMessage()).thenReturn("message");
46          when(serverErrorMessage.getPosition()).thenReturn(1);
47          PostgreSQLErrorResponsePacket actual = PostgreSQLErrorPacketFactory.newInstance(new PSQLException(serverErrorMessage));
48          Map<Character, String> fields = getFields(actual);
49          assertThat(fields.get(PostgreSQLErrorResponsePacket.FIELD_TYPE_SEVERITY), is(PostgreSQLMessageSeverityLevel.FATAL));
50          assertThat(fields.get(PostgreSQLErrorResponsePacket.FIELD_TYPE_CODE), is("sqlState"));
51          assertThat(fields.get(PostgreSQLErrorResponsePacket.FIELD_TYPE_MESSAGE), is("message"));
52          assertThat(fields.get(PostgreSQLErrorResponsePacket.FIELD_TYPE_POSITION), is("1"));
53      }
54      
55      @Test
56      void assertPSQLExceptionWithServerErrorMessageIsNull() throws ReflectiveOperationException {
57          PostgreSQLErrorResponsePacket actual = PostgreSQLErrorPacketFactory.newInstance(new PSQLException("psqlEx", PSQLState.UNEXPECTED_ERROR, new Exception("test")));
58          Map<Character, String> fields = getFields(actual);
59          assertThat(fields.get(PostgreSQLErrorResponsePacket.FIELD_TYPE_CODE), is(PSQLState.UNEXPECTED_ERROR.getState()));
60          assertThat(fields.get(PostgreSQLErrorResponsePacket.FIELD_TYPE_MESSAGE), is("psqlEx"));
61      }
62      
63      @Test
64      void assertRuntimeException() throws ReflectiveOperationException {
65          PostgreSQLErrorResponsePacket actual = PostgreSQLErrorPacketFactory.newInstance(new RuntimeException("No reason"));
66          Map<Character, String> fields = getFields(actual);
67          assertThat(fields.get(PostgreSQLErrorResponsePacket.FIELD_TYPE_MESSAGE), is("Unknown exception." + System.lineSeparator() + "More details: java.lang.RuntimeException: No reason"));
68      }
69      
70      @Test
71      void assertTableModifyInTransactionException() throws ReflectiveOperationException {
72          PostgreSQLErrorResponsePacket actual = PostgreSQLErrorPacketFactory.newInstance(new TableModifyInTransactionException("foo_table"));
73          Map<Character, String> fields = getFields(actual);
74          assertThat(fields.get(PostgreSQLErrorResponsePacket.FIELD_TYPE_SEVERITY), is(PostgreSQLMessageSeverityLevel.ERROR));
75          assertThat(fields.get(PostgreSQLErrorResponsePacket.FIELD_TYPE_CODE), is(PostgreSQLVendorError.METADATA_CHANGING_DDL_IN_TRANSACTION_NOT_SUPPORTED.getSqlState().getValue()));
76          assertThat(fields.get(PostgreSQLErrorResponsePacket.FIELD_TYPE_MESSAGE), is("DDL statements that modify metadata are not supported in transactions by ShardingSphere-Proxy."));
77      }
78      
79      @Test
80      void assertPostgreSQLExceptionWithServerErrorMessage() throws ReflectiveOperationException {
81          PostgreSQLException.ServerErrorMessage serverErrorMessage = new PostgreSQLException.ServerErrorMessage(
82                  PostgreSQLMessageSeverityLevel.ERROR, PostgreSQLVendorError.INVALID_PARAMETER_VALUE, "param", "value");
83          PostgreSQLErrorResponsePacket actual = PostgreSQLErrorPacketFactory.newInstance(new PostgreSQLException(serverErrorMessage));
84          Map<Character, String> fields = getFields(actual);
85          assertThat(fields.get(PostgreSQLErrorResponsePacket.FIELD_TYPE_SEVERITY), is(PostgreSQLMessageSeverityLevel.ERROR));
86          assertThat(fields.get(PostgreSQLErrorResponsePacket.FIELD_TYPE_CODE), is(PostgreSQLVendorError.INVALID_PARAMETER_VALUE.getSqlState().getValue()));
87          assertThat(fields.get(PostgreSQLErrorResponsePacket.FIELD_TYPE_MESSAGE), is("invalid value for parameter \"param\": \"value\""));
88      }
89      
90      @Test
91      void assertSQLExceptionWithoutMessageAndSQLState() throws ReflectiveOperationException {
92          PostgreSQLException cause = new PostgreSQLException(null, null);
93          PostgreSQLErrorResponsePacket actual = PostgreSQLErrorPacketFactory.newInstance(cause);
94          Map<Character, String> fields = getFields(actual);
95          assertThat(fields.get(PostgreSQLErrorResponsePacket.FIELD_TYPE_CODE), is(PostgreSQLVendorError.SYSTEM_ERROR.getSqlState().getValue()));
96          assertThat(fields.get(PostgreSQLErrorResponsePacket.FIELD_TYPE_MESSAGE), is(cause.toString()));
97      }
98      
99      @SuppressWarnings("unchecked")
100     private Map<Character, String> getFields(final PostgreSQLErrorResponsePacket packet) throws ReflectiveOperationException {
101         return (Map<Character, String>) Plugins.getMemberAccessor().get(PostgreSQLErrorResponsePacket.class.getDeclaredField("fields"), packet);
102     }
103 }
