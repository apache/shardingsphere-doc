31  /**
32   * MCP guidance payload builder.
33   */
34  @NoArgsConstructor(access = AccessLevel.PRIVATE)
35  final class MCPGuidancePayloadBuilder {
36      
37      private static final String GUIDANCE_RESOURCE_URI = "shardingsphere://guidance";
38      
39      static Map<String, Object> build() {
40          Map<String, Object> result = new LinkedHashMap<>(6, 1F);
41          result.put("response_mode", MCPResponseMode.GUIDANCE);
42          result.put("discovery", createDiscovery());
43          result.put("model_contract", createModelContract());
44          result.put("next_action_contract", createNextActionContract());
45          result.put("common_flows", createCommonFlows());
46          result.put("security_hints", createSecurityHints());
47          return result;
48      }
49      
50      static Map<String, Object> createDiscovery() {
51          Map<String, Object> result = new LinkedHashMap<>(3, 1F);
52          result.put("official_discovery_methods", createOfficialDiscoveryMethods());
53          result.put("argument_completion_method", "completion/complete");
54          result.put("guidance_resource_role", GUIDANCE_RESOURCE_URI + " complements MCP list methods with ShardingSphere domain guidance, workflow guidance, and side-effect notes.");
55          return result;
56      }
57      
58      static Map<String, Object> createModelContract() {
59          Map<String, Object> result = new LinkedHashMap<>(12, 1F);
60          result.put("metadata_first_resource", "shardingsphere://databases");
61          result.put("preflight_rule", "Use database_gateway_validate_runtime_database with a configured database name before onboarding or troubleshooting runtime connectivity.");
62          Map<String, Object> sqlToolSelection = new LinkedHashMap<>(3, 1F);
63          sqlToolSelection.put("read_only", "Use database_gateway_execute_query for one parser-approved SELECT statement.");
64          sqlToolSelection.put("explain", "Use database_gateway_execute_explain_query with the original SELECT and a model-generated database-native EXPLAIN SQL; do not use EXPLAIN ANALYZE.");
65          sqlToolSelection.put("side_effecting", "Use database_gateway_execute_update with execution_mode=preview before execution.");
66          result.put("sql_tool_selection", sqlToolSelection);
67          result.put("workflow_session_rule", "Reuse the current-session plan_id returned by a planning tool; re-plan when the plan is unavailable.");
68          result.put("side_effect_rule", "Preview before side effects and continue only when the requested side effect is still intended.");
69          result.put("completion_rule", "Use completion/complete for one uncertain argument at a time; when completion reports missing context, follow meta.next_actions before guessing.");
70          result.put("resource_template_rule", "Use resources/templates/list to discover URI variables, then read the nearest concrete resource before filling dependent completion context.");
71          result.put("next_action_rule", "Use canonical next_actions fields: type, tool_name, resource_uri, and arguments.");
72          result.put("detail_resource_rule", "Use resource descriptors, outputSchema, and returned payload keys before assuming detail fields.");
73          result.put("recovery_rule", "When a call fails, follow top-level next_actions before inventing a new call.");
74          result.put("payload_field_rule", "ShardingSphere-owned structured payload fields use snake_case.");
75          result.put("descriptor_field_rule", "Descriptor-derived MCP schema fields keep the protocol or JSON Schema field names required by MCP clients.");
76          return result;
77      }

79      static List<Map<String, Object>> createNextActionContract() {
80          return List.of(
81                  createNextActionContractEntry("resource_read", "Read an MCP resource before choosing another call."),
82                  createNextActionContractEntry("tool_call", "Call another MCP tool with server-suggested arguments."),
83                  createNextActionContractEntry("completion", "Use MCP completion for one argument before retrying."),
84                  createNextActionContractEntry("ask_user", "Ask the user for missing input."),
85                  createNextActionContractEntry("terminal", "Stop the current MCP flow and report the result."));
86      }
87      
88      static List<Map<String, Object>> createCommonFlows() {
89          return List.of(
90                  createCommonFlow("inspect_metadata", List.of("resources/list", "resources/templates/list", "read_resource shardingsphere://databases",
91                          "call_tool database_gateway_search_metadata", "read_resource returned resource.uri"),
92                          "Stop when the requested metadata detail resource is read."),
93                  createCommonFlow("validate_runtime_database", List.of("read_resource shardingsphere://databases", "call_tool database_gateway_validate_runtime_database"),
94                          "Stop after the configured runtime database reports ready or returns structured recovery guidance."),
95                  createCommonFlow("read_only_sql", List.of("read_resource shardingsphere://databases/{database}/capabilities", "call_tool database_gateway_execute_query"),
96                          "Use one SELECT statement and stop after the result is reported."),
97                  createCommonFlow("explain_query", List.of("read_resource shardingsphere://databases/{database}/capabilities",
98                          "generate database-native EXPLAIN SQL without ANALYZE", "call_tool database_gateway_execute_explain_query"),
99                          "Use the original SELECT as sql, the generated EXPLAIN as explain_sql, and stop after the plan rows are reported."),
100                 createCommonFlow("side_effecting_sql", List.of("call_tool database_gateway_execute_update execution_mode=preview",
101                         "call_tool database_gateway_execute_update execution_mode=execute"),
102                         "Execute only after reviewing the previewed SQL and side-effect scope."),
103                 createCommonFlow("complete_uncertain_argument", List.of("resources/templates/list", "call completion/complete for one argument",
104                         "follow completion meta.next_actions when diagnostic is missing_context, prefix_filtered_all_candidates, or no_candidates"),
105                         "Stop when a completion value is selected or the nearest resource proves the argument is unavailable."),
106                 createCommonFlow("workflow_plan_apply_validate", List.of("call_tool descriptor-backed feature planning tool", "call_tool database_gateway_apply_workflow execution_mode=preview",
107                         "call_tool database_gateway_apply_workflow execution_mode=review-then-execute approved_steps=<preview_artifacts.approval_step>",
108                         "call_tool database_gateway_validate_workflow"),
109                         "Reuse the same current-session plan_id and stop after validation succeeds."),
110                 createCommonFlow("recover_from_error", List.of("follow top-level next_actions", "prefer official list discovery methods when surface information is needed",
111                         "read_resource shardingsphere://guidance only when the server suggests the guidance resource", "ask_user only when uncertain"),
112                         "Do not invent replacement calls while structured recovery actions are present."));
113     }

115     static Map<String, Object> createSecurityHints() {
116         Map<String, Object> result = new LinkedHashMap<>(4, 1F);
117         result.put("http_transport", "Streamable HTTP is unauthenticated by default; prefer loopback binding or put remote exposure behind a trusted gateway.");
118         result.put("origin_header", "Present Origin headers must be valid loopback origins for loopback HTTP bindings; missing Origin is accepted.");
119         result.put("stdio_stdout", "STDIO transport must keep MCP protocol frames on stdout and send logs to stderr or files.");
120         result.put("client_safety_policy", createClientSafetyPolicy());
121         return result;
122     }
123     
124     private static Map<String, Object> createClientSafetyPolicy() {
125         Map<String, Object> result = new LinkedHashMap<>(5, 1F);
126         result.put("identity_scope", "mcp_session");
127         result.put("transport_scope",
128                 "HTTP transport can bind trusted session attribution when configured; "
129                         + "the MCP runtime does not provide built-in authentication or authorization. "
130                         + "STDIO inherits the local process boundary.");
131         result.put("runtime_protection", MCPRuntimeProtectionPolicy.createRuntimeProtectionPayload());
132         result.put("abuse_guard", "Every tool call is counted before dispatch, including invalid calls, so runaway model loops stop at the session quota.");
133         result.put("external_model_boundary", "The MCP runtime never calls external model providers; live LLM E2E clients call configured endpoints outside the server.");
134         return result;
135     }

137     private static Map<String, Object> createOfficialDiscoveryMethods() {
138         Map<String, Object> result = new LinkedHashMap<>(4, 1F);
139         result.put("tools", "tools/list");
140         result.put("resources", "resources/list");
141         result.put("resource_templates", "resources/templates/list");
142         result.put("prompts", "prompts/list");
143         return result;
144     }
145     
146     private static Map<String, Object> createNextActionContractEntry(final String actionType, final String description) {
147         Collection<String> requiredFields = MCPModelFacingPayloadContract.getNextActionRequiredFields(actionType);
148         List<String> optionalFields = MCPModelFacingPayloadContract.getNextActionAllowedFields(actionType).stream().filter(each -> !requiredFields.contains(each)).toList();
149         Map<String, Object> result = new LinkedHashMap<>(4, 1F);
150         result.put("type", actionType);
151         result.put("required_fields", requiredFields);
152         result.put("optional_fields", optionalFields);
153         result.put("description", description);
154         return result;
155     }
156     
157     private static Map<String, Object> createCommonFlow(final String flowId, final List<String> steps, final String stopCondition) {
158         Map<String, Object> result = new LinkedHashMap<>(3, 1F);
159         result.put("flow_id", flowId);
160         result.put("steps", steps);
161         result.put("stop_condition", stopCondition);
162         return result;
163     }
