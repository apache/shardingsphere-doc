18  package org.apache.shardingsphere.infra.metadata.database.schema.util;
19  
20  import org.apache.shardingsphere.database.connector.core.type.DatabaseType;
21  import org.apache.shardingsphere.infra.metadata.database.ShardingSphereDatabase;
22  import org.apache.shardingsphere.infra.spi.type.typed.TypedSPILoader;
23  import org.apache.shardingsphere.sql.parser.statement.core.segment.dml.item.ExpressionProjectionSegment;
24  import org.apache.shardingsphere.sql.parser.statement.core.segment.dml.item.ProjectionSegment;
25  import org.junit.jupiter.params.ParameterizedTest;
26  import org.junit.jupiter.params.provider.Arguments;
27  import org.junit.jupiter.params.provider.MethodSource;
28  
29  import java.util.Arrays;
30  import java.util.Collection;
31  import java.util.Collections;
32  import java.util.stream.Stream;
33  
34  import static org.hamcrest.MatcherAssert.assertThat;
35  import static org.hamcrest.Matchers.is;
36  import static org.mockito.Mockito.mock;
37  import static org.mockito.Mockito.when;
38  
39  class SystemSchemaUtilsTest {
40      
41      private static final DatabaseType MYSQL_DATABASE_TYPE = TypedSPILoader.getService(DatabaseType.class, "MySQL");
42      
43      private static final DatabaseType POSTGRESQL_DATABASE_TYPE = TypedSPILoader.getService(DatabaseType.class, "PostgreSQL");
44      
45      private static final DatabaseType OPEN_GAUSS_DATABASE_TYPE = TypedSPILoader.getService(DatabaseType.class, "openGauss");
46      
47      private static final DatabaseType FIXTURE_DATABASE_TYPE = TypedSPILoader.getService(DatabaseType.class, "FIXTURE");
48      
49      @ParameterizedTest(name = "{0}")
50      @MethodSource("containsSystemSchemaArguments")
51      void assertContainsSystemSchema(final String name, final DatabaseType databaseType, final Collection<String> schemaNames,
52                                      final String databaseName, final boolean complete, final boolean expectedContainsSystemSchema) {
53          assertThat(SystemSchemaUtils.containsSystemSchema(databaseType, schemaNames, mockDatabase(databaseName, complete, databaseType)), is(expectedContainsSystemSchema));
54      }
55      
56      @ParameterizedTest(name = "{0}")
57      @MethodSource("isSystemSchemaArguments")
58      void assertIsSystemSchema(final String name, final DatabaseType databaseType, final String databaseName, final boolean complete, final boolean expectedSystemSchema) {
59          assertThat(SystemSchemaUtils.isSystemSchema(mockDatabase(databaseName, complete, databaseType)), is(expectedSystemSchema));
60      }
61      
62      private ShardingSphereDatabase mockDatabase(final String databaseName, final boolean complete, final DatabaseType databaseType) {
63          ShardingSphereDatabase result = mock(ShardingSphereDatabase.class);
64          when(result.getName()).thenReturn(databaseName);
65          when(result.isComplete()).thenReturn(complete);
66          when(result.getProtocolType()).thenReturn(databaseType);
67          return result;
68      }
69      
70      @ParameterizedTest(name = "{0}")
71      @MethodSource("isDriverQuerySystemCatalogArguments")
72      void assertIsDriverQuerySystemCatalog(final String name, final DatabaseType databaseType, final Collection<ProjectionSegment> projections, final boolean expectedDriverQuerySystemCatalog) {
73          assertThat(SystemSchemaUtils.isDriverQuerySystemCatalog(databaseType, projections), is(expectedDriverQuerySystemCatalog));
74      }
75      
76      private static Stream<Arguments> containsSystemSchemaArguments() {
77          return Stream.of(
78                  Arguments.of("returns true when input schema names contain a MySQL system schema", MYSQL_DATABASE_TYPE,
79                          Arrays.asList("information_schema", "mysql"), "information_schema", false, true),
80                  Arguments.of("returns false immediately for a complete MySQL database without default schema", MYSQL_DATABASE_TYPE,
81                          Collections.emptyList(), "information_schema", true, false),
82                  Arguments.of("returns true when incomplete MySQL database name is a system schema", MYSQL_DATABASE_TYPE,
83                          Collections.emptyList(), "information_schema", false, true),
84                  Arguments.of("returns false when schema names do not contain a MySQL system schema", MYSQL_DATABASE_TYPE,
85                          Collections.singletonList("sharding_db"), "sharding_db", false, false),
86                  Arguments.of("returns false when PostgreSQL falls back to database name with default schema", POSTGRESQL_DATABASE_TYPE,
87                          Collections.emptyList(), "pg_catalog", true, false));
88      }
89      
90      private static Stream<Arguments> isSystemSchemaArguments() {
91          return Stream.of(
92                  Arguments.of("returns true for an incomplete MySQL system database", MYSQL_DATABASE_TYPE, "information_schema", false, true),
93                  Arguments.of("returns true for a complete PostgreSQL system schema", POSTGRESQL_DATABASE_TYPE, "pg_catalog", true, true),
94                  Arguments.of("returns false for a complete MySQL database without default schema", MYSQL_DATABASE_TYPE, "information_schema", true, false),
95                  Arguments.of("returns false for a non-system fixture database", FIXTURE_DATABASE_TYPE, "foo_db", false, false));
96      }
97      
98      private static Stream<Arguments> isDriverQuerySystemCatalogArguments() {
99          return Stream.of(
100                 Arguments.of("returns false when multiple projections are present", OPEN_GAUSS_DATABASE_TYPE,
101                         Arrays.asList(new ExpressionProjectionSegment(0, "version()".length(), "version()"), new ExpressionProjectionSegment(0, "current_database()".length(), "current_database()")),
102                         false),
103                 Arguments.of("returns false when the only projection is not an expression", OPEN_GAUSS_DATABASE_TYPE,
104                         Collections.singletonList(mock(ProjectionSegment.class)), false),
105                 Arguments.of("returns false when the database type does not define system catalog expressions", MYSQL_DATABASE_TYPE,
106                         Collections.singletonList(new ExpressionProjectionSegment(0, "version()".length(), "version()")), false),
107                 Arguments.of("returns true when openGauss projection matches a system catalog expression", OPEN_GAUSS_DATABASE_TYPE,
108                         Collections.singletonList(new ExpressionProjectionSegment(0, "version()".length(), "version()")), true));
109     }
