174     @ParameterizedTest(name = "{0}")
175     @MethodSource("getLocalDateTimeArguments")
176     void assertConvertLocalDateTime(final String name, final LocalDateTime dateTime, @Nullable final RelDataType dataType, final String expectedTimestamp, final int expectedPrecision) {
177         SqlTimestampLiteral actual = (SqlTimestampLiteral) LiteralExpressionConverter.convert(new LiteralExpressionSegment(0, 0, dateTime), dataType).orElse(null);
178         assertNotNull(actual);
179         assertThat(actual.getTypeName(), is(SqlTypeName.TIMESTAMP));
180         assertThat(actual.getTimestamp().toString(), is(expectedTimestamp));
181         assertThat(actual.getPrec(), is(expectedPrecision));
182     }
183     
184     private static Stream<Arguments> getLocalDateTimeArguments() {
185         RelDataType dataType = mock(RelDataType.class);
186         when(dataType.getScale()).thenReturn(6);
187         return Stream.of(
188                 Arguments.of("without fraction", LocalDateTime.of(2020, 1, 1, 1, 1, 1), null, "2020-01-01 01:01:01", 0),
189                 Arguments.of("inferred fractional precision", LocalDateTime.of(2020, 1, 1, 1, 1, 1, 123000000), null, "2020-01-01 01:01:01.123", 4),
190                 Arguments.of("data type precision", LocalDateTime.of(2020, 1, 1, 1, 1, 1, 123000000), dataType, "2020-01-01 01:01:01.123", 6));
191     }
