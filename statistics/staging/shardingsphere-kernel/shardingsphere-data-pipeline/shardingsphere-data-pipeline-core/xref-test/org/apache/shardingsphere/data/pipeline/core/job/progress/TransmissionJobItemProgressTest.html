44  class TransmissionJobItemProgressTest {
45      
46      private static final YamlTransmissionJobItemProgressSwapper SWAPPER = new YamlTransmissionJobItemProgressSwapper();
47      
48      @Test
49      void assertInit() {
50          TransmissionJobItemProgress actual = getJobItemProgress(SystemResourceFileUtils.readFile("job-progress.yaml"));
51          assertThat(actual.getStatus(), is(JobStatus.RUNNING));
52          assertThat(actual.getSourceDatabaseType().getType(), is("H2"));
53          assertThat(actual.getInventory().getProgresses().size(), is(4));
54          assertNotNull(actual.getIncremental().getIncrementalTaskProgress());
55      }
56      
57      @Test
58      void assertGetIncrementalPosition() {
59          TransmissionJobItemProgress actual = getJobItemProgress(SystemResourceFileUtils.readFile("job-progress.yaml"));
60          Optional<IngestPosition> position = actual.getIncremental().getIncrementalPosition();
61          assertTrue(position.isPresent());
62          assertThat(position.get(), isA(IngestPlaceholderPosition.class));
63      }
64      
65      @Test
66      void assertGetInventoryPosition() {
67          TransmissionJobItemProgress actual = getJobItemProgress(SystemResourceFileUtils.readFile("job-progress.yaml"));
68          assertThat(actual.getInventory().getInventoryPosition("t_1").get("ds0.t_1#1"), isA(IngestFinishedPosition.class));
69          assertThat(actual.getInventory().getInventoryPosition("t_1").get("ds1.t_1#1"), isA(IngestPlaceholderPosition.class));
70          assertThat(actual.getInventory().getInventoryPosition("t_2").get("ds0.t_2#2"), isA(IngestFinishedPosition.class));
71          assertThat(actual.getInventory().getInventoryPosition("t_2").get("ds1.t_2#2"), isA(UniqueKeyIngestPosition.class));
72      }
73      
74      @Test
75      void assertGetIncrementalLatestActiveTimeMillis() {
76          assertThat(getJobItemProgress(SystemResourceFileUtils.readFile("job-progress.yaml")).getIncremental().getIncrementalLatestActiveTimeMillis(), is(0L));
77      }
78      
79      @Test
80      void assertGetIncrementalDataLatestActiveTimeMillis() {
81          assertThat(getJobItemProgress(SystemResourceFileUtils.readFile("job-progress-all-finished.yaml")).getIncremental().getIncrementalLatestActiveTimeMillis(), is(50L));
82      }
83      
84      @Test
85      void assertGetProgressesCorrectly() {
86          Map<String, InventoryTaskProgress> progresses = new HashMap<>(4, 1F);
87          progresses.put("ds.order_item#0", new InventoryTaskProgress(UniqueKeyIngestPosition.ofInteger(Range.closed(BigInteger.ONE, BigInteger.valueOf(100L)))));
88          progresses.put("ds.order_item#1", new InventoryTaskProgress(UniqueKeyIngestPosition.ofUnsplit()));
89          progresses.put("ds.order#0", new InventoryTaskProgress(new IngestFinishedPosition()));
90          progresses.put("ds.test_order#0", new InventoryTaskProgress(UniqueKeyIngestPosition.ofString(Range.closed("1", "100"))));
91          JobItemInventoryTasksProgress progress = new JobItemInventoryTasksProgress(progresses);
92          Map<String, IngestPosition> orderPosition = progress.getInventoryPosition("order");
93          assertThat(orderPosition.size(), is(1));
94          assertThat(orderPosition.get("ds.order#0"), isA(IngestFinishedPosition.class));
95          Map<String, IngestPosition> testOrderPosition = progress.getInventoryPosition("test_order");
96          assertThat(testOrderPosition.size(), is(1));
97          assertThat(testOrderPosition.get("ds.test_order#0"), isA(UniqueKeyIngestPosition.class));
98          Map<String, IngestPosition> orderItemPosition = progress.getInventoryPosition("order_item");
99          assertThat(orderItemPosition.size(), is(2));
100         assertThat(orderItemPosition.get("ds.order_item#0"), isA(UniqueKeyIngestPosition.class));
101         assertThat(orderItemPosition.get("ds.order_item#1"), isA(UniqueKeyIngestPosition.class));
102     }
103     
104     private TransmissionJobItemProgress getJobItemProgress(final String data) {
105         return SWAPPER.swapToObject(YamlEngine.unmarshal(data, YamlTransmissionJobItemProgress.class));
106     }
107 }
