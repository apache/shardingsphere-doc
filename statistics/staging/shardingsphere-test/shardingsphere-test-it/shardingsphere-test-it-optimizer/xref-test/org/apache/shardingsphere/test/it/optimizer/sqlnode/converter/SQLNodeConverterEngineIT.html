46  import static org.hamcrest.MatcherAssert.assertThat;
47  import static org.hamcrest.Matchers.is;
48  
49  public abstract class SQLNodeConverterEngineIT {
50      
51      private static final SQLCases SQL_CASES = SQLConverterCasesRegistry.getInstance().getCases();
52      
53      private static final SQLNodeConverterTestCases SQL_NODE_CONVERTER_TEST_CASES = SQLNodeConverterTestCasesRegistry.getInstance().getCases();
54      
55      private static final String SELECT_STATEMENT_PREFIX = "SELECT";
56      
57      private static final String DELETE_STATEMENT_PREFIX = "DELETE";
58      
59      private static final String EXPLAIN_STATEMENT_PREFIX = "EXPLAIN";
60      
61      private static final String UPDATE_STATEMENT_PREFIX = "UPDATE";
62      
63      private static final String INSERT_STATEMENT_PREFIX = "INSERT";
64      
65      private static final String MERGE_STATEMENT_PREFIX = "MERGE";
66      
67      @ParameterizedTest(name = "{0} ({1}) -> {2}")
68      @ArgumentsSource(TestCaseArgumentsProvider.class)
69      void assertConvert(final String sqlCaseId, final SQLCaseType sqlCaseType, final String databaseType) {
70          String sql = SQL_CASES.getSQL(sqlCaseId, sqlCaseType, SQL_NODE_CONVERTER_TEST_CASES.get(sqlCaseId, sqlCaseType, databaseType).getParameters());
71          String actual = SQLNodeConverterEngine.convert(parseSQLStatement(databaseType, sql)).toSqlString(SQLDialectFactory.getSQLDialect(databaseType)).getSql().replace("\n", " ").replace("\r", "");
72          String expected = SQL_NODE_CONVERTER_TEST_CASES.get(sqlCaseId, sqlCaseType, databaseType).getExpectedSQL();
73          assertThat(actual, is(expected));
74      }
75      
76      private SQLStatement parseSQLStatement(final String databaseType, final String sql) {
77          return new SQLStatementVisitorEngine(databaseType).visit(new SQLParserEngine(databaseType, new CacheOption(128, 1024L)).parse(sql, false));
78      }
79      
80      private static final class TestCaseArgumentsProvider implements ArgumentsProvider {
81          
82          @Override
83          public Stream<? extends Arguments> provideArguments(final ParameterDeclarations parameters, final ExtensionContext context) {
84              SQLNodeConverterEngineITSettings settings = context.getRequiredTestClass().getAnnotation(SQLNodeConverterEngineITSettings.class);
85              Preconditions.checkNotNull(settings, "Annotation SQLBinderITSettings is required.");
86              return getTestParameters(settings.value()).stream();
87          }
88          
89          private Collection<Arguments> getTestParameters(final String... databaseTypes) {
90              Collection<Arguments> result = new LinkedList<>();
91              for (InternalSQLParserTestParameter each : SQL_CASES.generateTestParameters(Arrays.stream(databaseTypes).collect(Collectors.toSet()))) {
92                  if (null != SQL_NODE_CONVERTER_TEST_CASES.get(each.getSqlCaseId(), each.getSqlCaseType(), each.getDatabaseType()) && !isPlaceholderWithoutParameter(each) && isSupportedSQLCase(each)) {
93                      result.add(Arguments.of(each.getSqlCaseId(), each.getSqlCaseType(), "H2".equals(each.getDatabaseType()) ? "MySQL" : each.getDatabaseType()));
94                  }
95              }
96              return result;
97          }
98          
99          private boolean isPlaceholderWithoutParameter(final InternalSQLParserTestParameter testParam) {
100             return SQLCaseType.PLACEHOLDER == testParam.getSqlCaseType()
101                     && SQL_NODE_CONVERTER_TEST_CASES.get(testParam.getSqlCaseId(), testParam.getSqlCaseType(), testParam.getDatabaseType()).getParameters().isEmpty();
102         }
103         
104         private boolean isSupportedSQLCase(final InternalSQLParserTestParameter testParam) {
105             return testParam.getSqlCaseId().toUpperCase().startsWith(SELECT_STATEMENT_PREFIX)
106                     || testParam.getSqlCaseId().toUpperCase().startsWith(DELETE_STATEMENT_PREFIX)
107                     || testParam.getSqlCaseId().toUpperCase().startsWith(EXPLAIN_STATEMENT_PREFIX)
108                     || testParam.getSqlCaseId().toUpperCase().startsWith(UPDATE_STATEMENT_PREFIX)
109                     || testParam.getSqlCaseId().toUpperCase().startsWith(INSERT_STATEMENT_PREFIX)
110                     || testParam.getSqlCaseId().toUpperCase().startsWith(MERGE_STATEMENT_PREFIX);
111         }
112     }
113 }
