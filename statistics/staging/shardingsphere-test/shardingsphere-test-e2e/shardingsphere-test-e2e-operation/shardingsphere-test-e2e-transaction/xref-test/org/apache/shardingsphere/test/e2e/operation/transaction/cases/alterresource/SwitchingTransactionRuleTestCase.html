52  /**
53   * Switching transaction rule test case.
54   */
55  @Slf4j
56  @TransactionTestCase(adapters = TransactionTestConstants.PROXY, dbTypes = TransactionTestConstants.MYSQL, transactionTypes = TransactionType.LOCAL)
57  public final class SwitchingTransactionRuleTestCase extends BaseTransactionTestCase {
58      
59      private static final int THREAD_SIZE = 1;
60      
61      private static final int TRANSACTION_SIZE = 1000;
62      
63      private static final int MAX_SWITCH_COUNT = 6;
64      
65      private static final AtomicBoolean IS_FINISHED = new AtomicBoolean(false);
66      
67      private static final AtomicInteger SWITCH_COUNT = new AtomicInteger();
68      
69      public SwitchingTransactionRuleTestCase(final TransactionTestCaseParameter testCaseParam) {
70          super(testCaseParam);
71      }
72      
73      @Override
74      protected void executeTest(final TransactionContainerComposer containerComposer) {
75          innerRun(containerComposer);
76      }
77      
78      @SneakyThrows(InterruptedException.class)
79      private void innerRun(final TransactionContainerComposer containerComposer) {
80          List<Thread> tasks = new ArrayList<>(THREAD_SIZE);
81          for (int i = 0; i < THREAD_SIZE; i++) {
82              Thread updateThread = new Thread(new TransactionOperationsTask(getDataSource()));
83              updateThread.start();
84              tasks.add(updateThread);
85          }
86          ExecutorService executor = Executors.newSingleThreadExecutor();
87          executor.execute(new AlterTransactionRuleTask(containerComposer, getBaseTransactionITCase().getCommonSQL()));
88          for (Thread each : tasks) {
89              each.join();
90          }
91          IS_FINISHED.set(true);
92          executor.shutdown();
93      }
94      
95      @RequiredArgsConstructor
96      @Getter
97      private static final class AlterTransactionRuleTask implements Runnable {
98          
99          private final TransactionContainerComposer containerComposer;
100         
101         private final CommonSQLCommand commonSQL;
102         
103         @SneakyThrows(SQLException.class)
104         @Override
105         public void run() {
106             while (!IS_FINISHED.get()) {
107                 alterLocalTransactionRule();
108                 Awaitility.await().atMost(20L, TimeUnit.SECONDS).pollInterval(19L, TimeUnit.SECONDS).until(() -> true);
109                 alterXaTransactionRule("Narayana");
110                 Awaitility.await().atMost(20L, TimeUnit.SECONDS).pollInterval(19L, TimeUnit.SECONDS).until(() -> true);
111                 if (SWITCH_COUNT.incrementAndGet() >= MAX_SWITCH_COUNT) {
112                     alterLocalTransactionRule();
113                     IS_FINISHED.set(true);
114                     break;
115                 }
116             }
117         }
118         
119         private void alterLocalTransactionRule() throws SQLException {
120             try (Connection connection = containerComposer.getDataSource().getConnection()) {
121                 if (isExpectedTransactionRule(connection, TransactionType.LOCAL, "")) {
122                     return;
123                 }
124                 String alterLocalTransactionRule = commonSQL.getAlterLocalTransactionRule();
125                 log.info("Alter local transaction rule: {}", alterLocalTransactionRule);
126                 SWITCH_COUNT.getAndIncrement();
127                 executeWithLog(connection, alterLocalTransactionRule);
128             }
129             assertTrue(waitExpectedTransactionRule(TransactionType.LOCAL, "", containerComposer));
130         }
131         
132         private void alterXaTransactionRule(final String providerType) throws SQLException {
133             try (Connection connection = containerComposer.getDataSource().getConnection()) {
134                 if (isExpectedTransactionRule(connection, TransactionType.XA, providerType)) {
135                     return;
136                 }
137                 String alterXaTransactionRule = commonSQL.getAlterXATransactionRule().replace("${providerType}", providerType);
138                 log.info("Alter XA transaction rule: {}", alterXaTransactionRule);
139                 SWITCH_COUNT.getAndIncrement();
140                 executeWithLog(connection, alterXaTransactionRule);
141             }
142             assertTrue(waitExpectedTransactionRule(TransactionType.XA, providerType, containerComposer));
143         }
144         
145         private boolean isExpectedTransactionRule(final Connection connection, final TransactionType expectedTransType, final String expectedProviderType) throws SQLException {
146             Map<String, String> transactionRuleMap = executeShowTransactionRule(connection);
147             return Objects.equals(transactionRuleMap.get(TransactionTestConstants.DEFAULT_TYPE), expectedTransType.toString())
148                     && Objects.equals(transactionRuleMap.get(TransactionTestConstants.PROVIDER_TYPE), expectedProviderType);
149         }
150         
151         private Map<String, String> executeShowTransactionRule(final Connection connection) throws SQLException {
152             Statement statement = connection.createStatement();
153             ResultSet resultSet = statement.executeQuery("SHOW TRANSACTION RULE;");
154             Map<String, String> result = new HashMap<>();
155             while (resultSet.next()) {
156                 String defaultType = resultSet.getString(TransactionTestConstants.DEFAULT_TYPE);
157                 String providerType = resultSet.getString(TransactionTestConstants.PROVIDER_TYPE);
158                 result.put(TransactionTestConstants.DEFAULT_TYPE, defaultType);
159                 result.put(TransactionTestConstants.PROVIDER_TYPE, providerType);
160             }
161             statement.close();
162             return result;
163         }
164         
165         private boolean waitExpectedTransactionRule(final TransactionType expectedTransType, final String expectedProviderType,
166                                                     final TransactionContainerComposer containerComposer) throws SQLException {
167             Awaitility.await().pollDelay(5L, TimeUnit.SECONDS).until(() -> true);
168             try (Connection connection = containerComposer.getDataSource().getConnection()) {
169                 int waitTimes = 0;
170                 do {
171                     if (isExpectedTransactionRule(connection, expectedTransType, expectedProviderType)) {
172                         return true;
173                     }
174                     Awaitility.await().pollDelay(2L, TimeUnit.SECONDS).until(() -> true);
175                     waitTimes++;
176                 } while (waitTimes <= 3);
177                 return false;
178             }
179         }
180     }

182     @RequiredArgsConstructor
183     @Getter
184     private static final class TransactionOperationsTask implements Runnable {
185         
186         private static final AtomicInteger ID_COUNTER = new AtomicInteger();
187         
188         private final DataSource dataSource;
189         
190         @SneakyThrows(SQLException.class)
191         public void run() {
192             Connection connection = dataSource.getConnection();
193             for (int i = 0; i < TRANSACTION_SIZE; i++) {
194                 log.info("Transaction {} start", i);
195                 executeOneTransaction(connection);
196                 if (IS_FINISHED.get()) {
197                     break;
198                 }
199                 log.info("Transaction {} end", i);
200             }
201             connection.close();
202         }
203         
204         private static void executeOneTransaction(final Connection connection) throws SQLException {
205             boolean isErrorOccured = false;
206             ThreadLocalRandom random = ThreadLocalRandom.current();
207             try {
208                 connection.setAutoCommit(false);
209                 int id = ID_COUNTER.incrementAndGet();
210                 PreparedStatement insertStatement = connection.prepareStatement("INSERT INTO account(id, balance, transaction_id) VALUES(?, ?, ?)");
211                 insertStatement.setObject(1, id);
212                 insertStatement.setObject(2, id);
213                 insertStatement.setObject(3, id);
214                 insertStatement.execute();
215                 PreparedStatement updateStatement = connection.prepareStatement("UPDATE account SET balance = balance - 1 WHERE id = ?");
216                 updateStatement.setObject(1, id);
217                 updateStatement.execute();
218                 PreparedStatement selectStatement = connection.prepareStatement("SELECT * FROM account WHERE id = ?");
219                 selectStatement.setObject(1, id);
220                 selectStatement.executeQuery();
221                 PreparedStatement deleteStatement = connection.prepareStatement("DELETE FROM account WHERE id = ?");
222                 deleteStatement.setObject(1, id);
223                 deleteStatement.execute();
224                 long time = random.nextLong(900) + 100;
225                 Awaitility.await().atMost(time + 10L, TimeUnit.MILLISECONDS).pollInterval(time, TimeUnit.MILLISECONDS).until(() -> true);
226                 // CHECKSTYLE:OFF
227             } catch (final Exception ex) {
228                 // CHECKSTYLE:ON
229                 log.error("Execute transaction exception occurred", ex);
230                 isErrorOccured = true;
231                 connection.rollback();
232             }
233             if (!isErrorOccured) {
234                 connection.commit();
235             }
236             connection.setAutoCommit(true);
237         }
238     }
