32  import static org.hamcrest.MatcherAssert.assertThat;
33  import static org.hamcrest.Matchers.is;
34  import static org.junit.jupiter.api.Assertions.assertFalse;
35  import static org.junit.jupiter.api.Assertions.assertTrue;

37  class EncryptDescriptorContractTest {
38      
39      @Test
40      void assertEncryptDistSQLExamples() {
41          assertEncryptDistSQLExampleValue(findToolDescriptor().getOutputSchema().get("examples"));
42      }
43      
44      @Test
45      void assertPlanningOutputDeclaresSummary() {
46          Map<?, ?> actualOutputSchema = findToolDescriptor().getOutputSchema();
47          Map<?, ?> actualProperties = (Map<?, ?>) actualOutputSchema.get("properties");
48          assertTrue(actualProperties.containsKey("summary"));
49      }
50      
51      @Test
52      void assertPromptCompletionArguments() {
53          assertCompletionTargetArguments(EncryptFeatureDefinition.PLAN_PROMPT_NAME, "database", "schema", "table", "column", "algorithm_type", "assisted_query_algorithm_type",
54                  "like_query_algorithm_type", "plan_id");
55      }
56      
57      @Test
58      void assertPlanIdInputGuidesNewPlanOmission() {
59          MCPToolDescriptor actualDescriptor = findToolDescriptor();
60          Map<?, ?> actualPlanIdInput = MCPToolDescriptorValidationUtils.findToolInputProperty(actualDescriptor, "plan_id").orElseThrow();
61          String actualDescription = actualPlanIdInput.get("description").toString();
62          assertFalse(MCPToolDescriptorValidationUtils.isRequiredToolInput(actualDescriptor, "plan_id"));
63          assertTrue(actualDescription.contains("Omit when starting a new plan"));
64          assertTrue(actualDescription.contains("never use placeholder values"));
65      }
66      
67      @Test
68      void assertPlanToolOperationTypesExposeOnlySupportedLifecycleActions() {
69          Map<?, ?> actualOperationType = MCPToolDescriptorValidationUtils.findToolInputProperty(findToolDescriptor(), "operation_type").orElseThrow();
70          assertThat(actualOperationType.get("enum"), is(List.of("create", "drop")));
71      }
72      
73      @Test
74      void assertPlanToolExposesEncryptRequirementsAtTopLevel() {
75          MCPToolDescriptor actualDescriptor = findToolDescriptor();
76          assertThat(MCPToolDescriptorValidationUtils.findToolInputProperty(actualDescriptor, "requires_decrypt").orElseThrow().get("type"), is("boolean"));
77          assertThat(MCPToolDescriptorValidationUtils.findToolInputProperty(actualDescriptor, "requires_equality_filter").orElseThrow().get("type"), is("boolean"));
78          assertThat(MCPToolDescriptorValidationUtils.findToolInputProperty(actualDescriptor, "requires_like_query").orElseThrow().get("type"), is("boolean"));
79          assertFalse(MCPToolDescriptorValidationUtils.findToolInputProperty(actualDescriptor, "structured_intent_evidence").isPresent());
80      }
81      
82      private MCPToolDescriptor findToolDescriptor() {
83          MCPDescriptorCatalog catalog = MCPDescriptorCatalogLoader.load();
84          return catalog.getProtocolDescriptors().getToolDescriptors().stream()
85                  .filter(each -> EncryptFeatureDefinition.PLAN_TOOL_NAME.equals(each.getName())).findFirst().orElseThrow();
86      }
87      
88      private void assertCompletionTargetArguments(final String promptName, final String... expectedArguments) {
89          MCPCompletionTargetDescriptor actual = MCPDescriptorCatalogLoader.load().getShardingSphereDescriptors().getCompletionTargetDescriptors().stream()
90                  .filter(each -> "prompt".equals(each.getReferenceType()) && promptName.equals(each.getReference())).findFirst().orElseThrow();
91          assertThat(actual.getArguments(), is(List.of(expectedArguments)));
92      }
93      
94      private void assertEncryptDistSQLExampleValue(final Object value) {
95          if (value instanceof Map) {
96              assertEncryptDistSQLExampleMap((Map<?, ?>) value);
97          } else if (value instanceof Collection) {
98              for (Object each : (Collection<?>) value) {
99                  assertEncryptDistSQLExampleValue(each);
100             }
101         }
102     }
103     
104     private void assertEncryptDistSQLExampleMap(final Map<?, ?> value) {
105         Object sql = value.get("sql");
106         if (null != sql && isEncryptRuleDistSQL(sql.toString())) {
107             assertEncryptRuleDistSQL(sql.toString());
108         }
109         for (Object each : value.values()) {
110             assertEncryptDistSQLExampleValue(each);
111         }
112     }
113     
114     private boolean isEncryptRuleDistSQL(final String sql) {
115         String actualSQL = sql.toUpperCase(Locale.ENGLISH);
116         return actualSQL.contains("CREATE ENCRYPT RULE");
117     }
118     
119     private void assertEncryptRuleDistSQL(final String sql) {
120         String actualSQL = sql.toLowerCase(Locale.ENGLISH);
121         assertFalse(actualSQL.contains("type(name=aes"));
122         assertFalse(actualSQL.contains("'aes-key-value'") && !actualSQL.contains("'digest-algorithm-name'"));
123     }
124 }
