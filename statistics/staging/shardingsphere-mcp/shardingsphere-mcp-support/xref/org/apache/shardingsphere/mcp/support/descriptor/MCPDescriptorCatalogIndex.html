68      private static Map<String, MCPResourceDescriptor> createResourceDescriptors() {
69          int expectedSize = CATALOG.getProtocolDescriptors().getResourceDescriptors().size() + CATALOG.getProtocolDescriptors().getResourceTemplateDescriptors().size();
70          Map<String, MCPResourceDescriptor> result = new LinkedHashMap<>(expectedSize, 1F);
71          for (MCPResourceDescriptor each : CATALOG.getProtocolDescriptors().getResourceDescriptors()) {
72              result.put(each.getUriTemplate(), each);
73          }
74          for (MCPResourceDescriptor each : CATALOG.getProtocolDescriptors().getResourceTemplateDescriptors()) {
75              result.put(each.getUriTemplate(), each);
76          }
77          return result;
78      }

80      private static Map<String, ShardingSphereMCPResourceMetadata> createShardingSphereResourceMetadata() {
81          return CATALOG.getShardingSphereDescriptors().getResourceMetadata().stream()
82                  .collect(Collectors.toMap(ShardingSphereMCPResourceMetadata::getUriTemplate, each -> each));
83      }

86          return CATALOG.getShardingSphereDescriptors().getResourceMetadata().stream().map(each -> Map.entry(new MCPUriTemplate(each.getUriTemplate()), each)).toList();
87      }

89      private static Map<String, MCPToolDescriptor> createToolDescriptors() {
90          return CATALOG.getProtocolDescriptors().getToolDescriptors().stream()
91                  .collect(Collectors.toMap(MCPToolDescriptor::getName, each -> each, (a, b) -> b, () -> new LinkedHashMap<>(CATALOG.getProtocolDescriptors().getToolDescriptors().size(), 1F)));
92      }

94      private static Map<String, Collection<MCPResourceNavigationDescriptor>> createResourceNavigationDescriptors() {
95          Map<String, Collection<MCPResourceNavigationDescriptor>> result = new LinkedHashMap<>(CATALOG.getShardingSphereDescriptors().getResourceNavigationDescriptors().size(), 1F);
96          for (MCPResourceNavigationDescriptor each : CATALOG.getShardingSphereDescriptors().getResourceNavigationDescriptors()) {
97              result.computeIfAbsent(each.getFrom(), unused -> new LinkedList<>()).add(each);
98          }
99          return result;
100     }

110     private static Map<String, String> createPlanningToolNamesByWorkflowKind() {
111         Map<String, String> result = new LinkedHashMap<>(CATALOG.getShardingSphereDescriptors().getToolRuntimeDescriptors().size(), 1F);
112         for (MCPToolRuntimeDescriptor each : CATALOG.getShardingSphereDescriptors().getToolRuntimeDescriptors()) {
113             if (!"plan".equals(each.getWorkflowRole())) {
114                 continue;
115             }
116             MCPToolDescriptor toolDescriptor = TOOL_DESCRIPTORS.get(each.getToolName());
117             if (null != toolDescriptor && toolDescriptor.getMeta().containsKey(MCPShardingSphereMetadataKeys.WORKFLOW_KIND)) {
118                 result.put(String.valueOf(toolDescriptor.getMeta().get(MCPShardingSphereMetadataKeys.WORKFLOW_KIND)), each.getToolName());
119             }
120         }
121         return result;
122     }
123     
124     private static Map<String, Collection<String>> createWorkflowKindsByPromptName() {
125         Map<String, Collection<String>> result = new LinkedHashMap<>(PROMPT_DESCRIPTORS.size(), 1F);
126         for (MCPPromptDescriptor each : PROMPT_DESCRIPTORS) {
127             Collection<String> workflowKinds = findPromptPlanningWorkflowKinds(each);
128             if (!workflowKinds.isEmpty()) {
129                 result.put(each.getName(), workflowKinds);
130             }
131         }
132         return result;
133     }
134     
135     private static Collection<String> findPromptPlanningWorkflowKinds(final MCPPromptDescriptor prompt) {
136         Object relatedTools = prompt.getMeta().get(MCPShardingSphereMetadataKeys.RELATED_TOOLS);
137         if (!(relatedTools instanceof Collection<?>)) {
138             return List.of();
139         }
140         String planningToolName = "";
141         for (Object each : (Collection<?>) relatedTools) {
142             String toolName = Objects.toString(each, "");
143             MCPToolRuntimeDescriptor runtimeDescriptor = TOOL_RUNTIME_DESCRIPTORS.get(toolName);
144             if (null == runtimeDescriptor) {
145                 if (TOOL_DESCRIPTORS.containsKey(toolName)) {
146                     continue;
147                 }
148                 return List.of();
149             }
150             if (!"plan".equals(runtimeDescriptor.getWorkflowRole())) {
151                 continue;
152             }
153             if (!planningToolName.isEmpty()) {
154                 return List.of();
155             }
156             planningToolName = toolName;
157         }
158         MCPToolDescriptor toolDescriptor = TOOL_DESCRIPTORS.get(planningToolName);
159         if (null == toolDescriptor) {
160             return List.of();
161         }
162         String workflowKind = Objects.toString(toolDescriptor.getMeta().get(MCPShardingSphereMetadataKeys.WORKFLOW_KIND), "");
163         return workflowKind.isEmpty() ? List.of() : List.of(workflowKind);
164     }

196     /**
197      * Resolve resource kind from a descriptor URI template or expanded resource URI.
198      *
199      * @param uri resource URI or URI template
200      * @return descriptor resource kind, or generic resource kind when the URI is unknown
201      */
202     public static String resolveResourceKind(final String uri) {
203         ShardingSphereMCPResourceMetadata metadata = SHARDINGSPHERE_RESOURCE_METADATA.get(uri);
204         if (null == metadata) {
205             metadata = RESOURCE_METADATA_TEMPLATES.stream().filter(each -> each.getKey().parse(uri).isPresent())
206                     .map(Entry::getValue).findFirst().orElse(null);
207         }
208         if (null == metadata) {
209             return "resource";
210         }
211         return null == metadata.getObjectScope() ? metadata.getResourceKind() : metadata.getObjectScope();
212     }

275      *
276      * @param descriptor completion target descriptor
277      * @return related workflow kinds
278      */

280         if (!"prompt".equals(descriptor.getReferenceType())) {
