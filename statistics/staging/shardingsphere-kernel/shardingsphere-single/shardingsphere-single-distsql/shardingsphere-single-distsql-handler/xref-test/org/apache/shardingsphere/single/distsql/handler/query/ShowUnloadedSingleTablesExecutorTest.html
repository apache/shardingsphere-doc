128     @ParameterizedTest(name = "{0}")
129     @MethodSource("getRowsArguments")
130     void assertGetRows(final String name, final boolean schemaAvailable, final Map<String, Collection<DataNode>> actualDataNodes,
131                        final Map<String, Collection<DataNode>> loadedDataNodes, final List<List<String>> expectedRows) {
132         mockRowsDependencies(actualDataNodes, loadedDataNodes);

143     void assertGetRowsWithDifferentProtocolAndStorageTypes() throws SQLException {
144         Map<String, DataSource> actualDataSourceMap = Collections.singletonMap("foo_ds",
145                 mockDataSource("foo_ds", null, "jdbc:mock://127.0.0.1/foo_ds", Collections.singletonList("foo_tbl1")));
146         when(DatabaseTypeEngine.getStorageType(any(DataSource.class))).thenReturn(storageDatabaseType);

149         when(SingleTableLoadUtils.splitTableLines(anyCollection())).thenCallRealMethod();
150         when(SingleTableDataNodeLoader.load(eq("foo_db"), eq(protocolDatabaseType), anyMap(), anyCollection(), anyCollection())).thenCallRealMethod();
151         when(SingleTableDataNodeLoader.load(eq("foo_db"), anyMap(), anyCollection(), anyCollection(), anyMap())).thenCallRealMethod();
152         Map<String, Collection<DataNode>> loadedDataNodes = SingleTableDataNodeLoader.load(
153                 "foo_db", protocolDatabaseType, actualDataSourceMap, Collections.emptyList(), Collections.singleton("foo_ds.foo_tbl1"));
154         when(database.getName()).thenReturn("foo_db");
155         when(database.getProtocolType()).thenReturn(protocolDatabaseType);
156         when(database.getResourceMetaData()).thenReturn(resourceMetaData);
157         when(database.getRuleMetaData()).thenReturn(ruleMetaData);

167         when(database.getName()).thenReturn("foo_db");
168         when(database.getProtocolType()).thenReturn(databaseType);
169         when(database.getResourceMetaData()).thenReturn(resourceMetaData);
170         when(database.getRuleMetaData()).thenReturn(ruleMetaData);
171         when(ruleMetaData.getRules()).thenReturn(Collections.emptyList());
172         when(rule.getSingleTableDataNodes()).thenReturn(loadedDataNodes);
173         when(resourceMetaData.getStorageUnits()).thenReturn(Collections.singletonMap("ds_0", storageUnit));
174         when(storageUnit.getDataSource()).thenReturn(dataSource);
175         when(PhysicalDataSourceAggregator.getAggregatedDataSources(any(), any())).thenReturn(Collections.singletonMap("ds_0", dataSource));
176         when(SingleTableLoadUtils.getExcludedTables(Collections.emptyList())).thenReturn(Collections.emptySet());

178         when(SingleTableDataNodeLoader.load(eq("foo_db"), any(), anyCollection(), anyCollection(), anyMap())).thenReturn(actualDataNodes);

181     private void assertRows(final List<List<String>> expectedRows) {
182         List<LocalDataQueryResultRow> actualRows = new LinkedList<>(executor.getRows(new ShowUnloadedSingleTablesStatement(null, null, null), mock(ContextManager.class)));
183         assertThat(actualRows.size(), is(expectedRows.size()));
184         for (int i = 0; i < actualRows.size(); i++) {
185             List<String> expectedRow = expectedRows.get(i);
186             assertThat(actualRows.get(i).getCell(1), is(expectedRow.get(0)));
187             assertThat(actualRows.get(i).getCell(2), is(expectedRow.get(1)));
188             if (3 == expectedRow.size()) {
189                 assertThat(actualRows.get(i).getCell(3), is(expectedRow.get(2)));
190             }
191         }
192     }

200     private DataSource mockDataSource(final String dataSourceName, final String schemaName, final String url, final List<String> tableNames) throws SQLException {
201         Connection connection = mock(Connection.class, RETURNS_DEEP_STUBS);
202         lenient().when(connection.getCatalog()).thenReturn(dataSourceName);
203         lenient().when(connection.getMetaData().getURL()).thenReturn(url);
204         ResultSet tableResultSet = mockTableResultSet(tableNames);
205         if (null == schemaName) {
206             when(connection.getMetaData().getTables(dataSourceName, null, null, new String[]{"TABLE", "PARTITIONED TABLE", "VIEW", "SYSTEM TABLE", "SYSTEM VIEW"}))
207                     .thenReturn(tableResultSet);
208         } else {
209             ResultSet schemaResultSet = mockSchemaResultSet(schemaName);
210             when(connection.getMetaData().getSchemas()).thenReturn(schemaResultSet);
211             when(connection.getMetaData().getTables(dataSourceName, schemaName, null, new String[]{"TABLE", "PARTITIONED TABLE", "VIEW", "SYSTEM TABLE", "SYSTEM VIEW"}))
212                     .thenReturn(tableResultSet);
213         }
214         return new MockedDataSource(connection);
215     }
216     
217     private ResultSet mockSchemaResultSet(final String schemaName) throws SQLException {
218         ResultSet result = mock(ResultSet.class);
219         when(result.next()).thenReturn(true, false);
220         when(result.getString("TABLE_SCHEM")).thenReturn(schemaName);
221         return result;
222     }
223     
224     private ResultSet mockTableResultSet(final List<String> tableNames) throws SQLException {
225         ResultSet result = mock(ResultSet.class);
226         Collection<String> remainTableNames = tableNames.subList(1, tableNames.size());
227         Collection<Boolean> remainNextResults = new LinkedList<>();
228         for (int i = 0; i < remainTableNames.size(); i++) {
229             remainNextResults.add(true);
230         }
231         remainNextResults.add(false);
232         String firstTableName = tableNames.get(0);
233         lenient().when(result.next()).thenReturn(true, remainNextResults.toArray(new Boolean[tableNames.size()]));
234         lenient().when(result.getString("TABLE_NAME")).thenReturn(firstTableName, remainTableNames.toArray(new String[tableNames.size() - 1]));
235         return result;
236     }

243     private static Stream<Arguments> getRowsArguments() {
244         Map<String, Collection<DataNode>> allLoadedActualDataNodes = new HashMap<>(
245                 Collections.singletonMap("t_order", Collections.singleton(new DataNode("ds_0", (String) null, "t_order"))));
246         Map<String, Collection<DataNode>> allLoadedRuleDataNodes = new HashMap<>(
247                 Collections.singletonMap("t_order", Collections.singleton(new DataNode("ds_0", (String) null, "t_order"))));
248         Map<String, Collection<DataNode>> allLoadedUpperCaseActualDataNodes = new HashMap<>(
249                 Collections.singletonMap("T_ORDER", Collections.singleton(new DataNode("ds_0", (String) null, "T_ORDER"))));
250         Map<String, Collection<DataNode>> allLoadedUpperCaseRuleDataNodes = new HashMap<>(
251                 Collections.singletonMap("T_ORDER", Collections.singleton(new DataNode("ds_0", (String) null, "T_ORDER"))));
252         Map<String, Collection<DataNode>> partiallyLoadedActualDataNodes = new HashMap<>(
253                 Collections.singletonMap("t_order", new LinkedList<>(Arrays.asList(new DataNode("ds_0", "public", "t_order"), new DataNode("ds_1", "public", "t_order")))));
254         Map<String, Collection<DataNode>> partiallyLoadedRuleDataNodes = new HashMap<>(
255                 Collections.singletonMap("t_order", Collections.singleton(new DataNode("ds_0", "public", "t_order"))));
256         Map<String, Collection<DataNode>> unmatchedActualDataNodes = new HashMap<>(
257                 Collections.singletonMap("t_order_item", Collections.singleton(new DataNode("ds_2", (String) null, "t_order_item"))));
258         Map<String, Collection<DataNode>> unmatchedRuleDataNodes = new HashMap<>(
259                 Collections.singletonMap("t_order", Collections.singleton(new DataNode("ds_0", (String) null, "t_order"))));
260         Map<String, Collection<DataNode>> unorderedActualDataNodes = new LinkedHashMap<>(2, 1F);
261         unorderedActualDataNodes.put("t_order_item", Collections.singleton(new DataNode("ds_1", (String) null, "t_order_item")));
262         unorderedActualDataNodes.put("t_order", Collections.singleton(new DataNode("ds_0", (String) null, "t_order")));
263         Map<String, Collection<DataNode>> unorderedStorageUnitActualDataNodes = new HashMap<>(
264                 Collections.singletonMap("t_order", new LinkedList<>(Arrays.asList(new DataNode("ds_1", "public", "t_order"), new DataNode("ds_0", "public", "t_order")))));
265         Map<String, Collection<DataNode>> unorderedSchemaActualDataNodes = new HashMap<>(
266                 Collections.singletonMap("t_order", new LinkedList<>(Arrays.asList(new DataNode("ds_0", "schema_b", "t_order"), new DataNode("ds_0", "schema_a", "t_order")))));
267         return Stream.of(
268                 Arguments.of("all loaded tables are excluded", false, allLoadedActualDataNodes, allLoadedRuleDataNodes, Collections.<List<String>>emptyList()),
269                 Arguments.of("all loaded uppercase tables are excluded", false, allLoadedUpperCaseActualDataNodes, allLoadedUpperCaseRuleDataNodes, Collections.<List<String>>emptyList()),
270                 Arguments.of("remaining tables are sorted by table name",
271                         false, unorderedActualDataNodes, Collections.emptyMap(), Arrays.asList(Arrays.asList("t_order", "ds_0"), Arrays.asList("t_order_item", "ds_1"))),
272                 Arguments.of("partially loaded table keeps remaining nodes",
273                         true, partiallyLoadedActualDataNodes, partiallyLoadedRuleDataNodes, Collections.singletonList(Arrays.asList("t_order", "ds_1", "public"))),
274                 Arguments.of("same table nodes are sorted by storage unit name",
275                         true, unorderedStorageUnitActualDataNodes, Collections.emptyMap(), Arrays.asList(Arrays.asList("t_order", "ds_0", "public"), Arrays.asList("t_order", "ds_1", "public"))),
276                 Arguments.of("same table and storage unit nodes are sorted by schema name",
277                         true, unorderedSchemaActualDataNodes, Collections.emptyMap(), Arrays.asList(Arrays.asList("t_order", "ds_0", "schema_a"), Arrays.asList("t_order", "ds_0", "schema_b"))),
278                 Arguments.of("unmatched rule key keeps actual table nodes",
279                         false, unmatchedActualDataNodes, unmatchedRuleDataNodes, Collections.singletonList(Arrays.asList("t_order_item", "ds_2"))));
280     }
