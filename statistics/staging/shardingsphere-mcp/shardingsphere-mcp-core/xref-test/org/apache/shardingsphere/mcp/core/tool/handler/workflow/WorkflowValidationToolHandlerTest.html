43  class WorkflowValidationToolHandlerTest {
44      
45      @Test
46      void assertGetToolDescriptor() {
47          WorkflowValidationToolHandler handler = new WorkflowValidationToolHandler(new WorkflowRuntimeDefinitionRegistry(List.of(WorkflowHandlerTestFixture.createDefinition("encrypt.rule"))));
48          MCPToolDescriptor actual = MCPDescriptorCatalogIndex.getRequiredToolDescriptor(handler.getToolName());
49          assertThat(actual.getName(), is("database_gateway_validate_workflow"));
50      }
51      
52      @Test
53      void assertHandle() {
54          MCPWorkflowRuntimeHandler workflowRuntimeHandler = mock(MCPWorkflowRuntimeHandler.class);
55          ValidationReport validationReport = new ValidationReport();
56          validationReport.setOverallStatus(WorkflowLifecycle.STATUS_PASSED);
57          when(workflowRuntimeHandler.validate(any(), any())).thenReturn(validationReport);
58          WorkflowContextSnapshot snapshot = WorkflowHandlerTestFixture.createSnapshot();
59          snapshot.setStatus(WorkflowLifecycle.STATUS_EXECUTED);
60          WorkflowHandlerTestFixture.Context fixture = WorkflowHandlerTestFixture.createContext(snapshot);
61          WorkflowValidationToolHandler handler = new WorkflowValidationToolHandler(new WorkflowRuntimeDefinitionRegistry(List.of(
62                  WorkflowHandlerTestFixture.createDefinition("encrypt.rule", workflowRuntimeHandler, MCPWorkflowApplyArtifactValidator.NO_OP))));
63          MCPSuccessPayload actual = handler.handle(fixture.requestContext(), Map.of("plan_id", "plan-1"));
64          verify(workflowRuntimeHandler).validate(snapshot, fixture.queryFacade());
65          assertThat(actual.toPayload().get("status"), is("validated"));
66      }
67      
68      @Test
69      void assertHandleRejectsDifferentSession() {
70          MCPWorkflowRuntimeHandler workflowRuntimeHandler = mock(MCPWorkflowRuntimeHandler.class);
71          WorkflowContextSnapshot snapshot = WorkflowHandlerTestFixture.createSnapshot();
72          snapshot.setSessionId("session-2");
73          snapshot.setStatus(WorkflowLifecycle.STATUS_EXECUTED);
74          WorkflowHandlerTestFixture.Context fixture = WorkflowHandlerTestFixture.createContext(snapshot);
75          WorkflowValidationToolHandler handler = new WorkflowValidationToolHandler(new WorkflowRuntimeDefinitionRegistry(List.of(
76                  WorkflowHandlerTestFixture.createDefinition("encrypt.rule", workflowRuntimeHandler, MCPWorkflowApplyArtifactValidator.NO_OP))));
77          MCPSuccessPayload actual = handler.handle(fixture.requestContext(), Map.of("plan_id", "plan-1"));
78          assertThat(actual.toPayload().get("status"), is(WorkflowLifecycle.STATUS_FAILED));
79          assertThat(((Map<?, ?>) ((List<?>) actual.toPayload().get("issues")).getFirst()).get("code"), is(WorkflowIssueCode.SESSION_OWNERSHIP_MISMATCH));
80          verify(workflowRuntimeHandler, never()).validate(any(), any());
81      }
82  }
