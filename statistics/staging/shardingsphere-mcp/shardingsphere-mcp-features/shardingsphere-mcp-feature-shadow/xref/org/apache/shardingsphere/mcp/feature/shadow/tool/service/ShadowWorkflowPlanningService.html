93      /**
94       * Plan shadow rule workflow.
95       *
96       * @param workflowSessionContext workflow session context
97       * @param queryFacade query facade
98       * @param request workflow request
99       * @return workflow snapshot
100      */
101     public WorkflowContextSnapshot planRule(final WorkflowSessionContext workflowSessionContext, final MCPFeatureQueryFacade queryFacade,
102                                             final ShadowRuleWorkflowRequest request) {
103         WorkflowContextSnapshot result = workflowSessionContext.getOrCreate(request.getPlanId());
104         ShadowRuleWorkflowRequest mergedRequest = prepareSnapshot(result, request, ShadowFeatureDefinition.RULE_WORKFLOW_KIND,
105                 WorkflowLifecycle.OPERATION_CREATE, "Shadow rule workflow plan.", RULE_INTERACTION_STEPS);
106         result.getResourceUriTemplates().addAll(List.of(ShadowFeatureDefinition.STORAGE_UNITS_RESOURCE_URI,
107                 ShadowFeatureDefinition.SINGLE_TABLES_RESOURCE_URI, ShadowFeatureDefinition.SINGLE_TABLE_RESOURCE_URI));
108         planningSupport.applyResolvedIntent(mergedRequest, result.getClarifiedIntent());
109         if (!planningSupport.ensureSupportedOperationType(result.getClarifiedIntent(), SUPPORTED_RULE_OPERATION_TYPES, result)) {
110             return planningSupport.persistPlanningInterruption(workflowSessionContext, result);
111         }
112         planAlgorithmsIfRequired(queryFacade, mergedRequest, result);
113         if (!ensureRulePlanningContext(mergedRequest, result.getClarifiedIntent(), result)) {
114             return planningSupport.persistPlanningInterruption(workflowSessionContext, result);
115         }
116         if (!isReadyForAlgorithmArtifactPlanning(mergedRequest, result)) {
117             return workflowSessionContext.persist(result, WorkflowLifecycle.STEP_CLARIFYING, WorkflowLifecycle.STATUS_CLARIFYING);
118         }
119         queryFacade.checkDatabaseCapability(mergedRequest.getDatabase());
120         if (!ensureRuleLifecycle(result.getClarifiedIntent(), mergedRequest, inspectionService.queryRules(queryFacade, mergedRequest.getDatabase()), result, queryFacade)) {
121             return workflowSessionContext.persist(result, WorkflowLifecycle.STEP_FAILED, WorkflowLifecycle.STATUS_FAILED);
122         }
123         addRuleArtifact(result, mergedRequest, result.getClarifiedIntent().getOperationType());
124         return workflowSessionContext.persist(result, WorkflowLifecycle.STEP_REVIEW, WorkflowLifecycle.STATUS_PLANNED);
125     }

127     /**
128      * Plan default shadow algorithm workflow.
129      *
130      * @param workflowSessionContext workflow session context
131      * @param queryFacade query facade
132      * @param request workflow request
133      * @return workflow snapshot
134      */
135     public WorkflowContextSnapshot planDefaultAlgorithm(final WorkflowSessionContext workflowSessionContext, final MCPFeatureQueryFacade queryFacade,
136                                                         final ShadowDefaultAlgorithmWorkflowRequest request) {
137         WorkflowContextSnapshot result = workflowSessionContext.getOrCreate(request.getPlanId());
138         ShadowDefaultAlgorithmWorkflowRequest mergedRequest = prepareSnapshot(result, request, ShadowFeatureDefinition.DEFAULT_ALGORITHM_WORKFLOW_KIND,
139                 WorkflowLifecycle.OPERATION_CREATE, "Default shadow algorithm workflow plan.", DEFAULT_ALGORITHM_INTERACTION_STEPS);
140         planningSupport.applyResolvedIntent(mergedRequest, result.getClarifiedIntent());
141         if (!planningSupport.ensureSupportedOperationType(result.getClarifiedIntent(), SUPPORTED_RULE_OPERATION_TYPES, result)) {
142             return planningSupport.persistPlanningInterruption(workflowSessionContext, result);
143         }
144         planAlgorithmsIfRequired(queryFacade, mergedRequest, result);
145         if (!ensureDefaultAlgorithmPlanningContext(mergedRequest, result.getClarifiedIntent(), result)) {
146             return planningSupport.persistPlanningInterruption(workflowSessionContext, result);
147         }
148         if (!ensureDefaultAlgorithmType(mergedRequest, result)) {
149             return workflowSessionContext.persist(result, WorkflowLifecycle.STEP_FAILED, WorkflowLifecycle.STATUS_FAILED);
150         }
151         boolean exists = !inspectionService.queryDefaultAlgorithm(queryFacade, mergedRequest.getDatabase()).isEmpty();
152         if (!planningSupport.ensureLifecycleState("Default shadow algorithm", result.getClarifiedIntent(), exists, result)) {
153             return workflowSessionContext.persist(result, WorkflowLifecycle.STEP_FAILED, WorkflowLifecycle.STATUS_FAILED);
154         }
155         addDefaultAlgorithmArtifact(result, mergedRequest, result.getClarifiedIntent().getOperationType());
156         return workflowSessionContext.persist(result, WorkflowLifecycle.STEP_REVIEW, WorkflowLifecycle.STATUS_PLANNED);
157     }

159     /**
160      * Plan shadow algorithm cleanup workflow.
161      *
162      * @param workflowSessionContext workflow session context
163      * @param queryFacade query facade
164      * @param request workflow request
165      * @return workflow snapshot
166      */
167     public WorkflowContextSnapshot planAlgorithmCleanup(final WorkflowSessionContext workflowSessionContext, final MCPFeatureQueryFacade queryFacade,
168                                                         final ShadowAlgorithmCleanupWorkflowRequest request) {
169         WorkflowContextSnapshot result = workflowSessionContext.getOrCreate(request.getPlanId());
170         ShadowAlgorithmCleanupWorkflowRequest mergedRequest = prepareSnapshot(result, request, ShadowFeatureDefinition.ALGORITHM_CLEANUP_WORKFLOW_KIND,
171                 WorkflowLifecycle.OPERATION_DROP, "Shadow algorithm cleanup workflow plan.", CLEANUP_INTERACTION_STEPS);
172         planningSupport.applyResolvedIntent(mergedRequest, result.getClarifiedIntent());
173         if (!planningSupport.ensureSupportedOperationType(result.getClarifiedIntent(), SUPPORTED_CLEANUP_OPERATION_TYPES, result)) {
174             return planningSupport.persistPlanningInterruption(workflowSessionContext, result);
175         }
176         if (!ensureCleanupPlanningContext(mergedRequest, result.getClarifiedIntent(), result)) {
177             return planningSupport.persistPlanningInterruption(workflowSessionContext, result);
178         }
179         List<Map<String, Object>> algorithms = inspectionService.queryAlgorithms(queryFacade, mergedRequest.getDatabase());
180         List<Map<String, Object>> tableRules = inspectionService.queryTableRules(queryFacade, mergedRequest.getDatabase());
181         List<Map<String, Object>> defaultAlgorithm = inspectionService.queryDefaultAlgorithm(queryFacade, mergedRequest.getDatabase());
182         queryFacade.checkDatabaseCapability(mergedRequest.getDatabase());
183         if (!ensureAlgorithmCleanupState(mergedRequest, algorithms, tableRules, defaultAlgorithm, result, queryFacade)) {
184             return workflowSessionContext.persist(result, WorkflowLifecycle.STEP_FAILED, WorkflowLifecycle.STATUS_FAILED);
185         }
186         result.getRuleArtifacts().add(distSQLPlanningService.planDropAlgorithm(mergedRequest.getAlgorithmName()));
187         return workflowSessionContext.persist(result, WorkflowLifecycle.STEP_REVIEW, WorkflowLifecycle.STATUS_PLANNED);
188     }

191                                                       final String defaultOperationType, final String summary, final List<String> interactionSteps) {

213     private boolean ensureRulePlanningContext(final ShadowRuleWorkflowRequest request, final ClarifiedIntent clarifiedIntent, final WorkflowContextSnapshot snapshot) {
214         if (!ensureDatabase(request, clarifiedIntent, snapshot) || !ensureSupportedIdentifiers(snapshot, request.getDatabase(), request.getRuleName(), request.getSourceStorageUnit(),
215                 request.getShadowStorageUnit(), request.getTableName())) {
216             return false;
217         }
218         List<String> missingInputs = new LinkedList<>();
219         addMissingInput(missingInputs, request.getRuleName(), ShadowFeatureDefinition.RULE_FIELD);
220         if (!WorkflowLifecycle.OPERATION_DROP.equalsIgnoreCase(request.getOperationType())) {
221             addMissingInput(missingInputs, request.getSourceStorageUnit(), ShadowFeatureDefinition.SOURCE_STORAGE_UNIT_FIELD);
222             addMissingInput(missingInputs, request.getShadowStorageUnit(), ShadowFeatureDefinition.SHADOW_STORAGE_UNIT_FIELD);
223             addMissingInput(missingInputs, request.getTableName(), ShadowFeatureDefinition.TABLE_FIELD);
224             addMissingInput(missingInputs, request.getAlgorithmType(), ShadowFeatureDefinition.ALGORITHM_TYPE_FIELD);
225         }
226         return ensureNoMissingInputs(missingInputs, clarifiedIntent, snapshot, "Shadow rule DistSQL requires explicit rule, storage unit, table and algorithm inputs.");
227     }
228     
229     private boolean ensureDefaultAlgorithmPlanningContext(final ShadowDefaultAlgorithmWorkflowRequest request, final ClarifiedIntent clarifiedIntent, final WorkflowContextSnapshot snapshot) {
230         if (!ensureDatabase(request, clarifiedIntent, snapshot)) {
231             return false;
232         }
233         List<String> missingInputs = new LinkedList<>();
234         if (!WorkflowLifecycle.OPERATION_DROP.equalsIgnoreCase(request.getOperationType())) {
235             addMissingInput(missingInputs, request.getAlgorithmType(), ShadowFeatureDefinition.ALGORITHM_TYPE_FIELD);
236         }
237         return ensureNoMissingInputs(missingInputs, clarifiedIntent, snapshot, "Default shadow algorithm DistSQL requires an explicit algorithm type and properties.");
238     }

240     private boolean ensureDefaultAlgorithmType(final ShadowDefaultAlgorithmWorkflowRequest request, final WorkflowContextSnapshot snapshot) {
241         if (WorkflowLifecycle.OPERATION_DROP.equalsIgnoreCase(request.getOperationType())
242                 || ShadowFeatureDefinition.DEFAULT_ALGORITHM_TYPE.equalsIgnoreCase(request.getAlgorithmType())) {
243             return true;
244         }
245         snapshot.getIssues().add(new WorkflowIssue(WorkflowIssueCode.ALGORITHM_CAPABILITY_CONFLICT, "error", WorkflowLifecycle.STEP_SELECTING_ALGORITHM,
246                 "Default shadow algorithm must implement the hint algorithm contract.",
247                 "Use algorithm_type=SQL_HINT for the default shadow algorithm.", false, Map.of("algorithm_type", request.getAlgorithmType())));
248         snapshot.setStatus(WorkflowLifecycle.STATUS_FAILED);
249         return false;
250     }
251     
252     private boolean ensureCleanupPlanningContext(final ShadowAlgorithmCleanupWorkflowRequest request, final ClarifiedIntent clarifiedIntent, final WorkflowContextSnapshot snapshot) {
253         return ensureDatabase(request, clarifiedIntent, snapshot) && ensureSupportedIdentifiers(snapshot, request.getDatabase(), request.getAlgorithmName())
254                 && ensureNoMissingInputs(request.getAlgorithmName().isEmpty() ? List.of(ShadowFeatureDefinition.ALGORITHM_NAME_FIELD) : List.of(), clarifiedIntent, snapshot,
255                         "Shadow algorithm cleanup requires the configured algorithm name.");
256     }
257     
258     private boolean ensureDatabase(final WorkflowRequest request, final ClarifiedIntent clarifiedIntent, final WorkflowContextSnapshot snapshot) {
259         if (request.getDatabase().isEmpty()) {
260             clarifiedIntent.getClarificationMessages().add("Please provide logical database first.");
261             snapshot.getIssues().add(new WorkflowIssue(WorkflowIssueCode.DATABASE_REQUIRED, "error", WorkflowLifecycle.STEP_INTAKING,
262                     "Database is required before planning shadow DistSQL.", "Provide the logical database name.", true, Map.of()));
263             snapshot.setStatus(WorkflowLifecycle.STATUS_CLARIFYING);
264             return false;
265         }
266         if (!ensureSupportedIdentifiers(snapshot, request.getDatabase())) {
267             snapshot.setStatus(WorkflowLifecycle.STATUS_FAILED);
268             return false;
269         }
270         return true;
271     }
272     
273     private boolean ensureSupportedIdentifiers(final WorkflowContextSnapshot snapshot, final String... identifiers) {
274         if (planningSupport.ensureOptionalSupportedIdentifiers("", List.of(identifiers), snapshot, WorkflowLifecycle.STEP_INTAKING)) {
275             return true;
276         }
277         snapshot.setStatus(WorkflowLifecycle.STATUS_FAILED);
278         return false;
279     }

281     private void planAlgorithmsIfRequired(final MCPFeatureQueryFacade queryFacade, final WorkflowRequest request, final WorkflowContextSnapshot snapshot) {
282         if (!isDropWorkflow(snapshot.getClarifiedIntent()) && !request.getDatabase().isEmpty()) {
283             planAlgorithms(queryFacade, request, snapshot);
284         }
285     }

287     private boolean isReadyForAlgorithmArtifactPlanning(final WorkflowRequest request, final WorkflowContextSnapshot snapshot) {
288         return isDropWorkflow(snapshot.getClarifiedIntent())
289                 || planningSupport.isReadyForArtifactPlanning(request, snapshot.getClarifiedIntent(), snapshot, findPropertyRequirements(request), ALGORITHM_ARTIFACT_READY_MESSAGE);
290     }

292     private boolean isDropWorkflow(final ClarifiedIntent clarifiedIntent) {
293         return WorkflowLifecycle.OPERATION_DROP.equalsIgnoreCase(clarifiedIntent.getOperationType());
294     }

296     private void addMissingInput(final Collection<String> missingInputs, final String value, final String fieldName) {
297         if (value.trim().isEmpty()) {
298             missingInputs.add(fieldName);
299         }
300     }
301     
302     private boolean ensureNoMissingInputs(final Collection<String> missingInputs, final ClarifiedIntent clarifiedIntent,
303                                           final WorkflowContextSnapshot snapshot, final String message) {
304         if (missingInputs.isEmpty()) {
305             return true;
306         }
307         clarifiedIntent.getClarificationMessages().add(message);
308         snapshot.getIssues().add(new WorkflowIssue(WorkflowIssueCode.RULE_INPUT_REQUIRED, "error", WorkflowLifecycle.STEP_INTAKING, message,
309                 "Provide the missing fields instead of guessing storage units, tables or algorithms.", true, Map.of("missing_inputs", missingInputs)));
310         snapshot.setStatus(WorkflowLifecycle.STATUS_CLARIFYING);
311         return false;
312     }

314     private boolean ensureRuleLifecycle(final ClarifiedIntent clarifiedIntent, final ShadowRuleWorkflowRequest request, final List<Map<String, Object>> rules,
315                                         final WorkflowContextSnapshot snapshot, final MCPFeatureQueryFacade queryFacade) {
316         return planningSupport.ensureLifecycleState("Shadow rule", clarifiedIntent,
317                 rules.stream().anyMatch(each -> queryFacade.isSameIdentifier(
318                         request.getDatabase(), IdentifierScope.TABLE, request.getRuleName(), WorkflowRuleValueUtils.getRuleValue(each, "rule_name"))),
319                 snapshot);
320     }

322     private boolean ensureAlgorithmCleanupState(final ShadowAlgorithmCleanupWorkflowRequest request, final List<Map<String, Object>> algorithms,
323                                                 final List<Map<String, Object>> tableRules, final List<Map<String, Object>> defaultAlgorithm,
324                                                 final WorkflowContextSnapshot snapshot, final MCPFeatureQueryFacade queryFacade) {
325         boolean configured = algorithms.stream().anyMatch(each -> queryFacade.isSameIdentifier(
326                 request.getDatabase(), IdentifierScope.TABLE, request.getAlgorithmName(), WorkflowRuleValueUtils.getRuleValue(each, "shadow_algorithm_name")));
327         if (!configured) {
328             snapshot.getIssues().add(new WorkflowIssue(WorkflowIssueCode.RULE_STATE_MISMATCH, "error", WorkflowLifecycle.STEP_DISCOVERING,
329                     String.format("Shadow algorithm `%s` is not configured.", request.getAlgorithmName()),
330                     "Inspect configured shadow algorithms before cleanup.", false, Map.of("algorithm", request.getAlgorithmName())));
331             return false;
332         }
333         if (isReferencedByTableRule(queryFacade, request.getDatabase(), request.getAlgorithmName(), tableRules)
334                 || isDefaultAlgorithm(queryFacade, request.getDatabase(), request.getAlgorithmName(), defaultAlgorithm)) {
335             snapshot.getIssues().add(new WorkflowIssue(WorkflowIssueCode.RULE_STATE_MISMATCH, "error", WorkflowLifecycle.STEP_DISCOVERING,
336                     String.format("Shadow algorithm `%s` is still referenced.", request.getAlgorithmName()),
337                     "Remove table-rule or default-algorithm references before cleanup.", false, Map.of("algorithm", request.getAlgorithmName())));
338             return false;
339         }
340         return true;
341     }
342     
343     private boolean isReferencedByTableRule(final MCPFeatureQueryFacade queryFacade, final String databaseName,
344                                             final String algorithmName, final List<Map<String, Object>> tableRules) {
345         return tableRules.stream().map(each -> WorkflowRuleValueUtils.getRuleValue(each, "shadow_algorithm_name"))
346                 .flatMap(each -> Arrays.stream(each.split(","))).map(String::trim).filter(each -> !each.isEmpty())
347                 .anyMatch(each -> queryFacade.isSameIdentifier(databaseName, IdentifierScope.TABLE, algorithmName, each));
348     }
349     
350     private boolean isDefaultAlgorithm(final MCPFeatureQueryFacade queryFacade, final String databaseName,
351                                        final String algorithmName, final List<Map<String, Object>> defaultAlgorithm) {
352         return defaultAlgorithm.stream().anyMatch(each -> queryFacade.isSameIdentifier(databaseName, IdentifierScope.TABLE, algorithmName,
353                 WorkflowRuleValueUtils.getRuleValue(each, "shadow_algorithm_name")));
354     }

356     private void planAlgorithms(final MCPFeatureQueryFacade queryFacade, final WorkflowRequest request, final WorkflowContextSnapshot snapshot) {
357         List<AlgorithmCandidate> algorithmCandidates = algorithmRecommendationService.recommendShadowAlgorithms(
358                 request, inspectionService.queryAlgorithmPlugins(queryFacade), snapshot.getIssues());
359         snapshot.getAlgorithmCandidates().addAll(algorithmCandidates);
360         if (!algorithmCandidates.isEmpty()) {
361             request.setAlgorithmType(algorithmCandidates.getFirst().getAlgorithmType());
362         }
363     }
364     
365     private List<AlgorithmPropertyRequirement> findPropertyRequirements(final WorkflowRequest request) {
366         return algorithmPropertyTemplateService.findRequirements(request.getAlgorithmType());
367     }
368     
369     private void addRuleArtifact(final WorkflowContextSnapshot snapshot, final ShadowRuleWorkflowRequest request, final String operationType) {
370         if (WorkflowLifecycle.OPERATION_DROP.equalsIgnoreCase(operationType)) {
371             snapshot.getRuleArtifacts().add(distSQLPlanningService.planDropRule(request.getRuleName()));
372         } else if (WorkflowLifecycle.OPERATION_ALTER.equalsIgnoreCase(operationType)) {
373             snapshot.getRuleArtifacts().add(distSQLPlanningService.planAlterRule(request));
374         } else {
375             snapshot.getRuleArtifacts().add(distSQLPlanningService.planCreateRule(request));
376         }
377     }
378     
379     private void addDefaultAlgorithmArtifact(final WorkflowContextSnapshot snapshot, final ShadowDefaultAlgorithmWorkflowRequest request, final String operationType) {
380         if (WorkflowLifecycle.OPERATION_DROP.equalsIgnoreCase(operationType)) {
381             snapshot.getRuleArtifacts().add(distSQLPlanningService.planDropDefaultAlgorithm());
382         } else if (WorkflowLifecycle.OPERATION_ALTER.equalsIgnoreCase(operationType)) {
383             snapshot.getRuleArtifacts().add(distSQLPlanningService.planAlterDefaultAlgorithm(request));
384         } else {
385             snapshot.getRuleArtifacts().add(distSQLPlanningService.planCreateDefaultAlgorithm(request));
386         }
