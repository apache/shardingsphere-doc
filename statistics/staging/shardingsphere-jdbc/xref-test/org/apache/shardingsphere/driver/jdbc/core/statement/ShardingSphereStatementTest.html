335     @Test
336     void assertExecuteDistSQLDuringDatabaseRemoval() {
337         DatabaseType databaseType = TypedSPILoader.getService(DatabaseType.class, "MySQL");
338         ShardingSphereStatement statement = createStatement(databaseType);
339         ShardingSphereConnection connection = statement.getConnection();
340         ShardingSphereMetaData metaData = connection.getContextManager().getMetaDataContexts().getMetaData();
341         SQLParserRule parserRule = mock(SQLParserRule.class, RETURNS_DEEP_STUBS);
342         RuleMetaData globalRuleMetaData = new RuleMetaData(Collections.singletonList(parserRule));
343         when(metaData.getGlobalRuleMetaData()).thenReturn(globalRuleMetaData);
344         ParseStatement sqlStatement = new ParseStatement("SELECT 1");
345         UnknownDatabaseException expected = new UnknownDatabaseException("foo_db");
346         when(parserRule.getSQLParserEngine(databaseType).parse("foo_distsql", false)).thenAnswer(invocation -> {
347             when(metaData.getDatabase("foo_db")).thenReturn(null);
348             when(connection.getContextManager().getDatabase("foo_db")).thenThrow(expected);
349             return sqlStatement;
350         });
351         SQLException actual = assertThrows(SQLException.class, () -> statement.execute("foo_distsql"));
352         assertThat(actual.getCause(), sameInstance(expected));
353     }

407     private ShardingSphereConnection createConnection(final ShardingSphereMetaData metaData) {
408         ShardingSphereConnection result = mock(ShardingSphereConnection.class, RETURNS_DEEP_STUBS);
409         when(result.getContextManager().getMetaDataContexts().getMetaData()).thenReturn(metaData);
410         when(result.getCurrentDatabaseName()).thenReturn("foo_db");
411         ShardingSphereDatabase database = metaData.getDatabase("foo_db");
412         when(result.getContextManager().getDatabase("foo_db")).thenReturn(database);
413         return result;
414     }

416     private ShardingSphereMetaData createMetaData(final DatabaseType databaseType) {
417         ShardingSphereDatabase database = mock(ShardingSphereDatabase.class);
418         when(database.getProtocolType()).thenReturn(databaseType);
419         when(database.getRuleMetaData()).thenReturn(new RuleMetaData(Collections.emptyList()));
420         ShardingSphereMetaData result = mock(ShardingSphereMetaData.class);
421         when(result.getDatabase("foo_db")).thenReturn(database);
422         when(result.getProtocolType()).thenReturn(databaseType);
423         when(result.getGlobalRuleMetaData()).thenReturn(new RuleMetaData(Arrays.asList(
424                 new SQLParserRule(new DefaultSQLParserRuleConfigurationBuilder().build()),
425                 new SQLFederationRule(new DefaultSQLFederationRuleConfigurationBuilder().build(), Collections.emptyList()))));
426         when(result.getProps()).thenReturn(new ConfigurationProperties(new Properties()));
427         return result;
428     }

430     @SneakyThrows(ReflectiveOperationException.class)
431     private void setInsertStatementContext(final ShardingSphereStatement statement) {
432         GeneratedKeyContext generatedKeyContext = mock(GeneratedKeyContext.class);
433         when(generatedKeyContext.getColumnName()).thenReturn("id");
434         when(generatedKeyContext.getGeneratedValues()).thenReturn(Collections.emptyList());
435         InsertStatementContext insertStatementContext = mock(InsertStatementContext.class);
436         when(insertStatementContext.getGeneratedKeyContext()).thenReturn(Optional.of(generatedKeyContext));
437         QueryContext queryContext = mock(QueryContext.class);
438         when(queryContext.getSqlStatementContext()).thenReturn(insertStatementContext);
439         Plugins.getMemberAccessor().set(ShardingSphereStatement.class.getDeclaredField("queryContext"), statement, queryContext);
440         Plugins.getMemberAccessor().set(ShardingSphereStatement.class.getDeclaredField("returnGeneratedKeys"), statement, true);
441     }

443     private void addStatement(final ShardingSphereStatement shardingSphereStatement, final ResultSet generatedKeys) throws SQLException {
444         Statement statement = mock(Statement.class);
445         when(statement.getGeneratedKeys()).thenReturn(generatedKeys);
446         getStatements(shardingSphereStatement).add(statement);
447     }
