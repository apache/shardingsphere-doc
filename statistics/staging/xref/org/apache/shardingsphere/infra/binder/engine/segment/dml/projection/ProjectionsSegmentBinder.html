50  /**
51   * Projections segment binder.
52   */
53  @NoArgsConstructor(access = AccessLevel.PRIVATE)
54  public final class ProjectionsSegmentBinder {
55      
56      /**
57       * Bind projections segment.
58       *
59       * @param segment table segment
60       * @param binderContext statement binder context
61       * @param boundTableSegment bound table segment
62       * @param tableBinderContexts table binder contexts
63       * @param outerTableBinderContexts outer table binder contexts
64       * @return bound projections segment
65       */
66      public static ProjectionsSegment bind(final ProjectionsSegment segment, final SQLStatementBinderContext binderContext, final TableSegment boundTableSegment,
67                                            final Multimap<CaseInsensitiveString, TableSegmentBinderContext> tableBinderContexts,
68                                            final Multimap<CaseInsensitiveString, TableSegmentBinderContext> outerTableBinderContexts) {
69          ProjectionsSegment result = new ProjectionsSegment(segment.getStartIndex(), segment.getStopIndex());
70          result.setDistinctRow(segment.isDistinctRow());
71          for (ProjectionSegment each : segment.getProjections()) {
72              Multimap<CaseInsensitiveString, TableSegmentBinderContext> currentTableBinderContexts = createCurrentTableBinderContexts(binderContext, result.getProjections());
73              result.getProjections().add(bind(binderContext, boundTableSegment, currentTableBinderContexts, tableBinderContexts, outerTableBinderContexts, each));
74          }
75          return result;
76      }
77      
78      private static ProjectionSegment bind(final SQLStatementBinderContext binderContext, final TableSegment boundTableSegment,
79                                            final Multimap<CaseInsensitiveString, TableSegmentBinderContext> currentTableBinderContexts,
80                                            final Multimap<CaseInsensitiveString, TableSegmentBinderContext> tableBinderContexts,
81                                            final Multimap<CaseInsensitiveString, TableSegmentBinderContext> outerTableBinderContexts,
82                                            final ProjectionSegment projectionSegment) {
83          try {
84              return bind(projectionSegment, binderContext, boundTableSegment, tableBinderContexts, outerTableBinderContexts);
85          } catch (final ColumnNotFoundException ignored) {
86              return bind(projectionSegment, binderContext, boundTableSegment, currentTableBinderContexts, outerTableBinderContexts);
87          }
88      }
89      
90      private static ProjectionSegment bind(final ProjectionSegment projectionSegment, final SQLStatementBinderContext binderContext, final TableSegment boundTableSegment,
91                                            final Multimap<CaseInsensitiveString, TableSegmentBinderContext> tableBinderContexts,
92                                            final Multimap<CaseInsensitiveString, TableSegmentBinderContext> outerTableBinderContexts) {
93          if (projectionSegment instanceof ColumnProjectionSegment) {
94              return ColumnProjectionSegmentBinder.bind((ColumnProjectionSegment) projectionSegment, binderContext, tableBinderContexts, outerTableBinderContexts);
95          }
96          if (projectionSegment instanceof ShorthandProjectionSegment) {
97              return ShorthandProjectionSegmentBinder.bind((ShorthandProjectionSegment) projectionSegment, boundTableSegment, tableBinderContexts);
98          }
99          if (projectionSegment instanceof SubqueryProjectionSegment) {
100             Multimap<CaseInsensitiveString, TableSegmentBinderContext> newOuterTableBinderContexts = LinkedHashMultimap.create();
101             newOuterTableBinderContexts.putAll(outerTableBinderContexts);
102             newOuterTableBinderContexts.putAll(tableBinderContexts);
103             return SubqueryProjectionSegmentBinder.bind((SubqueryProjectionSegment) projectionSegment, binderContext, newOuterTableBinderContexts);
104         }
105         if (projectionSegment instanceof ExpressionProjectionSegment) {
106             ExpressionSegment boundExpressionSegment = ExpressionSegmentBinder.bind(
107                     ((ExpressionProjectionSegment) projectionSegment).getExpr(), SegmentType.PROJECTION, binderContext, tableBinderContexts, outerTableBinderContexts);
108             ExpressionProjectionSegment result = new ExpressionProjectionSegment(
109                     projectionSegment.getStartIndex(), projectionSegment.getStopIndex(), ((ExpressionProjectionSegment) projectionSegment).getText(), boundExpressionSegment);
110             ((ExpressionProjectionSegment) projectionSegment).getAliasSegment().ifPresent(result::setAlias);
111             return result;
112         }
113         if (projectionSegment instanceof AggregationDistinctProjectionSegment) {
114             return bindAggregationDistinctProjection((AggregationDistinctProjectionSegment) projectionSegment, binderContext, tableBinderContexts, outerTableBinderContexts);
115         }
116         if (projectionSegment instanceof AggregationProjectionSegment) {
117             return bindAggregationProjection((AggregationProjectionSegment) projectionSegment, binderContext, tableBinderContexts, outerTableBinderContexts);
118         }
119         // TODO support more ProjectionSegment bound
120         return projectionSegment;
121     }
122     
123     private static AggregationDistinctProjectionSegment bindAggregationDistinctProjection(final AggregationDistinctProjectionSegment aggregationDistinctSegment,
124                                                                                           final SQLStatementBinderContext binderContext,
125                                                                                           final Multimap<CaseInsensitiveString, TableSegmentBinderContext> tableBinderContexts,
126                                                                                           final Multimap<CaseInsensitiveString, TableSegmentBinderContext> outerTableBinderContexts) {
127         AggregationDistinctProjectionSegment result = new AggregationDistinctProjectionSegment(aggregationDistinctSegment.getStartIndex(), aggregationDistinctSegment.getStopIndex(),
128                 aggregationDistinctSegment.getType(), aggregationDistinctSegment.getExpression(), aggregationDistinctSegment.getDistinctInnerExpression(),
129                 aggregationDistinctSegment.getSeparator().orElse(null));
130         aggregationDistinctSegment.getParameters()
131                 .forEach(each -> result.getParameters().add(ExpressionSegmentBinder.bind(each, SegmentType.PROJECTION, binderContext, tableBinderContexts, outerTableBinderContexts)));
132         aggregationDistinctSegment.getAliasSegment().ifPresent(result::setAlias);
133         return result;
134     }
135     
136     private static AggregationProjectionSegment bindAggregationProjection(final AggregationProjectionSegment aggregationSegment, final SQLStatementBinderContext binderContext,
137                                                                           final Multimap<CaseInsensitiveString, TableSegmentBinderContext> tableBinderContexts,
138                                                                           final Multimap<CaseInsensitiveString, TableSegmentBinderContext> outerTableBinderContexts) {
139         AggregationProjectionSegment result =
140                 new AggregationProjectionSegment(aggregationSegment.getStartIndex(), aggregationSegment.getStopIndex(), aggregationSegment.getType(), aggregationSegment.getExpression(),
141                         aggregationSegment.getSeparator().orElse(null));
142         aggregationSegment.getParameters()
143                 .forEach(each -> result.getParameters().add(ExpressionSegmentBinder.bind(each, SegmentType.PROJECTION, binderContext, tableBinderContexts, outerTableBinderContexts)));
144         aggregationSegment.getAliasSegment().ifPresent(result::setAlias);
145         return result;
146     }
147     
148     private static Multimap<CaseInsensitiveString, TableSegmentBinderContext> createCurrentTableBinderContexts(final SQLStatementBinderContext binderContext,
149                                                                                                                final Collection<ProjectionSegment> projections) {
150         Multimap<CaseInsensitiveString, TableSegmentBinderContext> result = LinkedHashMultimap.create();
151         Collection<ProjectionSegment> subqueryProjections = SubqueryTableBindUtils.createSubqueryProjections(
152                 projections, new IdentifierValue(""), binderContext.getSqlStatement().getDatabaseType(), TableSourceType.TEMPORARY_TABLE);
153         result.put(new CaseInsensitiveString(""), new SimpleTableSegmentBinderContext(subqueryProjections, TableSourceType.TEMPORARY_TABLE));
154         return result;
155     }
