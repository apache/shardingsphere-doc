45  /**
46   * Shadow workflow validation service.
47   */
48  public final class ShadowWorkflowValidationService implements MCPWorkflowRuntimeHandler, MCPWorkflowApplyArtifactValidator {
49      
50      private final WorkflowValidationSupport validationSupport = new WorkflowValidationSupport();
51      
52      private final ShadowInspectionService inspectionService = new ShadowInspectionService();
53      
54      @Override
55      public ValidationReport validate(final WorkflowContextSnapshot snapshot, final MCPFeatureQueryFacade queryFacade) {
56          ValidationReport result = new ValidationReport();
57          queryFacade.checkDatabaseCapability(snapshot.getRequest().getDatabase());
58          result.setRuleValidation(validateByRequestType(snapshot, queryFacade, result));
59          result.setOverallStatus(validationSupport.resolveOverallStatus(result.getRuleValidation()));
60          return result;
61      }
62      
63      @Override
64      public List<Map<String, Object>> validate(final WorkflowContextSnapshot snapshot, final Collection<ExecutableWorkflowArtifact> artifacts) {
65          List<Map<String, Object>> result = new LinkedList<>();
66          for (ExecutableWorkflowArtifact each : artifacts) {
67              addRuleDistSQLIssues(result, snapshot, each.sql(), each.displaySql());
68          }
69          return result;
70      }
71      
72      private void addRuleDistSQLIssues(final List<Map<String, Object>> issues, final WorkflowContextSnapshot snapshot, final String sql, final String displaySql) {
73          if (!isCreateOrAlterShadowDistSQL(sql)) {
74              return;
75          }
76          if (snapshot.getRequest() instanceof ShadowRuleWorkflowRequest request) {
77              addShadowAlgorithmIssue(issues, request.getAlgorithmType(), request.getAlgorithmProperties(), displaySql);
78          }
79          if (snapshot.getRequest() instanceof ShadowDefaultAlgorithmWorkflowRequest request) {
80              addShadowAlgorithmIssue(issues, request.getAlgorithmType(), request.getAlgorithmProperties(), displaySql);
81          }
82      }
83      
84      private boolean isCreateOrAlterShadowDistSQL(final String sql) {
85          String actualSQL = sql.trim().toUpperCase(Locale.ENGLISH);
86          return actualSQL.startsWith("CREATE SHADOW RULE") || actualSQL.startsWith("ALTER SHADOW RULE")
87                  || actualSQL.startsWith("CREATE DEFAULT SHADOW ALGORITHM") || actualSQL.startsWith("ALTER DEFAULT SHADOW ALGORITHM");
88      }
89      
90      private void addShadowAlgorithmIssue(final List<Map<String, Object>> issues, final String algorithmType, final Map<String, String> properties, final String displaySql) {
91          if (algorithmType.isEmpty()) {
92              return;
93          }
94          if (!WorkflowAlgorithmUtils.isAlgorithmServiceAvailable(ShadowAlgorithm.class, algorithmType, properties)) {
95              issues.add(validationSupport.createSQLExecutabilityIssue(String.format("Generated shadow DistSQL references algorithm `%s`, but it cannot be loaded or initialized by ShadowAlgorithm SPI.",
96                      algorithmType), displaySql));
97          }
98      }
99      
100     private ValidationSection validateByRequestType(final WorkflowContextSnapshot snapshot, final MCPFeatureQueryFacade queryFacade,
101                                                     final ValidationReport validationReport) {
102         if (snapshot.getRequest() instanceof ShadowRuleWorkflowRequest) {
103             return validateRule(snapshot, queryFacade, validationReport);
104         }
105         if (snapshot.getRequest() instanceof ShadowDefaultAlgorithmWorkflowRequest) {
106             return validateDefaultAlgorithm(snapshot, queryFacade, validationReport);
107         }
108         return validateAlgorithmCleanup(snapshot, queryFacade, validationReport);
109     }
110     
111     private ValidationSection validateRule(final WorkflowContextSnapshot snapshot, final MCPFeatureQueryFacade queryFacade,
112                                            final ValidationReport validationReport) {
113         ShadowRuleWorkflowRequest request = (ShadowRuleWorkflowRequest) snapshot.getRequest();
114         List<Map<String, Object>> rules = inspectionService.queryRules(queryFacade, request.getDatabase());
115         boolean ruleExists = containsRule(rules, queryFacade, request.getDatabase(), request.getRuleName());
116         boolean dropWorkflow = WorkflowLifecycleUtils.isDropWorkflow(snapshot);
117         if (dropWorkflow == ruleExists) {
118             addMismatch(validationReport, "shadow_rule", request.getRuleName(), "Shadow rule state does not match the planned DistSQL artifact.");
119             return new ValidationSection(WorkflowLifecycle.STATUS_FAILED, rules, "Shadow rule state does not match the planned DistSQL artifact.");
120         }
121         if (!dropWorkflow && !validateRuleShape(rules, queryFacade, request, validationReport)) {
122             return new ValidationSection(WorkflowLifecycle.STATUS_FAILED, rules, "Shadow rule shape differs from the planned DistSQL artifact.");
123         }
124         return new ValidationSection(WorkflowLifecycle.STATUS_PASSED, rules, "Shadow rule state matches the planned DistSQL artifact.");
125     }
126     
127     private ValidationSection validateDefaultAlgorithm(final WorkflowContextSnapshot snapshot, final MCPFeatureQueryFacade queryFacade,
128                                                        final ValidationReport validationReport) {
129         ShadowDefaultAlgorithmWorkflowRequest request = (ShadowDefaultAlgorithmWorkflowRequest) snapshot.getRequest();
130         List<Map<String, Object>> defaultAlgorithm = inspectionService.queryDefaultAlgorithm(queryFacade, request.getDatabase());
131         boolean exists = !defaultAlgorithm.isEmpty();
132         boolean dropWorkflow = WorkflowLifecycleUtils.isDropWorkflow(snapshot);
133         if (dropWorkflow == exists) {
134             addMismatch(validationReport, "default_shadow_algorithm", request.getAlgorithmType(), "Default shadow algorithm state does not match the planned DistSQL artifact.");
135             return new ValidationSection(WorkflowLifecycle.STATUS_FAILED, defaultAlgorithm, "Default shadow algorithm state does not match the planned DistSQL artifact.");
136         }
137         if (!dropWorkflow && !matchesDefaultAlgorithm(defaultAlgorithm, request)) {
138             addMismatch(validationReport, "default_shadow_algorithm", request.getAlgorithmType(), "Default shadow algorithm type or properties differ from the planned artifact.");
139             return new ValidationSection(WorkflowLifecycle.STATUS_FAILED, defaultAlgorithm, "Default shadow algorithm configuration differs from the planned artifact.");
140         }
141         return new ValidationSection(WorkflowLifecycle.STATUS_PASSED, defaultAlgorithm, "Default shadow algorithm state matches the planned DistSQL artifact.");
142     }
143     
144     private ValidationSection validateAlgorithmCleanup(final WorkflowContextSnapshot snapshot, final MCPFeatureQueryFacade queryFacade,
145                                                        final ValidationReport validationReport) {
146         ShadowAlgorithmCleanupWorkflowRequest request = (ShadowAlgorithmCleanupWorkflowRequest) snapshot.getRequest();
147         List<Map<String, Object>> algorithms = inspectionService.queryAlgorithms(queryFacade, request.getDatabase());
148         if (containsAlgorithm(algorithms, queryFacade, request.getDatabase(), request.getAlgorithmName())) {
149             addMismatch(validationReport, "shadow_algorithm", request.getAlgorithmName(), "Shadow algorithm still exists after cleanup.");
150             return new ValidationSection(WorkflowLifecycle.STATUS_FAILED, algorithms, "Shadow algorithm still exists after cleanup.");
151         }
152         return new ValidationSection(WorkflowLifecycle.STATUS_PASSED, algorithms, "Shadow algorithm cleanup state matches the planned DistSQL artifact.");
153     }
154     
155     private boolean containsRule(final List<Map<String, Object>> rules, final MCPFeatureQueryFacade queryFacade, final String databaseName, final String ruleName) {
156         return rules.stream().anyMatch(each -> queryFacade.isSameIdentifier(
157                 databaseName, IdentifierScope.TABLE, ruleName, WorkflowRuleValueUtils.getRuleValue(each, "rule_name")));
158     }
159     
160     private boolean validateRuleShape(final List<Map<String, Object>> rules, final MCPFeatureQueryFacade queryFacade,
161                                       final ShadowRuleWorkflowRequest request, final ValidationReport validationReport) {
162         List<Map<String, Object>> matchingRules = rules.stream().filter(each -> queryFacade.isSameIdentifier(
163                 request.getDatabase(), IdentifierScope.TABLE, request.getRuleName(), WorkflowRuleValueUtils.getRuleValue(each, "rule_name"))).toList();
164         if (1 != matchingRules.size()) {
165             addMismatch(validationReport, "shadow_rule.row_count", "1", String.valueOf(matchingRules.size()),
166                     "Shadow rule row count differs from the single table and algorithm planned by this workflow.");
167             return false;
168         }
169         Map<String, Object> rule = matchingRules.getFirst();
170         if (!queryFacade.isSameIdentifier(request.getDatabase(), IdentifierScope.TABLE,
171                 request.getSourceStorageUnit(), WorkflowRuleValueUtils.getRuleValue(rule, "source_name"))) {
172             addMismatch(validationReport, "shadow_rule.source_name", request.getSourceStorageUnit(), WorkflowRuleValueUtils.getRuleValue(rule, "source_name"),
173                     "Shadow rule source storage unit differs from the planned artifact.");
174             return false;
175         }
176         if (!queryFacade.isSameIdentifier(request.getDatabase(), IdentifierScope.TABLE,
177                 request.getShadowStorageUnit(), WorkflowRuleValueUtils.getRuleValue(rule, "shadow_name"))) {
178             addMismatch(validationReport, "shadow_rule.shadow_name", request.getShadowStorageUnit(), WorkflowRuleValueUtils.getRuleValue(rule, "shadow_name"),
179                     "Shadow rule shadow storage unit differs from the planned artifact.");
180             return false;
181         }
182         if (!queryFacade.isSameIdentifier(request.getDatabase(), IdentifierScope.TABLE,
183                 request.getTableName(), WorkflowRuleValueUtils.getRuleValue(rule, "shadow_table"))) {
184             addMismatch(validationReport, "shadow_rule.shadow_table", request.getTableName(), WorkflowRuleValueUtils.getRuleValue(rule, "shadow_table"),
185                     "Shadow table differs from the planned artifact.");
186             return false;
187         }
188         String actualAlgorithmType = WorkflowRuleValueUtils.getRuleValue(rule, "algorithm_type");
189         if (!actualAlgorithmType.equalsIgnoreCase(request.getAlgorithmType())) {
190             addMismatch(validationReport, "shadow_rule.algorithm_type", request.getAlgorithmType(), actualAlgorithmType,
191                     "Shadow algorithm type differs from the planned artifact.");
192             return false;
193         }
194         Map<String, String> actualAlgorithmProperties = WorkflowAlgorithmUtils.createPropertyMap(rule.get("algorithm_props"));
195         if (!actualAlgorithmProperties.equals(request.getAlgorithmProperties())) {
196             addMismatch(validationReport, "shadow_rule.algorithm_properties", request.getAlgorithmProperties().toString(), actualAlgorithmProperties.toString(),
197                     "Shadow algorithm properties differ from the planned artifact.");
198             return false;
199         }
200         return true;
201     }
202     
203     private boolean matchesDefaultAlgorithm(final List<Map<String, Object>> rows, final ShadowDefaultAlgorithmWorkflowRequest request) {
204         return 1 == rows.size() && WorkflowRuleValueUtils.getRuleValue(rows.getFirst(), "type").equalsIgnoreCase(request.getAlgorithmType())
205                 && WorkflowAlgorithmUtils.createPropertyMap(rows.getFirst().get("props")).equals(request.getAlgorithmProperties());
206     }
207     
208     private boolean containsAlgorithm(final List<Map<String, Object>> algorithms, final MCPFeatureQueryFacade queryFacade,
209                                       final String databaseName, final String algorithmName) {
210         return algorithms.stream().anyMatch(each -> queryFacade.isSameIdentifier(databaseName, IdentifierScope.TABLE, algorithmName,
211                 WorkflowRuleValueUtils.getRuleValue(each, "shadow_algorithm_name")));
212     }
213     
214     private void addMismatch(final ValidationReport validationReport, final String field, final String expected, final String impact) {
215         addMismatch(validationReport, field, expected, "", impact);
216     }
217     
218     private void addMismatch(final ValidationReport validationReport, final String field, final String expected, final String actual, final String impact) {
219         validationReport.getMismatches().add(validationSupport.createMismatch(WorkflowIssueCode.RULE_STATE_MISMATCH, field, expected, actual,
220                 impact, "Inspect current shadow DistSQL-visible state and re-apply or re-plan the workflow."));
221     }
