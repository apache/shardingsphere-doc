46  /**
47   * Projection metadata resolver for MySQL prepared statements.
48   */
49  @NoArgsConstructor(access = AccessLevel.PRIVATE)
50  public final class MySQLProjectionMetadataResolver {
51      
52      private static final String BINARY_COLUMN_TYPE_KEYWORD = "BINARY";
53      
54      private static final String BLOB_COLUMN_TYPE_KEYWORD = "BLOB";
55      
56      private static final Collection<Integer> BINARY_TYPES = new HashSet<>(Arrays.asList(Types.BINARY, Types.VARBINARY, Types.LONGVARBINARY));
57      
58      /**
59       * Resolve projection metadata packets by JDBC metadata.
60       *
61       * @param connectionSession connection session
62       * @param preparedStatement prepared statement
63       * @param selectStatementContext select statement context
64       * @param characterSet character set
65       * @return column definition packets
66       * @throws SQLException SQL exception
67       */
68      public static Collection<MySQLPacket> resolveProjectionPackets(final ConnectionSession connectionSession, final MySQLServerPreparedStatement preparedStatement,
69                                                                     final SelectStatementContext selectStatementContext, final int characterSet) throws SQLException {
70          try (PreparedStatement actualPreparedStatement = MySQLPreparedStatementMetadataFactory.load(connectionSession, preparedStatement)) {
71              ResultSetMetaData actualResultSetMetaData = actualPreparedStatement.getMetaData();
72              if (null == actualResultSetMetaData) {
73                  return createDefaultPackets(selectStatementContext, characterSet);
74              }
75              String databaseName = selectStatementContext.getTablesContext().getDatabaseName().orElseGet(connectionSession::getCurrentDatabaseName);
76              ShardingSphereDatabase database = ProxyContext.getInstance().getContextManager().getMetaDataContexts().getMetaData().getDatabase(databaseName);
77              ShardingSphereResultSetMetaData resultSetMetaData = new ShardingSphereResultSetMetaData(actualResultSetMetaData, database, selectStatementContext);
78              QueryHeaderBuilderEngine queryHeaderBuilderEngine = new QueryHeaderBuilderEngine(database.getProtocolType());
79              int visibleColumnCount = ShardingSphereResultSetUtils.getVisibleColumnCount(selectStatementContext, actualResultSetMetaData);
80              Collection<MySQLPacket> result = new ArrayList<>(visibleColumnCount);
81              for (int columnIndex = 1; columnIndex <= visibleColumnCount; columnIndex++) {
82                  QueryHeader queryHeader = queryHeaderBuilderEngine.build(selectStatementContext, resultSetMetaData, database, columnIndex);
83                  result.add(createMySQLColumnDefinition41Packet(queryHeader, characterSet));
84              }
85              return result;
86          }
87      }
88      
89      private static Collection<MySQLPacket> createDefaultPackets(final SelectStatementContext selectStatementContext, final int characterSet) {
90          Collection<MySQLPacket> result = new ArrayList<>(selectStatementContext.getProjectionsContext().getExpandProjections().size());
91          for (int index = 0; index < selectStatementContext.getProjectionsContext().getExpandProjections().size(); index++) {
92              result.add(new MySQLColumnDefinition41Packet(characterSet, 0, "", "", "", "", "", 0, MySQLBinaryColumnType.VAR_STRING, 0, false));
93          }
94          return result;
95      }
96      
97      private static MySQLColumnDefinition41Packet createMySQLColumnDefinition41Packet(final QueryHeader queryHeader, final int characterSet) {
98          int actualCharacterSet = BINARY_TYPES.contains(queryHeader.getColumnType()) ? MySQLCharacterSets.BINARY.getId() : characterSet;
99          return new MySQLColumnDefinition41Packet(actualCharacterSet, getColumnDefinitionFlag(queryHeader), queryHeader.getSchema(), queryHeader.getTable(), queryHeader.getTable(),
100                 queryHeader.getColumnLabel(), queryHeader.getColumnName(), queryHeader.getColumnLength(), MySQLBinaryColumnType.valueOfJDBCType(queryHeader.getColumnType()),
101                 queryHeader.getDecimals(), false);
102     }
103     
104     private static int getColumnDefinitionFlag(final QueryHeader queryHeader) {
105         int result = 0;
106         boolean characterLargeObject = isCharacterLargeObject(queryHeader.getColumnType());
107         if (queryHeader.isPrimaryKey()) {
108             result += MySQLColumnDefinitionFlag.PRIMARY_KEY.getValue();
109         }
110         if (queryHeader.isNotNull()) {
111             result += MySQLColumnDefinitionFlag.NOT_NULL.getValue();
112         }
113         if (!queryHeader.isSigned() && !characterLargeObject) {
114             result += MySQLColumnDefinitionFlag.UNSIGNED.getValue();
115         }
116         if (queryHeader.isAutoIncrement()) {
117             result += MySQLColumnDefinitionFlag.AUTO_INCREMENT.getValue();
118         }
119         if (!characterLargeObject
120                 && (queryHeader.getColumnTypeName().contains(BINARY_COLUMN_TYPE_KEYWORD) || queryHeader.getColumnTypeName().contains(BLOB_COLUMN_TYPE_KEYWORD))) {
121             result += MySQLColumnDefinitionFlag.BINARY_COLLATION.getValue();
122         }
123         if (BINARY_TYPES.contains(queryHeader.getColumnType()) || characterLargeObject) {
124             result += MySQLColumnDefinitionFlag.BLOB.getValue();
125         }
126         return result;
127     }
128     
129     private static boolean isCharacterLargeObject(final int jdbcType) {
130         return Types.CLOB == jdbcType || Types.NCLOB == jdbcType;
131     }
132 }
