50  class OpenGaussSystemTableQueryExecutorFactoryTest {
51      
52      @Test
53      void assertSelectDatCompatibilityFromPgDatabase() {
54          String sql = "SELECT datcompatibility FROM pg_database WHERE datname='sharding_db'";
55          SelectStatementContext sqlStatementContext = mockSelectStatementContext("pg_catalog", "pg_database", "datcompatibility");
56          Optional<DatabaseAdminExecutor> actual = OpenGaussSystemTableQueryExecutorFactory.newInstance(sqlStatementContext, sql, Collections.emptyList());
57          assertTrue(actual.isPresent());
58          assertThat(actual.get(), isA(OpenGaussSelectDatCompatibilityExecutor.class));
59      }
60      
61      @Test
62      void assertSelectFromNotCollectedTable() {
63          String sql = "SELECT name FROM pg_type'";
64          SelectStatementContext sqlStatementContext = mockSelectStatementContext("pg_catalog", "pg_type", "name");
65          Optional<DatabaseAdminExecutor> actual = OpenGaussSystemTableQueryExecutorFactory.newInstance(sqlStatementContext, sql, Collections.emptyList());
66          assertTrue(actual.isPresent());
67          assertThat(actual.get(), isA(DatabaseMetaDataExecutor.class));
68      }
69      
70      @Test
71      void assertSelectFromPgCatalogPgDatabaseViaParserShouldBeRecognizedAsSystemTable() {
72          String sql = "SELECT d.datname AS \"Name\",\n"
73                  + "       pg_catalog.pg_get_userbyid(d.datdba) AS \"Owner\",\n"
74                  + "       pg_catalog.pg_encoding_to_char(d.encoding) AS \"Encoding\",\n"
75                  + "       d.datcollate AS \"Collate\",\n"
76                  + "       d.datctype AS \"Ctype\",\n"
77                  + "       pg_catalog.array_to_string(d.datacl, E'\\\\n') AS \"Access privileges\"\n"
78                  + "FROM pg_catalog.pg_database d\n"
79                  + "ORDER BY 1;";
80          SelectStatementContext sqlStatementContext = createSelectStatementContext(sql);
81          Optional<DatabaseAdminExecutor> actual = OpenGaussSystemTableQueryExecutorFactory.newInstance(sqlStatementContext, sql, Collections.emptyList());
82          assertTrue(actual.isPresent());
83          assertThat(actual.get(), isA(DatabaseMetaDataExecutor.class));
84      }
85      
86      private SelectStatementContext createSelectStatementContext(final String sql) {
87          DatabaseType databaseType = TypedSPILoader.getService(DatabaseType.class, "openGauss");
88          ShardingSphereSQLParserEngine parserEngine = new ShardingSphereSQLParserEngine(databaseType, new CacheOption(128, 64), new CacheOption(128, 64));
89          SQLStatement sqlStatement = parserEngine.parse(sql, false);
90          return new SelectStatementContext((SelectStatement) sqlStatement, mock(ShardingSphereMetaData.class), null, Collections.emptyList());
91      }
92      
93      private SelectStatementContext mockSelectStatementContext(final String schemaName, final String tableName, final String columnName) {
94          TableSegmentBoundInfo tableSegmentBoundInfo = mock(TableSegmentBoundInfo.class, RETURNS_DEEP_STUBS);
95          when(tableSegmentBoundInfo.getOriginalSchema().getValue()).thenReturn(schemaName);
96          TableNameSegment tableNameSegment = mock(TableNameSegment.class, RETURNS_DEEP_STUBS);
97          when(tableNameSegment.getIdentifier().getValue()).thenReturn(tableName);
98          when(tableNameSegment.getTableBoundInfo()).thenReturn(Optional.of(tableSegmentBoundInfo));
99          SimpleTableSegment tableSegment = mock(SimpleTableSegment.class, RETURNS_DEEP_STUBS);
100         when(tableSegment.getTableName()).thenReturn(tableNameSegment);
101         TablesContext tablesContext = mock(TablesContext.class, RETURNS_DEEP_STUBS);
102         when(tablesContext.getSimpleTables()).thenReturn(Collections.singletonList(tableSegment));
103         SelectStatementContext result = mock(SelectStatementContext.class, RETURNS_DEEP_STUBS);
104         when(result.getTablesContext()).thenReturn(tablesContext);
105         if (null != columnName) {
106             ColumnSegmentBoundInfo columnSegmentBoundInfo = mock(ColumnSegmentBoundInfo.class, RETURNS_DEEP_STUBS);
107             when(columnSegmentBoundInfo.getOriginalTable().getValue()).thenReturn(tableName);
108             ColumnProjection columnProjection = mock(ColumnProjection.class, RETURNS_DEEP_STUBS);
109             when(columnProjection.getName()).thenReturn(new IdentifierValue(columnName));
110             when(columnProjection.getColumnBoundInfo()).thenReturn(columnSegmentBoundInfo);
111             when(result.getProjectionsContext().getProjections()).thenReturn(Collections.singletonList(columnProjection));
112         }
113         return result;
114     }
