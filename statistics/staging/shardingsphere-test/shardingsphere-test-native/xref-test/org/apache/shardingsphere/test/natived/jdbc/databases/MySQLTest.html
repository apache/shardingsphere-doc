44  /**
45   * Unable to use `org.testcontainers:mysql:1.20.0` under GraalVM Native Image.
46   * Background comes from <a href="https://github.com/testcontainers/testcontainers-java/issues/7954">testcontainers/testcontainers-java#7954</a>.
47   */
48  @EnabledInNativeImage
49  @Testcontainers
50  class MySQLTest {
51      
52      private final String systemPropKeyPrefix = "fixture.test-native.yaml.database.mysql.";
53      
54      private final String password = "example";
55      
56      @SuppressWarnings("resource")
57      @Container
58      private final GenericContainer<?> container = new GenericContainer<>("mysql:8.0.40").withEnv("MYSQL_ROOT_PASSWORD", password).withExposedPorts(3306);
59      
60      private DataSource logicDataSource;
61      
62      private String jdbcUrlPrefix;
63      
64      private TestShardingService testShardingService;
65      
66      @BeforeEach
67      void beforeEach() {
68          assertNull(System.getProperty(systemPropKeyPrefix + "ds0.jdbc-url"));
69          assertNull(System.getProperty(systemPropKeyPrefix + "ds1.jdbc-url"));
70          assertNull(System.getProperty(systemPropKeyPrefix + "ds2.jdbc-url"));
71      }
72      
73      @AfterEach
74      void afterEach() throws SQLException {
75          ResourceUtils.closeJdbcDataSource(logicDataSource);
76          System.clearProperty(systemPropKeyPrefix + "ds0.jdbc-url");
77          System.clearProperty(systemPropKeyPrefix + "ds1.jdbc-url");
78          System.clearProperty(systemPropKeyPrefix + "ds2.jdbc-url");
79      }
80      
81      @Test
82      void assertShardingInLocalTransactions() throws SQLException {
83          jdbcUrlPrefix = "jdbc:mysql://localhost:" + container.getMappedPort(3306) + "/";
84          logicDataSource = createDataSource();
85          testShardingService = new TestShardingService(logicDataSource);
86          initEnvironment();
87          testShardingService.processSuccess();
88          testShardingService.cleanEnvironment();
89      }
90      
91      private void initEnvironment() throws SQLException {
92          testShardingService.getOrderRepository().createTableIfNotExistsInMySQL();
93          testShardingService.getOrderItemRepository().createTableIfNotExistsInMySQL();
94          testShardingService.getAddressRepository().createTableIfNotExistsInMySQL();
95          testShardingService.getOrderRepository().truncateTable();
96          testShardingService.getOrderItemRepository().truncateTable();
97          testShardingService.getAddressRepository().truncateTable();
98      }
99      
100     private Connection openConnection() throws SQLException {
101         Properties props = new Properties();
102         props.setProperty("user", "root");
103         props.setProperty("password", password);
104         return DriverManager.getConnection(jdbcUrlPrefix, props);
105     }
106     
107     @SuppressWarnings({"SqlDialectInspection", "SqlNoDataSourceInspection"})
108     private DataSource createDataSource() throws SQLException {
109         Awaitility.await().atMost(Duration.ofMinutes(1L)).ignoreExceptionsMatching(CommunicationsException.class::isInstance).until(() -> {
110             openConnection().close();
111             return true;
112         });
113         try (
114                 Connection connection = openConnection();
115                 Statement statement = connection.createStatement()) {
116             statement.executeUpdate("CREATE DATABASE demo_ds_0");
117             statement.executeUpdate("CREATE DATABASE demo_ds_1");
118             statement.executeUpdate("CREATE DATABASE demo_ds_2");
119         }
120         HikariConfig config = new HikariConfig();
121         config.setDriverClassName("org.apache.shardingsphere.driver.ShardingSphereDriver");
122         config.setJdbcUrl("jdbc:shardingsphere:classpath:test-native/yaml/jdbc/databases/mysql.yaml?placeholder-type=system_props");
123         System.setProperty(systemPropKeyPrefix + "ds0.jdbc-url", jdbcUrlPrefix + "demo_ds_0");
124         System.setProperty(systemPropKeyPrefix + "ds1.jdbc-url", jdbcUrlPrefix + "demo_ds_1");
125         System.setProperty(systemPropKeyPrefix + "ds2.jdbc-url", jdbcUrlPrefix + "demo_ds_2");
126         return new HikariDataSource(config);
127     }
128 }
