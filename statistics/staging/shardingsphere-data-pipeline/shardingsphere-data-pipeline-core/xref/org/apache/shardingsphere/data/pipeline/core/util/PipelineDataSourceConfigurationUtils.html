20  import lombok.AccessLevel;
21  import lombok.NoArgsConstructor;
22  import lombok.extern.slf4j.Slf4j;
23  import org.apache.shardingsphere.data.pipeline.api.PipelineDataSourceConfiguration;
24  import org.apache.shardingsphere.data.pipeline.api.type.ShardingSpherePipelineDataSourceConfiguration;
25  import org.apache.shardingsphere.data.pipeline.api.type.StandardPipelineDataSourceConfiguration;
26  import org.apache.shardingsphere.infra.datasource.pool.props.domain.DataSourcePoolProperties;
27  import org.apache.shardingsphere.infra.metadata.database.resource.unit.StorageUnit;
28  import org.apache.shardingsphere.infra.yaml.config.swapper.resource.YamlDataSourceConfigurationSwapper;
29  
30  import java.util.Map;
31  import java.util.Map.Entry;
32  import java.util.Objects;
33  
34  /**
35   * Utility class for pipeline data source configuration.
36   */
37  @NoArgsConstructor(access = AccessLevel.NONE)
38  @Slf4j
39  public final class PipelineDataSourceConfigurationUtils {
40      
41      /**
42       * Transform the given pipeline data source properties.
43       *
44       * @param jobId job id
45       * @param pipelineDataSourceConfig the pipeline data source configuration to transform
46       * @param storageUnits current storage units
47       * @return transformed pipeline data source configuration
48       */
49      public static PipelineDataSourceConfiguration transformPipelineDataSourceConfiguration(final String jobId, final PipelineDataSourceConfiguration pipelineDataSourceConfig,
50                                                                                             final Map<String, StorageUnit> storageUnits) {
51          if (null == storageUnits || storageUnits.isEmpty()) {
52              return pipelineDataSourceConfig;
53          }
54          if (pipelineDataSourceConfig instanceof ShardingSpherePipelineDataSourceConfiguration) {
55              return transformPipelineDataSourceConfiguration(jobId, (ShardingSpherePipelineDataSourceConfiguration) pipelineDataSourceConfig, storageUnits);
56          }
57          return pipelineDataSourceConfig;
58      }
59      
60      /**
61       * Transform the given pipeline data source properties.
62       *
63       * @param jobId job id
64       * @param dataSourceName data source name
65       * @param pipelineDataSourceConfig the pipeline data source configuration to transform
66       * @param storageUnits current storage units
67       * @return transformed pipeline data source configuration
68       */
69      public static PipelineDataSourceConfiguration transformPipelineDataSourceConfiguration(final String jobId, final String dataSourceName,
70                                                                                             final PipelineDataSourceConfiguration pipelineDataSourceConfig,
71                                                                                             final Map<String, StorageUnit> storageUnits) {
72          if (null == storageUnits || storageUnits.isEmpty()) {
73              return pipelineDataSourceConfig;
74          }
75          if (pipelineDataSourceConfig instanceof StandardPipelineDataSourceConfiguration) {
76              StorageUnit storageUnit = storageUnits.get(dataSourceName);
77              return null == storageUnit ? pipelineDataSourceConfig
78                      : transformStandardPipelineDataSourceConfiguration(jobId, dataSourceName, (StandardPipelineDataSourceConfiguration) pipelineDataSourceConfig, storageUnit);
79          }
80          return transformPipelineDataSourceConfiguration(jobId, pipelineDataSourceConfig, storageUnits);
81      }
82      
83      /**
84       * Transform the given pipeline data source properties.
85       *
86       * @param jobId job id
87       * @param dataSourceName data source name
88       * @param pipelineDataSourceConfig the pipeline data source configuration to transform
89       * @param dataSourcePoolProps current data source pool properties
90       * @return transformed pipeline data source configuration
91       */
92      public static PipelineDataSourceConfiguration transformPipelineDataSourceConfiguration(final String jobId, final String dataSourceName,
93                                                                                             final PipelineDataSourceConfiguration pipelineDataSourceConfig,
94                                                                                             final DataSourcePoolProperties dataSourcePoolProps) {
95          return pipelineDataSourceConfig instanceof StandardPipelineDataSourceConfiguration
96                  ? transformStandardPipelineDataSourceConfiguration(jobId, dataSourceName, (StandardPipelineDataSourceConfiguration) pipelineDataSourceConfig, dataSourcePoolProps)
97                  : pipelineDataSourceConfig;
98      }
99      
100     /**
101      * Transform the given ShardingSphere pipeline data source properties.
102      *
103      * @param jobId job id
104      * @param pipelineDataSourceConfig the pipeline data source configuration to transform
105      * @param storageUnits current storage units
106      * @return transformed pipeline data source configuration
107      */
108     public static ShardingSpherePipelineDataSourceConfiguration transformPipelineDataSourceConfiguration(final String jobId,
109                                                                                                          final ShardingSpherePipelineDataSourceConfiguration pipelineDataSourceConfig,
110                                                                                                          final Map<String, StorageUnit> storageUnits) {
111         if (null == storageUnits || storageUnits.isEmpty()) {
112             return pipelineDataSourceConfig;
113         }
114         for (Entry<String, DataSourcePoolProperties> entry : pipelineDataSourceConfig.getDataSourcePoolPropertiesMap().entrySet()) {
115             StorageUnit storageUnit = storageUnits.get(entry.getKey());
116             if (null == storageUnit) {
117                 continue;
118             }
119             DataSourcePoolProperties dataSourcePoolProps = entry.getValue();
120             Map<String, Object> jobDataSourceProps = dataSourcePoolProps.getAllLocalProperties();
121             Map<String, Object> storageUnitStandardProps = storageUnit.getDataSourcePoolProperties().getPoolPropertySynonyms().getStandardProperties();
122             logTransformPoolSize(jobId, entry.getKey(), jobDataSourceProps, storageUnitStandardProps);
123             transformPoolSize(jobDataSourceProps, storageUnitStandardProps);
124             entry.setValue(new DataSourcePoolProperties(dataSourcePoolProps.getPoolClassName(), jobDataSourceProps));
125         }
126         return pipelineDataSourceConfig;
127     }
128     
129     private static PipelineDataSourceConfiguration transformStandardPipelineDataSourceConfiguration(final String jobId, final String dataSourceName,
130                                                                                                     final StandardPipelineDataSourceConfiguration pipelineDataSourceConfig,
131                                                                                                     final StorageUnit storageUnit) {
132         return transformStandardPipelineDataSourceConfiguration(jobId, dataSourceName, pipelineDataSourceConfig, storageUnit.getDataSourcePoolProperties());
133     }
134     
135     private static PipelineDataSourceConfiguration transformStandardPipelineDataSourceConfiguration(final String jobId, final String dataSourceName,
136                                                                                                     final StandardPipelineDataSourceConfiguration pipelineDataSourceConfig,
137                                                                                                     final DataSourcePoolProperties dataSourcePoolProps) {
138         DataSourcePoolProperties jobDataSourcePoolProps = (DataSourcePoolProperties) pipelineDataSourceConfig.getDataSourceConfiguration();
139         Map<String, Object> jobStandardProps = jobDataSourcePoolProps.getPoolPropertySynonyms().getStandardProperties();
140         Map<String, Object> currentStandardProps = dataSourcePoolProps.getPoolPropertySynonyms().getStandardProperties();
141         if (!isPoolSizeChanged(jobStandardProps, currentStandardProps)) {
142             return pipelineDataSourceConfig;
143         }
144         logTransformPoolSize(jobId, dataSourceName, jobStandardProps, currentStandardProps);
145         return new StandardPipelineDataSourceConfiguration(new YamlDataSourceConfigurationSwapper().swapToMap(dataSourcePoolProps));
146     }
147     
148     private static boolean isPoolSizeChanged(final Map<String, Object> jobDataSourceProps, final Map<String, Object> storageUnitStandardProps) {
149         return isPoolSizeChanged("maxPoolSize", jobDataSourceProps, storageUnitStandardProps)
150                 || isPoolSizeChanged("maximumPoolSize", jobDataSourceProps, storageUnitStandardProps);
151     }
152     
153     private static boolean isPoolSizeChanged(final String key, final Map<String, Object> jobDataSourceProps, final Map<String, Object> storageUnitStandardProps) {
154         return storageUnitStandardProps.containsKey(key) && !Objects.equals(String.valueOf(jobDataSourceProps.get(key)), String.valueOf(storageUnitStandardProps.get(key)));
155     }
156     
157     private static void logTransformPoolSize(final String jobId, final String dataSourceName, final Map<String, Object> jobDataSourceProps, final Map<String, Object> storageUnitStandardProps) {
158         if (storageUnitStandardProps.containsKey("maxPoolSize")) {
159             log.info("Transform maxPoolSize from '{}' to '{}' for {} data source: {}",
160                     jobDataSourceProps.get("maxPoolSize"), storageUnitStandardProps.get("maxPoolSize"), jobId, dataSourceName);
161         }
162         if (storageUnitStandardProps.containsKey("maximumPoolSize")) {
163             log.info("Transform maximumPoolSize from '{}' to '{}' for {} data source: {}",
164                     jobDataSourceProps.get("maximumPoolSize"), storageUnitStandardProps.get("maximumPoolSize"), jobId, dataSourceName);
165         }
166     }
167     
168     private static void transformPoolSize(final Map<String, Object> jobDataSourceProps, final Map<String, Object> storageUnitStandardProps) {
169         if (storageUnitStandardProps.containsKey("maxPoolSize")) {
170             jobDataSourceProps.put("maxPoolSize", storageUnitStandardProps.get("maxPoolSize"));
171         }
172         if (storageUnitStandardProps.containsKey("maximumPoolSize")) {
173             jobDataSourceProps.put("maximumPoolSize", storageUnitStandardProps.get("maximumPoolSize"));
174         }
175     }
