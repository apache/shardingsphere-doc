44  /**
45   * Hive meta data loader.
46   * As of the HiveServer2 of apache/hive 4.0.1, the table `INFORMATION_SCHEMA.INDEXES` does not exist,
47   * and `INFORMATION_SCHEMA.COLUMNS` does not have a column `IS_VISIBLE`.
48   * The current implementation does not record the table's index, primary keys, generated info, or column visibility.
49   */
50  public final class HiveMetaDataLoader implements DialectMetaDataLoader {
51      
52      private static final String VIEW_META_DATA_SQL = "SELECT TABLE_NAME FROM INFORMATION_SCHEMA.TABLES"
53              + " WHERE TABLE_CATALOG=? AND TABLE_SCHEMA=? AND TABLE_TYPE='VIEW' AND UPPER(TABLE_NAME) IN (%s)";
54      
55      @Override
56      public Collection<SchemaMetaData> load(final MetaDataLoaderMaterial material) throws SQLException {
57          Collection<TableMetaData> tableMetaData = new LinkedList<>();
58          if (loadInformationSchemaFlag(material)) {
59              try (Connection connection = material.getDataSource().getConnection()) {
60                  String schemaName = connection.getSchema();
61                  Map<String, Collection<ColumnMetaData>> columnMetaDataMap = loadColumnMetaDataMap(connection, schemaName, material.getActualTableNames());
62                  Collection<String> viewNames = columnMetaDataMap.isEmpty() ? Collections.emptySet() : loadViewNames(connection, schemaName, columnMetaDataMap.keySet());
63                  for (Entry<String, Collection<ColumnMetaData>> entry : columnMetaDataMap.entrySet()) {
64                      tableMetaData.add(new TableMetaData(
65                              entry.getKey(), entry.getValue(), Collections.emptyList(), Collections.emptyList(), viewNames.contains(entry.getKey()) ? TableType.VIEW : TableType.TABLE));
66                  }
67              }
68              return Collections.singleton(new SchemaMetaData(material.getDefaultSchemaName(), tableMetaData));
69          }
70          for (String each : material.getActualTableNames()) {
71              TableMetaDataLoader.loadNormalized(material.getDataSource(), each, material.getStorageType()).ifPresent(tableMetaData::add);
72          }
73          return Collections.singleton(new SchemaMetaData(material.getDefaultSchemaName(), tableMetaData));
74      }
75      
76      private boolean loadInformationSchemaFlag(final MetaDataLoaderMaterial material) throws SQLException {
77          String sql = "SHOW DATABASES LIKE 'INFORMATION_SCHEMA'";
78          try (
79                  Connection connection = material.getDataSource().getConnection();
80                  Statement statement = connection.createStatement();
81                  ResultSet resultSet = statement.executeQuery(sql)) {
82              return resultSet.next();
83          }
84      }
85      
86      @SuppressWarnings("CollectionWithoutInitialCapacity")
87      private Map<String, Collection<ColumnMetaData>> loadColumnMetaDataMap(final Connection connection, final String schemaName, final Collection<String> tables) throws SQLException {
88          Map<String, Collection<ColumnMetaData>> result = new HashMap<>();
89          try (PreparedStatement preparedStatement = connection.prepareStatement(getTableMetaDataSQL(tables))) {
90              preparedStatement.setString(1, "default");
91              preparedStatement.setString(2, schemaName);
92              try (ResultSet resultSet = preparedStatement.executeQuery()) {
93                  while (resultSet.next()) {
94                      String tableName = resultSet.getString("TABLE_NAME");
95                      ColumnMetaData columnMetaData = loadColumnMetaData(resultSet);
96                      if (!result.containsKey(tableName)) {
97                          result.put(tableName, new LinkedList<>());
98                      }
99                      result.get(tableName).add(columnMetaData);
100                 }
101             }
102         }
103         return result;
104     }
105     
106     private String getTableMetaDataSQL(final Collection<String> tables) {
107         if (tables.isEmpty()) {
108             return "SELECT TABLE_CATALOG,TABLE_NAME,COLUMN_NAME,DATA_TYPE,ORDINAL_POSITION,IS_NULLABLE FROM INFORMATION_SCHEMA.COLUMNS"
109                     + " WHERE TABLE_CATALOG=? AND TABLE_SCHEMA=? ORDER BY ORDINAL_POSITION";
110         }
111         String tableNames = tables.stream().map(each -> String.format("'%s'", each).toUpperCase()).collect(Collectors.joining(","));
112         return String.format("SELECT TABLE_CATALOG,TABLE_NAME,COLUMN_NAME,DATA_TYPE,ORDINAL_POSITION,IS_NULLABLE FROM INFORMATION_SCHEMA.COLUMNS"
113                 + " WHERE TABLE_CATALOG=? AND TABLE_SCHEMA=? AND UPPER(TABLE_NAME) IN (%s) ORDER BY ORDINAL_POSITION", tableNames);
114     }
115     
116     private ColumnMetaData loadColumnMetaData(final ResultSet resultSet) throws SQLException {
117         String columnName = resultSet.getString("COLUMN_NAME");
118         String dataType = resultSet.getString("DATA_TYPE");
119         boolean isNullable = "YES".equals(resultSet.getString("IS_NULLABLE"));
120         return new ColumnMetaData(columnName, DataTypeRegistry.getDataType(getDatabaseType(), dataType).orElse(Types.OTHER), Boolean.FALSE, Boolean.FALSE,
121                 isStringDataType(dataType), Boolean.TRUE, false, isNullable);
122     }
123     
124     private boolean isStringDataType(final String dataType) {
125         String normalizedDataType = dataType.toLowerCase(Locale.ENGLISH);
126         return "string".equals(normalizedDataType) || normalizedDataType.startsWith("varchar") || normalizedDataType.startsWith("char");
127     }
128     
129     private Collection<String> loadViewNames(final Connection connection, final String schemaName, final Collection<String> tableNames) throws SQLException {
130         Collection<String> result = new LinkedList<>();
131         try (PreparedStatement preparedStatement = connection.prepareStatement(getViewMetaDataSQL(tableNames))) {
132             preparedStatement.setString(1, "default");
133             preparedStatement.setString(2, schemaName);
134             try (ResultSet resultSet = preparedStatement.executeQuery()) {
135                 while (resultSet.next()) {
136                     result.add(resultSet.getString("TABLE_NAME"));
137                 }
138             }
139         }
140         return result;
141     }
142     
143     private String getViewMetaDataSQL(final Collection<String> tableNames) {
144         return String.format(VIEW_META_DATA_SQL, tableNames.stream().map(each -> String.format("'%s'", each).toUpperCase(Locale.ENGLISH)).collect(Collectors.joining(",")));
145     }
146     
147     @Override
148     public String getDatabaseType() {
149         return "Hive";
150     }
151 }
