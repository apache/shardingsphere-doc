18  package org.apache.shardingsphere.proxy.frontend.firebird.command.query.batch;
19  
20  import lombok.AccessLevel;
21  import lombok.NoArgsConstructor;
22  import org.apache.shardingsphere.database.exception.firebird.exception.protocol.BatchTooBigException;
23  import org.apache.shardingsphere.database.exception.firebird.exception.protocol.InvalidBatchHandleException;
24  import org.apache.shardingsphere.database.protocol.firebird.packet.command.query.batch.FirebirdBatchColumnDescriptor;
25  import org.apache.shardingsphere.database.protocol.firebird.packet.command.query.batch.FirebirdBatchMessageCommandPacket;
26  import org.apache.shardingsphere.database.protocol.firebird.packet.command.query.batch.FirebirdBatchRegistry;
27  import org.apache.shardingsphere.database.protocol.firebird.packet.command.query.batch.FirebirdBatchStatement;
28  import org.apache.shardingsphere.infra.exception.ShardingSpherePreconditions;
29  
30  import java.util.List;
31  
32  /**
33   * Batch statement manager for Firebird.
34   */
35  @NoArgsConstructor(access = AccessLevel.NONE)
36  public final class FirebirdBatchStatementManager {
37      
38      private static final FirebirdBatchStatementManager INSTANCE = new FirebirdBatchStatementManager();
39      
40      /**
41       * Get batch statement manager instance.
42       *
43       * @return batch statement manager instance
44       */
45      public static FirebirdBatchStatementManager getInstance() {
46          return INSTANCE;
47      }
48      
49      /**
50       * Register connection.
51       *
52       * @param connectionId connection ID
53       */
54      public void registerConnection(final int connectionId) {
55          FirebirdBatchRegistry.getInstance().registerConnection(connectionId);
56      }
57      
58      /**
59       * Unregister connection and its batch statements.
60       *
61       * @param connectionId connection ID
62       */
63      public void unregisterConnection(final int connectionId) {
64          FirebirdBatchRegistry.getInstance().unregisterConnection(connectionId);
65      }
66      
67      /**
68       * Get batch statement.
69       *
70       * @param connectionId connection ID
71       * @param statementId statement ID
72       * @return batch statement, or {@code null} if absent
73       */
74      public FirebirdBatchStatement getBatchStatement(final int connectionId, final int statementId) {
75          return FirebirdBatchRegistry.getInstance().getBatchStatement(connectionId, statementId);
76      }
77      
78      /**
79       * Register batch statement.
80       *
81       * @param connectionId connection ID
82       * @param statementId statement ID
83       * @param columnDescriptors column descriptors
84       * @param bufferSize buffer size
85       * @param recordCounts whether record counts are requested
86       * @param multiError whether multiple errors are requested
87       */
88      public void registerBatchStatement(final int connectionId, final int statementId, final List<FirebirdBatchColumnDescriptor> columnDescriptors,
89                                         final long bufferSize, final boolean recordCounts, final boolean multiError) {
90          FirebirdBatchRegistry.getInstance().registerBatchStatement(connectionId, statementId, new FirebirdBatchStatement(statementId, columnDescriptors, bufferSize, recordCounts, multiError));
91      }
92      
93      /**
94       * Append batch message.
95       *
96       * @param connectionId connection ID
97       * @param packet batch message packet
98       * @throws InvalidBatchHandleException if batch statement does not exist
99       * @throws BatchTooBigException if appending the message exceeds the batch buffer size
100      */
101     public void appendBatchMessage(final int connectionId, final FirebirdBatchMessageCommandPacket packet) {
102         FirebirdBatchStatement batchStatement = getBatchStatement(connectionId, packet.getStatementHandle());
103         ShardingSpherePreconditions.checkNotNull(batchStatement, () -> new InvalidBatchHandleException(packet.getStatementHandle()));
104         int dataLength = packet.getDataLength();
105         ShardingSpherePreconditions.checkState(batchStatement.getAccumulatedSize() + dataLength <= batchStatement.getBufferSize(),
106                 () -> new BatchTooBigException(packet.getStatementHandle(), batchStatement.getAccumulatedSize(), dataLength, batchStatement.getBufferSize()));
107         for (List<Object> each : packet.readParameterValues(batchStatement.getColumnDescriptors())) {
108             batchStatement.addParameterValues(each);
109         }
110         batchStatement.addSize(dataLength);
111     }
112     
113     /**
114      * Reset batch statement.
115      *
116      * @param batchStatement batch statement
117      */
118     public void resetBatchStatement(final FirebirdBatchStatement batchStatement) {
119         batchStatement.reset();
120     }
121     
122     /**
123      * Unregister batch statement.
124      *
125      * @param connectionId connection ID
126      * @param statementId statement ID
127      */
128     public void unregisterBatchStatement(final int connectionId, final int statementId) {
129         FirebirdBatchRegistry.getInstance().unregisterBatchStatement(connectionId, statementId);
130     }
131 }
