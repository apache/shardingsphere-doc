242     @Override
243     public final ASTNode visitParameterMarker(final ParameterMarkerContext ctx) {
244         return new ParameterMarkerValue(getParameterMarkerIndex(ctx), ParameterMarkerType.QUESTION);
245     }
246     
247     private int getParameterMarkerIndex(final ParameterMarkerContext ctx) {
248         int startIndex = ctx.getStart().getStartIndex();
249         if (startIndex <= 0) {
250             return 0;
251         }
252         String sql = ctx.getStart().getInputStream().getText(new Interval(0, startIndex - 1));
253         return countParameterMarkers(sql);
254     }
255     
256     private int countParameterMarkers(final String sql) {
257         ParameterMarkerScanState state = new ParameterMarkerScanState();
258         while (state.index < sql.length()) {
259             if (advanceInLineComment(sql, state)) {
260                 continue;
261             }
262             if (advanceInBlockComment(sql, state)) {
263                 continue;
264             }
265             if (advanceInAlternativeQuotedText(sql, state)) {
266                 continue;
267             }
268             if (advanceInOrToggleStringLiteral(sql, state)) {
269                 continue;
270             }
271             if (enterLineOrBlockComment(sql, state)) {
272                 continue;
273             }
274             if (!state.inStringLiteral && sql.charAt(state.index) == '?') {
275                 state.result++;
276             }
277             state.index++;
278         }
279         return state.result;
280     }
281     
282     private boolean advanceInLineComment(final String sql, final ParameterMarkerScanState state) {
283         if (!state.inLineComment) {
284             return false;
285         }
286         char ch = sql.charAt(state.index);
287         if ('\n' == ch || '\r' == ch) {
288             state.inLineComment = false;
289         }
290         state.index++;
291         return true;
292     }
293     
294     private boolean advanceInBlockComment(final String sql, final ParameterMarkerScanState state) {
295         if (!state.inBlockComment) {
296             return false;
297         }
298         char ch = sql.charAt(state.index);
299         if ('*' == ch && state.index + 1 < sql.length() && '/' == sql.charAt(state.index + 1)) {
300             state.inBlockComment = false;
301             state.index += 2;
302         } else {
303             state.index++;
304         }
305         return true;
306     }
307     
308     private boolean advanceInAlternativeQuotedText(final String sql, final ParameterMarkerScanState state) {
309         if (state.inStringLiteral) {
310             return false;
311         }
312         int quoteIndicatorIndex = isNationalOrUnicodeLiteralIndicator(sql.charAt(state.index)) ? state.index + 1 : state.index;
313         if (quoteIndicatorIndex + 2 >= sql.length() || !isAlternativeQuoteIndicator(sql.charAt(quoteIndicatorIndex)) || '\'' != sql.charAt(quoteIndicatorIndex + 1)) {
314             return false;
315         }
316         char quoteDelimiter = sql.charAt(quoteIndicatorIndex + 2);
317         if (isInvalidAlternativeQuoteDelimiter(quoteDelimiter)) {
318             return false;
319         }
320         char closeDelimiter = getAlternativeQuoteCloseDelimiter(quoteDelimiter);
321         for (int index = quoteIndicatorIndex + 3; index + 1 < sql.length(); index++) {
322             if (closeDelimiter == sql.charAt(index) && '\'' == sql.charAt(index + 1)) {
323                 state.index = index + 2;
324                 return true;
325             }
326         }
327         return false;
328     }
329     
330     private boolean isAlternativeQuoteIndicator(final char ch) {
331         return 'q' == ch || 'Q' == ch;
332     }
333     
334     private boolean isNationalOrUnicodeLiteralIndicator(final char ch) {
335         return 'n' == ch || 'N' == ch || 'u' == ch || 'U' == ch;
336     }
337     
338     private boolean isInvalidAlternativeQuoteDelimiter(final char quoteDelimiter) {
339         return ' ' == quoteDelimiter || '\t' == quoteDelimiter || '\r' == quoteDelimiter || '\n' == quoteDelimiter;
340     }
341     
342     private char getAlternativeQuoteCloseDelimiter(final char quoteDelimiter) {
343         switch (quoteDelimiter) {
344             case '[':
345                 return ']';
346             case '{':
347                 return '}';
348             case '(':
349                 return ')';
350             case '<':
351                 return '>';
352             default:
353                 return quoteDelimiter;
354         }
355     }
356     
357     private boolean advanceInOrToggleStringLiteral(final String sql, final ParameterMarkerScanState state) {
358         if (sql.charAt(state.index) != '\'') {
359             return false;
360         }
361         if (state.inStringLiteral && state.index + 1 < sql.length() && '\'' == sql.charAt(state.index + 1)) {
362             state.index += 2;
363         } else {
364             state.inStringLiteral = !state.inStringLiteral;
365             state.index++;
366         }
367         return true;
368     }
369     
370     private boolean enterLineOrBlockComment(final String sql, final ParameterMarkerScanState state) {
371         if (state.inStringLiteral) {
372             return false;
373         }
374         char ch = sql.charAt(state.index);
375         if ('-' == ch && state.index + 1 < sql.length() && '-' == sql.charAt(state.index + 1)) {
376             state.inLineComment = true;
377             state.index += 2;
378             return true;
379         }
380         if ('/' == ch && state.index + 1 < sql.length() && '*' == sql.charAt(state.index + 1)) {
381             state.inBlockComment = true;
382             state.index += 2;
383             return true;
384         }
385         return false;
386     }

388     @Override

391             return visit(ctx.stringLiterals());
392         }
393         if (null != ctx.numberLiterals()) {
394             return visit(ctx.numberLiterals());
395         }
396         if (null != ctx.hexadecimalLiterals()) {

743     private InExpression createInSegment(final PredicateContext ctx) {
744         ExpressionSegment left = createInExpressionLeft(ctx);
745         ExpressionSegment right;
746         if (null == ctx.subquery()) {

749             } else {

751                 for (ExprContext each : ctx.expr()) {
752                     listExpression.getItems().add((ExpressionSegment) visit(each));

754                 right = listExpression;
755             }
756         } else {
757             right = new SubqueryExpressionSegment(
758                     new SubquerySegment(ctx.subquery().start.getStartIndex(), ctx.subquery().stop.getStopIndex(), (SelectStatement) visit(ctx.subquery()), getOriginalText(ctx.subquery())));
759         }
760         boolean not = null != ctx.NOT();

815     private ASTNode createExpressionSegment(final ASTNode astNode, final ParserRuleContext context) {
816         if (astNode instanceof StringLiteralValue) {
817             return new LiteralExpressionSegment(context.start.getStartIndex(), context.stop.getStopIndex(), ((StringLiteralValue) astNode).getValue());
818         }
819         if (astNode instanceof NumberLiteralValue) {
820             return new LiteralExpressionSegment(context.start.getStartIndex(), context.stop.getStopIndex(), ((NumberLiteralValue) astNode).getValue());
821         }
822         if (astNode instanceof BooleanLiteralValue) {
823             return new LiteralExpressionSegment(context.start.getStartIndex(), context.stop.getStopIndex(), ((BooleanLiteralValue) astNode).getValue());
824         }
825         if (astNode instanceof ParameterMarkerValue) {
826             return createParameterMarkerExpressionSegment((ParameterMarkerValue) astNode, context.start.getStartIndex(), context.stop.getStopIndex());
827         }
828         if (astNode instanceof SubquerySegment) {
829             return new SubqueryExpressionSegment((SubquerySegment) astNode);
830         }
831         if (astNode instanceof OtherLiteralValue) {
832             return new CommonExpressionSegment(context.getStart().getStartIndex(), context.getStop().getStopIndex(), context.getText(), Collections.emptyList());
833         }
834         return astNode;
835     }

874                 result.getItems().add((ExpressionSegment) visit(each));
875             }

877         }
878         return visitRemainSimpleExpr(ctx, startIndex, stopIndex);
879     }

904     private ASTNode visitRemainSimpleExpr(final SimpleExprContext ctx, final int startIndex, final int stopIndex) {
905         if (null != ctx.OR_()) {
906             ExpressionSegment left = (ExpressionSegment) visit(ctx.simpleExpr(0));
907             ExpressionSegment right = (ExpressionSegment) visit(ctx.simpleExpr(1));
908             String text = ctx.start.getInputStream().getText(new Interval(ctx.start.getStartIndex(), ctx.stop.getStopIndex()));
909             return new BinaryOperationExpression(ctx.start.getStartIndex(), ctx.stop.getStopIndex(), left, right, ctx.OR_().getText(), text);
910         }
911         if (null != ctx.caseExpression()) {
912             return visit(ctx.caseExpression());
913         }
914         if (null != ctx.TILDE_()) {
915             return new UnaryOperationExpression(ctx.start.getStartIndex(), ctx.stop.getStopIndex(), (ExpressionSegment) visit(ctx.simpleExpr(0)), "~", ctx.getText());
916         }
917         if (null != ctx.PLUS_()) {

1687         Collection<String> suffixKeywords = new LinkedList<>();
1688         for (int i = 0; i < ctx.getChildCount(); i++) {
1689             ParseTree child = ctx.getChild(i);
1690             if (child instanceof TerminalNode && OracleStatementParser.LP_ != ((TerminalNode) child).getSymbol().getType()
1691                     && OracleStatementParser.RP_ != ((TerminalNode) child).getSymbol().getType()) {
1692                 suffixKeywords.add(child.getText());

1696         if (null == ctx.fractionalSecondPrecision()) {
