42  /**
43   * Validator for workflow planning context.
44   */
45  public final class WorkflowPlanningContextValidator {
46      
47      /**
48       * Ensure workflow identifiers can be rendered into reviewable DistSQL.
49       *
50       * @param fieldName field name for issue details
51       * @param identifiers identifiers to check
52       * @param snapshot workflow snapshot
53       * @param issueStage issue stage
54       * @return whether all identifiers are supported
55       */
56      public boolean ensureSupportedIdentifiers(final String fieldName, final Collection<String> identifiers, final WorkflowContextSnapshot snapshot, final String issueStage) {
57          return ensureIdentifiers(fieldName, identifiers, snapshot, issueStage, false);
58      }
59      
60      /**
61       * Ensure optional workflow identifiers can be rendered into reviewable DistSQL when present.
62       *
63       * @param fieldName field name for issue details
64       * @param identifiers identifiers to check
65       * @param snapshot workflow snapshot
66       * @param issueStage issue stage
67       * @return whether all present identifiers are supported
68       */
69      public boolean ensureOptionalSupportedIdentifiers(final String fieldName, final Collection<String> identifiers, final WorkflowContextSnapshot snapshot, final String issueStage) {
70          return ensureIdentifiers(fieldName, identifiers, snapshot, issueStage, true);
71      }
72      
73      /**
74       * Ensure workflow planning context is complete and valid.
75       *
76       * @param metadataQueryFacade metadata query facade
77       * @param queryFacade query facade for database capability resolution
78       * @param request workflow request
79       * @param clarifiedIntent clarified intent
80       * @param snapshot workflow snapshot
81       * @return whether planning context is ready
82       * @throws DatabaseCapabilityNotFoundException when database profile or capability does not exist
83       */
84      public boolean ensurePlanningContext(final MCPMetadataQueryFacade metadataQueryFacade, final MCPFeatureQueryFacade queryFacade, final WorkflowRequest request,
85                                           final ClarifiedIntent clarifiedIntent, final WorkflowContextSnapshot snapshot) {
86          if (isEmptyIdentifier(request.getDatabase())) {
87              clarifiedIntent.getClarificationMessages().add("Please provide logical database first.");
88              snapshot.getIssues().add(new WorkflowIssue(WorkflowIssueCode.DATABASE_REQUIRED, "error", WorkflowLifecycle.STEP_INTAKING,
89                      "Database is required before planning.", "Provide the logical database name.", true, Map.of()));
90              snapshot.setStatus(WorkflowLifecycle.STATUS_CLARIFYING);
91              return false;
92          }
93          if (!ensureSupportedIdentifiers(WorkflowFieldNames.DATABASE, List.of(request.getDatabase()), snapshot, WorkflowLifecycle.STEP_DISCOVERING)
94                  || !ensureSupportedIdentifiers(WorkflowFieldNames.TABLE, List.of(request.getTable()), snapshot, WorkflowLifecycle.STEP_DISCOVERING)
95                  || !ensureSupportedIdentifiers(WorkflowFieldNames.COLUMN, List.of(request.getColumn()), snapshot, WorkflowLifecycle.STEP_DISCOVERING)) {
96              snapshot.setStatus(WorkflowLifecycle.STATUS_FAILED);
97              return false;
98          }
99          String databaseName = WorkflowSQLUtils.normalizeIdentifier(request.getDatabase());
100         metadataQueryFacade.queryDatabase(databaseName).orElseThrow(DatabaseCapabilityNotFoundException::new);
101         queryFacade.checkDatabaseCapability(databaseName);
102         List<ShardingSphereSchema> schemas = isEmptyIdentifier(request.getSchema()) ? metadataQueryFacade.querySchemas(databaseName) : List.of();
103         request.setSchema(resolveSchema(schemas, request, clarifiedIntent, queryFacade, databaseName));
104         if (!ensureSupportedIdentifiers(WorkflowFieldNames.SCHEMA, List.of(request.getSchema()), snapshot, WorkflowLifecycle.STEP_DISCOVERING)) {
105             snapshot.setStatus(WorkflowLifecycle.STATUS_FAILED);
106             return false;
107         }
108         addMissingQuestions(request, clarifiedIntent, snapshot);
109         if (isEmptyIdentifier(request.getSchema()) || isEmptyIdentifier(request.getTable()) || isEmptyIdentifier(request.getColumn())) {
110             snapshot.setStatus(WorkflowLifecycle.STATUS_CLARIFYING);
111             return false;
112         }
113         if (schemas.isEmpty()) {
114             schemas = metadataQueryFacade.querySchemas(databaseName);
115         }
116         Optional<ShardingSphereSchema> schema = findSchema(schemas, request, queryFacade, databaseName);
117         Optional<ShardingSphereTable> table = findTable(schema, request, queryFacade, databaseName);
118         if (!ensureTableExists(table, request, snapshot)) {
119             snapshot.setStatus(WorkflowLifecycle.STATUS_FAILED);
120             return false;
121         }
122         if (!ensureColumnExists(metadataQueryFacade.queryTableColumns(databaseName, schema.orElseThrow().getName(), table.orElseThrow().getName()),
123                 request, snapshot, queryFacade, databaseName)) {
124             snapshot.setStatus(WorkflowLifecycle.STATUS_FAILED);
125             return false;
126         }
127         return true;
128     }
129     
130     private Optional<ShardingSphereSchema> findSchema(final Collection<ShardingSphereSchema> schemas, final WorkflowRequest request,
131                                                       final MCPFeatureQueryFacade queryFacade, final String databaseName) {
132         return schemas.stream().filter(each -> matchesMetadataIdentifier(
133                 queryFacade, databaseName, IdentifierScope.SCHEMA, request.getSchema(), each.getName())).findFirst();
134     }
135     
136     private boolean ensureIdentifiers(final String fieldName, final Collection<String> identifiers, final WorkflowContextSnapshot snapshot, final String issueStage, final boolean allowEmpty) {
137         for (String each : identifiers) {
138             if (!ensureSupportedIdentifier(fieldName, each, snapshot, issueStage, allowEmpty)) {
139                 return false;
140             }
141         }
142         return true;
143     }
144     
145     private boolean isEmptyIdentifier(final String identifier) {
146         return WorkflowSQLUtils.normalizeIdentifier(identifier).isEmpty();
147     }
148     
149     private void addMissingQuestions(final WorkflowRequest request, final ClarifiedIntent clarifiedIntent, final WorkflowContextSnapshot snapshot) {
150         if (isEmptyIdentifier(request.getSchema())) {
151             clarifiedIntent.getClarificationMessages().add("Please specify schema.");
152         }
153         addMissingTableQuestion(request, clarifiedIntent, snapshot, "Table is required before planning.");
154         addMissingColumnQuestion(request, clarifiedIntent, snapshot, "Column is required before planning.");
155     }
156     
157     private void addMissingTableQuestion(final WorkflowRequest request, final ClarifiedIntent clarifiedIntent, final WorkflowContextSnapshot snapshot, final String message) {
158         if (isEmptyIdentifier(request.getTable())) {
159             clarifiedIntent.getClarificationMessages().add("Please specify target table.");
160             snapshot.getIssues().add(new WorkflowIssue(WorkflowIssueCode.TABLE_REQUIRED, "error", WorkflowLifecycle.STEP_INTAKING,
161                     message, "Provide the logical table name.", true, Map.of()));
162         }
163     }
164     
165     private void addMissingColumnQuestion(final WorkflowRequest request, final ClarifiedIntent clarifiedIntent, final WorkflowContextSnapshot snapshot, final String message) {
166         if (isEmptyIdentifier(request.getColumn())) {
167             clarifiedIntent.getClarificationMessages().add("Please specify target column.");
168             snapshot.getIssues().add(new WorkflowIssue(WorkflowIssueCode.COLUMN_REQUIRED, "error", WorkflowLifecycle.STEP_INTAKING,
169                     message, "Provide the logical column name.", true, Map.of()));
170         }
171     }
172     
173     private boolean ensureSupportedIdentifier(final String fieldName, final String identifier, final WorkflowContextSnapshot snapshot, final String issueStage, final boolean allowEmpty) {
174         if (allowEmpty && identifier.isEmpty() || WorkflowSQLUtils.isSupportedIdentifier(identifier)) {
175             return true;
176         }
177         snapshot.getIssues().add(new WorkflowIssue(WorkflowIssueCode.UNSUPPORTED_IDENTIFIER, "error", issueStage, createUnsupportedIdentifierMessage(fieldName, identifier),
178                 createUnsupportedIdentifierAction(fieldName), false, createUnsupportedIdentifierDetails(fieldName, identifier)));
179         return false;
180     }
181     
182     private String createUnsupportedIdentifierMessage(final String fieldName, final String identifier) {
183         return fieldName.isEmpty()
184                 ? String.format("Identifier `%s` contains unsupported characters.", identifier)
185                 : String.format("%s identifier `%s` contains unsupported characters.", fieldName, identifier);
186     }
187     
188     private String createUnsupportedIdentifierAction(final String fieldName) {
189         return fieldName.isEmpty()
190                 ? "Use reviewable logical identifiers without NUL or line terminators."
191                 : "Use a reviewable logical identifier without NUL or line terminators.";
192     }
193     
194     private Map<String, Object> createUnsupportedIdentifierDetails(final String fieldName, final String identifier) {
195         return fieldName.isEmpty() ? Map.of("identifier", identifier) : Map.of("field", fieldName, "identifier", identifier);
196     }
197     
198     private String resolveSchema(final Collection<ShardingSphereSchema> schemas, final WorkflowRequest request, final ClarifiedIntent clarifiedIntent,
199                                  final MCPFeatureQueryFacade queryFacade, final String databaseName) {
200         String actualSchema = request.getSchema();
201         if (!isEmptyIdentifier(actualSchema)) {
202             return actualSchema;
203         }
204         if (schemas.isEmpty()) {
205             return "";
206         }
207         if (!WorkflowSQLUtils.normalizeIdentifier(request.getTable()).isEmpty()) {
208             List<String> result = new LinkedList<>();
209             for (ShardingSphereSchema each : schemas) {
210                 if (containsTable(queryFacade, databaseName, request.getTable(), each.getAllTables())) {
211                     result.add(each.getName());
212                 }
213             }
214             if (1 == result.size()) {
215                 return recordInferredSchema(clarifiedIntent, result.get(0));
216             }
217         }
218         return 1 == schemas.size() ? recordInferredSchema(clarifiedIntent, schemas.iterator().next().getName()) : "";
219     }
220     
221     private boolean containsTable(final MCPFeatureQueryFacade queryFacade, final String databaseName, final String tableName, final Collection<ShardingSphereTable> tables) {
222         for (ShardingSphereTable each : tables) {
223             if (TableType.TABLE == each.getType() && queryFacade.isSameIdentifier(databaseName, IdentifierScope.TABLE, tableName, each.getName())) {
224                 return true;
225             }
226         }
227         return false;
228     }
229     
230     private String recordInferredSchema(final ClarifiedIntent clarifiedIntent, final String schema) {
231         if (null != clarifiedIntent) {
232             clarifiedIntent.getInferredValues().put(WorkflowFieldNames.SCHEMA, schema);
233         }
234         return schema;
235     }
236     
237     private Optional<ShardingSphereTable> findTable(final Optional<ShardingSphereSchema> schema, final WorkflowRequest request,
238                                                     final MCPFeatureQueryFacade queryFacade, final String databaseName) {
239         return schema.flatMap(optional -> optional.getAllTables().stream()
240                 .filter(each -> TableType.TABLE == each.getType()
241                         && matchesMetadataIdentifier(queryFacade, databaseName, IdentifierScope.TABLE, request.getTable(), each.getName()))
242                 .findFirst());
243     }
244     
245     private boolean matchesMetadataIdentifier(final MCPFeatureQueryFacade queryFacade, final String databaseName, final IdentifierScope identifierScope,
246                                               final String identifier, final String existingIdentifier) {
247         return WorkflowSQLUtils.requiresExactIdentifierMatch(identifier)
248                 ? WorkflowSQLUtils.normalizeIdentifier(identifier).equals(existingIdentifier)
249                 : queryFacade.isSameIdentifier(databaseName, identifierScope, identifier, existingIdentifier);
250     }
251     
252     private boolean ensureTableExists(final Optional<ShardingSphereTable> table, final WorkflowRequest request, final WorkflowContextSnapshot snapshot) {
253         if (table.isPresent()) {
254             return true;
255         }
256         snapshot.getIssues().add(new WorkflowIssue(WorkflowIssueCode.TABLE_NOT_FOUND, "error", WorkflowLifecycle.STEP_DISCOVERING,
257                 String.format("Table `%s` does not exist in Proxy logical metadata.", request.getTable()), "Check database, schema and table name.", false, Map.of()));
258         return false;
259     }
260     
261     private boolean ensureColumnExists(final Collection<MCPColumnMetadata> columns, final WorkflowRequest request, final WorkflowContextSnapshot snapshot,
262                                        final MCPFeatureQueryFacade queryFacade, final String databaseName) {
263         for (MCPColumnMetadata each : columns) {
264             if (matchesMetadataIdentifier(queryFacade, databaseName, IdentifierScope.COLUMN, request.getColumn(), each.getName())) {
265                 return true;
266             }
267         }
268         snapshot.getIssues().add(new WorkflowIssue(WorkflowIssueCode.COLUMN_NOT_FOUND, "error", WorkflowLifecycle.STEP_DISCOVERING,
269                 String.format("Column `%s` does not exist in Proxy logical metadata.", request.getColumn()), "Check column name.", false, Map.of()));
270         return false;
271     }
