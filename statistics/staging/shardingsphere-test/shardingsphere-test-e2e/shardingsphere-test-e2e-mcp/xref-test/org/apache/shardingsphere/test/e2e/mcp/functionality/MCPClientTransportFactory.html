21  import io.modelcontextprotocol.client.McpSyncClient;
22  import io.modelcontextprotocol.client.transport.HttpClientStreamableHttpTransport;
23  import io.modelcontextprotocol.client.transport.ServerParameters;
24  import io.modelcontextprotocol.client.transport.StdioClientTransport;
25  import io.modelcontextprotocol.spec.McpClientTransport;
26  import io.modelcontextprotocol.spec.McpSchema;
27  import lombok.AccessLevel;
28  import lombok.NoArgsConstructor;
29  import org.apache.shardingsphere.mcp.bootstrap.MCPBootstrap;
30  import org.apache.shardingsphere.mcp.bootstrap.transport.MCPTransportJsonMapperFactory;
31  import org.apache.shardingsphere.test.e2e.mcp.support.transport.MCPInteractionProtocolSupport;
32  import org.apache.shardingsphere.test.e2e.mcp.support.transport.client.MCPStdioLogbackConfiguration;
33  
34  import java.io.IOException;
35  import java.net.URI;
36  import java.nio.file.Path;
37  import java.nio.file.Paths;
38  import java.time.Duration;
39  import java.util.List;
40  import java.util.function.BiFunction;
41  
42  @NoArgsConstructor(access = AccessLevel.PRIVATE)
43  final class MCPClientTransportFactory {
44      
45      private static final String STDIO_LOGBACK_CONFIG_FILE_NAME = "mcp-functionality-e2e-sdk-stdio-logback.xml";
46      
47      private static final List<String> SUPPORTED_PROTOCOL_VERSIONS = List.of(MCPInteractionProtocolSupport.PROTOCOL_VERSION);
48      
49      static McpSyncClient createElicitationClient(final McpClientTransport clientTransport, final List<McpSchema.ElicitRequest> elicitationRequests,
50                                                   final BiFunction<List<McpSchema.ElicitRequest>, McpSchema.ElicitRequest, McpSchema.ElicitResult> elicitationHandler) {
51          return McpClient.sync(clientTransport)
52                  .clientInfo(new McpSchema.Implementation("mcp-functionality-e2e-elicitation", "MCP Functionality E2E Elicitation", "1.0.0"))
53                  .capabilities(McpSchema.ClientCapabilities.builder().elicitation().build())
54                  .requestTimeout(Duration.ofSeconds(30L))
55                  .initializationTimeout(Duration.ofSeconds(30L))
56                  .elicitation(request -> elicitationHandler.apply(elicitationRequests, request))
57                  .build();
58      }
59      
60      static McpClientTransport createHttpClientTransport(final URI endpointUri) {
61          return HttpClientStreamableHttpTransport.builder(String.format("%s://%s:%d", endpointUri.getScheme(), endpointUri.getHost(), endpointUri.getPort()))
62                  .endpoint(endpointUri.getPath()).supportedProtocolVersions(SUPPORTED_PROTOCOL_VERSIONS).build();
63      }
64      
65      static StdioClientTransport createStdioClientTransport(final Path configFile) throws IOException {
66          return new ProtocolAwareStdioClientTransport(ServerParameters.builder(Paths.get(System.getProperty("java.home"), "bin", "java").toString())
67                  .args("-Dlogback.configurationFile=" + MCPStdioLogbackConfiguration.createForConfig(configFile, STDIO_LOGBACK_CONFIG_FILE_NAME),
68                          "-cp", System.getProperty("java.class.path"), MCPBootstrap.class.getName(), configFile.toString())
69                  .build());
70      }
71      
72      private static final class ProtocolAwareStdioClientTransport extends StdioClientTransport {
73          
74          private ProtocolAwareStdioClientTransport(final ServerParameters params) {
75              super(params, MCPTransportJsonMapperFactory.create());
76          }
77          
78          @Override
79          public List<String> protocolVersions() {
80              return SUPPORTED_PROTOCOL_VERSIONS;
81          }
82      }
83  }
