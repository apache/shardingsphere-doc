66          assertThat(actual.getXid(), is(616281L));
67      }

77      @Test
78      void assertDecodeWriteRowEvent() {
79          ByteBuffer data = ByteBuffer.wrap(("table public.test: INSERT: data[character varying]:' 1 2 3'' 😊中' t_json_empty[json]:'{}' t_json[json]:'{\"test\":\"中中{中中}' 中\"}'"
80                  + " t_jsonb[jsonb]:'{\"test\":\"😊Emoji中\"}'").getBytes(StandardCharsets.UTF_8));
81          WriteRowEvent actual = (WriteRowEvent) new TestDecodingPlugin(null).decode(data, logSequenceNumber);
82          assertThat(actual.getLogSequenceNumber(), is(logSequenceNumber));
83          assertThat(actual.getSchemaName(), is("public"));
84          assertThat(actual.getTableName(), is("test"));
85          assertThat(actual.getAfterRow().get(0), is(" 1 2 3' 😊中"));
86          assertThat(actual.getAfterRow().get(1), is("{}"));
87          assertThat(actual.getAfterRow().get(2), is("{\"test\":\"中中{中中}' 中\"}"));
88          assertThat(actual.getAfterRow().get(3), is("{\"test\":\"😊Emoji中\"}"));
89      }

169     @Test
170     void assertDecodeVariousColumnTypes() {
171         BaseTimestampUtils timestampUtils = new BaseTimestampUtils() {
172             
173             @Override
174             public Time toTime(final Calendar cal, final String input) {
175                 return Time.valueOf(input);
176             }
177             
178             @Override
179             public Timestamp toTimestamp(final Calendar cal, final String input) {
180                 return Timestamp.valueOf(input);
181             }
182         };
183         String wal = "table public.test: INSERT: foo_numeric[numeric]:123.45 foo_bit[bit]:101 foo_smallint[smallint]:1 foo_bigint[bigint]:1234567890123"
184                 + " foo_real[real]:1.5 foo_double[double precision]:2.5 foo_boolean[boolean]:true foo_date[date]:'2020-01-02'"
185                 + " foo_time[time without time zone]:'12:34:56' foo_timestamp[timestamp without time zone]:'2020-01-02 03:04:05'"
186                 + " foo_bytea[bytea]:'\\x' foo_json[json]:'{\"a\":{\"b\":1}}' foo_text[character varying]:'abc''def' foo_null[integer]:null";
187         WriteRowEvent actual = (WriteRowEvent) new TestDecodingPlugin(timestampUtils).decode(ByteBuffer.wrap(wal.getBytes(StandardCharsets.UTF_8)), logSequenceNumber);
188         assertThat(actual.getAfterRow().get(0), is(new BigDecimal("123.45")));
189         assertThat(actual.getAfterRow().get(1), is("101"));
190         assertThat(actual.getAfterRow().get(2), is((short) 1));
191         assertThat(actual.getAfterRow().get(3), is(1234567890123L));
192         assertThat(actual.getAfterRow().get(4), is(1.5F));
193         assertThat(actual.getAfterRow().get(5), is(2.5D));
194         assertTrue((boolean) actual.getAfterRow().get(6));
195         assertThat(actual.getAfterRow().get(7), is(Date.valueOf("2020-01-02")));
196         assertThat(actual.getAfterRow().get(8), is(Time.valueOf("12:34:56")));
197         assertThat(actual.getAfterRow().get(9), is(Timestamp.valueOf("2020-01-02 03:04:05")));
198         assertThat(actual.getAfterRow().get(10), is(new byte[0]));
199         assertThat(actual.getAfterRow().get(11), is("{\"a\":{\"b\":1}}"));
200         assertThat(actual.getAfterRow().get(12), is("abc'def"));
201         assertNull(actual.getAfterRow().get(13));
202     }
