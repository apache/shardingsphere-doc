18  package org.apache.shardingsphere.mcp.bootstrap.transport.server.http;
19  
20  import io.modelcontextprotocol.json.McpJsonMapper;
21  import io.modelcontextprotocol.server.McpSyncServer;
22  import org.apache.catalina.Context;
23  import org.apache.catalina.LifecycleException;
24  import org.apache.catalina.Wrapper;
25  import org.apache.catalina.connector.Connector;
26  import org.apache.catalina.core.StandardContext;
27  import org.apache.catalina.startup.Tomcat;
28  import org.apache.shardingsphere.mcp.bootstrap.config.HttpTransportConfiguration;
29  import org.apache.shardingsphere.mcp.bootstrap.transport.MCPTransportJsonMapperFactory;
30  import org.apache.shardingsphere.mcp.bootstrap.transport.server.MCPRuntimeServer;
31  import org.apache.shardingsphere.mcp.bootstrap.transport.server.MCPSyncServerFactory;
32  import org.apache.shardingsphere.mcp.core.context.MCPRuntimeContext;
33  
34  import java.io.IOException;
35  import java.nio.file.Files;
36  import java.nio.file.Path;
37  import java.util.Comparator;
38  
39  /**
40   * Streamable HTTP MCP server.
41   */
42  public final class StreamableHttpMCPServer implements MCPRuntimeServer {
43      
44      private final HttpTransportConfiguration config;
45      
46      private final MCPSyncServerFactory syncServerFactory;
47      
48      private final StreamableHttpMCPServlet transportServlet;
49      
50      private McpSyncServer syncServer;
51      
52      private Tomcat tomcat;
53      
54      private Connector connector;
55      
56      private Path baseDirectory;
57      
58      public StreamableHttpMCPServer(final HttpTransportConfiguration config, final MCPRuntimeContext runtimeContext) {
59          this.config = config;
60          McpJsonMapper jsonMapper = MCPTransportJsonMapperFactory.create();
61          syncServerFactory = new MCPSyncServerFactory(runtimeContext, jsonMapper);
62          transportServlet = new StreamableHttpMCPServlet(runtimeContext.getSessionManager(), jsonMapper, config);
63      }
64      
65      @Override
66      public void start() throws IOException {
67          if (null == syncServer) {
68              syncServer = syncServerFactory.create(transportServlet);
69              startEmbeddedTomcat();
70          }
71      }
72      
73      private void startEmbeddedTomcat() throws IOException {
74          try {
75              tomcat = new Tomcat();
76              connector = new Connector();
77              connector.setPort(config.getPort());
78              connector.setProperty("address", config.getBindHost());
79              tomcat.setConnector(connector);
80              baseDirectory = Files.createTempDirectory("shardingsphere-mcp-tomcat");
81              tomcat.setBaseDir(baseDirectory.toString());
82              Context context = tomcat.addContext("", baseDirectory.toString());
83              ((StandardContext) context).setClearReferencesRmiTargets(false);
84              registerTransportServlet(context);
85              tomcat.start();
86          } catch (final LifecycleException ex) {
87              stop();
88              throw new IOException("Failed to start embedded Tomcat runtime.", ex);
89          }
90      }
91      
92      private void registerTransportServlet(final Context context) {
93          Wrapper servletWrapper = Tomcat.addServlet(context, "mcp-streamable-http", transportServlet);
94          servletWrapper.setAsyncSupported(true);
95          context.addServletMappingDecoded(config.getEndpointPath(), "mcp-streamable-http");
96      }
97      
98      @Override
99      public void stop() {
100         closeTomcat();
101         closeSyncServer();
102         connector = null;
103         deleteBaseDirectory();
104     }
105     
106     private void closeTomcat() {
107         if (null == tomcat) {
108             return;
109         }
110         try {
111             tomcat.stop();
112         } catch (final LifecycleException ignored) {
113         }
114         try {
115             tomcat.destroy();
116         } catch (final LifecycleException ignored) {
117         }
118         tomcat = null;
119     }
120     
121     private void closeSyncServer() {
122         if (null != syncServer) {
123             syncServer.closeGracefully();
124             syncServer = null;
125         }
126     }
127     
128     private void deleteBaseDirectory() {
129         if (null == baseDirectory) {
130             return;
131         }
132         try {
133             Files.walk(baseDirectory)
134                     .sorted(Comparator.reverseOrder())
135                     .forEach(each -> {
136                         try {
137                             Files.deleteIfExists(each);
138                         } catch (final IOException ignored) {
139                         }
140                     });
141         } catch (final IOException ignored) {
142         }
143         baseDirectory = null;
144     }
145     
146     /**
147      * Get the local port that the listener actually bound to.
148      *
149      * @return bound port
150      */
151     public int getLocalPort() {
152         return null == connector ? config.getPort() : connector.getLocalPort();
153     }
154 }
