37  import static org.junit.jupiter.api.Assertions.assertThrows;
38  import static org.mockito.Mockito.inOrder;
39  import static org.mockito.Mockito.verify;
40  
41  @ExtendWith(MockitoExtension.class)
42  class FirebirdDatabaseInfoReturnPacketTest {
43      
44      @Mock
45      private FirebirdPacketPayload payload;
46      
47      @ParameterizedTest(name = "{0}")
48      @MethodSource("databaseInfoTypes")
49      void assertWriteWithDatabaseInfoType(final String name, final FirebirdDatabaseInfoPacketType infoType, final int expectedLength, final int expectedValue, final boolean int1Value) {
50          new FirebirdDatabaseInfoReturnPacket(Collections.singletonList(infoType)).write(payload);
51          InOrder order = inOrder(payload);
52          order.verify(payload).writeInt1(infoType.getCode());
53          order.verify(payload).writeInt2LE(expectedLength);
54          if (int1Value) {
55              order.verify(payload).writeInt1(expectedValue);
56          } else {
57              order.verify(payload).writeInt4LE(expectedValue);
58          }
59      }
60      
61      @Test
62      void assertWriteFirebirdVersion() {
63          new FirebirdDatabaseInfoReturnPacket(Collections.singletonList(FirebirdDatabaseInfoPacketType.FIREBIRD_VERSION)).write(payload);
64          String fbVersion = String.format("%s-%s%d.%d.%d.%d Firebird %d.%d (ShardingSphere-Proxy)", FirebirdArchType.ARCHITECTURE.getIdentifier(), "V", 5, 0, 0, 0, 5, 0);
65          InOrder order = inOrder(payload);
66          order.verify(payload).writeInt1(FirebirdDatabaseInfoPacketType.FIREBIRD_VERSION.getCode());
67          order.verify(payload).writeInt2LE(fbVersion.length() + 2);
68          order.verify(payload).writeInt1(1);
69          order.verify(payload).writeInt1(fbVersion.length());
70          order.verify(payload).writeBytes(fbVersion.getBytes(StandardCharsets.UTF_8));
71      }
72      
73      @Test
74      void assertParseDatabaseInfoWithUnknownType() {
75          assertThrows(DatabaseProtocolException.class, () -> new FirebirdDatabaseInfoReturnPacket(Collections.singletonList(FirebirdDatabaseInfoPacketType.DB_ID)).write(payload));
76      }
77      
78      @Test
79      void assertWriteCommonInfo() {
80          new FirebirdDatabaseInfoReturnPacket(Collections.singletonList(FirebirdCommonInfoPacketType.END)).write(payload);
81          verify(payload).writeInt1(FirebirdCommonInfoPacketType.END.getCode());
82      }
83      
84      private static Stream<Arguments> databaseInfoTypes() {
85          return Stream.of(
86                  Arguments.of("db sql dialect", FirebirdDatabaseInfoPacketType.DB_SQL_DIALECT, 1, 3, true),
87                  Arguments.of("ods version", FirebirdDatabaseInfoPacketType.ODS_VERSION, 4, 13, false),
88                  Arguments.of("ods minor version", FirebirdDatabaseInfoPacketType.ODS_MINOR_VERSION, 4, 1, false));
89      }
90  }
