41  /**
42   * Run one LLM conversation without harness corrections.
43   */
44  public final class LLMConversationRunner {
45      
46      static final String READ_RESOURCE_TOOL_NAME = "mcp_read_resource";
47      
48      private static final String TOOL_CALL_KIND = "tool_call";
49      
50      private static final String RESOURCE_READ_KIND = "resource_read";
51      
52      private static final String SYSTEM_PROMPT = """
53              You are evaluating a live ShardingSphere MCP server. Use the available MCP functions to inspect current state and complete the user's task.
54              Before constructing an UPDATE statement for preview, retrieve the target table's complete column list from MCP.
55              Use the returned column names for both assignments and row selection.
56              Choose tools from their advertised names, descriptions, and input schemas. Pass resource URIs only to mcp_read_resource, never as SQL.
57              Never guess or stop before retrieving the requested evidence. When target metadata is explicitly marked as already verified, use the requested
58              planning response as evidence instead of repeating metadata discovery. Preview side effects without executing them. Use function calls for
59              every tool invocation rather than printing a tool-call object as the answer. When you have enough evidence, answer the user's question concisely.
60              """;

62      private final int maxTurns;
63      
64      private final LLMChatModelClient llmChatClient;
65      
66      private final MCPInteractionClient mcpInteractionClient;
67      
68      private final String modelName;
69      
70      private final LLMMCPToolDefinitionFactory toolDefinitionFactory = new LLMMCPToolDefinitionFactory();
71      
72      private final LLMMCPSafetyValidator safetyValidator = new LLMMCPSafetyValidator();
73      
74      public LLMConversationRunner(final int maxTurns, final LLMChatModelClient llmChatClient, final MCPInteractionClient mcpInteractionClient,
75                                   final String modelName) {
76          this.maxTurns = maxTurns;
77          this.llmChatClient = llmChatClient;
78          this.mcpInteractionClient = mcpInteractionClient;
79          this.modelName = modelName;
80      }
81      
82      /**
83       * Run one scenario. The supplied MCP interaction client is opened and closed for this invocation.
84       *
85       * @param scenario scenario
86       * @return conversation result
87       */
88      public Result run(final Scenario scenario) {
89          ConversationArtifacts artifacts = new ConversationArtifacts();
90          try {
91              mcpInteractionClient.open();
92              List<Map<String, Object>> advertisedTools = mcpInteractionClient.listTools();
93              List<Map<String, Object>> toolDefinitions = toolDefinitionFactory.createFromRemote(advertisedTools, scenario.allowedToolNames());
94              artifacts.setToolDefinitions(toolDefinitions);
95              return runTurns(scenario, artifacts, toolDefinitions);
96          } catch (final IOException ex) {
97              return artifacts.createResult(scenario, modelName,
98                      LLME2EAssertionReport.failure("mcp_runtime_unavailable", ex.getMessage()));
99          } catch (final InterruptedException ex) {
100             Thread.currentThread().interrupt();
101             return artifacts.createResult(scenario, modelName,
102                     LLME2EAssertionReport.failure("conversation_interrupted", "Conversation was interrupted."));
103         } catch (final IllegalStateException ex) {
104             return artifacts.createResult(scenario, modelName,
105                     LLME2EAssertionReport.failure("mcp_runtime_unavailable", ex.getMessage()));
106         } finally {
107             closeInteractionClient();
108         }
109     }
110     
111     private void closeInteractionClient() {
112         try {
113             mcpInteractionClient.close();
114         } catch (final InterruptedException ignored) {
115             Thread.currentThread().interrupt();
116         } catch (final IOException ignored) {
117         }
118     }
119     
120     private Result runTurns(final Scenario scenario, final ConversationArtifacts artifacts,
121                             final List<Map<String, Object>> toolDefinitions) throws InterruptedException {
122         List<LLMChatMessage> messages = new LinkedList<>();
123         messages.add(LLMChatMessage.system(SYSTEM_PROMPT));
124         messages.add(LLMChatMessage.user(scenario.question()));
125         Set<String> availableToolNames = toolDefinitions.stream()
126                 .map(each -> LLMMCPJsonValues.castToMap(each.get("function")))
127                 .map(each -> String.valueOf(each.get("name")))
128                 .collect(Collectors.toSet());
129         for (int turnIndex = 0; turnIndex < maxTurns; turnIndex++) {
130             LLMChatCompletion completion;
131             Map<String, Object> responseFormat = scenario.finalResponseFormat().apply(artifacts.getTrace());
132             try {
133                 completion = llmChatClient.complete(messages, responseFormat.isEmpty() ? toolDefinitions : List.of(), responseFormat.isEmpty() ? "auto" : "none", responseFormat);
134             } catch (final IOException | IllegalStateException ex) {
135                 return artifacts.createResult(scenario, modelName,
136                         LLME2EAssertionReport.failure("model_service_unavailable", ex.getMessage()));
137             }
138             artifacts.addRawModelOutput(completion.getRawResponse());
139             if (completion.getToolCalls().isEmpty()) {
140                 return createFinalResult(scenario, completion.getContent(), artifacts);
141             }
142             messages.add(LLMChatMessage.assistant(completion.getContent(), completion.getToolCalls()));
143             Optional<Result> failure = executeToolCalls(scenario, turnIndex + 1, completion.getToolCalls(), messages, artifacts, availableToolNames);
144             if (failure.isPresent()) {
145                 return failure.get();
146             }
147         }
148         return artifacts.createResult(scenario, modelName,
149                 LLME2EAssertionReport.failure("turn_limit_exhausted", "Model did not produce a final answer within the configured turn limit."));
150     }
151     
152     private Optional<Result> executeToolCalls(final Scenario scenario, final int modelTurn, final List<LLMToolCall> toolCalls, final List<LLMChatMessage> messages,
153                                               final ConversationArtifacts artifacts, final Set<String> availableToolNames) throws InterruptedException {
154         for (LLMToolCall each : toolCalls) {
155             if (!availableToolNames.contains(each.getName())) {
156                 artifacts.addTrace(MCPInteractionTraceRecord.createInvalidAction(artifacts.nextSequence(), modelTurn, TOOL_CALL_KIND, each.getName(),
157                         Map.of("rawArgumentsJson", each.getArgumentsJson()), "unexpected_tool_requested"));
158                 return Optional.of(artifacts.createResult(scenario, modelName,
159                         LLME2EAssertionReport.failure("unexpected_tool_requested", "Model requested a tool that was not advertised for this scenario.")));
160             }
161             Map<String, Object> arguments;
162             try {
163                 arguments = LLMMCPJsonValues.parseToolArguments(each.getArgumentsJson());
164             } catch (final IllegalArgumentException ex) {
165                 artifacts.addTrace(MCPInteractionTraceRecord.createInvalidAction(artifacts.nextSequence(), modelTurn, TOOL_CALL_KIND, each.getName(),
166                         Map.of("rawArgumentsJson", each.getArgumentsJson()), "invalid_tool_arguments"));
167                 return Optional.of(artifacts.createResult(scenario, modelName,
168                         LLME2EAssertionReport.failure("invalid_tool_arguments", "Model returned invalid tool arguments JSON.")));
169             }
170             Optional<LLMMCPSafetyValidator.ValidationFailure> validationFailure = safetyValidator.validate(each.getName(), arguments);
171             if (validationFailure.isPresent()) {
172                 LLMMCPSafetyValidator.ValidationFailure failure = validationFailure.get();
173                 artifacts.addTrace(MCPInteractionTraceRecord.createInvalidAction(
174                         artifacts.nextSequence(), modelTurn, getActionKind(each.getName()), each.getName(), arguments, failure.failureType()));
175                 return Optional.of(artifacts.createResult(scenario, modelName,
176                         LLME2EAssertionReport.failure(failure.failureType(), failure.message())));
177             }
178             long startTime = System.currentTimeMillis();
179             Map<String, Object> response;
180             try {
181                 response = executeAction(each.getName(), arguments);
182             } catch (final IllegalArgumentException ex) {
183                 artifacts.addTrace(MCPInteractionTraceRecord.createInvalidAction(
184                         artifacts.nextSequence(), modelTurn, TOOL_CALL_KIND, each.getName(), arguments, "invalid_tool_arguments"));
185                 return Optional.of(artifacts.createResult(scenario, modelName,
186                         LLME2EAssertionReport.failure("invalid_tool_arguments", ex.getMessage())));
187             } catch (final IllegalStateException ex) {
188                 return Optional.of(artifacts.createResult(scenario, modelName,
189                         LLME2EAssertionReport.failure("mcp_runtime_unavailable", ex.getMessage())));
190             }
191             long latencyMillis = System.currentTimeMillis() - startTime;
192             artifacts.addTrace(new MCPInteractionTraceRecord(
193                     artifacts.nextSequence(), modelTurn, getActionKind(each.getName()), MCPInteractionTraceRecord.MODEL_TOOL_CALL_ORIGIN,
194                     each.getName(), getTraceArguments(each.getName(), arguments), response, true, latencyMillis));
195             messages.add(LLMChatMessage.tool(each.getId(), JsonEngine.marshal(response)));
196         }
197         return Optional.empty();
198     }
199     
200     private String getActionKind(final String toolName) {
201         return READ_RESOURCE_TOOL_NAME.equals(toolName) ? RESOURCE_READ_KIND : TOOL_CALL_KIND;
202     }
203     
204     private Map<String, Object> executeAction(final String actionName, final Map<String, Object> arguments) throws InterruptedException {
205         try {
206             return READ_RESOURCE_TOOL_NAME.equals(actionName)
207                     ? mcpInteractionClient.readResource(getRequiredResourceUri(arguments))
208                     : mcpInteractionClient.call(actionName, arguments);
209         } catch (final IOException | IllegalStateException ex) {
210             throw new IllegalStateException(String.format("MCP action `%s` failed: %s", actionName, ex.getMessage()), ex);
211         }
212     }
213     
214     private String getRequiredResourceUri(final Map<String, Object> arguments) {
215         Object value = arguments.get("uri");
216         if (!(value instanceof String) || ((String) value).trim().isEmpty()) {
217             throw new IllegalArgumentException("Resource URI is required.");
218         }
219         return ((String) value).trim();
220     }
221     
222     private Map<String, Object> getTraceArguments(final String toolName, final Map<String, Object> arguments) {
223         return READ_RESOURCE_TOOL_NAME.equals(toolName)
224                 ? Map.of("uri", Objects.toString(arguments.get("uri"), "").trim())
225                 : arguments;
226     }
227     
228     private Result createFinalResult(final Scenario scenario, final String actualAnswer, final ConversationArtifacts artifacts) {
229         artifacts.setActualAnswer(actualAnswer.trim());
230         if (artifacts.getTrace().stream().noneMatch(MCPInteractionTraceRecord::isValid)) {
231             return artifacts.createResult(scenario, modelName,
232                     LLME2EAssertionReport.failure("missing_mcp_evidence", "Model returned an answer without retrieving MCP evidence."));
233         }
234         LLME2EAssertionReport assertionReport;
235         try {
236             assertionReport = scenario.evaluator().apply(artifacts.getActualAnswer(), artifacts.getTrace());
237         } catch (final IllegalArgumentException | IllegalStateException | ClassCastException ex) {
238             assertionReport = LLME2EAssertionReport.failure("invalid_scenario_evidence", ex.getMessage());
239         }
240         return artifacts.createResult(scenario, modelName, assertionReport);
241     }

243     /**
244      * LLM scenario.
245      *
246      * @param id scenario ID
247      * @param question question
248      * @param allowedToolNames tools exposed only to this scenario
249      * @param finalResponseFormat final response format derived from sufficient MCP evidence, or an empty map while tools are needed
250      * @param evaluator scenario evidence evaluator
251      */
252     public record Scenario(String id, String question, Set<String> allowedToolNames,
253                            Function<List<MCPInteractionTraceRecord>, Map<String, Object>> finalResponseFormat,
254                            BiFunction<String, List<MCPInteractionTraceRecord>, LLME2EAssertionReport> evaluator) {
255     }
256     
257     /**
258      * Autonomous conversation result.
259      *
260      * @param scenario scenario
261      * @param systemPrompt system prompt
262      * @param modelProvider model provider
263      * @param modelName model name
264      * @param actualAnswer actual answer
265      * @param evidence model and MCP evidence
266      * @param assertionReport assertion report
267      */
268     public record Result(Scenario scenario, String systemPrompt, String modelProvider, String modelName, String actualAnswer,
269                          Evidence evidence, LLME2EAssertionReport assertionReport) {
270     }
271     
272     /**
273      * Model and MCP evidence captured for one conversation.
274      *
275      * @param rawModelOutputs raw model outputs
276      * @param toolDefinitions tool definitions derived from MCP discovery
277      * @param interactionTrace interaction trace
278      */
279     public record Evidence(List<String> rawModelOutputs, List<Map<String, Object>> toolDefinitions,
280                            List<MCPInteractionTraceRecord> interactionTrace) {
281     }
282     
283     private static final class ConversationArtifacts {
284         
285         private String actualAnswer = "";
286         
287         private final List<String> rawModelOutputs = new LinkedList<>();
288         
289         private List<Map<String, Object>> toolDefinitions = List.of();
290         
291         private final List<MCPInteractionTraceRecord> trace = new LinkedList<>();
292         
293         private int nextSequence() {
294             return trace.size() + 1;
295         }
296         
297         private void setActualAnswer(final String actualAnswer) {
298             this.actualAnswer = actualAnswer;
299         }
300         
301         private String getActualAnswer() {
302             return actualAnswer;
303         }
304         
305         private void addRawModelOutput(final String rawModelOutput) {
306             rawModelOutputs.add(rawModelOutput);
307         }
308         
309         private void setToolDefinitions(final List<Map<String, Object>> toolDefinitions) {
310             this.toolDefinitions = toolDefinitions;
311         }
312         
313         private void addTrace(final MCPInteractionTraceRecord traceRecord) {
314             trace.add(traceRecord);
315         }
316         
317         private List<MCPInteractionTraceRecord> getTrace() {
318             return trace;
319         }
320         
321         private Result createResult(final Scenario scenario, final String modelName, final LLME2EAssertionReport assertionReport) {
322             return new Result(scenario, SYSTEM_PROMPT, LLME2EConfiguration.MODEL_PROVIDER, modelName, actualAnswer,
323                     new Evidence(rawModelOutputs, toolDefinitions, trace), assertionReport);
324         }
325     }
