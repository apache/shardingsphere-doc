33  import java.util.List;
34  import java.util.Map;
35  
36  /**
37   * Broadcast workflow validation service.
38   */
39  public final class BroadcastWorkflowValidationService implements MCPWorkflowRuntimeHandler {
40      
41      private final WorkflowValidationSupport validationSupport = new WorkflowValidationSupport();
42      
43      private final BroadcastRuleInspectionService ruleInspectionService = new BroadcastRuleInspectionService();
44      
45      @Override
46      public ValidationReport validate(final WorkflowContextSnapshot snapshot, final MCPFeatureQueryFacade queryFacade) {
47          ValidationReport result = new ValidationReport();
48          queryFacade.checkDatabaseCapability(snapshot.getRequest().getDatabase());
49          List<Map<String, Object>> broadcastRules = ruleInspectionService.queryBroadcastRules(queryFacade, snapshot.getRequest().getDatabase());
50          result.setRuleValidation(validateRules(snapshot, broadcastRules, result, queryFacade));
51          result.setOverallStatus(validationSupport.resolveOverallStatus(result.getRuleValidation()));
52          return result;
53      }
54      
55      private ValidationSection validateRules(final WorkflowContextSnapshot snapshot, final List<Map<String, Object>> broadcastRules,
56                                              final ValidationReport validationReport, final MCPFeatureQueryFacade queryFacade) {
57          BroadcastWorkflowRequest request = (BroadcastWorkflowRequest) snapshot.getRequest();
58          boolean dropWorkflow = WorkflowLifecycleUtils.isDropWorkflow(snapshot);
59          boolean expectedRuleExists = !dropWorkflow;
60          for (String each : request.getTargetTables()) {
61              boolean ruleExists = broadcastRules.stream().anyMatch(rule -> queryFacade.isSameIdentifier(
62                      request.getDatabase(), IdentifierScope.TABLE, each, WorkflowRuleValueUtils.getRuleValue(rule, "broadcast_table")));
63              if (expectedRuleExists != ruleExists) {
64                  addRuleMismatch(validationReport, dropWorkflow, each);
65                  return new ValidationSection(WorkflowLifecycle.STATUS_FAILED, broadcastRules, "Broadcast rule state does not match the planned DistSQL artifact.");
66              }
67          }
68          return new ValidationSection(WorkflowLifecycle.STATUS_PASSED, broadcastRules, "Broadcast rule state matches the planned DistSQL artifact.");
69      }
70      
71      private void addRuleMismatch(final ValidationReport validationReport, final boolean dropWorkflow, final String tableName) {
72          validationReport.getMismatches().add(validationSupport.createMismatch(WorkflowIssueCode.RULE_STATE_MISMATCH, "rule",
73                  dropWorkflow ? "table absent" : "table present", tableName,
74                  dropWorkflow ? "Broadcast table still exists after drop." : "Broadcast table is missing after create.",
75                  dropWorkflow ? "Drop the broadcast table rule again or investigate the failure." : "Create the broadcast table rule again."));
76      }
77  }
