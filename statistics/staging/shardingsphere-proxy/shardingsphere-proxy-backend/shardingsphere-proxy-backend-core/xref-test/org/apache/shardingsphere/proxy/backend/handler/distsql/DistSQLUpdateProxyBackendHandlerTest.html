72      private final DatabaseType databaseType = TypedSPILoader.getService(DatabaseType.class, "FIXTURE");

77      @Test
78      void assertEmptyStorageUnit() {
79          ResourceMetaData resourceMetaData = mock(ResourceMetaData.class);
80          RuleMetaData ruleMetaData = mock(RuleMetaData.class);
81          ShardingSphereDatabase database = new ShardingSphereDatabase("foo_db", databaseType, resourceMetaData, ruleMetaData,
82                  Collections.emptyList(), new ConfigurationProperties(new Properties()));
83          when(contextManager.getDatabase("foo_db")).thenReturn(database);
84          RefreshTableMetaDataStatement sqlStatement = new RefreshTableMetaDataStatement();
85          DistSQLUpdateProxyBackendHandler backendHandler = new DistSQLUpdateProxyBackendHandler(sqlStatement, mockQueryContext(), mockConnectionSession(), contextManager);
86          assertThrows(EmptyStorageUnitException.class, backendHandler::execute);
87      }

94          when(resourceMetaData.getStorageUnits()).thenReturn(Collections.singletonMap("ds_0", storageUnit));

98          when(contextManager.getDatabase("foo_db")).thenReturn(database);
99          RefreshTableMetaDataStatement sqlStatement = new RefreshTableMetaDataStatement(new IdentifierValue("t_order"), "ds_1", null);
100         DistSQLUpdateProxyBackendHandler backendHandler = new DistSQLUpdateProxyBackendHandler(sqlStatement, mockQueryContext(), mockConnectionSession(), contextManager);

102     }

133     @Test
134     void assertUpdate() throws SQLException {
135         ResourceMetaData resourceMetaData = mock(ResourceMetaData.class);
136         when(resourceMetaData.getStorageUnits()).thenReturn(Collections.singletonMap("ds_0", mock(StorageUnit.class)));
137         ShardingSphereDatabase database = mock(ShardingSphereDatabase.class);
138         when(database.getDefaultSchemaName()).thenReturn("foo_schema");
139         when(database.getSchema(new IdentifierValue("foo_schema"))).thenReturn(mock(ShardingSphereSchema.class));
140         when(database.getResourceMetaData()).thenReturn(resourceMetaData);
141         when(contextManager.getDatabase("foo_db")).thenReturn(database);
142         RefreshTableMetaDataStatement sqlStatement = new RefreshTableMetaDataStatement();
143         assertThat(new DistSQLUpdateProxyBackendHandler(sqlStatement, mockQueryContext(), mockConnectionSession(), contextManager).execute(), isA(UpdateResponseHeader.class));
144     }
145     
146     @ParameterizedTest(name = "{0}")
147     @MethodSource("provideLockClusterStatements")
148     void assertUpdateWithLockCluster(final String name, final LockClusterStatement sqlStatement,
149                                      final long expectedTimeoutMillis, final ShardingSphereState expectedState) throws SQLException {
150         when(contextManager.getComputeNodeInstanceContext().getModeConfiguration().isCluster()).thenReturn(true);
151         when(contextManager.getStateContext().getState()).thenReturn(ShardingSphereState.OK);
152         ExclusiveOperatorEngine exclusiveOperatorEngine = contextManager.getExclusiveOperatorEngine();
153         doAnswer(invocation -> {
154             ExclusiveOperationVoidCallback callback = invocation.getArgument(2);
155             callback.execute();
156             return null;
157         }).when(exclusiveOperatorEngine).operate(any(), anyLong(), any(ExclusiveOperationVoidCallback.class));
158         assertThat(new DistSQLUpdateProxyBackendHandler(sqlStatement, mockQueryContext(), mockConnectionSession(), contextManager).execute(), isA(UpdateResponseHeader.class));
159         verify(exclusiveOperatorEngine).operate(any(), eq(expectedTimeoutMillis), any(ExclusiveOperationVoidCallback.class));
160         verify(contextManager.getPersistServiceFacade().getStateService()).update(expectedState);
161     }
162     
163     private static Stream<Arguments> provideLockClusterStatements() {
164         return Stream.of(
165                 Arguments.of("write lock with explicit timeout", new LockClusterStatement(new AlgorithmSegment("WRITE", new Properties()), 2000L), 2000L, ShardingSphereState.READ_ONLY),
166                 Arguments.of("write lock with default timeout", new LockClusterStatement(new AlgorithmSegment("WRITE", new Properties())), 3000L, ShardingSphereState.READ_ONLY),
167                 Arguments.of("read-write lock with explicit timeout", new LockClusterStatement(new AlgorithmSegment("READ_WRITE", new Properties()), 2000L), 2000L, ShardingSphereState.UNAVAILABLE));
168     }

178         return result;
