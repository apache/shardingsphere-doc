18  package org.apache.shardingsphere.mcp.feature.readwritesplitting.tool.service;
19  
20  import org.apache.shardingsphere.mcp.support.database.spi.MCPFeatureQueryFacade;
21  import org.apache.shardingsphere.mcp.support.workflow.service.WorkflowRuleValueUtils;
22  import org.apache.shardingsphere.mcp.support.workflow.service.WorkflowSQLUtils;
23  
24  import java.util.LinkedHashMap;
25  import java.util.List;
26  import java.util.Locale;
27  import java.util.Map;
28  
29  /**
30   * Readwrite-splitting rule inspection service.
31   */
32  public final class ReadwriteSplittingInspectionService {
33      
34      String queryProxyMode(final MCPFeatureQueryFacade queryFacade, final String databaseName) {
35          String result = queryFacade.query(databaseName, "SHOW COMPUTE NODE INFO").stream().findFirst()
36                  .map(each -> WorkflowRuleValueUtils.getRuleValue(each, "mode_type")).orElse("");
37          return result.isEmpty() ? "unknown" : result;
38      }
39      
40      /**
41       * Query readwrite-splitting rules.
42       *
43       * @param queryFacade query facade
44       * @param databaseName logical database name
45       * @return readwrite-splitting rules
46       */
47      public List<Map<String, Object>> queryRules(final MCPFeatureQueryFacade queryFacade, final String databaseName) {
48          return queryFacade.query(databaseName, String.format("SHOW READWRITE_SPLITTING RULES FROM %s", WorkflowSQLUtils.formatDistSQLIdentifier(databaseName)));
49      }
50      
51      /**
52       * Query one readwrite-splitting rule.
53       *
54       * @param queryFacade query facade
55       * @param databaseName logical database name
56       * @param ruleName rule name
57       * @return readwrite-splitting rule rows
58       */
59      public List<Map<String, Object>> queryRule(final MCPFeatureQueryFacade queryFacade, final String databaseName, final String ruleName) {
60          return queryFacade.query(databaseName, String.format("SHOW READWRITE_SPLITTING RULE %s FROM %s",
61                  WorkflowSQLUtils.formatDistSQLIdentifier(ruleName), WorkflowSQLUtils.formatDistSQLIdentifier(databaseName)));
62      }
63      
64      /**
65       * Query readwrite-splitting rule count.
66       *
67       * @param queryFacade query facade
68       * @param databaseName logical database name
69       * @return count rows
70       */
71      public List<Map<String, Object>> queryRuleCount(final MCPFeatureQueryFacade queryFacade, final String databaseName) {
72          return queryFacade.query(databaseName, String.format("COUNT READWRITE_SPLITTING RULE FROM %s", WorkflowSQLUtils.formatDistSQLIdentifier(databaseName)));
73      }
74      
75      /**
76       * Query all readwrite-splitting storage-unit statuses.
77       *
78       * @param queryFacade query facade
79       * @param databaseName logical database name
80       * @return status rows
81       */
82      public List<Map<String, Object>> queryStatuses(final MCPFeatureQueryFacade queryFacade, final String databaseName) {
83          return queryFacade.query(databaseName, String.format("SHOW STATUS FROM READWRITE_SPLITTING RULES FROM %s", WorkflowSQLUtils.formatDistSQLIdentifier(databaseName)));
84      }
85      
86      /**
87       * Query one readwrite-splitting rule's storage-unit statuses.
88       *
89       * @param queryFacade query facade
90       * @param databaseName logical database name
91       * @param ruleName rule name
92       * @return status rows
93       */
94      public List<Map<String, Object>> queryRuleStatus(final MCPFeatureQueryFacade queryFacade, final String databaseName, final String ruleName) {
95          return queryFacade.query(databaseName, String.format("SHOW STATUS FROM READWRITE_SPLITTING RULE %s FROM %s",
96                  WorkflowSQLUtils.formatDistSQLIdentifier(ruleName), WorkflowSQLUtils.formatDistSQLIdentifier(databaseName)));
97      }
98      
99      /**
100      * Query load-balance algorithm plugin catalog.
101      *
102      * @param queryFacade query facade
103      * @return algorithm plugin query result with built-in property hints
104      */
105     public List<Map<String, Object>> queryLoadBalanceAlgorithmPlugins(final MCPFeatureQueryFacade queryFacade) {
106         return queryFacade.queryWithAnyDatabase("SHOW LOAD BALANCE ALGORITHM PLUGINS").stream().map(this::appendPropertyGuidance).toList();
107     }
108     
109     private Map<String, Object> appendPropertyGuidance(final Map<String, Object> row) {
110         Map<String, Object> result = new LinkedHashMap<>(row);
111         result.putIfAbsent("property_guidance", resolvePropertyGuidance(String.valueOf(row.getOrDefault("type", row.getOrDefault("name", "")))));
112         return result;
113     }
114     
115     private String resolvePropertyGuidance(final String algorithmType) {
116         String actualType = algorithmType.trim().toUpperCase(Locale.ENGLISH);
117         if ("WEIGHT".equals(actualType)) {
118             return "Provide one numeric property per read storage unit, for example read_ds_0=2 and read_ds_1=1.";
119         }
120         if ("RANDOM".equals(actualType) || "ROUND_ROBIN".equals(actualType)) {
121             return "No required properties.";
122         }
123         return "Read the plugin documentation for required properties before planning.";
124     }
125 }
