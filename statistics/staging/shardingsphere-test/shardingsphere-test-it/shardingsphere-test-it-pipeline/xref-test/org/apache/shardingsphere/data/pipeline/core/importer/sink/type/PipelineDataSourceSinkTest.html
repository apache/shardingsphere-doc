63  @ExtendWith(MockitoExtension.class)
64  class PipelineDataSourceSinkTest {
65      
66      private static final String TABLE_NAME = "test_table";
67      
68      private PipelineDataSourceConfiguration dataSourceConfig;
69      
70      @Mock
71      private PipelineChannel channel;
72      
73      @Mock
74      private Connection connection;
75      
76      @Mock
77      private PreparedStatement preparedStatement;
78      
79      private SingleChannelConsumerImporter importer;
80      
81      @BeforeEach
82      void setUp() throws SQLException {
83          Map<String, Object> poolProps = new HashMap<>(3, 1F);
84          poolProps.put("url", "jdbc:h2:mem:test_db;DB_CLOSE_DELAY=-1;DATABASE_TO_UPPER=false;MODE=MySQL;USER=root;PASSWORD=root");
85          poolProps.put("username", "root");
86          poolProps.put("password", "root");
87          dataSourceConfig = new StandardPipelineDataSourceConfiguration(poolProps);
88          PipelineSink pipelineSink = new PipelineDataSourceSink(mockImporterConfiguration(), mockPipelineDataSourceManager());
89          importer = new SingleChannelConsumerImporter(channel, 100, 1000L, pipelineSink, new FixtureTransmissionJobItemContext());
90      }
91      
92      private ImporterConfiguration mockImporterConfiguration() {
93          Map<ShardingSphereIdentifier, Collection<String>> tableAndRequiredColumnsMap = Collections.singletonMap(new ShardingSphereIdentifier("test_table"), Collections.singleton("user"));
94          return new ImporterConfiguration(dataSourceConfig, tableAndRequiredColumnsMap, new TableAndSchemaNameMapper(Collections.emptyMap()), 1000, null, 3, 3);
95      }
96      
97      private PipelineDataSourceManager mockPipelineDataSourceManager() throws SQLException {
98          PipelineDataSourceManager result = mock(PipelineDataSourceManager.class, RETURNS_DEEP_STUBS);
99          when(result.getDataSource(dataSourceConfig).getConnection()).thenReturn(connection);
100         return result;
101     }
102     
103     @Test
104     void assertWriteInsertDataRecord() throws SQLException {
105         DataRecord insertRecord = getDataRecord(PipelineSQLOperationType.INSERT);
106         when(connection.prepareStatement(any())).thenReturn(preparedStatement);
107         when(channel.fetch(anyInt(), anyLong())).thenReturn(mockRecords(insertRecord));
108         importer.run();
109         verify(preparedStatement).setObject(1, 1);
110         verify(preparedStatement).setObject(2, 10);
111         verify(preparedStatement).setObject(3, PipelineSQLOperationType.INSERT);
112         verify(preparedStatement).addBatch();
113     }
114     
115     @Test
116     void assertDeleteDataRecord() throws SQLException {
117         DataRecord deleteRecord = getDataRecord(PipelineSQLOperationType.DELETE);
118         when(connection.prepareStatement(any())).thenReturn(preparedStatement);
119         when(channel.fetch(anyInt(), anyLong())).thenReturn(mockRecords(deleteRecord));
120         when(preparedStatement.executeBatch()).thenReturn(new int[]{1});
121         importer.run();
122         verify(preparedStatement).setObject(1, 1);
123         verify(preparedStatement).setObject(2, 10);
124         verify(preparedStatement).addBatch();
125     }
126     
127     @Test
128     void assertUpdateDataRecord() throws SQLException {
129         DataRecord updateRecord = getDataRecord(PipelineSQLOperationType.UPDATE);
130         when(connection.prepareStatement(any())).thenReturn(preparedStatement);
131         when(channel.fetch(anyInt(), anyLong())).thenReturn(mockRecords(updateRecord));
132         importer.run();
133         verify(preparedStatement).setObject(1, 20);
134         verify(preparedStatement).setObject(2, PipelineSQLOperationType.UPDATE);
135         verify(preparedStatement).setObject(3, 1);
136         verify(preparedStatement).setObject(4, 10);
137         verify(preparedStatement).executeUpdate();
138     }
139     
140     @Test
141     void assertUpdatePrimaryKeyDataRecord() throws SQLException {
142         DataRecord updateRecord = getUpdatePrimaryKeyDataRecord();
143         when(connection.prepareStatement(any())).thenReturn(preparedStatement);
144         when(channel.fetch(anyInt(), anyLong())).thenReturn(mockRecords(updateRecord));
145         importer.run();
146         InOrder inOrder = inOrder(preparedStatement);
147         inOrder.verify(preparedStatement).setObject(1, 2);
148         inOrder.verify(preparedStatement).setObject(2, 10);
149         inOrder.verify(preparedStatement).setObject(3, PipelineSQLOperationType.UPDATE);
150         inOrder.verify(preparedStatement).setObject(4, 1);
151         inOrder.verify(preparedStatement).setObject(5, 0);
152         inOrder.verify(preparedStatement).executeUpdate();
153     }
154     
155     private DataRecord getUpdatePrimaryKeyDataRecord() {
156         DataRecord result = new DataRecord(PipelineSQLOperationType.UPDATE, TABLE_NAME, new IngestPlaceholderPosition(), 3);
157         result.addColumn(new NormalColumn("id", 1, 2, true, true));
158         result.addColumn(new NormalColumn("user", 0, 10, true, false));
159         result.addColumn(new NormalColumn("status", null, PipelineSQLOperationType.UPDATE, true, false));
160         return result;
161     }
162     
163     private List<Record> mockRecords(final DataRecord dataRecord) {
164         List<Record> result = new LinkedList<>();
165         result.add(dataRecord);
166         result.add(new FinishedRecord(new IngestFinishedPosition()));
167         return result;
168     }
169     
170     private DataRecord getDataRecord(final PipelineSQLOperationType type) {
171         Integer idOldValue = null;
172         Integer userOldValue = null;
173         Integer idValue = null;
174         Integer userValue = null;
175         PipelineSQLOperationType statusOldValue = null;
176         PipelineSQLOperationType statusValue = null;
177         if (PipelineSQLOperationType.INSERT == type) {
178             idValue = 1;
179             userValue = 10;
180             statusValue = type;
181         }
182         if (PipelineSQLOperationType.UPDATE == type) {
183             idOldValue = 1;
184             idValue = idOldValue;
185             userOldValue = 10;
186             userValue = 20;
187             statusValue = type;
188         }
189         if (PipelineSQLOperationType.DELETE == type) {
190             idOldValue = 1;
191             userOldValue = 10;
192             statusOldValue = type;
193         }
194         DataRecord result = new DataRecord(type, TABLE_NAME, new IngestPlaceholderPosition(), 3);
195         result.addColumn(new NormalColumn("id", idOldValue, idValue, false, true));
196         result.addColumn(new NormalColumn("user", userOldValue, userValue, true, false));
197         result.addColumn(new NormalColumn("status", statusOldValue, statusValue, true, false));
198         return result;
199     }
