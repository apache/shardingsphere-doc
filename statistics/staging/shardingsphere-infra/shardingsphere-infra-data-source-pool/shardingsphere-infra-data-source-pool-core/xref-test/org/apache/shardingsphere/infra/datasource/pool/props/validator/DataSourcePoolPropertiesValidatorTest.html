82      @MethodSource("validateWithMockedDataSourcePrivilegesArguments")
83      void assertValidateWithMockedDataSourcePrivileges(final String name, final Collection<PrivilegeCheckType> expectedPrivileges) {

88      @Test
89      void assertValidateWithInvalidContentProperties() {
90          DataSourcePoolPropertiesContentValidator validator = mock(DataSourcePoolPropertiesContentValidator.class);
91          DataSourcePoolProperties dataSourcePoolProperties = new DataSourcePoolProperties(HikariDataSource.class.getName(), createHikariDataSourceProperties(MOCKED_URL));
92          doThrow(new IllegalArgumentException("mock invalid properties")).when(validator).validate(dataSourcePoolProperties);
93          try (MockedStatic<TypedSPILoader> typedSPILoader = mockStatic(TypedSPILoader.class, CALLS_REAL_METHODS)) {
94              typedSPILoader.when(() -> TypedSPILoader.findService(DataSourcePoolPropertiesContentValidator.class, HikariDataSource.class.getName())).thenReturn(Optional.of(validator));
95              assertInvalidDataSource(DataSourcePoolPropertiesValidator.validate(
96                      Collections.singletonMap("foo_ds", dataSourcePoolProperties), Collections.emptySet()), "Invalid data source `foo_ds`, error message is: mock invalid properties");
97          }
98      }

105         try (MockedStatic<TypedSPILoader> typedSPILoader = mockStatic(TypedSPILoader.class, CALLS_REAL_METHODS)) {
106             typedSPILoader.when(() -> TypedSPILoader.findService(DataSourcePoolPropertiesContentValidator.class, HikariDataSource.class.getName())).thenReturn(Optional.of(validator));
107             assertInvalidDataSource(DataSourcePoolPropertiesValidator.validate(
108                     Collections.singletonMap("foo_ds", dataSourcePoolProperties), Collections.emptySet()), "Invalid data source `foo_ds`, error message is: mock illegal state properties");
109         }

112     @Test
113     void assertValidateWithCreateFailure() {
114         Map<String, Object> props = createMockedDataSourceProperties();
115         props.put("connectionTimeout", "invalid");
116         assertInvalidDataSource(DataSourcePoolPropertiesValidator.validate(Collections.singletonMap("foo_ds", new DataSourcePoolProperties(MockedDataSource.class.getName(), props)),

120     @Test
121     void assertValidateWithCreateFailureBeforeAssignment() {
122         Map<String, Object> props = createMockedDataSourceProperties();
123         DataSourcePoolProperties dataSourcePoolProperties = new DataSourcePoolProperties(MockedDataSource.class.getName(), props);
124         try (MockedStatic<DataSourcePoolCreator> dataSourcePoolCreator = mockStatic(DataSourcePoolCreator.class)) {
125             dataSourcePoolCreator.when(() -> DataSourcePoolCreator.create(dataSourcePoolProperties)).thenThrow(new IllegalStateException("mock create failure"));
126             assertInvalidDataSource(DataSourcePoolPropertiesValidator.validate(Collections.singletonMap("foo_ds", dataSourcePoolProperties), Collections.emptySet()),
127                     "Invalid data source `foo_ds`, error message is: mock create failure");
128         }

131     private void assertInvalidDataSource(final Map<String, Exception> actual, final String expectedMessage) {
132         assertThat(actual.size(), is(1));
133         assertThat(actual.get("foo_ds"), isA(InvalidDataSourcePoolPropertiesException.class));
134         assertThat(actual.get("foo_ds").getMessage(), is(expectedMessage));
135     }

137     @Test
138     void assertValidateWithConnectionFailure() {
139         Map<String, Exception> actual = DataSourcePoolPropertiesValidator.validate(
140                 Collections.singletonMap("foo_ds", new DataSourcePoolProperties(HikariDataSource.class.getName(), createHikariDataSourceProperties("jdbc:mock://127.0.0.1/invalid"))),
141                 Collections.singleton(PrivilegeCheckType.SELECT));
142         assertThat(actual.size(), is(1));
143         assertThat(actual.get("foo_ds"), isA(InvalidDataSourcePoolPropertiesException.class));
144     }

146     private Map<String, Object> createHikariDataSourceProperties(final String jdbcUrl) {
147         Map<String, Object> result = new LinkedHashMap<>(2, 1F);

149         result.put("jdbcUrl", jdbcUrl);
150         return result;

153     @Test
154     void assertValidateWithNullConnection() throws SQLException {
155         DataSource dataSource = mock(DataSource.class);
156         Map<String, Object> props = createMockedDataSourceProperties();
157         DataSourcePoolProperties dataSourcePoolProperties = new DataSourcePoolProperties(MockedDataSource.class.getName(), props);
158         when(dataSource.getConnection()).thenReturn(null);
159         try (MockedStatic<DataSourcePoolCreator> dataSourcePoolCreator = mockStatic(DataSourcePoolCreator.class)) {
160             dataSourcePoolCreator.when(() -> DataSourcePoolCreator.create(dataSourcePoolProperties)).thenReturn(dataSource);
161             assertTrue(DataSourcePoolPropertiesValidator.validate(Collections.singletonMap("foo_ds", dataSourcePoolProperties), Collections.emptySet()).isEmpty());
162         }
163     }

165     @Test
166     void assertValidateWithPrivilegesAndChecker() {
167         DatabaseType databaseType = mock(DatabaseType.class);
168         DialectDatabasePrivilegeChecker checker = mock(DialectDatabasePrivilegeChecker.class);
169         try (
170                 MockedStatic<DatabaseTypeFactory> databaseTypeFactory = mockStatic(DatabaseTypeFactory.class);
171                 MockedStatic<DatabaseTypedSPILoader> databaseTypedSPILoader = mockStatic(DatabaseTypedSPILoader.class)) {
172             databaseTypeFactory.when(() -> DatabaseTypeFactory.get(MOCKED_URL)).thenReturn(databaseType);
173             databaseTypedSPILoader.when(() -> DatabaseTypedSPILoader.findService(DialectDatabasePrivilegeChecker.class, databaseType)).thenReturn(Optional.of(checker));
174             Map<String, Object> props = createMockedDataSourceProperties();
175             assertTrue(DataSourcePoolPropertiesValidator.validate(Collections.singletonMap("foo_ds", new DataSourcePoolProperties(MockedDataSource.class.getName(), props)),
176                     Arrays.asList(PrivilegeCheckType.SELECT, PrivilegeCheckType.XA)).isEmpty());
177             verify(checker).check(any(DataSource.class), eq(PrivilegeCheckType.SELECT));
178             verify(checker).check(any(DataSource.class), eq(PrivilegeCheckType.XA));
179         }
180     }

186         result.put("password", "root");
187         result.put("driverClassName", MockedDataSource.class.getName());
188         result.put("connectionTimeout", "120");
189         return result;
190     }
