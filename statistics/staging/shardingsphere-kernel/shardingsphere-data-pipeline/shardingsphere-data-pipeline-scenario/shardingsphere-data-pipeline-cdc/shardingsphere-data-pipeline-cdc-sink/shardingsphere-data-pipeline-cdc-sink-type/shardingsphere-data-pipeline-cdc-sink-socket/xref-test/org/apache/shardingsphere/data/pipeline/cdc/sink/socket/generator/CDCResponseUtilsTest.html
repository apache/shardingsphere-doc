18  package org.apache.shardingsphere.data.pipeline.cdc.sink.socket.generator;
19  
20  import com.google.protobuf.Message;
21  import org.apache.shardingsphere.data.pipeline.cdc.protocol.response.CDCResponse;
22  import org.apache.shardingsphere.data.pipeline.cdc.protocol.response.DataRecordResult;
23  import org.apache.shardingsphere.data.pipeline.cdc.protocol.response.ServerGreetingResult;
24  import org.apache.shardingsphere.data.pipeline.cdc.protocol.response.StreamDataResult;
25  import org.apache.shardingsphere.infra.exception.external.sql.sqlstate.XOpenSQLState;
26  import org.junit.jupiter.api.Test;
27  
28  import static org.hamcrest.MatcherAssert.assertThat;
29  import static org.hamcrest.Matchers.is;
30  import static org.junit.jupiter.api.Assertions.assertNotNull;
31  
32  class CDCResponseUtilsTest {
33      
34      @Test
35      void assertSucceedWhenResponseNotSet() {
36          CDCResponse actualResponse = CDCResponseUtils.succeed("request_id_1");
37          assertThat(actualResponse.getStatus(), is(CDCResponse.Status.SUCCEED));
38          assertThat(actualResponse.getRequestId(), is("request_id_1"));
39      }
40      
41      @Test
42      void assertSucceedWhenResponseCaseServerGreetingResult() {
43          Message msg = ServerGreetingResult.newBuilder().build();
44          CDCResponse actualResponse = CDCResponseUtils.succeed("request_id_1", CDCResponse.ResponseCase.SERVER_GREETING_RESULT, msg);
45          assertThat(actualResponse.getStatus(), is(CDCResponse.Status.SUCCEED));
46          assertThat(actualResponse.getRequestId(), is("request_id_1"));
47          assertNotNull(actualResponse.getServerGreetingResult());
48      }
49      
50      @Test
51      void assertSucceedWhenResponseCaseDataRecordResult() {
52          Message msg = DataRecordResult.newBuilder().build();
53          CDCResponse actualResponse = CDCResponseUtils.succeed("request_id_1", CDCResponse.ResponseCase.DATA_RECORD_RESULT, msg);
54          assertThat(actualResponse.getStatus(), is(CDCResponse.Status.SUCCEED));
55          assertThat(actualResponse.getRequestId(), is("request_id_1"));
56          assertNotNull(actualResponse.getDataRecordResult());
57      }
58      
59      @Test
60      void assertSucceedWhenResponseCaseStreamDataResult() {
61          Message msg = StreamDataResult.newBuilder().build();
62          CDCResponse actualResponse = CDCResponseUtils.succeed("request_id_1", CDCResponse.ResponseCase.STREAM_DATA_RESULT, msg);
63          assertThat(actualResponse.getStatus(), is(CDCResponse.Status.SUCCEED));
64          assertThat(actualResponse.getRequestId(), is("request_id_1"));
65          assertNotNull(actualResponse.getStreamDataResult());
66      }
67      
68      @Test
69      void assertFailed() {
70          CDCResponse actualResponse = CDCResponseUtils.failed("request_id_1", XOpenSQLState.GENERAL_ERROR.getValue(), "Error");
71          assertThat(actualResponse.getStatus(), is(CDCResponse.Status.FAILED));
72          assertThat(actualResponse.getRequestId(), is("request_id_1"));
73          assertThat(actualResponse.getErrorCode(), is(XOpenSQLState.GENERAL_ERROR.getValue()));
74          assertThat(actualResponse.getErrorMessage(), is("Error"));
75      }
76  }
