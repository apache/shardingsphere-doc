421     private RelDataType createRowType(final boolean nullable, final int precision, final int scale) {
422         RelDataType result = mock(RelDataType.class);
423         when(result.isNullable()).thenReturn(nullable);
424         when(result.getPrecision()).thenReturn(precision);
425         when(result.getScale()).thenReturn(scale);
426         return result;
427     }

429     @SuppressWarnings("unchecked")
430     private Schema createSchemaWithTable(final String tableName, final RelDataType rowType) {
431         Table table = mock(Table.class);
432         when(table.getRowType(any(RelDataTypeFactory.class))).thenReturn(rowType);
433         Lookup<Table> tables = mock(Lookup.class);
434         when(tables.get(tableName)).thenReturn(table);
435         Schema result = mock(Schema.class);
436         when(result.tables()).thenReturn(tables);
437         return result;
438     }
439     
440     private RelDataType createResultType(final String[] names, final RelDataType... types) {
441         List<RelDataTypeField> fields = new ArrayList<>(names.length);
442         for (int i = 0; i < names.length; i++) {
443             RelDataTypeField field = mock(RelDataTypeField.class);
444             when(field.getName()).thenReturn(names[i]);
445             when(field.getType()).thenReturn(types[i]);
446             fields.add(field);
447         }
448         RelDataType result = mock(RelDataType.class);
449         when(result.getFieldList()).thenReturn(fields);
450         return result;
451     }
