40      @Test
41      void assertCreateBroadListResponseGuidesSearch() {
42          Map<String, Object> actual = createResponse("shardingsphere://databases", new MCPResourceURIVariables(Map.of()), createDatabases(101)).toPayload();
43          assertThat(((List<?>) actual.get("items")).size(), is(100));
44          assertThat(actual.get("count"), is(100));
45          assertThat(actual.get("total_count"), is(101));
46          assertThat(actual.get("summary"), is("Returned 100 of 101 logical-database metadata entries."));
47          assertTrue((Boolean) actual.get("truncated"));
48          assertThat(actual.get("continuation_mode"), is("metadata_search"));
49          assertThat(((Map<?, ?>) actual.get("large_result_guidance")).get("state"), is("broad_metadata_list"));
50          assertThat(((Map<?, ?>) actual.get("large_result_guidance")).get("threshold"), is(100));
51          Map<?, ?> nextAction = (Map<?, ?>) ((List<?>) actual.get("next_actions")).getFirst();
52          assertThat(nextAction.get("tool_name"), is("database_gateway_search_metadata"));
53          assertFalse(((Map<?, ?>) nextAction.get("arguments")).containsKey("page_size"));
54          assertThat(((Map<?, ?>) nextAction.get("arguments")).get("object_types"), is(List.of("database")));
55      }
56      
57      @Test
58      void assertCreateDetailResponse() {
59          MCPResourceURIVariables uriVariables = new MCPResourceURIVariables(Map.of("database", "逻辑 库/2026?"));
60          Map<String, Object> actual = createResponse("shardingsphere://databases/{database}", uriVariables,
61                  List.of(Map.of("database", uriVariables.getValue("database")))).toPayload();
62          assertThat(actual.get("response_mode"), is("detail"));
63          assertThat(actual.get("summary"), is("Returned logical-database detail for this resource URI."));
64          assertThat(actual.get("resource_kind"), is("detail"));
65          assertThat(actual.get("object_scope"), is("logical-database"));
66          assertThat(actual.get("items"), is(List.of(Map.of("database", "逻辑 库/2026?"))));
67          String expectedSelfUri = "shardingsphere://databases/%E9%80%BB%E8%BE%91%20%E5%BA%93%2F2026%3F";
68          Map<?, ?> actualSelfResource = (Map<?, ?>) actual.get("self_resource");
69          assertThat(actualSelfResource.get("uri"), is(expectedSelfUri));
70          assertThat(actualSelfResource.get("resource_kind"), is("logical-database"));
71          Map<?, ?> actualParentResource = (Map<?, ?>) actual.get("parent_resource");
72          assertThat(actualParentResource.get("uri"), is("shardingsphere://databases"));
73          assertThat(actualParentResource.get("resource_kind"), is("logical-database"));
74          List<String> nextResourceUris = ((List<?>) actual.get("next_resources")).stream().map(each -> (String) ((Map<?, ?>) each).get("uri")).toList();
75          assertThat(nextResourceUris, is(List.of(
76                  expectedSelfUri + "/schemas",
77                  expectedSelfUri + "/storage-units",
78                  expectedSelfUri + "/single-tables",
79                  expectedSelfUri + "/single-table/default-storage-unit")));
80          List<String> nextResourceKinds = ((List<?>) actual.get("next_resources")).stream().map(each -> (String) ((Map<?, ?>) each).get("resource_kind")).toList();
81          assertThat(nextResourceKinds, is(List.of("schema", "storage-unit", "single-table", "single-table")));
82      }
83      
84      @Test
85      void assertCreateDefaultStorageUnitResponsePreservesNavigationResourceKinds() {
86          MCPResourceURIVariables uriVariables = new MCPResourceURIVariables(Map.of("database", "logic_db"));
87          Map<String, Object> actual = createResponse("shardingsphere://databases/{database}/single-table/default-storage-unit", uriVariables,
88                  List.of(Map.of("storage_unit", "ds_0"))).toPayload();
89          assertThat(((Map<?, ?>) actual.get("self_resource")).get("resource_kind"), is("single-table"));
90          Map<?, ?> actualParentResource = (Map<?, ?>) actual.get("parent_resource");
91          assertThat(actualParentResource.get("uri"), is("shardingsphere://databases/logic_db/single-table"));
92          assertThat(actualParentResource.get("resource_kind"), is("logical-database"));
93      }
94      
95      @Test
96      void assertCreateMissingDetailResponse() {
97          Map<String, Object> actual = createResponse("shardingsphere://databases/{database}", mock(MCPResourceURIVariables.class), List.of()).toPayload();
98          assertThat(actual.get("summary"), is("No logical-database detail item matched this resource URI."));
99          assertThat(actual.get("items"), is(List.of()));
100         assertThat(((Map<?, ?>) actual.get("empty_state")).get("reason"), is("logical-database detail resource was not found for this URI."));
101         assertThat(((Map<?, ?>) actual.get("recovery")).get("recovery_category"), is("not_found"));
102         assertThat(((Map<?, ?>) ((List<?>) actual.get("next_actions")).getFirst()).get("type"), is("terminal"));
103     }
104     
105     private MCPSuccessPayload createResponse(final String uriTemplate, final MCPResourceURIVariables uriVariables, final List<?> items) {
106         MCPResourceDescriptor descriptor = MCPDescriptorCatalogIndex.getRequiredResourceDescriptor(uriTemplate);
107         ShardingSphereMCPResourceMetadata metadata = MCPDescriptorCatalogIndex.getRequiredShardingSphereResourceMetadata(uriTemplate);
108         return new MetadataResourceResponseFactory().create(mock(MCPFeatureRequestContext.class), uriVariables, descriptor, metadata, items);
109     }
110     
111     private List<Map<String, String>> createDatabases(final int count) {
112         List<Map<String, String>> result = new LinkedList<>();
113         for (int i = 0; i < count; i++) {
114             result.add(Map.of("database", String.format("logic_db_%03d", i)));
115         }
116         return result;
117     }
