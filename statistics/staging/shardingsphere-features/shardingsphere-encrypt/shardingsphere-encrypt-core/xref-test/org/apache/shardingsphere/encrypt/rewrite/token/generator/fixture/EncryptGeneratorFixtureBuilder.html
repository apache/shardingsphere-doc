85      public static EncryptRule createEncryptRule() {
86          Map<String, AlgorithmConfiguration> encryptors = new LinkedHashMap<>(2, 1F);
87          encryptors.put("standard_encryptor", new AlgorithmConfiguration("CORE.FIXTURE", new Properties()));
88          encryptors.put("assisted_encryptor", new AlgorithmConfiguration("CORE.QUERY_ASSISTED.FIXTURE", new Properties()));
89          encryptors.put("like_encryptor", new AlgorithmConfiguration("CORE.QUERY_LIKE.FIXTURE", new Properties()));
90          EncryptColumnRuleConfiguration pwdColumnConfig =
91                  new EncryptColumnRuleConfiguration("pwd", new EncryptColumnItemRuleConfiguration("pwd_cipher", "standard_encryptor"));
92          pwdColumnConfig.setAssistedQuery(new EncryptColumnItemRuleConfiguration("pwd_assist", "assisted_encryptor"));
93          pwdColumnConfig.setLikeQuery(new EncryptColumnItemRuleConfiguration("pwd_like", "like_encryptor"));
94          EncryptColumnRuleConfiguration userNameColumnConfig = new EncryptColumnRuleConfiguration("user_name", new EncryptColumnItemRuleConfiguration("user_name_cipher", "standard_encryptor"));
95          EncryptColumnRuleConfiguration userIdColumnConfig = new EncryptColumnRuleConfiguration("user_id", new EncryptColumnItemRuleConfiguration("user_id_cipher", "standard_encryptor"));
96          userIdColumnConfig.setAssistedQuery(new EncryptColumnItemRuleConfiguration("user_id_assist", "assisted_encryptor"));
97          EncryptColumnRuleConfiguration orderIdColumnConfig = new EncryptColumnRuleConfiguration("order_id", new EncryptColumnItemRuleConfiguration("order_id_cipher", "standard_encryptor"));
98          return new EncryptRule("foo_db",
99                  new EncryptRuleConfiguration(Arrays.asList(
100                         new EncryptTableRuleConfiguration("t_user", Arrays.asList(pwdColumnConfig, userNameColumnConfig, userIdColumnConfig)),
101                         new EncryptTableRuleConfiguration("t_order", Collections.singletonList(orderIdColumnConfig)),
102                         new EncryptTableRuleConfiguration("t_order_item", Collections.singletonList(new EncryptColumnRuleConfiguration("order_id",
103                                 new EncryptColumnItemRuleConfiguration("order_id_cipher", "standard_encryptor"))))),
104                         encryptors));
105     }
