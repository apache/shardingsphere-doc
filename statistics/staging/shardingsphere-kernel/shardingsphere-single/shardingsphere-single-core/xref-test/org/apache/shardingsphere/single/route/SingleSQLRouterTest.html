93      @Test
94      void assertCreateRouteContextWithSingleDataSource() throws SQLException {
95          ShardingSphereDatabase database = mock(ShardingSphereDatabase.class, RETURNS_DEEP_STUBS);
96          when(database.getResourceMetaData().getStorageUnits()).thenReturn(Collections.singletonMap("foo_ds", mock()));
97          SingleRule rule = createSingleRule();
98          RouteContext actual = singleSQLRouter.createRouteContext(createQueryContext(), mock(), database, rule, Collections.singletonList("foo_tbl"), new ConfigurationProperties(new Properties()));
99          assertThat(actual.getRouteUnits().size(), is(1));
100         RouteUnit routeUnit = actual.getRouteUnits().iterator().next();
101         assertThat(routeUnit.getDataSourceMapper().getLogicName(), is("foo_ds"));
102         assertThat(routeUnit.getDataSourceMapper().getActualName(), is("foo_ds"));
103         assertThat(routeUnit.getTableMappers().size(), is(1));
104         RouteMapper tableMapper = routeUnit.getTableMappers().iterator().next();
105         assertThat(tableMapper.getLogicName(), is("foo_tbl"));
106         assertThat(tableMapper.getActualName(), is("foo_tbl"));
107     }

259         Map<String, StorageUnit> storageUnits = new HashMap<>(2, 1F);

263         DataSourcePoolProperties dataSourcePoolProps1 = mock(DataSourcePoolProperties.class, RETURNS_DEEP_STUBS);
264         when(dataSourcePoolProps1.getConnectionPropertySynonyms().getStandardProperties()).thenReturn(ImmutableMap.of("url", "jdbc:mock://127.0.0.1/ds_1", "username", "test"));
265         storageUnits.put("ds_1", new StorageUnit(mock(StorageNode.class), dataSourcePoolProps1, new MockedDataSource()));
266         ShardingSphereDatabase result = mock(ShardingSphereDatabase.class, RETURNS_DEEP_STUBS);
267         when(result.getResourceMetaData().getStorageUnits()).thenReturn(storageUnits);
268         when(result.getName()).thenReturn("foo_db");
269         ShardingSphereSchema schema = mock(ShardingSphereSchema.class);
270         when(schema.containsTable("foo_tbl")).thenReturn(true);
271         when(result.getSchema(anyString())).thenReturn(schema);

298     private QueryContext createQueryContext() {
299         TableNameSegment tableNameSegment = new TableNameSegment(1, 2, new IdentifierValue("foo_tbl"));
300         tableNameSegment.setTableBoundInfo(new TableSegmentBoundInfo(new IdentifierValue("foo_db"), new IdentifierValue("foo_schema")));
301         ConnectionContext connectionContext = mock(ConnectionContext.class);
302         when(connectionContext.getCurrentDatabaseName()).thenReturn(Optional.of("foo_db"));
303         ShardingSphereMetaData metaData = mock(ShardingSphereMetaData.class);
304         when(metaData.containsDatabase("foo_db")).thenReturn(true);
305         when(metaData.getDatabase("foo_db")).thenReturn(mock(ShardingSphereDatabase.class));
306         return new QueryContext(new CommonSQLStatementContext(CreateTableStatement.builder().databaseType(databaseType).table(new SimpleTableSegment(tableNameSegment)).build()),
307                 "CREATE TABLE", Collections.emptyList(), new HintValueContext(), connectionContext, metaData);
308     }

319                 .thenReturn(Collections.singletonList(new SingleDataNodeRuleAttribute(new LinkedHashMap<>())));
320         RouteContext actual = new SingleSQLRouter().createRouteContext(
321                 createQueryContext(), mock(RuleMetaData.class), database, rule, Collections.singletonList("t_order"), new ConfigurationProperties(new Properties()));
322         assertThat(actual.getRouteUnits().size(), is(1));
323         RouteUnit routeUnit = actual.getRouteUnits().iterator().next();
324         assertThat(routeUnit.getDataSourceMapper().getLogicName(), is("logical_ds"));
325         assertThat(routeUnit.getDataSourceMapper().getActualName(), is("logical_ds"));
