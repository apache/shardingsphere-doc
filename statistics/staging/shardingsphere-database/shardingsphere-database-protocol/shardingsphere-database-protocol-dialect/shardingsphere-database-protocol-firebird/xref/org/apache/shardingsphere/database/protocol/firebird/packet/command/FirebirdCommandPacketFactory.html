52  /**
53   * Command packet factory for Firebird.
54   */
55  @NoArgsConstructor(access = AccessLevel.PRIVATE)
56  public final class FirebirdCommandPacketFactory {
57      
58      /**
59       * Create new instance of command packet.
60       *
61       * @param commandPacketType command packet type for Firebird
62       * @param payload packet payload for Firebird
63       * @param protocolVersion protocol version of Firebird
64       * @param connectionId connection ID
65       * @return created instance
66       */
67      public static FirebirdCommandPacket newInstance(final FirebirdCommandPacketType commandPacketType, final FirebirdPacketPayload payload, final FirebirdProtocolVersion protocolVersion,
68                                                      final int connectionId) {
69          switch (commandPacketType) {
70              case INFO_DATABASE:
71                  return FirebirdDatabaseInfoPacketType.createPacket(payload);
72              case INFO_BLOB:
73                  return FirebirdBlobInfoPacketType.createPacket(payload);
74              case TRANSACTION:
75                  return new FirebirdStartTransactionPacket(payload);
76              case CREATE_BLOB:
77              case CREATE_BLOB2:
78                  return new FirebirdCreateBlobCommandPacket(commandPacketType, payload);
79              case OPEN_BLOB:
80              case OPEN_BLOB2:
81                  return new FirebirdOpenBlobCommandPacket(commandPacketType, payload);
82              case GET_SEGMENT:
83                  return new FirebirdGetBlobSegmentCommandPacket(payload);
84              case PUT_SEGMENT:
85                  return new FirebirdPutBlobSegmentCommandPacket(payload);
86              case BATCH_SEGMENTS:
87                  return new FirebirdBatchBlobSegmentsCommandPacket(payload);
88              case SEEK_BLOB:
89                  return new FirebirdSeekBlobCommandPacket(payload);
90              case CANCEL_BLOB:
91                  return new FirebirdCancelBlobCommandPacket(payload);
92              case CLOSE_BLOB:
93                  return new FirebirdCloseBlobCommandPacket(payload);
94              case ALLOCATE_STATEMENT:
95                  return new FirebirdAllocateStatementPacket(payload);
96              case PREPARE_STATEMENT:
97                  return new FirebirdPrepareStatementPacket(payload);
98              case EXECUTE:
99              case EXECUTE2:
100                 return new FirebirdExecuteStatementPacket(payload, protocolVersion);
101             case FETCH:
102                 return new FirebirdFetchStatementPacket(payload);
103             case INFO_SQL:
104                 return FirebirdSQLInfoPacketType.createPacket(payload);
105             case COMMIT:
106                 return new FirebirdCommitTransactionPacket(payload);
107             case ROLLBACK:
108                 return new FirebirdRollbackTransactionPacket(payload);
109             case FREE_STATEMENT:
110                 return new FirebirdFreeStatementPacket(payload);
111             case BATCH_CREATE:
112                 return new FirebirdBatchCreateCommandPacket(payload);
113             case BATCH_MSG:
114                 return new FirebirdBatchMessageCommandPacket(payload);
115             case BATCH_EXEC:
116                 return new FirebirdBatchExecuteCommandPacket(payload);
117             case BATCH_RLS:
118                 return new FirebirdBatchReleaseCommandPacket(payload);
119             case BATCH_CANCEL:
120                 return new FirebirdBatchCancelCommandPacket(payload);
121             case BATCH_SYNC:
122                 return new FirebirdBatchSyncCommandPacket(payload);
123             default:
124                 return new FirebirdUnsupportedCommandPacket(commandPacketType);
125         }
126     }
127     
128     /**
129      * Get expected length of command packet including message type.
130      *
131      * @param commandPacketType command packet type for Firebird
132      * @param payload packet payload for Firebird
133      * @param protocolVersion protocol version of Firebird
134      * @param connectionId connection ID
135      * @return expected length of packet, or 0 if length is variable
136      */
137     public static int getExpectedLength(final FirebirdCommandPacketType commandPacketType, final FirebirdPacketPayload payload, final FirebirdProtocolVersion protocolVersion, final int connectionId) {
138         switch (commandPacketType) {
139             case INFO_DATABASE:
140             case INFO_SQL:
141             case INFO_BLOB:
142                 return FirebirdInfoPacket.getLength(payload);
143             case TRANSACTION:
144                 return FirebirdStartTransactionPacket.getLength(payload);
145             case CREATE_BLOB:
146             case CREATE_BLOB2:
147                 return FirebirdCreateBlobCommandPacket.getLength(commandPacketType, payload);
148             case OPEN_BLOB:
149             case OPEN_BLOB2:
150                 return FirebirdOpenBlobCommandPacket.getLength(commandPacketType, payload);
151             case GET_SEGMENT:
152                 return FirebirdGetBlobSegmentCommandPacket.getLength(payload);
153             case PUT_SEGMENT:
154                 return FirebirdPutBlobSegmentCommandPacket.getLength(payload);
155             case BATCH_SEGMENTS:
156                 return FirebirdBatchBlobSegmentsCommandPacket.getLength(payload);
157             case SEEK_BLOB:
158                 return FirebirdSeekBlobCommandPacket.getLength();
159             case CANCEL_BLOB:
160                 return FirebirdCancelBlobCommandPacket.getLength();
161             case CLOSE_BLOB:
162                 return FirebirdCloseBlobCommandPacket.getLength();
163             case ALLOCATE_STATEMENT:
164                 return FirebirdAllocateStatementPacket.getLength();
165             case PREPARE_STATEMENT:
166                 return FirebirdPrepareStatementPacket.getLength(payload);
167             case EXECUTE:
168             case EXECUTE2:
169                 return FirebirdExecuteStatementPacket.getLength(payload, protocolVersion);
170             case FETCH:
171                 return FirebirdFetchStatementPacket.getLength(payload);
172             case COMMIT:
173                 return FirebirdCommitTransactionPacket.getLength();
174             case ROLLBACK:
175                 return FirebirdRollbackTransactionPacket.getLength();
176             case FREE_STATEMENT:
177                 return FirebirdFreeStatementPacket.getLength();
178             case BATCH_CREATE:
179                 return FirebirdBatchCreateCommandPacket.getLength(payload);
180             case BATCH_MSG:
181                 return FirebirdBatchMessageCommandPacket.getLength(payload, connectionId);
182             case BATCH_EXEC:
183                 return FirebirdBatchExecuteCommandPacket.getLength();
184             case BATCH_RLS:
185                 return FirebirdBatchReleaseCommandPacket.getLength();
186             case BATCH_CANCEL:
187                 return FirebirdBatchCancelCommandPacket.getLength();
188             case BATCH_SYNC:
189                 return FirebirdBatchSyncCommandPacket.getLength();
190             default:
191                 return 0;
192         }
193     }
