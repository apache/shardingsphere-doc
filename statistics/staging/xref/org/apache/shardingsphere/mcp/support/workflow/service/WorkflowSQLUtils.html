37  /**
38   * Workflow SQL utilities.
39   */
40  @NoArgsConstructor(access = AccessLevel.PRIVATE)
41  public final class WorkflowSQLUtils {
42      
43      private static final Pattern UNQUOTED_IDENTIFIER_PATTERN = Pattern.compile("[A-Za-z_][A-Za-z0-9_$]*");
44      
45      private static final Set<String> DIST_SQL_RESERVED_IDENTIFIERS = Set.of(
46              "address_random_replace", "aes", "algorithm", "alter", "and", "assisted_query", "assisted_query_algorithm", "assisted_query_column", "by", "cipher", "column", "columns", "count",
47              "create", "delete", "drop", "encrypt", "encrypt_algorithm", "exists", "false", "from", "generic_table_random_replace", "group", "if", "index", "insert", "keep_first_n_last_m",
48              "keep_from_x_to_y", "key", "like_query", "like_query_algorithm", "like_query_column", "mask", "mask_after_special_chars", "mask_before_special_chars", "mask_first_n_last_m",
49              "mask_from_x_to_y", "md5", "name", "not", "order", "plugins", "properties", "rule", "rules", "select", "show", "table", "true", "type", "update", "where");
50      
51      private static final char BACK_QUOTE = '`';
52      
53      private static final char DOUBLE_QUOTE = '"';
54      
55      /**
56       * Normalize a SQL identifier from user input.
57       *
58       * @param identifier identifier to normalize
59       * @return normalized identifier
60       */
61      public static String normalizeIdentifier(final String identifier) {
62          String result = trimToEmpty(identifier);
63          if (isDelimitedIdentifier(result, BACK_QUOTE, BACK_QUOTE)) {
64              return result.substring(1, result.length() - 1).replace("``", "`");
65          }
66          if (isDelimitedIdentifier(result, DOUBLE_QUOTE, DOUBLE_QUOTE)) {
67              return result.substring(1, result.length() - 1).replace("\"\"", "\"");
68          }
69          if (isDelimitedIdentifier(result, '[', ']')) {
70              return result.substring(1, result.length() - 1).replace("]]", "]");
71          }
72          return result;
73      }
74      
75      /**
76       * Check whether an identifier is supported by workflow planning.
77       *
78       * @param identifier identifier to check
79       * @return whether the identifier is supported
80       */
81      public static boolean isSupportedIdentifier(final String identifier) {
82          return !containsUnsupportedIdentifierCharacter(normalizeIdentifier(identifier));
83      }
84      
85      /**
86       * Validate a workflow SQL identifier.
87       *
88       * @param fieldName field name for error reporting
89       * @param identifier identifier to check
90       * @throws MCPInvalidRequestException when the identifier contains unsupported characters
91       */
92      public static void checkSupportedIdentifier(final String fieldName, final String identifier) {
93          String actualIdentifier = normalizeIdentifier(identifier);
94          ShardingSpherePreconditions.checkState(!containsUnsupportedIdentifierCharacter(actualIdentifier),
95                  () -> new MCPInvalidRequestException(String.format(
96                          "%s `%s` contains unsupported characters that cannot be rendered as a reviewable SQL identifier.", fieldName, actualIdentifier)));
97      }
98      
99      /**
100      * Format a DistSQL identifier.
101      *
102      * @param identifier identifier to format
103      * @return formatted DistSQL identifier
104      */
105     public static String formatDistSQLIdentifier(final String identifier) {
106         String rawIdentifier = trimToEmpty(identifier);
107         String actualIdentifier = normalizeIdentifier(rawIdentifier);
108         checkSupportedIdentifier("identifier", actualIdentifier);
109         return actualIdentifier.isEmpty() || !isSpecialDistSQLIdentifier(actualIdentifier) && !isDelimitedIdentifier(rawIdentifier)
110                 ? actualIdentifier
111                 : wrapIdentifier(QuoteCharacter.BACK_QUOTE, actualIdentifier);
112     }
113     
114     /**
115      * Format an identifier rendered by generated rule DistSQL artifacts.
116      *
117      * @param identifier identifier to format
118      * @return formatted DistSQL identifier
119      */
120     public static String formatGeneratedRuleDistSQLIdentifier(final String identifier) {
121         String actualIdentifier = normalizeIdentifier(trimToEmpty(identifier));
122         checkSupportedIdentifier("identifier", actualIdentifier);
123         return actualIdentifier.isEmpty() ? actualIdentifier : wrapIdentifier(QuoteCharacter.BACK_QUOTE, actualIdentifier);
124     }
125     
126     /**
127      * Judge whether a workflow identifier token references an existing identifier under the target database policy.
128      *
129      * @param identifierContext identifier context
130      * @param identifierScope identifier scope
131      * @param identifier identifier token
132      * @param existingIdentifier existing identifier
133      * @return whether the identifier references the existing identifier
134      */
135     public static boolean isSameIdentifier(final DatabaseIdentifierContext identifierContext, final IdentifierScope identifierScope,
136                                            final String identifier, final String existingIdentifier) {
137         String actualIdentifier = normalizeIdentifier(identifier);
138         String actualExistingIdentifier = normalizeIdentifier(existingIdentifier);
139         return identifierContext.matchesMetaData(identifierScope, actualExistingIdentifier, new IdentifierValue(actualIdentifier, getQuoteCharacter(identifier)));
140     }
141     
142     static boolean requiresExactIdentifierMatch(final String identifier) {
143         String rawIdentifier = trimToEmpty(identifier);
144         return isDelimitedIdentifier(rawIdentifier) || isSpecialSQLIdentifier(normalizeIdentifier(rawIdentifier));
145     }
146     
147     /**
148      * Escape a SQL string literal value.
149      *
150      * @param value raw value
151      * @return escaped literal value
152      */
153     public static String escapeLiteral(final String value) {
154         return trimToEmpty(value).replace("'", "''");
155     }
156     
157     /**
158      * Create an algorithm fragment for DistSQL.
159      *
160      * @param algorithmType algorithm type
161      * @param properties algorithm properties
162      * @return DistSQL algorithm fragment
163      */
164     public static String createAlgorithmFragment(final String algorithmType, final Map<String, String> properties) {
165         String actualType = trimToEmpty(algorithmType).toLowerCase(Locale.ENGLISH);
166         return createAlgorithmFragmentWithType(actualType, properties);
167     }
168     
169     /**
170      * Create an algorithm fragment for DistSQL while preserving the algorithm type case.
171      *
172      * @param algorithmType algorithm type
173      * @param properties algorithm properties
174      * @return DistSQL algorithm fragment
175      */
176     public static String createAlgorithmFragmentWithExactType(final String algorithmType, final Map<String, String> properties) {
177         return createAlgorithmFragmentWithType(trimToEmpty(algorithmType), properties);
178     }
179     
180     private static String createAlgorithmFragmentWithType(final String algorithmType, final Map<String, String> properties) {
181         if (algorithmType.isEmpty()) {
182             return "";
183         }
184         Properties actualProperties = WorkflowAlgorithmUtils.createProperties(properties);
185         return actualProperties.isEmpty()
186                 ? String.format("TYPE(NAME='%s')", escapeLiteral(algorithmType))
187                 : String.format("TYPE(NAME='%s', PROPERTIES(%s))", escapeLiteral(algorithmType), createPropertiesFragment(actualProperties));
188     }
189     
190     private static String createPropertiesFragment(final Properties props) {
191         return new TreeMap<>(props).entrySet().stream()
192                 .map(entry -> String.format("'%s'='%s'", escapeLiteral(String.valueOf(entry.getKey())), escapeLiteral(String.valueOf(entry.getValue()))))
193                 .collect(Collectors.joining(", "));
194     }
195     
196     private static boolean isDelimitedIdentifier(final String identifier) {
197         return isDelimitedIdentifier(identifier, BACK_QUOTE, BACK_QUOTE) || isDelimitedIdentifier(identifier, DOUBLE_QUOTE, DOUBLE_QUOTE) || isDelimitedIdentifier(identifier, '[', ']');
198     }
199     
200     private static boolean isDelimitedIdentifier(final String identifier, final char startDelimiter, final char endDelimiter) {
201         return identifier.length() >= 2 && startDelimiter == identifier.charAt(0) && endDelimiter == identifier.charAt(identifier.length() - 1);
202     }
203     
204     private static boolean containsUnsupportedIdentifierCharacter(final String identifier) {
205         return identifier.chars().anyMatch(each -> BACK_QUOTE == each || 0 == each || '\r' == each || '\n' == each);
206     }
207     
208     private static boolean isSpecialDistSQLIdentifier(final String identifier) {
209         return !UNQUOTED_IDENTIFIER_PATTERN.matcher(identifier).matches() || DIST_SQL_RESERVED_IDENTIFIERS.contains(identifier.toLowerCase(Locale.ENGLISH));
210     }
211     
212     private static boolean isSpecialSQLIdentifier(final String identifier) {
213         return !UNQUOTED_IDENTIFIER_PATTERN.matcher(identifier).matches();
214     }
215     
216     private static QuoteCharacter getQuoteCharacter(final String identifier) {
217         String actualIdentifier = trimToEmpty(identifier);
218         return isDelimitedIdentifier(actualIdentifier) ? QuoteCharacter.getQuoteCharacter(actualIdentifier) : QuoteCharacter.NONE;
219     }
220     
221     private static String wrapIdentifier(final QuoteCharacter quoteCharacter, final String value) {
222         return QuoteCharacter.NONE == quoteCharacter
223                 ? value
224                 : quoteCharacter.getStartDelimiter() + value.replace(quoteCharacter.getEndDelimiter(), quoteCharacter.getEndDelimiter() + quoteCharacter.getEndDelimiter())
225                         + quoteCharacter.getEndDelimiter();
226     }
227     
228     private static String trimToEmpty(final String value) {
229         return null == value ? "" : value.trim();
230     }
