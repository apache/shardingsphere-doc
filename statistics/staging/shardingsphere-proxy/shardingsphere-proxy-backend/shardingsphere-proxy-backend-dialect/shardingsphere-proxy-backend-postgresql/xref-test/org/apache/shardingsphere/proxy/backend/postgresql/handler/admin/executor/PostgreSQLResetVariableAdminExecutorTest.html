40  import static org.mockito.Mockito.RETURNS_DEEP_STUBS;

43  import static org.mockito.Mockito.never;

47  class PostgreSQLResetVariableAdminExecutorTest {
48      
49      private final DatabaseType databaseType = TypedSPILoader.getService(DatabaseType.class, "PostgreSQL");
50      
51      @Test
52      void assertExecute() {
53          PostgreSQLResetVariableAdminExecutor executor = new PostgreSQLResetVariableAdminExecutor(new PostgreSQLResetParameterStatement(databaseType, "key"));
54          ConnectionSession connectionSession = mock(ConnectionSession.class, RETURNS_DEEP_STUBS);
55          RequiredSessionVariableRecorder requiredSessionVariableRecorder = mock(RequiredSessionVariableRecorder.class);
56          ProxyDatabaseConnectionManager databaseConnectionManager = mock(ProxyDatabaseConnectionManager.class);
57          when(connectionSession.getRequiredSessionVariableRecorder()).thenReturn(requiredSessionVariableRecorder);
58          when(connectionSession.getDatabaseConnectionManager()).thenReturn(databaseConnectionManager);
59          try (MockedStatic<DatabaseTypedSPILoader> databaseTypedSPILoader = mockStatic(DatabaseTypedSPILoader.class)) {
60              databaseTypedSPILoader.when(() -> DatabaseTypedSPILoader.findService(CharsetVariableProvider.class, databaseType)).thenReturn(Optional.empty());
61              ReplayedSessionVariableProvider replayedSessionVariableProvider = mock(ReplayedSessionVariableProvider.class);
62              when(replayedSessionVariableProvider.isNeedToReplay("key")).thenReturn(true);
63              databaseTypedSPILoader.when(() -> DatabaseTypedSPILoader.findService(ReplayedSessionVariableProvider.class, databaseType)).thenReturn(Optional.of(replayedSessionVariableProvider));
64              executor.execute(connectionSession, mock());
65              verify(requiredSessionVariableRecorder).setVariable("key", "DEFAULT");
66              verify(databaseConnectionManager).markSessionVariablesDirty();
67          }
68      }
69      
70      @Test
71      void assertExecuteWithClientEncoding() {
72          final PostgreSQLResetVariableAdminExecutor executor = new PostgreSQLResetVariableAdminExecutor(new PostgreSQLResetParameterStatement(databaseType, "client_encoding"));
73          ConnectionSession connectionSession = mock(ConnectionSession.class);
74          RequiredSessionVariableRecorder requiredSessionVariableRecorder = mock(RequiredSessionVariableRecorder.class);
75          AttributeMap attributeMap = mock(AttributeMap.class);
76          @SuppressWarnings("unchecked")
77          Attribute<Charset> attribute = mock(Attribute.class);
78          when(connectionSession.getRequiredSessionVariableRecorder()).thenReturn(requiredSessionVariableRecorder);
79          when(connectionSession.getAttributeMap()).thenReturn(attributeMap);
80          when(attributeMap.attr(CommonConstants.CHARSET_ATTRIBUTE_KEY)).thenReturn(attribute);
81          try (MockedStatic<DatabaseTypedSPILoader> databaseTypedSPILoader = mockStatic(DatabaseTypedSPILoader.class)) {
82              databaseTypedSPILoader.when(() -> DatabaseTypedSPILoader.findService(CharsetVariableProvider.class, databaseType)).thenReturn(Optional.of(new PostgreSQLCharsetVariableProvider()));
83              ReplayedSessionVariableProvider replayedSessionVariableProvider = mock(ReplayedSessionVariableProvider.class);
84              when(replayedSessionVariableProvider.isNeedToReplay("client_encoding")).thenReturn(true);
85              databaseTypedSPILoader.when(() -> DatabaseTypedSPILoader.findService(ReplayedSessionVariableProvider.class, databaseType)).thenReturn(Optional.of(replayedSessionVariableProvider));
86              executor.execute(connectionSession, mock());
87              verify(attribute).set(StandardCharsets.UTF_8);
88              verify(requiredSessionVariableRecorder, never()).setVariable("client_encoding", "DEFAULT");
89          }
90      }
91  }
