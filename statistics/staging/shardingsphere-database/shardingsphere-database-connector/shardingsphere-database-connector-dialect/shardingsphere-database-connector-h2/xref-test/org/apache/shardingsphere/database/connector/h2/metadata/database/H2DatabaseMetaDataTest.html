32  import static org.hamcrest.MatcherAssert.assertThat;
33  import static org.hamcrest.Matchers.is;
34  import static org.hamcrest.Matchers.isA;
35  import static org.junit.jupiter.api.Assertions.assertFalse;
36  import static org.junit.jupiter.api.Assertions.assertTrue;
37  
38  class H2DatabaseMetaDataTest {
39      
40      private final DialectDatabaseMetaData dialectDatabaseMetaData = DatabaseTypedSPILoader.getService(DialectDatabaseMetaData.class, TypedSPILoader.getService(DatabaseType.class, "H2"));
41      
42      @Test
43      void assertGetQuoteCharacter() {
44          assertThat(dialectDatabaseMetaData.getQuoteCharacter(), is(QuoteCharacter.QUOTE));
45      }
46      
47      @Test
48      void assertGetDefaultNullsOrderType() {
49          assertThat(dialectDatabaseMetaData.getDefaultNullsOrderType(), is(NullsOrderType.LOW));
50      }
51      
52      @Test
53      void assertGetIdentifierPatternType() {
54          assertThat(dialectDatabaseMetaData.getIdentifierPatternType(), is(IdentifierPatternType.KEEP_ORIGIN));
55      }
56      
57      @Test
58      void assertGetTransactionOption() {
59          DialectTransactionOption actual = dialectDatabaseMetaData.getTransactionOption();
60          assertFalse(actual.isSupportGlobalCSN());
61          assertThat(actual.getDDLCommitPolicy(), is(DDLCommitPolicy.NO_ADDITIONAL_COMMIT));
62          assertFalse(actual.isSupportAutoCommitInNestedTransaction());
63          assertFalse(actual.isSupportDDLInXATransaction());
64          assertTrue(actual.isSupportMetaDataRefreshInTransaction());
65          assertFalse(actual.isReturnRollbackStatementWhenCommitFailed());
66          assertFalse(actual.isAllowCommitAndRollbackOnlyWhenTransactionFailed());
67          assertThat(actual.getXaDriverClassNames().size(), is(1));
68          assertTrue(actual.getXaDriverClassNames().contains("org.h2.jdbcx.JdbcDataSource"));
69      }
70      
71      @Test
72      void assertGetFunctionOption() {
73          assertThat(dialectDatabaseMetaData.getFunctionOption(), isA(H2FunctionOption.class));
74      }
75  }
