38  /**
39   * Alter view statement context.
40   */
41  @Getter
42  public final class AlterViewStatementContext extends CommonSQLStatementContext implements TableAvailable {
43      
44      private final TablesContext tablesContext;
45      
46      private final SelectStatementContext selectStatementContext;
47      
48      public AlterViewStatementContext(final ShardingSphereMetaData metaData, final List<Object> params, final AlterViewStatement sqlStatement, final String currentDatabaseName) {
49          super(sqlStatement);
50          Collection<SimpleTableSegment> tables = new LinkedList<>();
51          tables.add(sqlStatement.getView());
52          Optional<SelectStatement> selectStatement = sqlStatement.getSelect();
53          selectStatement.ifPresent(optional -> extractTables(optional, tables));
54          sqlStatement.getRenameView().ifPresent(tables::add);
55          tablesContext = new TablesContext(tables);
56          selectStatementContext = selectStatement.map(optional -> createSelectStatementContext(metaData, params, optional, currentDatabaseName)).orElse(null);
57      }
58      
59      private SelectStatementContext createSelectStatementContext(final ShardingSphereMetaData metaData, final List<Object> params, final SelectStatement selectStatement,
60                                                                  final String currentDatabaseName) {
61          SelectStatementContext result = new SelectStatementContext(metaData, params, selectStatement, currentDatabaseName, Collections.emptyList());
62          result.setSubqueryType(SubqueryType.VIEW_DEFINITION);
63          return result;
64      }
65      
66      private void extractTables(final SelectStatement selectStatement, final Collection<SimpleTableSegment> tables) {
67          TableExtractor extractor = new TableExtractor();
68          extractor.extractTablesFromSelect(selectStatement);
69          tables.addAll(extractor.getRewriteTables());
70      }
71      
72      /**
73       * Get select statement context.
74       *
75       * @return select statement context
76       */
77      public Optional<SelectStatementContext> getSelectStatementContext() {
78          return Optional.ofNullable(selectStatementContext);
79      }
80      
81      @Override
82      public AlterViewStatement getSqlStatement() {
83          return (AlterViewStatement) super.getSqlStatement();
84      }
85  }
