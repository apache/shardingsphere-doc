18  package org.apache.shardingsphere.database.protocol.postgresql.packet.command.query.extended;
19  
20  import org.apache.shardingsphere.database.exception.core.exception.protocol.DatabaseProtocolException;
21  import org.junit.jupiter.api.Test;
22  import org.junit.jupiter.params.ParameterizedTest;
23  import org.junit.jupiter.params.provider.Arguments;
24  import org.junit.jupiter.params.provider.MethodSource;
25  
26  import java.sql.Types;
27  import java.util.stream.Stream;
28  
29  import static org.hamcrest.MatcherAssert.assertThat;
30  import static org.hamcrest.Matchers.is;
31  import static org.junit.jupiter.api.Assertions.assertThrows;
32  
33  class PostgreSQLBinaryColumnTypeTest {
34      
35      @ParameterizedTest(name = "{0}")
36      @MethodSource("assertValueOfJDBCTypeArguments")
37      void assertValueOfJDBCType(final String name, final int jdbcType, final PostgreSQLBinaryColumnType expectedColumnType) {
38          assertThat(PostgreSQLBinaryColumnType.valueOfJDBCType(jdbcType), is(expectedColumnType));
39      }
40      
41      @Test
42      void assertValueOfJDBCTypeThrowsIllegalArgumentException() {
43          assertThrows(IllegalArgumentException.class, () -> PostgreSQLBinaryColumnType.valueOfJDBCType(Types.REF_CURSOR));
44      }
45      
46      @ParameterizedTest(name = "{0}")
47      @MethodSource("assertValueOfJDBCTypeWithColumnTypeNameArguments")
48      void assertValueOfJDBCTypeWithColumnTypeName(final String name, final int jdbcType, final String columnTypeName, final PostgreSQLBinaryColumnType expectedColumnType) {
49          assertThat(PostgreSQLBinaryColumnType.valueOfJDBCType(jdbcType, columnTypeName), is(expectedColumnType));
50      }
51      
52      @ParameterizedTest(name = "{0}")
53      @MethodSource("assertIsBitArguments")
54      void assertIsBit(final String name, final int jdbcType, final String columnTypeName, final boolean expected) {
55          assertThat(PostgreSQLBinaryColumnType.isBit(jdbcType, columnTypeName), is(expected));
56      }
57      
58      @ParameterizedTest(name = "{0}")
59      @MethodSource("assertIsBoolArguments")
60      void assertIsBool(final String name, final int jdbcType, final String columnTypeName, final boolean expected) {
61          assertThat(PostgreSQLBinaryColumnType.isBool(jdbcType, columnTypeName), is(expected));
62      }
63      
64      @ParameterizedTest(name = "{0}")
65      @MethodSource("assertIsUUIDArguments")
66      void assertIsUUID(final String name, final int jdbcType, final String columnTypeName, final boolean expected) {
67          assertThat(PostgreSQLBinaryColumnType.isUUID(jdbcType, columnTypeName), is(expected));
68      }
69      
70      @Test
71      void assertValueOf() {
72          assertThat(PostgreSQLBinaryColumnType.valueOf(20), is(PostgreSQLBinaryColumnType.INT8));
73      }
74      
75      @Test
76      void assertValueOfThrowsDatabaseProtocolException() {
77          assertThrows(DatabaseProtocolException.class, () -> PostgreSQLBinaryColumnType.valueOf(9999));
78      }
79      
80      private static Stream<Arguments> assertValueOfJDBCTypeArguments() {
81          return Stream.of(
82                  Arguments.of("tinyint", Types.TINYINT, PostgreSQLBinaryColumnType.INT2),
83                  Arguments.of("smallint", Types.SMALLINT, PostgreSQLBinaryColumnType.INT2),
84                  Arguments.of("integer", Types.INTEGER, PostgreSQLBinaryColumnType.INT4),
85                  Arguments.of("bigint", Types.BIGINT, PostgreSQLBinaryColumnType.INT8),
86                  Arguments.of("numeric", Types.NUMERIC, PostgreSQLBinaryColumnType.NUMERIC),
87                  Arguments.of("decimal", Types.DECIMAL, PostgreSQLBinaryColumnType.NUMERIC),
88                  Arguments.of("real", Types.REAL, PostgreSQLBinaryColumnType.FLOAT4),
89                  Arguments.of("double", Types.DOUBLE, PostgreSQLBinaryColumnType.FLOAT8),
90                  Arguments.of("char", Types.CHAR, PostgreSQLBinaryColumnType.CHAR),
91                  Arguments.of("varchar", Types.VARCHAR, PostgreSQLBinaryColumnType.VARCHAR),
92                  Arguments.of("binary", Types.BINARY, PostgreSQLBinaryColumnType.BYTEA),
93                  Arguments.of("bit", Types.BIT, PostgreSQLBinaryColumnType.BIT),
94                  Arguments.of("date", Types.DATE, PostgreSQLBinaryColumnType.DATE),
95                  Arguments.of("time", Types.TIME, PostgreSQLBinaryColumnType.TIME),
96                  Arguments.of("timestamp", Types.TIMESTAMP, PostgreSQLBinaryColumnType.TIMESTAMP),
97                  Arguments.of("other", Types.OTHER, PostgreSQLBinaryColumnType.JSON),
98                  Arguments.of("sqlxml", Types.SQLXML, PostgreSQLBinaryColumnType.XML),
99                  Arguments.of("boolean", Types.BOOLEAN, PostgreSQLBinaryColumnType.BOOL),
100                 Arguments.of("struct", Types.STRUCT, PostgreSQLBinaryColumnType.VARCHAR),
101                 Arguments.of("array", Types.ARRAY, PostgreSQLBinaryColumnType.TEXT_ARRAY));
102     }
103     
104     private static Stream<Arguments> assertValueOfJDBCTypeWithColumnTypeNameArguments() {
105         return Stream.of(
106                 Arguments.of("bit override", Types.BIT, "bit", PostgreSQLBinaryColumnType.BIT),
107                 Arguments.of("bool override", Types.BIT, "bool", PostgreSQLBinaryColumnType.BOOL),
108                 Arguments.of("uuid override", Types.OTHER, "uuid", PostgreSQLBinaryColumnType.UUID),
109                 Arguments.of("fallback to jdbc type map", Types.INTEGER, "INT4", PostgreSQLBinaryColumnType.INT4));
110     }
111     
112     private static Stream<Arguments> assertIsBitArguments() {
113         return Stream.of(
114                 Arguments.of("bit exact", Types.BIT, "bit", true),
115                 Arguments.of("bit ignore case", Types.BIT, "BiT", true),
116                 Arguments.of("bit type with bool name", Types.BIT, "bool", false),
117                 Arguments.of("non bit jdbc type", Types.INTEGER, "bit", false));
118     }
119     
120     private static Stream<Arguments> assertIsBoolArguments() {
121         return Stream.of(
122                 Arguments.of("bool exact", Types.BIT, "bool", true),
123                 Arguments.of("bool ignore case", Types.BIT, "BoOl", true),
124                 Arguments.of("bit type with bit name", Types.BIT, "bit", false),
125                 Arguments.of("non bit jdbc type", Types.BOOLEAN, "bool", false));
126     }
127     
128     private static Stream<Arguments> assertIsUUIDArguments() {
129         return Stream.of(
130                 Arguments.of("uuid exact", Types.OTHER, "uuid", true),
131                 Arguments.of("uuid ignore case", Types.OTHER, "UUID", true),
132                 Arguments.of("other type with non uuid name", Types.OTHER, "json", false),
133                 Arguments.of("non other jdbc type", Types.VARCHAR, "uuid", false));
134     }
135 }
