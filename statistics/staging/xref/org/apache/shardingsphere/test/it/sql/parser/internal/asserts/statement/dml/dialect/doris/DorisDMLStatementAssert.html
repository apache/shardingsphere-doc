38  /**
39   * Doris DML statement assert.
40   */
41  @NoArgsConstructor(access = AccessLevel.PRIVATE)
42  public final class DorisDMLStatementAssert {
43      
44      /**
45       * Assert Doris DML statement is correct with expected parser result.
46       *
47       * @param assertContext assert context
48       * @param actual actual DML statement
49       * @param expected expected parser result
50       */
51      public static void assertIs(final SQLCaseAssertContext assertContext, final DMLStatement actual, final SQLParserTestCase expected) {
52          if (actual instanceof DorisCreateRoutineLoadStatement) {
53              DorisCreateRoutineLoadStatementAssert.assertIs(assertContext, (DorisCreateRoutineLoadStatement) actual, (DorisCreateRoutineLoadStatementTestCase) expected);
54          } else if (actual instanceof DorisAlterRoutineLoadStatement) {
55              DorisAlterRoutineLoadStatementAssert.assertIs(assertContext, (DorisAlterRoutineLoadStatement) actual, (DorisAlterRoutineLoadStatementTestCase) expected);
56          } else if (actual instanceof DorisPauseRoutineLoadStatement) {
57              DorisPauseRoutineLoadStatementAssert.assertIs(assertContext, (DorisPauseRoutineLoadStatement) actual, (DorisPauseRoutineLoadStatementTestCase) expected);
58          } else if (actual instanceof DorisResumeRoutineLoadStatement) {
59              DorisResumeRoutineLoadStatementAssert.assertIs(assertContext, (DorisResumeRoutineLoadStatement) actual, (DorisResumeRoutineLoadStatementTestCase) expected);
60          } else if (actual instanceof DorisStopRoutineLoadStatement) {
61              DorisStopRoutineLoadStatementAssert.assertIs(assertContext, (DorisStopRoutineLoadStatement) actual, (DorisStopRoutineLoadStatementTestCase) expected);
62          } else if (actual instanceof DorisBrokerLoadStatement) {
63              DorisBrokerLoadStatementAssert.assertIs(assertContext, (DorisBrokerLoadStatement) actual, (DorisBrokerLoadStatementTestCase) expected);
64          }
65      }
66  }
