58  @NoArgsConstructor(access = AccessLevel.PRIVATE)

60      
61      /**
62       * Create meta data contexts.
63       *
64       * @param persistService persist service
65       * @param param context manager builder parameter
66       * @param instanceContext compute node instance context
67       * @return meta data contexts
68       * @throws SQLException SQL exception
69       */
70      public static MetaDataContexts create(final MetaDataPersistService persistService,
71                                            final ContextManagerBuilderParameter param, final ComputeNodeInstanceContext instanceContext) throws SQLException {
72          return isCreateByLocal(persistService) ? createByLocal(persistService, param, instanceContext) : createByRepository(persistService, param, instanceContext);
73      }
74      
75      private static boolean isCreateByLocal(final MetaDataPersistService persistService) {
76          return persistService.getDatabaseMetaDataFacade().getDatabase().loadAllDatabaseNames().isEmpty();
77      }
78      
79      private static MetaDataContexts createByLocal(final MetaDataPersistService persistService,
80                                                    final ContextManagerBuilderParameter param, final ComputeNodeInstanceContext instanceContext) throws SQLException {
81          Collection<RuleConfiguration> globalRuleConfigs = param.getGlobalRuleConfigs();
82          ConfigurationProperties props = new ConfigurationProperties(param.getProps());
83          Map<String, ShardingSphereDatabase> databases = ExternalMetaDataFactory.create(param.getDatabaseConfigs(), props, instanceContext);
84          MetaDataContexts result = newMetaDataContexts(persistService, param, globalRuleConfigs, databases, props);
85          persistDatabaseConfigurations(result, param, persistService, instanceContext);
86          persistMetaData(result, persistService);
87          return result;
88      }
89      
90      private static MetaDataContexts createByRepository(final MetaDataPersistService persistService, final ContextManagerBuilderParameter param, final ComputeNodeInstanceContext instanceContext) {
91          Map<String, DatabaseConfiguration> effectiveDatabaseConfigs = createEffectiveDatabaseConfigurations(
92                  getDatabaseNames(instanceContext, param.getDatabaseConfigs(), persistService), param.getDatabaseConfigs(), persistService);
93          Collection<RuleConfiguration> globalRuleConfigs = persistService.getGlobalRuleService().load();
94          ConfigurationProperties props = new ConfigurationProperties(persistService.getPropsService().load());
95          Map<String, ShardingSphereDatabase> databases = InternalMetaDataFactory.create(persistService, effectiveDatabaseConfigs, props, instanceContext);
96          return newMetaDataContexts(persistService, param, globalRuleConfigs, databases, props);
97      }
98      
99      private static MetaDataContexts newMetaDataContexts(final MetaDataPersistService persistService, final ContextManagerBuilderParameter param,
100                                                         final Collection<RuleConfiguration> globalRuleConfigs, final Map<String, ShardingSphereDatabase> databases,
101                                                         final ConfigurationProperties props) {
102         // TODO load global data sources from persist service
103         ResourceMetaData globalResourceMetaData = new ResourceMetaData(param.getGlobalDataSources());
104         RuleMetaData globalRuleMetaData = new RuleMetaData(GlobalRulesBuilder.buildRules(globalRuleConfigs, databases.values(), props));
105         ShardingSphereMetaData metaData = new ShardingSphereMetaData(databases.values(), globalResourceMetaData, globalRuleMetaData, props);
106         return new MetaDataContexts(metaData, ShardingSphereStatisticsFactory.create(persistService, metaData));
107     }
108     
109     private static Collection<String> getDatabaseNames(final ComputeNodeInstanceContext instanceContext,
110                                                        final Map<String, DatabaseConfiguration> databaseConfigs, final MetaDataPersistService persistService) {
111         return instanceContext.getInstance().getMetaData() instanceof JDBCInstanceMetaData
112                 ? databaseConfigs.keySet()
113                 : persistService.getDatabaseMetaDataFacade().getDatabase().loadAllDatabaseNames();
114     }
115     
116     private static Map<String, DatabaseConfiguration> createEffectiveDatabaseConfigurations(final Collection<String> databaseNames,
117                                                                                             final Map<String, DatabaseConfiguration> databaseConfigs, final MetaDataPersistService persistService) {
118         return databaseNames.stream().collect(Collectors.toMap(each -> each, each -> createEffectiveDatabaseConfiguration(each, databaseConfigs, persistService)));
119     }
120     
121     private static DatabaseConfiguration createEffectiveDatabaseConfiguration(final String databaseName,
122                                                                               final Map<String, DatabaseConfiguration> databaseConfigs, final MetaDataPersistService persistService) {
123         closeGeneratedDataSources(databaseName, databaseConfigs);
124         Map<String, DataSourceConfiguration> dataSources = persistService.loadDataSourceConfigurations(databaseName);
125         Collection<RuleConfiguration> databaseRuleConfigs = persistService.getDatabaseRulePersistService().load(databaseName);
126         return new DataSourceGeneratedDatabaseConfiguration(dataSources, databaseRuleConfigs);
127     }
128     
129     private static void closeGeneratedDataSources(final String databaseName, final Map<String, ? extends DatabaseConfiguration> databaseConfigs) {
130         if (databaseConfigs.containsKey(databaseName) && !databaseConfigs.get(databaseName).getStorageUnits().isEmpty()) {
131             databaseConfigs.get(databaseName).getDataSources().values().forEach(each -> new DataSourcePoolDestroyer(each).asyncDestroy());
132         }
133     }
134     
135     private static void persistDatabaseConfigurations(final MetaDataContexts metadataContexts,
136                                                       final ContextManagerBuilderParameter param, final MetaDataPersistService persistService, final ComputeNodeInstanceContext instanceContext) {
137         Collection<RuleConfiguration> globalRuleConfigs = metadataContexts.getMetaData().getGlobalRuleMetaData().getConfigurations();
138         persistService.persistGlobalRuleConfiguration(globalRuleConfigs, param.getProps());
139         for (Entry<String, ? extends DatabaseConfiguration> entry : param.getDatabaseConfigs().entrySet()) {
140             String databaseName = entry.getKey();
141             persistService.persistConfigurations(entry.getKey(), entry.getValue(),
142                     metadataContexts.getMetaData().getDatabase(databaseName).getResourceMetaData().getStorageUnits().entrySet().stream()
143                             .collect(Collectors.toMap(Entry::getKey, each -> each.getValue().getDataSource(), (oldValue, currentValue) -> oldValue, LinkedHashMap::new)),
144                     metadataContexts.getMetaData().getDatabase(databaseName).getRuleMetaData().getRules());
145         }
146     }
147     
148     private static void persistMetaData(final MetaDataContexts metaDataContexts, final MetaDataPersistService persistService) {
149         metaDataContexts.getMetaData().getAllDatabases().forEach(each -> each.getAllSchemas().forEach(schema -> {
150             if (schema.isEmpty()) {
151                 persistService.getDatabaseMetaDataFacade().getSchema().add(each.getName(), schema.getName());
152             }
153             persistService.getDatabaseMetaDataFacade().getTable().persist(each.getName(), schema.getName(), schema.getAllTables());
154         }));
155         for (Entry<String, ShardingSphereDatabaseData> databaseDataEntry : metaDataContexts.getStatistics().getDatabaseData().entrySet()) {
156             for (Entry<String, ShardingSphereSchemaData> schemaDataEntry : databaseDataEntry.getValue().getSchemaData().entrySet()) {
157                 persistService.getShardingSphereDataPersistService().persist(
158                         metaDataContexts.getMetaData().getDatabase(databaseDataEntry.getKey()), schemaDataEntry.getKey(), schemaDataEntry.getValue());
159             }
160         }
161     }

163     /**
164      * Create meta data contexts by switch resource.
165      *
166      * @param databaseName database name
167      * @param internalLoadMetaData internal load meta data
168      * @param switchingResource switching resource
169      * @param originalMetaDataContexts original meta data contexts
170      * @param persistService meta data persist service
171      * @param instanceContext compute node instance context
172      * @return meta data contexts
173      * @throws SQLException SQL exception
174      */
175     public static MetaDataContexts createBySwitchResource(final String databaseName, final boolean internalLoadMetaData, final SwitchingResource switchingResource,
176                                                           final MetaDataContexts originalMetaDataContexts, final MetaDataPersistService persistService,
177                                                           final ComputeNodeInstanceContext instanceContext) throws SQLException {
178         ShardingSphereDatabase changedDatabase = createChangedDatabase(
179                 databaseName, internalLoadMetaData, switchingResource, null, originalMetaDataContexts, persistService, instanceContext);
180         ConfigurationProperties props = originalMetaDataContexts.getMetaData().getProps();
181         ShardingSphereMetaData clonedMetaData = cloneMetaData(originalMetaDataContexts.getMetaData(), changedDatabase);
182         RuleMetaData changedGlobalMetaData = new RuleMetaData(
183                 GlobalRulesBuilder.buildRules(originalMetaDataContexts.getMetaData().getGlobalRuleMetaData().getConfigurations(), clonedMetaData.getAllDatabases(), props));
184         ShardingSphereMetaData metaData = new ShardingSphereMetaData(
185                 clonedMetaData.getAllDatabases(), originalMetaDataContexts.getMetaData().getGlobalResourceMetaData(), changedGlobalMetaData, props);
186         return new MetaDataContexts(metaData, ShardingSphereStatisticsFactory.create(persistService, metaData));
187     }

189     /**
190      * Create meta data contexts by alter rule.
191      *
192      * @param databaseName database name
193      * @param internalLoadMetaData internal load meta data
194      * @param ruleConfigs rule configs
195      * @param originalMetaDataContexts original meta data contexts
196      * @param persistService meta data persist service
197      * @param instanceContext compute node instance context
198      * @return meta data contexts
199      * @throws SQLException SQL exception
200      */
201     public static MetaDataContexts createByAlterRule(final String databaseName, final boolean internalLoadMetaData, final Collection<RuleConfiguration> ruleConfigs,
202                                                      final MetaDataContexts originalMetaDataContexts, final MetaDataPersistService persistService,
203                                                      final ComputeNodeInstanceContext instanceContext) throws SQLException {
204         ShardingSphereDatabase changedDatabase = createChangedDatabase(
205                 databaseName, internalLoadMetaData, null, ruleConfigs, originalMetaDataContexts, persistService, instanceContext);
206         ShardingSphereMetaData clonedMetaData = cloneMetaData(originalMetaDataContexts.getMetaData(), changedDatabase);
207         ConfigurationProperties props = originalMetaDataContexts.getMetaData().getProps();
208         RuleMetaData changedGlobalMetaData = new RuleMetaData(
209                 GlobalRulesBuilder.buildRules(originalMetaDataContexts.getMetaData().getGlobalRuleMetaData().getConfigurations(), clonedMetaData.getAllDatabases(), props));
210         ShardingSphereMetaData metaData = new ShardingSphereMetaData(
211                 clonedMetaData.getAllDatabases(), originalMetaDataContexts.getMetaData().getGlobalResourceMetaData(), changedGlobalMetaData, props);
212         return new MetaDataContexts(metaData, ShardingSphereStatisticsFactory.create(persistService, metaData));
213     }
214     
215     private static ShardingSphereMetaData cloneMetaData(final ShardingSphereMetaData originalMetaData, final ShardingSphereDatabase changedDatabase) {
216         ShardingSphereMetaData result = new ShardingSphereMetaData(
217                 originalMetaData.getAllDatabases(), originalMetaData.getGlobalResourceMetaData(), originalMetaData.getGlobalRuleMetaData(), originalMetaData.getProps());
218         result.putDatabase(changedDatabase);
219         return result;
220     }

222     /**
223      * Create changed database by switch resource.
224      *
225      * @param databaseName database name
226      * @param internalLoadMetaData internal load meta data
227      * @param switchingResource switching resource
228      * @param ruleConfigs rule configurations
229      * @param originalMetaDataContext original meta data contexts
230      * @param persistService meta data persist service
231      * @param instanceContext compute node instance context
232      * @return changed database
233      * @throws SQLException SQL exception
234      */
235     public static ShardingSphereDatabase createChangedDatabase(final String databaseName, final boolean internalLoadMetaData,
236                                                                final SwitchingResource switchingResource, final Collection<RuleConfiguration> ruleConfigs,
237                                                                final MetaDataContexts originalMetaDataContext, final MetaDataPersistService persistService,
238                                                                final ComputeNodeInstanceContext instanceContext) throws SQLException {
239         ResourceMetaData effectiveResourceMetaData = getEffectiveResourceMetaData(originalMetaDataContext.getMetaData().getDatabase(databaseName), switchingResource);
240         Collection<RuleConfiguration> toBeCreatedRuleConfigs = null == ruleConfigs
241                 ? originalMetaDataContext.getMetaData().getDatabase(databaseName).getRuleMetaData().getConfigurations()
242                 : ruleConfigs;
243         DatabaseConfiguration toBeCreatedDatabaseConfig = getDatabaseConfiguration(effectiveResourceMetaData, switchingResource, toBeCreatedRuleConfigs);
244         return createChangedDatabase(originalMetaDataContext.getMetaData().getDatabase(databaseName).getName(), internalLoadMetaData,
245                 persistService, toBeCreatedDatabaseConfig, originalMetaDataContext.getMetaData().getProps(), instanceContext);
246     }
247     
248     private static ShardingSphereDatabase createChangedDatabase(final String databaseName, final boolean internalLoadMetaData, final MetaDataPersistService persistService,
249                                                                 final DatabaseConfiguration databaseConfig, final ConfigurationProperties props,
250                                                                 final ComputeNodeInstanceContext instanceContext) throws SQLException {
251         return internalLoadMetaData
252                 ? InternalMetaDataFactory.create(databaseName, persistService, databaseConfig, props, instanceContext)
253                 : ExternalMetaDataFactory.create(databaseName, databaseConfig, props, instanceContext);
254     }
255     
256     private static ResourceMetaData getEffectiveResourceMetaData(final ShardingSphereDatabase database, final SwitchingResource resource) {
257         Map<StorageNode, DataSource> storageNodes = getStorageNodes(database.getResourceMetaData().getDataSources(), resource);
258         Map<String, StorageUnit> storageUnits = getStorageUnits(database.getResourceMetaData().getStorageUnits(), resource);
259         return new ResourceMetaData(storageNodes, storageUnits);
260     }
261     
262     private static Map<StorageNode, DataSource> getStorageNodes(final Map<StorageNode, DataSource> currentStorageNodes, final SwitchingResource resource) {
263         Map<StorageNode, DataSource> result = new LinkedHashMap<>(currentStorageNodes.size(), 1F);
264         for (Entry<StorageNode, DataSource> entry : currentStorageNodes.entrySet()) {
265             if (null == resource || !resource.getStaleDataSources().containsKey(entry.getKey())) {
266                 result.put(entry.getKey(), entry.getValue());
267             }
268         }
269         return result;
270     }
271     
272     private static Map<String, StorageUnit> getStorageUnits(final Map<String, StorageUnit> currentStorageUnits, final SwitchingResource resource) {
273         Map<String, StorageUnit> result = new LinkedHashMap<>(currentStorageUnits.size(), 1F);
274         for (Entry<String, StorageUnit> entry : currentStorageUnits.entrySet()) {
275             if (null == resource || !resource.getStaleStorageUnitNames().contains(entry.getKey())) {
276                 result.put(entry.getKey(), entry.getValue());
277             }
278         }
279         return result;
280     }
281     
282     private static DatabaseConfiguration getDatabaseConfiguration(final ResourceMetaData resourceMetaData, final SwitchingResource switchingResource,
283                                                                   final Collection<RuleConfiguration> toBeCreatedRuleConfigs) {
284         Map<String, DataSourcePoolProperties> propsMap = null == switchingResource ? resourceMetaData.getStorageUnits().entrySet().stream()
285                 .collect(Collectors.toMap(Entry::getKey, entry -> entry.getValue().getDataSourcePoolProperties(), (oldValue, currentValue) -> oldValue, LinkedHashMap::new))
286                 : switchingResource.getMergedDataSourcePoolPropertiesMap();
287         return new DataSourceProvidedDatabaseConfiguration(getMergedStorageNodeDataSources(resourceMetaData, switchingResource), toBeCreatedRuleConfigs, propsMap);
288     }
289     
290     private static Map<StorageNode, DataSource> getMergedStorageNodeDataSources(final ResourceMetaData currentResourceMetaData, final SwitchingResource switchingResource) {
291         Map<StorageNode, DataSource> result = currentResourceMetaData.getDataSources();
292         if (null != switchingResource && !switchingResource.getNewDataSources().isEmpty()) {
293             result.putAll(switchingResource.getNewDataSources());
294         }
295         return result;
296     }
