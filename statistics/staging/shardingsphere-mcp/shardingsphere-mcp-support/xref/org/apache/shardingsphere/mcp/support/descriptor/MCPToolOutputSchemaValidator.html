18  package org.apache.shardingsphere.mcp.support.descriptor;
19  
20  import lombok.AccessLevel;
21  import lombok.NoArgsConstructor;
22  import org.apache.shardingsphere.infra.exception.ShardingSpherePreconditions;
23  import org.apache.shardingsphere.mcp.api.capability.tool.MCPToolDescriptor;
24  import org.apache.shardingsphere.mcp.support.protocol.MCPModelFacingPayloadContract;
25  import org.apache.shardingsphere.mcp.support.protocol.MCPPayloadFieldNames;
26  import org.apache.shardingsphere.mcp.support.protocol.MCPResponseMode;
27  
28  import java.util.Collection;
29  import java.util.List;
30  import java.util.Map;
31  import java.util.Map.Entry;
32  
33  /**
34   * Validator for model-facing MCP tool output schema contracts.
35   */
36  @NoArgsConstructor(access = AccessLevel.PRIVATE)
37  public final class MCPToolOutputSchemaValidator {
38      
39      private static final Collection<String> CONTINUATION_MODES = List.of("none", "pagination", "metadata_search");
40      
41      private static final Collection<String> RECOVERY_CATEGORIES = List.of("not_found", "ambiguous", "empty_scope", "missing_context", "validation", "terminal",
42              "unsupported_target", "invalid_enum", "unsafe_sql", "invalid_explain_sql", "stale_workflow", "unavailable_runtime", "terminal_operator_action");
43      
44      /**
45       * Validate one tool output schema.
46       *
47       * @param descriptor tool descriptor
48       */
49      public static void validate(final MCPToolDescriptor descriptor) {
50          Map<String, Object> outputSchema = descriptor.getOutputSchema();
51          ShardingSpherePreconditions.checkState("object".equals(outputSchema.get("type")),
52                  () -> new IllegalStateException(String.format("Tool `%s` outputSchema must be an object.", descriptor.getName())));
53          Object properties = outputSchema.get("properties");
54          ShardingSpherePreconditions.checkState(properties instanceof Map && !((Map<?, ?>) properties).isEmpty(),
55                  () -> new IllegalStateException(String.format("Tool `%s` outputSchema must declare properties.", descriptor.getName())));
56          MCPToolDescriptorValidationUtils.validateModelFacingSchemaFields(descriptor, outputSchema);
57          validateOutputExamples(descriptor, outputSchema);
58          validateOutputExampleContractValues(descriptor, outputSchema);
59          validateModelCriticalOutputHints(descriptor, (Map<?, ?>) properties);
60      }
61      
62      private static void validateOutputExamples(final MCPToolDescriptor descriptor, final Map<String, Object> outputSchema) {
63          ShardingSpherePreconditions.checkState(isNonEmptyCollection(outputSchema.get("examples")),
64                  () -> new IllegalStateException(String.format("Tool `%s` outputSchema must declare examples.", descriptor.getName())));
65      }
66      
67      private static void validateOutputExampleContractValues(final MCPToolDescriptor descriptor, final Map<String, Object> outputSchema) {
68          Object examples = outputSchema.get("examples");
69          if (!(examples instanceof Collection)) {
70              return;
71          }
72          for (Object each : (Collection<?>) examples) {
73              validateExampleContractValue(descriptor, each);
74          }
75      }
76      
77      private static void validateExampleContractValue(final MCPToolDescriptor descriptor, final Object value) {
78          if (value instanceof Map) {
79              validateExampleContractMap(descriptor, (Map<?, ?>) value);
80          } else if (value instanceof Collection) {
81              for (Object each : (Collection<?>) value) {
82                  validateExampleContractValue(descriptor, each);
83              }
84          }
85      }
86      
87      private static void validateExampleContractMap(final MCPToolDescriptor descriptor, final Map<?, ?> value) {
88          Object responseMode = value.get("response_mode");
89          if (null != responseMode) {
90              ShardingSpherePreconditions.checkState(MCPResponseMode.isAllowed(responseMode.toString()),
91                      () -> new IllegalStateException(String.format("Tool `%s` output example uses unknown response_mode `%s`.", descriptor.getName(), responseMode)));
92          }
93          Object continuationMode = value.get("continuation_mode");
94          if (null != continuationMode) {
95              ShardingSpherePreconditions.checkState(CONTINUATION_MODES.contains(continuationMode.toString()),
96                      () -> new IllegalStateException(String.format("Tool `%s` output example uses unknown continuation_mode `%s`.", descriptor.getName(), continuationMode)));
97          }
98          Object recoveryCategory = value.get("recovery_category");
99          if (null != recoveryCategory) {
100             ShardingSpherePreconditions.checkState(RECOVERY_CATEGORIES.contains(recoveryCategory.toString()),
101                     () -> new IllegalStateException(String.format("Tool `%s` output example uses unknown recovery_category `%s`.", descriptor.getName(), recoveryCategory)));
102         }
103         for (Object each : value.values()) {
104             validateExampleContractValue(descriptor, each);
105         }
106         Object nextActions = value.get(MCPPayloadFieldNames.NEXT_ACTIONS);
107         if (null != nextActions) {
108             validateConcreteNextActions(descriptor, nextActions);
109         }
110     }
111     
112     private static void validateConcreteNextActions(final MCPToolDescriptor descriptor, final Object value) {
113         ShardingSpherePreconditions.checkState(value instanceof Collection,
114                 () -> new IllegalStateException(String.format("Tool `%s` next_actions example must be an array.", descriptor.getName())));
115         for (Object each : (Collection<?>) value) {
116             ShardingSpherePreconditions.checkState(each instanceof Map,
117                     () -> new IllegalStateException(String.format("Tool `%s` next_actions example item must be an object.", descriptor.getName())));
118             validateConcreteNextAction(descriptor, (Map<?, ?>) each);
119         }
120     }
121     
122     private static void validateConcreteNextAction(final MCPToolDescriptor descriptor, final Map<?, ?> action) {
123         String type = String.valueOf(action.get("type"));
124         Collection<String> allowedFields = MCPModelFacingPayloadContract.getNextActionAllowedFields(type);
125         ShardingSpherePreconditions.checkState(!allowedFields.isEmpty(),
126                 () -> new IllegalStateException(String.format("Tool `%s` next_actions example uses unknown type `%s`.", descriptor.getName(), type)));
127         for (String each : MCPModelFacingPayloadContract.getNextActionRequiredFields(type)) {
128             ShardingSpherePreconditions.checkState(action.containsKey(each),
129                     () -> new IllegalStateException(String.format("Tool `%s` next_actions example `%s` must contain `%s`.", descriptor.getName(), type, each)));
130         }
131         for (Object each : action.keySet()) {
132             String fieldName = String.valueOf(each);
133             ShardingSpherePreconditions.checkState(allowedFields.contains(fieldName),
134                     () -> new IllegalStateException(String.format("Tool `%s` next_actions example `%s` contains unsupported field `%s`.", descriptor.getName(), type, fieldName)));
135         }
136     }
137     
138     private static void validateModelCriticalOutputHints(final MCPToolDescriptor descriptor, final Map<?, ?> properties) {
139         for (Entry<?, ?> entry : properties.entrySet()) {
140             String fieldName = String.valueOf(entry.getKey());
141             if (MCPModelFacingPayloadContract.getModelCriticalFieldNames().contains(fieldName)) {
142                 validateModelCriticalOutputHint(descriptor, fieldName, entry.getValue());
143             }
144             validateNestedModelCriticalOutputHints(descriptor, entry.getValue());
145         }
146     }
147     
148     private static void validateNestedModelCriticalOutputHints(final MCPToolDescriptor descriptor, final Object value) {
149         if (!(value instanceof Map)) {
150             return;
151         }
152         Object properties = ((Map<?, ?>) value).get("properties");
153         if (properties instanceof Map) {
154             validateModelCriticalOutputHints(descriptor, (Map<?, ?>) properties);
155         }
156         Object items = ((Map<?, ?>) value).get("items");
157         if (items instanceof Map) {
158             validateNestedModelCriticalOutputHints(descriptor, items);
159         }
160     }
161     
162     private static void validateModelCriticalOutputHint(final MCPToolDescriptor descriptor, final String fieldName, final Object property) {
163         ShardingSpherePreconditions.checkState(property instanceof Map,
164                 () -> new IllegalStateException(String.format("Tool `%s` model-critical output field `%s` must be an object.", descriptor.getName(), fieldName)));
165         Object description = ((Map<?, ?>) property).get("description");
166         checkDescription(null == description ? "" : description.toString(), String.format("Tool model-critical output field `%s.%s` description", descriptor.getName(), fieldName));
167         if (MCPPayloadFieldNames.NEXT_ACTIONS.equals(fieldName)) {
168             validateNextActionsSchema(descriptor, (Map<?, ?>) property);
169         }
170     }
171     
172     private static void validateNextActionsSchema(final MCPToolDescriptor descriptor, final Map<?, ?> property) {
173         ShardingSpherePreconditions.checkState("array".equals(property.get("type")), () -> new IllegalStateException(String.format("Tool `%s` next_actions must be an array.", descriptor.getName())));
174         Object items = property.get("items");
175         ShardingSpherePreconditions.checkState(items instanceof Map, () -> new IllegalStateException(String.format("Tool `%s` next_actions items must be an object.", descriptor.getName())));
176         Object properties = ((Map<?, ?>) items).get("properties");
177         ShardingSpherePreconditions.checkState(properties instanceof Map,
178                 () -> new IllegalStateException(String.format("Tool `%s` next_actions items must declare properties.", descriptor.getName())));
179         for (String each : List.of("order", "type", "title")) {
180             ShardingSpherePreconditions.checkState(((Map<?, ?>) properties).containsKey(each),
181                     () -> new IllegalStateException(String.format("Tool `%s` next_actions item must declare `%s`.", descriptor.getName(), each)));
182             Object field = ((Map<?, ?>) properties).get(each);
183             ShardingSpherePreconditions.checkState(field instanceof Map,
184                     () -> new IllegalStateException(String.format("Tool `%s` next_actions item field `%s` must be an object.", descriptor.getName(), each)));
185             Object description = ((Map<?, ?>) field).get("description");
186             checkDescription(null == description ? "" : description.toString(), String.format("Tool next_actions item field `%s.%s` description", descriptor.getName(), each));
187         }
188         for (Object each : ((Map<?, ?>) properties).keySet()) {
189             String fieldName = String.valueOf(each);
190             ShardingSpherePreconditions.checkState(MCPModelFacingPayloadContract.getNextActionSchemaAllowedFields().contains(fieldName),
191                     () -> new IllegalStateException(String.format("Tool `%s` next_actions item contains unsupported field `%s`.", descriptor.getName(), fieldName)));
192         }
193     }
194     
195     private static boolean isNonEmptyCollection(final Object value) {
196         return value instanceof Collection && !((Collection<?>) value).isEmpty();
197     }
198     
199     private static void checkDescription(final String value, final String label) {
200         checkNotBlank(value, label);
201         ShardingSpherePreconditions.checkState(!value.startsWith(createPlaceholderPrefix("resource:")),
202                 () -> new IllegalStateException(String.format("%s must not be a placeholder description.", label)));
203         ShardingSpherePreconditions.checkState(!value.startsWith(createPlaceholderPrefix("resource template:")),
204                 () -> new IllegalStateException(String.format("%s must not be a placeholder description.", label)));
205     }
206     
207     private static String createPlaceholderPrefix(final String suffix) {
208         return "ShardingSphere MCP " + suffix;
209     }
210     
211     private static void checkNotBlank(final String value, final String label) {
212         ShardingSpherePreconditions.checkState(null != value && !value.isBlank(), () -> new IllegalStateException(String.format("%s is required.", label)));
213     }
