205         storageUnits.put("foo_ds_1", storageUnit);
206         Map<String, DataSourcePoolProperties> propsMap = new LinkedHashMap<>(2, 1F);
207         propsMap.put("foo_ds_0", mock(DataSourcePoolProperties.class));
208         propsMap.put("foo_ds_1", mock(DataSourcePoolProperties.class));
209         MetaDataContexts metaDataContexts = createRegistrationMetaDataContexts(Collections.emptyMap());

214             return null;
215         }).when(dataSourceUnitPersistService).persist("foo_db", propsMap);
216         try (
217                 MockedStatic<GenericSchemaBuilder> schemaBuilder = mockStatic(GenericSchemaBuilder.class);
218                 MockedConstruction<RetryExecutor> ignored = mockConstruction(RetryExecutor.class, (mock, context) -> when(mock.execute(any(), any())).thenAnswer(invocation -> {

229                     return true;
230                 }))) {
231             metaDataManagerPersistService.registerStorageUnits("foo_db", propsMap);
232             schemaBuilder.verify(() -> GenericSchemaBuilder.build(any(DatabaseType.class), argThat(material -> storageUnits.equals(material.getStorageUnits()))));
233             verify(metaDataPersistFacade.getDatabaseMetaDataFacade()).persistReloadDatabase(eq("foo_db"), argThat(each -> storageUnits.equals(each.getResourceMetaData().getStorageUnits())), any());
