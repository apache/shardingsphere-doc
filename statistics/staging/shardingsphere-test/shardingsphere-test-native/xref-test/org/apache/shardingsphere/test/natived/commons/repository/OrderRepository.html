247     /**
248      * Insert order to table.
249      *
250      * @param order order
251      * @return order ID of the insert statement
252      * @throws SQLException SQL Exception
253      */
254     public Long insert(final Order order) throws SQLException {
255         String sql = "INSERT INTO t_order (user_id, order_type, address_id, status) VALUES (?, ?, ?, ?)";
256         try (
257                 Connection connection = dataSource.getConnection();
258                 PreparedStatement preparedStatement = connection.prepareStatement(sql, Statement.RETURN_GENERATED_KEYS)) {
259             preparedStatement.setInt(1, order.getUserId());
260             preparedStatement.setInt(2, order.getOrderType());
261             preparedStatement.setLong(3, order.getAddressId());
262             preparedStatement.setString(4, order.getStatus());
263             preparedStatement.executeUpdate();
264             try (ResultSet resultSet = preparedStatement.getGeneratedKeys()) {
265                 if (resultSet.next()) {
266                     order.setOrderId(resultSet.getLong(1));
267                 }
268             }
269         }
270         return order.getOrderId();
271     }

331         return getOrders("SELECT * FROM t_order");
332     }
333     
334     private List<Order> getOrders(final String sql) throws SQLException {
335         List<Order> result = new LinkedList<>();
336         try (
337                 Connection connection = dataSource.getConnection();
338                 PreparedStatement preparedStatement = connection.prepareStatement(sql);
339                 ResultSet resultSet = preparedStatement.executeQuery()) {
340             while (resultSet.next()) {
341                 Order order = new Order();
342                 order.setOrderId(resultSet.getLong(1));
343                 order.setOrderType(resultSet.getInt(2));
344                 order.setUserId(resultSet.getInt(3));
345                 order.setAddressId(resultSet.getLong(4));
346                 order.setStatus(resultSet.getString(5));
347                 result.add(order);
348             }
349         }
350         return result;
351     }
