63      @Test
64      void assertExecuteWithCallbacks() throws SQLException, ReflectiveOperationException {
65          Object originalContextManager = Plugins.getMemberAccessor().get(ProxyContext.class.getDeclaredField("contextManager"), ProxyContext.getInstance());
66          assertExecuteWithCallbacks(originalContextManager);
67      }

69      private void assertExecuteWithCallbacks(final Object originalContextManager) throws SQLException, ReflectiveOperationException {
70          ContextManager contextManager = mock(ContextManager.class, RETURNS_DEEP_STUBS);
71          try {
72              ProxyContext.init(contextManager);
73              JDBCDriverType type = JDBCDriverType.STATEMENT;
74              ConnectionSession connectionSession = mock(ConnectionSession.class);
75              when(connectionSession.getUsedDatabaseName()).thenReturn("logic_db");
76              DatabaseProxyConnector databaseProxyConnector = mock(DatabaseProxyConnector.class);
77              JDBCExecutor jdbcExecutor = mock(JDBCExecutor.class);
78              ProxyJDBCExecutor executor = new ProxyJDBCExecutor(type, connectionSession, databaseProxyConnector, jdbcExecutor);
79              SQLStatement sqlStatement = mock(SQLStatement.class);
80              QueryContext queryContext = mockQueryContext(sqlStatement);
81              ExecutionGroupContext<JDBCExecutionUnit> executionGroupContext = new ExecutionGroupContext<>(Collections.emptyList(), new ExecutionGroupReportContext("pid", "logic_db"));
82              ProxyJDBCExecutorCallback firstCallback = mock(ProxyJDBCExecutorCallback.class);
83              ProxyJDBCExecutorCallback secondCallback = mock(ProxyJDBCExecutorCallback.class);
84              List<ExecuteResult> expected = Collections.singletonList(mock(ExecuteResult.class));
85              try (MockedStatic<ProxyJDBCExecutorCallbackFactory> mockedStatic = mockStatic(ProxyJDBCExecutorCallbackFactory.class)) {
86                  ResourceMetaData resourceMetaData = contextManager.getDatabase("logic_db").getResourceMetaData();
87                  DatabaseType protocolType = contextManager.getDatabase("logic_db").getProtocolType();
88                  mockedStatic.when(() -> ProxyJDBCExecutorCallbackFactory.newInstance(type, protocolType, resourceMetaData, sqlStatement, databaseProxyConnector, true, false, true))
89                          .thenReturn(firstCallback);
90                  mockedStatic.when(() -> ProxyJDBCExecutorCallbackFactory.newInstance(type, protocolType, resourceMetaData, sqlStatement, databaseProxyConnector, true, false, false))
91                          .thenReturn(secondCallback);
92                  when(jdbcExecutor.execute(executionGroupContext, firstCallback, secondCallback)).thenReturn(expected);
93                  assertThat(executor.execute(queryContext, executionGroupContext, true, false), is(expected));
94                  mockedStatic.verify(() -> ProxyJDBCExecutorCallbackFactory.newInstance(type, protocolType, resourceMetaData, sqlStatement, databaseProxyConnector, true, false, true));
95                  mockedStatic.verify(() -> ProxyJDBCExecutorCallbackFactory.newInstance(type, protocolType, resourceMetaData, sqlStatement, databaseProxyConnector, true, false, false));
96              }
97          } finally {
98              ProcessRegistry.getInstance().remove("pid");
99              Plugins.getMemberAccessor().set(ProxyContext.class.getDeclaredField("contextManager"), ProxyContext.getInstance(), originalContextManager);
100         }
101     }

103     @Test
104     void assertExecuteWithUnknownDatabase() throws ReflectiveOperationException {
105         Object originalContextManager = Plugins.getMemberAccessor().get(ProxyContext.class.getDeclaredField("contextManager"), ProxyContext.getInstance());
106         ContextManager contextManager = mock(ContextManager.class);
107         try {
108             ProxyContext.init(contextManager);
109             UnknownDatabaseException expected = new UnknownDatabaseException("foo_db");
110             when(contextManager.getDatabase("foo_db")).thenThrow(expected);
111             ConnectionSession connectionSession = mock(ConnectionSession.class);
112             when(connectionSession.getUsedDatabaseName()).thenReturn("foo_db");
113             JDBCExecutor jdbcExecutor = mock(JDBCExecutor.class);
114             ProxyJDBCExecutor executor = new ProxyJDBCExecutor(JDBCDriverType.STATEMENT, connectionSession, mock(DatabaseProxyConnector.class), jdbcExecutor);
115             ProcessEngine processEngine = mock(ProcessEngine.class);
116             Plugins.getMemberAccessor().set(ProxyJDBCExecutor.class.getDeclaredField("processEngine"), executor, processEngine);
117             ExecutionGroupContext<JDBCExecutionUnit> executionGroupContext = new ExecutionGroupContext<>(Collections.emptyList(), new ExecutionGroupReportContext("foo_pid", "foo_db"));
118             assertThat(assertThrows(UnknownDatabaseException.class, () -> executor.execute(mock(QueryContext.class), executionGroupContext, false, true)), sameInstance(expected));
119             verify(processEngine).completeSQLExecution("foo_pid");
120             verifyNoInteractions(jdbcExecutor);
121         } finally {
122             Plugins.getMemberAccessor().set(ProxyContext.class.getDeclaredField("contextManager"), ProxyContext.getInstance(), originalContextManager);
123         }
124     }

126     @Test
127     void assertExecuteWithSQLException() throws SQLException, ReflectiveOperationException {
128         Object originalContextManager = Plugins.getMemberAccessor().get(ProxyContext.class.getDeclaredField("contextManager"), ProxyContext.getInstance());
129         ContextManager contextManager = mock(ContextManager.class, RETURNS_DEEP_STUBS);
130         try {
131             ProxyContext.init(contextManager);
132             ConnectionSession connectionSession = mock(ConnectionSession.class);
133             when(connectionSession.getUsedDatabaseName()).thenReturn("foo_db");
134             JDBCExecutor jdbcExecutor = mock(JDBCExecutor.class);
135             ProxyJDBCExecutor executor = new ProxyJDBCExecutor(JDBCDriverType.STATEMENT, connectionSession, mock(DatabaseProxyConnector.class), jdbcExecutor);
136             ProcessEngine processEngine = mock(ProcessEngine.class);
137             Plugins.getMemberAccessor().set(ProxyJDBCExecutor.class.getDeclaredField("processEngine"), executor, processEngine);
138             ExecutionGroupContext<JDBCExecutionUnit> executionGroupContext = new ExecutionGroupContext<>(Collections.emptyList(), new ExecutionGroupReportContext("foo_pid", "foo_db"));
139             SQLException expected = new SQLException("foo execution failure");
140             when(jdbcExecutor.execute(eq(executionGroupContext), any(), any())).thenThrow(expected);
141             QueryContext queryContext = mockQueryContext(mock(SQLStatement.class));
142             assertThat(assertThrows(SQLException.class, () -> executor.execute(queryContext, executionGroupContext, false, true)), sameInstance(expected));
143             verify(processEngine).completeSQLExecution("foo_pid");
144         } finally {
145             Plugins.getMemberAccessor().set(ProxyContext.class.getDeclaredField("contextManager"), ProxyContext.getInstance(), originalContextManager);
146         }
147     }
148     
149     private QueryContext mockQueryContext(final SQLStatement sqlStatement) {
150         QueryContext result = mock(QueryContext.class);
151         when(result.getSql()).thenReturn("SELECT 1");
152         SQLStatementContext sqlStatementContext = mock(SQLStatementContext.class);
153         when(sqlStatementContext.getSqlStatement()).thenReturn(sqlStatement);
154         when(result.getSqlStatementContext()).thenReturn(sqlStatementContext);
155         return result;
156     }
