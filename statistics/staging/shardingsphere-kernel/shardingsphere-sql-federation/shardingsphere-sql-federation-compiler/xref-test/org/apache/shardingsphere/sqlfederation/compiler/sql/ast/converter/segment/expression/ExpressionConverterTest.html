113     @Test
114     void assertConvertDelegatesToAllSupportedConverters() {
115         SqlNode expectedLiteralNode = mock(SqlNode.class);
116         LiteralExpressionSegment literalSegment = new LiteralExpressionSegment(0, 0, "literal");
117         when(LiteralExpressionConverter.convert(literalSegment, null)).thenReturn(Optional.of(expectedLiteralNode));
118         SqlNode expectedListNode = mock(SqlNode.class);
119         ListExpression listSegment = new ListExpression(0, 0);
120         when(ListExpressionConverter.convert(listSegment)).thenReturn(Optional.of(expectedListNode));
121         SqlBasicCall expectedBinaryNode = mock(SqlBasicCall.class);
122         BinaryOperationExpression binarySegment = new BinaryOperationExpression(0, 0, literalSegment, literalSegment, "+", "text");
123         when(BinaryOperationExpressionConverter.convert(binarySegment)).thenReturn(expectedBinaryNode);
124         SqlIdentifier expectedColumnNode = mock(SqlIdentifier.class);
125         ColumnSegment columnSegment = new ColumnSegment(0, 0, new IdentifierValue("col"));
126         when(ColumnConverter.convert(columnSegment)).thenReturn(expectedColumnNode);
127         SqlNode expectedExistsSubqueryNode = mock(SqlNode.class);
128         SubquerySegment subquerySegment = new SubquerySegment(0, 0, SelectStatement.builder().databaseType(databaseType).build(), "sub");
129         ExistsSubqueryExpression existsSubqueryExpression = new ExistsSubqueryExpression(0, 0, subquerySegment);
130         when(ExistsSubqueryExpressionConverter.convert(existsSubqueryExpression)).thenReturn(expectedExistsSubqueryNode);
131         SqlNode expectedSubqueryNode = mock(SqlNode.class);
132         SubqueryExpressionSegment subqueryExpressionSegment = new SubqueryExpressionSegment(subquerySegment);
133         when(SubqueryExpressionConverter.convert(subqueryExpressionSegment)).thenReturn(expectedSubqueryNode);
134         SqlBasicCall expectedInNode = mock(SqlBasicCall.class);
135         InExpression inExpression = new InExpression(0, 0, literalSegment, literalSegment, false);
136         when(InExpressionConverter.convert(inExpression)).thenReturn(expectedInNode);
137         SqlBasicCall expectedBetweenNode = mock(SqlBasicCall.class);
138         BetweenExpression betweenExpression = new BetweenExpression(0, 0, literalSegment, literalSegment, literalSegment, false);
139         when(BetweenExpressionConverter.convert(betweenExpression)).thenReturn(expectedBetweenNode);
140         SqlDynamicParam expectedParameterNode = mock(SqlDynamicParam.class);
141         ParameterMarkerExpressionSegment parameterSegment = new ParameterMarkerExpressionSegment(0, 0, 0);
142         when(ParameterMarkerExpressionConverter.convert(parameterSegment)).thenReturn(expectedParameterNode);
143         SqlNode expectedFunctionNode = mock(SqlNode.class);
144         FunctionSegment functionSegment = new FunctionSegment(0, 0, "func", "func_text");
145         when(FunctionConverter.convert(functionSegment)).thenReturn(expectedFunctionNode);
146         SqlNode expectedAggregationNode = mock(SqlNode.class);
147         AggregationProjectionSegment aggregationSegment = new AggregationProjectionSegment(0, 0, AggregationType.COUNT, "count(expr)");
148         when(AggregationProjectionConverter.convert(aggregationSegment)).thenReturn(Optional.of(expectedAggregationNode));
149         DataTypeSegment dataTypeSegment = new DataTypeSegment();
150         dataTypeSegment.setStartIndex(0);
151         dataTypeSegment.setStopIndex(0);
152         dataTypeSegment.setDataTypeName("int");
153         SqlNode expectedDataTypeNode = mock(SqlNode.class);
154         when(DataTypeExpressionConverter.convert(dataTypeSegment)).thenReturn(expectedDataTypeNode);
155         SqlCase expectedCaseWhenNode = mock(SqlCase.class);
156         CaseWhenExpression caseWhenExpression = new CaseWhenExpression(0, 0, literalSegment, Collections.singleton(literalSegment), Collections.singleton(literalSegment), literalSegment);
157         when(CaseWhenExpressionConverter.convert(caseWhenExpression)).thenReturn(expectedCaseWhenNode);
158         SqlBasicCall expectedNotNode = mock(SqlBasicCall.class);
159         NotExpression notExpression = new NotExpression(0, 0, literalSegment, true);
160         when(NotExpressionConverter.convert(notExpression)).thenReturn(expectedNotNode);
161         SqlNode expectedTypeCastNode = mock(SqlNode.class);
162         TypeCastExpression typeCastExpression = new TypeCastExpression(0, 0, "text", literalSegment, "int");
163         when(TypeCastExpressionConverter.convert(typeCastExpression)).thenReturn(Optional.of(expectedTypeCastNode));
164         SqlIdentifier expectedExtractNode = mock(SqlIdentifier.class);
165         ExtractArgExpression extractArgExpression = new ExtractArgExpression(0, 0, "extract");
166         when(ExtractArgExpressionConverter.convert(extractArgExpression)).thenReturn(expectedExtractNode);
167         SqlBasicCall expectedMatchNode = mock(SqlBasicCall.class);
168         MatchAgainstExpression matchAgainstExpression = new MatchAgainstExpression(0, 0, literalSegment, "search", "text");
169         when(MatchExpressionConverter.convert(matchAgainstExpression)).thenReturn(expectedMatchNode);
170         SqlBasicCall expectedCollateNode = mock(SqlBasicCall.class);
171         CollateExpression collateExpression = new CollateExpression(0, 0, literalSegment, literalSegment);
172         when(CollateExpressionConverter.convert(collateExpression)).thenReturn(expectedCollateNode);
173         SqlNode expectedRowNode = mock(SqlNode.class);
174         RowExpression rowExpression = new RowExpression(0, 0, "row");
175         when(RowExpressionConverter.convert(rowExpression)).thenReturn(expectedRowNode);
176         SqlIdentifier expectedVariableNode = mock(SqlIdentifier.class);
177         VariableSegment variableSegment = new VariableSegment(0, 0, "@@session");
178         when(VariableSegmentConverter.convert(variableSegment)).thenReturn(expectedVariableNode);
179         SqlBasicCall expectedUnaryNode = mock(SqlBasicCall.class);
180         UnaryOperationExpression unaryOperationExpression = new UnaryOperationExpression(0, 0, literalSegment, "+", "text");
181         when(UnaryOperationExpressionConverter.convert(unaryOperationExpression)).thenReturn(expectedUnaryNode);
182         SqlBasicCall expectedIntervalNode = mock(SqlBasicCall.class);
183         IntervalExpression intervalExpression = new IntervalExpression(0, 0, literalSegment, IntervalUnit.DAY, "interval");
184         when(IntervalExpressionConverter.convert(intervalExpression)).thenReturn(expectedIntervalNode);
185         SqlNode expectedQuantifyNode = mock(SqlNode.class);
186         QuantifySubqueryExpression quantifySubqueryExpression = new QuantifySubqueryExpression(
187                 0, 0, new SubquerySegment(0, 0, SelectStatement.builder().databaseType(databaseType).build(), "sub"), "ALL");
188         when(QuantifySubqueryExpressionConverter.convert(quantifySubqueryExpression)).thenReturn(expectedQuantifyNode);
189         Map<ExpressionSegment, SqlNode> expectations = new LinkedHashMap<>(23, 1F);
190         expectations.put(literalSegment, expectedLiteralNode);
191         expectations.put(listSegment, expectedListNode);
192         expectations.put(binarySegment, expectedBinaryNode);
193         expectations.put(columnSegment, expectedColumnNode);
194         expectations.put(existsSubqueryExpression, expectedExistsSubqueryNode);
195         expectations.put(subqueryExpressionSegment, expectedSubqueryNode);
196         expectations.put(inExpression, expectedInNode);
197         expectations.put(betweenExpression, expectedBetweenNode);
198         expectations.put(parameterSegment, expectedParameterNode);
199         expectations.put(functionSegment, expectedFunctionNode);
200         expectations.put(aggregationSegment, expectedAggregationNode);
201         expectations.put(dataTypeSegment, expectedDataTypeNode);
202         expectations.put(caseWhenExpression, expectedCaseWhenNode);
203         expectations.put(notExpression, expectedNotNode);
204         expectations.put(typeCastExpression, expectedTypeCastNode);
205         expectations.put(extractArgExpression, expectedExtractNode);
206         expectations.put(matchAgainstExpression, expectedMatchNode);
207         expectations.put(collateExpression, expectedCollateNode);
208         expectations.put(rowExpression, expectedRowNode);
209         expectations.put(variableSegment, expectedVariableNode);
210         expectations.put(unaryOperationExpression, expectedUnaryNode);
211         expectations.put(intervalExpression, expectedIntervalNode);
212         expectations.put(quantifySubqueryExpression, expectedQuantifyNode);
213         for (Entry<ExpressionSegment, SqlNode> entry : expectations.entrySet()) {
214             Optional<SqlNode> actualOptional = ExpressionConverter.convert(entry.getKey());
215             assertTrue(actualOptional.isPresent());
216             assertThat(actualOptional.orElse(null), is(entry.getValue()));
217         }
218     }

220     @Test
221     void assertConvertThrowsUnsupportedForCommonExpression() {
222         CommonExpressionSegment expressionSegment = new CommonExpressionSegment(0, 0, "text", Collections.emptyList());
223         UnsupportedSQLOperationException actualException = assertThrows(UnsupportedSQLOperationException.class, () -> ExpressionConverter.convert(expressionSegment));
224         assertThat(actualException.getMessage(), is("Unsupported SQL operation: unsupported CommonExpressionSegment."));
225     }
226     
227     @Test
228     void assertConvertReturnsEmptyForNullSegment() {
229         assertFalse(ExpressionConverter.convert(null).isPresent());
230     }
231     
232     @Test
233     void assertConvertThrowsUnsupportedForUnknownExpression() {
234         ExpressionSegment expressionSegment = mock(ExpressionSegment.class);
235         UnsupportedSQLOperationException actualException = assertThrows(UnsupportedSQLOperationException.class, () -> ExpressionConverter.convert(expressionSegment));
236         assertThat(actualException.getMessage(), is("Unsupported SQL operation: unsupported TableSegment type: " + expressionSegment.getClass() + "."));
237     }
