277     @CsvSource({"false, false, false, false", "true, false, false, false", "false, true, false, false", "false, false, true, false", "true, true, true, false", "true, true, true, true"})
278     void assertExecuteIncrementalFailure(final boolean notificationFailure, final boolean stopFailure, final boolean disableFailure, final boolean reuseOriginalError,
279                                          final LogCaptureAssertion logCaptureAssertion) {
280         CDCJobConfiguration jobConfig = mockJobConfiguration(
281                 Collections.singletonList(new JobDataNodeLine(Collections.singletonList(new JobDataNodeEntry("logic_tbl", Collections.singletonList(new DataNode("ds_0.tbl_0")))))));

286         try (
287                 MockedStatic<TypedSPILoader> typedSPILoader = mockStatic(TypedSPILoader.class);
288                 MockedConstruction<PipelineProcessConfigurationPersistService> ignoredProcess = mockPersistService(
289                         new PipelineProcessConfiguration(new PipelineReadConfiguration(1, 1, 1, null), new PipelineWriteConfiguration(1, 1, null), null));
290                 MockedConstruction<CDCJobPreparer> ignoredPreparer = mockConstruction(CDCJobPreparer.class, (mock, context) -> doAnswer(invocation -> {
291                     CDCJobItemContext jobItemContext = ((Collection<CDCJobItemContext>) invocation.getArgument(0)).iterator().next();
292                     jobItemContext.getInventoryTasks().add(mockTask(new IngestPlaceholderPosition(), Collections.singletonList(CompletableFuture.completedFuture(null))));
293                     jobItemContext.getIncrementalTasks().add(mockTask(new IngestPlaceholderPosition(), Collections.singletonList(CompletableFuture.completedFuture(null))));
294                     return null;
295                 }).when(mock).initTasks(anyCollection()));
296                 MockedStatic<PipelineExecuteEngine> triggerMocked = mockStatic(PipelineExecuteEngine.class);
297                 MockedStatic<PipelineJobRegistry> jobRegistryMocked = mockStatic(PipelineJobRegistry.class)) {
298             typedSPILoader.when(() -> TypedSPILoader.getService(TransmissionJobAPI.class, "STREAMING")).thenReturn(jobAPI);
299             PipelineGovernanceFacade governanceFacade = mock(PipelineGovernanceFacade.class, RETURNS_DEEP_STUBS);
300             when(PipelineAPIFactory.getPipelineGovernanceFacade(CONTEXT_KEY)).thenReturn(governanceFacade);
301             when(PipelineDistributedBarrier.getInstance(CONTEXT_KEY)).thenReturn(mock(PipelineDistributedBarrier.class));
302             when(OrderedSPILoader.getServices(eq(PipelineRequiredColumnsExtractor.class), anyCollection())).thenReturn(Collections.emptyMap());
303             when(PipelineDataSourceConfigurationFactory.newInstance(anyString(), anyString())).thenReturn(mock(PipelineDataSourceConfiguration.class));
304             CDCSink sink = mock(CDCSink.class);
305             Collection<Throwable> expectedSuppressed = new ArrayList<>(3);
306             if (notificationFailure) {
307                 RuntimeException notificationException = reuseOriginalError ? expectedException : new IllegalStateException("notification failed");
308                 doThrow(notificationException).when(sink).onFailure(expectedException);
309                 if (!reuseOriginalError) {
310                     expectedSuppressed.add(notificationException);
311                 }
312             }
313             if (stopFailure) {
314                 RuntimeException stopException = reuseOriginalError ? expectedException : new IllegalStateException("stop failed");
315                 jobRegistryMocked.when(() -> PipelineJobRegistry.stop("foo_job_id")).thenThrow(stopException);
316                 if (!reuseOriginalError) {
317                     expectedSuppressed.add(stopException);
318                 }
319             }
320             if (disableFailure) {
321                 RuntimeException disableException = reuseOriginalError ? expectedException : new IllegalStateException("disable failed");
322                 doThrow(disableException).when(jobAPI).disable("foo_job_id");
323                 if (!reuseOriginalError) {
324                     expectedSuppressed.add(disableException);
325                 }
326             }
327             AtomicInteger triggerCounter = new AtomicInteger();
328             triggerMocked.when(() -> PipelineExecuteEngine.trigger(anyCollection(), any())).thenAnswer(invocation -> {
329                 ExecuteCallback callback = invocation.getArgument(1);
330                 if (0 == triggerCounter.getAndIncrement()) {
331                     callback.onSuccess();
332                 } else {
333                     callback.onFailure(expectedException);
334                 }
335                 return null;
336             });
337             CDCJob job = new CDCJob(sink);
338             job.execute(shardingContext);
339             verify(sink).onFailure(expectedException);
340             assertThat(Arrays.asList(expectedException.getSuppressed()), is(expectedSuppressed));
341             verify(governanceFacade.getJobItemFacade().getErrorMessage()).update(eq("foo_job_id"), anyInt(), eq(expectedException));
342             verify(jobAPI).disable("foo_job_id");
343             jobRegistryMocked.verify(() -> PipelineJobRegistry.stop("foo_job_id"));
344         }
345         logCaptureAssertion.assertErrorLog(actualException -> assertThat(((ThrowableProxy) actualException).getThrowable(), sameInstance(expectedException)));
346     }
