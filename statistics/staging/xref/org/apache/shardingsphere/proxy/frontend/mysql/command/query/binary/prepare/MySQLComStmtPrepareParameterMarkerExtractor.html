49  /**
50   * Parameter marker extractor for MySQL COM_STMT_PREPARE.
51   */
52  @NoArgsConstructor(access = AccessLevel.PRIVATE)
53  public final class MySQLComStmtPrepareParameterMarkerExtractor {
54      
55      /**
56       * Resolve columns for parameter markers.
57       *
58       * @param sqlStatementContext SQL statement context
59       * @param schema schema
60       * @return corresponding columns of parameter markers
61       */
62      public static List<ShardingSphereColumn> resolveColumnsForParameterMarkers(final SQLStatementContext sqlStatementContext, final ShardingSphereSchema schema) {
63          int parameterCount = sqlStatementContext.getSqlStatement().getParameterCount();
64          if (0 == parameterCount) {
65              return Collections.emptyList();
66          }
67          List<ShardingSphereColumn> result = new ArrayList<>(Collections.nCopies(parameterCount, null));
68          if (sqlStatementContext instanceof InsertStatementContext) {
69              collectColumnsForInsert((InsertStatementContext) sqlStatementContext, schema, result);
70          }
71          if (sqlStatementContext instanceof UpdateStatementContext) {
72              collectColumnsForUpdateAssignments((UpdateStatementContext) sqlStatementContext, schema, result);
73          }
74          if (sqlStatementContext instanceof WhereContextAvailable) {
75              collectColumnsForWhereSegments(sqlStatementContext, schema, result);
76          }
77          return result;
78      }
79      
80      private static void collectColumnsForInsert(final InsertStatementContext sqlStatementContext, final ShardingSphereSchema schema, final List<ShardingSphereColumn> columns) {
81          Optional<String> tableName = sqlStatementContext.getSqlStatement().getTable().map(optional -> optional.getTableName().getIdentifier().getValue());
82          if (!tableName.isPresent() || !schema.containsTable(tableName.get())) {
83              return;
84          }
85          ShardingSphereTable table = schema.getTable(tableName.get());
86          for (InsertValueContext each : sqlStatementContext.getInsertValueContexts()) {
87              collectColumnsForInsertValue(each, table, sqlStatementContext.getColumnNames(), columns);
88          }
89          OnDuplicateUpdateContext onDuplicateUpdateContext = sqlStatementContext.getOnDuplicateKeyUpdateValueContext();
90          if (null != onDuplicateUpdateContext) {
91              collectColumnsForOnDuplicateUpdate(onDuplicateUpdateContext, schema, sqlStatementContext, columns);
92          }
93      }
94      
95      private static void collectColumnsForInsertValue(final InsertValueContext insertValueContext, final ShardingSphereTable table,
96                                                       final List<String> insertColumnNames, final List<ShardingSphereColumn> columns) {
97          List<ExpressionSegment> valueExpressions = insertValueContext.getValueExpressions();
98          for (int index = 0; index < valueExpressions.size(); index++) {
99              if (index < insertColumnNames.size()) {
100                 setParameterMarkerColumns(columns, valueExpressions.get(index), getColumn(table, insertColumnNames.get(index)));
101             }
102         }
103     }
104     
105     private static void collectColumnsForOnDuplicateUpdate(final OnDuplicateUpdateContext onDuplicateUpdateContext, final ShardingSphereSchema schema,
106                                                            final SQLStatementContext sqlStatementContext, final List<ShardingSphereColumn> columns) {
107         for (int index = 0; index < onDuplicateUpdateContext.getValueExpressions().size(); index++) {
108             setParameterMarkerColumns(columns,
109                     onDuplicateUpdateContext.getValueExpressions().get(index), resolveColumn(schema, sqlStatementContext, onDuplicateUpdateContext.getColumns().get(index)));
110         }
111     }
112     
113     private static void collectColumnsForUpdateAssignments(final UpdateStatementContext sqlStatementContext, final ShardingSphereSchema schema, final List<ShardingSphereColumn> columns) {
114         for (ColumnAssignmentSegment each : sqlStatementContext.getSqlStatement().getSetAssignment().getAssignments()) {
115             setParameterMarkerColumns(columns, each.getValue(), resolveColumn(schema, sqlStatementContext, each.getColumns().get(0)));
116         }
117     }
118     
119     private static void collectColumnsForWhereSegments(final SQLStatementContext sqlStatementContext, final ShardingSphereSchema schema, final List<ShardingSphereColumn> columns) {
120         for (WhereSegment each : ((WhereContextAvailable) sqlStatementContext).getWhereSegments()) {
121             collectColumnsForWhereExpression(each.getExpr(), sqlStatementContext, schema, columns);
122         }
123     }
124     
125     private static void collectColumnsForWhereExpression(final ExpressionSegment expression,
126                                                          final SQLStatementContext sqlStatementContext, final ShardingSphereSchema schema, final List<ShardingSphereColumn> columns) {
127         if (expression instanceof BinaryOperationExpression) {
128             BinaryOperationExpression binaryExpression = (BinaryOperationExpression) expression;
129             collectColumnsForWhereExpression(binaryExpression.getLeft(), sqlStatementContext, schema, columns);
130             collectColumnsForWhereExpression(binaryExpression.getRight(), sqlStatementContext, schema, columns);
131             setParameterMarkerColumns(columns, binaryExpression.getRight(), resolveColumn(schema, sqlStatementContext, binaryExpression.getLeft()));
132             setParameterMarkerColumns(columns, binaryExpression.getLeft(), resolveColumn(schema, sqlStatementContext, binaryExpression.getRight()));
133             return;
134         }
135         if (expression instanceof InExpression) {
136             InExpression inExpression = (InExpression) expression;
137             ShardingSphereColumn column = resolveColumn(schema, sqlStatementContext, inExpression.getLeft());
138             if (null != column) {
139                 for (ExpressionSegment each : inExpression.getExpressionList()) {
140                     setParameterMarkerColumns(columns, each, column);
141                 }
142             }
143             return;
144         }
145         if (expression instanceof BetweenExpression) {
146             BetweenExpression betweenExpression = (BetweenExpression) expression;
147             ShardingSphereColumn column = resolveColumn(schema, sqlStatementContext, betweenExpression.getLeft());
148             if (null != column) {
149                 setParameterMarkerColumns(columns, betweenExpression.getBetweenExpr(), column);
150                 setParameterMarkerColumns(columns, betweenExpression.getAndExpr(), column);
151             }
152         }
153     }
154     
155     private static ShardingSphereColumn resolveColumn(final ShardingSphereSchema schema, final SQLStatementContext sqlStatementContext, final ExpressionSegment expressionSegment) {
156         return expressionSegment instanceof ColumnSegment ? resolveColumn(schema, sqlStatementContext, (ColumnSegment) expressionSegment) : null;
157     }
158     
159     private static ShardingSphereColumn resolveColumn(final ShardingSphereSchema schema, final SQLStatementContext sqlStatementContext, final ColumnSegment columnSegment) {
160         String tableName = Optional.ofNullable(columnSegment.getColumnBoundInfo()).map(optional -> optional.getOriginalTable().getValue()).orElse("");
161         if (tableName.isEmpty()) {
162             tableName = columnSegment.getOwner().map(optional -> optional.getIdentifier().getValue()).orElseGet(() -> getSingleTableName(sqlStatementContext).orElse(""));
163         }
164         if (tableName.isEmpty() || !schema.containsTable(tableName)) {
165             return null;
166         }
167         ShardingSphereTable table = schema.getTable(tableName);
168         String columnName = Optional.ofNullable(columnSegment.getColumnBoundInfo()).map(optional -> optional.getOriginalColumn().getValue()).orElse("");
169         if (columnName.isEmpty()) {
170             columnName = columnSegment.getIdentifier().getValue();
171         }
172         return table.containsColumn(columnName) ? table.getColumn(columnName) : null;
173     }
174     
175     private static Optional<String> getSingleTableName(final SQLStatementContext sqlStatementContext) {
176         Collection<String> tableNames = sqlStatementContext.getTablesContext().getTableNames();
177         return 1 == tableNames.size() ? Optional.of(tableNames.iterator().next()) : Optional.empty();
178     }
179     
180     private static ShardingSphereColumn getColumn(final ShardingSphereTable table, final String columnName) {
181         ShardingSpherePreconditions.checkState(table.containsColumn(columnName), () -> new ColumnNotFoundException(table.getName(), columnName));
182         return table.getColumn(columnName);
183     }
184     
185     private static void setParameterMarkerColumns(final List<ShardingSphereColumn> columns, final ExpressionSegment expression, final ShardingSphereColumn column) {
186         if (null == column) {
187             return;
188         }
189         for (ParameterMarkerExpressionSegment each : ExpressionExtractor.getParameterMarkerExpressions(Collections.singleton(expression))) {
190             if (each.getParameterMarkerIndex() < columns.size()) {
191                 columns.set(each.getParameterMarkerIndex(), column);
192             }
193         }
194     }
