55  @ExtendWith(MockitoExtension.class)
56  class MySQLTimeBinaryProtocolValueTest {
57      
58      private final MySQLTimeBinaryProtocolValue binaryProtocolValue = new MySQLTimeBinaryProtocolValue();
59      
60      @Mock
61      private MySQLPacketPayload payload;
62      
63      @ParameterizedTest(name = "{0}")
64      @MethodSource("readArguments")
65      void assertRead(final String name, final int[] int1Values, final int[] int4Values, final Timestamp expected) throws SQLException {
66          when(payload.readInt1()).thenReturn(int1Values[0], int1Values[1], int1Values[2], int1Values[3], int1Values[4]);
67          when(payload.readInt4()).thenReturn(int4Values[0], int4Values[1]);
68          assertThat(binaryProtocolValue.read(payload, false), is(expected));
69      }
70      
71      @Test
72      void assertReadZeroLengthDoesNotConsumeNextParameterBytes() throws SQLException {
73          byte[] data = {0x00, (byte) 0xd2, 0x04, 0x00, 0x00};
74          MySQLPacketPayload realPayload = new MySQLPacketPayload(Unpooled.wrappedBuffer(data), StandardCharsets.UTF_8);
75          assertThat(binaryProtocolValue.read(realPayload, false), is(new Timestamp(0L)));
76          assertThat(realPayload.readInt4(), is(1234));
77      }
78      
79      @Test
80      void assertReadWithUnsupportedLength() {
81          when(payload.readInt1()).thenReturn(100);
82          SQLFeatureNotSupportedException actual = assertThrows(SQLFeatureNotSupportedException.class, () -> binaryProtocolValue.read(payload, false));
83          assertThat(actual.getMessage(), is("Wrong length `100` of MYSQL_TYPE_DATE"));
84      }
85      
86      @ParameterizedTest(name = "{0}")
87      @MethodSource("writeArguments")
88      void assertWrite(final String name, final Time value, final List<Integer> expectedInt1Values, final List<Integer> expectedInt4Values) {
89          binaryProtocolValue.write(payload, value);
90          ArgumentCaptor<Integer> int1Captor = ArgumentCaptor.forClass(Integer.class);
91          verify(payload, times(expectedInt1Values.size())).writeInt1(int1Captor.capture());
92          List<Integer> actualInt1Values = int1Captor.getAllValues();
93          assertThat(actualInt1Values, is(expectedInt1Values));
94          if (expectedInt4Values.isEmpty()) {
95              verify(payload, never()).writeInt4(anyInt());
96          } else {
97              ArgumentCaptor<Integer> int4Captor = ArgumentCaptor.forClass(Integer.class);
98              verify(payload, times(expectedInt4Values.size())).writeInt4(int4Captor.capture());
99              List<Integer> actualInt4Values = int4Captor.getAllValues();
100             assertThat(actualInt4Values, is(expectedInt4Values));
101         }
102         verifyNoMoreInteractions(payload);
103     }
104     
105     private static Stream<Arguments> readArguments() {
106         return Stream.of(
107                 Arguments.of("eight_bytes", new int[]{8, 0, 10, 59, 0}, new int[]{0, 0}, createTimestamp(0)),
108                 Arguments.of("twelve_bytes", new int[]{12, 0, 10, 59, 0}, new int[]{0, 1000}, createTimestamp(1_000_000)),
109                 Arguments.of("twelve_bytes_fractional_second", new int[]{12, 0, 10, 59, 0}, new int[]{0, 230000}, createTimestamp(230_000_000)));
110     }
111     
112     private static Stream<Arguments> writeArguments() {
113         return Stream.of(
114                 Arguments.of("time_and_nanos_absent", Time.valueOf(LocalTime.of(0, 0)), Collections.singletonList(0), Collections.emptyList()),
115                 Arguments.of("hour_present_without_nanos", Time.valueOf(LocalTime.of(1, 30, 10)), Arrays.asList(8, 0, 1, 30, 10), Collections.singletonList(0)),
116                 Arguments.of("minutes_present_without_nanos", Time.valueOf(LocalTime.of(0, 30)), Arrays.asList(8, 0, 0, 30, 0), Collections.singletonList(0)),
117                 Arguments.of("seconds_present_without_nanos", Time.valueOf(LocalTime.of(0, 0, 10)), Arrays.asList(8, 0, 0, 0, 10), Collections.singletonList(0)),
118                 Arguments.of("nanos_only", createTimeAtLocalDateStart(), Arrays.asList(12, 0, 0, 0, 0), Arrays.asList(0, 1000)));
119     }
120     
121     private static Timestamp createTimestamp(final int nanos) {
122         Timestamp result = Timestamp.valueOf(LocalDateTime.of(0, 1, 1, 10, 59, 0));
123         result.setNanos(nanos);
124         return result;
125     }
126     
127     private static Time createTimeAtLocalDateStart() {
128         return new Time(LocalDate.of(1970, 1, 1).atStartOfDay(ZoneId.systemDefault()).toInstant().toEpochMilli() + 1);
129     }
130 }
