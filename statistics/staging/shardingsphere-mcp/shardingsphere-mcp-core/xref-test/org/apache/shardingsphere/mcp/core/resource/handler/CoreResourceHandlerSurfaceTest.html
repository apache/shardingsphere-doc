72      @ParameterizedTest(name = "{0}")
73      @MethodSource("handlerCases")
74      void assertHandle(final HandlerCase handlerCase) {
75          try (RequestContextFixture requestContextFixture = ResourceTestDataFactory.createRequestContextFixture(runtimeContext, ResourceTestDataFactory.createDatabaseMetadata())) {
76              MCPFeatureRuntimeRequestContext requestContext = requestContextFixture.getRequestContext();
77              MCPSuccessPayload actual = handle(handlerCase.getHandler(), requestContext, parseUriVariables(handlerCase.getExpectedUriTemplate(), handlerCase.getResourceUri()));
78              Map<String, Object> actualPayload = actual.toPayload();
79              if (HandlerResultType.DATABASE_CAPABILITY == handlerCase.getExpectedType()) {
80                  assertThat(actual, isA(MCPDatabaseCapabilityPayload.class));
81                  assertThat(actualPayload.get("database"), is(handlerCase.getExpectedDatabase()));
82                  return;
83              }
84              if (HandlerResultType.SERVICE_CAPABILITY == handlerCase.getExpectedType()) {
85                  assertThat(actual, isA(MCPMapPayload.class));
86                  assertFalse(((Collection<?>) actualPayload.get("supportedStatementClasses")).isEmpty());
87                  assertTrue(((List<?>) actualPayload.get("completionTargets")).stream().map(String::valueOf).anyMatch(each -> each.contains("inspect_metadata")));
88                  assertTrue(((List<?>) actualPayload.get("resourceNavigation")).stream().map(String::valueOf).anyMatch(each -> each.contains("database_gateway_apply_workflow")));
89                  assertFalse(actualPayload.containsKey("resources"));
90                  assertFalse(actualPayload.containsKey("tools"));
91                  assertFalse(actualPayload.containsKey("fingerprints"));
92                  return;
93              }
94              if (HandlerResultType.SERVICE_GUIDANCE == handlerCase.getExpectedType()) {
95                  assertThat(actual, isA(MCPMapPayload.class));
96                  assertThat(actualPayload.get("response_mode"), is("guidance"));
97                  assertTrue(actualPayload.containsKey("model_contract"));
98                  assertTrue(actualPayload.containsKey("common_flows"));
99                  return;
100             }
101             assertMetadataResponse(handlerCase, actual, actualPayload);
102         }
103     }

149     private MCPResourceURIVariables parseUriVariables(final String uriTemplate, final String resourceUri) {
150         return new MCPUriTemplate(uriTemplate).parse(resourceUri).orElseThrow();
151     }

153     private <T extends MCPRequestContext> MCPSuccessPayload handle(final MCPResourceHandler<T> handler,
154                                                                    final MCPFeatureRuntimeRequestContext requestContext, final MCPResourceURIVariables uriVariables) {
155         return handler.handle(handler.getContextType().cast(requestContext), uriVariables);
156     }

158     private void assertMetadataResponse(final HandlerCase handlerCase, final MCPSuccessPayload actual, final Map<String, Object> actualPayload) {
159         if (actualPayload.containsKey("resource_kind")) {
160             assertThat(actual, isA(MCPMapPayload.class));
161             assertThat(actualPayload.get("resource_kind"), is("detail"));
162         } else {
163             assertThat(actual, isA(MCPItemsPayload.class));
164         }
165         assertThat(actualPayload.get("count"), is(handlerCase.getExpectedObjectNames().size()));
166         assertThat(((Map<?, ?>) actualPayload.get("self_resource")).get("uri"), is(handlerCase.getResourceUri()));
167         assertParentResource(handlerCase.getResourceUri(), actualPayload);
168         assertNextResources(handlerCase.getResourceUri(), actualPayload);
169         assertThat(extractMetadataNames(actualPayload), is(handlerCase.getExpectedObjectNames()));
170     }
171     
172     private void assertParentResource(final String resourceUri, final Map<String, Object> actualPayload) {
173         String expectedParentUri = createParentUri(resourceUri);
174         if (expectedParentUri.isEmpty()) {
175             assertFalse(actualPayload.containsKey("parent_resource"));
176             return;
177         }
178         assertThat(((Map<?, ?>) actualPayload.get("parent_resource")).get("uri"), is(expectedParentUri));
179     }
180     
181     private String createParentUri(final String resourceUri) {
182         String prefix = "shardingsphere://";
183         String path = resourceUri.substring(prefix.length());
184         int lastSeparatorIndex = path.lastIndexOf('/');
185         return 0 > lastSeparatorIndex ? "" : prefix + path.substring(0, lastSeparatorIndex);
186     }
187     
188     private void assertNextResources(final String resourceUri, final Map<String, Object> actualPayload) {
189         if (!"shardingsphere://databases/logic_db/schemas/public/tables/orders".equals(resourceUri)) {
190             return;
191         }
192         assertThat(extractResourceUris((List<?>) actualPayload.get("next_resources")), is(List.of(
193                 "shardingsphere://databases/logic_db/schemas/public/tables/orders/columns",
194                 "shardingsphere://databases/logic_db/schemas/public/tables/orders/indexes")));
195     }
196     
197     private List<String> extractResourceUris(final List<?> resources) {
198         List<String> result = new LinkedList<>();
199         for (Object each : resources) {
200             result.add((String) ((Map<?, ?>) each).get("uri"));
201         }
202         return result;
203     }
204     
205     private List<String> extractMetadataNames(final Map<String, Object> payload) {
206         List<String> result = new LinkedList<>();
207         for (Object each : getMetadataItems(payload)) {
208             if (each instanceof Map) {
209                 result.add(extractMetadataName((Map<?, ?>) each));
210             }
211         }
212         return result;
213     }
214     
215     private String extractMetadataName(final Map<?, ?> metadata) {
216         for (String each : List.of("sequence", "index", "column", "view", "table", "schema", "database")) {
217             if (metadata.containsKey(each)) {
218                 return String.valueOf(metadata.get(each));
219             }
220         }
221         if (metadata.containsKey("name")) {
222             return String.valueOf(metadata.get("name"));
223         }
224         if (metadata.containsKey("table_name")) {
225             return String.valueOf(metadata.get("table_name"));
226         }
227         return metadata.containsKey("storage_unit_name") ? String.valueOf(metadata.get("storage_unit_name")) : "";
228     }

230     @SuppressWarnings("unchecked")
231     private List<Object> getMetadataItems(final Map<String, Object> payload) {
232         return (List<Object>) payload.get("items");
233     }
