37  /**
38   * Readwrite-splitting status workflow planning service.
39   */
40  public final class ReadwriteSplittingStatusWorkflowPlanningService {
41      
42      private static final String CLUSTER_MODE = "Cluster";
43      
44      private static final List<String> INTERACTION_STEPS = List.of(
45              "Confirm database, rule, read storage unit and target status",
46              "Confirm Cluster mode and inspect DistSQL-visible readwrite-splitting status",
47              "Generate readwrite-splitting status DistSQL artifact",
48              "Review artifacts and choose execution mode",
49              "Execute or export artifacts",
50              "Validate and summarize");
51      
52      private static final List<String> VALIDATION_LAYERS = List.of("status");
53      
54      private final WorkflowPlanningSupport planningSupport = new WorkflowPlanningSupport();
55      
56      private final ReadwriteSplittingInspectionService inspectionService = new ReadwriteSplittingInspectionService();
57      
58      private final ReadwriteSplittingStatusDistSQLPlanningService distSQLPlanningService = new ReadwriteSplittingStatusDistSQLPlanningService();
59      
60      /**
61       * Plan readwrite-splitting storage-unit status workflow.
62       *
63       * @param workflowSessionContext workflow session context
64       * @param queryFacade query facade
65       * @param request workflow request
66       * @return workflow snapshot
67       */
68      public WorkflowContextSnapshot plan(final WorkflowSessionContext workflowSessionContext, final MCPFeatureQueryFacade queryFacade,
69                                          final ReadwriteSplittingStatusWorkflowRequest request) {
70          WorkflowContextSnapshot result = workflowSessionContext.getOrCreate(request.getPlanId());
71          ReadwriteSplittingStatusWorkflowRequest mergedRequest = prepareSnapshot(result, request);
72          ClarifiedIntent clarifiedIntent = result.getClarifiedIntent();
73          if (!ensurePlanningContext(mergedRequest, clarifiedIntent, result)) {
74              return workflowSessionContext.persist(result, WorkflowLifecycle.STEP_CLARIFYING, result.getStatus());
75          }
76          queryFacade.checkDatabaseCapability(mergedRequest.getDatabase());
77          if (!ensureClusterMode(inspectionService.queryProxyMode(queryFacade, mergedRequest.getDatabase()), result)) {
78              return workflowSessionContext.persist(result, WorkflowLifecycle.STEP_FAILED, WorkflowLifecycle.STATUS_FAILED);
79          }
80          List<Map<String, Object>> statuses = inspectionService.queryRuleStatus(queryFacade, mergedRequest.getDatabase(), mergedRequest.getRuleName());
81          if (!ensureTargetStatusRow(mergedRequest, statuses, result, queryFacade)) {
82              return workflowSessionContext.persist(result, WorkflowLifecycle.STEP_FAILED, WorkflowLifecycle.STATUS_FAILED);
83          }
84          result.getRuleArtifacts().add(distSQLPlanningService.planStatus(mergedRequest));
85          return workflowSessionContext.persist(result, WorkflowLifecycle.STEP_REVIEW, WorkflowLifecycle.STATUS_PLANNED);
86      }
87      
88      private ReadwriteSplittingStatusWorkflowRequest prepareSnapshot(final WorkflowContextSnapshot snapshot, final ReadwriteSplittingStatusWorkflowRequest request) {
89          ReadwriteSplittingStatusWorkflowRequest result = ReadwriteSplittingStatusWorkflowRequest.merge(snapshot.getRequest(), request);
90          planningSupport.prepareSnapshot(snapshot, ReadwriteSplittingFeatureDefinition.STATUS_WORKFLOW_KIND, result, null,
91                  new ClarifiedIntent(), "Readwrite-splitting status workflow plan.", INTERACTION_STEPS, VALIDATION_LAYERS);
92          snapshot.getResourceUriTemplates().add(ReadwriteSplittingFeatureDefinition.STORAGE_UNITS_RESOURCE_URI);
93          return result;
94      }
95      
96      private boolean ensurePlanningContext(final ReadwriteSplittingStatusWorkflowRequest request, final ClarifiedIntent clarifiedIntent, final WorkflowContextSnapshot snapshot) {
97          if (request.getDatabase().isEmpty()) {
98              clarifiedIntent.getClarificationMessages().add("Please provide logical database first.");
99              snapshot.getIssues().add(new WorkflowIssue(WorkflowIssueCode.DATABASE_REQUIRED, "error", WorkflowLifecycle.STEP_INTAKING,
100                     "Database is required before planning readwrite-splitting status DistSQL.", "Provide the logical database name.", true, Map.of()));
101             snapshot.setStatus(WorkflowLifecycle.STATUS_CLARIFYING);
102             return false;
103         }
104         if (!planningSupport.ensureOptionalSupportedIdentifiers("database", List.of(request.getDatabase()), snapshot, WorkflowLifecycle.STEP_INTAKING)
105                 || !planningSupport.ensureOptionalSupportedIdentifiers(ReadwriteSplittingFeatureDefinition.RULE_FIELD, List.of(request.getRuleName()),
106                         snapshot, WorkflowLifecycle.STEP_INTAKING)
107                 || !planningSupport.ensureOptionalSupportedIdentifiers(ReadwriteSplittingFeatureDefinition.STORAGE_UNIT_FIELD, List.of(request.getStorageUnit()),
108                         snapshot, WorkflowLifecycle.STEP_INTAKING)) {
109             snapshot.setStatus(WorkflowLifecycle.STATUS_FAILED);
110             return false;
111         }
112         List<String> missingInputs = new LinkedList<>();
113         addMissingInput(missingInputs, request.getRuleName(), ReadwriteSplittingFeatureDefinition.RULE_FIELD);
114         addMissingInput(missingInputs, request.getStorageUnit(), ReadwriteSplittingFeatureDefinition.STORAGE_UNIT_FIELD);
115         addMissingInput(missingInputs, distSQLPlanningService.resolveStatusOperation(request), ReadwriteSplittingFeatureDefinition.TARGET_STATUS_FIELD);
116         if (!missingInputs.isEmpty()) {
117             clarifiedIntent.getClarificationMessages().add("Please provide readwrite-splitting rule name, read storage unit and target status.");
118             snapshot.getIssues().add(new WorkflowIssue(WorkflowIssueCode.RULE_INPUT_REQUIRED, "error", WorkflowLifecycle.STEP_INTAKING,
119                     "Readwrite-splitting status DistSQL requires explicit rule, storage unit and status inputs.",
120                     "Provide target_status as enable or disable.", true, Map.of("missing_inputs", missingInputs)));
121             snapshot.setStatus(WorkflowLifecycle.STATUS_CLARIFYING);
122             return false;
123         }
124         return true;
125     }
126     
127     private boolean ensureClusterMode(final String proxyMode, final WorkflowContextSnapshot snapshot) {
128         if (CLUSTER_MODE.equalsIgnoreCase(proxyMode)) {
129             return true;
130         }
131         snapshot.getIssues().add(new WorkflowIssue(WorkflowIssueCode.CLUSTER_MODE_REQUIRED, "error", WorkflowLifecycle.STEP_DISCOVERING,
132                 String.format("Readwrite-splitting storage-unit status changes require Cluster mode; current Proxy mode is `%s`.", proxyMode),
133                 "Connect MCP to a Cluster-mode ShardingSphere Proxy, then start a new readwrite-splitting status plan.", false,
134                 Map.of("required_mode", CLUSTER_MODE, "actual_mode", proxyMode)));
135         return false;
136     }
137     
138     private void addMissingInput(final List<String> missingInputs, final String value, final String fieldName) {
139         if (value.isEmpty()) {
140             missingInputs.add(fieldName);
141         }
142     }
143     
144     private boolean ensureTargetStatusRow(final ReadwriteSplittingStatusWorkflowRequest request, final List<Map<String, Object>> statuses,
145                                           final WorkflowContextSnapshot snapshot, final MCPFeatureQueryFacade queryFacade) {
146         if (statuses.stream().anyMatch(each -> matchesStatusTarget(request, each, queryFacade))) {
147             return true;
148         }
149         snapshot.getIssues().add(new WorkflowIssue(WorkflowIssueCode.DROP_TARGET_RULE_NOT_FOUND, "error", WorkflowLifecycle.STEP_DISCOVERING,
150                 String.format("Readwrite-splitting status target `%s.%s` does not exist.", request.getRuleName(), request.getStorageUnit()),
151                 "Confirm the target rule and read storage unit before planning status changes.", false,
152                 Map.of("rule", request.getRuleName(), "storage_unit", request.getStorageUnit())));
153         return false;
154     }
155     
156     private boolean matchesStatusTarget(final ReadwriteSplittingStatusWorkflowRequest request, final Map<String, Object> status, final MCPFeatureQueryFacade queryFacade) {
157         return queryFacade.isSameIdentifier(request.getDatabase(), IdentifierScope.TABLE, request.getRuleName(), WorkflowRuleValueUtils.getRuleValue(status, "name"))
158                 && queryFacade.isSameIdentifier(request.getDatabase(), IdentifierScope.TABLE, request.getStorageUnit(), WorkflowRuleValueUtils.getRuleValue(status, "storage_unit"));
159     }
