53  class ReadwriteSplittingContextManagerLifecycleListenerTest {
54      
55      private final ReadwriteSplittingContextManagerLifecycleListener listener = new ReadwriteSplittingContextManagerLifecycleListener();
56      
57      @ParameterizedTest(name = "{0}")
58      @MethodSource("onInitializedArguments")
59      void assertOnInitialized(final String name, final Collection<String> databaseNames, final Map<String, DataSourceState> states,
60                               final List<String> expectedQualifiedDataSources, final List<DataSourceState> expectedStates, final int expectedUpdateCount) {
61          ContextManager contextManager = mock(ContextManager.class, RETURNS_DEEP_STUBS);
62          StaticDataSourceRuleAttribute attribute = mock(StaticDataSourceRuleAttribute.class);
63          Map<String, QualifiedDataSourceState> qualifiedDataSourceStates = states.entrySet().stream().collect(Collectors.toMap(Entry::getKey, entry -> new QualifiedDataSourceState(entry.getValue())));
64          Collection<ShardingSphereDatabase> databases = createDatabases(databaseNames, attribute);
65          when(contextManager.getPersistServiceFacade().getQualifiedDataSourceStateService().load()).thenReturn(qualifiedDataSourceStates);
66          when(contextManager.getMetaDataContexts().getMetaData().getAllDatabases()).thenReturn(databases);
67          listener.onInitialized(contextManager);
68          ArgumentCaptor<QualifiedDataSource> qualifiedDataSourceCaptor = ArgumentCaptor.forClass(QualifiedDataSource.class);
69          ArgumentCaptor<DataSourceState> stateCaptor = ArgumentCaptor.forClass(DataSourceState.class);
70          verify(attribute, times(expectedUpdateCount)).updateStatus(qualifiedDataSourceCaptor.capture(), stateCaptor.capture());
71          assertThat(qualifiedDataSourceCaptor.getAllValues().stream().map(QualifiedDataSource::toString).collect(Collectors.toList()), is(expectedQualifiedDataSources));
72          assertThat(stateCaptor.getAllValues(), is(expectedStates));
73      }
74      
75      private Collection<ShardingSphereDatabase> createDatabases(final Collection<String> databaseNames, final StaticDataSourceRuleAttribute attribute) {
76          Collection<ShardingSphereDatabase> result = new LinkedList<>();
77          for (String each : databaseNames) {
78              ShardingSphereDatabase database = mock(ShardingSphereDatabase.class);
79              when(database.getName()).thenReturn(each);
80              RuleMetaData ruleMetaData = mock(RuleMetaData.class);
81              when(ruleMetaData.getAttributes(StaticDataSourceRuleAttribute.class)).thenReturn(Collections.singletonList(attribute));
82              when(database.getRuleMetaData()).thenReturn(ruleMetaData);
83              result.add(database);
84          }
85          return result;
86      }
87      
88      @Test
89      void assertOnDestroyed() {
90          assertDoesNotThrow(() -> listener.onDestroyed(mock(ContextManager.class)));
91      }
92      
93      private static Stream<Arguments> onInitializedArguments() {
94          return Stream.of(
95                  Arguments.of("matched database", Collections.singletonList("foo_db"), Collections.singletonMap("foo_db.group_0.ds_0", DataSourceState.DISABLED),
96                          Collections.singletonList("foo_db.group_0.ds_0"), Collections.singletonList(DataSourceState.DISABLED), 1),
97                  Arguments.of("mismatched database", Collections.singletonList("bar_db"), Collections.singletonMap("foo_db.group_0.ds_0", DataSourceState.DISABLED),
98                          Collections.emptyList(), Collections.emptyList(), 0),
99                  Arguments.of("matched and mismatched databases", Arrays.asList("foo_db", "bar_db"), createDataSourceStates(),
100                         Arrays.asList("foo_db.group_0.ds_0", "foo_db.group_0.ds_1"), Arrays.asList(DataSourceState.DISABLED, DataSourceState.ENABLED), 2));
101     }
102     
103     private static Map<String, DataSourceState> createDataSourceStates() {
104         Map<String, DataSourceState> result = new LinkedHashMap<>(2, 1F);
105         result.put("foo_db.group_0.ds_0", DataSourceState.DISABLED);
106         result.put("foo_db.group_0.ds_1", DataSourceState.ENABLED);
107         return result;
108     }
