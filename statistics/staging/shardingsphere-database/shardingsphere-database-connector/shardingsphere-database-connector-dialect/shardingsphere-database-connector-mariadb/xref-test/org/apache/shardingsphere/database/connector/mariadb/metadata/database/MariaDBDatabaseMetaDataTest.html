40  import static org.junit.jupiter.api.Assertions.assertFalse;
41  import static org.junit.jupiter.api.Assertions.assertTrue;
42  
43  class MariaDBDatabaseMetaDataTest {
44      
45      private final DatabaseType databaseType = TypedSPILoader.getService(DatabaseType.class, "MariaDB");
46      
47      private final DialectDatabaseMetaData metaData = DatabaseTypedSPILoader.getService(DialectDatabaseMetaData.class, databaseType);
48      
49      @Test
50      void assertGetQuoteCharacter() {
51          assertThat(metaData.getQuoteCharacter(), is(QuoteCharacter.BACK_QUOTE));
52      }
53      
54      @Test
55      void assertGetIdentifierPatternType() {
56          assertThat(metaData.getIdentifierPatternType(), is(IdentifierPatternType.KEEP_ORIGIN));
57      }
58      
59      @Test
60      void assertGetDefaultNullsOrderType() {
61          assertThat(metaData.getDefaultNullsOrderType(), is(NullsOrderType.LOW));
62      }
63      
64      @Test
65      void assertGetDataTypeOption() {
66          assertThat(metaData.getDataTypeOption(), isA(MySQLDataTypeOption.class));
67      }
68      
69      @Test
70      void assertGetColumnOption() {
71          assertFalse(metaData.getColumnOption().isColumnNameEqualsLabelInColumnProjection());
72      }
73      
74      @Test
75      void assertGetConnectionOption() {
76          DialectConnectionOption actual = metaData.getConnectionOption();
77          assertTrue(actual.isInstanceConnectionAvailable());
78          assertTrue(actual.isSupportThreeTierStorageStructure());
79      }
80      
81      @Test
82      void assertGetTransactionOption() {
83          DialectTransactionOption actual = metaData.getTransactionOption();
84          assertFalse(actual.isSupportGlobalCSN());
85          assertFalse(actual.isDDLNeedImplicitCommit());
86          assertTrue(actual.isSupportAutoCommitInNestedTransaction());
87          assertFalse(actual.isSupportDDLInXATransaction());
88          assertTrue(actual.isSupportMetaDataRefreshInTransaction());
89          assertThat(actual.getDefaultIsolationLevel(), is(Connection.TRANSACTION_REPEATABLE_READ));
90          assertFalse(actual.isReturnRollbackStatementWhenCommitFailed());
91          assertFalse(actual.isAllowCommitAndRollbackOnlyWhenTransactionFailed());
92          assertThat(actual.getXaDriverClassNames().size(), is(1));
93          assertTrue(actual.getXaDriverClassNames().contains("org.mariadb.jdbc.MariaDbDataSource"));
94      }
95      
96      @Test
97      void assertGetJoinOption() {
98          DialectJoinOption actual = metaData.getJoinOption();
99          assertTrue(actual.isUsingColumnsByProjectionOrder());
100         assertTrue(actual.isRightColumnsByFirstOrder());
101     }
102     
103     @Test
104     void assertGetGeneratedKeyOption() {
105         Optional<DialectGeneratedKeyOption> actual = metaData.getGeneratedKeyOption();
106         assertTrue(actual.isPresent());
107         assertThat(actual.map(DialectGeneratedKeyOption::getColumnName).orElse(""), is("GENERATED_KEY"));
108     }
109     
110     @Test
111     void assertGetProtocolVersionOption() {
112         assertThat(metaData.getProtocolVersionOption().getDefaultVersion(), is("5.7.22"));
113     }
114 }
