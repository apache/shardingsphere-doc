38      @Test
39      void assertHandle() {
40          MCPFeatureRuntimeRequestContext requestContext =
41                  new MCPFeatureRuntimeRequestContext(ResourceTestDataFactory.createRuntimeContext(ResourceTestDataFactory.createDatabaseMetadata(), MCPTransportType.HTTP),
42                          new MCPSessionIdentity("session-1", "", "", Map.of()));
43          Map<String, Object> actual = new RuntimeStatusHandler().handle(requestContext, new MCPResourceURIVariables(Map.of())).toPayload();
44          assertThat(actual.get("response_mode"), is("runtime"));
45          assertThat(actual.get("summary"), is("Runtime is ready with 3 configured logical database(s)."));
46          assertThat(actual.get("status"), is("available"));
47          assertThat(actual.get("active_transport"), is("http"));
48          assertThat(((Map<?, ?>) actual.get("transport_security_summary")).get("recommended_exposure"), is("loopback_or_trusted_gateway"));
49          assertThat(actual.get("configured_database_count"), is(3));
50          assertTrue(((List<?>) actual.get("databases")).stream().map(each -> ((Map<?, ?>) each).get("database")).anyMatch("logic_db"::equals));
51          assertRuntimeDiagnostics(actual, "ready");
52          assertRuntimeProtection(actual);
53          assertFalse(actual.containsKey("capability_fingerprint"));
54          assertRuntimeCapability((List<?>) actual.get("databases"), "logic_db");
55          assertThat(extractResourceUris((List<?>) actual.get("resources_to_read")), is(List.of("shardingsphere://capabilities", "shardingsphere://databases")));
56          assertThat(((Map<?, ?>) ((List<?>) actual.get("next_actions")).getFirst()).get("resource_uri"), is("shardingsphere://databases"));
57      }

69      @Test
70      void assertHandleWithEmptyRuntimeDatabase() {
71          MCPFeatureRuntimeRequestContext requestContext = new MCPFeatureRuntimeRequestContext(ResourceTestDataFactory.createRuntimeContext(List.of(), MCPTransportType.HTTP),
72                  new MCPSessionIdentity("session-1", "", "", Map.of()));
73          Map<String, Object> actual = new RuntimeStatusHandler().handle(requestContext, new MCPResourceURIVariables(Map.of())).toPayload();
74          assertThat(actual.get("summary"), is("Runtime requires at least one configured logical database before metadata discovery or SQL execution."));
75          assertThat(actual.get("status"), is("configuration_required"));
76          assertThat(actual.get("configured_database_count"), is(0));
77          assertThat(((List<?>) actual.get("databases")).size(), is(0));
78          Map<?, ?> readiness = (Map<?, ?>) actual.get("readiness");
79          assertFalse((Boolean) readiness.get("ready"));
80          assertThat(readiness.get("reason"), is("No runtime databases are configured."));
81          assertRuntimeDiagnostics(actual, "invalid_configuration");
82          assertRuntimeProtection(actual);
83          assertThat(extractResourceUris((List<?>) actual.get("resources_to_read")), is(List.of("shardingsphere://capabilities")));
84          List<?> nextActions = (List<?>) actual.get("next_actions");
85          assertThat(((Map<?, ?>) nextActions.get(0)).get("type"), is("resource_read"));
86          assertThat(((Map<?, ?>) nextActions.get(1)).get("type"), is("ask_user"));
87          assertThat(((Map<?, ?>) nextActions.get(1)).get("required_inputs"), is(List.of("runtimeDatabases")));
88      }

90      private void assertRuntimeDiagnostics(final Map<String, Object> payload, final String expectedCategory) {
91          Map<?, ?> actualDiagnostics = (Map<?, ?>) payload.get("diagnostics");
92          assertThat(actualDiagnostics.get("current_category"), is(expectedCategory));
93          List<?> actualSafeCategories = (List<?>) actualDiagnostics.get("safe_categories");
94          assertTrue(actualSafeCategories.contains("missing_jdbc_driver"));
95          assertTrue(actualSafeCategories.contains("authentication_failed"));
96          assertTrue(actualSafeCategories.contains("authorization_failed"));
97          assertTrue(actualSafeCategories.contains("connection_timeout"));
98          assertTrue(actualSafeCategories.contains("invalid_configuration"));
99          assertTrue(actualSafeCategories.contains("database_unavailable"));
100         assertTrue(actualSafeCategories.contains("connection_failed"));
101         assertTrue(actualSafeCategories.contains("database_not_visible"));
102         List<?> actualOperatorNextActions = (List<?>) actualDiagnostics.get("operator_next_actions");
103         assertThat(actualOperatorNextActions.size(), is(8));
104         Map<?, ?> actualInvalidConfigurationAction = (Map<?, ?>) actualOperatorNextActions.get(4);
105         assertThat(actualInvalidConfigurationAction.get("category"), is("invalid_configuration"));
106         assertThat(actualInvalidConfigurationAction.get("operator_action"), is("Fix runtimeDatabases JDBC URL, driver, or binding configuration."));
107         assertTrue((Boolean) actualInvalidConfigurationAction.get("secret_safe"));
108         assertThat(((Map<?, ?>) actualOperatorNextActions.get(7)).get("category"), is("database_not_visible"));
109     }
110     
111     private void assertRuntimeProtection(final Map<String, Object> payload) {
112         Map<?, ?> actualRuntimeProtection = (Map<?, ?>) payload.get("runtime_protection");
113         Map<?, ?> actualToolCallLimit = (Map<?, ?>) actualRuntimeProtection.get("tool_call_limit");
114         assertThat(actualToolCallLimit.get("scope"), is("session"));
115         assertThat(actualToolCallLimit.get("property"), is(MCPRuntimeProtectionPolicy.MAX_TOOL_CALLS_PER_SESSION_PROPERTY));
116         Map<?, ?> actualSQLExecutionLimits = (Map<?, ?>) actualRuntimeProtection.get("sql_execution_limits");
117         Map<?, ?> actualMaxRows = (Map<?, ?>) actualSQLExecutionLimits.get("max_rows");
118         assertThat(actualMaxRows.get("default_value"), is(MCPRuntimeProtectionPolicy.DEFAULT_MAX_ROWS));
119         assertThat(actualMaxRows.get("maximum_value"), is(MCPRuntimeProtectionPolicy.MAX_ROWS_LIMIT));
120         assertThat(actualMaxRows.get("applied_field"), is("applied_max_rows"));
121         assertThat(actualMaxRows.get("truncation_field"), is("truncated"));
122         Map<?, ?> actualTimeout = (Map<?, ?>) actualSQLExecutionLimits.get("timeout_ms");
123         assertThat(actualTimeout.get("default_value"), is(MCPRuntimeProtectionPolicy.DEFAULT_TIMEOUT_MILLISECONDS));
124         assertThat(actualTimeout.get("maximum_value"), is(MCPRuntimeProtectionPolicy.MAX_TIMEOUT_MILLISECONDS));
125         assertThat(actualTimeout.get("applied_field"), is("applied_timeout_ms"));
126         assertThat(actualTimeout.get("zero_means"), is("server_default"));
127     }
128     
129     private void assertRuntimeCapability(final List<?> databases, final String databaseName) {
130         Map<?, ?> actualDatabase = databases.stream().map(each -> (Map<?, ?>) each).filter(each -> databaseName.equals(each.get("database"))).findFirst().orElseThrow();
131         Map<?, ?> actualCapabilities = (Map<?, ?>) actualDatabase.get("capabilities");
132         assertTrue((Boolean) actualCapabilities.get("available"));
133         assertTrue(((List<?>) actualCapabilities.get("supported_statement_classes")).contains("QUERY"));
134         assertTrue(((List<?>) actualCapabilities.get("supported_metadata_object_types")).contains("TABLE"));
135     }
136     
137     private List<String> extractResourceUris(final List<?> resources) {
138         return resources.stream().map(each -> (String) ((Map<?, ?>) each).get("uri")).toList();
139     }
