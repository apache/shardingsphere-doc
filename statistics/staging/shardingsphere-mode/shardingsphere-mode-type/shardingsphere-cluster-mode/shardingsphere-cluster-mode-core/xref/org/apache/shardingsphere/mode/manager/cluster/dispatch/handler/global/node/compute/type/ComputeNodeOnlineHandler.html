41  /**
42   *  Compute node online handler.
43   */
44  public final class ComputeNodeOnlineHandler implements ComputeNodeChangedHandler {
45      
46      @Override
47      public NodePath getSubscribedNodePath() {
48          return new OnlineNodePath(null, null);
49      }
50      
51      @Override
52      public Collection<Type> getSubscribedTypes() {
53          return Arrays.asList(Type.ADDED, Type.DELETED);
54      }
55      
56      @Override
57      public void handle(final ContextManager contextManager, final DataChangedEvent event) {
58          if (!NodePathSearcher.isMatchedPath(event.getKey(), OnlineNodePath.createInstanceIdSearchCriteria())) {
59              return;
60          }
61          ClusterInstanceRegistry clusterInstanceRegistry = contextManager.getComputeNodeInstanceContext().getClusterInstanceRegistry();
62          InstanceType instanceType = InstanceType.valueOf(NodePathSearcher.get(event.getKey(), OnlineNodePath.createInstanceTypeSearchCriteria()).toUpperCase());
63          String instanceId = NodePathSearcher.get(event.getKey(), OnlineNodePath.createInstanceIdSearchCriteria());
64          ComputeNodeData computeNodeData = new YamlComputeNodeDataSwapper().swapToObject(YamlEngine.unmarshal(event.getValue(), YamlComputeNodeData.class));
65          InstanceMetaData instanceMetaData = InstanceMetaDataFactory.create(instanceId, instanceType, computeNodeData);
66          ClusterPersistServiceFacade clusterPersistServiceFacade = (ClusterPersistServiceFacade) contextManager.getPersistServiceFacade().getModeFacade();
67          switch (event.getType()) {
68              case ADDED:
69                  clusterInstanceRegistry.add(clusterPersistServiceFacade.getComputeNodeService().loadInstance(instanceMetaData));
70                  break;
71              case DELETED:
72                  clusterInstanceRegistry.delete(new ComputeNodeInstance(instanceMetaData));
73                  break;
74              default:
75          }
76      }
77  }
