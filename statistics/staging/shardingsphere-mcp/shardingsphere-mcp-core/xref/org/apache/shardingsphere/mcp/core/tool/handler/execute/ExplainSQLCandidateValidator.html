18  package org.apache.shardingsphere.mcp.core.tool.handler.execute;
19  
20  import lombok.AccessLevel;
21  import lombok.RequiredArgsConstructor;
22  import org.apache.shardingsphere.infra.exception.ShardingSpherePreconditions;
23  import org.apache.shardingsphere.mcp.api.exception.MCPInvalidRequestException;
24  import org.apache.shardingsphere.mcp.support.database.capability.MCPDatabaseCapability;
25  import org.apache.shardingsphere.mcp.support.database.capability.SupportedMCPStatement;
26  
27  import java.util.Set;
28  
29  @RequiredArgsConstructor(access = AccessLevel.PACKAGE)
30  final class ExplainSQLCandidateValidator {
31      
32      private static final Set<String> STATEMENT_START_KEYWORDS = Set.of(
33              "EXPLAIN", "SELECT", "WITH", "VALUES", "INSERT", "UPDATE", "DELETE", "MERGE", "REPLACE", "UPSERT",
34              "CREATE", "ALTER", "DROP", "TRUNCATE", "RENAME", "ATTACH", "DETACH", "EXCHANGE", "UNDROP", "OPTIMIZE",
35              "GRANT", "REVOKE", "CALL", "EXEC", "EXECUTE", "DECLARE", "PREPARE", "DEALLOCATE", "DESCRIBE", "SHOW",
36              "DO", "COPY", "LOAD", "UNLOAD", "IMPORT", "EXPORT", "BEGIN", "START", "COMMIT", "ROLLBACK", "SAVEPOINT", "RELEASE",
37              "SET", "RESET", "USE", "PARALLEL");
38      
39      private final MCPStatementAnalyzer statementAnalyzer;
40      
41      ClassificationResult validate(final String sql, final String explainSql, final MCPDatabaseCapability databaseCapability) {
42          ClassificationResult explainedStatement = statementAnalyzer.analyze(sql, databaseCapability);
43          ShardingSpherePreconditions.checkState(SupportedMCPStatement.QUERY == explainedStatement.getStatementClass(),
44                  () -> new MCPInvalidRequestException("database_gateway_execute_explain_query only supports QUERY statements as the explained SQL."));
45          SQLStatementScanner scanner = new SQLStatementScanner(databaseCapability.getDatabaseType(), explainSql);
46          String actualExplainSql = scanner.sql();
47          ShardingSpherePreconditions.checkState(!scanner.containsExecutableComment(),
48                  () -> new MCPInvalidRequestException("Executable comments are not supported by the MCP explain query tool."));
49          ShardingSpherePreconditions.checkState(scanner.startsWithKeyword("EXPLAIN"),
50                  () -> new MCPInvalidRequestException("explain_sql must start with EXPLAIN."));
51          SQLStatementScanner prefixScanner = scanner.scan(extractExplainPrefix(actualExplainSql, explainedStatement.getNormalizedSql()));
52          ShardingSpherePreconditions.checkState(!prefixScanner.containsKeywordSequence("EXPLAIN", "PLAN", "FOR"),
53                  () -> new MCPInvalidRequestException("EXPLAIN PLAN FOR workflows are not supported by the MCP explain query tool."));
54          ShardingSpherePreconditions.checkState(!prefixScanner.containsKeyword("ANALYZE", "ANALYSE"),
55                  () -> new MCPInvalidRequestException("EXPLAIN ANALYZE is not supported by the MCP explain query tool."));
56          ShardingSpherePreconditions.checkState(!prefixScanner.containsKeyword("INTO"),
57                  () -> new MCPInvalidRequestException("EXPLAIN output redirection is not supported by the MCP explain query tool."));
58          ShardingSpherePreconditions.checkState(!prefixScanner.containsKeywordAfterFirst(STATEMENT_START_KEYWORDS),
59                  () -> new MCPInvalidRequestException("explain_sql must not wrap the original sql argument in another statement."));
60          return new ClassificationResult(SupportedMCPStatement.EXPLAIN, "EXPLAIN", actualExplainSql, "", explainedStatement.getReferencedObjects(), false);
61      }
62      
63      private String extractExplainPrefix(final String explainSql, final String sql) {
64          ShardingSpherePreconditions.checkState(explainSql.endsWith(sql),
65                  () -> new MCPInvalidRequestException("explain_sql must include the original sql argument without rewriting it."));
66          return explainSql.substring(0, explainSql.length() - sql.length());
67      }
68  }
