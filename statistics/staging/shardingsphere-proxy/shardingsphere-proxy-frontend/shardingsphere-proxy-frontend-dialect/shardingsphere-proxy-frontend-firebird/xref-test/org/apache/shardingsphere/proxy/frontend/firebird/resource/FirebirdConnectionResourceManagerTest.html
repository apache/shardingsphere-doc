45  import static org.mockito.Mockito.verify;

48  @ExtendWith(AutoMockExtension.class)
49  @StaticMockSettings({
50          FirebirdTransactionIdGenerator.class, FirebirdStatementIdGenerator.class, FirebirdBlobIdGenerator.class, FirebirdBlobHandleGenerator.class,
51          FirebirdBlobWriteCache.class, FirebirdBlobReadCache.class, FirebirdConnectionProtocolVersion.class, FirebirdFetchStatementCache.class,
52          FirebirdBatchStatementManager.class
53  })
54  class FirebirdConnectionResourceManagerTest {
55      
56      private static final int CONNECTION_ID = 1;
57      
58      @Mock
59      private FirebirdTransactionIdGenerator transactionIdGenerator;
60      
61      @Mock
62      private FirebirdStatementIdGenerator statementIdGenerator;
63      
64      @Mock
65      private FirebirdBlobIdGenerator blobIdGenerator;
66      
67      @Mock
68      private FirebirdBlobHandleGenerator blobHandleGenerator;
69      
70      @Mock
71      private FirebirdBlobWriteCache blobWriteCache;
72      
73      @Mock
74      private FirebirdBlobReadCache blobReadCache;
75      
76      @Mock
77      private FirebirdConnectionProtocolVersion connectionProtocolVersion;
78      
79      @Mock
80      private FirebirdFetchStatementCache fetchStatementCache;
81      
82      @Mock
83      private FirebirdBatchStatementManager batchStatementManager;
84      
85      @BeforeEach
86      void setUp() {
87          when(FirebirdTransactionIdGenerator.getInstance()).thenReturn(transactionIdGenerator);
88          when(FirebirdStatementIdGenerator.getInstance()).thenReturn(statementIdGenerator);
89          when(FirebirdBlobIdGenerator.getInstance()).thenReturn(blobIdGenerator);
90          when(FirebirdBlobHandleGenerator.getInstance()).thenReturn(blobHandleGenerator);
91          when(FirebirdBlobWriteCache.getInstance()).thenReturn(blobWriteCache);
92          when(FirebirdBlobReadCache.getInstance()).thenReturn(blobReadCache);
93          when(FirebirdConnectionProtocolVersion.getInstance()).thenReturn(connectionProtocolVersion);
94          when(FirebirdFetchStatementCache.getInstance()).thenReturn(fetchStatementCache);
95          when(FirebirdBatchStatementManager.getInstance()).thenReturn(batchStatementManager);
96      }
97      
98      @Test
99      void assertRegisterConnection() {
100         FirebirdConnectionResourceManager.getInstance().registerConnection(CONNECTION_ID);
101         InOrder inOrder = inOrder(transactionIdGenerator, statementIdGenerator, blobIdGenerator, blobHandleGenerator, blobWriteCache, blobReadCache,
102                 fetchStatementCache, batchStatementManager);
103         inOrder.verify(transactionIdGenerator).registerConnection(CONNECTION_ID);
104         inOrder.verify(statementIdGenerator).registerConnection(CONNECTION_ID);
105         inOrder.verify(blobIdGenerator).registerConnection(CONNECTION_ID);
106         inOrder.verify(blobHandleGenerator).registerConnection(CONNECTION_ID);
107         inOrder.verify(blobWriteCache).registerConnection(CONNECTION_ID);
108         inOrder.verify(blobReadCache).registerConnection(CONNECTION_ID);
109         inOrder.verify(fetchStatementCache).registerConnection(CONNECTION_ID);
110         inOrder.verify(batchStatementManager).registerConnection(CONNECTION_ID);
111     }
112     
113     @Test
114     void assertUnregisterConnection() {
115         byte[] expectedBlobContent = new byte[]{1};
116         FirebirdPacketPayload payload = mock(FirebirdPacketPayload.class);
117         when(payload.getConnectionId()).thenReturn(CONNECTION_ID);
118         new FirebirdBlobBinaryProtocolValue().write(payload, expectedBlobContent);
119         ArgumentCaptor<Long> blobIdCaptor = ArgumentCaptor.forClass(Long.class);
120         verify(payload).writeInt8(blobIdCaptor.capture());
121         assertThat(FirebirdBlobBinaryProtocolValue.getBlobContent(CONNECTION_ID, blobIdCaptor.getValue()), is(expectedBlobContent));
122         FirebirdConnectionResourceManager.getInstance().unregisterConnection(CONNECTION_ID);
123         InOrder inOrder = inOrder(statementIdGenerator, transactionIdGenerator, blobIdGenerator, blobHandleGenerator, blobWriteCache, blobReadCache,
124                 connectionProtocolVersion, fetchStatementCache, batchStatementManager);
125         inOrder.verify(statementIdGenerator).unregisterConnection(CONNECTION_ID);
126         inOrder.verify(transactionIdGenerator).unregisterConnection(CONNECTION_ID);
127         inOrder.verify(blobIdGenerator).unregisterConnection(CONNECTION_ID);
128         inOrder.verify(blobHandleGenerator).unregisterConnection(CONNECTION_ID);
129         inOrder.verify(blobWriteCache).unregisterConnection(CONNECTION_ID);
130         inOrder.verify(blobReadCache).unregisterConnection(CONNECTION_ID);
131         inOrder.verify(connectionProtocolVersion).unsetProtocolVersion(CONNECTION_ID);
132         inOrder.verify(fetchStatementCache).unregisterConnection(CONNECTION_ID);
133         inOrder.verify(batchStatementManager).unregisterConnection(CONNECTION_ID);
134         assertNull(FirebirdBlobBinaryProtocolValue.getBlobContent(CONNECTION_ID, blobIdCaptor.getValue()));
135     }
136 }
