20  import com.google.common.base.Preconditions;
21  import lombok.Getter;
22  import lombok.RequiredArgsConstructor;
23  import org.apache.shardingsphere.database.protocol.binary.BinaryColumnType;
24  import org.firebirdsql.gds.BlrConstants;
25  
26  import java.sql.Types;
27  import java.util.HashMap;
28  import java.util.Map;
29  
30  /**
31   * Column type for Firebird.
32   */
33  @RequiredArgsConstructor
34  @Getter
35  public enum FirebirdBinaryColumnType implements BinaryColumnType {
36      
37      // TODO add support for retrieving length for ARRAY types.
38      // Currently not possible due to existing issues with ARRAY handling in the database itself.
39      TEXT(452, 255),
40      VARYING(448, 0),
41      LEGACY_TEXT(452, 255),
42      LEGACY_VARYING(448, 0),
43      SHORT(500, 2),
44      LONG(496, 4),
45      FLOAT(482, 4),
46      DOUBLE(480, 8),
47      D_FLOAT(530, 8),
48      TIMESTAMP(510, 8),
49      BLOB(520, 0),
50      ARRAY(540, 255),
51      QUAD(550, 4),
52      TIME(560, 4),
53      DATE(570, 4),
54      INT64(580, 8),
55      NUMERIC(580, 8, 1),
56      DECIMAL(580, 8, 2),
57      TIMESTAMP_TZ_EX(32748, 10),
58      TIME_TZ_EX(32750, 6),
59      INT128(32752, 16),
60      TIMESTAMP_TZ(32754, 10),
61      TIME_TZ(32756, 6),
62      DEC16(32760, 2),
63      DEC34(32762, 4),
64      BOOLEAN(32764, 1),
65      NULL(32766, 0);
66      
67      private static final Map<Integer, FirebirdBinaryColumnType> JDBC_TYPE_AND_COLUMN_TYPE_MAP = new HashMap<>(values().length, 1F);
68      
69      private static final Map<Integer, FirebirdBinaryColumnType> BLR_TYPE_AND_COLUMN_TYPE_MAP = new HashMap<>(values().length, 1F);
70      
71      private static final Map<Integer, FirebirdBinaryColumnType> VALUE_AND_COLUMN_TYPE_MAP = new HashMap<>(values().length, 1F);
72      
73      private final int value;
74      
75      private final int length;
76      
77      private final int subtype;
78      
79      static {
80          JDBC_TYPE_AND_COLUMN_TYPE_MAP.put(Types.TINYINT, SHORT);
81          JDBC_TYPE_AND_COLUMN_TYPE_MAP.put(Types.SMALLINT, SHORT);
82          JDBC_TYPE_AND_COLUMN_TYPE_MAP.put(Types.INTEGER, LONG);
83          JDBC_TYPE_AND_COLUMN_TYPE_MAP.put(Types.BIGINT, INT64);
84          JDBC_TYPE_AND_COLUMN_TYPE_MAP.put(Types.FLOAT, FLOAT);
85          JDBC_TYPE_AND_COLUMN_TYPE_MAP.put(Types.REAL, FLOAT);
86          JDBC_TYPE_AND_COLUMN_TYPE_MAP.put(Types.DOUBLE, DOUBLE);
87          JDBC_TYPE_AND_COLUMN_TYPE_MAP.put(Types.NUMERIC, NUMERIC);
88          JDBC_TYPE_AND_COLUMN_TYPE_MAP.put(Types.DECIMAL, DECIMAL);
89          // replace VARYING with TEXT when add proper length
90          JDBC_TYPE_AND_COLUMN_TYPE_MAP.put(Types.CHAR, VARYING);
91          JDBC_TYPE_AND_COLUMN_TYPE_MAP.put(Types.VARCHAR, VARYING);
92          JDBC_TYPE_AND_COLUMN_TYPE_MAP.put(Types.LONGVARCHAR, BLOB);
93          JDBC_TYPE_AND_COLUMN_TYPE_MAP.put(Types.DATE, DATE);
94          JDBC_TYPE_AND_COLUMN_TYPE_MAP.put(Types.TIME, TIME);
95          JDBC_TYPE_AND_COLUMN_TYPE_MAP.put(Types.TIMESTAMP, TIMESTAMP);
96          JDBC_TYPE_AND_COLUMN_TYPE_MAP.put(Types.BINARY, TEXT);
97          JDBC_TYPE_AND_COLUMN_TYPE_MAP.put(Types.VARBINARY, VARYING);
98          JDBC_TYPE_AND_COLUMN_TYPE_MAP.put(Types.LONGVARBINARY, BLOB);
99          JDBC_TYPE_AND_COLUMN_TYPE_MAP.put(Types.NULL, NULL);
100         JDBC_TYPE_AND_COLUMN_TYPE_MAP.put(Types.BLOB, BLOB);
101         JDBC_TYPE_AND_COLUMN_TYPE_MAP.put(Types.BOOLEAN, BOOLEAN);
102         JDBC_TYPE_AND_COLUMN_TYPE_MAP.put(Types.ARRAY, ARRAY);
103         JDBC_TYPE_AND_COLUMN_TYPE_MAP.put(Types.TIME_WITH_TIMEZONE, TIME_TZ);
104         JDBC_TYPE_AND_COLUMN_TYPE_MAP.put(Types.TIMESTAMP_WITH_TIMEZONE, TIMESTAMP_TZ);
105         
106         BLR_TYPE_AND_COLUMN_TYPE_MAP.put(BlrConstants.blr_varying2, VARYING);
107         BLR_TYPE_AND_COLUMN_TYPE_MAP.put(BlrConstants.blr_text2, TEXT);
108         BLR_TYPE_AND_COLUMN_TYPE_MAP.put(BlrConstants.blr_text, LEGACY_TEXT);
109         BLR_TYPE_AND_COLUMN_TYPE_MAP.put(BlrConstants.blr_varying, LEGACY_VARYING);
110         BLR_TYPE_AND_COLUMN_TYPE_MAP.put(BlrConstants.blr_double, DOUBLE);
111         BLR_TYPE_AND_COLUMN_TYPE_MAP.put(BlrConstants.blr_float, FLOAT);
112         BLR_TYPE_AND_COLUMN_TYPE_MAP.put(BlrConstants.blr_d_float, D_FLOAT);
113         BLR_TYPE_AND_COLUMN_TYPE_MAP.put(BlrConstants.blr_sql_date, DATE);
114         BLR_TYPE_AND_COLUMN_TYPE_MAP.put(BlrConstants.blr_sql_time, TIME);
115         BLR_TYPE_AND_COLUMN_TYPE_MAP.put(BlrConstants.blr_timestamp, TIMESTAMP);
116         BLR_TYPE_AND_COLUMN_TYPE_MAP.put(BlrConstants.blr_quad, BLOB);
117         BLR_TYPE_AND_COLUMN_TYPE_MAP.put(BlrConstants.blr_long, LONG);
118         BLR_TYPE_AND_COLUMN_TYPE_MAP.put(BlrConstants.blr_short, SHORT);
119         BLR_TYPE_AND_COLUMN_TYPE_MAP.put(BlrConstants.blr_int64, INT64);
120         BLR_TYPE_AND_COLUMN_TYPE_MAP.put(BlrConstants.blr_bool, BOOLEAN);
121         BLR_TYPE_AND_COLUMN_TYPE_MAP.put(BlrConstants.blr_dec64, DEC16);
122         BLR_TYPE_AND_COLUMN_TYPE_MAP.put(BlrConstants.blr_dec128, DEC34);
123         BLR_TYPE_AND_COLUMN_TYPE_MAP.put(BlrConstants.blr_int128, INT128);
124         BLR_TYPE_AND_COLUMN_TYPE_MAP.put(BlrConstants.blr_timestamp_tz, TIMESTAMP_TZ);
125         BLR_TYPE_AND_COLUMN_TYPE_MAP.put(BlrConstants.blr_sql_time_tz, TIME_TZ);
126         BLR_TYPE_AND_COLUMN_TYPE_MAP.put(BlrConstants.blr_ex_timestamp_tz, TIMESTAMP_TZ_EX);
127         BLR_TYPE_AND_COLUMN_TYPE_MAP.put(BlrConstants.blr_ex_time_tz, TIME_TZ_EX);
128         
129         for (FirebirdBinaryColumnType each : values()) {
130             VALUE_AND_COLUMN_TYPE_MAP.put(each.value, each);
131         }
132     }
133     
134     FirebirdBinaryColumnType(final int value, final int length) {
135         this.value = value;
136         this.length = length;
137         subtype = 0;
138     }
139     
140     /**
141      * Value of JDBC type.
142      *
143      * @param jdbcType JDBC type
144      * @return column type enum
145      */
146     public static FirebirdBinaryColumnType valueOfJDBCType(final int jdbcType) {
147         Preconditions.checkArgument(JDBC_TYPE_AND_COLUMN_TYPE_MAP.containsKey(jdbcType), "Can not find JDBC type `%d` in column type", jdbcType);
148         return JDBC_TYPE_AND_COLUMN_TYPE_MAP.get(jdbcType);
149     }
150     
151     /**
152      * Value of BLR type.
153      *
154      * @param blrType BLR type
155      * @return column type enum
156      */
157     public static FirebirdBinaryColumnType valueOfBLRType(final int blrType) {
158         Preconditions.checkArgument(BLR_TYPE_AND_COLUMN_TYPE_MAP.containsKey(blrType), "Can not find BLR type `%d` in column type", blrType);
159         return BLR_TYPE_AND_COLUMN_TYPE_MAP.get(blrType);
160     }
161     
162     /**
163      * Value of.
164      *
165      * @param value value
166      * @return column type
167      */
168     public static FirebirdBinaryColumnType valueOf(final int value) {
169         Preconditions.checkArgument(VALUE_AND_COLUMN_TYPE_MAP.containsKey(value), "Can not find value `%d` in column type", value);
170         return VALUE_AND_COLUMN_TYPE_MAP.get(value);
171     }
