18  package org.apache.shardingsphere.test.it.rewriter.engine.mocker.type;
19  
20  import lombok.SneakyThrows;
21  import org.apache.shardingsphere.test.it.rewriter.engine.mocker.DialectStorageUnitMetaDataMocker;
22  
23  import java.sql.Connection;
24  import java.sql.DatabaseMetaData;
25  import java.sql.PreparedStatement;
26  import java.sql.ResultSet;
27  import java.sql.SQLException;
28  import java.sql.Types;
29  import java.util.LinkedHashMap;
30  import java.util.Map;
31  import java.util.Map.Entry;
32  import java.util.concurrent.atomic.AtomicReference;
33  
34  import static org.mockito.Mockito.mock;
35  import static org.mockito.Mockito.when;
36  
37  /**
38   * MySQL storage unit meta data mocker.
39   */
40  public final class MySQLStorageUnitMetaDataMocker implements DialectStorageUnitMetaDataMocker {
41      
42      @Override
43      @SneakyThrows(SQLException.class)
44      public void mockStorageUnitMetaData(final Connection connection, final DatabaseMetaData databaseMetaData) {
45          mockIdentifierCasePolicy(connection);
46          mockDataTypeInfo(databaseMetaData);
47      }
48      
49      private void mockIdentifierCasePolicy(final Connection connection) throws SQLException {
50          PreparedStatement preparedStatement = mock(PreparedStatement.class);
51          ResultSet resultSet = mock(ResultSet.class);
52          when(connection.prepareStatement("SELECT @@lower_case_table_names")).thenReturn(preparedStatement);
53          when(preparedStatement.executeQuery()).thenReturn(resultSet);
54          when(resultSet.next()).thenReturn(true);
55          when(resultSet.getInt(1)).thenReturn(0);
56      }
57      
58      private void mockDataTypeInfo(final DatabaseMetaData databaseMetaData) throws SQLException {
59          Map<String, Integer> dataTypes = new LinkedHashMap<>(30, 1F);
60          dataTypes.put("TINYINT", Types.TINYINT);
61          dataTypes.put("SMALLINT", Types.SMALLINT);
62          dataTypes.put("MEDIUMINT", Types.INTEGER);
63          dataTypes.put("INT", Types.INTEGER);
64          dataTypes.put("INTEGER", Types.INTEGER);
65          dataTypes.put("BIGINT", Types.BIGINT);
66          dataTypes.put("DECIMAL", Types.DECIMAL);
67          dataTypes.put("NUMERIC", Types.NUMERIC);
68          dataTypes.put("FLOAT", Types.FLOAT);
69          dataTypes.put("REAL", Types.REAL);
70          dataTypes.put("DOUBLE", Types.DOUBLE);
71          dataTypes.put("BIT", Types.BIT);
72          dataTypes.put("DATE", Types.DATE);
73          dataTypes.put("DATETIME", Types.TIMESTAMP);
74          dataTypes.put("TIMESTAMP", Types.TIMESTAMP);
75          dataTypes.put("TIME", Types.TIME);
76          dataTypes.put("CHAR", Types.CHAR);
77          dataTypes.put("VARCHAR", Types.VARCHAR);
78          dataTypes.put("BINARY", Types.BINARY);
79          dataTypes.put("VARBINARY", Types.VARBINARY);
80          dataTypes.put("TINYBLOB", Types.BLOB);
81          dataTypes.put("BLOB", Types.BLOB);
82          dataTypes.put("MEDIUMBLOB", Types.BLOB);
83          dataTypes.put("LONGBLOB", Types.BLOB);
84          dataTypes.put("TINYTEXT", Types.LONGVARCHAR);
85          dataTypes.put("TEXT", Types.LONGVARCHAR);
86          dataTypes.put("MEDIUMTEXT", Types.LONGVARCHAR);
87          dataTypes.put("LONGTEXT", Types.LONGVARCHAR);
88          dataTypes.put("ENUM", Types.VARCHAR);
89          dataTypes.put("SET", Types.VARCHAR);
90          ResultSet resultSet = mock(ResultSet.class);
91          AtomicReference<Entry<String, Integer>> current = new AtomicReference<>();
92          when(resultSet.next()).thenAnswer(invocation -> {
93              if (dataTypes.isEmpty()) {
94                  return false;
95              }
96              Entry<String, Integer> next = dataTypes.entrySet().iterator().next();
97              dataTypes.remove(next.getKey());
98              current.set(next);
99              return true;
100         });
101         when(resultSet.getString("TYPE_NAME")).thenAnswer(invocation -> current.get().getKey());
102         when(resultSet.getInt("DATA_TYPE")).thenAnswer(invocation -> current.get().getValue());
103         when(databaseMetaData.getTypeInfo()).thenReturn(resultSet);
104     }
105     
106     @Override
107     public String getDatabaseType() {
108         return "MySQL";
109     }
110 }
