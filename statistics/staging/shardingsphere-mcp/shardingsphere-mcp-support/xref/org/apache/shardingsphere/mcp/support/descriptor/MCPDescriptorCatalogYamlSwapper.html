50  @NoArgsConstructor(access = AccessLevel.PRIVATE)
51  final class MCPDescriptorCatalogYamlSwapper {
52      
53      static MCPDescriptorCatalog swap(final Collection<YamlMCPDescriptorCatalog> yamlCatalogs) {
54          Collection<MCPResourceDescriptor> resourceDescriptors = new LinkedList<>();
55          Collection<MCPResourceDescriptor> resourceTemplateDescriptors = new LinkedList<>();
56          Collection<ShardingSphereMCPResourceMetadata> shardingSphereResourceMetadata = new LinkedList<>();
57          Collection<MCPToolDescriptor> toolDescriptors = new LinkedList<>();
58          Collection<MCPPromptDescriptor> promptDescriptors = new LinkedList<>();
59          Collection<MCPPromptTemplateBinding> promptTemplateBindings = new LinkedList<>();
60          Collection<MCPCompletionTargetDescriptor> completionTargetDescriptors = new LinkedList<>();
61          Collection<MCPResourceNavigationDescriptor> resourceNavigationDescriptors = new LinkedList<>();
62          Collection<MCPToolRuntimeDescriptor> toolRuntimeDescriptors = new LinkedList<>();
63          for (YamlMCPDescriptorCatalog each : yamlCatalogs) {
64              MCPConfigurationValidator.validate(each, "MCP descriptor catalog");
65              swapFixedResourceDescriptors(each.getResources(), resourceDescriptors, shardingSphereResourceMetadata);
66              swapResourceTemplateDescriptors(each.getResourceTemplates(), resourceTemplateDescriptors, shardingSphereResourceMetadata);
67              toolDescriptors.addAll(swapToolDescriptors(each.getTools()));
68              swapPromptDescriptors(each.getPrompts(), promptDescriptors, promptTemplateBindings);
69              completionTargetDescriptors.addAll(swapCompletionTargetDescriptors(each.getCompletionTargets()));
70              resourceNavigationDescriptors.addAll(swapResourceNavigationDescriptors(each.getResourceNavigation()));
71              toolRuntimeDescriptors.addAll(swapToolRuntimeDescriptors(each.getTools()));
72          }
73          return new MCPDescriptorCatalog(new MCPProtocolDescriptorCatalog(resourceDescriptors, resourceTemplateDescriptors, toolDescriptors, promptDescriptors),
74                  new MCPShardingSphereDescriptorCatalog(shardingSphereResourceMetadata, promptTemplateBindings, completionTargetDescriptors, resourceNavigationDescriptors, toolRuntimeDescriptors));
75      }
76      
77      private static void swapFixedResourceDescriptors(final Collection<YamlMCPResourceDescriptor> yamlDescriptors, final Collection<MCPResourceDescriptor> resources,
78                                                       final Collection<ShardingSphereMCPResourceMetadata> resourceMetadata) {
79          for (YamlMCPResourceDescriptor each : emptyIfNull(yamlDescriptors)) {
80              swapResourceDescriptor(each.getUri(), each, resources, resourceMetadata);
81          }
82      }
83      
84      private static void swapResourceTemplateDescriptors(final Collection<YamlMCPResourceDescriptor> yamlDescriptors, final Collection<MCPResourceDescriptor> resourceTemplates,
85                                                          final Collection<ShardingSphereMCPResourceMetadata> resourceMetadata) {
86          for (YamlMCPResourceDescriptor each : emptyIfNull(yamlDescriptors)) {
87              swapResourceDescriptor(each.getUriTemplate(), each, resourceTemplates, resourceMetadata);
88          }
89      }
90      
91      private static void swapResourceDescriptor(final String uriTemplate, final YamlMCPResourceDescriptor yamlDescriptor, final Collection<MCPResourceDescriptor> resources,
92                                                 final Collection<ShardingSphereMCPResourceMetadata> resourceMetadata) {
93          ShardingSphereMCPResourceMetadata metadata = swapShardingSphereResourceMetadata(uriTemplate, yamlDescriptor.getShardingSphereMetadata());
94          resources.add(new MCPResourceDescriptor(uriTemplate, yamlDescriptor.getName(), yamlDescriptor.getTitle(), yamlDescriptor.getDescription(), yamlDescriptor.getMimeType(),
95                  swapResourceAnnotations(yamlDescriptor.getAnnotations()), createResourceMeta(uriTemplate, emptyMapIfNull(yamlDescriptor.getMeta()), metadata)));
96          resourceMetadata.add(metadata);
97      }
98      
99      private static ShardingSphereMCPResourceMetadata swapShardingSphereResourceMetadata(final String uriTemplate, final YamlShardingSphereMCPResourceMetadata yamlMetadata) {
100         if (null == yamlMetadata) {
101             return new ShardingSphereMCPResourceMetadata(uriTemplate, List.of(), null, null, null, List.of(), List.of(), List.of());
102         }
103         return new ShardingSphereMCPResourceMetadata(uriTemplate, swapUriVariables(yamlMetadata.getUriVariables()), yamlMetadata.getResourceKind(), yamlMetadata.getObjectScope(),
104                 yamlMetadata.getFeature(), emptyIfNull(yamlMetadata.getRelatedTools()), emptyIfNull(yamlMetadata.getRelatedResources()), emptyIfNull(yamlMetadata.getUseBefore()));
105     }
106     
107     private static Map<String, Object> createResourceMeta(final String uriTemplate, final Map<String, Object> rawMeta, final ShardingSphereMCPResourceMetadata metadata) {
108         validateNoTypedResourceMetadataKeys(uriTemplate, rawMeta);
109         Map<String, Object> result = new LinkedHashMap<>(rawMeta.size() + MCPShardingSphereMetadataKeys.TYPED_RESOURCE_METADATA_KEYS.size(), 1F);
110         result.putAll(rawMeta);
111         result.putAll(metadata.toMeta());
112         return result;
113     }
114     
115     private static void validateNoTypedResourceMetadataKeys(final String uriTemplate, final Map<String, Object> rawMeta) {
116         for (String each : MCPShardingSphereMetadataKeys.TYPED_RESOURCE_METADATA_KEYS) {
117             if (rawMeta.containsKey(each)) {
118                 throw new IllegalStateException(String.format("MCP resource descriptor `%s` raw meta must not define typed ShardingSphere resource metadata key `%s`.", uriTemplate, each));
119             }
120         }
121     }
122     
123     private static List<MCPUriVariableDescriptor> swapUriVariables(final Collection<YamlMCPUriVariableDescriptor> yamlUriVariables) {
124         List<MCPUriVariableDescriptor> result = new LinkedList<>();
125         for (YamlMCPUriVariableDescriptor each : emptyIfNull(yamlUriVariables)) {
126             result.add(new MCPUriVariableDescriptor(each.getName(), each.getTitle(), each.getDescription(), each.isRequired(), each.getScope()));
127         }
128         return result;
129     }
130     
131     private static MCPResourceAnnotations swapResourceAnnotations(final YamlMCPResourceAnnotations yamlAnnotations) {
132         return null == yamlAnnotations ? MCPResourceAnnotations.EMPTY : new MCPResourceAnnotations(yamlAnnotations.getAudience(), yamlAnnotations.getPriority(), yamlAnnotations.getLastModified());
133     }
134     
135     private static Collection<MCPToolDescriptor> swapToolDescriptors(final Collection<YamlMCPToolDescriptor> yamlDescriptors) {
136         Collection<MCPToolDescriptor> result = new LinkedList<>();
137         for (YamlMCPToolDescriptor each : emptyIfNull(yamlDescriptors)) {
138             result.add(new MCPToolDescriptor(each.getName(), each.getTitle(), each.getDescription(), emptyMapIfNull(each.getInputSchema()), emptyMapIfNull(each.getOutputSchema()),
139                     swapToolAnnotations(each.getAnnotations()), emptyMapIfNull(each.getMeta())));
140         }
141         return result;
142     }
143     
144     private static void swapPromptDescriptors(final Collection<YamlMCPPromptDescriptor> yamlDescriptors, final Collection<MCPPromptDescriptor> prompts,
145                                               final Collection<MCPPromptTemplateBinding> promptTemplateBindings) {
146         for (YamlMCPPromptDescriptor each : emptyIfNull(yamlDescriptors)) {
147             prompts.add(new MCPPromptDescriptor(each.getName(), each.getTitle(), each.getDescription(), swapPromptArguments(each.getArguments()), emptyMapIfNull(each.getMeta())));
148             promptTemplateBindings.add(new MCPPromptTemplateBinding(each.getName(), null == each.getBinding() ? null : each.getBinding().getTemplateResource()));
149         }
150     }
151     
152     private static List<MCPPromptArgumentDescriptor> swapPromptArguments(final Collection<YamlMCPPromptArgumentDescriptor> yamlArguments) {
153         List<MCPPromptArgumentDescriptor> result = new LinkedList<>();
154         for (YamlMCPPromptArgumentDescriptor each : emptyIfNull(yamlArguments)) {
155             result.add(new MCPPromptArgumentDescriptor(each.getName(), each.getTitle(), each.getDescription(), each.isRequired()));
156         }
157         return result;
158     }
159     
160     private static Collection<MCPCompletionTargetDescriptor> swapCompletionTargetDescriptors(final Collection<YamlMCPCompletionTargetDescriptor> yamlDescriptors) {
161         Collection<MCPCompletionTargetDescriptor> result = new LinkedList<>();
162         for (YamlMCPCompletionTargetDescriptor each : emptyIfNull(yamlDescriptors)) {
163             result.add(new MCPCompletionTargetDescriptor(each.getReferenceType(), each.getReference(), emptyIfNull(each.getArguments()), each.getMaxValues(), emptyMapIfNull(each.getMeta())));
164         }
165         return result;
166     }
167     
168     private static Collection<MCPResourceNavigationDescriptor> swapResourceNavigationDescriptors(final Collection<YamlMCPResourceNavigationDescriptor> yamlDescriptors) {
169         Collection<MCPResourceNavigationDescriptor> result = new LinkedList<>();
170         for (YamlMCPResourceNavigationDescriptor each : emptyIfNull(yamlDescriptors)) {
171             result.add(new MCPResourceNavigationDescriptor(each.getFrom(), each.getTo(), emptyIfNull(each.getRequiredArguments()), emptyIfNull(each.getCarriedArguments()), each.getDescription()));
172         }
173         return result;
174     }
175     
176     private static MCPToolAnnotations swapToolAnnotations(final YamlMCPToolAnnotations yamlAnnotations) {
177         return MCPToolAnnotations.builder()
178                 .title(yamlAnnotations.getTitle())
179                 .readOnlyHint(Boolean.TRUE.equals(yamlAnnotations.getReadOnlyHint()))
180                 .destructiveHint(Boolean.TRUE.equals(yamlAnnotations.getDestructiveHint()))
181                 .idempotentHint(Boolean.TRUE.equals(yamlAnnotations.getIdempotentHint()))
182                 .openWorldHint(Boolean.TRUE.equals(yamlAnnotations.getOpenWorldHint()))
183                 .build();
184     }
185     
186     private static Collection<MCPToolRuntimeDescriptor> swapToolRuntimeDescriptors(final Collection<YamlMCPToolDescriptor> yamlDescriptors) {
187         Collection<MCPToolRuntimeDescriptor> result = new LinkedList<>();
188         for (YamlMCPToolDescriptor each : emptyIfNull(yamlDescriptors)) {
189             if (null != each.getRuntime()) {
190                 result.add(swapToolRuntimeDescriptor(each.getName(), each.getRuntime()));
191             }
192         }
193         return result;
194     }
195     
196     private static MCPToolRuntimeDescriptor swapToolRuntimeDescriptor(final String toolName, final YamlMCPToolRuntimeDescriptor yamlRuntime) {
197         return new MCPToolRuntimeDescriptor(toolName, null == yamlRuntime.getWorkflowRole() ? "" : yamlRuntime.getWorkflowRole());
198     }
199     
200     private static <T> Collection<T> emptyIfNull(final Collection<T> values) {
201         return null == values ? Collections.emptyList() : values;
202     }
203     
204     private static Map<String, Object> emptyMapIfNull(final Map<String, Object> values) {
205         return null == values ? Collections.emptyMap() : values;
206     }
