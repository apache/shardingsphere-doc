99                               final Class<? extends ShardingSphereRule> ruleClass, final Class<?> expectedType, final Class<? extends Throwable> expectedExceptionClass) {
100         if (cached) {
101             assertTrue(ruleMetaData.findSingleRule(RuleMetaDataShardingSphereRuleFixture.class).isPresent());

104             assertThat(ruleMetaData.getSingleRule(ruleClass), isA(expectedType));
105         } else {
106             assertThat(assertThrows(expectedExceptionClass, () -> ruleMetaData.getSingleRule(ruleClass)), isA(expectedExceptionClass));
107         }

169     @Test
170     void assertCopyAndPutDataNode() {
171         MutableDataNodeRuleAttribute attribute = mock(MutableDataNodeRuleAttribute.class);
172         ShardingSphereRule mutableRule = mock(ShardingSphereRule.class);
173         ShardingSphereRule copiedRule = mock(ShardingSphereRule.class);
174         when(mutableRule.getAttributes()).thenReturn(new RuleAttributes(attribute));
175         when(attribute.copyRuleAndPut("foo_ds", "foo_schema", "foo_tbl")).thenReturn(copiedRule);
176         RuleMetaData original = new RuleMetaData(Arrays.asList(fixtureRule, mutableRule, dataNodeRule));
177         RuleMetaData actual = original.copyAndPutDataNode("foo_ds", "foo_schema", "foo_tbl");
178         assertThat(actual, not(sameInstance(original)));
179         assertThat(actual.getRules(), is(Arrays.asList(fixtureRule, copiedRule, dataNodeRule)));
180         assertThat(original.getRules(), is(Arrays.asList(fixtureRule, mutableRule, dataNodeRule)));
181     }

183     @Test
184     void assertCopyAndPutDataNodeWhenDataNodeExists() {
185         MutableDataNodeRuleAttribute attribute = mock(MutableDataNodeRuleAttribute.class);
186         ShardingSphereRule mutableRule = mock(ShardingSphereRule.class);
187         when(mutableRule.getAttributes()).thenReturn(new RuleAttributes(attribute));

189         RuleMetaData original = new RuleMetaData(Collections.singleton(mutableRule));

193     @Test
194     void assertCopyAndRemoveDataNode() {
195         MutableDataNodeRuleAttribute attribute = mock(MutableDataNodeRuleAttribute.class);
196         ShardingSphereRule mutableRule = mock(ShardingSphereRule.class);
197         ShardingSphereRule copiedRule = mock(ShardingSphereRule.class);
198         when(mutableRule.getAttributes()).thenReturn(new RuleAttributes(attribute));
199         when(attribute.findTableDataNode("foo_schema", "foo_tbl")).thenReturn(Optional.of(new DataNode("foo_ds.foo_schema.foo_tbl")));
200         when(attribute.copyRuleAndRemove("foo_schema", "foo_tbl")).thenReturn(copiedRule);
201         RuleMetaData original = new RuleMetaData(Arrays.asList(fixtureRule, mutableRule, dataNodeRule));
202         RuleMetaData actual = original.copyAndRemoveDataNode("foo_schema", "foo_tbl");
203         assertThat(actual, not(sameInstance(original)));
204         assertThat(actual.getRules(), is(Arrays.asList(fixtureRule, copiedRule, dataNodeRule)));
205         assertThat(original.getRules(), is(Arrays.asList(fixtureRule, mutableRule, dataNodeRule)));
206     }

266     private ShardingSphereRule mockDataSourceMapperRule(final String storageUnitName) {
267         ShardingSphereRule result = mock(ShardingSphereRule.class, RETURNS_DEEP_STUBS);
268         DataSourceMapperRuleAttribute ruleAttribute = mock(DataSourceMapperRuleAttribute.class, RETURNS_DEEP_STUBS);
269         when(ruleAttribute.getDataSourceMapper().values()).thenReturn(Collections.singletonList(Collections.singletonList(storageUnitName)));
270         when(result.getAttributes().findAttribute(DataSourceMapperRuleAttribute.class)).thenReturn(Optional.of(ruleAttribute));
271         return result;
272     }

274     private ShardingSphereRule mockDataNodeRule() {
275         ShardingSphereRule result = mock(ShardingSphereRule.class, RETURNS_DEEP_STUBS);
276         DataNodeRuleAttribute ruleAttribute = mock(DataNodeRuleAttribute.class, RETURNS_DEEP_STUBS);
277         when(ruleAttribute.getAllDataNodes().values()).thenReturn(Collections.singleton(Collections.singleton(new DataNode("foo_db.foo_tbl"))));
278         when(result.getAttributes().findAttribute(DataNodeRuleAttribute.class)).thenReturn(Optional.of(ruleAttribute));
279         return result;
