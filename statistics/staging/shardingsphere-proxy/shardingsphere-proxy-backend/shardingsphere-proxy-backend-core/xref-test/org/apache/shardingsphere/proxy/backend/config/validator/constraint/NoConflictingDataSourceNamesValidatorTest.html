55  class NoConflictingDataSourceNamesValidatorTest {
56      
57      private static final Validator VALIDATOR = Validation.byProvider(ApacheValidationProvider.class).configure().buildValidatorFactory().getValidator();
58      
59      private final NoConflictingDataSourceNamesValidator validator = new NoConflictingDataSourceNamesValidator();
60      
61      @ParameterizedTest(name = "{0}")
62      @MethodSource("validConfigsArguments")
63      void assertIsValidWithoutConflicts(final String name, final ProxyConfigurationLoadResult config) {
64          ConstraintValidatorContext context = mock(ConstraintValidatorContext.class);
65          assertTrue(validator.isValid(config, context));
66          verifyNoInteractions(context);
67      }
68      
69      private static Stream<Arguments> validConfigsArguments() {
70          Map<String, Collection<String>> sharedLocalNames = new LinkedHashMap<>(2, 1F);
71          sharedLocalNames.put("foo_db", Collections.singleton("foo_ds"));
72          sharedLocalNames.put("bar_db", Collections.singleton("foo_ds"));
73          return Stream.of(
74                  Arguments.of("No databases", createConfig(Collections.singleton("foo_ds"), Collections.emptyMap())),
75                  Arguments.of("No global sources", createConfig(Collections.emptyList(), Collections.singletonMap("foo_db", Collections.singleton("foo_ds")))),
76                  Arguments.of("No database sources", createConfig(Collections.singleton("foo_ds"), Collections.singletonMap("foo_db", Collections.emptyList()))),
77                  Arguments.of("Distinct names", createConfig(Collections.singleton("foo_ds"), Collections.singletonMap("foo_db", Collections.singleton("bar_ds")))),
78                  Arguments.of("Case-sensitive names", createConfig(Collections.singleton("foo_ds"), Collections.singletonMap("foo_db", Collections.singleton("FOO_DS")))),
79                  Arguments.of("Same local names across databases", createConfig(Collections.singleton("global_ds"), sharedLocalNames)));
80      }
81      
82      private static ProxyConfigurationLoadResult createConfig(final Collection<String> globalDataSourceNames, final Map<String, Collection<String>> databaseDataSourceNames) {
83          Map<String, ProxyDatabaseConfiguration> databaseConfigs = new LinkedHashMap<>(databaseDataSourceNames.size(), 1F);
84          for (Entry<String, Collection<String>> entry : databaseDataSourceNames.entrySet()) {
85              databaseConfigs.put(entry.getKey(), new ProxyDatabaseConfiguration(entry.getKey(), createDataSources(entry.getValue()), Collections.emptyList()));
86          }
87          return new ProxyConfigurationLoadResult(null, createDataSources(globalDataSourceNames), Collections.emptyList(), new Properties(), databaseConfigs);
88      }
89      
90      private static Map<String, DataSourceConfiguration> createDataSources(final Collection<String> dataSourceNames) {
91          Map<String, DataSourceConfiguration> result = new HashMap<>(dataSourceNames.size(), 1F);
92          for (String each : dataSourceNames) {
93              result.put(each, mock(DataSourceConfiguration.class));
94          }
95          return result;
96      }
97      
98      @ParameterizedTest(name = "{0}")
99      @MethodSource("conflictingConfigsArguments")
100     void assertIsValidWithConflicts(final String name, final ProxyConfigurationLoadResult config, final String expectedDatabaseKey, final Collection<String> expectedDataSourceNames) {
101         Collection<ConstraintViolation<ProxyConfigurationLoadResult>> actual = VALIDATOR.validate(config);
102         assertThat(actual.size(), is(expectedDataSourceNames.size()));
103         Collection<String> actualDataSourceNames = new ArrayList<>(actual.size());
104         for (ConstraintViolation<ProxyConfigurationLoadResult> each : actual) {
105             List<Path.Node> actualPath = StreamSupport.stream(each.getPropertyPath().spliterator(), false).collect(Collectors.toList());
106             assertThat(actualPath.size(), is(3));
107             assertThat(actualPath.get(0).getName(), is("databaseConfigurations"));
108             assertThat(actualPath.get(1).getName(), is("dataSources"));
109             assertThat(actualPath.get(1).getKey(), is(expectedDatabaseKey));
110             assertThat(actualPath.get(2).getKind(), is(ElementKind.BEAN));
111             actualDataSourceNames.add((String) actualPath.get(2).getKey());
112             assertThat(each.getMessage(), is("must not contain data source names shared by global and database configurations"));
113         }
114         assertThat(actualDataSourceNames, containsInAnyOrder(expectedDataSourceNames.toArray(new String[0])));
115     }
116     
117     private static Stream<Arguments> conflictingConfigsArguments() {
118         Map<String, Collection<String>> multipleDatabases = new LinkedHashMap<>(3, 1F);
119         multipleDatabases.put("foo_db", Collections.singleton("other_ds"));
120         multipleDatabases.put("bar_db", Arrays.asList("foo_ds", "bar_ds"));
121         multipleDatabases.put("baz_db", Collections.singleton("baz_ds"));
122         String specialDataSourceName = "${validatedValue}[foo].{bar}";
123         return Stream.of(
124                 Arguments.of("One conflicting source", createConfig(Collections.singleton("foo_ds"), Collections.singletonMap("foo_db", Collections.singleton("foo_ds"))),
125                         "foo_db", Collections.singleton("foo_ds")),
126                 Arguments.of("First conflicting database and all its sources", createConfig(Arrays.asList("foo_ds", "bar_ds", "baz_ds"), multipleDatabases), "bar_db",
127                         Arrays.asList("foo_ds", "bar_ds")),
128                 Arguments.of("Special source key", createConfig(Collections.singleton(specialDataSourceName), Collections.singletonMap("foo_db", Collections.singleton(specialDataSourceName))),
129                         "foo_db", Collections.singleton(specialDataSourceName)));
130     }
