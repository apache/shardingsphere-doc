18  package org.apache.shardingsphere.db.protocol.firebird.packet.command.query.info.type.sql;
19  
20  import lombok.Getter;
21  import lombok.RequiredArgsConstructor;
22  import org.apache.shardingsphere.db.protocol.firebird.exception.FirebirdProtocolException;
23  import org.apache.shardingsphere.db.protocol.firebird.packet.FirebirdPacket;
24  import org.apache.shardingsphere.db.protocol.firebird.packet.command.query.info.FirebirdInfoPacketType;
25  import org.apache.shardingsphere.db.protocol.firebird.packet.command.query.info.type.common.FirebirdCommonInfoPacketType;
26  import org.apache.shardingsphere.db.protocol.firebird.payload.FirebirdPacketPayload;
27  
28  import java.util.List;
29  
30  /**
31   * SQL info return data packet for Firebird.
32   */
33  @RequiredArgsConstructor
34  @Getter
35  public final class FirebirdSQLInfoReturnPacket extends FirebirdPacket {
36      
37      private final List<FirebirdInfoPacketType> infoItems;
38      
39      @Override
40      protected void write(final FirebirdPacketPayload payload) {
41          for (FirebirdInfoPacketType type : infoItems) {
42              if (type.isCommon()) {
43                  FirebirdCommonInfoPacketType.parseCommonInfo(payload, (FirebirdCommonInfoPacketType) type);
44              } else {
45                  parseSQLInfo(payload, (FirebirdSQLInfoPacketType) type);
46              }
47          }
48      }
49      
50      private void parseSQLInfo(final FirebirdPacketPayload payload, final FirebirdSQLInfoPacketType type) {
51          // TODO implement other request types handle
52          switch (type) {
53              case RECORDS:
54                  // TODO handle actual update count
55                  processRecords(payload);
56                  return;
57              default:
58                  throw new FirebirdProtocolException("Unknown database information request type %d", type.getCode());
59          }
60      }
61      
62      private void processRecords(final FirebirdPacketPayload payload) {
63          // TODO handle actual update count
64          payload.writeInt1(FirebirdSQLInfoPacketType.RECORDS.getCode());
65          payload.writeInt2LE(0);
66          payload.writeInt1(FirebirdSQLInfoReturnValue.SELECT.getCode());
67          payload.writeInt2LE(4);
68          payload.writeInt4LE(0);
69          payload.writeInt1(FirebirdSQLInfoReturnValue.INSERT.getCode());
70          payload.writeInt2LE(4);
71          payload.writeInt4LE(0);
72          payload.writeInt1(FirebirdSQLInfoReturnValue.UPDATE.getCode());
73          payload.writeInt2LE(4);
74          payload.writeInt4LE(0);
75          payload.writeInt1(FirebirdSQLInfoReturnValue.DELETE.getCode());
76          payload.writeInt2LE(4);
77          payload.writeInt4LE(0);
78      }
79  }
