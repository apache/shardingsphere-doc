47  class MethodAdvisorBuilderInterceptorTest {
48      
49      @AfterEach
50      void resetCache() throws ReflectiveOperationException {
51          ((Map<?, ?>) Plugins.getMemberAccessor().get(AdviceFactory.class.getDeclaredField("CACHED_ADVICES"), null)).clear();
52          ((Map<?, ?>) Plugins.getMemberAccessor().get(ClassLoaderContext.class.getDeclaredField("AGENT_CLASS_LOADERS"), null)).clear();
53      }
54      
55      @Test
56      void assertInterceptWhenNoMatchedAdvice() throws ReflectiveOperationException {
57          AdvisorConfiguration advisorConfig = new AdvisorConfiguration(TargetObjectFixture.class.getName());
58          TypeDescription typeDescription = mock(TypeDescription.class);
59          when(typeDescription.getDeclaredMethods()).thenReturn(new Explicit<>(
60                  Collections.singletonList(new MethodDescription.ForLoadedMethod(TargetObjectFixture.class.getMethod("call", List.class)))));
61          MethodAdvisorBuilderInterceptor interceptor = new MethodAdvisorBuilderInterceptor(
62                  typeDescription, new ClassLoaderContext(new URLClassLoader(new URL[0], getClass().getClassLoader()), Collections.emptyList()), advisorConfig);
63          Builder<?> builder = mock(Builder.class);
64          assertThat(interceptor.intercept(builder), is(builder));
65      }
66      
67      @SuppressWarnings({"unchecked", "rawtypes"})
68      @Test
69      void assertInterceptWhenAdviceMatched() throws ReflectiveOperationException {
70          TypeDescription typeDescription = mock(TypeDescription.class);
71          MethodDescription.InDefinedShape methodDescription = mock(MethodDescription.InDefinedShape.class);
72          when(typeDescription.getDeclaredMethods()).thenReturn(new Explicit<>(Collections.singletonList(methodDescription)));
73          MethodAdvisorBuilderInterceptor interceptor = new MethodAdvisorBuilderInterceptor(
74                  typeDescription, new ClassLoaderContext(new URLClassLoader(new URL[0], getClass().getClassLoader()), Collections.emptyList()), new AdvisorConfiguration("ignored"));
75          AdviceExecutorFactory factory = mock(AdviceExecutorFactory.class);
76          AdviceExecutor adviceExecutor = mock(AdviceExecutor.class);
77          Builder intercepted = mock(Builder.class);
78          when(factory.findMatchedAdviceExecutor(methodDescription)).thenReturn(Optional.of(adviceExecutor));
79          when(adviceExecutor.intercept(any(Builder.class), any(MethodDescription.class))).thenReturn(intercepted);
80          Plugins.getMemberAccessor().set(MethodAdvisorBuilderInterceptor.class.getDeclaredField("adviceExecutorFactory"), interceptor, factory);
81          Builder<?> builder = mock(Builder.class);
82          assertThat(interceptor.intercept(builder), is(intercepted));
83      }
84      
85      @Test
86      void assertInterceptWhenAdviceInterceptThrows() throws ReflectiveOperationException {
87          TypeDescription typeDescription = mock(TypeDescription.class);
88          MethodDescription.InDefinedShape methodDescription = mock(MethodDescription.InDefinedShape.class);
89          when(typeDescription.getDeclaredMethods()).thenReturn(new Explicit<>(Collections.singletonList(methodDescription)));
90          MethodAdvisorBuilderInterceptor interceptor = new MethodAdvisorBuilderInterceptor(
91                  typeDescription, new ClassLoaderContext(new URLClassLoader(new URL[0], getClass().getClassLoader()), Collections.emptyList()), new AdvisorConfiguration("ignored"));
92          AdviceExecutorFactory factory = mock(AdviceExecutorFactory.class);
93          AdviceExecutor adviceExecutor = mock(AdviceExecutor.class);
94          when(factory.findMatchedAdviceExecutor(methodDescription)).thenReturn(Optional.of(adviceExecutor));
95          when(adviceExecutor.intercept(any(Builder.class), any(MethodDescription.class))).thenThrow(new IllegalStateException("error"));
96          Plugins.getMemberAccessor().set(MethodAdvisorBuilderInterceptor.class.getDeclaredField("adviceExecutorFactory"), interceptor, factory);
97          Builder<?> builder = mock(Builder.class);
98          assertThat(interceptor.intercept(builder), is(builder));
99      }
