44      static Map<String, Object> createSQLToolMismatchRecovery(final SQLToolMismatchException cause) {
45          Map<String, Object> result = MCPRecoveryPayloadSupport.createBaseRecovery(createSQLToolMismatchCategory(cause),
46                  CoreToolNames.EXECUTE_UPDATE.equals(cause.getTargetTool())
47                          ? "Use database_gateway_execute_update in preview mode, then execute only when the requested side effect is still intended."
48                          : "Use database_gateway_execute_query for this read-only SQL.");
49          result.put("source_tool", cause.getSourceTool());
50          result.put("statement_class", cause.getClassificationResult().getStatementClass().name().toLowerCase(Locale.ENGLISH));
51          result.put("statement_type", cause.getClassificationResult().getStatementType());
52          result.put("normalized_sql", cause.getClassificationResult().getNormalizedSql());
53          cause.getClassificationResult().getTargetObjectName().ifPresent(optional -> result.put("target_object", optional));
54          cause.getClassificationResult().getSavepointName().ifPresent(optional -> result.put("savepoint", optional));
55          result.put("suggested_arguments", cause.getSuggestedArguments());
56          result.put(MCPPayloadFieldNames.NEXT_ACTIONS, List.of(MCPNextActionUtils.callTool(cause.getTargetTool(), createSQLToolMismatchActionReason(cause), cause.getSuggestedArguments())));
57          return result;
58      }

60      static Map<String, Object> createExplainSQLSyntaxRecovery(final ExplainSQLSyntaxException cause) {
61          Map<String, Object> result = MCPRecoveryPayloadSupport.createBaseRecovery("invalid_explain_sql",
62                  "Regenerate explain_sql for the same sql using database-native EXPLAIN syntax, then retry database_gateway_execute_explain_query. Do not use EXPLAIN ANALYZE.");
63          result.put("database", cause.getDatabase());
64          result.put("schema", cause.getSchema());
65          result.put("sql", cause.getSql());
66          result.put("rejected_explain_sql", cause.getExplainSql());
67          result.put("suggested_arguments", createExplainRetryArguments(cause));
68          result.put(MCPPayloadFieldNames.NEXT_ACTIONS, MCPNextActionUtils.ordered(
69                  MCPNextActionUtils.readResource(createDatabaseCapabilityUri(cause.getDatabase()),
70                          "Read the target database type before regenerating database-native explain_sql."),
71                  MCPNextActionUtils.dependsOn(MCPNextActionUtils.callTool(CoreToolNames.EXECUTE_EXPLAIN_QUERY,
72                          "Regenerate explain_sql from sql without changing sql, then retry the explain tool with the generated explain_sql.",
73                          createExplainRetryArguments(cause)), 1)));
74          return result;
75      }
76      
77      private static Map<String, Object> createExplainRetryArguments(final ExplainSQLSyntaxException cause) {
78          Map<String, Object> result = new LinkedHashMap<>(4, 1F);
79          result.put("database", cause.getDatabase());
80          if (!cause.getSchema().isEmpty()) {
81              result.put("schema", cause.getSchema());
82          }
83          result.put("sql", cause.getSql());
84          return result;
85      }

87      static Map<String, Object> createRuleDistSQLExecutionRecovery(final RuleDistSQLExecutionException cause) {
88          Map<String, Object> result = MCPRecoveryPayloadSupport.createBaseRecovery("rule_distsql_execution_failed",
89                  "Do not ask the user to rewrite the SQL yet; read workflow guidance and verify the runtime database can execute ShardingSphere rule DistSQL.");
90          ClassificationResult classificationResult = cause.getClassificationResult();
91          result.put("database", cause.getDatabase());
92          result.put("statement_class", classificationResult.getStatementClass().name().toLowerCase(Locale.ENGLISH));
93          result.put("statement_type", classificationResult.getStatementType());
94          result.put("side_effect_scope", List.of(classificationResult.getSideEffectScope()));
95          result.put("secret_safe", true);
96          result.put(MCPPayloadFieldNames.RESOURCES_TO_READ, createRuleDistSQLExecutionResources(cause.getDatabase()));
97          result.put(MCPPayloadFieldNames.NEXT_ACTIONS, createRuleDistSQLExecutionNextActions(cause.getDatabase()));
98          return result;
99      }

101     private static List<Map<String, Object>> createRuleDistSQLExecutionResources(final String database) {
102         return List.of(
103                 MCPResourceHintUtils.create("shardingsphere://guidance", "guidance", "read_first",
104                         "Read workflow guidance before retrying rule DistSQL execution.", MCPPayloadFieldNames.RESOURCES_TO_READ),
105                 createDatabaseCapabilityResourceHint(database));
106     }
107     
108     private static Map<String, Object> createDatabaseCapabilityResourceHint(final String database) {
109         return MCPResourceHintUtils.create(createDatabaseCapabilityUri(database), database.isEmpty() ? "logical-database" : "logical-database-capability", "read_first",
110                 createDatabaseCapabilityReason(database), MCPPayloadFieldNames.RESOURCES_TO_READ);
111     }
112     
113     private static String createDatabaseCapabilityUri(final String database) {
114         return database.isEmpty() ? "shardingsphere://databases" : String.format("shardingsphere://databases/%s/capabilities", MCPUriPathSegmentUtils.encodePathSegment(database));
115     }
116     
117     private static String createDatabaseCapabilityReason(final String database) {
118         return database.isEmpty()
119                 ? "Choose a configured logical database before retrying rule DistSQL execution."
120                 : "Verify the runtime database capabilities before retrying rule DistSQL execution.";
121     }
122     
123     private static List<Map<String, Object>> createRuleDistSQLExecutionNextActions(final String database) {
124         return MCPNextActionUtils.ordered(
125                 MCPNextActionUtils.readResource("shardingsphere://guidance", "Read workflow guidance and choose the matching database_gateway_plan_* workflow tool for rule changes."),
126                 MCPNextActionUtils.dependsOn(MCPNextActionUtils.readResource(createDatabaseCapabilityUri(database), createDatabaseCapabilityReason(database)), 1));
127     }

129     static Map<String, Object> createMetadataIntrospectionSQLRecovery(final MetadataIntrospectionSQLStatementException cause) {
130         Map<String, Object> result = MCPRecoveryPayloadSupport.createBaseRecovery(
131                 "metadata_introspection_sql", "Use logical metadata resources or database_gateway_search_metadata instead of console-style metadata SQL.");
132         result.put("statement_type", cause.getStatementType());
133         List<Map<String, Object>> resourcesToRead = createMetadataIntrospectionResources(cause.getStatementType());
134         result.put(MCPPayloadFieldNames.RESOURCES_TO_READ, resourcesToRead);
135         Map<String, Object> suggestedArguments = createMetadataSearchArguments(cause.getStatementType());
136         result.put("suggested_arguments", suggestedArguments);
137         result.put(MCPPayloadFieldNames.NEXT_ACTIONS, createMetadataIntrospectionNextActions(resourcesToRead, suggestedArguments));
138         return result;
139     }
140     
141     private static List<Map<String, Object>> createMetadataIntrospectionResources(final String statementType) {
142         switch (statementType) {
143             case "SHOW STORAGE UNITS":
144                 return List.of(createResourceHint("shardingsphere://databases/{database}/storage-units", "storage-unit",
145                         "Read storage units from the target logical database instead of executing SHOW STORAGE UNITS."));
146             case "SHOW RULES USED STORAGE UNIT":
147                 return List.of(
148                         createResourceHint("shardingsphere://databases/{database}/storage-units", "storage-unit",
149                                 "Choose the storage unit from the target logical database."),
150                         createResourceHint("shardingsphere://databases/{database}/storage-units/{storageUnit}/used-by-rules", "storage-unit",
151                                 "Read rules that use the chosen storage unit."));
152             case "SHOW SINGLE TABLES", "SHOW SINGLE TABLE":
153                 return List.of(createResourceHint("shardingsphere://databases/{database}/single-tables", "single-table",
154                         "Read single table mappings from the target logical database instead of executing SHOW SINGLE TABLE."));
155             case "SHOW DEFAULT SINGLE TABLE STORAGE UNIT":
156                 return List.of(createResourceHint("shardingsphere://databases/{database}/single-table/default-storage-unit", "single-table",
157                         "Read the default single table storage unit resource instead of executing SHOW DEFAULT SINGLE TABLE STORAGE UNIT."));
158             default:
159                 return MCPRecoveryPayloadSupport.createResourceHintList(
160                         "shardingsphere://databases", "logical-database", "Read logical databases before choosing a metadata scope.");
161         }
162     }
163     
164     private static Map<String, Object> createMetadataSearchArguments(final String statementType) {
165         return statementType.startsWith("SHOW STORAGE") || statementType.startsWith("SHOW RULES USED STORAGE UNIT")
166                 ? Map.of("object_types", List.of("storage_unit"))
167                 : Map.of("object_types", List.of("database"));
168     }
169     
170     private static List<Map<String, Object>> createMetadataIntrospectionNextActions(final List<Map<String, Object>> resourcesToRead,
171                                                                                     final Map<String, Object> suggestedArguments) {
172         Map<String, Object> resource = resourcesToRead.getFirst();
173         Map<String, Object> readResource = MCPNextActionUtils.readResource(
174                 String.valueOf(resource.get(MCPPayloadFieldNames.URI)), String.valueOf(resource.get(MCPPayloadFieldNames.REASON)));
175         if (List.of("storage_unit").equals(suggestedArguments.get("object_types"))) {
176             return MCPNextActionUtils.ordered(readResource,
177                     MCPNextActionUtils.dependsOn(MCPNextActionUtils.callTool(CoreToolNames.SEARCH_METADATA,
178                             "Search storage units with an explicit database or query scope instead of executing metadata SQL.", suggestedArguments), 1));
179         }
180         return MCPNextActionUtils.ordered(readResource,
181                 MCPNextActionUtils.dependsOn(MCPNextActionUtils.callTool(CoreToolNames.SEARCH_METADATA,
182                         "Search metadata with an explicit database, schema, query, or object_types scope instead of executing metadata SQL.", suggestedArguments), 1));
183     }
184     
185     private static Map<String, Object> createResourceHint(final String uri, final String resourceKind, final String reason) {
186         return MCPResourceHintUtils.create(uri, resourceKind, "read_first", reason, MCPPayloadFieldNames.RESOURCES_TO_READ);
187     }

189     static Map<String, Object> createMultipleStatementsRecovery() {
190         Map<String, Object> result = MCPRecoveryPayloadSupport.createBaseRecovery(
191                 "multiple_sql_statements", "Split the user intent into separate MCP calls and handle each statement independently.");
192         result.put("suggested_arguments", Map.of(MCPPayloadFieldNames.EXECUTION_MODE, "preview"));
193         result.put(MCPPayloadFieldNames.NEXT_ACTIONS, List.of(MCPNextActionUtils.askUser(
194                 "Ask the user which single statement should be handled first.", List.of("single_sql_statement"))));
195         return result;
196     }

198     static Map<String, Object> createUnsupportedStatementRecovery() {
199         Map<String, Object> result = MCPRecoveryPayloadSupport.createBaseRecovery(
200                 "unsupported_sql_statement", "Ask the user for a supported SELECT, EXPLAIN, DML, DDL, DCL, transaction, or savepoint statement.");
201         result.put(MCPPayloadFieldNames.RESOURCES_TO_READ, MCPRecoveryPayloadSupport.createResourceHintList(
202                 "shardingsphere://capabilities", "capability", "Read supported SQL statement classes before retrying."));
203         result.put(MCPPayloadFieldNames.NEXT_ACTIONS, List.of(MCPNextActionUtils.readResource("shardingsphere://capabilities", "Read supported statement classes before retrying.")));
204         return result;
205     }
206     
207     static Map<String, Object> createBannedStatementRecovery() {
208         Map<String, Object> result = MCPRecoveryPayloadSupport.createBaseRecovery(
209                 "banned_sql_statement", "Do not execute this SQL through MCP; ask the user for a safer supported operation.");
210         result.put(MCPPayloadFieldNames.RESOURCES_TO_READ, MCPRecoveryPayloadSupport.createResourceHintList(
211                 "shardingsphere://capabilities", "capability", "Read supported safe alternatives before asking the user."));
212         result.put(MCPPayloadFieldNames.NEXT_ACTIONS, List.of(MCPNextActionUtils.askUser(
213                 "Ask for a safer supported operation instead of executing the banned SQL.", List.of("safe_sql_or_metadata_request"))));
214         return result;
215     }
216     
217     private static String createSQLToolMismatchCategory(final SQLToolMismatchException cause) {
218         return CoreToolNames.EXECUTE_UPDATE.equals(cause.getTargetTool()) ? "unsafe_sql_attempted" : "read_only_sql_sent_to_update_tool";
219     }
220     
221     private static String createSQLToolMismatchActionReason(final SQLToolMismatchException cause) {
222         return CoreToolNames.EXECUTE_UPDATE.equals(cause.getTargetTool())
223                 ? "Retry side-effecting SQL in preview mode with the normalized SQL and preserved context."
224                 : "Retry the read-only SQL with database_gateway_execute_query using the normalized SQL and preserved context.";
225     }
