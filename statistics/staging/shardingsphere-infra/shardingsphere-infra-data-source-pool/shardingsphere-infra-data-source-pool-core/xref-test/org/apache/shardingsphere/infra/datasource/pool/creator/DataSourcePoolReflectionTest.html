59      @Test
60      void assertConvertToProperties() {
61          DataSourcePoolReflectionFixture dataSource = new DataSourcePoolReflectionFixture();
62          Properties jdbcUrlProperties = PropertiesBuilder.build(new Property("socketTimeout", "30"));
63          dataSource.setStringValue("foo_value");
64          dataSource.setEnabled(true);
65          dataSource.setListValue(Collections.singletonList("init_sql"));
66          dataSource.setJdbcUrlProperties(jdbcUrlProperties);
67          Map<String, Object> actual = new DataSourcePoolReflection(dataSource).convertToProperties();
68          assertThat(actual.size(), is(4));
69          assertThat(actual.get("stringValue"), is("foo_value"));
70          assertTrue((Boolean) actual.get("enabled"));
71          assertThat(actual.get("listValue"), is(Collections.singletonList("init_sql")));
72          assertThat(actual.get("jdbcUrlProperties"), is(jdbcUrlProperties));
73          assertFalse(actual.containsKey("loginTimeout"));
74          assertFalse(actual.containsKey("nullValue"));
75          assertFalse(actual.containsKey("objectValue"));
76          assertFalse(actual.containsKey("ignoredValue"));
77      }

100     @ParameterizedTest(name = "{0}")
101     @MethodSource("addDefaultDataSourcePoolPropertiesWithEarlyReturn")
102     void assertAddDefaultDataSourcePoolPropertiesWithEarlyReturn(final String name, final boolean hasSpiMetaData,
103                                                                  final String jdbcUrl, final Properties jdbcUrlProperties,
104                                                                  final Properties expectedJdbcUrlProperties) {
105         DataSourcePoolReflectionFixture dataSource = new DataSourcePoolReflectionFixture();
106         DataSourcePoolMetaData metaData = mockDataSourcePoolMetaData();
107         dataSource.setUrl(jdbcUrl);
108         dataSource.setJdbcUrlProperties(jdbcUrlProperties);
109         try (MockedStatic<TypedSPILoader> typedSPILoader = mockStatic(TypedSPILoader.class)) {
110             typedSPILoader.when(() -> TypedSPILoader.findService(DataSourcePoolMetaData.class, DataSourcePoolReflectionFixture.class.getName()))
111                     .thenReturn(hasSpiMetaData ? Optional.of(metaData) : Optional.empty());
112             new DataSourcePoolReflection(dataSource).addDefaultDataSourcePoolProperties(metaData);
113         }
114         Properties actualJdbcUrlProperties = dataSource.getJdbcUrlProperties();
115         if (null == expectedJdbcUrlProperties) {
116             assertNull(actualJdbcUrlProperties);
117         } else {
118             assertThat(actualJdbcUrlProperties, is(expectedJdbcUrlProperties));
119         }
120     }

122     @ParameterizedTest(name = "{0}")
123     @MethodSource("addDefaultDataSourcePoolPropertiesWithDefaultQueryProperties")
124     void assertAddDefaultDataSourcePoolPropertiesWithDefaultQueryProperties(final String name, final Properties jdbcUrlProperties,
125                                                                             final Properties queryProperties, final Properties defaultQueryProps, final Properties expectedJdbcUrlProps) {
126         DataSourcePoolReflectionFixture dataSource = new DataSourcePoolReflectionFixture();
127         ConnectionProperties connectionProperties = mock(ConnectionProperties.class);
128         dataSource.setUrl("jdbc:mock://127.0.0.1/foo_ds");
129         dataSource.setJdbcUrlProperties(jdbcUrlProperties);
130         when(connectionProperties.getQueryProperties()).thenReturn(queryProperties);
131         ConnectionPropertiesParser connectionPropertiesParser = mock(ConnectionPropertiesParser.class);
132         when(connectionPropertiesParser.parse("jdbc:mock://127.0.0.1/foo_ds", null, null)).thenReturn(connectionProperties);
133         DialectDefaultQueryPropertiesProvider defaultQueryPropsProvider = mock(DialectDefaultQueryPropertiesProvider.class);
134         DataSourcePoolMetaData metaData = mockDataSourcePoolMetaData();
135         DatabaseType databaseType = mock(DatabaseType.class);
136         try (
137                 MockedStatic<TypedSPILoader> typedSPILoader = mockStatic(TypedSPILoader.class);
138                 MockedStatic<DatabaseTypedSPILoader> databaseTypedSPILoader = mockStatic(DatabaseTypedSPILoader.class);
139                 MockedStatic<DatabaseTypeFactory> databaseTypeFactory = mockStatic(DatabaseTypeFactory.class)) {
140             typedSPILoader.when(() -> TypedSPILoader.findService(DataSourcePoolMetaData.class, DataSourcePoolReflectionFixture.class.getName())).thenReturn(Optional.of(metaData));
141             databaseTypeFactory.when(() -> DatabaseTypeFactory.get("jdbc:mock://127.0.0.1/foo_ds")).thenReturn(databaseType);
142             databaseTypedSPILoader.when(() -> DatabaseTypedSPILoader.getService(ConnectionPropertiesParser.class, databaseType)).thenReturn(connectionPropertiesParser);
143             databaseTypedSPILoader.when(() -> DatabaseTypedSPILoader.findService(DialectDefaultQueryPropertiesProvider.class, databaseType))
144                     .thenReturn(null == defaultQueryProps ? Optional.empty() : Optional.of(defaultQueryPropsProvider));
145             if (null != defaultQueryProps) {
146                 when(defaultQueryPropsProvider.getDefaultQueryProperties()).thenReturn(defaultQueryProps);
147             }
148             new DataSourcePoolReflection(dataSource).addDefaultDataSourcePoolProperties(metaData);
149         }
150         assertThat(dataSource.getJdbcUrlProperties(), is(expectedJdbcUrlProps));
151     }
