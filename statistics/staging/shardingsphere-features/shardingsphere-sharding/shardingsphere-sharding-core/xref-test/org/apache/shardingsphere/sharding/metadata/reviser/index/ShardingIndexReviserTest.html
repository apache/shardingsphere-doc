45      private final DatabaseType databaseType = TypedSPILoader.getService(DatabaseType.class, "PostgreSQL");
46      
47      @Test
48      void assertReviseWithEmptyActualDataNode() {
49          assertThat(new ShardingIndexReviser(mock(ShardingTable.class)).revise("foo_tbl", new IndexMetaData("foo_idx"), Collections.emptyList(), Collections.emptyList(), mock(ShardingRule.class)),
50                  is(Optional.empty()));
51      }
52      
53      @Test
54      void assertReviseWithHashedActualIndexName() {
55          Optional<IndexMetaData> actual = revise("tbl_0", new IndexMetaData(IndexMetaDataUtils.getActualIndexName(
56                  "named_index_boundary_case_abcdefghijklmnopqrstuvwxyz", "tbl_0", databaseType), Collections.singletonList("foo_col")));
57          assertTrue(actual.isPresent());
58          assertThat(actual.get().getName(), is("named_index_boundary_case_abcdefghijklmnopqrstuvwxyz"));
59          assertThat(actual.get().getColumns().size(), is(1));
60      }
61      
62      @Test
63      void assertReviseWithTruncatedNamedActualIndexNameUsingCandidates() {
64          String logicIndexName = "named_index_boundary_case_abcdefghijklmnopqrstuvwxyz_0123456789";
65          IndexMetaData truncatedActualIndexMetaData = new IndexMetaData(
66                  IndexMetaDataUtils.getActualIndexName(logicIndexName, "tbl_0", databaseType), Collections.singletonList("foo_col"));
67          IndexMetaData legacyActualIndexMetaData = new IndexMetaData(IndexMetaDataUtils.getLegacyActualIndexName(logicIndexName, "tbl_1"), Collections.singletonList("foo_col"));
68          Optional<IndexMetaData> actual = revise("tbl_0", truncatedActualIndexMetaData,
69                  tableMetaData("tbl_0", truncatedActualIndexMetaData), tableMetaData("tbl_1", legacyActualIndexMetaData));
70          assertTrue(actual.isPresent());
71          assertThat(actual.get().getName(), is(logicIndexName));
72      }
73      
74      @SuppressWarnings({"rawtypes", "unchecked"})
75      @Test
76      void assertReviseThroughSharedIndexReviseEngineWithTruncatedNamedActualIndexName() {
77          String logicIndexName = "named_index_boundary_case_abcdefghijklmnopqrstuvwxyz_0123456789";
78          IndexMetaData truncatedActualIndexMetaData = new IndexMetaData(
79                  IndexMetaDataUtils.getActualIndexName(logicIndexName, "tbl_0", databaseType), Collections.singletonList("foo_col"));
80          IndexMetaData legacyActualIndexMetaData = new IndexMetaData(IndexMetaDataUtils.getLegacyActualIndexName(logicIndexName, "tbl_1"), Collections.singletonList("foo_col"));
81          ShardingRule rule = mock(ShardingRule.class);
82          MetaDataReviseEntry reviseEntry = mock(MetaDataReviseEntry.class);
83          ShardingIndexReviser indexReviser = new ShardingIndexReviser(mockShardingTable());
84          when(reviseEntry.getIndexReviser(rule, "tbl_0")).thenReturn(Optional.of(indexReviser));
85          Collection<IndexMetaData> actual = new IndexReviseEngine<>(rule, reviseEntry).revise("tbl_0", Collections.singleton(truncatedActualIndexMetaData),
86                  Arrays.asList(tableMetaData("tbl_0", truncatedActualIndexMetaData), tableMetaData("tbl_1", legacyActualIndexMetaData)), Collections.emptyList());
87          assertThat(actual.size(), is(1));
88          assertThat(actual.iterator().next().getName(), is(logicIndexName));
89      }

91      @SuppressWarnings({"rawtypes", "unchecked"})
92      @Test
93      void assertReviseThroughSharedIndexReviseEngineWithAllTruncatedNamedActualIndexNamesUsingRevisionCandidates() {
94          String logicIndexName = "named_index_boundary_case_abcdefghijklmnopqrstuvwxyz_0123456789";
95          IndexMetaData firstTruncatedActualIndexMetaData = new IndexMetaData(
96                  IndexMetaDataUtils.getActualIndexName(logicIndexName, "tbl_0", databaseType), Collections.singletonList("foo_col"));
97          IndexMetaData secondTruncatedActualIndexMetaData = new IndexMetaData(
98                  IndexMetaDataUtils.getActualIndexName(logicIndexName, "tbl_1", databaseType), Collections.singletonList("foo_col"));
99          ShardingRule rule = mock(ShardingRule.class);
100         MetaDataReviseEntry reviseEntry = mock(MetaDataReviseEntry.class);
101         ShardingIndexReviser indexReviser = new ShardingIndexReviser(mockShardingTable());
102         when(reviseEntry.getIndexReviser(rule, "tbl_0")).thenReturn(Optional.of(indexReviser));
103         Collection<IndexMetaData> actual = new IndexReviseEngine<>(rule, reviseEntry).revise("tbl_0", Collections.singleton(firstTruncatedActualIndexMetaData),
104                 Arrays.asList(tableMetaData("tbl_0", firstTruncatedActualIndexMetaData), tableMetaData("tbl_1", secondTruncatedActualIndexMetaData)),
105                 Collections.singleton(tableMetaData("tbl", new IndexMetaData(logicIndexName, Collections.singletonList("foo_col")))));
106         assertThat(actual.size(), is(1));
107         assertThat(actual.iterator().next().getName(), is(logicIndexName));
108     }

149     private Optional<IndexMetaData> revise(final String tableName, final IndexMetaData originalMetaData, final TableMetaData... originalTableMetaDataList) {
150         return new ShardingIndexReviser(mockShardingTable()).revise(tableName, originalMetaData,
151                 0 == originalTableMetaDataList.length ? Collections.singleton(tableMetaData(tableName, originalMetaData)) : Arrays.asList(originalTableMetaDataList), Collections.emptyList(),
152                 mock(ShardingRule.class));
153     }
154     
155     private static TableMetaData tableMetaData(final String tableName, final IndexMetaData... indexes) {
156         return new TableMetaData(tableName, Collections.emptyList(), Arrays.asList(indexes), Collections.emptyList());
157     }
158     
159     private static ShardingTable mockShardingTable() {
160         ShardingTable result = mock(ShardingTable.class);
161         when(result.getLogicTable()).thenReturn("tbl");
162         when(result.getActualDataNodes()).thenReturn(Arrays.asList(new DataNode("foo_schema", (String) null, "tbl_0"), new DataNode("foo_schema", (String) null, "tbl_1")));
163         return result;
164     }
