125     void assertDecorateUsesFirstDataSourceDefaultSchema() {
126         DataSource firstDataSource = mock(DataSource.class);
127         DatabaseType firstDatabaseType = mock(DatabaseType.class);
128         when(DatabaseTypeEngine.getStorageType(firstDataSource)).thenReturn(firstDatabaseType);
129         DataSource secondDataSource = mock(DataSource.class);
130         when(DatabaseTypeEngine.getStorageType(secondDataSource)).thenReturn(mock(DatabaseType.class));
131         Map<String, DataSource> dataSources = new LinkedHashMap<>(2, 1F);
132         dataSources.put("foo_ds", firstDataSource);
133         dataSources.put("bar_ds", secondDataSource);
134         when(DefaultSchemaNameResolver.resolveStorage(firstDatabaseType, firstDataSource, "foo_db")).thenReturn("foo_schema");
135         Map<String, Collection<DataNode>> actualDataNodes = Collections.singletonMap("foo_tbl", Collections.singleton(new DataNode("foo_ds", "foo_schema", "foo_tbl")));
136         when(SingleTableDataNodeLoader.load(anyString(), anyMap(), anyCollection(), anyCollection(), anyMap())).thenReturn(actualDataNodes);
137         Collection<String> splitTables = Collections.singleton("foo_ds.*");
138         when(SingleTableLoadUtils.splitTableLines(anyCollection())).thenReturn(splitTables);
139         when(SingleTableLoadUtils.getExcludedTables(anyCollection())).thenReturn(Collections.emptyList());
140         when(SingleTableLoadUtils.getFeatureRequiredSingleTables(anyCollection())).thenReturn(Collections.emptyList());
141         when(SingleTableLoadUtils.convertToDataNodesWithDefaultSchemaName(anyString(), any(DatabaseType.class), anyCollection())).thenAnswer(invocation -> {
142             assertThat(invocation.<String>getArgument(0), is("foo_schema"));
143             assertThat(invocation.<DatabaseType>getArgument(1), is(firstDatabaseType));
144             assertThat(invocation.<Collection<String>>getArgument(2), contains("foo_ds.*"));
145             return Collections.singleton(new DataNode("foo_ds", "foo_schema", "*"));
146         });
147         SingleRuleConfiguration ruleConfig = new SingleRuleConfiguration(splitTables, null);
148         try (MockedConstruction<DatabaseTypeRegistry> ignored = mockSchemaRegistry(true)) {
149             assertThat(decorator.decorate("foo_db", dataSources, Collections.emptyList(), ruleConfig).getTables(), contains("foo_ds.foo_schema.foo_tbl"));
150         }

309     private MockedConstruction<DatabaseTypeRegistry> mockSchemaRegistry(final boolean schemaAvailable) {
310         DialectDatabaseMetaData dialectDatabaseMetaData = mock(DialectDatabaseMetaData.class, RETURNS_DEEP_STUBS);
311         when(dialectDatabaseMetaData.getSchemaOption().isSchemaAvailable()).thenReturn(schemaAvailable);
312         return mockConstruction(DatabaseTypeRegistry.class, (mock, context) -> {
313             when(mock.getDialectDatabaseMetaData()).thenReturn(dialectDatabaseMetaData);
314         });
315     }

317     private void mockSplitAndConvert(final Collection<String> splitTables, final Collection<DataNode> configuredDataNodes,
318                                      final Collection<String> excludedTables, final Collection<String> featureRequiredTables) {
319         when(SingleTableLoadUtils.splitTableLines(anyCollection())).thenReturn(splitTables);
320         when(SingleTableLoadUtils.getExcludedTables(anyCollection())).thenReturn(excludedTables);
321         when(SingleTableLoadUtils.convertToDataNodes(anyString(), any(DatabaseType.class), anyCollection())).thenReturn(configuredDataNodes);
322         when(SingleTableLoadUtils.convertToDataNodesWithDefaultSchemaName(anyString(), any(DatabaseType.class), anyCollection())).thenReturn(configuredDataNodes);
323         when(SingleTableLoadUtils.getFeatureRequiredSingleTables(anyCollection())).thenReturn(featureRequiredTables);
324     }
