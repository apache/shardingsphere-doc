36  /**
37   * MCP workflow recovery payload factory.
38   */
39  @NoArgsConstructor(access = AccessLevel.PRIVATE)
40  final class MCPWorkflowRecoveryPayloadFactory {
41      
42      static Map<String, Object> createMissingExecutionModeRecovery(final MCPExecutionModeRequiredException cause) {
43          return WorkflowToolDescriptors.APPLY_TOOL_NAME.equals(cause.getToolName()) ? createMissingWorkflowExecutionModeRecovery(cause) : createMissingUpdateExecutionModeRecovery(cause);
44      }
45      
46      static Map<String, Object> createInvalidExecutionModeRecovery(final MCPInvalidExecutionModeException cause) {
47          return WorkflowToolDescriptors.APPLY_TOOL_NAME.equals(cause.getToolName()) ? createInvalidWorkflowExecutionModeRecovery(cause) : createInvalidUpdateExecutionModeRecovery(cause);
48      }
49      
50      static Map<String, Object> createInvalidApprovedStepsRecovery(final MCPInvalidApprovedStepsException cause) {
51          Map<String, Object> result = MCPRecoveryPayloadSupport.createBaseRecovery(
52                  "invalid_enum_value", "Retry database_gateway_apply_workflow with execution_mode=preview, review the returned preview_artifacts, "
53                          + "then pass explicit approved_steps copied from visible preview_artifacts.approval_step values.");
54          Map<String, Object> suggestedArguments = MCPRecoveryPayloadSupport.getSuggestedArguments(cause.getSuggestedArguments(),
55                  Map.of(WorkflowFieldNames.EXECUTION_MODE, WorkflowLifecycle.EXECUTION_MODE_PREVIEW));
56          result.put(MCPPayloadFieldNames.FIELD, WorkflowFieldNames.APPROVED_STEPS);
57          result.put(MCPPayloadFieldNames.ALLOWED_VALUES, cause.getAllowedValues());
58          result.put("suggested_arguments", suggestedArguments);
59          result.put(MCPPayloadFieldNames.NEXT_ACTIONS, List.of(MCPNextActionUtils.callTool(
60                  WorkflowToolDescriptors.APPLY_TOOL_NAME, "Preview again, then copy only visible approval_step values into approved_steps.", suggestedArguments)));
61          return result;
62      }
63      
64      static Map<String, Object> createWorkflowStateRecovery(final MCPWorkflowStateException cause) {
65          Map<String, Object> result = MCPRecoveryPayloadSupport.createBaseRecovery(
66                  "stale_workflow", "Use current-session completion to select an available plan_id, or re-run the matching planning tool.");
67          result.put(WorkflowFieldNames.PLAN_ID, cause.getPlanId());
68          result.put(MCPPayloadFieldNames.RESOURCES_TO_READ, MCPRecoveryPayloadSupport.createResourceHintList(
69                  "shardingsphere://capabilities", "capability", "Read workflow-capable MCP tools before re-planning."));
70          result.put("completion_first", Map.of("argument", WorkflowFieldNames.PLAN_ID, "scope", "current MCP session"));
71          result.put(MCPPayloadFieldNames.NEXT_ACTIONS, MCPNextActionUtils.ordered(
72                  MCPNextActionUtils.completeArgument(MCPCompletionAction.builder()
73                          .referenceType("resource")
74                          .reference("shardingsphere://workflows/{plan_id}")
75                          .argumentName(WorkflowFieldNames.PLAN_ID)
76                          .resumeTargetType("resource")
77                          .resumeTarget("shardingsphere://workflows/{plan_id}")
78                          .reason("Use MCP completion for plan_id to pick an available current-session workflow plan.")
79                          .build()),
80                  MCPNextActionUtils.dependsOn(MCPNextActionUtils.readResource(
81                          "shardingsphere://capabilities", "Read current workflow tools, then re-run the matching planning tool if completion has no usable plan."), 1)));
82          return result;
83      }
84      
85      private static Map<String, Object> createMissingUpdateExecutionModeRecovery(final MCPExecutionModeRequiredException cause) {
86          return createExecutionModeRecovery("missing_execution_mode",
87                  "Retry the same side-effecting tool with execution_mode=preview, then execute only when the requested side effect is still intended.",
88                  cause.getToolName(), cause.getAllowedValues(), cause.getSuggestedArguments(), "Retry the same side-effecting tool in preview mode.", true, true);
89      }
90      
91      private static Map<String, Object> createMissingWorkflowExecutionModeRecovery(final MCPExecutionModeRequiredException cause) {
92          return createExecutionModeRecovery("missing_execution_mode",
93                  "Retry database_gateway_apply_workflow with execution_mode=preview, then use review-then-execute or manual-only after review.",
94                  cause.getToolName(), cause.getAllowedValues(), cause.getSuggestedArguments(), "Retry database_gateway_apply_workflow with execution_mode=preview first.", true, true);
95      }
96      
97      private static Map<String, Object> createInvalidUpdateExecutionModeRecovery(final MCPInvalidExecutionModeException cause) {
98          return createExecutionModeRecovery("invalid_enum_value", "Retry with execution_mode=preview or execution_mode=execute.", cause.getToolName(), cause.getAllowedValues(),
99                  cause.getSuggestedArguments(), "Retry database_gateway_execute_update with execution_mode=preview first.", false, false);
100     }
101     
102     private static Map<String, Object> createInvalidWorkflowExecutionModeRecovery(final MCPInvalidExecutionModeException cause) {
103         return createExecutionModeRecovery("invalid_enum_value",
104                 "Retry database_gateway_apply_workflow with execution_mode=preview, then use review-then-execute or manual-only after review.",
105                 cause.getToolName(), cause.getAllowedValues(), cause.getSuggestedArguments(), "Retry database_gateway_apply_workflow with execution_mode=preview first.", false, false);
106     }
107     
108     private static Map<String, Object> createExecutionModeRecovery(final String category, final String modelAction, final String toolName, final List<String> allowedValues,
109                                                                    final Map<String, Object> sourceSuggestedArguments, final String retryReason, final boolean missingField,
110                                                                    final boolean retryTool) {
111         Map<String, Object> result = MCPRecoveryPayloadSupport.createBaseRecovery(category, modelAction);
112         if (missingField) {
113             result.put("missing_fields", List.of(WorkflowFieldNames.EXECUTION_MODE));
114         }
115         result.put(MCPPayloadFieldNames.FIELD, WorkflowFieldNames.EXECUTION_MODE);
116         result.put("tool_name", toolName);
117         result.put(MCPPayloadFieldNames.ALLOWED_VALUES, allowedValues);
118         Map<String, Object> suggestedArguments = MCPRecoveryPayloadSupport.getSuggestedArguments(sourceSuggestedArguments,
119                 Map.of(WorkflowFieldNames.EXECUTION_MODE, WorkflowLifecycle.EXECUTION_MODE_PREVIEW));
120         result.put("suggested_arguments", suggestedArguments);
121         result.put(MCPPayloadFieldNames.NEXT_ACTIONS, List.of(retryTool
122                 ? MCPNextActionUtils.retryTool(toolName, retryReason, suggestedArguments)
123                 : MCPNextActionUtils.callTool(toolName, retryReason, suggestedArguments)));
124         return result;
125     }
