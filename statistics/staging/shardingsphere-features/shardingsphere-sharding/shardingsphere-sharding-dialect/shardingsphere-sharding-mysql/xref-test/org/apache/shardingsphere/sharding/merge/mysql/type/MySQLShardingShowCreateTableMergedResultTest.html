69      private ShardingRule buildShardingRule() {
70          ShardingRuleConfiguration shardingRuleConfig = new ShardingRuleConfiguration();
71          shardingRuleConfig.getTables().add(new ShardingTableRuleConfiguration("foo_tbl", "ds.foo_tbl_${0..2}"));
72          shardingRuleConfig.getTables().add(new ShardingTableRuleConfiguration("bar_tbl", "ds.bar_tbl_${0..2}"));
73          return new ShardingRule(shardingRuleConfig, Collections.singletonMap("ds", new MockedDataSource()), mock(ComputeNodeInstanceContext.class), Collections.emptyList());
74      }

76      private ShardingSphereSchema createSchema() {
77          Collection<ShardingSphereTable> tables = new LinkedList<>();
78          tables.add(new ShardingSphereTable("foo_tbl", Collections.emptyList(), Collections.emptyList(), Collections.singleton(new ShardingSphereConstraint("foo_tbl_foreign_key", "bar_tbl"))));
79          tables.add(new ShardingSphereTable("bar_tbl", Collections.emptyList(), Collections.emptyList(), Collections.emptyList()));
80          return new ShardingSphereSchema("foo_db", mock(DatabaseType.class), tables, Collections.emptyList());
81      }

133     @Test
134     void assertGetValueWithDollarSignInTableNames() throws SQLException {
135         MySQLShardingShowCreateTableMergedResult actual = new MySQLShardingShowCreateTableMergedResult(
136                 buildShardingRuleWithDollarSign(), mockSQLStatementContext("foo$tbl"), createSchemaWithDollarSign(), Collections.singletonList(mockQueryResultWithDollarSign()));
137         assertTrue(actual.next());
138         assertThat(actual.getValue(2, String.class), is("CREATE TABLE `foo$tbl` (FOREIGN KEY (`bar_id`) REFERENCES `bar$tbl` (`bar_id`))"));
139     }

141     private ShardingRule buildShardingRuleWithDollarSign() {
142         ShardingRule result = mock(ShardingRule.class);
143         ShardingTable shardingTable = mock(ShardingTable.class);
144         when(shardingTable.getLogicTable()).thenReturn("foo$tbl");
145         when(shardingTable.getActualDataNodes()).thenReturn(Collections.singletonList(new DataNode("ds.foo$tbl_0")));
146         when(result.findShardingTableByActualTable("foo$tbl_0")).thenReturn(Optional.of(shardingTable));
147         when(result.findShardingTable("foo$tbl")).thenReturn(Optional.of(shardingTable));
148         BindingTableRule bindingTableRule = mock(BindingTableRule.class);
149         when(bindingTableRule.getAllLogicTables()).thenReturn(Collections.singleton("bar$tbl"));
150         when(result.findBindingTableRule("foo$tbl")).thenReturn(Optional.of(bindingTableRule));
151         when(result.getLogicAndActualTablesFromBindingTable("ds", "foo$tbl", "foo$tbl_0", Collections.singleton("bar$tbl")))
152                 .thenReturn(Collections.singletonMap("bar$tbl", "bar$tbl_0"));
153         return result;
154     }
