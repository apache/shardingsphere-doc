265     private InsertStatementContext createInsertStatementContext(final SimpleTableSegment table, final List<ColumnSegment> insertColumns,
266                                                                 final List<ExpressionSegment> values, final List<ColumnAssignmentSegment> onDuplicateAssignments) {
267         InsertStatement insertStatement = InsertStatement.builder()
268                 .databaseType(databaseType)
269                 .table(table)
270                 .insertColumns(insertColumns.isEmpty() ? null : new InsertColumnsSegment(0, 0, insertColumns))
271                 .values(Collections.singletonList(new InsertValuesSegment(0, 0, values)))
272                 .onDuplicateKeyColumns(onDuplicateAssignments.isEmpty() ? null : new OnDuplicateKeyColumnsSegment(0, 0, onDuplicateAssignments))
273                 .build();
274         insertStatement.addParameterMarkers(createParameterMarkerSegments(values));
275         if (!onDuplicateAssignments.isEmpty()) {
276             List<ExpressionSegment> onDuplicateExpressions = new LinkedList<>();
277             for (ColumnAssignmentSegment each : onDuplicateAssignments) {
278                 onDuplicateExpressions.add(each.getValue());
279             }
280             insertStatement.addParameterMarkers(createParameterMarkerSegments(onDuplicateExpressions));
281         }
282         return new InsertStatementContext(insertStatement, createMetaData(), "foo_db");
283     }

285     private UpdateStatementContext createUpdateStatementContext(final TableSegment table, final Collection<ColumnAssignmentSegment> assignments, final ExpressionSegment whereExpression) {
286         UpdateStatement updateStatement = UpdateStatement.builder()
287                 .databaseType(databaseType)
288                 .table(table)
289                 .setAssignment(new SetAssignmentSegment(0, 0, assignments))
290                 .where(null == whereExpression ? null : new WhereSegment(0, 0, whereExpression))
291                 .build();
292         List<ExpressionSegment> expressions = new LinkedList<>();
293         for (ColumnAssignmentSegment each : assignments) {
294             expressions.add(each.getValue());
295         }
296         if (null != whereExpression) {
297             expressions.add(whereExpression);
298         }
299         updateStatement.addParameterMarkers(createParameterMarkerSegments(expressions));
300         return new UpdateStatementContext(updateStatement);
301     }

324     private Collection<ParameterMarkerSegment> createParameterMarkerSegments(final Collection<ExpressionSegment> expressions) {
325         Collection<ParameterMarkerSegment> result = new LinkedList<>();
326         for (ExpressionSegment each : expressions) {
327             result.addAll(createParameterMarkerSegments(each));
328         }
329         return result;
330     }
331     
332     private Collection<ParameterMarkerSegment> createParameterMarkerSegments(final ExpressionSegment expression) {
333         Collection<ParameterMarkerSegment> result = new LinkedList<>();
334         if (expression instanceof ParameterMarkerExpressionSegment) {
335             result.add((ParameterMarkerExpressionSegment) expression);
336             return result;
337         }
338         if (expression instanceof BinaryOperationExpression) {
339             BinaryOperationExpression binaryOperationExpression = (BinaryOperationExpression) expression;
340             result.addAll(createParameterMarkerSegments(binaryOperationExpression.getLeft()));
341             result.addAll(createParameterMarkerSegments(binaryOperationExpression.getRight()));
342             return result;
343         }
344         if (expression instanceof InExpression) {
345             InExpression inExpression = (InExpression) expression;
346             result.addAll(createParameterMarkerSegments(inExpression.getLeft()));
347             result.addAll(createParameterMarkerSegments(inExpression.getExpressionList()));
348             return result;
349         }
350         if (expression instanceof BetweenExpression) {
351             BetweenExpression betweenExpression = (BetweenExpression) expression;
352             result.addAll(createParameterMarkerSegments(betweenExpression.getLeft()));
353             result.addAll(createParameterMarkerSegments(betweenExpression.getBetweenExpr()));
354             result.addAll(createParameterMarkerSegments(betweenExpression.getAndExpr()));
355             return result;
356         }
357         if (expression instanceof ListExpression) {
358             result.addAll(createParameterMarkerSegments(((ListExpression) expression).getItems()));
359         }
360         return result;
361     }
362     
363     private ListExpression createListExpression(final ExpressionSegment... expressions) {
364         ListExpression result = new ListExpression(0, 0);
365         result.getItems().addAll(Arrays.asList(expressions));
366         return result;
