82      @MethodSource("provideSelectStatements")
83      void assertExtractFromSelectStatement(final String name, final SelectStatement selectStatement, final boolean containsSubQuery,

85          Collection<ColumnSegment> columnSegments = new LinkedList<>();
86          ColumnExtractor.extractFromSelectStatement(columnSegments, selectStatement, containsSubQuery);
87          assertThat(toColumnNames(columnSegments), is(expectedColumnNames));
88      }

151     private static Collection<WhereSegment> createWhereSegments() {
152         BinaryOperationExpression leftExpression = new BinaryOperationExpression(10, 24,
153                 new ColumnSegment(10, 13, new IdentifierValue("foo_name")), new LiteralExpressionSegment(18, 22, "LiLei"), "=", "foo_name = 'LiLei'");
154         BinaryOperationExpression rightExpression = new BinaryOperationExpression(30, 44,

157     }

159     private static SelectStatement createSelectStatementForExtraction() {
160         ProjectionsSegment projections = new ProjectionsSegment(0, 0);
161         projections.getProjections().add(new ColumnProjectionSegment(createColumnSegment("foo_projection_column")));
162         AggregationProjectionSegment aggregationProjectionSegment = new AggregationProjectionSegment(0, 0, AggregationType.COUNT, "COUNT(foo_projection_agg_param)");
163         aggregationProjectionSegment.getParameters().add(createColumnSegment("foo_projection_agg_param"));
164         projections.getProjections().add(aggregationProjectionSegment);
165         projections.getProjections().add(new DatetimeProjectionSegment(0, 0, createColumnSegment("foo_datetime_left"), createColumnSegment("foo_datetime_right"), "date_add"));
166         projections.getProjections().add(new ExpressionProjectionSegment(0, 0, "foo_expression_projection", createColumnSegment("foo_expression_projection")));
167         projections.getProjections().add(new IntervalExpressionProjection(0, 0, createColumnSegment("foo_interval_left"),
168                 createColumnSegment("foo_interval_minus"), createColumnSegment("foo_interval_right"), "interval expr"));
169         projections.getProjections().add(new SubqueryProjectionSegment(new SubquerySegment(0, 0, createSelectStatementWithProjection("foo_subquery_projection_column"), ""), "subquery"));
170         SelectStatement result = new SelectStatement(mock(DatabaseType.class));
171         result.setProjections(projections);
172         result.setFrom(createJoinTableForExtraction());
173         result.setWhere(new WhereSegment(0, 0, createBinaryOperation("foo_where_left", "bar_where_right")));
174         result.setGroupBy(createGroupBySegment());
175         result.setHaving(new HavingSegment(0, 0, createBinaryOperation("foo_having_left", "bar_having_right")));
176         result.setOrderBy(createOrderBySegment());
177         result.setCombine(createCombineSegment());
178         return result;
179     }

210     private static CombineSegment createCombineSegment() {
211         SubquerySegment left = new SubquerySegment(0, 0, createSelectStatementWithProjection("foo_combine_left_column"), "");
212         SubquerySegment right = new SubquerySegment(0, 0, createSelectStatementWithProjection("bar_combine_right_column"), "");
213         return new CombineSegment(0, 0, left, CombineType.UNION, right);

216     private static JoinTableSegment createJoinTableForExtraction() {
217         JoinTableSegment innerJoin = new JoinTableSegment();
218         innerJoin.setLeft(new CollectionTableSegment(createColumnSegment("foo_collection_expr")));
219         CommonTableExpressionSegment commonTableExpressionSegment = new CommonTableExpressionSegment(0, 0, new AliasSegment(0, 0, new IdentifierValue("cte_alias")),
220                 new SubquerySegment(0, 0, createSelectStatementWithProjection("foo_cte_subquery_column"), ""));
221         innerJoin.setRight(commonTableExpressionSegment);
222         innerJoin.setCondition(createBinaryOperation("foo_inner_join_left", "bar_inner_join_right"));
223         JoinTableSegment result = new JoinTableSegment();
224         result.setLeft(innerJoin);
225         result.setRight(new SubqueryTableSegment(0, 0, new SubquerySegment(0, 0, createSelectStatementWithProjection("foo_subquery_table_column"), "")));
226         result.setCondition(createBinaryOperation("foo_join_left", "bar_join_right"));
227         result.setUsing(Collections.singletonList(createColumnSegment("foo_using_column")));
228         result.setDerivedUsing(Collections.singletonList(createColumnSegment("bar_derived_using_column")));
229         return result;
230     }
