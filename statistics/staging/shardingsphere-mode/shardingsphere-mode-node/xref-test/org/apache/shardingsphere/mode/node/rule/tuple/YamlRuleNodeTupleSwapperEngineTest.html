38  class YamlRuleNodeTupleSwapperEngineTest {

40      @Test
41      void assertSwapToTuple() {
42          RuleNodeTuple actual = new YamlRuleNodeTupleSwapperEngine().swapToTuple(new YamlLeafRuleConfiguration("foo"));
43          assertThat(actual.getPath(), is("/rules/leaf"));
44          assertThat(actual.getContent(), is("value: foo" + System.lineSeparator()));
45      }

49          assertTrue(new YamlRuleNodeTupleSwapperEngine().swapToTuples("foo_db", new YamlNodeRuleConfiguration()).isEmpty());
50      }
51      
52      @Test
53      void assertSwapToTuplesWithNodeYamlRuleConfiguration() {
54          YamlNodeRuleConfiguration yamlRuleConfig = new YamlNodeRuleConfiguration();
55          yamlRuleConfig.setMapValue(Collections.singletonMap("k", new YamlLeafRuleConfiguration("v")));
56          yamlRuleConfig.setCollectionValue(Collections.singletonList(new YamlLeafRuleConfiguration("foo")));
57          yamlRuleConfig.setStringValue("str");
58          yamlRuleConfig.setBooleanValue(true);
59          yamlRuleConfig.setIntegerValue(1);
60          yamlRuleConfig.setLongValue(10L);
61          yamlRuleConfig.setEnumValue(YamlNodeRuleConfigurationEnum.FOO);
62          YamlLeafRuleConfiguration leaf = new YamlLeafRuleConfiguration();
63          leaf.setValue("leaf");
64          yamlRuleConfig.setLeaf(leaf);
65          yamlRuleConfig.setGens(Collections.singleton("value"));
66          yamlRuleConfig.setGen("single_gen");
67          List<RuleNodeTuple> actual = new ArrayList<>(new YamlRuleNodeTupleSwapperEngine().swapToTuples("foo_db", yamlRuleConfig));
68          assertThat(actual.size(), is(10));
69          assertThat(actual.get(0).getPath(), is("/metadata/foo_db/rules/node/map_value/k"));
70          assertThat(actual.get(0).getContent(), is("value: v" + System.lineSeparator()));
71          assertThat(actual.get(1).getPath(), is("/metadata/foo_db/rules/node/collection_value"));
72          assertThat(actual.get(1).getContent(), is("- !LEAF" + System.lineSeparator() + "  value: foo" + System.lineSeparator()));
73          assertThat(actual.get(2).getPath(), is("/metadata/foo_db/rules/node/string_value"));
74          assertThat(actual.get(2).getContent(), is("str"));
75          assertThat(actual.get(3).getPath(), is("/metadata/foo_db/rules/node/boolean_value"));
76          assertThat(actual.get(3).getContent(), is("true"));
77          assertThat(actual.get(4).getPath(), is("/metadata/foo_db/rules/node/integer_value"));
78          assertThat(actual.get(4).getContent(), is("1"));
79          assertThat(actual.get(5).getPath(), is("/metadata/foo_db/rules/node/long_value"));
80          assertThat(actual.get(5).getContent(), is("10"));
81          assertThat(actual.get(6).getPath(), is("/metadata/foo_db/rules/node/enum_value"));
82          assertThat(actual.get(6).getContent(), is("FOO"));
83          assertThat(actual.get(7).getPath(), is("/metadata/foo_db/rules/node/leaf"));
84          assertThat(actual.get(7).getContent(), is("value: leaf" + System.lineSeparator()));
85          assertThat(actual.get(8).getPath(), is("/metadata/foo_db/rules/node/gens/gen: value"));
86          assertThat(actual.get(8).getContent(), is("value"));
87          assertThat(actual.get(9).getPath(), is("/metadata/foo_db/rules/node/gen"));
88          assertThat(actual.get(9).getContent(), is("single_gen"));
89      }
90      
91      @Test
92      void assertSwapToYamlRuleConfigurationWithEmptyNodeYamlRuleConfiguration() {
93          DatabaseRuleNodePath databaseRuleNodePath = new DatabaseRuleNodePath("foo_db", "node", new DatabaseRuleItem("string_value"));
94          YamlNodeRuleConfiguration actual = (YamlNodeRuleConfiguration) new YamlRuleNodeTupleSwapperEngine().swapToYamlDatabaseRuleConfiguration("foo_db", "node",
95                  Collections.singleton(new RuleNodeTuple(databaseRuleNodePath, "")));
96          assertThat(actual.getStringValue(), is(""));
97      }
98      
99      @Test
100     void assertSwapToYamlRuleConfigurationWithNodeYamlRuleConfiguration() {
101         YamlNodeRuleConfiguration actual = (YamlNodeRuleConfiguration) new YamlRuleNodeTupleSwapperEngine().swapToYamlDatabaseRuleConfiguration("foo_db", "node", Arrays.asList(
102                 new RuleNodeTuple(new DatabaseRuleNodePath("foo_db", "node", new DatabaseRuleItem("map_value/k")), "v"),
103                 new RuleNodeTuple(new DatabaseRuleNodePath("foo_db", "node", new DatabaseRuleItem("map_value/k:qualified")), "k:qualified"),
104                 new RuleNodeTuple(new DatabaseRuleNodePath("foo_db", "node", new DatabaseRuleItem("collection_value")), "- !LEAF" + System.lineSeparator() + "  value: foo"),
105                 new RuleNodeTuple(new DatabaseRuleNodePath("foo_db", "node", new DatabaseRuleItem("string_value")), "str"),
106                 new RuleNodeTuple(new DatabaseRuleNodePath("foo_db", "node", new DatabaseRuleItem("boolean_value")), "true"),
107                 new RuleNodeTuple(new DatabaseRuleNodePath("foo_db", "node", new DatabaseRuleItem("integer_value")), "1"),
108                 new RuleNodeTuple(new DatabaseRuleNodePath("foo_db", "node", new DatabaseRuleItem("long_value")), "10"),
109                 new RuleNodeTuple(new DatabaseRuleNodePath("foo_db", "node", new DatabaseRuleItem("enum_value")), "FOO")));
110         assertThat(actual.getMapValue().size(), is(2));
111         assertThat(actual.getMapValue().get("k").getValue(), is("v"));
112         assertThat(actual.getMapValue().get("k:qualified").getValue(), is("k:qualified"));
113         assertThat(actual.getCollectionValue().size(), is(1));
114         assertThat(actual.getCollectionValue().iterator().next().getValue(), is("foo"));
115         assertThat(actual.getStringValue(), is("str"));
116         assertTrue(actual.getBooleanValue());
117         assertThat(actual.getIntegerValue(), is(1));
118         assertThat(actual.getLongValue(), is(10L));
119         assertThat(actual.getEnumValue(), is(YamlNodeRuleConfigurationEnum.FOO));
120     }
