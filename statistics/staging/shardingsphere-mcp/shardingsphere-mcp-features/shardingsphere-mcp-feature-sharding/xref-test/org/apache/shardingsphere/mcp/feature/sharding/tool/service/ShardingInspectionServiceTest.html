40  class ShardingInspectionServiceTest {
41      
42      @ParameterizedTest(name = "{0}")
43      @MethodSource("assertDatabaseScopedQueryArguments")
44      void assertDatabaseScopedQuery(final String name, final QueryInvocation invocation, final String expectedSQL) {
45          MCPFeatureQueryFacade queryFacade = mock(MCPFeatureQueryFacade.class);
46          invocation.invoke(new ShardingInspectionService(), queryFacade);
47          verify(queryFacade).query(eq("logic_db"), eq(expectedSQL));
48      }
49      
50      @ParameterizedTest(name = "{0}")
51      @MethodSource("assertQueryAlgorithmPluginsArguments")
52      void assertQueryAlgorithmPlugins(final String name, final String algorithmType, final String expectedPropertyGuidance) {
53          MCPFeatureQueryFacade queryFacade = mock(MCPFeatureQueryFacade.class);
54          when(queryFacade.queryWithAnyDatabase("SHOW SHARDING ALGORITHM PLUGINS")).thenReturn(List.of(Map.of("type", algorithmType)));
55          List<Map<String, Object>> actual = new ShardingInspectionService().queryAlgorithmPlugins(queryFacade);
56          assertThat(actual.getFirst().get("property_guidance").toString(), is(expectedPropertyGuidance));
57      }
58      
59      @Test
60      void assertQueryAlgorithmPluginsWithUnavailableDistSQL() {
61          MCPFeatureQueryFacade queryFacade = mock(MCPFeatureQueryFacade.class);
62          MCPQueryFailedException expected = new MCPQueryFailedException(
63                  "syntax error near 'SHARDING ALGORITHM PLUGINS'", new SQLSyntaxErrorException("syntax error"));
64          when(queryFacade.queryWithAnyDatabase("SHOW SHARDING ALGORITHM PLUGINS")).thenThrow(expected);
65          MCPQueryFailedException actual = assertThrows(MCPQueryFailedException.class, () -> new ShardingInspectionService().queryAlgorithmPlugins(queryFacade));
66          assertThat(actual, is(expected));
67      }
68      
69      @Test
70      void assertQueryKeyGenerateAlgorithmPlugins() {
71          MCPFeatureQueryFacade queryFacade = mock(MCPFeatureQueryFacade.class);
72          when(queryFacade.queryWithAnyDatabase("SHOW KEY GENERATE ALGORITHM PLUGINS")).thenReturn(List.of(Map.of("type", "UUID")));
73          List<Map<String, Object>> actual = new ShardingInspectionService().queryKeyGenerateAlgorithmPlugins(queryFacade);
74          assertThat(actual.getFirst().get("property_guidance").toString(), is("No required properties."));
75      }
76      
77      @Test
78      void assertQueryKeyGenerateAlgorithmPluginsWithUnavailableDistSQL() {
79          MCPFeatureQueryFacade queryFacade = mock(MCPFeatureQueryFacade.class);
80          MCPQueryFailedException expected = new MCPQueryFailedException(
81                  "syntax error near 'KEY GENERATE ALGORITHM PLUGINS'", new SQLSyntaxErrorException("syntax error"));
82          when(queryFacade.queryWithAnyDatabase("SHOW KEY GENERATE ALGORITHM PLUGINS")).thenThrow(expected);
83          MCPQueryFailedException actual = assertThrows(MCPQueryFailedException.class, () -> new ShardingInspectionService().queryKeyGenerateAlgorithmPlugins(queryFacade));
84          assertThat(actual, is(expected));
85      }

87      private static Stream<Arguments> assertQueryAlgorithmPluginsArguments() {
88          return Stream.of(
89                  Arguments.of("inline", "INLINE", "Usually requires algorithm-expression."),
90                  Arguments.of("mod", "MOD", "Usually requires sharding-count."),
91                  Arguments.of("hash mod", "HASH_MOD", "Usually requires sharding-count."),
92                  Arguments.of("unknown inline", "CUSTOM_INLINE", "Read the plugin documentation for required properties before planning."));
93      }
