53  /**
54   * Assignment generator for encrypt.
55   */
56  @HighFrequencyInvocation
57  @AllArgsConstructor
58  @Slf4j
59  public final class EncryptAssignmentTokenGenerator {
60      
61      private static final String UNSUPPORTED_ASSIGNMENT_MESSAGE = "Encrypted assignments with multiple columns, nested fields or unsupported expressions";
62      
63      private final EncryptRule rule;
64      
65      private final ShardingSphereDatabase database;
66      
67      private final DatabaseType databaseType;
68      
69      /**
70       * Generate SQL tokens.
71       *
72       * @param tablesContext SQL statement context
73       * @param setAssignmentSegment set assignment segment
74       * @return generated SQL tokens
75       */
76      public Collection<SQLToken> generateSQLTokens(final TablesContext tablesContext, final SetAssignmentSegment setAssignmentSegment) {
77          Collection<SQLToken> result = new LinkedList<>();
78          DatabaseTypeRegistry databaseTypeRegistry = new DatabaseTypeRegistry(databaseType);
79          String schemaName = tablesContext.getSchemaName().orElseGet(database::getDefaultSchemaName);
80          QuoteCharacter quoteCharacter = databaseTypeRegistry.getDialectDatabaseMetaData().getQuoteCharacter();
81          for (ColumnAssignmentSegment each : setAssignmentSegment.getAssignments()) {
82              ColumnSegment assignedColumn = getAssignedColumn(each);
83              if (each.getColumns().size() > 1 || null != assignedColumn.getNestedObjectAttributes() && !assignedColumn.getNestedObjectAttributes().isEmpty()) {
84                  checkUnsupportedAssignment(each);
85                  continue;
86              }
87              findEncryptTable(assignedColumn).ifPresent(encryptTable -> {
88                  String columnName = assignedColumn.getIdentifier().getValue();
89                  if (encryptTable.isEncryptColumn(columnName)) {
90                      ExpressionSegment value = each.getValue();
91                      ShardingSpherePreconditions.checkState(null == value || !isUnsupportedExpression(value), () -> new UnsupportedSQLOperationException(UNSUPPORTED_ASSIGNMENT_MESSAGE));
92                      result.addAll(generateAssignmentSQLTokens(schemaName, encryptTable.getTable(), encryptTable.getEncryptColumn(columnName), each, quoteCharacter));
93                  }
94              });
95          }
96          return result;
97      }
98      
99      private void checkUnsupportedAssignment(final ColumnAssignmentSegment assignment) {
100         for (ColumnSegment each : assignment.getColumns()) {
101             ShardingSpherePreconditions.checkState(!findEncryptTable(each).map(optional -> optional.isEncryptColumn(each.getIdentifier().getValue())).orElse(false),
102                     () -> new UnsupportedSQLOperationException(UNSUPPORTED_ASSIGNMENT_MESSAGE));
103         }
104     }
105     
106     private boolean isUnsupportedExpression(final ExpressionSegment expression) {
107         if (!(expression instanceof FunctionSegment)) {
108             return !(expression instanceof ParameterMarkerExpressionSegment) && !(expression instanceof LiteralExpressionSegment);
109         }
110         for (ExpressionSegment each : ((FunctionSegment) expression).getParameters()) {
111             if (!(each instanceof ColumnSegment) && isUnsupportedExpression(each)) {
112                 return true;
113             }
114         }
115         return false;
116     }
117     
118     private Collection<SQLToken> generateAssignmentSQLTokens(final String schemaName, final String tableName, final EncryptColumn encryptColumn,
119                                                              final ColumnAssignmentSegment segment, final QuoteCharacter quoteCharacter) {
120         ExpressionSegment value = segment.getValue();
121         if (value instanceof ParameterMarkerExpressionSegment) {
122             return Collections.singleton(generateParameterSQLToken(encryptColumn, segment, quoteCharacter));
123         }
124         if (value instanceof LiteralExpressionSegment) {
125             return Collections.singleton(generateLiteralSQLToken(schemaName, tableName, encryptColumn, segment, quoteCharacter));
126         }
127         return Collections.emptyList();
128     }
129     
130     private EncryptAssignmentToken generateParameterSQLToken(final EncryptColumn encryptColumn, final ColumnAssignmentSegment segment, final QuoteCharacter quoteCharacter) {
131         ColumnSegment leftColumn = getAssignedColumn(segment);
132         EncryptParameterAssignmentToken result = new EncryptParameterAssignmentToken(leftColumn.getStartIndex(), segment.getStopIndex(), quoteCharacter);
133         appendEncryptColumnTokens(leftColumn, encryptColumn, (targetName, suffix) -> result.addColumnName(targetName));
134         return result;
135     }
136     
137     private String getColumnName(final ColumnSegment columnSegment, final EncryptDerivedColumnSuffix derivedColumnSuffix, final String actualColumnName) {
138         return TableSourceType.TEMPORARY_TABLE == columnSegment.getColumnBoundInfo().getTableSourceType()
139                 ? derivedColumnSuffix.getDerivedColumnName(columnSegment.getIdentifier().getValue(), database.getIdentifierContext())
140                 : actualColumnName;
141     }
142     
143     private EncryptAssignmentToken generateLiteralSQLToken(final String schemaName, final String tableName, final EncryptColumn encryptColumn, final ColumnAssignmentSegment segment,
144                                                            final QuoteCharacter quoteCharacter) {
145         ColumnSegment leftColumn = getAssignedColumn(segment);
146         EncryptLiteralAssignmentToken result = new EncryptLiteralAssignmentToken(leftColumn.getStartIndex(), segment.getStopIndex(), quoteCharacter);
147         Object literalValue = ((LiteralExpressionSegment) segment.getValue()).getLiterals();
148         appendEncryptColumnTokens(leftColumn, encryptColumn, (targetName, suffix) -> addLiteralSQLToken(schemaName, tableName, encryptColumn, targetName, suffix, result, literalValue));
149         return result;
150     }
151     
152     private void addLiteralSQLToken(final String schemaName, final String tableName, final EncryptColumn encryptColumn, final String targetColumnName, final EncryptDerivedColumnSuffix suffix,
153                                     final EncryptLiteralAssignmentToken result, final Object literalValue) {
154         if (null == suffix) {
155             result.addAssignment(targetColumnName, literalValue);
156         } else {
157             Object encryptValue = encrypt(encryptColumn, suffix, database.getName(), schemaName, tableName, encryptColumn.getName(), literalValue);
158             result.addAssignment(targetColumnName, encryptValue);
159         }
160     }
161     
162     private Object encrypt(final EncryptColumn encryptColumn, final EncryptDerivedColumnSuffix suffix, final String databaseName, final String schemaName,
163                            final String tableName, final String logicColumnName, final Object originalValue) {
164         List<Object> originalValues = Collections.singletonList(originalValue);
165         switch (suffix) {
166             case CIPHER:
167                 return encryptColumn.getCipher().encrypt(databaseName, schemaName, tableName, logicColumnName, originalValues).iterator().next();
168             case ASSISTED_QUERY:
169                 return encryptColumn.getAssistedQuery().map(optional -> optional.encrypt(databaseName, schemaName, tableName, logicColumnName, originalValues).iterator().next()).orElse(null);
170             case LIKE_QUERY:
171                 return encryptColumn.getLikeQuery().map(optional -> optional.encrypt(databaseName, schemaName, tableName, logicColumnName, originalValues).iterator().next()).orElse(null);
172             default:
173                 return null;
174         }
175     }
176     
177     private Optional<EncryptTable> findEncryptTable(final ColumnSegment columnSegment) {
178         return rule.findEncryptTable(columnSegment.getColumnBoundInfo().getOriginalTable().getValue());
179     }
180     
181     private void appendEncryptColumnTokens(final ColumnSegment leftColumn, final EncryptColumn encryptColumn, final EncryptColumnConsumer consumer) {
182         consumer.accept(getColumnName(leftColumn, EncryptDerivedColumnSuffix.CIPHER, encryptColumn.getCipher().getName()), EncryptDerivedColumnSuffix.CIPHER);
183         encryptColumn.getAssistedQuery()
184                 .ifPresent(optional -> consumer.accept(getColumnName(leftColumn, EncryptDerivedColumnSuffix.ASSISTED_QUERY, optional.getName()), EncryptDerivedColumnSuffix.ASSISTED_QUERY));
185         encryptColumn.getLikeQuery()
186                 .ifPresent(optional -> consumer.accept(getColumnName(leftColumn, EncryptDerivedColumnSuffix.LIKE_QUERY, optional.getName()), EncryptDerivedColumnSuffix.LIKE_QUERY));
187     }
188     
189     private ColumnSegment getAssignedColumn(final ColumnAssignmentSegment assignmentSegment) {
190         return assignmentSegment.getColumns().get(0);
191     }
192     
193     @FunctionalInterface
194     private interface EncryptColumnConsumer {
195         
196         void accept(String targetColumnName, EncryptDerivedColumnSuffix derivedColumnSuffix);
197     }
