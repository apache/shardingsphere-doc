18  package org.apache.shardingsphere.mcp.support.workflow.service;
19  
20  import org.junit.jupiter.api.Test;
21  import org.junit.jupiter.params.ParameterizedTest;
22  import org.junit.jupiter.params.provider.Arguments;
23  import org.junit.jupiter.params.provider.MethodSource;
24  
25  import java.util.Map;
26  import java.util.concurrent.atomic.AtomicReference;
27  import java.util.stream.Stream;
28  
29  import static org.hamcrest.MatcherAssert.assertThat;
30  import static org.hamcrest.Matchers.is;
31  
32  class WorkflowPlanningArgumentsTest {
33      
34      @ParameterizedTest(name = "{0}")
35      @MethodSource("getStringArgumentCases")
36      void assertGetStringArgument(final String name, final Map<String, Object> rawArguments, final String expectedValue) {
37          assertThat(new WorkflowPlanningArguments(rawArguments).getStringArgument("name"), is(expectedValue));
38      }
39      
40      @ParameterizedTest(name = "{0}")
41      @MethodSource("getBooleanArgumentCases")
42      void assertGetBooleanArgument(final String name, final Map<String, Object> rawArguments, final Boolean expectedValue) {
43          assertThat(new WorkflowPlanningArguments(rawArguments).getBooleanArgument("enabled"), is(expectedValue));
44      }
45      
46      @Test
47      void assertApplyStringArgument() {
48          AtomicReference<String> actual = new AtomicReference<>("unchanged");
49          new WorkflowPlanningArguments(Map.of("name", " foo_name ")).applyStringArgument("name", actual::set);
50          assertThat(actual.get(), is("foo_name"));
51      }
52      
53      @Test
54      void assertApplyStringArgumentWithMissingArgument() {
55          AtomicReference<String> actual = new AtomicReference<>("unchanged");
56          new WorkflowPlanningArguments(Map.of()).applyStringArgument("name", actual::set);
57          assertThat(actual.get(), is("unchanged"));
58      }
59      
60      @ParameterizedTest(name = "{0}")
61      @MethodSource("getMapArgumentCases")
62      void assertGetMapArgument(final String name, final Map<String, Object> rawArguments, final Map<String, String> expectedValue) {
63          assertThat(new WorkflowPlanningArguments(rawArguments).getMapArgument("props"), is(expectedValue));
64      }
65      
66      @ParameterizedTest(name = "{0}")
67      @MethodSource("getAlgorithmPropertyMapArgumentCases")
68      void assertGetAlgorithmPropertyMapArgument(final String name, final Map<String, Object> rawArguments, final Map<String, String> expectedValue) {
69          assertThat(new WorkflowPlanningArguments(rawArguments).getAlgorithmPropertyMapArgument("props", "primary"), is(expectedValue));
70      }
71      
72      @ParameterizedTest(name = "{0}")
73      @MethodSource("getSecretReferenceMapArgumentCases")
74      void assertGetSecretReferenceMapArgument(final String name, final Map<String, Object> rawArguments, final boolean expectedMalformed) {
75          assertThat(new WorkflowPlanningArguments(rawArguments).getSecretReferenceMapArgument("props").get("aes-key-value").isMalformed(), is(expectedMalformed));
76      }
77      
78      private static Stream<Arguments> getStringArgumentCases() {
79          return Stream.of(
80                  Arguments.of("missing string", Map.of(), ""),
81                  Arguments.of("trimmed string", Map.of("name", " foo_name "), "foo_name"),
82                  Arguments.of("number string", Map.of("name", 42), "42"));
83      }
84      
85      private static Stream<Arguments> getBooleanArgumentCases() {
86          return Stream.of(
87                  Arguments.of("missing boolean", Map.of(), null),
88                  Arguments.of("true boolean", Map.of("enabled", true), true),
89                  Arguments.of("false boolean", Map.of("enabled", false), false));
90      }
91      
92      private static Stream<Arguments> getMapArgumentCases() {
93          return Stream.of(
94                  Arguments.of("missing map", Map.of(), Map.of()),
95                  Arguments.of("map argument", Map.of("props", Map.of("aes-key-value", "123456", "digest-algorithm-name", "SHA-1")),
96                          Map.of("aes-key-value", "123456", "digest-algorithm-name", "SHA-1")));
97      }
98      
99      private static Stream<Arguments> getAlgorithmPropertyMapArgumentCases() {
100         return Stream.of(
101                 Arguments.of("secret reference object", Map.of("props", Map.of("aes-key-value", Map.of("secret_ref", "placeholder://secret-value-1"))),
102                         Map.of("aes-key-value", "secret_reference:primary.aes-key-value")),
103                 Arguments.of("literal map argument", Map.of("props", Map.of("digest-algorithm-name", "SHA-256")), Map.of("digest-algorithm-name", "SHA-256")));
104     }
105     
106     private static Stream<Arguments> getSecretReferenceMapArgumentCases() {
107         return Stream.of(
108                 Arguments.of("valid secret reference", Map.of("props", Map.of("aes-key-value", Map.of("secret_ref", "placeholder://secret-value-1"))), false),
109                 Arguments.of("malformed secret reference", Map.of("props", Map.of("aes-key-value", Map.of("label", "placeholder"))), true));
110     }
111 }
