57  @ExtendWith(MockitoExtension.class)
58  class AlterReadwriteSplittingStorageUnitStatusExecutorTest {
59      
60      private final AlterReadwriteSplittingStorageUnitStatusExecutor executor = (AlterReadwriteSplittingStorageUnitStatusExecutor) TypedSPILoader.getService(
61              DistSQLUpdateExecutor.class, AlterReadwriteSplittingStorageUnitStatusStatement.class);
62      
63      @Mock
64      private ShardingSphereDatabase database;
65      
66      @Mock
67      private ReadwriteSplittingRule rule;
68      
69      @Mock(answer = Answers.RETURNS_DEEP_STUBS)
70      private ContextManager contextManager;
71      
72      @BeforeEach
73      void setUp() {
74          executor.setDatabase(database);
75          executor.setRule(rule);
76      }
77      
78      @ParameterizedTest(name = "{0}")
79      @MethodSource("executeUpdateFailureArguments")
80      void assertExecuteUpdateWithInvalidStatus(final String name, final String configuredRuleName, final Collection<String> readDataSources,
81                                                final Collection<String> disabledDataSourceNames, final boolean enable, final Class<? extends Exception> expectedException) {
82          lenient().when(database.getName()).thenReturn("test_db");
83          setRuleGroup(configuredRuleName, readDataSources, disabledDataSourceNames);
84          assertThrows(expectedException, () -> executor.executeUpdate(new AlterReadwriteSplittingStorageUnitStatusStatement(null, "readwrite_group", "read_ds_0", enable), contextManager));
85      }
86      
87      @ParameterizedTest(name = "{0}")
88      @MethodSource("executeUpdateSuccessArguments")
89      void assertExecuteUpdate(final String name, final String statementRuleName, final String configuredRuleName,
90                               final boolean enable, final Collection<String> disabledDataSourceNames, final DataSourceState expectedState) {
91          when(database.getName()).thenReturn("test_db");
92          QualifiedDataSourceStatePersistService qualifiedDataSourceStateService = mock(QualifiedDataSourceStatePersistService.class);
93          when(contextManager.getPersistServiceFacade().getQualifiedDataSourceStateService()).thenReturn(qualifiedDataSourceStateService);
94          setRuleGroup(configuredRuleName, Collections.singleton("read_ds_0"), disabledDataSourceNames);
95          assertDoesNotThrow(() -> executor.executeUpdate(new AlterReadwriteSplittingStorageUnitStatusStatement(null, statementRuleName, "read_ds_0", enable), contextManager));
96          verify(qualifiedDataSourceStateService).update("test_db", statementRuleName, "read_ds_0", expectedState);
97      }
98      
99      private void setRuleGroup(final String configuredRuleName, final Collection<String> readDataSources, final Collection<String> disabledDataSourceNames) {
100         if (null == configuredRuleName) {
101             when(rule.getDataSourceRuleGroups()).thenReturn(Collections.emptyMap());
102             return;
103         }
104         ReadwriteSplittingDataSourceGroupRule dataSourceGroupRule = mock(ReadwriteSplittingDataSourceGroupRule.class);
105         when(dataSourceGroupRule.getName()).thenReturn(configuredRuleName);
106         ReadwriteSplittingGroup readwriteSplittingGroup = mock(ReadwriteSplittingGroup.class);
107         when(readwriteSplittingGroup.getReadDataSources()).thenReturn(new LinkedList<>(readDataSources));
108         when(dataSourceGroupRule.getReadwriteSplittingGroup()).thenReturn(readwriteSplittingGroup);
109         lenient().when(dataSourceGroupRule.getDisabledDataSourceNames()).thenReturn(Collections.unmodifiableCollection(disabledDataSourceNames));
110         when(rule.getDataSourceRuleGroups()).thenReturn(Collections.singletonMap(configuredRuleName, dataSourceGroupRule));
111     }
112     
113     @Test
114     void assertGetRuleClass() {
115         assertThat(executor.getRuleClass(), is(ReadwriteSplittingRule.class));
116     }
117     
118     private static Stream<Arguments> executeUpdateFailureArguments() {
119         return Stream.of(
120                 Arguments.of("missing readwrite-splitting rule", null, Collections.singleton("read_ds_0"), Collections.emptySet(), true,
121                         MissingRequiredRuleException.class),
122                 Arguments.of("missing read storage unit", "readwrite_group", Collections.singleton("read_ds_1"), Collections.emptySet(), true,
123                         ReadwriteSplittingActualDataSourceNotFoundException.class),
124                 Arguments.of("enable storage unit that is not disabled", "readwrite_group", Collections.singleton("read_ds_0"), Collections.emptySet(), true,
125                         InvalidStorageUnitStatusException.class),
126                 Arguments.of("disable storage unit that is already disabled", "readwrite_group", Collections.singleton("read_ds_0"), Collections.singleton("read_ds_0"), false,
127                         InvalidStorageUnitStatusException.class));
128     }
129     
130     private static Stream<Arguments> executeUpdateSuccessArguments() {
131         return Stream.of(
132                 Arguments.of("enable storage unit", "readwrite_group", "readwrite_group", true, Collections.singleton("read_ds_0"), DataSourceState.ENABLED),
133                 Arguments.of("disable storage unit", "readwrite_group", "readwrite_group", false, Collections.emptySet(), DataSourceState.DISABLED),
134                 Arguments.of("match rule name case insensitively", "readwrite_group", "READWRITE_GROUP", true, Collections.singleton("read_ds_0"), DataSourceState.ENABLED));
135     }
