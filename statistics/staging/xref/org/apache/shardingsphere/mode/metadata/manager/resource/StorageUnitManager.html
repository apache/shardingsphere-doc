18  package org.apache.shardingsphere.mode.metadata.manager.resource;
19  
20  import lombok.RequiredArgsConstructor;
21  import lombok.SneakyThrows;
22  import lombok.extern.slf4j.Slf4j;
23  import org.apache.shardingsphere.infra.datasource.pool.props.domain.DataSourcePoolProperties;
24  import org.apache.shardingsphere.infra.instance.ComputeNodeInstanceContext;
25  import org.apache.shardingsphere.infra.metadata.database.ShardingSphereDatabase;
26  import org.apache.shardingsphere.infra.metadata.database.schema.model.ShardingSphereSchema;
27  import org.apache.shardingsphere.infra.rule.ShardingSphereRule;
28  import org.apache.shardingsphere.mode.metadata.MetaDataContexts;
29  import org.apache.shardingsphere.mode.metadata.factory.MetaDataContextsFactory;
30  import org.apache.shardingsphere.mode.metadata.persist.MetaDataPersistFacade;
31  
32  import java.sql.SQLException;
33  import java.util.Collection;
34  import java.util.Collections;
35  import java.util.Map;
36  import java.util.stream.Collectors;
37  
38  /**
39   * Storage unit manager.
40   */
41  @RequiredArgsConstructor
42  @Slf4j
43  public final class StorageUnitManager {
44      
45      private final MetaDataContexts metaDataContexts;
46      
47      private final ComputeNodeInstanceContext computeNodeInstanceContext;
48      
49      private final ResourceSwitchManager resourceSwitchManager;
50      
51      private final MetaDataPersistFacade metaDataPersistFacade;
52      
53      /**
54       * Register storage unit.
55       *
56       * @param databaseName database name
57       * @param propsMap data source pool properties map
58       */
59      public synchronized void register(final String databaseName, final Map<String, DataSourcePoolProperties> propsMap) {
60          ShardingSphereDatabase database = metaDataContexts.getMetaData().getDatabase(databaseName);
61          try {
62              closeStaleRules(database);
63              SwitchingResource switchingResource = resourceSwitchManager.switchByRegisterStorageUnit(database.getResourceMetaData(), propsMap);
64              buildNewMetaDataContext(databaseName, switchingResource, true);
65          } catch (final SQLException ex) {
66              log.error("Alter database: {} register storage unit failed.", databaseName, ex);
67          }
68      }
69      
70      /**
71       * Alter storage unit.
72       *
73       * @param databaseName database name
74       * @param propsMap data source pool properties map
75       */
76      public synchronized void alter(final String databaseName, final Map<String, DataSourcePoolProperties> propsMap) {
77          ShardingSphereDatabase database = metaDataContexts.getMetaData().getDatabase(databaseName);
78          try {
79              closeStaleRules(database);
80              SwitchingResource switchingResource = resourceSwitchManager.switchByAlterStorageUnit(database.getResourceMetaData(), propsMap);
81              buildNewMetaDataContext(databaseName, switchingResource, true);
82          } catch (final SQLException ex) {
83              log.error("Alter database: {} alter storage unit failed.", databaseName, ex);
84          }
85      }
86      
87      /**
88       * UnRegister storage unit.
89       *
90       * @param databaseName database name
91       * @param storageUnitName storage unit name
92       */
93      public synchronized void unregister(final String databaseName, final String storageUnitName) {
94          ShardingSphereDatabase database = metaDataContexts.getMetaData().getDatabase(databaseName);
95          try {
96              closeStaleRules(database);
97              SwitchingResource switchingResource = resourceSwitchManager.switchByUnregisterStorageUnit(database.getResourceMetaData(), Collections.singleton(storageUnitName));
98              buildNewMetaDataContext(databaseName, switchingResource, false);
99          } catch (final SQLException ex) {
100             log.error("Alter database: {} register storage unit failed.", databaseName, ex);
101         }
102     }
103     
104     private void buildNewMetaDataContext(final String databaseName, final SwitchingResource switchingResource, final boolean isLoadSchemasFromRegisterCenter) throws SQLException {
105         MetaDataContexts reloadMetaDataContexts = new MetaDataContextsFactory(metaDataPersistFacade, computeNodeInstanceContext).createBySwitchResource(
106                 databaseName, isLoadSchemasFromRegisterCenter, switchingResource, metaDataContexts);
107         metaDataContexts.update(reloadMetaDataContexts);
108         metaDataContexts.getMetaData().putDatabase(buildDatabase(reloadMetaDataContexts.getMetaData().getDatabase(databaseName)));
109         switchingResource.closeStaleDataSources();
110     }
111     
112     private ShardingSphereDatabase buildDatabase(final ShardingSphereDatabase originalDatabase) {
113         return new ShardingSphereDatabase(
114                 originalDatabase.getName(), originalDatabase.getProtocolType(), originalDatabase.getResourceMetaData(), originalDatabase.getRuleMetaData(), buildSchemas(originalDatabase));
115     }
116     
117     private Collection<ShardingSphereSchema> buildSchemas(final ShardingSphereDatabase originalDatabase) {
118         return originalDatabase.getAllSchemas().stream().map(each -> buildSchema(originalDatabase, each)).collect(Collectors.toList());
119     }
120     
121     private ShardingSphereSchema buildSchema(final ShardingSphereDatabase originalDatabase, final ShardingSphereSchema schema) {
122         return new ShardingSphereSchema(schema.getName(), schema.getAllTables(), metaDataPersistFacade.getDatabaseMetaDataFacade().getView().load(originalDatabase.getName(), schema.getName()));
123     }
124     
125     @SneakyThrows(Exception.class)
126     private void closeStaleRules(final ShardingSphereDatabase database) {
127         for (ShardingSphereRule each : database.getRuleMetaData().getRules()) {
128             if (each instanceof AutoCloseable) {
129                 ((AutoCloseable) each).close();
130             }
131         }
132     }
133 }
