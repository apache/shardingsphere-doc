227 /**
228  * DDL statement visitor for Doris.
229  */
230 public final class DorisDDLStatementVisitor extends DorisStatementVisitor implements DDLStatementVisitor {
231     
232     public DorisDDLStatementVisitor(final DatabaseType databaseType) {
233         super(databaseType);
234     }
235     
236     @Override
237     public ASTNode visitCreateView(final CreateViewContext ctx) {
238         CreateViewStatement result = new CreateViewStatement(getDatabaseType());
239         result.setReplaceView(null != ctx.REPLACE());
240         result.setView((SimpleTableSegment) visit(ctx.viewName()));
241         result.setViewDefinition(getOriginalText(ctx.select()));
242         result.setSelect((SelectStatement) visit(ctx.select()));
243         return result;
244     }
245     
246     @Override
247     public ASTNode visitAlterView(final AlterViewContext ctx) {
248         AlterViewStatement result = new AlterViewStatement(getDatabaseType());
249         result.setView((SimpleTableSegment) visit(ctx.viewName()));
250         if (null != ctx.columnNames()) {
251             CollectionValue<ColumnSegment> columns = (CollectionValue<ColumnSegment>) visit(ctx.columnNames());
252             for (ColumnSegment each : columns.getValue()) {
253                 result.getColumns().add(new ViewColumnSegment(each.getStartIndex(), each.getStopIndex(), each, null));
254             }
255         }
256         if (null != ctx.viewColumnDefinitions()) {
257             CollectionValue<ViewColumnSegment> columns = (CollectionValue<ViewColumnSegment>) visit(ctx.viewColumnDefinitions());
258             result.getColumns().addAll(columns.getValue());
259         }
260         result.setViewDefinition(getOriginalText(ctx.select()));
261         result.setSelect((SelectStatement) visit(ctx.select()));
262         return result;
263     }
264     
265     @SuppressWarnings("unchecked")
266     @Override
267     public ASTNode visitDropView(final DropViewContext ctx) {
268         DropViewStatement result = new DropViewStatement(getDatabaseType());
269         result.setIfExists(null != ctx.ifExists());
270         result.getViews().addAll(((CollectionValue<SimpleTableSegment>) visit(ctx.viewNames())).getValue());
271         return result;
272     }
273     
274     @Override
275     public ASTNode visitCreateDatabase(final CreateDatabaseContext ctx) {
276         return new CreateDatabaseStatement(getDatabaseType(), new IdentifierValue(ctx.databaseName().getText()).getValue(), null != ctx.ifNotExists());
277     }
278     
279     @Override
280     public ASTNode visitAlterDatabase(final AlterDatabaseContext ctx) {
281         AlterDatabaseStatement result = new AlterDatabaseStatement(getDatabaseType());
282         if (null != ctx.databaseName()) {
283             result.setDatabaseName(new IdentifierValue(ctx.databaseName().getText()).getValue());
284         }
285         if (null != ctx.RENAME() && null != ctx.identifier()) {
286             result.setRenameDatabaseName(new IdentifierValue(ctx.identifier().getText()).getValue());
287         }
288         if (null != ctx.QUOTA() && null != ctx.NUMBER_()) {
289             if (null != ctx.DATA()) {
290                 result.setQuotaType("DATA");
291             } else if (null != ctx.REPLICA()) {
292                 result.setQuotaType("REPLICA");
293             } else if (null != ctx.TRANSACTION()) {
294                 result.setQuotaType("TRANSACTION");
295             }
296             result.setQuotaValue(Long.parseLong(ctx.NUMBER_().getText()));
297         }
298         if (null != ctx.PROPERTIES() && null != ctx.properties()) {
299             PropertiesSegment propertiesSegment = new PropertiesSegment(ctx.properties().getStart().getStartIndex(), ctx.properties().getStop().getStopIndex());
300             for (int i = 0; i < ctx.properties().property().size(); i++) {
301                 String key = getPropertyKey(ctx.properties().property(i));
302                 String value = SQLUtils.getExactlyValue(ctx.properties().property(i).literals().getText());
303                 PropertySegment propertySegment = new PropertySegment(ctx.properties().property(i).getStart().getStartIndex(), ctx.properties().property(i).getStop().getStopIndex(), key, value);
304                 propertiesSegment.getProperties().add(propertySegment);
305             }
306             result.setProperties(propertiesSegment);
307         }
308         return result;
309     }
310     
311     @Override
312     public ASTNode visitDropDatabase(final DropDatabaseContext ctx) {
313         return new DropDatabaseStatement(getDatabaseType(), new IdentifierValue(ctx.databaseName().getText()).getValue(), null != ctx.ifExists());
314     }
315     
316     @Override
317     public ASTNode visitAlterCatalog(final AlterCatalogContext ctx) {
318         AlterCatalogStatement result = new AlterCatalogStatement(getDatabaseType());
319         result.setCatalogName(new IdentifierValue(ctx.catalogName().getText()).getValue());
320         if (null != ctx.RENAME()) {

498     private Optional<AlterDefinitionSegment> getAlterDefinitionSegment(final AlterListContext alterListContext, final AlterListItemContext alterListItemContext) {
499         if (alterListItemContext instanceof AddColumnContext) {
500             return Optional.of((AddColumnDefinitionSegment) visit(alterListItemContext));
501         }
502         if (alterListItemContext instanceof AlterConstraintContext || alterListItemContext instanceof AlterCheckContext) {
503             return Optional.of((AlterDefinitionSegment) visit(alterListItemContext));
504         }
505         if (alterListItemContext instanceof ChangeColumnContext) {
506             return Optional.of(generateModifyColumnDefinitionSegment((ChangeColumnContext) alterListItemContext));
507         }
508         if (alterListItemContext instanceof ModifyColumnContext) {
509             return Optional.of(generateModifyColumnDefinitionSegment((ModifyColumnContext) alterListItemContext));
510         }
511         if (alterListItemContext instanceof AlterTableDropContext) {
512             return getDropItemDefinitionSegment(alterListContext, (AlterTableDropContext) alterListItemContext);
513         }
514         if (alterListItemContext instanceof AddTableConstraintContext) {
515             return Optional.of((AddConstraintDefinitionSegment) visit(alterListItemContext));
516         }
517         if (alterListItemContext instanceof AlterRenameTableContext) {
518             return Optional.of((RenameTableDefinitionSegment) visit(alterListItemContext));
519         }
520         if (alterListItemContext instanceof AlterConvertContext) {
521             return Optional.of((ConvertTableDefinitionSegment) visit(alterListItemContext));
522         }
523         if (alterListItemContext instanceof RenameColumnContext) {
524             return Optional.of((RenameColumnSegment) visit(alterListItemContext));
525         }
526         if (alterListItemContext instanceof RenameIndexContext) {
527             return Optional.of((RenameIndexDefinitionSegment) visit(alterListItemContext));
528         }
529         if (alterListItemContext instanceof RenameRollupContext) {
530             return Optional.of((RenameRollupDefinitionSegment) visit(alterListItemContext));
531         }
532         if (alterListItemContext instanceof RenamePartitionContext) {
533             return Optional.of((RenamePartitionDefinitionSegment) visit(alterListItemContext));

538     private Optional<AlterDefinitionSegment> getDropItemDefinitionSegment(final AlterListContext alterListContext, final AlterTableDropContext alterTableDrop) {
539         if (null != alterTableDrop.CHECK() || null != alterTableDrop.CONSTRAINT()) {
540             ConstraintSegment constraint = new ConstraintSegment(alterTableDrop.identifier().getStart().getStartIndex(), alterTableDrop.identifier().getStop().getStopIndex(),
541                     (IdentifierValue) visit(alterTableDrop.identifier()));
542             return Optional.of(new DropConstraintDefinitionSegment(alterListContext.getStart().getStartIndex(), alterListContext.getStop().getStopIndex(), constraint));
543         }
544         if (null == alterTableDrop.KEY() && null == alterTableDrop.keyOrIndex()) {
545             ColumnSegment column = new ColumnSegment(alterTableDrop.columnInternalRef.start.getStartIndex(), alterTableDrop.columnInternalRef.stop.getStopIndex(),
546                     (IdentifierValue) visit(alterTableDrop.columnInternalRef));
547             return Optional.of(new DropColumnDefinitionSegment(alterTableDrop.getStart().getStartIndex(), alterTableDrop.getStop().getStopIndex(), Collections.singleton(column)));
548         }
549         if (null != alterTableDrop.keyOrIndex()) {
550             return Optional.of(
551                     new DropIndexDefinitionSegment(alterListContext.getStart().getStartIndex(), alterListContext.getStop().getStopIndex(), (IndexSegment) visit(alterTableDrop.indexName())));
552         }
553         return Optional.empty();
554     }

915     @Override
916     public ASTNode visitCreateFunction(final CreateFunctionContext ctx) {
917         DorisCreateFunctionStatement result = new DorisCreateFunctionStatement(getDatabaseType());
918         result.setFunctionName((FunctionNameSegment) visit(ctx.functionName()));
919         if (null != ctx.routineBody()) {
920             int paramIndex = 0;
921             for (int i = 0; i < ctx.dataType().size(); i++) {
922                 DataTypeSegment dataType = (DataTypeSegment) visit(ctx.dataType(i));
923                 if (i == ctx.dataType().size() - 1 && null != ctx.RETURNS()) {
924                     result.setReturnType(dataType);
925                 } else if (paramIndex < ctx.identifier().size()) {
926                     IdentifierValue paramName = (IdentifierValue) visit(ctx.identifier(paramIndex));
927                     result.getNamedParameters().put(paramName, dataType);
928                     paramIndex++;
929                 }
930             }
931             result.setRoutineBody((RoutineBodySegment) visit(ctx.routineBody()));
932         } else {
933             if (null != ctx.GLOBAL()) {
934                 result.setGlobal(true);
935             }
936             if (null != ctx.AGGREGATE()) {
937                 result.setFunctionType(DorisCreateFunctionStatement.FunctionType.AGGREGATE);
938             } else if (null != ctx.TABLES()) {
939                 result.setFunctionType(DorisCreateFunctionStatement.FunctionType.TABLES);
940             } else if (null != ctx.ALIAS()) {
941                 result.setFunctionType(DorisCreateFunctionStatement.FunctionType.ALIAS);
942             }
943             int dataTypeCount = ctx.dataType().size();
944             int parameterCount = dataTypeCount;
945             if (null != ctx.INTERMEDIATE()) {
946                 parameterCount--;
947             }
948             if (null != ctx.RETURNS()) {
949                 parameterCount--;
950             }
951             for (int i = 0; i < parameterCount; i++) {
952                 result.getParameterDataTypes().add((DataTypeSegment) visit(ctx.dataType(i)));
953             }
954             if (null != ctx.RETURNS()) {
955                 result.setReturnType((DataTypeSegment) visit(ctx.dataType(parameterCount)));
956             }
957             if (null != ctx.INTERMEDIATE()) {
958                 int intermediateTypeIndex = dataTypeCount - 1;
959                 result.setIntermediateType((DataTypeSegment) visit(ctx.dataType(intermediateTypeIndex)));
960             }
961             if (null != ctx.PARAMETER()) {
962                 for (int i = 0; i < ctx.identifier().size(); i++) {
963                     result.getWithParameters().add((IdentifierValue) visit(ctx.identifier(i)));
964                 }
965             }
966             if (null != ctx.expr()) {
967                 result.setAliasExpression((ExpressionSegment) visit(ctx.expr()));
968             }
969             if (null != ctx.PROPERTIES()) {
970                 fillCreateFunctionProperties(ctx, result);
971             }
972         }
973         return result;
974     }

977         for (int i = 0; i < ctx.properties().property().size(); i++) {
978             String key = getPropertyKey(ctx.properties().property(i));
979             String value = SQLUtils.getExactlyValue(ctx.properties().property(i).literals().getText());
980             statement.getProperties().put(key, value);
981         }
982     }
