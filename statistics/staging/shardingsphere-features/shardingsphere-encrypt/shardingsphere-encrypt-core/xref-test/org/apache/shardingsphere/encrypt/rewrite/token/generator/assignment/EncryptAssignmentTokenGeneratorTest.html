131     @Test
132     void assertGenerateSQLTokenWithUpdateLiteralExpressionSegment() {
133         ShardingSphereDatabase database = mock(ShardingSphereDatabase.class);
134         when(database.getName()).thenReturn("foo_db");
135         when(database.getDefaultSchemaName()).thenReturn("foo_default_schema");
136         EncryptRule rule = mockEncryptRule();
137         EncryptColumn encryptColumn = rule.findEncryptTable("table").get().getEncryptColumn("columns");
138         when(encryptColumn.getName()).thenReturn("columns");
139         when(encryptColumn.getCipher().getName()).thenReturn("cipher_columns");
140         when(encryptColumn.getCipher().encrypt("foo_db", "foo_default_schema", "table", "columns", Collections.singletonList("foo_value")))
141                 .thenReturn(Collections.singletonList("encrypted_value"));
142         tokenGenerator = new EncryptAssignmentTokenGenerator(rule, database, TypedSPILoader.getService(DatabaseType.class, "FIXTURE"));
143         when(tablesContext.getSchemaName()).thenReturn(Optional.empty());
144         when(assignmentSegment.getValue()).thenReturn(new LiteralExpressionSegment(0, 0, "foo_value"));
145         assertThat(tokenGenerator.generateSQLTokens(tablesContext, setAssignmentSegment).iterator().next().toString(), is("cipher_columns = 'encrypted_value'"));
146     }

164     @ParameterizedTest(name = "{0}")
165     @MethodSource("getEncryptedTupleArguments")
166     void assertGenerateSQLTokensWithEncryptedTuple(final String name, final Collection<String> columnNames, final String encryptedColumnName) {
167         EncryptRule rule = mock(EncryptRule.class);
168         EncryptTable encryptTable = mock(EncryptTable.class);
169         when(rule.findEncryptTable("foo_table")).thenReturn(Optional.of(encryptTable));
170         when(encryptTable.isEncryptColumn(anyString())).thenAnswer(invocation -> encryptedColumnName.equals(invocation.getArgument(0)));
171         when(assignmentSegment.getColumns()).thenReturn(columnNames.stream().map(this::createBoundColumn).collect(Collectors.toList()));
172         tokenGenerator = new EncryptAssignmentTokenGenerator(rule, mock(ShardingSphereDatabase.class), TypedSPILoader.getService(DatabaseType.class, "FIXTURE"));
173         UnsupportedSQLOperationException actualException = assertThrows(UnsupportedSQLOperationException.class, () -> tokenGenerator.generateSQLTokens(tablesContext, setAssignmentSegment));
174         assertThat(actualException.getMessage(), is("Unsupported SQL operation: Encrypted assignments with multiple columns, nested fields or unsupported expressions."));
175     }

247     private static Stream<Arguments> getUnsupportedAssignmentValues() {
248         UnaryOperationExpression unary = new UnaryOperationExpression(0, 1, new ParameterMarkerExpressionSegment(1, 1, 0), "~", "~?");
249         FunctionSegment function = new FunctionSegment(0, 11, "coalesce", "coalesce(~?)");
250         function.getParameters().add(unary);
251         FunctionSegment nestedFunction = new FunctionSegment(0, 21, "coalesce", "coalesce(coalesce(~?))");
252         nestedFunction.getParameters().add(function);
253         FunctionSegment commonFunction = new FunctionSegment(0, 15, "coalesce", "coalesce((?)[1])");
254         commonFunction.getParameters().add(new CommonExpressionSegment(9, 14, "(?)[1]", Collections.singletonList(new ParameterMarkerExpressionSegment(10, 10, 0))));
255         return Stream.of(
256                 Arguments.of("common children", new CommonExpressionSegment(0, 5, "(?)[1]", Collections.singletonList(new ParameterMarkerExpressionSegment(1, 1, 0)))),
257                 Arguments.of("opaque common", new CommonExpressionSegment(0, 6, "DEFAULT", Collections.emptyList())),
258                 Arguments.of("cast unary value", new TypeCastExpression(0, 14, "CAST(~? AS text)", unary, "text")),
259                 Arguments.of("unary value", unary),
260                 Arguments.of("function unary value", function),
261                 Arguments.of("nested function unary value", nestedFunction),
262                 Arguments.of("function common value", commonFunction));
263     }
