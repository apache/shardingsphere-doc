52      @Test
53      void assertMapDatabaseDetail() {
54          MCPMetadataQueryFacade metadataQueryFacade = mock(MCPMetadataQueryFacade.class);
55          when(metadataQueryFacade.querySchemas("logic_db")).thenReturn(List.of(createSchemaMetadata()));
56          when(metadataQueryFacade.isSupportedMetadataObjectType("logic_db", SupportedMCPMetadataObjectType.SEQUENCE)).thenReturn(true);
57          when(metadataQueryFacade.querySequences("logic_db", "public")).thenReturn(List.of(new MCPSequenceMetadata("logic_db", "public", "order_seq")));
58          List<?> actual = new MetadataResourcePayloadMapper(metadataQueryFacade, new MCPResourceURIVariables(Map.of()), true)
59                  .map(createMetadata("logical-database"), List.of(
60                          new RuntimeDatabaseProfile("logic_db", "FixtureDB", "1.0", TransactionCapability.LOCAL_WITH_SAVEPOINT,
61                                  new DatabaseIdentifierContext(IdentifierCasePolicyFactory.newInsensitivePolicySet()))));
62          Map<?, ?> actualDatabase = (Map<?, ?>) actual.getFirst();
63          assertThat(actualDatabase.get("database"), is("logic_db"));
64          assertThat(actualDatabase.get("databaseType"), is("FixtureDB"));
65          Map<?, ?> actualSchema = getFirstMap(actualDatabase, "schemas");
66          assertThat(actualSchema.get("schema"), is("public"));
67          assertThat(getNames(actualSchema, "tables", "table"), is(List.of("order_items", "orders")));
68          assertThat(getNames(actualSchema, "views", "view"), is(List.of("active_orders")));
69          assertThat(getNames(actualSchema, "sequences", "sequence"), is(List.of("order_seq")));
70      }

72      @Test
73      void assertMapSchemaDetailWithoutSequenceSupport() {
74          MCPResourceURIVariables uriVariables = new MCPResourceURIVariables(Map.of("database", "logic_db"));
75          MCPMetadataQueryFacade metadataQueryFacade = mock(MCPMetadataQueryFacade.class);
76          List<?> actual = new MetadataResourcePayloadMapper(metadataQueryFacade, uriVariables, true)
77                  .map(createMetadata("schema"), List.of(createSchemaMetadata()));
78          assertThat(((Map<?, ?>) actual.getFirst()).get("sequences"), is(List.of()));
79          verify(metadataQueryFacade, never()).querySequences("logic_db", "public");
80      }

82      @Test
83      void assertMapSequence() {
84          MCPSequenceMetadata sequence = new MCPSequenceMetadata("logic_db", "public", "order_seq");
85          List<?> actual = new MetadataResourcePayloadMapper(mock(MCPMetadataQueryFacade.class), new MCPResourceURIVariables(Map.of()), false)
86                  .map(createMetadata("sequence"), List.of(sequence));
87          assertThat(actual, is(List.of(Map.of("database", "logic_db", "schema", "public", "sequence", "order_seq"))));
88      }

90      @Test
91      void assertMapTableDetail() {
92          MCPResourceURIVariables uriVariables = new MCPResourceURIVariables(Map.of("database", "logic_db", "schema", "public"));
93          MCPMetadataQueryFacade metadataQueryFacade = mock(MCPMetadataQueryFacade.class);
94          when(metadataQueryFacade.queryTableColumns("logic_db", "public", "orders")).thenReturn(createColumns("orders"));
95          when(metadataQueryFacade.queryIndexes("logic_db", "public", "orders")).thenReturn(createIndexes());
96          List<?> actual = new MetadataResourcePayloadMapper(metadataQueryFacade, uriVariables, true)
97                  .map(createMetadata("table"), List.of(createTable("orders", TableType.TABLE)));
98          Map<?, ?> actualTable = (Map<?, ?>) actual.getFirst();
99          assertThat(actualTable.get("database"), is("logic_db"));
100         assertThat(actualTable.get("schema"), is("public"));
101         assertThat(actualTable.get("table"), is("orders"));
102         assertThat(getNames(actualTable, "columns", "column"), is(List.of("amount", "order_id")));
103         assertThat(getNames(actualTable, "indexes", "index"), is(List.of("idx_amount", "idx_order_id")));
104         Map<?, ?> firstColumn = getFirstMap(actualTable, "columns");
105         assertThat(firstColumn.get("ordinalPosition"), is(2));
106         assertThat(firstColumn.get("jdbcType"), is(Types.DECIMAL));
107         assertThat(firstColumn.get("nativeTypeName"), is("DECIMAL"));
108         assertThat(firstColumn.get("nullability"), is("nullable"));
109         Map<?, ?> firstIndex = getFirstMap(actualTable, "indexes");
110         assertThat(firstIndex.get("columns"), is(List.of("amount")));
111         assertFalse((Boolean) firstIndex.get("unique"));
112     }

114     @Test
115     void assertMapViewDetail() {
116         MCPResourceURIVariables uriVariables = new MCPResourceURIVariables(Map.of("database", "logic_db", "schema", "public"));
117         MCPMetadataQueryFacade metadataQueryFacade = mock(MCPMetadataQueryFacade.class);
118         when(metadataQueryFacade.queryViewColumns("logic_db", "public", "active_orders")).thenReturn(createColumns("active_orders"));
119         List<?> actual = new MetadataResourcePayloadMapper(metadataQueryFacade, uriVariables, true)
120                 .map(createMetadata("view"), List.of(createTable("active_orders", TableType.VIEW)));
121         Map<?, ?> actualView = (Map<?, ?>) actual.getFirst();
122         assertThat(actualView.get("database"), is("logic_db"));
123         assertThat(actualView.get("schema"), is("public"));
124         assertThat(actualView.get("view"), is("active_orders"));
125         assertThat(getNames(actualView, "columns", "column"), is(List.of("amount", "order_id")));
126     }

128     private ShardingSphereMCPResourceMetadata createMetadata(final String objectScope) {
129         return new ShardingSphereMCPResourceMetadata("", List.of(), "detail", objectScope, "", List.of(), List.of(), List.of());
130     }
131     
132     private ShardingSphereSchema createSchemaMetadata() {
133         return new ShardingSphereSchema("public", mock(DatabaseType.class), List.of(
134                 createTable("orders", TableType.TABLE),
135                 createTable("order_items", TableType.TABLE),
136                 createTable("active_orders", TableType.VIEW)), List.of());
137     }
138     
139     private ShardingSphereTable createTable(final String name, final TableType tableType) {
140         return new ShardingSphereTable(name, List.of(), List.of(), List.of(), tableType);
141     }
142     
143     private List<MCPColumnMetadata> createColumns(final String relationName) {
144         return List.of(
145                 new MCPColumnMetadata(relationName, "amount", 2, Types.DECIMAL, "DECIMAL", Nullability.NULLABLE),
146                 new MCPColumnMetadata(relationName, "order_id", 1, Types.BIGINT, "BIGINT", Nullability.NOT_NULLABLE));
147     }
148     
149     private List<ShardingSphereIndex> createIndexes() {
150         return List.of(new ShardingSphereIndex("idx_amount", List.of("amount"), false), new ShardingSphereIndex("idx_order_id", List.of("order_id"), true));
151     }
152     
153     private Map<?, ?> getFirstMap(final Map<?, ?> payload, final String key) {
154         return (Map<?, ?>) ((List<?>) payload.get(key)).getFirst();
155     }
156     
157     private List<String> getNames(final Map<?, ?> payload, final String collectionKey, final String nameKey) {
158         return ((List<?>) payload.get(collectionKey)).stream().map(each -> String.valueOf(((Map<?, ?>) each).get(nameKey))).toList();
159     }
