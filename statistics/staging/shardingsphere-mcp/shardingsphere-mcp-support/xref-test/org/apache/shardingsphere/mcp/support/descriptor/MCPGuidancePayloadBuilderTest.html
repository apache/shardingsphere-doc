22  import java.util.Collection;
23  import java.util.List;

26  import static org.hamcrest.MatcherAssert.assertThat;
27  import static org.hamcrest.Matchers.is;
28  import static org.junit.jupiter.api.Assertions.assertFalse;
29  import static org.junit.jupiter.api.Assertions.assertTrue;

31  class MCPGuidancePayloadBuilderTest {
32      
33      @Test
34      void assertBuild() {
35          Map<String, Object> actual = MCPGuidancePayloadBuilder.build();
36          assertThat(actual.get("response_mode"), is("guidance"));
37          assertTrue(actual.containsKey("discovery"));
38          assertTrue(actual.containsKey("model_contract"));
39          assertTrue(actual.containsKey("security_hints"));
40          assertFalse(actual.containsKey("model_first_summary"));
41          assertFalse(actual.containsKey("surface_summary"));
42          assertFalse(actual.containsKey("field_naming_contract"));
43      }
44      
45      @Test
46      void assertCreateDiscovery() {
47          Map<String, Object> actual = MCPGuidancePayloadBuilder.createDiscovery();
48          assertThat(castToMap(actual.get("official_discovery_methods")), is(createOfficialDiscoveryMethods()));
49          assertThat(actual.get("argument_completion_method"), is("completion/complete"));
50          assertThat(actual.get("guidance_resource_role"),
51                  is("shardingsphere://guidance complements MCP list methods with ShardingSphere domain guidance, workflow guidance, and side-effect notes."));
52      }
53      
54      @Test
55      void assertCreateModelContract() {
56          Map<String, Object> actual = MCPGuidancePayloadBuilder.createModelContract();
57          assertThat(actual.get("metadata_first_resource"), is("shardingsphere://databases"));
58          assertTrue(String.valueOf(actual.get("preflight_rule")).contains("database_gateway_validate_runtime_database"));
59          Map<?, ?> actualSqlToolSelection = castToMap(actual.get("sql_tool_selection"));
60          assertThat(actualSqlToolSelection.keySet().stream().toList(), is(List.of("read_only", "explain", "side_effecting")));
61          assertThat(actual.get("side_effect_rule"), is("Preview before side effects and continue only when the requested side effect is still intended."));
62          assertThat(actual.get("completion_rule"),
63                  is("Use completion/complete for one uncertain argument at a time; when completion reports missing context, follow meta.next_actions before guessing."));
64          assertThat(actual.get("resource_template_rule"),
65                  is("Use resources/templates/list to discover URI variables, then read the nearest concrete resource before filling dependent completion context."));
66          assertThat(actual.get("detail_resource_rule"), is("Use resource descriptors, outputSchema, and returned payload keys before assuming detail fields."));
67          assertThat(actual.get("recovery_rule"), is("When a call fails, follow top-level next_actions before inventing a new call."));
68          assertThat(actual.get("payload_field_rule"), is("ShardingSphere-owned structured payload fields use snake_case."));
69          assertFalse(actual.containsKey("official_discovery_methods"));
70      }
71      
72      @Test
73      void assertCreateNextActionContract() {
74          List<Map<String, Object>> actual = MCPGuidancePayloadBuilder.createNextActionContract();
75          assertThat(actual.size(), is(5));
76          Map<?, ?> actualToolCall = findByType(actual, "tool_call");
77          assertTrue(((Collection<?>) actualToolCall.get("required_fields")).contains("tool_name"));
78          Map<?, ?> actualCompletion = findByType(actual, "completion");
79          assertTrue(((Collection<?>) actualCompletion.get("required_fields")).contains("ref"));
80          assertTrue(((Collection<?>) actualCompletion.get("required_fields")).contains("argument"));
81          assertTrue(((Collection<?>) actualCompletion.get("optional_fields")).contains("resume_ref"));
82          assertTrue(((Collection<?>) actualCompletion.get("optional_fields")).contains("resume_arguments"));
83      }
84      
85      @Test
86      void assertCreateCommonFlows() {
87          List<Map<String, Object>> actual = MCPGuidancePayloadBuilder.createCommonFlows();
88          Map<?, ?> actualInspectMetadata = findByKey(actual, "flow_id", "inspect_metadata");
89          assertTrue(((Collection<?>) actualInspectMetadata.get("steps")).contains("resources/list"));
90          Map<?, ?> actualValidateRuntimeDatabase = findByKey(actual, "flow_id", "validate_runtime_database");
91          assertTrue(((Collection<?>) actualValidateRuntimeDatabase.get("steps")).contains("call_tool database_gateway_validate_runtime_database"));
92          Map<?, ?> actualExplainQuery = findByKey(actual, "flow_id", "explain_query");
93          assertTrue(((Collection<?>) actualExplainQuery.get("steps")).contains("call_tool database_gateway_execute_explain_query"));
94          Map<?, ?> actualCompleteArgument = findByKey(actual, "flow_id", "complete_uncertain_argument");
95          assertTrue(((Collection<?>) actualCompleteArgument.get("steps")).contains("call completion/complete for one argument"));
96          Map<?, ?> actualSideEffectingSql = findByKey(actual, "flow_id", "side_effecting_sql");
97          assertTrue(((Collection<?>) actualSideEffectingSql.get("steps")).contains("call_tool database_gateway_execute_update execution_mode=preview"));
98          Map<?, ?> actualWorkflow = findByKey(actual, "flow_id", "workflow_plan_apply_validate");
99          assertTrue(((Collection<?>) actualWorkflow.get("steps")).contains(
100                 "call_tool database_gateway_apply_workflow execution_mode=review-then-execute approved_steps=<preview_artifacts.approval_step>"));
101         Map<?, ?> actualRecovery = findByKey(actual, "flow_id", "recover_from_error");
102         assertThat(((Collection<?>) actualRecovery.get("steps")).iterator().next(), is("follow top-level next_actions"));
103     }
104     
105     @Test
106     void assertCreateSecurityHints() {
107         Map<String, Object> actual = MCPGuidancePayloadBuilder.createSecurityHints();
108         assertTrue(String.valueOf(actual.get("http_transport")).contains("unauthenticated by default"));
109         assertTrue(String.valueOf(actual.get("origin_header")).contains("loopback origins"));
110         Map<?, ?> actualClientSafetyPolicy = castToMap(actual.get("client_safety_policy"));
111         assertThat(actualClientSafetyPolicy.get("identity_scope"), is("mcp_session"));
112         assertTrue(String.valueOf(actualClientSafetyPolicy.get("transport_scope")).contains("trusted session attribution"));
113         assertThat(castToMap(castToMap(actualClientSafetyPolicy.get("runtime_protection")).get("tool_call_limit")).get("scope"), is("session"));
114         assertTrue(String.valueOf(actualClientSafetyPolicy.get("external_model_boundary")).contains("never calls external model providers"));
115     }
116     
117     private Map<?, ?> findByType(final List<Map<String, Object>> values, final String type) {
118         return findByKey(values, "type", type);
119     }
120     
121     private Map<?, ?> findByKey(final List<Map<String, Object>> values, final String key, final String value) {
122         return values.stream().filter(each -> value.equals(each.get(key))).findFirst().orElseThrow();
123     }
124     
125     private Map<?, ?> castToMap(final Object value) {
126         return (Map<?, ?>) value;
127     }
128     
129     private Map<String, Object> createOfficialDiscoveryMethods() {
130         return Map.of("tools", "tools/list", "resources", "resources/list", "resource_templates", "resources/templates/list", "prompts", "prompts/list");
131     }
