18  package org.apache.shardingsphere.mode.repository.standalone.jdbc.sql;
19  
20  import lombok.AccessLevel;
21  import lombok.NoArgsConstructor;
22  import lombok.SneakyThrows;
23  import org.apache.shardingsphere.infra.util.directory.ClasspathResourceDirectoryReader;
24  
25  import javax.xml.XMLConstants;
26  import javax.xml.stream.XMLInputFactory;
27  import javax.xml.stream.XMLStreamConstants;
28  import javax.xml.stream.XMLStreamException;
29  import javax.xml.stream.XMLStreamReader;
30  import java.io.IOException;
31  import java.io.InputStream;
32  import java.util.Objects;
33  import java.util.stream.Stream;
34  
35  /**
36   * JDBC repository SQL Loader.
37   */
38  @NoArgsConstructor(access = AccessLevel.PRIVATE)
39  public final class JDBCRepositorySQLLoader {
40      
41      private static final String ROOT_DIRECTORY = "sql";
42      
43      private static final String FILE_EXTENSION = ".xml";
44      
45      /**
46       * Load JDBC repository SQL.
47       *
48       * @param type type of JDBC repository SQL
49       * @return loaded JDBC repository SQL
50       */
51      @SneakyThrows(IOException.class)
52      public static JDBCRepositorySQL load(final String type) {
53          JDBCRepositorySQL result = null;
54          XMLInputFactory factory = createXMLInputFactory();
55          try (Stream<String> resourceNameStream = ClasspathResourceDirectoryReader.read(JDBCRepositorySQLLoader.class.getClassLoader(), ROOT_DIRECTORY)) {
56              Iterable<String> resourceNameIterable = resourceNameStream::iterator;
57              for (String each : resourceNameIterable) {
58                  if (!each.endsWith(FILE_EXTENSION)) {
59                      continue;
60                  }
61                  JDBCRepositorySQL provider;
62                  try (InputStream inputStream = Objects.requireNonNull(JDBCRepositorySQLLoader.class.getClassLoader().getResourceAsStream(each))) {
63                      provider = parse(factory, inputStream);
64                  }
65                  if (provider.isDefault()) {
66                      result = provider;
67                  }
68                  if (Objects.equals(provider.getType(), type)) {
69                      result = provider;
70                      break;
71                  }
72              }
73          }
74          return result;
75      }
76      
77      private static XMLInputFactory createXMLInputFactory() {
78          XMLInputFactory result = XMLInputFactory.newFactory();
79          result.setProperty(XMLInputFactory.SUPPORT_DTD, false);
80          result.setProperty(XMLInputFactory.IS_SUPPORTING_EXTERNAL_ENTITIES, false);
81          if (result.isPropertySupported(XMLConstants.ACCESS_EXTERNAL_DTD)) {
82              result.setProperty(XMLConstants.ACCESS_EXTERNAL_DTD, "");
83          }
84          result.setXMLResolver((publicId, systemId, baseUri, namespace) -> {
85              throw new XMLStreamException("External XML resources are not allowed.");
86          });
87          return result;
88      }
89      
90      private static JDBCRepositorySQL parse(final XMLInputFactory factory, final InputStream inputStream) throws IOException {
91          try {
92              XMLStreamReader reader = factory.createXMLStreamReader(inputStream);
93              try {
94                  return parse(reader);
95              } finally {
96                  reader.close();
97              }
98          } catch (final XMLStreamException ex) {
99              throw new IOException("Failed to parse JDBC repository SQL XML.", ex);
100         }
101     }
102     
103     private static JDBCRepositorySQL parse(final XMLStreamReader reader) throws XMLStreamException {
104         while (reader.hasNext()) {
105             int event = reader.next();
106             if (XMLStreamConstants.DTD == event) {
107                 throw new XMLStreamException("DTD is not allowed.");
108             }
109             if (XMLStreamConstants.START_ELEMENT == event) {
110                 validateRootElement(reader);
111                 JDBCRepositorySQL result = createJDBCRepositorySQL(reader);
112                 parseSQLElements(reader, result);
113                 return result;
114             }
115         }
116         throw new XMLStreamException("JDBC repository SQL root element is missing.");
117     }
118     
119     private static void validateRootElement(final XMLStreamReader reader) throws XMLStreamException {
120         if (!"sql".equals(reader.getLocalName())) {
121             throw new XMLStreamException("Unexpected JDBC repository SQL root element: " + reader.getLocalName());
122         }
123         for (int i = 0; i < reader.getAttributeCount(); i++) {
124             String attributeName = reader.getAttributeLocalName(i);
125             if (!"type".equals(attributeName) && !"driver-class-name".equals(attributeName) && !"default".equals(attributeName)) {
126                 throw new XMLStreamException("Unexpected JDBC repository SQL attribute: " + attributeName);
127             }
128         }
129     }
130     
131     private static JDBCRepositorySQL createJDBCRepositorySQL(final XMLStreamReader reader) {
132         JDBCRepositorySQL result = new JDBCRepositorySQL();
133         result.setType(reader.getAttributeValue(null, "type"));
134         result.setDriverClassName(reader.getAttributeValue(null, "driver-class-name"));
135         result.setDefault(Boolean.parseBoolean(reader.getAttributeValue(null, "default")));
136         return result;
137     }
138     
139     private static void parseSQLElements(final XMLStreamReader reader, final JDBCRepositorySQL result) throws XMLStreamException {
140         while (reader.hasNext()) {
141             int event = reader.next();
142             if (XMLStreamConstants.DTD == event) {
143                 throw new XMLStreamException("DTD is not allowed.");
144             }
145             if (XMLStreamConstants.START_ELEMENT == event) {
146                 parseSQLElement(reader, result);
147             } else if (XMLStreamConstants.END_ELEMENT == event) {
148                 validateDocumentEnd(reader);
149                 return;
150             }
151         }
152         throw new XMLStreamException("JDBC repository SQL root element is not closed.");
153     }
154     
155     private static void parseSQLElement(final XMLStreamReader reader, final JDBCRepositorySQL result) throws XMLStreamException {
156         String elementName = reader.getLocalName();
157         if (0 != reader.getAttributeCount()) {
158             throw new XMLStreamException("Unexpected attribute on JDBC repository SQL element: " + elementName);
159         }
160         switch (elementName) {
161             case "create-table":
162                 result.setCreateTableSQL(reader.getElementText());
163                 break;
164             case "select-by-key":
165                 result.setSelectByKeySQL(reader.getElementText());
166                 break;
167             case "select-by-parent":
168                 result.setSelectByParentKeySQL(reader.getElementText());
169                 break;
170             case "insert":
171                 result.setInsertSQL(reader.getElementText());
172                 break;
173             case "update":
174                 result.setUpdateSQL(reader.getElementText());
175                 break;
176             case "delete":
177                 result.setDeleteSQL(reader.getElementText());
178                 break;
179             default:
180                 throw new XMLStreamException("Unexpected JDBC repository SQL element: " + elementName);
181         }
182     }
183     
184     private static void validateDocumentEnd(final XMLStreamReader reader) throws XMLStreamException {
185         while (reader.hasNext()) {
186             int event = reader.next();
187             if (XMLStreamConstants.START_ELEMENT == event || XMLStreamConstants.DTD == event) {
188                 throw new XMLStreamException("Unexpected content after JDBC repository SQL root element.");
189             }
190             if (XMLStreamConstants.CHARACTERS == event && !reader.isWhiteSpace()) {
191                 throw new XMLStreamException("Unexpected text after JDBC repository SQL root element.");
192             }
193         }
194     }
