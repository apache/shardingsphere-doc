82      @ParameterizedTest(name = "{0}")
83      @MethodSource("decodeArguments")
84      void assertDecode(final String name, final int payloadLength, final int readableBytes, final int expectedOutSize) {
85          when(byteBuf.markReaderIndex()).thenReturn(byteBuf);
86          when(byteBuf.readUnsignedMediumLE()).thenReturn(payloadLength);
87          when(byteBuf.readableBytes()).thenReturn(readableBytes);
88          List<Object> out = new LinkedList<>();
89          new MySQLPacketCodecEngine().decode(context, byteBuf, out);
90          assertThat(out.size(), is(expectedOutSize));
91      }
92      
93      @Test
94      void assertDecodePacketMoreThan16MB() {
95          MySQLPacketCodecEngine engine = new MySQLPacketCodecEngine();
96          when(context.alloc().compositeBuffer(3)).thenReturn(new CompositeByteBuf(UnpooledByteBufAllocator.DEFAULT, false, 3));
97          List<Object> actual = new ArrayList<>(1);
98          for (ByteBuf each : preparePacketMoreThan16MB()) {
99              engine.decode(context, each, actual);
100         }
101         assertThat(actual.size(), is(1));
102         assertThat(((ByteBuf) actual.get(0)).readableBytes(), is(1 << 24));
103     }
104     
105     @Test
106     void assertDecodePacketMoreThan32MB() {
107         MySQLPacketCodecEngine engine = new MySQLPacketCodecEngine();
108         List<ByteBuf> packets = preparePacketMoreThan32MB();
109         List<Object> actual = new ArrayList<>(1);
110         engine.decode(context, packets.get(0), actual);
111         engine.decode(context, packets.get(1), actual);
112         when(context.alloc().compositeBuffer(4)).thenReturn(new CompositeByteBuf(UnpooledByteBufAllocator.DEFAULT, false, 4));
113         engine.decode(context, packets.get(2), actual);
114         assertThat(actual.size(), is(1));
115         assertThat(((ByteBuf) actual.get(0)).readableBytes(), is(1 << 25));
116     }

118     private List<ByteBuf> preparePacketMoreThan16MB() {
119         byte[] firstPacketData = new byte[4 + (1 << 24) - 1];
120         firstPacketData[0] = firstPacketData[1] = firstPacketData[2] = (byte) 0xff;
121         firstPacketData[3] = (byte) 0;
122         byte[] secondPacketData = new byte[]{0x00, 0x00, 0x00, 0x01};
123         return Arrays.asList(Unpooled.wrappedBuffer(firstPacketData), Unpooled.wrappedBuffer(secondPacketData));
124     }
125     
126     private List<ByteBuf> preparePacketMoreThan32MB() {
127         byte[] firstPacketData = new byte[4 + (1 << 24) - 1];
128         firstPacketData[0] = firstPacketData[1] = firstPacketData[2] = (byte) 0xff;
129         byte[] secondPacketData = new byte[4 + (1 << 24) - 1];
130         secondPacketData[0] = secondPacketData[1] = secondPacketData[2] = (byte) 0xff;
131         byte[] thirdPacketData = new byte[]{0x01, 0x00, 0x00, 0x02, 0x01};
132         return Arrays.asList(Unpooled.wrappedBuffer(firstPacketData), Unpooled.wrappedBuffer(secondPacketData), Unpooled.wrappedBuffer(thirdPacketData));
133     }

135     @Test
136     void assertEncode() {
137         when(byteBuf.writeInt(anyInt())).thenReturn(byteBuf);
138         when(byteBuf.markWriterIndex()).thenReturn(byteBuf);
139         when(byteBuf.readableBytes()).thenReturn(8);
140         MySQLPacket actualMessage = mock(MySQLPacket.class);
141         context.channel().attr(MySQLConstants.SEQUENCE_ID_ATTRIBUTE_KEY).get().set(1);
142         new MySQLPacketCodecEngine().encode(context, actualMessage, byteBuf);
143         verify(byteBuf).writeInt(0);
144         verify(byteBuf).markWriterIndex();
145         ArgumentCaptor<MySQLPacketPayload> payloadCaptor = ArgumentCaptor.forClass(MySQLPacketPayload.class);
146         verify(actualMessage).write(payloadCaptor.capture());
147         assertThat(payloadCaptor.getValue().getCharset(), is(StandardCharsets.UTF_8));
148         verify(byteBuf).setMediumLE(0, 4);
149         verify(byteBuf).setByte(3, 1);
150     }

165     @Test
166     void assertEncodePacketMoreThan16MB() {
167         CompositeByteBuf expected = new CompositeByteBuf(UnpooledByteBufAllocator.DEFAULT, false, 6);
168         when(context.alloc().compositeBuffer(6)).thenReturn(expected);
169         when(context.alloc().ioBuffer(4, 4)).thenReturn(Unpooled.buffer(4, 4), Unpooled.buffer(4, 4), Unpooled.buffer(4, 4));
170         MySQLPacketCodecEngine engine = new MySQLPacketCodecEngine();
171         engine.encode(context, mock(MySQLPacket.class), prepareByteBufAlmost32MB());
172         verify(context).write(expected);
173         assertThat(expected.numComponents(), is(5));
174         assertThat(expected.component(0).getUnsignedByte(3), is((short) 0));
175         assertThat(expected.component(1).readableBytes(), is(0xFFFFFF));
176         assertThat(expected.component(2).getUnsignedByte(3), is((short) 1));
177         assertThat(expected.component(3).readableBytes(), is(0xFFFFFF));
178         assertThat(expected.component(4).getUnsignedByte(3), is((short) 2));
179     }
180     
181     private ByteBuf prepareByteBufAlmost32MB() {
182         int max = 0xFFFFFF << 1;
183         ByteBuf result = Unpooled.buffer(max + 4);
184         result.writerIndex(max);
185         return result.retain();
186     }

188     @Test
189     void assertEncodeOccursException() {
190         when(byteBuf.writeInt(anyInt())).thenReturn(byteBuf);
191         when(byteBuf.markWriterIndex()).thenReturn(byteBuf);
192         when(byteBuf.readableBytes()).thenReturn(12);
193         RuntimeException ex = mock(RuntimeException.class);
194         MySQLPacket actualMessage = mock(MySQLPacket.class);
195         doThrow(ex).when(actualMessage).write(any(MySQLPacketPayload.class));
196         context.channel().attr(MySQLConstants.SEQUENCE_ID_ATTRIBUTE_KEY).get().set(2);
197         new MySQLPacketCodecEngine().encode(context, actualMessage, byteBuf);
198         verify(byteBuf).writeInt(0);
199         verify(byteBuf).markWriterIndex();
200         verify(byteBuf).resetWriterIndex();
201         verify(byteBuf).setMediumLE(0, 8);
202         verify(byteBuf).setByte(3, 2);
203     }
