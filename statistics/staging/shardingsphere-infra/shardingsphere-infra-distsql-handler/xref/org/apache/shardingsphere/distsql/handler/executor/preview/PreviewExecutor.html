53  /**
54   * Preview executor.
55   */
56  @Setter
57  public final class PreviewExecutor implements DistSQLQueryExecutor<PreviewStatement>, DistSQLExecutorConnectionContextAware {
58      
59      private DistSQLConnectionContext connectionContext;
60      
61      @Override
62      public Collection<String> getColumnNames(final PreviewStatement sqlStatement) {
63          return Arrays.asList("data_source_name", "actual_sql");
64      }
65      
66      @Override
67      public Collection<LocalDataQueryResultRow> getRows(final PreviewStatement sqlStatement, final ContextManager contextManager) {
68          ShardingSphereMetaData metaData = contextManager.getMetaDataContexts().getMetaData();
69          String toBePreviewedSQL = sqlStatement.getSql();
70          SQLStatement toBePreviewedStatement =
71                  metaData.getGlobalRuleMetaData().getSingleRule(SQLParserRule.class).getSQLParserEngine(connectionContext.getProtocolType()).parse(toBePreviewedSQL, false);
72          HintValueContext hintValueContext = connectionContext.getQueryContext().getHintValueContext();
73          hintValueContext.setSkipMetadataValidate(true);
74          String currentDatabaseName = connectionContext.getQueryContext().getConnectionContext().getCurrentDatabaseName().orElse(null);
75          SQLStatementContext toBePreviewedStatementContext = new SQLBindEngine(metaData, currentDatabaseName, hintValueContext).bind(toBePreviewedStatement);
76          if (toBePreviewedStatementContext.getSqlStatement().getAttributes().findAttribute(CursorSQLStatementAttribute.class).isPresent()
77                  && toBePreviewedStatementContext instanceof CursorHeldSQLStatementContext) {
78              setUpCursorDefinition((CursorHeldSQLStatementContext) toBePreviewedStatementContext);
79          }
80          QueryContext queryContext = new QueryContext(
81                  toBePreviewedStatementContext, toBePreviewedSQL, Collections.emptyList(), hintValueContext, connectionContext.getQueryContext().getConnectionContext(), metaData);
82          ShardingSphereDatabase database = queryContext.getUsedDatabaseNames().size() > 1 ? contextManager.getDatabase(currentDatabaseName) : queryContext.getUsedDatabase();
83          ShardingSpherePreconditions.checkState(database.isComplete(), () -> new EmptyRuleException(database.getName()));
84          Collection<ExecutionUnit> executionUnits = getExecutionUnits(contextManager, metaData, queryContext, database);
85          return executionUnits.stream().map(each -> new LocalDataQueryResultRow(each.getDataSourceName(), each.getSqlUnit().getSql())).collect(Collectors.toList());
86      }
87      
88      private void setUpCursorDefinition(final CursorHeldSQLStatementContext toBePreviewedStatementContext) {
89          Optional<CursorNameSegment> cursorNameSegment = toBePreviewedStatementContext.getSqlStatement().getAttributes().getAttribute(CursorSQLStatementAttribute.class).getCursorName();
90          if (!cursorNameSegment.isPresent()) {
91              return;
92          }
93          String cursorName = cursorNameSegment.get().getIdentifier().getValue().toLowerCase();
94          CursorStatementContext cursorStatementContext = connectionContext.getQueryContext().getConnectionContext().getCursorContext().getCursorStatementContexts().get(cursorName);
95          Preconditions.checkNotNull(cursorStatementContext, "Cursor %s does not exist.", cursorName);
96          toBePreviewedStatementContext.setCursorStatementContext(cursorStatementContext);
97      }
98      
99      private Collection<ExecutionUnit> getExecutionUnits(final ContextManager contextManager, final ShardingSphereMetaData metaData, final QueryContext queryContext,
100                                                         final ShardingSphereDatabase database) {
101         for (PreviewExecutionUnitGenerator each : ShardingSphereServiceLoader.getServiceInstances(PreviewExecutionUnitGenerator.class)) {
102             Optional<Collection<ExecutionUnit>> result = each.generate(database, queryContext, contextManager, connectionContext);
103             if (result.isPresent()) {
104                 return result.get();
105             }
106         }
107         return new KernelProcessor().generateExecutionContext(queryContext, metaData.getGlobalRuleMetaData(), metaData.getProps()).getExecutionUnits();
108     }
109     
110     @Override
111     public Class<PreviewStatement> getType() {
112         return PreviewStatement.class;
113     }
114 }
