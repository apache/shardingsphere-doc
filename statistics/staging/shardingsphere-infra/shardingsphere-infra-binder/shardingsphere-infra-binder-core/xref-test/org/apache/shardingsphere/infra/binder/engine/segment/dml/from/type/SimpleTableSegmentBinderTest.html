88      @ParameterizedTest(name = "alias={0}")
89      @ValueSource(strings = {"", "gp_alias", "\"GP Alias\""})
90      void assertBindExternalTableProjectionOwner(final String alias) {
91          IdentifierValue tableName = new IdentifierValue("cte_orders");
92          IdentifierValue expectedOwner = alias.isEmpty() ? tableName : new IdentifierValue(alias);
93          SimpleTableSegment segment = new SimpleTableSegment(new TableNameSegment(0, 9, tableName));
94          if (!alias.isEmpty()) {
95              segment.setAlias(new AliasSegment(11, 11 + alias.length() - 1, expectedOwner));
96          }
97          SQLStatementBinderContext context = new SQLStatementBinderContext(createMetaData(), "foo_db", new HintValueContext(),
98                  SelectStatement.builder().databaseType(databaseType).build(), Collections.emptyList());
99          SimpleTableSegmentBinderContext externalContext = new SimpleTableSegmentBinderContext(Collections.emptyList(), TableSourceType.TEMPORARY_TABLE);
100         externalContext.setFromWithSegment(true);
101         context.getExternalTableBinderContexts().put(CaseInsensitiveString.of(tableName.getValue()), externalContext);
102         ProjectionSegment projection = new ColumnProjectionSegment(new ColumnSegment(0, 6, new IdentifierValue("user_id")));
103         Multimap<CaseInsensitiveString, TableSegmentBinderContext> tableBinderContexts = LinkedHashMultimap.create();
104         try (MockedStatic<SubqueryTableBindUtils> ignored = mockStatic(SubqueryTableBindUtils.class)) {
105             ignored.when(() -> SubqueryTableBindUtils.createSubqueryProjections(anyCollection(), eq(expectedOwner), eq(databaseType), eq(TableSourceType.TEMPORARY_TABLE)))
106                     .thenReturn(Collections.singletonList(projection));
107             SimpleTableSegmentBinder.bind(segment, context, tableBinderContexts);
108             SimpleTableSegmentBinderContext actual = (SimpleTableSegmentBinderContext) tableBinderContexts.get(CaseInsensitiveString.of(expectedOwner.getValue())).iterator().next();
109             assertThat(actual.getProjectionSegments().iterator().next(), is(projection));
110             assertTrue(actual.isFromWithSegment());
111         }
112     }
