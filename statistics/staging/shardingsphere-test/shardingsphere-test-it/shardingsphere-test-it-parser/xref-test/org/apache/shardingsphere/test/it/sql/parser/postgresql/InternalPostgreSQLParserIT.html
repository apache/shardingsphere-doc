88      @ParameterizedTest(name = "{0}")
89      @MethodSource("getQuotedIdentifierCases")
90      void assertQuotedIdentifierRoundTrip(final String name, final String identifier, final String expectedValue, final String expectedQuotedValue) {
91          String sql = "SELECT " + identifier + " AS " + identifier + " FROM " + identifier + "." + identifier + " AS " + identifier;
92          SelectStatement actual = (SelectStatement) new SQLStatementVisitorEngine("PostgreSQL").visit(new SQLParserEngine("PostgreSQL", new CacheOption(128, 1024L)).parse(sql, false));
93          ColumnProjectionSegment actualProjection = (ColumnProjectionSegment) actual.getProjections().getProjections().iterator().next();
94          SimpleTableSegment actualTable = (SimpleTableSegment) actual.getFrom().get();
95          ColumnSegment actualColumn = actualProjection.getColumn();
96          assertThat(sql.substring(actualColumn.getStartIndex(), actualColumn.getStopIndex() + 1), is(identifier));
97          assertThat(sql.substring(actualTable.getTableName().getStartIndex(), actualTable.getTableName().getStopIndex() + 1), is(identifier));
98          Collection<IdentifierValue> actualIdentifiers = Arrays.asList(actualColumn.getIdentifier(), actualProjection.getAlias().get(), actualTable.getOwner().get().getIdentifier(),
99                  actualTable.getTableName().getIdentifier(), actualTable.getAlias().get());
100         for (IdentifierValue each : actualIdentifiers) {
101             assertThat(each.getValue(), is(expectedValue));
102             assertThat(each.getValueWithQuoteCharacters(), is(expectedQuotedValue));
103             String roundTripSQL = "SELECT " + each.getValueWithQuoteCharacters();
104             SelectStatement actualRoundTrip = (SelectStatement) new SQLStatementVisitorEngine("PostgreSQL")
105                     .visit(new SQLParserEngine("PostgreSQL", new CacheOption(128, 1024L)).parse(roundTripSQL, false));
106             assertThat(((ColumnProjectionSegment) actualRoundTrip.getProjections().getProjections().iterator().next()).getColumn().getIdentifier().getValue(), is(expectedValue));
107         }
108     }

219     @ParameterizedTest(name = "{0}")
220     @MethodSource("getUnfilteredAggregateCases")
221     void assertUnfilteredAggregate(final String name, final String expression, final Class<?> expectedType, final int expectedArgCount,
222                                    final boolean expectedWindow, final Collection<Integer> expectedParams) {
223         String sql = "SELECT " + expression + " AS aggregate_value FROM foo";
224         SelectStatement actual = (SelectStatement) new SQLStatementVisitorEngine("PostgreSQL").visit(new SQLParserEngine("PostgreSQL", new CacheOption(128, 1024L)).parse(sql, false));
225         ProjectionSegment actualProjection = actual.getProjections().getProjections().iterator().next();
226         ExpressionSegment actualExpression = actualProjection instanceof ExpressionProjectionSegment
227                 ? ((ExpressionProjectionSegment) actualProjection).getExpr()
228                 : (ExpressionSegment) actualProjection;
229         assertThat(actualProjection.getColumnLabel(), is("aggregate_value"));
230         assertThat(actualExpression.getClass(), is(expectedType));
231         assertThat(sql.substring(actualExpression.getStartIndex(), actualExpression.getStopIndex() + 1), is(actualExpression.getText()));
232         if (actualExpression instanceof FunctionSegment) {
233             FunctionSegment actualFunction = (FunctionSegment) actualExpression;
234             assertThat(actualFunction.getParameters().size(), is(expectedArgCount));
235             assertThat(actualFunction.getWindow().isPresent(), is(expectedWindow));
236         } else {
237             AggregationProjectionSegment actualAggregation = (AggregationProjectionSegment) actualExpression;
238             assertThat(actualAggregation.getParameters().size(), is(expectedArgCount));
239             assertThat(actualAggregation.getWindow().isPresent(), is(expectedWindow));
240         }
241         assertThat(actual.getParameterMarkers().stream().map(ParameterMarkerSegment::getParameterIndex).collect(Collectors.toList()), is(expectedParams));
242     }

269         assertThat(actualMarker.getStartIndex(), is(sql.indexOf("SELECT ?") + 7));
270         assertThat(actualMarker.getStopIndex(), is(actualMarker.getStartIndex()));
271         assertThat(ExpressionExtractor.getParameterMarkerExpressions(actualValues).stream().map(ParameterMarkerSegment::getParameterIndex).collect(Collectors.toList()), is(expectedParams));
