53  import static org.mockito.Mockito.mock;

55  
56  class ShardingProjectionsTokenGeneratorTest {
57      
58      private final DatabaseType databaseType = TypedSPILoader.getService(DatabaseType.class, "FIXTURE");
59      
60      private RouteUnit routeUnit;
61      
62      private ShardingProjectionsTokenGenerator generator;
63      
64      @BeforeEach
65      void setup() {
66          routeUnit = new RouteUnit(mock(RouteMapper.class), Collections.singleton(new RouteMapper("foo_tbl", "foo_tbl_0")));
67          generator = createProjectionsTokenGenerator();
68      }
69      
70      private ShardingProjectionsTokenGenerator createProjectionsTokenGenerator() {
71          RouteContext routeContext = new RouteContext();
72          routeContext.getRouteUnits().add(routeUnit);
73          ShardingProjectionsTokenGenerator result = new ShardingProjectionsTokenGenerator();
74          result.setRouteContext(routeContext);
75          return result;
76      }
77      
78      @Test
79      void assertIsNotGenerateSQLTokenWithNoSelectStatementContext() {
80          assertFalse(generator.isGenerateSQLToken(mock(SQLStatementContext.class)));
81      }
82      
83      @Test
84      void assertIsNotGenerateSQLTokenWithNotContainsDerivedProjection() {
85          SelectStatementContext sqlStatementContext = mock(SelectStatementContext.class, RETURNS_DEEP_STUBS);
86          when(sqlStatementContext.getProjectionsContext().getProjections()).thenReturn(Collections.singleton(mock()));
87          assertFalse(generator.isGenerateSQLToken(sqlStatementContext));
88      }
89      
90      @Test
91      void assertIsGenerateSQLTokenWithDerivedProjections() {
92          SelectStatementContext sqlStatementContext = mock(SelectStatementContext.class, RETURNS_DEEP_STUBS);
93          when(sqlStatementContext.getProjectionsContext().getProjections()).thenReturn(Arrays.asList(mock(AggregationProjection.class), mock(DerivedProjection.class)));
94          assertTrue(generator.isGenerateSQLToken(sqlStatementContext));
95      }
96      
97      @Test
98      void assertIsGenerateSQLTokenWithDerivedAggregationProjections() {
99          SelectStatementContext sqlStatementContext = mock(SelectStatementContext.class, RETURNS_DEEP_STUBS);
100         AggregationProjection aggregationProjection = mock(AggregationProjection.class, RETURNS_DEEP_STUBS);
101         when(sqlStatementContext.getProjectionsContext().getProjections()).thenReturn(Collections.singleton(aggregationProjection));
102         assertTrue(generator.isGenerateSQLToken(sqlStatementContext));
103     }
104     
105     @Test
106     void assertGenerateSQLToken() {
107         SelectStatementContext selectStatementContext = mock(SelectStatementContext.class, RETURNS_DEEP_STUBS);
108         Collection<Projection> projections = Arrays.asList(
109                 createAggregationProjection(), createDerivedProjectionWithOwner(), createDerivedProjectionWithoutOwner(), createOtherDerivedProjection(), mock());
110         when(selectStatementContext.getProjectionsContext().getProjections()).thenReturn(projections);
111         when(selectStatementContext.getProjectionsContext().getStopIndex()).thenReturn(2);
112         when(selectStatementContext.getSqlStatement()).thenReturn(SelectStatement.builder().databaseType(databaseType).build());
113         ProjectionsToken actual = generator.generateSQLToken(selectStatementContext);
114         assertThat(actual.toString(routeUnit),
115                 is(", foo_agg_expr AS foo_agg_alias , foo_tbl_0.foo_derived_col AS foo_derived_alias , bar_derived_col AS bar_derived_alias , other_expr AS other_alias "));
116     }
117     
118     private AggregationProjection createAggregationProjection() {
119         AggregationDistinctProjection derivedProjection = new AggregationDistinctProjection(0, 0, AggregationType.COUNT, new AggregationProjectionSegment(0, 0, AggregationType.COUNT, ""),
120                 new IdentifierValue("foo_agg_alias"), "foo_agg_expr", databaseType);
121         AggregationProjection result = new AggregationDistinctProjection(0, 0, AggregationType.COUNT, new AggregationProjectionSegment(0, 0, AggregationType.COUNT, ""), null, "", databaseType);
122         result.getDerivedAggregationProjections().add(derivedProjection);
123         return result;
124     }
125     
126     private DerivedProjection createDerivedProjectionWithOwner() {
127         ColumnSegment columnSegment = new ColumnSegment(0, 0, new IdentifierValue("foo_derived_col"));
128         columnSegment.setOwner(new OwnerSegment(0, 0, new IdentifierValue("foo_tbl")));
129         return new DerivedProjection("", new IdentifierValue("foo_derived_alias"), new ColumnOrderByItemSegment(columnSegment, OrderDirection.DESC, NullsOrderType.FIRST));
130     }
131     
132     private DerivedProjection createDerivedProjectionWithoutOwner() {
133         ColumnSegment columnSegment = new ColumnSegment(0, 0, new IdentifierValue("bar_derived_col"));
134         return new DerivedProjection("", new IdentifierValue("bar_derived_alias"), new ColumnOrderByItemSegment(columnSegment, OrderDirection.DESC, NullsOrderType.FIRST));
135     }
136     
137     private DerivedProjection createOtherDerivedProjection() {
138         return new DerivedProjection("other_expr", new IdentifierValue("other_alias"), mock());
139     }
140 }
