138 import static org.hamcrest.MatcherAssert.assertThat;
139 import static org.hamcrest.Matchers.isA;

141 /**
142  * Standard DDL statement assert.
143  */
144 @NoArgsConstructor(access = AccessLevel.PRIVATE)
145 public final class StandardDDLStatementAssert {
146     
147     /**
148      * Assert DDL statement is correct with expected parser result.
149      *
150      * @param assertContext assert context
151      * @param actual actual DDL statement
152      * @param expected expected parser result
153      */
154     public static void assertIs(final SQLCaseAssertContext assertContext, final DDLStatement actual, final SQLParserTestCase expected) {
155         if (expected instanceof CreateServerStatementTestCase) {
156             assertThat(actual, isA(CreateServerStatement.class));
157         } else if (actual instanceof CreateTableStatement) {
158             CreateTableStatementAssert.assertIs(assertContext, (CreateTableStatement) actual, (CreateTableStatementTestCase) expected);
159         } else if (actual instanceof AlterTableStatement) {
160             AlterTableStatementAssert.assertIs(assertContext, (AlterTableStatement) actual, (AlterTableStatementTestCase) expected);
161         } else if (actual instanceof RenameTableStatement) {
162             RenameTableStatementAssert.assertIs(assertContext, (RenameTableStatement) actual, (RenameTableStatementTestCase) expected);
163         } else if (actual instanceof DropTableStatement) {
164             DropTableStatementAssert.assertIs(assertContext, (DropTableStatement) actual, (DropTableStatementTestCase) expected);
165         } else if (actual instanceof TruncateStatement) {
166             TruncateStatementAssert.assertIs(assertContext, (TruncateStatement) actual, (TruncateStatementTestCase) expected);
167         } else if (actual instanceof CreateIndexStatement) {
168             CreateIndexStatementAssert.assertIs(assertContext, (CreateIndexStatement) actual, (CreateIndexStatementTestCase) expected);
169         } else if (actual instanceof AlterIndexStatement) {
170             AlterIndexStatementAssert.assertIs(assertContext, (AlterIndexStatement) actual, (AlterIndexStatementTestCase) expected);
171         } else if (actual instanceof DropIndexStatement) {
172             DropIndexStatementAssert.assertIs(assertContext, (DropIndexStatement) actual, (DropIndexStatementTestCase) expected);
173         } else if (actual instanceof AlterSynonymStatement) {
174             AlterSynonymStatementAssert.assertIs(assertContext, (AlterSynonymStatement) actual, (AlterSynonymStatementTestCase) expected);
175         } else if (actual instanceof CursorStatement) {
176             CursorStatementAssert.assertIs(assertContext, (CursorStatement) actual, (CursorStatementTestCase) expected);
177         } else if (actual instanceof CloseStatement) {
178             CloseStatementAssert.assertIs(assertContext, (CloseStatement) actual, (CloseStatementTestCase) expected);
179         } else if (actual instanceof MoveStatement) {
180             MoveStatementAssert.assertIs(assertContext, (MoveStatement) actual, (MoveStatementTestCase) expected);
181         } else if (actual instanceof FetchStatement) {
182             FetchStatementAssert.assertIs(assertContext, (FetchStatement) actual, (FetchStatementTestCase) expected);
183         } else if (actual instanceof CommentStatement) {
184             CommentStatementAssert.assertIs(assertContext, (CommentStatement) actual, (CommentStatementTestCase) expected);
185         } else if (actual instanceof RefreshMatViewStmtStatement) {
186             RefreshMatViewStmtStatementAssert.assertIs(assertContext, (RefreshMatViewStmtStatement) actual, (RefreshMatViewStmtStatementTestCase) expected);
187         } else if (actual instanceof CreateViewStatement) {
188             CreateViewStatementAssert.assertIs(assertContext, (CreateViewStatement) actual, (CreateViewStatementTestCase) expected);
189         } else if (actual instanceof AlterViewStatement) {
190             AlterViewStatementAssert.assertIs(assertContext, (AlterViewStatement) actual, (AlterViewStatementTestCase) expected);
191         } else if (actual instanceof DropViewStatement) {
192             DropViewStatementAssert.assertIs(assertContext, (DropViewStatement) actual, (DropViewStatementTestCase) expected);
193         } else if (actual instanceof AlterTablespaceStatement) {
194             AlterTablespaceStatementAssert.assertIs(assertContext, (AlterTablespaceStatement) actual, (AlterTablespaceStatementTestCase) expected);
195         } else if (actual instanceof CreateSequenceStatement) {
196             CreateSequenceStatementAssert.assertIs(assertContext, (CreateSequenceStatement) actual, (CreateSequenceStatementTestCase) expected);
197         } else if (actual instanceof CreateProcedureStatement) {
198             CreateProcedureStatementAssert.assertIs(assertContext, (CreateProcedureStatement) actual, (CreateProcedureTestCase) expected);
199         } else if (actual instanceof CreateTriggerStatement) {
200             CreateTriggerStatementAssert.assertIs(assertContext, (CreateTriggerStatement) actual, (CreateTriggerStatementTestCase) expected);
201         } else if (actual instanceof AlterPackageStatement && expected instanceof AlterPackageStatementTestCase) {
202             RoutineDDLStatementAssert.assertAlterPackage(assertContext, (AlterPackageStatement) actual, (AlterPackageStatementTestCase) expected);
203         } else if (actual instanceof DropPackageStatement && expected instanceof OracleDropPackageStatementTestCase) {
204             RoutineDDLStatementAssert.assertDropPackage(assertContext, (DropPackageStatement) actual, (OracleDropPackageStatementTestCase) expected);
205         } else if (actual instanceof AlterFunctionStatement && expected instanceof AlterFunctionStatementTestCase) {
206             RoutineDDLStatementAssert.assertAlterFunction(assertContext, (AlterFunctionStatement) actual, (AlterFunctionStatementTestCase) expected);
207         } else if (actual instanceof DropFunctionStatement && expected instanceof DropFunctionStatementTestCase) {
208             RoutineDDLStatementAssert.assertDropFunction(assertContext, (DropFunctionStatement) actual, (DropFunctionStatementTestCase) expected);
209         } else if (actual instanceof AlterTriggerStatement && expected instanceof AlterTriggerStatementTestCase) {
210             RoutineDDLStatementAssert.assertAlterTrigger(assertContext, (AlterTriggerStatement) actual, (AlterTriggerStatementTestCase) expected);
211         } else if (actual instanceof DropTriggerStatement && expected instanceof DropTriggerStatementTestCase) {
212             RoutineDDLStatementAssert.assertDropTrigger(assertContext, (DropTriggerStatement) actual, (DropTriggerStatementTestCase) expected);
213         } else if (actual instanceof AlterCatalogStatement) {
214             AlterCatalogStatementAssert.assertIs(assertContext, (AlterCatalogStatement) actual, (AlterCatalogStatementTestCase) expected);
215         } else if (actual instanceof AlterDatabaseStatement) {
216             if (!(expected instanceof OracleAlterPluggableDatabaseStatementTestCase)) {
217                 AlterDatabaseStatementAssert.assertIs(assertContext, (AlterDatabaseStatement) actual, (AlterDatabaseStatementTestCase) expected);
218             }
219         } else if (actual instanceof BuildIndexStatement) {
220             BuildIndexStatementAssert.assertIs(assertContext, (BuildIndexStatement) actual, (BuildIndexStatementTestCase) expected);
221         } else if (actual instanceof CancelBuildIndexStatement) {
222             CancelBuildIndexStatementAssert.assertIs(assertContext, (CancelBuildIndexStatement) actual, (CancelBuildIndexStatementTestCase) expected);
223         } else if (actual instanceof CreateFileStatement) {
224             CreateFileStatementAssert.assertIs(assertContext, (CreateFileStatement) actual, (CreateFileStatementTestCase) expected);
225         } else if (actual instanceof DropFileStatement) {
226             DropFileStatementAssert.assertIs(assertContext, (DropFileStatement) actual, (DropFileStatementTestCase) expected);
227         } else if (actual instanceof CancelAlterTableStatement) {
228             CancelAlterTableStatementAssert.assertIs(assertContext, (CancelAlterTableStatement) actual, (CancelAlterTableStatementTestCase) expected);
229         } else if (actual instanceof DropEncryptKeyStatement) {
230             DropEncryptKeyStatementAssert.assertIs(assertContext, (DropEncryptKeyStatement) actual, (DropEncryptKeyStatementTestCase) expected);
231         } else if (actual instanceof CreateDatabaseStatement) {
232             CreateDatabaseStatementAssert.assertIs(assertContext, (CreateDatabaseStatement) actual, (CreateDatabaseStatementTestCase) expected);
233         } else if (actual instanceof DropDatabaseStatement) {
234             DropDatabaseStatementAssert.assertIs(assertContext, (DropDatabaseStatement) actual, (DropDatabaseStatementTestCase) expected);
235         }
236     }
