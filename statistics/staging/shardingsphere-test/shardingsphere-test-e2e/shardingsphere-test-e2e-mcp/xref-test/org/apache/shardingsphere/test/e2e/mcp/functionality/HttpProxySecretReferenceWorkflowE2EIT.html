18  package org.apache.shardingsphere.test.e2e.mcp.functionality;
19  
20  import org.apache.shardingsphere.mcp.support.diagnostic.MCPDiagnosticCategory;
21  import org.apache.shardingsphere.mcp.support.workflow.descriptor.WorkflowToolDescriptors;
22  import org.apache.shardingsphere.test.e2e.mcp.support.transport.client.MCPInteractionClient;
23  import org.junit.jupiter.api.Test;
24  import org.junit.jupiter.api.condition.EnabledIf;
25  
26  import java.io.IOException;
27  import java.util.List;
28  import java.util.Map;
29  
30  import static org.hamcrest.MatcherAssert.assertThat;
31  import static org.hamcrest.Matchers.containsString;
32  import static org.hamcrest.Matchers.is;
33  import static org.junit.jupiter.api.Assertions.assertFalse;
34  import static org.junit.jupiter.api.Assertions.assertTrue;
35  
36  @EnabledIf("org.apache.shardingsphere.test.e2e.mcp.env.MCPE2ECondition#isDockerEnabled")
37  class HttpProxySecretReferenceWorkflowE2EIT extends AbstractHttpProxyWorkflowE2EIT {
38      
39      private static final String PLAN_TOOL_NAME = "database_gateway_plan_encrypt_rule";
40      
41      private static final String APPLY_TOOL_NAME = WorkflowToolDescriptors.APPLY_TOOL_NAME;
42      
43      private static final String RULES_RESOURCE_URI = "shardingsphere://features/encrypt/databases/%s/rules";
44      
45      private static final String SECRET_REF = "placeholder://secret-value-1";
46      
47      private static final String INPUT_LABEL = "placeholder-secret-1";
48      
49      @Test
50      void assertSecretReferenceApplyRequiresManualExecution() throws IOException, InterruptedException {
51          try (MCPInteractionClient interactionClient = createOpenedInteractionClient()) {
52              Map<String, Object> planResponse = planSecretReferencedEncryptRule(interactionClient);
53              String planId = String.valueOf(planResponse.get("plan_id"));
54              assertPlannedSecretReferencePayload(planResponse);
55              Map<String, Object> previewResponse = previewWorkflow(interactionClient, planId);
56              assertSecretReferencedPreview(previewResponse);
57              Map<String, Object> applyResponse = interactionClient.call(APPLY_TOOL_NAME, createReviewThenExecuteArguments(planId, getApprovedSteps(previewResponse)));
58              assertThat(String.valueOf(applyResponse.get("response_mode")), is("recovery"));
59              assertThat(String.valueOf(applyResponse.get("status")), is("failed"));
60              assertThat(String.valueOf(applyResponse.get("category")), is(MCPDiagnosticCategory.SECRET_REFERENCE_MANUAL_EXECUTION_REQUIRED));
61              assertModelFacingPayloadContract(applyResponse);
62              assertThat(getObjectListOrEmpty(applyResponse.get("step_results")).size(), is(0));
63              assertSecretReferenceRedacted(applyResponse);
64              assertTrue(getPayloadItems(interactionClient.readResource(String.format(RULES_RESOURCE_URI, getLogicalDatabaseName()))).isEmpty());
65          }
66      }
67      
68      private Map<String, Object> planSecretReferencedEncryptRule(final MCPInteractionClient interactionClient) throws IOException, InterruptedException {
69          return interactionClient.call(PLAN_TOOL_NAME,
70                  Map.of("database", getLogicalDatabaseName(), "table", "orders", "column", "status",
71                          "operation_type", "create", "requires_decrypt", true, "requires_equality_filter", false, "requires_like_query", false,
72                          "algorithm_type", "AES", "cipher_column_name", "status_cipher",
73                          "primary_algorithm_properties", Map.of("aes-key-value", Map.of("secret_ref", SECRET_REF, "label", INPUT_LABEL))));
74      }
75      
76      private void assertPlannedSecretReferencePayload(final Map<String, Object> planResponse) {
77          assertThat(String.valueOf(planResponse.get("status")), is("planned"));
78          assertModelFacingPayloadContract(planResponse);
79          Map<String, Object> secretReferenceSummary = getObjectOrEmpty(planResponse.get("secret_reference_summary"));
80          assertTrue((Boolean) secretReferenceSummary.get("required"));
81          assertThat(secretReferenceSummary.get("reference_count"), is(1));
82          Map<String, Object> secretReference = getObjectListOrEmpty(secretReferenceSummary.get("references")).getFirst();
83          assertThat(String.valueOf(secretReference.get("algorithm_role")), is("primary"));
84          assertThat(String.valueOf(secretReference.get("property_key")), is("aes-key-value"));
85          assertThat(String.valueOf(secretReference.get("label")), is("secret_placeholder:primary.aes-key-value"));
86          assertThat(String.valueOf(secretReference.get("manual_placeholder")), is("<SECRET_VALUE_PRIMARY_AES_KEY_VALUE>"));
87          Map<String, Object> maskedPropertyPreview = getObjectOrEmpty(planResponse.get("masked_property_preview"));
88          assertThat(String.valueOf(getObjectOrEmpty(maskedPropertyPreview.get("primary")).get("aes-key-value")), is("******"));
89          assertSecretReferenceRedacted(planResponse);
90      }
91      
92      private void assertSecretReferencedPreview(final Map<String, Object> previewResponse) {
93          assertThat(String.valueOf(previewResponse.get("status")), is("preview"));
94          assertModelFacingPayloadContract(previewResponse);
95          List<Map<String, Object>> previewArtifacts = getObjectListOrEmpty(previewResponse.get("preview_artifacts"));
96          assertThat(previewArtifacts.size(), is(1));
97          assertThat(String.valueOf(previewArtifacts.getFirst().get("sql")), containsString("'aes-key-value'='<SECRET_VALUE_PRIMARY_AES_KEY_VALUE>'"));
98          assertFalse(String.valueOf(previewArtifacts.getFirst().get("sql")).contains("secret_reference:primary.aes-key-value"));
99          assertSecretReferenceRedacted(previewResponse);
100     }
101     
102     private void assertSecretReferenceRedacted(final Object payload) {
103         String actual = String.valueOf(payload);
104         assertFalse(actual.contains(SECRET_REF));
105         assertFalse(actual.contains(INPUT_LABEL));
106         assertFalse(actual.contains("secret_reference:"));
107     }
108 }
