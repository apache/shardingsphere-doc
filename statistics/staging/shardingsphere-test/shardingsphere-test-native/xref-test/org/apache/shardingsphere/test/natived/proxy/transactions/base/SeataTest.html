71      @BeforeEach
72      void beforeEach() {
73          assertNull(System.getProperty(serviceDefaultGroupListKey));
74          System.setProperty(serviceDefaultGroupListKey, "127.0.0.1:" + container.getMappedPort(8091));
75          Awaitility.await().atMost(Duration.ofSeconds(30L)).ignoreExceptions().until(() -> {
76              openConnection("test", "test", "jdbc:postgresql://127.0.0.1:" + postgresContainer.getMappedPort(5432) + "/")
77                      .close();
78              return true;
79          });
80          String absolutePath = Paths.get("src/test/resources/test-native/yaml/proxy/transactions/base").toAbsolutePath().toString();
81          proxyTestingServer = new ProxyTestingServer(absolutePath);
82          Awaitility.await().atMost(Duration.ofSeconds(30L)).ignoreExceptions().until(() -> {
83              openConnection("root", "root", "jdbc:postgresql://127.0.0.1:" + proxyTestingServer.getProxyPort() + "/postgres").close();
84              return true;
85          });
86      }

98      /**
99       * {@link groovy.lang.Closure} related classes are not available on GraalVM Native Image.
100      * This CLASS_BASE algorithm class is designed to emulate INLINE's {@code ds_${user_id % 2}}.
101      * See <a href="https://github.com/oracle/graal/issues/5522">oracle/graal#5522</a> .
102      *
103      * @throws SQLException SQL Exception
104      */
105     @Test
106     void assertShardingInLocalTransactions() throws SQLException {
107         try (
108                 Connection connection = openConnection("root", "root", "jdbc:postgresql://127.0.0.1:" + proxyTestingServer.getProxyPort() + "/postgres");
109                 Statement statement = connection.createStatement()) {
110             statement.execute("CREATE DATABASE sharding_db");
111         }
112         try (
113                 Connection connection = openConnection("root", "root", "jdbc:postgresql://127.0.0.1:" + proxyTestingServer.getProxyPort() + "/sharding_db");
114                 Statement statement = connection.createStatement()) {
115             statement.execute("REGISTER STORAGE UNIT ds_0 (URL='jdbc:postgresql://127.0.0.1:" + postgresContainer.getMappedPort(5432) + "/demo_ds_0',USER='test',PASSWORD='test'),"
116                     + "ds_1 (URL='jdbc:postgresql://127.0.0.1:" + postgresContainer.getMappedPort(5432) + "/demo_ds_1',USER='test',PASSWORD='test'),"
117                     + "ds_2 (URL='jdbc:postgresql://127.0.0.1:" + postgresContainer.getMappedPort(5432) + "/demo_ds_2',USER='test',PASSWORD='test')");
118             statement.execute("CREATE DEFAULT SHARDING DATABASE STRATEGY (TYPE='standard', SHARDING_COLUMN=user_id, SHARDING_ALGORITHM(TYPE(NAME=CLASS_BASED,"
119                     + "PROPERTIES('strategy'='STANDARD','algorithmClassName'='org.apache.shardingsphere.test.natived.commons.algorithm.ClassBasedInlineShardingAlgorithmFixture'))))");
120             statement.execute("CREATE SHARDING TABLE RULE t_order (DATANODES('<LITERAL>ds_0.t_order, ds_1.t_order, ds_2.t_order'),"
121                     + "KEY_GENERATE_STRATEGY(COLUMN=order_id,TYPE(NAME='SNOWFLAKE'))), t_order_item (DATANODES('<LITERAL>ds_0.t_order_item, ds_1.t_order_item, ds_2.t_order_item'),"
122                     + "KEY_GENERATE_STRATEGY(COLUMN=order_item_id,TYPE(NAME='SNOWFLAKE')))");
123             statement.execute("CREATE BROADCAST TABLE RULE t_address");
124         }
125         HikariConfig config = new HikariConfig();
126         config.setDriverClassName("org.postgresql.Driver");
127         config.setJdbcUrl("jdbc:postgresql://127.0.0.1:" + proxyTestingServer.getProxyPort() + "/sharding_db");
128         config.setUsername("root");
129         config.setPassword("root");
130         DataSource dataSource = new HikariDataSource(config);
131         testShardingService = new TestShardingService(dataSource);
132         initEnvironment();
133         testShardingService.processSuccess();
134         testShardingService.cleanEnvironment();
135     }
136     
137     private void initEnvironment() throws SQLException {
138         testShardingService.getOrderRepository().createTableIfNotExistsInPostgres();
139         testShardingService.getOrderItemRepository().createTableIfNotExistsInPostgres();
140         testShardingService.getAddressRepository().createTableIfNotExistsInMySQL();
141         testShardingService.getOrderRepository().truncateTable();
142         testShardingService.getOrderItemRepository().truncateTable();
143         testShardingService.getAddressRepository().truncateTable();
144     }
145     
146     private Connection openConnection(final String username, final String password, final String jdbcUrl) throws SQLException {
147         Properties props = new Properties();
148         props.setProperty("user", username);
149         props.setProperty("password", password);
150         return DriverManager.getConnection(jdbcUrl, props);
151     }
