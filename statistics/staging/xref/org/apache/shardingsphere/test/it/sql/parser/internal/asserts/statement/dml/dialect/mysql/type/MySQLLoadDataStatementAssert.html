35  import static org.hamcrest.MatcherAssert.assertThat;
36  import static org.hamcrest.Matchers.is;
37  import static org.junit.jupiter.api.Assertions.assertFalse;
38  import static org.junit.jupiter.api.Assertions.assertNotNull;
39  import static org.junit.jupiter.api.Assertions.assertNull;

41  /**
42   * Load data statement assert for MySQL.
43   */
44  @NoArgsConstructor(access = AccessLevel.PRIVATE)
45  public final class MySQLLoadDataStatementAssert {
46      
47      /**
48       * Assert load data statement is correct with expected parser result.
49       *
50       * @param assertContext assert context
51       * @param actual actual load data statement
52       * @param expected expected load data statement test case
53       */
54      public static void assertIs(final SQLCaseAssertContext assertContext, final MySQLLoadDataStatement actual, final MySQLLoadDataStatementTestCase expected) {
55          assertLocal(assertContext, actual, expected);
56          assertFileName(assertContext, actual, expected);
57          assertTable(assertContext, actual, expected);
58          assertPartitions(assertContext, actual, expected);
59          assertColumnSeparator(assertContext, actual, expected);
60          assertLineDelimiter(assertContext, actual, expected);
61          assertIgnoreLines(assertContext, actual, expected);
62          assertColumnList(assertContext, actual, expected);
63          assertSetAssignments(assertContext, actual, expected);
64          assertProperties(assertContext, actual, expected);
65      }
66      
67      private static void assertLocal(final SQLCaseAssertContext assertContext, final MySQLLoadDataStatement actual, final MySQLLoadDataStatementTestCase expected) {
68          assertThat(assertContext.getText("LOCAL flag does not match: "), actual.isLocal(), is(null != expected.getLocal() && expected.getLocal()));
69      }
70      
71      private static void assertFileName(final SQLCaseAssertContext assertContext, final MySQLLoadDataStatement actual, final MySQLLoadDataStatementTestCase expected) {
72          if (null != expected.getFileName()) {
73              assertNotNull(actual.getFileName(), assertContext.getText("File name segment should not be null"));
74              assertThat(assertContext.getText("File name value does not match: "), actual.getFileName().getValue(), is(expected.getFileName().getValue()));
75              SQLSegmentAssert.assertIs(assertContext, actual.getFileName(), expected.getFileName());
76          }
77      }
78      
79      private static void assertTable(final SQLCaseAssertContext assertContext, final MySQLLoadDataStatement actual, final MySQLLoadDataStatementTestCase expected) {
80          if (null == expected.getTable()) {
81              assertNull(actual.getTable(), assertContext.getText("Actual table should not exist."));
82          } else {
83              TableAssert.assertIs(assertContext, actual.getTable(), expected.getTable());
84          }
85      }
86      
87      private static void assertPartitions(final SQLCaseAssertContext assertContext, final MySQLLoadDataStatement actual, final MySQLLoadDataStatementTestCase expected) {
88          if (!expected.getPartitions().isEmpty()) {
89              assertThat(assertContext.getText("Partitions size does not match: "), actual.getPartitions().size(), is(expected.getPartitions().size()));
90              int count = 0;
91              for (PartitionSegment each : actual.getPartitions()) {
92                  PartitionAssert.assertIs(assertContext, each, expected.getPartitions().get(count));
93                  count++;
94              }
95          }
96      }
97      
98      private static void assertColumnSeparator(final SQLCaseAssertContext assertContext, final MySQLLoadDataStatement actual, final MySQLLoadDataStatementTestCase expected) {
99          if (null != expected.getColumnSeparator()) {
100             assertNotNull(actual.getColumnSeparator().orElse(null), assertContext.getText("Column separator should not be null"));
101             assertThat(assertContext.getText("Column separator value does not match: "), actual.getColumnSeparator().get().getValue(),
102                     is(expected.getColumnSeparator().getValue()));
103             SQLSegmentAssert.assertIs(assertContext, actual.getColumnSeparator().get(), expected.getColumnSeparator());
104         }
105     }
106     
107     private static void assertLineDelimiter(final SQLCaseAssertContext assertContext, final MySQLLoadDataStatement actual, final MySQLLoadDataStatementTestCase expected) {
108         if (null != expected.getLineDelimiter()) {
109             assertNotNull(actual.getLineDelimiter().orElse(null), assertContext.getText("Line delimiter should not be null"));
110             assertThat(assertContext.getText("Line delimiter value does not match: "), actual.getLineDelimiter().get().getValue(),
111                     is(expected.getLineDelimiter().getValue()));
112             SQLSegmentAssert.assertIs(assertContext, actual.getLineDelimiter().get(), expected.getLineDelimiter());
113         }
114     }
115     
116     private static void assertIgnoreLines(final SQLCaseAssertContext assertContext, final MySQLLoadDataStatement actual, final MySQLLoadDataStatementTestCase expected) {
117         if (null != expected.getIgnoreLines()) {
118             assertNotNull(actual.getIgnoreLines().orElse(null), assertContext.getText("Ignore lines segment should not be null"));
119             assertThat(assertContext.getText("Ignore lines number does not match: "), actual.getIgnoreLines().get().getNumber(), is(expected.getIgnoreLines().getNumber()));
120             assertThat(assertContext.getText("Ignore lines unit does not match: "), actual.getIgnoreLines().get().getUnit(), is(expected.getIgnoreLines().getUnit()));
121             SQLSegmentAssert.assertIs(assertContext, actual.getIgnoreLines().get(), expected.getIgnoreLines());
122         }
123     }
124     
125     private static void assertColumnList(final SQLCaseAssertContext assertContext, final MySQLLoadDataStatement actual, final MySQLLoadDataStatementTestCase expected) {
126         if (!expected.getColumns().isEmpty()) {
127             assertThat(assertContext.getText("Column list size does not match: "), actual.getColumnList().size(), is(expected.getColumns().size()));
128             int count = 0;
129             for (ColNameOrUserVarSegment each : actual.getColumnList()) {
130                 ExpectedPartition exp = expected.getColumns().get(count);
131                 assertThat(assertContext.getText("Column name does not match: "), each.getIdentifier().getValue(), is(exp.getName()));
132                 SQLSegmentAssert.assertIs(assertContext, each, exp);
133                 count++;
134             }
135         }
136     }
137     
138     private static void assertSetAssignments(final SQLCaseAssertContext assertContext, final MySQLLoadDataStatement actual, final MySQLLoadDataStatementTestCase expected) {
139         if (null != expected.getSetAssignments()) {
140             assertNotNull(actual.getSetAssignments().orElse(null), assertContext.getText("Set assignments segment should not be null"));
141             SetClauseAssert.assertIs(assertContext, actual.getSetAssignments().get(), expected.getSetAssignments());
142         }
143     }
144     
145     private static void assertProperties(final SQLCaseAssertContext assertContext, final MySQLLoadDataStatement actual, final MySQLLoadDataStatementTestCase expected) {
146         if (!expected.getProperties().isEmpty()) {
147             assertNotNull(actual.getProperties().orElse(null), assertContext.getText("Actual properties should exist."));
148             assertThat(assertContext.getText("Properties size does not match: "), actual.getProperties().get().getProperties().size(), is(expected.getProperties().size()));
149             for (int i = 0; i < expected.getProperties().size(); i++) {
150                 assertProperty(assertContext, actual.getProperties().get().getProperties().get(i), expected.getProperties().get(i));
151             }
152         } else if (actual.getProperties().isPresent() && !actual.getProperties().get().getProperties().isEmpty()) {
153             assertFalse(expected.getProperties().isEmpty(), assertContext.getText("Expected properties should exist when actual has properties."));
154         }
155     }
156     
157     private static void assertProperty(final SQLCaseAssertContext assertContext, final PropertySegment actual, final PropertyTestCase expected) {
158         assertThat(assertContext.getText(String.format("Property key '%s' assertion error: ", expected.getKey())), actual.getKey(), is(expected.getKey()));
159         assertThat(assertContext.getText(String.format("Property value for key '%s' assertion error: ", expected.getKey())), actual.getValue(), is(expected.getValue()));
160         SQLSegmentAssert.assertIs(assertContext, actual, expected);
161     }
