30  
31  import java.util.Arrays;
32  import java.util.Collection;
33  import java.util.Optional;
34  
35  /**
36   * Global rule changed handler.
37   */
38  public final class GlobalRuleChangedHandler implements DataChangedEventHandler {
39      
40      @Override
41      public String getSubscribedKey() {
42          return GlobalNode.getGlobalRuleRootNode();
43      }
44      
45      @Override
46      public Collection<Type> getSubscribedTypes() {
47          return Arrays.asList(Type.ADDED, Type.UPDATED);
48      }
49      
50      @SuppressWarnings("unchecked")
51      @Override
52      public void handle(final ContextManager contextManager, final DataChangedEvent event) {
53          if (!GlobalNodePath.isRuleActiveVersionPath(event.getKey())) {
54              return;
55          }
56          Optional<String> ruleName = GlobalNodePath.getRuleName(event.getKey());
57          if (!ruleName.isPresent()) {
58              return;
59          }
60          Preconditions.checkArgument(event.getValue().equals(
61                  contextManager.getPersistServiceFacade().getMetaDataPersistService().getMetaDataVersionPersistService().getActiveVersionByFullPath(event.getKey())),
62                  "Invalid active version: %s of key: %s", event.getValue(), event.getKey());
63          Optional<RuleConfiguration> ruleConfig = contextManager.getPersistServiceFacade().getMetaDataPersistService().getGlobalRuleService().load(ruleName.get());
64          Preconditions.checkArgument(ruleConfig.isPresent(), "Can not find rule configuration with name: %s", ruleName.get());
65          contextManager.getMetaDataContextManager().getGlobalConfigurationManager().alterGlobalRuleConfiguration(
66                  TypedSPILoader.findService(RuleConfigurationPersistDecorator.class, ruleConfig.getClass()).map(optional -> optional.restore(ruleConfig.get())).orElse(ruleConfig.get()));
67      }
68  }
