106     public FirebirdBatchedStatementsExecutor(final ConnectionSession connectionSession, final FirebirdServerPreparedStatement preparedStatement, final List<List<Object>> parameterSets,
107                                              final boolean multiError) {
108         jdbcExecutor = new JDBCExecutor(BackendExecutorContext.getInstance().getExecutorEngine(), connectionSession.getConnectionContext());
109         this.multiError = multiError;
110         this.connectionSession = connectionSession;
111         metaDataContexts = ProxyContext.getInstance().getContextManager().getMetaDataContexts();
112         this.preparedStatement = preparedStatement;
113         batchMessageCount = parameterSets.size();
114         Iterator<List<Object>> parameterSetsIterator = parameterSets.iterator();
115         SQLStatementContext sqlStatementContext = null;
116         ExecutionContext executionContext = null;
117         int batchMessageIndex = 0;
118         if (parameterSetsIterator.hasNext()) {
119             List<Object> firstGroupOfParam = parameterSetsIterator.next();
120             sqlStatementContext = createSQLStatementContext(firstGroupOfParam);
121             executionContext = createExecutionContext(createQueryContext(sqlStatementContext, firstGroupOfParam));
122             addExecutionUnitParams(executionContext, batchMessageIndex++);
123         }
124         anyExecutionContext = executionContext;
125         prepareForRestOfParametersSet(parameterSetsIterator, sqlStatementContext, batchMessageIndex);
126     }
127     
128     private SQLStatementContext createSQLStatementContext(final List<Object> params) {
129         SQLStatementContext result = new SQLBindEngine(metaDataContexts.getMetaData(), connectionSession.getCurrentDatabaseName(), preparedStatement.getHintValueContext())
130                 .bind(preparedStatement.getSqlStatementContext().getSqlStatement());
131         if (result instanceof ParameterAware) {
132             ((ParameterAware) result).bindParameters(params);
133         }
134         return result;
135     }
136     
137     private void prepareForRestOfParametersSet(final Iterator<List<Object>> paramSetsIterator, final SQLStatementContext sqlStatementContext, final int firstBatchMessageIndex) {
138         int batchMessageIndex = firstBatchMessageIndex;
139         while (paramSetsIterator.hasNext()) {
140             List<Object> eachGroupOfParam = paramSetsIterator.next();
141             if (sqlStatementContext instanceof ParameterAware) {
142                 ((ParameterAware) sqlStatementContext).bindParameters(eachGroupOfParam);
143             }
144             ExecutionContext eachExecutionContext = createExecutionContext(createQueryContext(sqlStatementContext, eachGroupOfParam));
145             addExecutionUnitParams(eachExecutionContext, batchMessageIndex++);
146         }
147     }
148     
149     private void addExecutionUnitParams(final ExecutionContext executionContext, final int batchMessageIndex) {
150         for (ExecutionUnit each : executionContext.getExecutionUnits()) {
151             executionUnitParams.computeIfAbsent(each, key -> new LinkedList<>()).add(each.getSqlUnit().getParameters());
152             executionUnitBatchMessageIndexes.computeIfAbsent(each, key -> new LinkedList<>()).add(batchMessageIndex);
153         }
154     }
155     
156     private ExecutionContext createExecutionContext(final QueryContext queryContext) {
157         ShardingSphereDatabase currentDatabase = metaDataContexts.getMetaData().getDatabase(connectionSession.getCurrentDatabaseName());
158         SQLAuditEngine.audit(queryContext, currentDatabase);
159         return kernelProcessor.generateExecutionContext(queryContext, metaDataContexts.getMetaData().getGlobalRuleMetaData(), metaDataContexts.getMetaData().getProps());
160     }
161     
162     private QueryContext createQueryContext(final SQLStatementContext sqlStatementContext, final List<Object> params) {
163         return new QueryContext(sqlStatementContext, preparedStatement.getSql(), params, preparedStatement.getHintValueContext(), connectionSession.getConnectionContext(),
164                 metaDataContexts.getMetaData());
165     }
166     
167     /**
168      * Execute batch.
169      *
170      * @return batch completion in the Firebird domain
171      * @throws SQLException SQL exception
172      */
173     public FirebirdBatchCompletion executeBatch() throws SQLException {
174         connectionSession.getDatabaseConnectionManager().handleAutoCommit();
175         prepareBatchedStatements();
176         List<BatchExecution> batchExecutions;
177         List<BatchExecutionUnitResult> executeResults;
178         try {
179             batchExecutions = createBatchExecutionPlan();
180             executeResults = executeBatchedPreparedStatements(batchExecutions);
181         } finally {
182             closePreparedStatements();
183         }
184         return createBatchCompletion(batchExecutions, executeResults);
185     }
186     
187     private void prepareBatchedStatements() throws SQLException {
188         Collection<ShardingSphereRule> rules = metaDataContexts.getMetaData().getDatabase(connectionSession.getUsedDatabaseName()).getRuleMetaData().getRules();
189         int maxConnectionsSizePerQuery = metaDataContexts.getMetaData().getProps().<Integer>getValue(ConfigurationPropertyKey.MAX_CONNECTIONS_SIZE_PER_QUERY);
190         DriverExecutionPrepareEngine<JDBCExecutionUnit, Connection> prepareEngine = new DriverExecutionPrepareEngine<>(JDBCDriverType.PREPARED_STATEMENT, maxConnectionsSizePerQuery,
191                 connectionSession.getDatabaseConnectionManager(), (JDBCBackendStatement) connectionSession.getStatementManager(), new StatementOption(false), rules, metaDataContexts.getMetaData());
192         executionGroupContext = prepareEngine.prepare(connectionSession.getUsedDatabaseName(), anyExecutionContext, executionUnitParams.keySet(),
193                 new ExecutionGroupReportContext(connectionSession.getProcessId(), connectionSession.getUsedDatabaseName(), connectionSession.getConnectionContext().getGrantee()));
194     }
195     
196     private void addPendingBatch(final BatchExecution batchExecution) throws SQLException {
197         PreparedStatement preparedStatement = (PreparedStatement) batchExecution.executionUnit.getStorageResource();
198         List<List<Object>> params = getBatchParams(batchExecution);
199         pendingBatchParams.put(preparedStatement, params);
200         addBatchedParameters(preparedStatement, params);
201     }
202     
203     private void addBatchedParameters(final PreparedStatement preparedStatement, final List<List<Object>> params) throws SQLException {
204         for (List<Object> eachGroupParam : params) {
205             ListIterator<Object> eachParams = eachGroupParam.listIterator();
206             while (eachParams.hasNext()) {
207                 int paramIndex = eachParams.nextIndex() + 1;
208                 Object value = eachParams.next();
209                 preparedStatement.setObject(paramIndex, value);
210             }
211             preparedStatement.addBatch();
212         }
213     }
214     
215     private List<List<Object>> getBatchParams(final BatchExecution batchExecution) {
216         return executionUnitParams.getOrDefault(batchExecution.executionUnit.getExecutionUnit(), Collections.emptyList())
217                 .subList(batchExecution.fromOffset, batchExecution.toOffset);
218     }
219     
220     private List<Integer> getBatchMessageIndexes(final BatchExecution batchExecution) {
221         return executionUnitBatchMessageIndexes.getOrDefault(batchExecution.executionUnit.getExecutionUnit(), Collections.emptyList())
222                 .subList(batchExecution.fromOffset, batchExecution.toOffset);
223     }
224     
225     private void closePreparedStatements() throws SQLException {
226         for (ExecutionGroup<JDBCExecutionUnit> eachGroup : executionGroupContext.getInputGroups()) {
227             for (JDBCExecutionUnit each : eachGroup.getInputs()) {
228                 each.getStorageResource().close();
229             }
230         }
231     }
232     
233     private List<BatchExecutionUnitResult> executeBatchedPreparedStatements(final List<BatchExecution> batchExecutions) throws SQLException {
234         boolean isExceptionThrown = SQLExecutorExceptionHandler.isExceptionThrown();
235         ShardingSphereDatabase database = metaDataContexts.getMetaData().getDatabase(connectionSession.getUsedDatabaseName());
236         JDBCExecutorCallback<BatchExecutionUnitResult> callback =
237                 new BatchedStatementsJDBCExecutorCallback(database.getProtocolType(), database.getResourceMetaData(), preparedStatement.getSqlStatementContext().getSqlStatement(), isExceptionThrown);
238         return multiError ? jdbcExecutor.execute(executionGroupContext, callback) : executeInBatchMessageOrder(batchExecutions, callback);
239     }
240     
241     private List<BatchExecutionUnitResult> executeInBatchMessageOrder(final List<BatchExecution> batchExecutions, final JDBCExecutorCallback<BatchExecutionUnitResult> callback) throws SQLException {
242         List<BatchExecutionUnitResult> result = new LinkedList<>();
243         for (BatchExecution each : batchExecutions) {
244             addPendingBatch(each);
245             List<BatchExecutionUnitResult> executeResults = jdbcExecutor.execute(createSingleUnitExecutionGroupContext(each.executionUnit), callback);
246             result.addAll(executeResults);
247             if (executeResults.stream().anyMatch(eachResult -> !eachResult.failures.isEmpty())) {
248                 break;
249             }
250         }
251         return result;
252     }
253     
254     private ExecutionGroupContext<JDBCExecutionUnit> createSingleUnitExecutionGroupContext(final JDBCExecutionUnit executionUnit) {
255         return new ExecutionGroupContext<>(Collections.singleton(new ExecutionGroup<>(Collections.singletonList(executionUnit))), executionGroupContext.getReportContext());
256     }
257     
258     private FirebirdBatchCompletion createBatchCompletion(final List<BatchExecution> batchExecutions, final List<BatchExecutionUnitResult> executeResults) throws SQLException {
259         int[] messageUpdateCounts = new int[batchMessageCount];
260         Arrays.fill(messageUpdateCounts, NOT_EXECUTED);
261         NavigableMap<Integer, SQLException> failedMessages = new TreeMap<>();
262         Iterator<BatchExecution> executions = batchExecutions.iterator();
263         Iterator<BatchExecutionUnitResult> results = executeResults.iterator();
264         while (executions.hasNext() && results.hasNext()) {
265             List<Integer> batchMessageIndexes = getBatchMessageIndexes(executions.next());
266             BatchExecutionUnitResult eachResult = results.next();
267             mergeUpdateCounts(messageUpdateCounts, batchMessageIndexes, eachResult.updateCounts);
268             markFailedMessages(messageUpdateCounts, failedMessages, batchMessageIndexes, eachResult.failures);
269         }
270         int processedMessageCount = getProcessedMessageCount(messageUpdateCounts, failedMessages);
271         return new FirebirdBatchCompletion(processedMessageCount, toFirebirdUpdateCounts(messageUpdateCounts, processedMessageCount), createFailures(failedMessages, processedMessageCount));
272     }
273     
274     private void markFailedMessages(final int[] messageUpdateCounts, final Map<Integer, SQLException> failedMessages,
275                                     final List<Integer> batchMessageIndexes, final Map<Integer, SQLException> failedOffsets) {
276         for (Entry<Integer, SQLException> each : failedOffsets.entrySet()) {
277             markFailedMessage(messageUpdateCounts, failedMessages, getBatchMessageIndex(batchMessageIndexes, each.getKey()), each.getValue());
278         }
279     }
280     
281     private int getBatchMessageIndex(final List<Integer> batchMessageIndexes, final int offset) {
282         if (batchMessageIndexes.isEmpty()) {
283             return 0;
284         }
285         return offset < batchMessageIndexes.size() ? batchMessageIndexes.get(offset) : batchMessageIndexes.get(batchMessageIndexes.size() - 1);
286     }
287     
288     private void markFailedMessage(final int[] messageUpdateCounts, final Map<Integer, SQLException> failedMessages, final int messageIndex, final SQLException failureCause) {
289         messageUpdateCounts[messageIndex] = Statement.EXECUTE_FAILED;
290         failedMessages.putIfAbsent(messageIndex, failureCause);
291     }
292     
293     private int getProcessedMessageCount(final int[] messageUpdateCounts, final NavigableMap<Integer, SQLException> failedMessages) throws SQLException {
294         int notExecutedMessageIndex = batchMessageCount;
295         for (int i = 0; i < batchMessageCount; i++) {
296             if (NOT_EXECUTED == messageUpdateCounts[i]) {
297                 notExecutedMessageIndex = i;
298                 break;
299             }
300         }
301         if (failedMessages.isEmpty()) {
302             return notExecutedMessageIndex;
303         }
304         Entry<Integer, SQLException> firstFailedMessage = failedMessages.firstEntry();
305         if (notExecutedMessageIndex <= firstFailedMessage.getKey()) {
306             throw firstFailedMessage.getValue();
307         }
308         return multiError ? notExecutedMessageIndex : Math.min(notExecutedMessageIndex, firstFailedMessage.getKey() + 1);
309     }
310     
311     private Collection<FirebirdBatchCompletion.Failure> createFailures(final Map<Integer, SQLException> failedMessages, final int processedMessageCount) {
312         Collection<FirebirdBatchCompletion.Failure> result = new LinkedList<>();
313         for (Entry<Integer, SQLException> each : failedMessages.entrySet()) {
314             if (each.getKey() < processedMessageCount) {
315                 result.add(new FirebirdBatchCompletion.Failure(each.getKey(), each.getValue()));
316             }
317         }
318         return result;
319     }
320     
321     private int[] toFirebirdUpdateCounts(final int[] messageUpdateCounts, final int processedMessageCount) {
322         int[] result = Arrays.copyOf(messageUpdateCounts, processedMessageCount);
323         for (int i = 0; i < result.length; i++) {
324             if (Statement.EXECUTE_FAILED == result[i]) {
325                 result[i] = FirebirdBatchCompletion.EXECUTE_FAILED;
326             }
327         }
328         return result;
329     }
330     
331     private List<BatchExecution> createBatchExecutionPlan() throws SQLException {
332         List<JDBCExecutionUnit> executionUnits = new LinkedList<>();
333         for (ExecutionGroup<JDBCExecutionUnit> eachGroup : executionGroupContext.getInputGroups()) {
334             executionUnits.addAll(eachGroup.getInputs());
335         }
336         return multiError ? createWholeUnitBatchExecutions(executionUnits) : createBatchExecutionsInMessageOrder(executionUnits);
337     }
338     
339     private List<BatchExecution> createWholeUnitBatchExecutions(final Collection<JDBCExecutionUnit> executionUnits) throws SQLException {
340         List<BatchExecution> result = new ArrayList<>(executionUnits.size());
341         for (JDBCExecutionUnit each : executionUnits) {
342             BatchExecution batchExecution = new BatchExecution(each, 0, executionUnitParams.getOrDefault(each.getExecutionUnit(), Collections.emptyList()).size());
343             addPendingBatch(batchExecution);
344             result.add(batchExecution);
345         }
346         return result;
347     }
348     
349     private List<BatchExecution> createBatchExecutionsInMessageOrder(final Collection<JDBCExecutionUnit> executionUnits) {
350         Map<Integer, Collection<JDBCExecutionUnit>> unitsByBatchMessageIndex = new TreeMap<>();
351         for (JDBCExecutionUnit each : executionUnits) {
352             for (int eachMessageIndex : executionUnitBatchMessageIndexes.getOrDefault(each.getExecutionUnit(), Collections.emptyList())) {
353                 unitsByBatchMessageIndex.computeIfAbsent(eachMessageIndex, key -> new LinkedList<>()).add(each);
354             }
355         }
356         List<BatchExecution> result = new ArrayList<>(batchMessageCount);
357         Map<JDBCExecutionUnit, Integer> consumedOffsets = new IdentityHashMap<>();
358         for (Entry<Integer, Collection<JDBCExecutionUnit>> entry : unitsByBatchMessageIndex.entrySet()) {
359             for (JDBCExecutionUnit each : entry.getValue()) {
360                 int offset = consumedOffsets.merge(each, 1, Integer::sum) - 1;
361                 result.add(new BatchExecution(each, offset, offset + 1));
362             }
363         }
364         return result;
365     }
366     
367     private void mergeUpdateCounts(final int[] result, final List<Integer> batchMessageIndexes, final int[] updateCounts) {
368         int countSize = Math.min(batchMessageIndexes.size(), updateCounts.length);
369         for (int i = 0; i < countSize; i++) {
370             int batchMessageIndex = batchMessageIndexes.get(i);
371             result[batchMessageIndex] = mergeUpdateCount(result[batchMessageIndex], updateCounts[i]);
372         }
373     }
374     
375     private int mergeUpdateCount(final int current, final int updateCount) {
376         if (NOT_EXECUTED == current) {
377             return updateCount;
378         }
379         if (Statement.EXECUTE_FAILED == updateCount || Statement.EXECUTE_FAILED == current) {
380             return Statement.EXECUTE_FAILED;
381         }
382         if (Statement.SUCCESS_NO_INFO == updateCount || Statement.SUCCESS_NO_INFO == current) {
383             return Statement.SUCCESS_NO_INFO;
384         }
385         return current + updateCount;
386     }

388     private final class BatchedStatementsJDBCExecutorCallback extends JDBCExecutorCallback<BatchExecutionUnitResult> {
389         
390         private BatchedStatementsJDBCExecutorCallback(final DatabaseType protocolType, final ResourceMetaData resourceMetaData, final SQLStatement sqlStatement, final boolean isExceptionThrown) {
391             super(protocolType, resourceMetaData, sqlStatement, isExceptionThrown);
392         }
393         
394         @Override
395         protected BatchExecutionUnitResult executeSQL(final String sql, final Statement statement, final ConnectionMode connectionMode, final DatabaseType storageType) throws SQLException {
396             return executeBatchedStatement((PreparedStatement) statement);
397         }
398         
399         private BatchExecutionUnitResult executeBatchedStatement(final PreparedStatement statement) throws SQLException {
400             List<List<Object>> params = pendingBatchParams.getOrDefault(statement, Collections.emptyList());
401             List<Integer> updateCounts = new LinkedList<>();
402             Map<Integer, SQLException> failures = new TreeMap<>();
403             while (true) {
404                 int executedCount = updateCounts.size();
405                 try {
406                     updateCounts.addAll(Ints.asList(statement.executeBatch()));
407                     return new BatchExecutionUnitResult(Ints.toArray(updateCounts), failures);
408                 } catch (final SQLException ex) {
409                     int[] executedUpdateCounts = getExecutedUpdateCounts(ex);
410                     updateCounts.addAll(Ints.asList(executedUpdateCounts));
411                     registerFailures(failures, ex, updateCounts, executedCount, params.size() - executedCount > executedUpdateCounts.length);
412                 }
413                 if (!multiError || updateCounts.size() >= params.size()) {
414                     return new BatchExecutionUnitResult(Ints.toArray(updateCounts), failures);
415                 }
416                 statement.clearBatch();
417                 addBatchedParameters(statement, params.subList(updateCounts.size(), params.size()));
418             }
419         }
420         
421         private int[] getExecutedUpdateCounts(final SQLException ex) {
422             if (!(ex instanceof BatchUpdateException)) {
423                 return new int[0];
424             }
425             int[] result = ((BatchUpdateException) ex).getUpdateCounts();
426             return null == result ? new int[0] : result;
427         }
428         
429         private void registerFailures(final Map<Integer, SQLException> failures, final SQLException ex, final List<Integer> updateCounts,
430                                       final int executedCount, final boolean isHaltedBeforeRemainingMessages) {
431             List<Integer> failedOffsets = new LinkedList<>();
432             for (int i = executedCount; i < updateCounts.size(); i++) {
433                 if (Statement.EXECUTE_FAILED == updateCounts.get(i)) {
434                     failedOffsets.add(i);
435                 }
436             }
437             if (isHaltedBeforeRemainingMessages) {
438                 failedOffsets.add(updateCounts.size());
439                 updateCounts.add(Statement.EXECUTE_FAILED);
440             }
441             if (failedOffsets.isEmpty()) {
442                 failedOffsets.add(Math.max(executedCount, updateCounts.size() - 1));
443             }
444             Iterator<SQLException> causes = getFailureCauses(ex, failedOffsets.size()).iterator();
445             for (int each : failedOffsets) {
446                 failures.putIfAbsent(each, causes.next());
447             }
448         }
449         
450         private Collection<SQLException> getFailureCauses(final SQLException ex, final int failedCount) {
451             Collection<SQLException> chainedCauses = new LinkedList<>();
452             for (SQLException each = ex.getNextException(); null != each; each = each.getNextException()) {
453                 chainedCauses.add(each);
454             }
455             return chainedCauses.size() == failedCount ? chainedCauses : Collections.nCopies(failedCount, ex);
456         }
457         
458         @Override
459         protected Optional<BatchExecutionUnitResult> getSaneResult(final SQLStatement sqlStatement, final SQLException ex) {
460             return Optional.empty();
461         }
462     }
