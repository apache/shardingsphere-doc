18  package org.apache.shardingsphere.test.e2e.mcp.llm.conversation.artifact;
19  
20  import org.apache.shardingsphere.infra.util.json.JsonEngine;
21  import org.apache.shardingsphere.test.e2e.mcp.llm.conversation.LLMConversationRunner.Result;
22  import org.apache.shardingsphere.test.e2e.mcp.support.artifact.MCPArtifactUtils;
23  
24  import java.io.IOException;
25  import java.nio.file.Files;
26  import java.nio.file.Path;
27  import java.util.Collection;
28  import java.util.Map;
29  
30  /**
31   * LLM conversation artifact writer.
32   */
33  public final class LLMConversationArtifactWriter {
34      
35      /**
36       * Write one conversation result.
37       *
38       * @param artifactDirectory artifact directory
39       * @param conversationResult conversation result
40       * @param sensitiveValues concrete sensitive values
41       * @throws IOException IO exception
42       */
43      public void write(final Path artifactDirectory, final Result conversationResult, final Collection<String> sensitiveValues) throws IOException {
44          writeContent(artifactDirectory.resolve("run-context.json"), JsonEngine.marshal(createRunContext(conversationResult)), sensitiveValues);
45          writeContent(artifactDirectory.resolve("system-prompt.md"), conversationResult.systemPrompt(), sensitiveValues);
46          writeContent(artifactDirectory.resolve("question.txt"), conversationResult.scenario().question(), sensitiveValues);
47          writeContent(artifactDirectory.resolve("answer.txt"), conversationResult.actualAnswer(), sensitiveValues);
48          writeContent(artifactDirectory.resolve("raw-model-output.txt"),
49                  String.join(System.lineSeparator() + System.lineSeparator(), conversationResult.evidence().rawModelOutputs()), sensitiveValues);
50          writeContent(artifactDirectory.resolve("available-tools.json"), JsonEngine.marshal(conversationResult.evidence().toolDefinitions()), sensitiveValues);
51          writeContent(artifactDirectory.resolve("interaction-trace.json"), JsonEngine.marshal(conversationResult.evidence().interactionTrace()), sensitiveValues);
52          writeContent(artifactDirectory.resolve("assertion-report.json"), JsonEngine.marshal(conversationResult.assertionReport()), sensitiveValues);
53      }
54      
55      private void writeContent(final Path file, final String content, final Collection<String> sensitiveValues) throws IOException {
56          Files.createDirectories(file.getParent());
57          Files.writeString(file, MCPArtifactUtils.redact(content, sensitiveValues));
58      }
59      
60      private Map<String, Object> createRunContext(final Result conversationResult) {
61          return Map.of(
62                  "scenarioId", conversationResult.scenario().id(),
63                  "modelProvider", conversationResult.modelProvider(),
64                  "modelName", conversationResult.modelName(),
65                  "failureType", conversationResult.assertionReport().getFailureType());
66      }
67  }
