68   * Insert SQL statement context.

70  public final class InsertStatementContext extends CommonSQLStatementContext implements TableAvailable, ParameterAware, WhereAvailable, WithAvailable {
71      
72      private final ShardingSphereMetaData metaData;
73      
74      private final String currentDatabaseName;
75      
76      private final Map<String, Integer> insertColumnNamesAndIndexes;
77      
78      private final List<List<ExpressionSegment>> valueExpressions;
79      
80      @Getter
81      private final TablesContext tablesContext;
82      
83      @Getter
84      private final List<String> columnNames;
85      
86      @Getter
87      private List<InsertValueContext> insertValueContexts;
88      
89      @Getter
90      private InsertSelectContext insertSelectContext;
91      
92      @Getter
93      private OnDuplicateUpdateContext onDuplicateKeyUpdateValueContext;
94      
95      private GeneratedKeyContext generatedKeyContext;
96      
97      public InsertStatementContext(final ShardingSphereMetaData metaData, final List<Object> params, final InsertStatement sqlStatement, final String currentDatabaseName) {
98          super(sqlStatement);
99          this.metaData = metaData;
100         this.currentDatabaseName = currentDatabaseName;
101         valueExpressions = getAllValueExpressions(sqlStatement);
102         AtomicInteger parametersOffset = new AtomicInteger(0);
103         insertValueContexts = getInsertValueContexts(params, parametersOffset, valueExpressions);
104         insertSelectContext = getInsertSelectContext(metaData, params, parametersOffset, currentDatabaseName).orElse(null);
105         onDuplicateKeyUpdateValueContext = getOnDuplicateKeyUpdateValueContext(params, parametersOffset).orElse(null);
106         tablesContext = new TablesContext(getAllSimpleTableSegments());
107         List<String> insertColumnNames = getInsertColumnNames();
108         ShardingSphereSchema schema = getSchema(metaData, currentDatabaseName);
109         columnNames = containsInsertColumns()
110                 ? insertColumnNames
111                 : sqlStatement.getTable().map(optional -> schema.getVisibleColumnNames(optional.getTableName().getIdentifier().getValue())).orElseGet(Collections::emptyList);
112         insertColumnNamesAndIndexes = createInsertColumnNamesAndIndexes(insertColumnNames);
113         generatedKeyContext = new GeneratedKeyContextEngine(sqlStatement, schema).createGenerateKeyContext(insertColumnNamesAndIndexes, insertValueContexts, params).orElse(null);
114     }
115     
116     private Map<String, Integer> createInsertColumnNamesAndIndexes(final List<String> insertColumnNames) {
117         if (containsInsertColumns()) {
118             Map<String, Integer> result = new CaseInsensitiveMap<>(insertColumnNames.size(), 1F);
119             int index = 0;
120             for (String each : insertColumnNames) {
121                 result.put(each, index++);
122             }
123             return result;
124         }
125         return Collections.emptyMap();
126     }
127     
128     private List<InsertValueContext> getInsertValueContexts(final List<Object> params, final AtomicInteger paramsOffset, final List<List<ExpressionSegment>> valueExpressions) {
129         List<InsertValueContext> result = new LinkedList<>();
130         for (Collection<ExpressionSegment> each : valueExpressions) {
131             InsertValueContext insertValueContext = new InsertValueContext(each, params, paramsOffset.get());
132             result.add(insertValueContext);
133             paramsOffset.addAndGet(insertValueContext.getParameterCount());
134         }
135         return result;
136     }
137     
138     private Optional<InsertSelectContext> getInsertSelectContext(final ShardingSphereMetaData metaData, final List<Object> params,
139                                                                  final AtomicInteger paramsOffset, final String currentDatabaseName) {
140         if (!getSqlStatement().getInsertSelect().isPresent()) {
141             return Optional.empty();
142         }
143         SubquerySegment insertSelectSegment = getSqlStatement().getInsertSelect().get();
144         SelectStatementContext selectStatementContext = new SelectStatementContext(metaData, params, insertSelectSegment.getSelect(), currentDatabaseName, Collections.emptyList());
145         selectStatementContext.setSubqueryType(SubqueryType.INSERT_SELECT);
146         setCombineSelectSubqueryType(selectStatementContext);
147         setProjectionSelectSubqueryType(selectStatementContext);
148         InsertSelectContext insertSelectContext = new InsertSelectContext(selectStatementContext, params, paramsOffset.get());
149         paramsOffset.addAndGet(insertSelectContext.getParameterCount());
150         return Optional.of(insertSelectContext);
151     }
152     
153     private void setCombineSelectSubqueryType(final SelectStatementContext selectStatementContext) {
154         if (selectStatementContext.getSqlStatement().getCombine().isPresent()) {
155             CombineSegment combineSegment = selectStatementContext.getSqlStatement().getCombine().get();
156             Optional.ofNullable(selectStatementContext.getSubqueryContexts().get(combineSegment.getLeft().getStartIndex()))
157                     .ifPresent(optional -> optional.setSubqueryType(SubqueryType.INSERT_SELECT));
158             Optional.ofNullable(selectStatementContext.getSubqueryContexts().get(combineSegment.getRight().getStartIndex()))
159                     .ifPresent(optional -> optional.setSubqueryType(SubqueryType.INSERT_SELECT));
160         }
161     }
162     
163     private void setProjectionSelectSubqueryType(final SelectStatementContext selectStatementContext) {
164         for (Entry<Integer, SelectStatementContext> entry : selectStatementContext.getSubqueryContexts().entrySet()) {
165             if (entry.getKey() >= selectStatementContext.getProjectionsContext().getStartIndex() && entry.getKey() <= selectStatementContext.getProjectionsContext().getStopIndex()) {
166                 entry.getValue().setSubqueryType(SubqueryType.INSERT_SELECT);
167             }
168         }
169     }
170     
171     private Optional<OnDuplicateUpdateContext> getOnDuplicateKeyUpdateValueContext(final List<Object> params, final AtomicInteger parametersOffset) {
172         Optional<OnDuplicateKeyColumnsSegment> onDuplicateKeyColumnsSegment = getSqlStatement().getOnDuplicateKeyColumns();
173         if (!onDuplicateKeyColumnsSegment.isPresent()) {
174             return Optional.empty();
175         }
176         Collection<ColumnAssignmentSegment> onDuplicateKeyColumns = onDuplicateKeyColumnsSegment.get().getColumns();
177         OnDuplicateUpdateContext onDuplicateUpdateContext = new OnDuplicateUpdateContext(onDuplicateKeyColumns, params, parametersOffset.get());
178         parametersOffset.addAndGet(onDuplicateUpdateContext.getParameterCount());
179         return Optional.of(onDuplicateUpdateContext);
180     }
181     
182     private Collection<SimpleTableSegment> getAllSimpleTableSegments() {
183         TableExtractor tableExtractor = new TableExtractor();
184         tableExtractor.extractTablesFromInsert(getSqlStatement());
185         return tableExtractor.getRewriteTables();
186     }
187     
188     private ShardingSphereSchema getSchema(final ShardingSphereMetaData metaData, final String currentDatabaseName) {
189         String databaseName = tablesContext.getDatabaseName().orElse(currentDatabaseName);
190         ShardingSpherePreconditions.checkNotNull(databaseName, NoDatabaseSelectedException::new);
191         ShardingSphereDatabase database = metaData.getDatabase(databaseName);
192         ShardingSpherePreconditions.checkNotNull(database, () -> new UnknownDatabaseException(databaseName));
193         String defaultSchema = new DatabaseTypeRegistry(getDatabaseType()).getDefaultSchemaName(databaseName);
194         return tablesContext.getSchemaName().map(database::getSchema).orElseGet(() -> database.getSchema(defaultSchema));
195     }

197     /**
198      * Get column names for descending order.
199      *
200      * @return column names for descending order
201      */
202     public Iterator<String> getDescendingColumnNames() {
203         return new LinkedList<>(columnNames).descendingIterator();
204     }

206     /**
207      * Get grouped parameters.
208      *
209      * @return grouped parameters
210      */
211     public List<List<Object>> getGroupedParameters() {
212         List<List<Object>> result = new LinkedList<>();
213         for (InsertValueContext each : insertValueContexts) {
214             result.add(each.getParameters());
215         }
216         if (null != insertSelectContext && !insertSelectContext.getParameters().isEmpty()) {
217             result.add(insertSelectContext.getParameters());
218         }
219         return result;
220     }

222     /**
223      * Get on duplicate key update parameters.
224      *
225      * @return on duplicate key update parameters
226      */
227     public List<Object> getOnDuplicateKeyUpdateParameters() {
228         return null == onDuplicateKeyUpdateValueContext ? new ArrayList<>() : onDuplicateKeyUpdateValueContext.getParameters();
229     }

231     /**
232      * Get generated key context.
233      *
234      * @return generated key context
235      */
236     public Optional<GeneratedKeyContext> getGeneratedKeyContext() {
237         return Optional.ofNullable(generatedKeyContext);
238     }

240     /**
241      * Judge whether contains insert columns.
242      *
243      * @return contains insert columns or not
244      */
245     public boolean containsInsertColumns() {
246         InsertStatement insertStatement = getSqlStatement();
247         return !insertStatement.getColumns().isEmpty() || insertStatement.getSetAssignment().isPresent();
248     }
249     
250     /**
251      * Get value list count.
252      *
253      * @return value list count
254      */
255     public int getValueListCount() {
256         InsertStatement insertStatement = getSqlStatement();
257         return insertStatement.getSetAssignment().isPresent() ? 1 : insertStatement.getValues().size();
258     }
259     
260     /**
261      * Get insert column names.
262      *
263      * @return column names collection
264      */
265     public List<String> getInsertColumnNames() {
266         return getSqlStatement().getSetAssignment().map(this::getColumnNamesForSetAssignment).orElseGet(() -> getColumnNamesForInsertColumns(getSqlStatement().getColumns()));
267     }
268     
269     private List<String> getColumnNamesForSetAssignment(final SetAssignmentSegment setAssignment) {
270         List<String> result = new LinkedList<>();
271         for (ColumnAssignmentSegment each : setAssignment.getAssignments()) {
272             result.add(each.getColumns().get(0).getIdentifier().getValue().toLowerCase());
273         }
274         return result;
275     }
276     
277     private List<String> getColumnNamesForInsertColumns(final Collection<ColumnSegment> columns) {
278         List<String> result = new LinkedList<>();
279         for (ColumnSegment each : columns) {
280             result.add(each.getIdentifier().getValue().toLowerCase());
281         }
282         return result;
283     }
284     
285     private List<List<ExpressionSegment>> getAllValueExpressions(final InsertStatement insertStatement) {
286         Optional<SetAssignmentSegment> setAssignment = insertStatement.getSetAssignment();
287         return setAssignment
288                 .map(optional -> Collections.singletonList(getAllValueExpressionsFromSetAssignment(optional))).orElseGet(() -> getAllValueExpressionsFromValues(insertStatement.getValues()));
289     }
290     
291     private List<ExpressionSegment> getAllValueExpressionsFromSetAssignment(final SetAssignmentSegment setAssignment) {
292         List<ExpressionSegment> result = new ArrayList<>(setAssignment.getAssignments().size());
293         for (ColumnAssignmentSegment each : setAssignment.getAssignments()) {
294             result.add(each.getValue());
295         }
296         return result;
297     }
298     
299     private List<List<ExpressionSegment>> getAllValueExpressionsFromValues(final Collection<InsertValuesSegment> values) {
300         List<List<ExpressionSegment>> result = new ArrayList<>(values.size());
301         for (InsertValuesSegment each : values) {
302             result.add(each.getValues());
303         }
304         return result;
305     }

307     @Override
308     public InsertStatement getSqlStatement() {
309         return (InsertStatement) super.getSqlStatement();
310     }
311     
312     @Override
313     public void setUpParameters(final List<Object> params) {
314         AtomicInteger parametersOffset = new AtomicInteger(0);
315         insertValueContexts = getInsertValueContexts(params, parametersOffset, valueExpressions);
316         insertSelectContext = getInsertSelectContext(metaData, params, parametersOffset, currentDatabaseName).orElse(null);
317         onDuplicateKeyUpdateValueContext = getOnDuplicateKeyUpdateValueContext(params, parametersOffset).orElse(null);
318         ShardingSphereSchema schema = getSchema(metaData, currentDatabaseName);
319         generatedKeyContext = new GeneratedKeyContextEngine(getSqlStatement(), schema).createGenerateKeyContext(insertColumnNamesAndIndexes, insertValueContexts, params).orElse(null);
320     }
