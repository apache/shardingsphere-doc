65      @Test
66      void assertBindExistsSubqueryWithShadowedOuterTableContext() {
67          SubquerySegment subquerySegment = new SubquerySegment(0, 5, mock(SelectStatement.class), "SELECT 1");
68          ExistsSubqueryExpression boundSegment = new ExistsSubqueryExpression(0, 10, subquerySegment);
69          TableSegmentBinderContext outerTableBinderContext = mock(TableSegmentBinderContext.class);
70          TableSegmentBinderContext retainedOuterTableBinderContext = mock(TableSegmentBinderContext.class);
71          TableSegmentBinderContext currentTableBinderContext = mock(TableSegmentBinderContext.class);
72          Multimap<CaseInsensitiveString, TableSegmentBinderContext> outerTableBinderContexts = LinkedHashMultimap.create();
73          outerTableBinderContexts.put(CaseInsensitiveString.of("t"), outerTableBinderContext);
74          outerTableBinderContexts.put(CaseInsensitiveString.of("u"), retainedOuterTableBinderContext);
75          Multimap<CaseInsensitiveString, TableSegmentBinderContext> tableBinderContexts = LinkedHashMultimap.create();
76          tableBinderContexts.put(CaseInsensitiveString.of("T"), currentTableBinderContext);
77          AtomicReference<Multimap<CaseInsensitiveString, TableSegmentBinderContext>> mergedContexts = new AtomicReference<>();
78          when(ExistsSubqueryExpressionBinder.bind(any(), any(), any())).thenAnswer(invocation -> {
79              mergedContexts.set(invocation.getArgument(2));
80              return boundSegment;
81          });
82          ExistsSubqueryExpression segment = new ExistsSubqueryExpression(0, 10, subquerySegment);
83          SQLStatementBinderContext binderContext = mock(SQLStatementBinderContext.class);
84          assertThat(ExpressionSegmentBinder.bind(segment, SegmentType.PREDICATE, binderContext, tableBinderContexts, outerTableBinderContexts), is(boundSegment));
85          assertThat(mergedContexts.get().get(CaseInsensitiveString.of("t")), hasItem(currentTableBinderContext));
86          assertThat(mergedContexts.get().get(CaseInsensitiveString.of("u")), contains(retainedOuterTableBinderContext));
87      }

91          SubquerySegment boundSubquerySegment = new SubquerySegment(0, 5, mock(SelectStatement.class), "SELECT 1");
92          TableSegmentBinderContext outerTableBinderContext = mock(TableSegmentBinderContext.class);
93          TableSegmentBinderContext retainedOuterTableBinderContext = mock(TableSegmentBinderContext.class);
94          TableSegmentBinderContext currentTableBinderContext = mock(TableSegmentBinderContext.class);
95          Multimap<CaseInsensitiveString, TableSegmentBinderContext> outerTableBinderContexts = LinkedHashMultimap.create();
96          outerTableBinderContexts.put(CaseInsensitiveString.of("t"), outerTableBinderContext);
97          outerTableBinderContexts.put(CaseInsensitiveString.of("u"), retainedOuterTableBinderContext);
98          Multimap<CaseInsensitiveString, TableSegmentBinderContext> tableBinderContexts = LinkedHashMultimap.create();
99          tableBinderContexts.put(CaseInsensitiveString.of("T"), currentTableBinderContext);
100         AtomicReference<Multimap<CaseInsensitiveString, TableSegmentBinderContext>> mergedContexts = new AtomicReference<>();
101         when(SubquerySegmentBinder.bind(any(), any(), any())).thenAnswer(invocation -> {
102             mergedContexts.set(invocation.getArgument(2));
103             return boundSubquerySegment;
104         });
105         SubquerySegment subquerySegment = new SubquerySegment(0, 5, mock(SelectStatement.class), "SELECT 1");
106         SubqueryExpressionSegment segment = new SubqueryExpressionSegment(subquerySegment);
107         SQLStatementBinderContext binderContext = mock(SQLStatementBinderContext.class);
108         SubqueryExpressionSegment actual = (SubqueryExpressionSegment) ExpressionSegmentBinder.bind(
109                 segment, SegmentType.PREDICATE, binderContext, tableBinderContexts, outerTableBinderContexts);
110         assertThat(actual.getSubquery(), is(boundSubquerySegment));
111         assertThat(mergedContexts.get().get(CaseInsensitiveString.of("t")), hasItem(currentTableBinderContext));
112         assertThat(mergedContexts.get().get(CaseInsensitiveString.of("u")), contains(retainedOuterTableBinderContext));
113     }

115     @Test
116     void assertBindCommonExpression() {
117         ColumnSegment column = new ColumnSegment(0, 9, new IdentifierValue("foo_column"));
118         ColumnSegment boundColumn = new ColumnSegment(0, 9, new IdentifierValue("foo_column"));
119         SubquerySegment subquery = new SubquerySegment(13, 22, mock(SelectStatement.class), "(SELECT 1)");
120         SubquerySegment boundSubquery = new SubquerySegment(13, 22, mock(SelectStatement.class), "(SELECT 1)");
121         SubqueryExpressionSegment subqueryExpression = new SubqueryExpressionSegment(subquery);
122         ParameterMarkerExpressionSegment parameterMarker = new ParameterMarkerExpressionSegment(26, 26, 0);
123         CommonExpressionSegment segment = new CommonExpressionSegment(0, 26, "foo_column + (SELECT 1) + ?", Arrays.asList(column, subqueryExpression, parameterMarker));
124         SQLStatementBinderContext binderContext = mock(SQLStatementBinderContext.class);
125         Multimap<CaseInsensitiveString, TableSegmentBinderContext> tableBinderContexts = LinkedHashMultimap.create();
126         Multimap<CaseInsensitiveString, TableSegmentBinderContext> outerTableBinderContexts = LinkedHashMultimap.create();
127         when(ColumnSegmentBinder.bind(same(column), eq(SegmentType.PREDICATE), same(binderContext), same(tableBinderContexts), same(outerTableBinderContexts))).thenReturn(boundColumn);
128         when(SubquerySegmentBinder.bind(same(subquery), same(binderContext), any())).thenReturn(boundSubquery);
129         CommonExpressionSegment actual = (CommonExpressionSegment) ExpressionSegmentBinder.bind(segment, SegmentType.PREDICATE, binderContext, tableBinderContexts, outerTableBinderContexts);
130         assertThat(actual, not(sameInstance(segment)));
131         assertThat(actual.getStartIndex(), is(segment.getStartIndex()));
132         assertThat(actual.getStopIndex(), is(segment.getStopIndex()));
133         assertThat(actual.getText(), is(segment.getText()));
134         Iterator<ExpressionSegment> actualChildren = actual.getChildren().iterator();
135         assertThat(actualChildren.next(), sameInstance(boundColumn));
136         assertThat(((SubqueryExpressionSegment) actualChildren.next()).getSubquery(), sameInstance(boundSubquery));
137         assertThat(actualChildren.next(), sameInstance(parameterMarker));
138         assertFalse(actualChildren.hasNext());
139         assertThat(segment.getChildren(), contains(column, subqueryExpression, parameterMarker));
140         assertThat(subqueryExpression.getSubquery(), sameInstance(subquery));
141     }

143     @Test
144     void assertBindNestedCommonExpression() {
145         ColumnSegment column = new ColumnSegment(1, 10, new IdentifierValue("foo_column"));
146         ColumnSegment boundColumn = new ColumnSegment(1, 10, new IdentifierValue("foo_column"));
147         CommonExpressionSegment nestedSegment = new CommonExpressionSegment(1, 10, "foo_column", Collections.singleton(column));
148         CommonExpressionSegment segment = new CommonExpressionSegment(0, 11, "(foo_column)", Collections.singleton(nestedSegment));
149         SQLStatementBinderContext binderContext = mock(SQLStatementBinderContext.class);
150         Multimap<CaseInsensitiveString, TableSegmentBinderContext> tableBinderContexts = LinkedHashMultimap.create();
151         Multimap<CaseInsensitiveString, TableSegmentBinderContext> outerTableBinderContexts = LinkedHashMultimap.create();
152         when(ColumnSegmentBinder.bind(same(column), eq(SegmentType.PROJECTION), same(binderContext), same(tableBinderContexts), same(outerTableBinderContexts))).thenReturn(boundColumn);
153         CommonExpressionSegment actual = (CommonExpressionSegment) ExpressionSegmentBinder.bind(segment, SegmentType.PROJECTION, binderContext, tableBinderContexts, outerTableBinderContexts);
154         CommonExpressionSegment actualNestedSegment = (CommonExpressionSegment) actual.getChildren().iterator().next();
155         assertThat(actual, not(sameInstance(segment)));
156         assertThat(actualNestedSegment, not(sameInstance(nestedSegment)));
157         assertThat(actualNestedSegment.getStartIndex(), is(nestedSegment.getStartIndex()));
158         assertThat(actualNestedSegment.getStopIndex(), is(nestedSegment.getStopIndex()));
159         assertThat(actualNestedSegment.getText(), is(nestedSegment.getText()));
160         assertThat(actualNestedSegment.getChildren(), contains(boundColumn));
161         assertThat(segment.getChildren(), contains(nestedSegment));
162         assertThat(nestedSegment.getChildren(), contains(column));
163     }
