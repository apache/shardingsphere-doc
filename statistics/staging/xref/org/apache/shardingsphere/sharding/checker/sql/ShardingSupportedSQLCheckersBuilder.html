40  import java.util.Arrays;
41  import java.util.Collection;
42  
43  /**
44   * Sharding SQL supported checker factory.
45   */
46  public final class ShardingSupportedSQLCheckersBuilder implements SupportedSQLCheckersBuilder<ShardingRule> {
47      
48      private final Collection<SupportedSQLChecker<?, ShardingRule>> supportedSQLCheckers = Arrays.asList(
49              new ShardingAlterIndexSupportedChecker(),
50              new ShardingAlterTableSupportedChecker(),
51              new ShardingAlterViewSupportedChecker(),
52              new ShardingCreateFunctionSupportedChecker(),
53              new ShardingCreateIndexSupportedChecker(),
54              new ShardingCreateProcedureSupportedChecker(),
55              new ShardingCreateTableSupportedChecker(),
56              new ShardingCreateViewSupportedChecker(),
57              new ShardingDropIndexSupportedChecker(),
58              new ShardingDropTableSupportedChecker(),
59              new ShardingRenameTableSupportedChecker(),
60              new ShardingInsertSupportedChecker(),
61              new ShardingTableSupportedChecker(),
62              new ShardingMultipleTablesSupportedChecker());
63      
64      @HighFrequencyInvocation
65      @Override
66      public Collection<SupportedSQLChecker<?, ShardingRule>> getSupportedSQLCheckers() {
67          return supportedSQLCheckers;
68      }
69      
70      @Override
71      public int getOrder() {
72          return ShardingOrder.ORDER;
73      }
74      
75      @Override
76      public Class<ShardingRule> getTypeClass() {
77          return ShardingRule.class;
78      }
79  }
