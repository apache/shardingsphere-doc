47  /**
48   * Standard DCL statement assert.
49   */
50  @NoArgsConstructor(access = AccessLevel.PRIVATE)
51  public final class StandardDCLStatementAssert {
52      
53      /**
54       * Assert DCL statement is correct with expected parser result.
55       *
56       * @param assertContext assert context
57       * @param actual actual DCL statement
58       * @param expected expected DCL statement test case
59       */
60      public static void assertIs(final SQLCaseAssertContext assertContext, final DCLStatement actual, final SQLParserTestCase expected) {
61          if (actual instanceof CreateUserStatement) {
62              CreateUserStatementAssert.assertIs(assertContext, (CreateUserStatement) actual, (CreateUserStatementTestCase) expected);
63          } else if (actual instanceof AlterUserStatement) {
64              AlterUserStatementAssert.assertIs(assertContext, (AlterUserStatement) actual, (AlterUserStatementTestCase) expected);
65          } else if (actual instanceof DropUserStatement) {
66              DropUserStatementAssert.assertIs(assertContext, (DropUserStatement) actual, (DropUserStatementTestCase) expected);
67          } else if (actual instanceof CreateRoleStatement) {
68              CreateRoleStatementAssert.assertIs(assertContext, (CreateRoleStatement) actual, (CreateRoleStatementTestCase) expected);
69          } else if (actual instanceof AlterRoleStatement) {
70              AlterRoleStatementAssert.assertIs(assertContext, (AlterRoleStatement) actual, (AlterRoleStatementTestCase) expected);
71          } else if (actual instanceof DropRoleStatement) {
72              DropRoleStatementAssert.assertIs(assertContext, (DropRoleStatement) actual, (DropRoleStatementTestCase) expected);
73          } else if (actual instanceof SetRoleStatement) {
74              SetRoleStatementAssert.assertIs(assertContext, (SetRoleStatement) actual, (SetRoleStatementTestCase) expected);
75          }
76      }
77  }
