108     void assertLoadWithoutSystemCatalogForDatabaseAsSchema() throws SQLException {
109         Connection connection = mock(Connection.class);
110         DatabaseMetaData databaseMetaData = mock(DatabaseMetaData.class);
111         when(connection.getMetaData()).thenReturn(databaseMetaData);
112         when(databaseMetaData.getDatabaseProductVersion()).thenReturn("");
113         when(databaseMetaData.getURL()).thenReturn(getMetadataJdbcUrl("MySQL"));
114         mockEmptyScalarQueries(connection);
115         when(databaseMetaData.getTables(isNull(), isNull(), eq("%"), any(String[].class))).thenAnswer(invocation -> {
116             String[] tableTypes = invocation.getArgument(3);
117             return mockMultiRowResultSet("TABLE".equals(tableTypes[0])
118                     ? List.of(
119                             Map.of("TABLE_CAT", "information_schema", "TABLE_SCHEM", "", "TABLE_NAME", "GLOBAL_STATUS"),
120                             Map.of("TABLE_CAT", "logic_db", "TABLE_SCHEM", "", "TABLE_NAME", "orders"))
121                     : List.of());
122         });
123         ResultSet orderColumns = mockResultSet("COLUMN_NAME", "order_id");
124         when(databaseMetaData.getColumns(eq("logic_db"), isNull(), eq("orders"), eq("%"))).thenReturn(orderColumns);
125         ResultSet orderIndexes = mockResultSet("INDEX_NAME", "idx_orders_status");
126         when(databaseMetaData.getIndexInfo(eq("logic_db"), isNull(), eq("orders"), eq(false), eq(false))).thenReturn(orderIndexes);
127         when(databaseMetaData.getColumns(eq("information_schema"), isNull(), eq("GLOBAL_STATUS"), eq("%")))
128                 .thenThrow(new SQLException("system catalog should be skipped"));
129         RuntimeDatabaseConfiguration runtimeDatabaseConfiguration = createMockRuntimeDatabaseConfiguration(connection);
130         Collection<ShardingSphereSchema> actual = load(Map.of("logic_db", runtimeDatabaseConfiguration), List.of(),
131                 Map.of("MySQL", List.of("information_schema")), Map.of("MySQL", DialectSchemaSemantics.DATABASE_AS_SCHEMA)).findMetadata("logic_db").orElseThrow();
132         assertTrue(containsMetadata(actual, SupportedMCPMetadataObjectType.TABLE, "orders"));
133         assertFalse(containsMetadata(actual, SupportedMCPMetadataObjectType.TABLE, "GLOBAL_STATUS"));
134         assertThat(actual.iterator().next().getName(), is("logic_db"));
135     }
