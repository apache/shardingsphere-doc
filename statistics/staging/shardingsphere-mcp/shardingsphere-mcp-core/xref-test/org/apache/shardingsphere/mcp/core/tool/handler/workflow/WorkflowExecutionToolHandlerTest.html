45  class WorkflowExecutionToolHandlerTest {
46      
47      @Test
48      void assertGetExecutionToolDescriptor() {
49          WorkflowExecutionToolHandler handler = new WorkflowExecutionToolHandler(new WorkflowRuntimeDefinitionRegistry(List.of(WorkflowHandlerTestFixture.createDefinition("encrypt.rule"))));
50          MCPToolDescriptor actual = MCPDescriptorCatalogIndex.getRequiredToolDescriptor(handler.getToolName());
51          assertThat(actual.getName(), is("database_gateway_apply_workflow"));
52      }
53      
54      @Test
55      void assertHandleExecution() {
56          WorkflowContextSnapshot snapshot = WorkflowHandlerTestFixture.createSnapshot();
57          WorkflowHandlerTestFixture.Context fixture = WorkflowHandlerTestFixture.createContext(snapshot);
58          MCPWorkflowRuntimeHandler workflowRuntimeHandler = mock(MCPWorkflowRuntimeHandler.class);
59          MCPWorkflowApplyArtifactValidator workflowApplyArtifactValidator = mock(MCPWorkflowApplyArtifactValidator.class);
60          try (
61                  MockedConstruction<WorkflowExecutionService> mockedExecutionServices = mockConstruction(WorkflowExecutionService.class,
62                          (mock, context) -> when(mock.apply(any(), any(), any(), any(), any(), any(), any(), any(), any())).thenReturn(Map.of("status", "completed")))) {
63              WorkflowExecutionToolHandler handler = new WorkflowExecutionToolHandler(new WorkflowRuntimeDefinitionRegistry(List.of(
64                      WorkflowHandlerTestFixture.createDefinition("encrypt.rule", workflowRuntimeHandler, workflowApplyArtifactValidator))));
65              MCPSuccessPayload actual = handler.handle(fixture.requestContext(), Map.of(WorkflowFieldNames.PLAN_ID, "plan-1",
66                      WorkflowFieldNames.APPROVED_STEPS, List.of("rule_distsql"), WorkflowFieldNames.EXECUTION_MODE, "manual-only"));
67              WorkflowExecutionService executionService = mockedExecutionServices.constructed().getFirst();
68              verify(executionService).apply(fixture.workflowSessionContext(), fixture.queryFacade(), fixture.executionFacade(),
69                      workflowRuntimeHandler, workflowApplyArtifactValidator, "session-1", snapshot, List.of("rule_distsql"), "manual-only");
70              assertThat(actual.toPayload().get("status"), is("completed"));
71          }
72      }
73      
74      @Test
75      void assertHandleExecutionWithMissingWorkflowKind() {
76          WorkflowContextSnapshot snapshot = WorkflowHandlerTestFixture.createSnapshotWithoutWorkflowKind();
77          WorkflowHandlerTestFixture.Context fixture = WorkflowHandlerTestFixture.createContext(snapshot);
78          WorkflowExecutionToolHandler handler = new WorkflowExecutionToolHandler(new WorkflowRuntimeDefinitionRegistry(List.of(WorkflowHandlerTestFixture.createDefinition("encrypt.rule"))));
79          MCPInvalidRequestException actual = assertThrows(MCPInvalidRequestException.class,
80                  () -> handler.handle(fixture.requestContext(), Map.of(WorkflowFieldNames.PLAN_ID, "plan-1", WorkflowFieldNames.EXECUTION_MODE, "manual-only")));
81          assertThat(actual.getMessage(), is("Workflow kind is required for plan_id `plan-1`."));
82      }
83      
84      @Test
85      void assertHandleExecutionWithMissingExecutionMode() {
86          WorkflowContextSnapshot snapshot = WorkflowHandlerTestFixture.createSnapshot();
87          WorkflowHandlerTestFixture.Context fixture = WorkflowHandlerTestFixture.createContext(snapshot);
88          WorkflowExecutionToolHandler handler = new WorkflowExecutionToolHandler(new WorkflowRuntimeDefinitionRegistry(List.of(WorkflowHandlerTestFixture.createDefinition("encrypt.rule"))));
89          MCPInvalidRequestException actual = assertThrows(MCPInvalidRequestException.class,
90                  () -> handler.handle(fixture.requestContext(), Map.of(WorkflowFieldNames.PLAN_ID, "plan-1")));
91          assertThat(actual.getMessage(), is("database_gateway_apply_workflow execution_mode is required."));
92      }
