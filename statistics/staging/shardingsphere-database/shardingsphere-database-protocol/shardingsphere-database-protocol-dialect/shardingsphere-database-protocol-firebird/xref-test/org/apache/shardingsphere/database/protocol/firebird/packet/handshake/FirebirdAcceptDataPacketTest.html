18  package org.apache.shardingsphere.database.protocol.firebird.packet.handshake;
19  
20  import org.apache.shardingsphere.database.protocol.firebird.constant.FirebirdAuthenticationMethod;
21  import org.apache.shardingsphere.database.protocol.firebird.payload.FirebirdPacketPayload;
22  import org.apache.shardingsphere.database.protocol.payload.PacketPayload;
23  import org.junit.jupiter.api.extension.ExtendWith;
24  import org.junit.jupiter.params.ParameterizedTest;
25  import org.junit.jupiter.params.provider.Arguments;
26  import org.junit.jupiter.params.provider.MethodSource;
27  import org.mockito.InOrder;
28  import org.mockito.Mock;
29  import org.mockito.junit.jupiter.MockitoExtension;
30  
31  import java.nio.charset.StandardCharsets;
32  import java.util.stream.Stream;
33  
34  import static org.mockito.Mockito.inOrder;
35  
36  @ExtendWith(MockitoExtension.class)
37  class FirebirdAcceptDataPacketTest {
38      
39      @Mock
40      private FirebirdPacketPayload payload;
41      
42      @ParameterizedTest(name = "{0}")
43      @MethodSource("writeArguments")
44      void assertWrite(final String name, final byte[] salt, final String publicKey, final FirebirdAuthenticationMethod plugin,
45                       final int authenticated, final String keys, final int expectedPayloadLength, final boolean expectedPayloadData) {
46          new FirebirdAcceptDataPacket(salt, publicKey, plugin, authenticated, keys).write((PacketPayload) payload);
47          InOrder order = inOrder(payload);
48          order.verify(payload).writeInt4(expectedPayloadLength);
49          verifyPayloadData(order, salt, publicKey, expectedPayloadLength, expectedPayloadData);
50          order.verify(payload).writeString(plugin.getMethodName());
51          order.verify(payload).writeInt4(authenticated);
52          order.verify(payload).writeString(keys);
53          order.verifyNoMoreInteractions();
54      }
55      
56      private void verifyPayloadData(final InOrder order, final byte[] salt, final String publicKey, final int expectedPayloadLength, final boolean expectedPayloadData) {
57          if (!expectedPayloadData) {
58              return;
59          }
60          order.verify(payload).writeInt2LE(salt.length);
61          order.verify(payload).writeBytes(salt);
62          order.verify(payload).writeInt2LE(publicKey.length());
63          order.verify(payload).writeBytes(publicKey.getBytes(StandardCharsets.US_ASCII));
64          order.verify(payload).writePadding(expectedPayloadLength);
65      }
66      
67      private static Stream<Arguments> writeArguments() {
68          return Stream.of(
69                  Arguments.of("salt_and_public_key", new byte[]{1, 2}, "key", FirebirdAuthenticationMethod.SRP, 1, "k", 9, true),
70                  Arguments.of("empty_salt", new byte[0], "key", FirebirdAuthenticationMethod.SRP224, 0, "", 0, false),
71                  Arguments.of("empty_public_key", new byte[]{1, 2}, "", FirebirdAuthenticationMethod.LEGACY_AUTH, 7, "keys", 0, false));
72      }
73  }
