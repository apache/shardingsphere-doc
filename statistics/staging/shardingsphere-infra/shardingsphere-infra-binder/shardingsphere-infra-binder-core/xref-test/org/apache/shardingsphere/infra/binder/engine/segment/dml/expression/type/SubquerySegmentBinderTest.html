61  class SubquerySegmentBinderTest {
62      
63      @Test
64      void assertBind() {
65          SelectStatement selectStatement = new SQL92SelectStatement();
66          ColumnSegment columnSegment = new ColumnSegment(58, 65, new IdentifierValue("order_id"));
67          ProjectionsSegment projectionsSegment = new ProjectionsSegment(58, 65);
68          projectionsSegment.getProjections().add(new ColumnProjectionSegment(columnSegment));
69          selectStatement.setProjections(projectionsSegment);
70          selectStatement.setFrom(new SimpleTableSegment(new TableNameSegment(72, 78, new IdentifierValue("t_order"))));
71          ExpressionSegment whereExpressionSegment = new ColumnSegment(86, 91, new IdentifierValue("status"));
72          selectStatement.setWhere(new WhereSegment(80, 102, whereExpressionSegment));
73          SubquerySegment subquerySegment = new SubquerySegment(39, 103, selectStatement, "order_id = (SELECT order_id FROM t_order WHERE status = 'SUBMIT')");
74          SQLStatementBinderContext sqlStatementBinderContext = new SQLStatementBinderContext(createMetaData(), "foo_db", new HintValueContext(), mock(SQLStatement.class));
75          ColumnSegment boundNameColumn = new ColumnSegment(7, 13, new IdentifierValue("user_id"));
76          boundNameColumn.setColumnBoundInfo(new ColumnSegmentBoundInfo(new TableSegmentBoundInfo(
77                  new IdentifierValue("foo_db"), new IdentifierValue("foo_db")), new IdentifierValue("t_order_item"), new IdentifierValue("user_id"), TableSourceType.TEMPORARY_TABLE));
78          sqlStatementBinderContext.getExternalTableBinderContexts().put(new CaseInsensitiveString("t_order_item"),
79                  new SimpleTableSegmentBinderContext(Collections.singleton(new ColumnProjectionSegment(boundNameColumn)), TableSourceType.TEMPORARY_TABLE));
80          Multimap<CaseInsensitiveString, TableSegmentBinderContext> outerTableBinderContexts = LinkedHashMultimap.create();
81          SubquerySegment actual = SubquerySegmentBinder.bind(subquerySegment, sqlStatementBinderContext, outerTableBinderContexts);
82          assertNotNull(actual.getSelect());
83          assertTrue(actual.getSelect().getFrom().isPresent());
84          assertInstanceOf(SimpleTableSegment.class, actual.getSelect().getFrom().get());
85          assertThat(((SimpleTableSegment) actual.getSelect().getFrom().get()).getTableName().getIdentifier().getValue(), is("t_order"));
86          assertTrue(actual.getSelect().getWhere().isPresent());
87          assertInstanceOf(ColumnSegment.class, actual.getSelect().getWhere().get().getExpr());
88          assertThat(((ColumnSegment) actual.getSelect().getWhere().get().getExpr()).getIdentifier().getValue(), is("status"));
89          assertNotNull(((ColumnSegment) actual.getSelect().getWhere().get().getExpr()).getColumnBoundInfo());
90          assertThat(((ColumnSegment) actual.getSelect().getWhere().get().getExpr()).getColumnBoundInfo().getOriginalColumn().getValue(), is("status"));
91          assertThat(((ColumnSegment) actual.getSelect().getWhere().get().getExpr()).getColumnBoundInfo().getOriginalTable().getValue(), is("t_order"));
92          assertThat(((ColumnSegment) actual.getSelect().getWhere().get().getExpr()).getColumnBoundInfo().getOriginalSchema().getValue(), is("foo_db"));
93          assertThat(((ColumnSegment) actual.getSelect().getWhere().get().getExpr()).getColumnBoundInfo().getOriginalDatabase().getValue(), is("foo_db"));
94          assertNotNull(actual.getSelect().getProjections());
95          assertThat(actual.getSelect().getProjections().getProjections().size(), is(1));
96          ProjectionSegment column = actual.getSelect().getProjections().getProjections().iterator().next();
97          assertInstanceOf(ColumnProjectionSegment.class, column);
98          assertThat(((ColumnProjectionSegment) column).getColumn().getIdentifier().getValue(), is("order_id"));
99          assertNotNull(((ColumnProjectionSegment) column).getColumn().getColumnBoundInfo());
100         assertThat(((ColumnProjectionSegment) column).getColumn().getColumnBoundInfo().getOriginalColumn().getValue(), is("order_id"));
101         assertThat(((ColumnProjectionSegment) column).getColumn().getColumnBoundInfo().getOriginalTable().getValue(), is("t_order"));
102         assertThat(((ColumnProjectionSegment) column).getColumn().getColumnBoundInfo().getOriginalSchema().getValue(), is("foo_db"));
103         assertThat(((ColumnProjectionSegment) column).getColumn().getColumnBoundInfo().getOriginalDatabase().getValue(), is("foo_db"));
104     }
105     
106     private ShardingSphereMetaData createMetaData() {
107         ShardingSphereSchema schema = mock(ShardingSphereSchema.class, RETURNS_DEEP_STUBS);
108         when(schema.getTable("t_order").getAllColumns()).thenReturn(Arrays.asList(
109                 new ShardingSphereColumn("order_id", Types.INTEGER, true, false, false, true, false, false),
110                 new ShardingSphereColumn("user_id", Types.INTEGER, false, false, false, true, false, false),
111                 new ShardingSphereColumn("status", Types.INTEGER, false, false, false, true, false, false)));
112         ShardingSphereMetaData result = mock(ShardingSphereMetaData.class, RETURNS_DEEP_STUBS);
113         when(result.getDatabase("foo_db").getSchema("foo_db")).thenReturn(schema);
114         when(result.containsDatabase("foo_db")).thenReturn(true);
115         when(result.getDatabase("foo_db").containsSchema("foo_db")).thenReturn(true);
116         when(result.getDatabase("foo_db").getSchema("foo_db").containsTable("t_order")).thenReturn(true);
117         return result;
118     }
