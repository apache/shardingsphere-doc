86  /**
87   * Standard DDL statement assert.
88   */
89  @NoArgsConstructor(access = AccessLevel.PRIVATE)
90  public final class StandardDDLStatementAssert {
91      
92      /**
93       * Assert DDL statement is correct with expected parser result.
94       *
95       * @param assertContext assert context
96       * @param actual actual DDL statement
97       * @param expected expected parser result
98       */
99      public static void assertIs(final SQLCaseAssertContext assertContext, final DDLStatement actual, final SQLParserTestCase expected) {
100         if (actual instanceof CreateTableStatement) {
101             CreateTableStatementAssert.assertIs(assertContext, (CreateTableStatement) actual, (CreateTableStatementTestCase) expected);
102         } else if (actual instanceof AlterTableStatement) {
103             AlterTableStatementAssert.assertIs(assertContext, (AlterTableStatement) actual, (AlterTableStatementTestCase) expected);
104         } else if (actual instanceof RenameTableStatement) {
105             RenameTableStatementAssert.assertIs(assertContext, (RenameTableStatement) actual, (RenameTableStatementTestCase) expected);
106         } else if (actual instanceof DropTableStatement) {
107             DropTableStatementAssert.assertIs(assertContext, (DropTableStatement) actual, (DropTableStatementTestCase) expected);
108         } else if (actual instanceof TruncateStatement) {
109             TruncateStatementAssert.assertIs(assertContext, (TruncateStatement) actual, (TruncateStatementTestCase) expected);
110         } else if (actual instanceof CreateIndexStatement) {
111             CreateIndexStatementAssert.assertIs(assertContext, (CreateIndexStatement) actual, (CreateIndexStatementTestCase) expected);
112         } else if (actual instanceof AlterIndexStatement) {
113             AlterIndexStatementAssert.assertIs(assertContext, (AlterIndexStatement) actual, (AlterIndexStatementTestCase) expected);
114         } else if (actual instanceof DropIndexStatement) {
115             DropIndexStatementAssert.assertIs(assertContext, (DropIndexStatement) actual, (DropIndexStatementTestCase) expected);
116         } else if (actual instanceof AlterSynonymStatement) {
117             AlterSynonymStatementAssert.assertIs(assertContext, (AlterSynonymStatement) actual, (AlterSynonymStatementTestCase) expected);
118         } else if (actual instanceof CursorStatement) {
119             CursorStatementAssert.assertIs(assertContext, (CursorStatement) actual, (CursorStatementTestCase) expected);
120         } else if (actual instanceof CloseStatement) {
121             CloseStatementAssert.assertIs(assertContext, (CloseStatement) actual, (CloseStatementTestCase) expected);
122         } else if (actual instanceof MoveStatement) {
123             MoveStatementAssert.assertIs(assertContext, (MoveStatement) actual, (MoveStatementTestCase) expected);
124         } else if (actual instanceof FetchStatement) {
125             FetchStatementAssert.assertIs(assertContext, (FetchStatement) actual, (FetchStatementTestCase) expected);
126         } else if (actual instanceof CommentStatement) {
127             CommentStatementAssert.assertIs(assertContext, (CommentStatement) actual, (CommentStatementTestCase) expected);
128         } else if (actual instanceof RefreshMatViewStmtStatement) {
129             RefreshMatViewStmtStatementAssert.assertIs(assertContext, (RefreshMatViewStmtStatement) actual, (RefreshMatViewStmtStatementTestCase) expected);
130         } else if (actual instanceof CreateViewStatement) {
131             CreateViewStatementAssert.assertIs(assertContext, (CreateViewStatement) actual, (CreateViewStatementTestCase) expected);
132         } else if (actual instanceof AlterViewStatement) {
133             AlterViewStatementAssert.assertIs(assertContext, (AlterViewStatement) actual, (AlterViewStatementTestCase) expected);
134         } else if (actual instanceof DropViewStatement) {
135             DropViewStatementAssert.assertIs(assertContext, (DropViewStatement) actual, (DropViewStatementTestCase) expected);
136         } else if (actual instanceof AlterTablespaceStatement) {
137             AlterTablespaceStatementAssert.assertIs(assertContext, (AlterTablespaceStatement) actual, (AlterTablespaceStatementTestCase) expected);
138         } else if (actual instanceof CreateSequenceStatement) {
139             CreateSequenceStatementAssert.assertIs(assertContext, (CreateSequenceStatement) actual, (CreateSequenceStatementTestCase) expected);
140         }
141     }
