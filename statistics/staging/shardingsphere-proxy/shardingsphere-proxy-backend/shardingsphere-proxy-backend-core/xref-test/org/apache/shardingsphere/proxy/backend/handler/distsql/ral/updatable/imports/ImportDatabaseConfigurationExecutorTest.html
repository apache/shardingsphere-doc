65  class ImportDatabaseConfigurationExecutorTest {
66      
67      private final ImportDatabaseConfigurationExecutor executor =
68              (ImportDatabaseConfigurationExecutor) TypedSPILoader.getService(DistSQLUpdateExecutor.class, ImportDatabaseConfigurationStatement.class);
69      
70      @ParameterizedTest(name = "{0}")
71      @MethodSource("importSuccessCases")
72      void assertImportDatabaseExecutor(final String caseName, final String databaseName, final String filePath) {
73          assertExecute(mockContextManager(databaseName), filePath);
74      }
75      
76      @Test
77      void assertImportExistedDatabase() {
78          ContextManager contextManager = mockContextManager("sharding_db");
79          when(contextManager.getMetaDataContexts().getMetaData().containsDatabase("sharding_db")).thenReturn(true);
80          assertThrows(DatabaseCreateExistsException.class, () -> assertExecute(contextManager, "/conf/import/database-sharding.yaml"));
81      }
82      
83      @ParameterizedTest(name = "{0}")
84      @MethodSource("importFailureCases")
85      void assertImportFailure(final String caseName, final String databaseName, final String filePath, final Class<? extends Throwable> expected) {
86          assertThrows(expected, () -> assertExecute(mockContextManager(databaseName), filePath));
87      }
88      
89      @Test
90      void assertImportWhenRuleConversionFails() {
91          ContextManager contextManager = mock(ContextManager.class);
92          URL url = ImportDatabaseConfigurationExecutorTest.class.getResource("/conf/import/database-sharding.yaml");
93          assertNotNull(url);
94          try (
95                  MockedConstruction<YamlDatabaseConfigurationSwapper> ignored = mockConstruction(YamlDatabaseConfigurationSwapper.class,
96                          (mock, context) -> when(mock.swapRuleConfigurations(anyCollection())).thenThrow(IllegalStateException.class))) {
97              assertThrows(IllegalStateException.class, () -> executor.executeUpdate(new ImportDatabaseConfigurationStatement(url.getPath()), contextManager));
98          }
99          verifyNoInteractions(contextManager);
100     }
101     
102     @Test
103     void assertImportFileNotFound() {
104         assertThrows(FileIOException.class, () -> executor.executeUpdate(new ImportDatabaseConfigurationStatement("/__not_exist__/database-configuration.yaml"), mock(ContextManager.class)));
105     }
106     
107     private static Stream<Arguments> importSuccessCases() {
108         return Stream.of(
109                 Arguments.of("import sharding", "sharding_db", "/conf/import/database-sharding.yaml"),
110                 Arguments.of("import single", "single_db", "/conf/import/database-single.yaml"),
111                 Arguments.of("import readwrite splitting", "readwrite_splitting_db", "/conf/import/database-readwrite-splitting.yaml"),
112                 Arguments.of("import encrypt", "encrypt_db", "/conf/import/database-encrypt.yaml"),
113                 Arguments.of("import shadow", "shadow_db", "/conf/import/database-shadow.yaml"),
114                 Arguments.of("import mask", "mask_db", "/conf/import/database-mask.yaml"));
115     }
116     
117     private static Stream<Arguments> importFailureCases() {
118         return Stream.of(
119                 Arguments.of("missing database name", "sharding_db", "/conf/import/database-empty-database-name.yaml", MissingRequiredDatabaseException.class),
120                 Arguments.of("duplicated logic table", "sharding_db", "/conf/import/database-duplicated-logic-table.yaml", DuplicateRuleException.class),
121                 Arguments.of("invalid algorithm", "sharding_db", "/conf/import/database-invalid-algorithm.yaml", InvalidRuleConfigurationException.class));
122     }
123     
124     private void assertExecute(final ContextManager contextManager, final String filePath) {
125         URL url = ImportDatabaseConfigurationExecutorTest.class.getResource(filePath);
126         assertNotNull(url);
127         executor.executeUpdate(new ImportDatabaseConfigurationStatement(url.getPath()), contextManager);
128     }
129     
130     private ContextManager mockContextManager(final String databaseName) {
131         ContextManager result = mock(ContextManager.class, RETURNS_DEEP_STUBS);
132         ShardingSphereDatabase database = mock(ShardingSphereDatabase.class, RETURNS_DEEP_STUBS);
133         when(database.getName()).thenReturn(databaseName);
134         ResourceMetaData resourceMetaData = mock(ResourceMetaData.class);
135         when(database.getResourceMetaData()).thenReturn(resourceMetaData);
136         when(database.getRuleMetaData().getRules()).thenReturn(new LinkedList<>());
137         when(database.getProtocolType()).thenReturn(TypedSPILoader.getService(DatabaseType.class, "FIXTURE"));
138         StorageUnit storageUnit = mock(StorageUnit.class);
139         DataSource dataSource = new MockedDataSource();
140         when(storageUnit.getDataSource()).thenReturn(dataSource);
141         when(database.getResourceMetaData().getStorageUnits()).thenReturn(new HashMap<>(Collections.singletonMap("foo_ds", storageUnit)));
142         when(database.getResourceMetaData().getDataSourceMap()).thenReturn(Collections.singletonMap("foo_ds", dataSource));
143         when(database.getRuleMetaData().getAttributes(DataSourceMapperRuleAttribute.class)).thenReturn(Collections.emptyList());
144         when(result.getMetaDataContexts().getMetaData().getAllDatabases()).thenReturn(Collections.singleton(database));
145         when(result.getMetaDataContexts().getMetaData().getDatabase(databaseName)).thenReturn(database);
146         when(result.getMetaDataContexts().getMetaData().getProps()).thenReturn(
147                 new ConfigurationProperties(PropertiesBuilder.build(new Property(ConfigurationPropertyKey.PROXY_FRONTEND_DATABASE_PROTOCOL_TYPE.getKey(), "MySQL"))));
148         when(result.getMetaDataContexts().getMetaData().getTemporaryProps()).thenReturn(new TemporaryConfigurationProperties(new Properties()));
149         return result;
150     }
