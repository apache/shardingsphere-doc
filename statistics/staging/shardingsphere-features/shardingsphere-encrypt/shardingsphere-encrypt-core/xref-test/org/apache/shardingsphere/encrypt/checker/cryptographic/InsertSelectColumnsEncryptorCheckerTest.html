45  @ExtendWith(MockitoExtension.class)
46  class InsertSelectColumnsEncryptorCheckerTest {
47      
48      @Test
49      void assertInsertSelectIsSame() {
50          String databaseName = "foo_db";
51          IdentifierValue databaseValue = new IdentifierValue(databaseName);
52          IdentifierValue schemaValue = new IdentifierValue("schema");
53          ColumnSegment insertColumn1 = getInsertColumnSegment(databaseValue, schemaValue, "table1", "pwd");
54          ColumnSegment insertColumn2 = getInsertColumnSegment(databaseValue, schemaValue, "table1", "card");
55          ColumnProjection projection1 = getSelectProjection("pwd", databaseValue, schemaValue);
56          ColumnProjection projection2 = getSelectProjection("card", databaseValue, schemaValue);
57          EncryptRule encryptRule = new EncryptRule(databaseName, createEncryptRuleConfiguration());
58          assertTrue(InsertSelectColumnsEncryptorChecker.isSame(Arrays.asList(insertColumn1, insertColumn2), Arrays.asList(projection1, projection2), encryptRule));
59      }
60      
61      private ColumnProjection getSelectProjection(final String pwd, final IdentifierValue databaseValue, final IdentifierValue schemaValue) {
62          TableSegmentBoundInfo tableSegmentBoundInfo = new TableSegmentBoundInfo(databaseValue, schemaValue);
63          return new ColumnProjection(new IdentifierValue("table2"), new IdentifierValue(pwd), null, null, null, null,
64                  new ColumnSegmentBoundInfo(tableSegmentBoundInfo, new IdentifierValue("table2"), new IdentifierValue(pwd), TableSourceType.TEMPORARY_TABLE));
65      }
66      
67      private ColumnSegment getInsertColumnSegment(final IdentifierValue databaseValue, final IdentifierValue schemaValue, final String tableName, final String columnName) {
68          ColumnSegment result = mock(ColumnSegment.class);
69          TableSegmentBoundInfo tableSegmentBoundInfo = new TableSegmentBoundInfo(databaseValue, schemaValue);
70          when(result.getColumnBoundInfo())
71                  .thenReturn(new ColumnSegmentBoundInfo(tableSegmentBoundInfo, new IdentifierValue(tableName), new IdentifierValue(columnName), TableSourceType.TEMPORARY_TABLE));
72          return result;
73      }
74      
75      private EncryptRuleConfiguration createEncryptRuleConfiguration() {
76          return new EncryptRuleConfiguration(Arrays.asList(getEncryptTableRuleConfiguration("table1"), getEncryptTableRuleConfiguration("table2")),
77                  getEncryptors(new AlgorithmConfiguration("CORE.FIXTURE", new Properties()),
78                          new AlgorithmConfiguration("CORE.QUERY_ASSISTED.FIXTURE", new Properties()), new AlgorithmConfiguration("CORE.QUERY_LIKE.FIXTURE", new Properties())));
79      }
80      
81      private EncryptTableRuleConfiguration getEncryptTableRuleConfiguration(final String tableName) {
82          EncryptColumnRuleConfiguration pwdColumnConfig = createEncryptColumnRuleConfiguration("pwd", "standard_encryptor");
83          EncryptColumnRuleConfiguration cardColumnConfig = new EncryptColumnRuleConfiguration("card", new EncryptColumnItemRuleConfiguration("card_cipher", "standard_encryptor"));
84          return new EncryptTableRuleConfiguration(tableName, Arrays.asList(pwdColumnConfig, cardColumnConfig));
85      }
86      
87      private EncryptColumnRuleConfiguration createEncryptColumnRuleConfiguration(final String logicalName, final String encryptorName) {
88          return new EncryptColumnRuleConfiguration(logicalName, new EncryptColumnItemRuleConfiguration(logicalName, encryptorName));
89      }
90      
91      private Map<String, AlgorithmConfiguration> getEncryptors(final AlgorithmConfiguration standardEncryptConfig, final AlgorithmConfiguration queryAssistedEncryptConfig,
92                                                                final AlgorithmConfiguration queryLikeEncryptConfig) {
93          Map<String, AlgorithmConfiguration> result = new HashMap<>(3, 1F);
94          result.put("standard_encryptor", standardEncryptConfig);
95          result.put("assisted_encryptor", queryAssistedEncryptConfig);
96          result.put("like_encryptor", queryLikeEncryptConfig);
97          return result;
98      }
