198     @ParameterizedTest(name = "{0}")
199     @MethodSource("getConflictParameterCases")
200     void assertConflictParameterOrder(final String name, final boolean targetPredicate, final String action, final List<Object> expectedParams) {
201         String sql = "INSERT INTO foo_table (foo) VALUES (?) ON CONFLICT (foo)" + (targetPredicate ? " WHERE foo > ?" : "") + action;
202         boolean update = action.startsWith(" DO UPDATE");
203         boolean actionPredicate = action.contains(" WHERE ");
204         List<ParameterMarkerSegment> markers = new ArrayList<>();
205         for (int position = sql.indexOf('?'); position >= 0; position = sql.indexOf('?', position + 1)) {
206             markers.add(new ParameterMarkerExpressionSegment(position, position, markers.size()));
207         }
208         int assignmentIndex = targetPredicate ? 2 : 1;
209         OnDuplicateKeyColumnsSegment conflict = new OnDuplicateKeyColumnsSegment(sql.indexOf("ON CONFLICT"), sql.length() - 1,
210                 update ? Collections.singletonList(new ColumnAssignmentSegment(sql.indexOf("foo ="), markers.get(assignmentIndex).getStopIndex(), Collections.emptyList(),
211                         (ParameterMarkerExpressionSegment) markers.get(assignmentIndex))) : Collections.emptyList());
212         if (targetPredicate) {
213             conflict.setConflictTargetWhere(createConflictWhere(sql.indexOf("WHERE"), (ParameterMarkerExpressionSegment) markers.get(1)));
214         }
215         InsertStatement statement = InsertStatement.builder().databaseType(databaseType).onDuplicateKeyColumns(conflict)
216                 .where(actionPredicate ? createConflictWhere(sql.lastIndexOf("WHERE"), (ParameterMarkerExpressionSegment) markers.get(markers.size() - 1)) : null).build();
217         statement.addParameterMarkers(markers);
218         InsertStatementContext statementContext = mock(InsertStatementContext.class, RETURNS_DEEP_STUBS);
219         when(statementContext.getSqlStatement()).thenReturn(statement);
220         when(statementContext.getInsertSelectContext()).thenReturn(null);
221         when(statementContext.getGroupedParameters()).thenReturn(Collections.singletonList(Collections.singletonList(expectedParams.get(0))));
222         when(statementContext.getOnDuplicateKeyUpdateParameters()).thenReturn(update ? Collections.singletonList(expectedParams.get(assignmentIndex)) : Collections.emptyList());
223         QueryContext queryContext = mock(QueryContext.class, RETURNS_DEEP_STUBS);
224         when(queryContext.getSqlStatementContext()).thenReturn(statementContext);
225         when(queryContext.getParameters()).thenReturn(expectedParams);
226         SQLRewriteContext actual = new SQLRewriteContext(database, queryContext);
227         assertThat(actual.getParameterBuilder().getParameters(), is(expectedParams));
228     }
229     
230     private static Stream<Arguments> getConflictParameterCases() {
231         return Stream.of(Arguments.of("target predicate and do nothing", true, " DO NOTHING", Arrays.<Object>asList(1, 2)),
232                 Arguments.of("target predicate and conditional update", true, " DO UPDATE SET foo = ? WHERE foo > ?", Arrays.<Object>asList(1, 2, 3, 4)),
233                 Arguments.of("conditional update", false, " DO UPDATE SET foo = ? WHERE foo > ?", Arrays.<Object>asList(1, 3, 4)),
234                 Arguments.of("target predicate and conditional select", true, " DO SELECT WHERE foo > ? RETURNING foo", Arrays.<Object>asList(1, 2, 3)));
235     }
236     
237     private WhereSegment createConflictWhere(final int startIndex, final ParameterMarkerExpressionSegment marker) {
238         return new WhereSegment(startIndex, marker.getStopIndex(), new BinaryOperationExpression(startIndex + 6, marker.getStopIndex(),
239                 new ColumnSegment(startIndex + 6, startIndex + 8, new IdentifierValue("foo")), marker, ">", "foo > ?"));
240     }
