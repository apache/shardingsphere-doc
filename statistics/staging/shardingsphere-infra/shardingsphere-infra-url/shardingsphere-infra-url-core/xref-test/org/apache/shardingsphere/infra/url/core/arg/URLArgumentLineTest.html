18  package org.apache.shardingsphere.infra.url.core.arg;
19  
20  import org.junit.jupiter.api.Test;
21  
22  import java.util.Optional;
23  
24  import static org.hamcrest.CoreMatchers.is;
25  import static org.hamcrest.MatcherAssert.assertThat;
26  import static org.junit.jupiter.api.Assertions.assertFalse;
27  import static org.junit.jupiter.api.Assertions.assertTrue;
28  
29  class URLArgumentLineTest {
30      
31      private final String line = "key=$${value::default_value}";
32      
33      private final String lineMultiple1 = "key1=$${value1::default_value1}:key2=$${value2::default_value2}:tail";
34      
35      private final String lineMultiple2 = "key1=$${value1::}:key2=$${value2::}:tail ";
36      
37      @Test
38      void assertParseWithInvalidPattern() {
39          assertFalse(URLArgumentLine.parse("invalid").isPresent());
40      }
41      
42      @Test
43      void assertReplaceArgumentWithNone() {
44          Optional<URLArgumentLine> argLine = URLArgumentLine.parse(line);
45          Optional<URLArgumentLine> argLineMultiple1 = URLArgumentLine.parse(lineMultiple1);
46          Optional<URLArgumentLine> argLineMultiple2 = URLArgumentLine.parse(lineMultiple2);
47          assertTrue(argLine.isPresent());
48          assertTrue(argLineMultiple1.isPresent());
49          assertTrue(argLineMultiple2.isPresent());
50          assertThat(argLine.get().replaceArgument(URLArgumentPlaceholderType.NONE), is("key=default_value"));
51          assertThat(argLineMultiple1.get().replaceArgument(URLArgumentPlaceholderType.NONE), is("key1=default_value1:key2=default_value2:tail"));
52          assertThat(argLineMultiple2.get().replaceArgument(URLArgumentPlaceholderType.NONE), is("key1=:key2=:tail"));
53      }
54      
55      @Test
56      void assertReplaceArgumentWithEnvironment() {
57          Optional<URLArgumentLine> argLine = URLArgumentLine.parse(line);
58          Optional<URLArgumentLine> argLineMultiple1 = URLArgumentLine.parse(lineMultiple1);
59          Optional<URLArgumentLine> argLineMultiple2 = URLArgumentLine.parse(lineMultiple2);
60          assertTrue(argLine.isPresent());
61          assertTrue(argLineMultiple1.isPresent());
62          assertTrue(argLineMultiple2.isPresent());
63          assertThat(argLine.get().replaceArgument(URLArgumentPlaceholderType.ENVIRONMENT), is("key=default_value"));
64          assertThat(argLineMultiple1.get().replaceArgument(URLArgumentPlaceholderType.ENVIRONMENT), is("key1=default_value1:key2=default_value2:tail"));
65          assertThat(argLineMultiple2.get().replaceArgument(URLArgumentPlaceholderType.ENVIRONMENT), is("key1=:key2=:tail"));
66      }
67      
68      @Test
69      void assertReplaceArgumentWithSystemProperty() {
70          System.setProperty("value", "props_value");
71          System.setProperty("value1", "props_value1");
72          System.setProperty("value2", "props_value2");
73          Optional<URLArgumentLine> argLine = URLArgumentLine.parse(line);
74          Optional<URLArgumentLine> argLineMultiple1 = URLArgumentLine.parse(lineMultiple1);
75          Optional<URLArgumentLine> argLineMultiple2 = URLArgumentLine.parse(lineMultiple2);
76          assertTrue(argLine.isPresent());
77          assertTrue(argLineMultiple1.isPresent());
78          assertTrue(argLineMultiple2.isPresent());
79          assertThat(argLine.get().replaceArgument(URLArgumentPlaceholderType.SYSTEM_PROPS), is("key=props_value"));
80          assertThat(argLineMultiple1.get().replaceArgument(URLArgumentPlaceholderType.SYSTEM_PROPS), is("key1=props_value1:key2=props_value2:tail"));
81          assertThat(argLineMultiple2.get().replaceArgument(URLArgumentPlaceholderType.SYSTEM_PROPS), is("key1=props_value1:key2=props_value2:tail"));
82      }
83  }
