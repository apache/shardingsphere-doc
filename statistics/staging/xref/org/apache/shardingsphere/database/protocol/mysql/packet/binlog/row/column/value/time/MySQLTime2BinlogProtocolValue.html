18  package org.apache.shardingsphere.database.protocol.mysql.packet.binlog.row.column.value.time;
19  
20  import org.apache.shardingsphere.database.protocol.mysql.packet.binlog.row.column.MySQLBinlogColumnDef;
21  import org.apache.shardingsphere.database.protocol.mysql.packet.binlog.row.column.value.MySQLBinlogProtocolValue;
22  import org.apache.shardingsphere.database.protocol.mysql.payload.MySQLPacketPayload;
23  
24  import java.io.Serializable;
25  import java.time.LocalTime;
26  import java.util.Locale;
27  
28  /**
29   * TIME2 type value of MySQL binlog protocol.
30   * The 3-byte value is the packed time offset by 0x800000, so it is signed: MySQL TIME ranges from -838:59:59 to 838:59:59.
31   * Stored as 3-byte value The number of decimals for the fractional part is stored in the table metadata as a one byte value.
32   * The number of bytes that follow the 3 byte time value can be calculated with the following formula: (decimals + 1) / 2
33   * A negative value with a non-zero fraction is stored with its whole seconds one lower and its fraction counted up from there, so the two parts are only meaningful together.
34   *
35   * <p>
36   * TIME2 type applied after MySQL 5.6.4.
37   * </p>
38   *
39   * @see <a href="https://dev.mysql.com/doc/dev/mysql-server/latest/field__types_8h.html">field type</a>
40   * @see <a href="https://github.com/mysql/mysql-server/blob/mysql-8.0.46/mysys/my_time.cc">my_time_packed_from_binary</a>
41   */
42  public final class MySQLTime2BinlogProtocolValue implements MySQLBinlogProtocolValue {
43      
44      private static final int INT_OFFSET = 0x800000;
45      
46      @Override
47      public Serializable read(final MySQLBinlogColumnDef columnDef, final MySQLPacketPayload payload) {
48          long packedTime = readPackedTime(columnDef.getColumnMeta(), payload);
49          if (0L == packedTime) {
50              return MySQLTimeValueUtils.ZERO_OF_TIME;
51          }
52          long magnitude = Math.abs(packedTime);
53          int wholeSeconds = (int) (magnitude >> 24);
54          int hour = (wholeSeconds >> 12) % (1 << 10);
55          int minute = (wholeSeconds >> 6) % (1 << 6);
56          int second = wholeSeconds % (1 << 6);
57          int micros = (int) (magnitude % (1 << 24));
58          return packedTime > 0L && hour < 24
59                  ? LocalTime.of(hour, minute, second, micros * 1000)
60                  : formatOutOfLocalTimeRange(packedTime < 0L, hour, minute, second, micros);
61      }
62      
63      /**
64       * Read the value as MySQL's signed packed time, the whole seconds shifted left by 24 bits plus the microseconds, the same way my_time_packed_from_binary does.
65       * The fraction is read as stored rather than through {@link MySQLFractionalSeconds}, because for a negative value it can be any number the bytes hold.
66       *
67       * @param fractionalSecondsPrecision fractional seconds precision of the column
68       * @param payload payload
69       * @return signed packed time
70       */
71      private long readPackedTime(final int fractionalSecondsPrecision, final MySQLPacketPayload payload) {
72          long wholeSeconds = payload.getByteBuf().readUnsignedMedium() - INT_OFFSET;
73          switch (fractionalSecondsPrecision) {
74              case 1:
75              case 2:
76                  return toPackedTime(wholeSeconds, payload.readInt1(), 0x100, 10000);
77              case 3:
78              case 4:
79                  return toPackedTime(wholeSeconds, payload.getByteBuf().readUnsignedShort(), 0x10000, 100);
80              case 5:
81              case 6:
82                  return toPackedTime(wholeSeconds, payload.getByteBuf().readUnsignedMedium(), 0x1000000, 1);
83              default:
84                  return wholeSeconds << 24;
85          }
86      }
87      
88      /**
89       * Combine the stored whole seconds and fraction into a signed packed time.
90       * A negative value with a non-zero fraction is stored with its whole seconds one lower and its fraction as fractionCarry minus the real fraction, so both are brought back first.
91       *
92       * @param wholeSeconds stored whole seconds, already signed
93       * @param fraction stored fraction
94       * @param fractionCarry value at which the stored fraction carries into the whole seconds
95       * @param microsPerFractionUnit microseconds in one unit of the stored fraction
96       * @return signed packed time
97       */
98      private long toPackedTime(final long wholeSeconds, final int fraction, final int fractionCarry, final int microsPerFractionUnit) {
99          return wholeSeconds < 0L && 0 != fraction
100                 ? ((wholeSeconds + 1L) << 24) + (long) (fraction - fractionCarry) * microsPerFractionUnit
101                 : (wholeSeconds << 24) + (long) fraction * microsPerFractionUnit;
102     }
103     
104     /**
105      * Formats a value that {@link LocalTime} cannot hold, which is any negative time and any time from 24:00:00 up to the MySQL maximum of 838:59:59.
106      * The text form matches what {@link MySQLTimeBinlogProtocolValue} returns for every value and what {@link MySQLTimeValueUtils#ZERO_OF_TIME} returns from this same method.
107      * Digits are always ASCII, whatever the default locale.
108      *
109      * @param negative whether the time is negative
110      * @param hour hour
111      * @param minute minute
112      * @param second second
113      * @param micros microseconds
114      * @return time in MySQL text form
115      */
116     private String formatOutOfLocalTimeRange(final boolean negative, final int hour, final int minute, final int second, final int micros) {
117         String result = String.format(Locale.ROOT, "%s%02d:%02d:%02d", negative ? "-" : "", hour, minute, second);
118         return 0 == micros ? result : result + String.format(Locale.ROOT, ".%06d", micros);
119     }
120 }
