18  package org.apache.shardingsphere.sqlfederation.mysql;
19  
20  import org.apache.calcite.runtime.SqlFunctions;
21  import org.apache.calcite.schema.SchemaPlus;
22  import org.apache.calcite.schema.impl.ScalarFunctionImpl;
23  import org.apache.shardingsphere.sqlfederation.compiler.sql.function.DialectSQLFederationFunctionRegister;
24  import org.apache.shardingsphere.sqlfederation.compiler.sql.function.mysql.impl.MySQLBinFunction;
25  
26  import java.util.Arrays;
27  import java.util.Collection;
28  
29  /**
30   * SQL federation function register for MySQL.
31   */
32  public final class MySQLSQLFederationFunctionRegister implements DialectSQLFederationFunctionRegister {
33      
34      @Override
35      public void registerFunction(final SchemaPlus schemaPlus, final String schemaName) {
36          schemaPlus.add("bin", ScalarFunctionImpl.create(MySQLBinFunction.class, "bin"));
37          schemaPlus.add("atan", ScalarFunctionImpl.create(SqlFunctions.class, "atan2"));
38          schemaPlus.add("atan2", ScalarFunctionImpl.create(SqlFunctions.class, "atan"));
39      }
40      
41      @Override
42      public Collection<String> getUnsupportedFunctionNames() {
43          return Arrays.asList(
44                  "AES_DECRYPT",
45                  "AES_ENCRYPT",
46                  "BENCHMARK",
47                  "BIT_LENGTH",
48                  "CEILING",
49                  "CHAR",
50                  "CHARSET",
51                  "COERCIBILITY",
52                  "CONV",
53                  "CRC32",
54                  "DATABASE",
55                  "ELT",
56                  "EXPORT_SET",
57                  "FIELD",
58                  "FIND_IN_SET",
59                  "FORMAT",
60                  "FROM_UNIXTIME",
61                  "HEX",
62                  "INSERT",
63                  "LCASE",
64                  "LEAST",
65                  "LENGTH",
66                  "LOAD_FILE",
67                  "LOCATE",
68                  "LOG",
69                  "LOG2",
70                  "LPAD",
71                  "LTRIM",
72                  "MAKE_SET",
73                  "MID",
74                  "OCT",
75                  "OCTET_LENGTH",
76                  "ORD",
77                  "POW",
78                  "QUOTE",
79                  "REGEXP_INSTR",
80                  "REGEXP_LIKE",
81                  "REGEXP_SUBSTR",
82                  "RPAD",
83                  "RTRIM",
84                  "SHA",
85                  "SUBSTRING",
86                  "SUBSTRING_INDEX",
87                  "TIMEDIFF",
88                  "UCASE",
89                  "UNHEX",
90                  "WEIGHT_STRING");
91      }
92      
93      @Override
94      public String getDatabaseType() {
95          return "MySQL";
96      }
97  }
