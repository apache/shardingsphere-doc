56      @Test
57      void assertHandlePlanEncryptRule() {
58          try (MockedConstruction<EncryptWorkflowPlanningService> mockedConstruction = mockConstruction(EncryptWorkflowPlanningService.class)) {
59              PlanEncryptRuleToolHandler handler = new PlanEncryptRuleToolHandler();
60              EncryptWorkflowPlanningService planningService = mockedConstruction.constructed().getFirst();
61              when(planningService.plan(any(), any(), any(), any())).thenReturn(createSnapshot("plan-1", "planned"));
62              WorkflowContextFixture fixture = createWorkflowContextFixture();
63              MCPSuccessPayload actual = handler.handle(fixture.requestContext, Map.of(
64                      "database", "logic_db",
65                      "table", "orders",
66                      "column", "phone",
67                      "operation_type", "create",
68                      "algorithm_type", "AES",
69                      "cipher_column_name", "phone_cipher",
70                      "requires_decrypt", true,
71                      "requires_equality_filter", false,
72                      "requires_like_query", false));
73              assertThat(actual.toPayload().get("plan_id"), is("plan-1"));
74              ArgumentCaptor<EncryptWorkflowRequest> requestCaptor = ArgumentCaptor.forClass(EncryptWorkflowRequest.class);
75              verify(planningService).plan(eq(fixture.workflowSessionContext), eq(fixture.metadataQueryFacade), eq(fixture.queryFacade), requestCaptor.capture());
76              EncryptWorkflowRequest actualRequest = requestCaptor.getValue();
77              assertThat(actualRequest.getAlgorithmType(), is("AES"));
78              assertThat(actualRequest.getOptions().getCipherColumnName(), is("phone_cipher"));
79              assertTrue(actualRequest.getOptions().getRequiresDecrypt());
80              assertFalse(actualRequest.getOptions().getRequiresEqualityFilter());
81              assertFalse(actualRequest.getOptions().getRequiresLikeQuery());
82          }
83      }

113     @Test
114     void assertHandlePlanEncryptRuleWithSecretReferences() {
115         try (MockedConstruction<EncryptWorkflowPlanningService> mockedConstruction = mockConstruction(EncryptWorkflowPlanningService.class)) {
116             PlanEncryptRuleToolHandler handler = new PlanEncryptRuleToolHandler();
117             EncryptWorkflowPlanningService planningService = mockedConstruction.constructed().getFirst();
118             WorkflowContextSnapshot snapshot = createSnapshot("plan-1", "planned");
119             when(planningService.plan(any(), any(), any(), any())).thenAnswer(invocation -> {
120                 snapshot.setRequest(invocation.getArgument(3));
121                 return snapshot;
122             });
123             WorkflowContextFixture fixture = createWorkflowContextFixture();
124             Map<String, Object> actualPayload = handler.handle(fixture.requestContext, Map.of(
125                     "database", "logic_db",
126                     "primary_algorithm_properties", Map.of("aes-key-value", Map.of("secret_ref", "placeholder://secret-value-1")),
127                     "assisted_query_algorithm_properties", Map.of("salt", Map.of("secret_ref", "placeholder://secret-value-2")),
128                     "like_query_algorithm_properties", Map.of("token", Map.of("secret_ref", "placeholder://secret-value-3")))).toPayload();
129             Map<?, ?> actualMaskedPropertyPreview = (Map<?, ?>) actualPayload.get("masked_property_preview");
130             assertThat(((Map<?, ?>) actualMaskedPropertyPreview.get("primary")).get("aes-key-value"), is("******"));
131             assertThat(((Map<?, ?>) actualMaskedPropertyPreview.get("assisted_query")).get("salt"), is("******"));
132             assertThat(((Map<?, ?>) actualMaskedPropertyPreview.get("like_query")).get("token"), is("******"));
133             assertFalse(String.valueOf(actualPayload).contains("secret_reference:"));
134             ArgumentCaptor<EncryptWorkflowRequest> requestCaptor = ArgumentCaptor.forClass(EncryptWorkflowRequest.class);
135             verify(planningService).plan(eq(fixture.workflowSessionContext), eq(fixture.metadataQueryFacade), eq(fixture.queryFacade), requestCaptor.capture());
136             EncryptWorkflowRequest actualRequest = requestCaptor.getValue();
137             assertThat(actualRequest.getPrimaryAlgorithmProperties().get("aes-key-value"), is("secret_reference:primary.aes-key-value"));
138             assertFalse(actualRequest.getSecretReferences("primary").get("aes-key-value").isMalformed());
139             assertThat(actualRequest.getOptions().getAssistedQueryAlgorithmProperties().get("salt"), is("secret_reference:assisted_query.salt"));
140             assertFalse(actualRequest.getSecretReferences("assisted_query").get("salt").isMalformed());
141             assertThat(actualRequest.getOptions().getLikeQueryAlgorithmProperties().get("token"), is("secret_reference:like_query.token"));
142             assertFalse(actualRequest.getSecretReferences("like_query").get("token").isMalformed());
143         }
144     }

161     private WorkflowContextSnapshot createSnapshot(final String planId, final String status) {
162         WorkflowContextSnapshot result = new WorkflowContextSnapshot();
163         result.setPlanId(planId);
164         result.setWorkflowKind(EncryptFeatureDefinition.WORKFLOW_KIND);
165         result.setStatus(status);
166         result.setRequest(new EncryptWorkflowRequest());
167         result.setClarifiedIntent(new ClarifiedIntent());
168         result.setInteractionPlan(createInteractionPlan());
169         return result;
170     }
171     
172     private WorkflowContextSnapshot createClarifyingSnapshot() {
173         EncryptWorkflowRequest request = new EncryptWorkflowRequest();
174         request.setAlgorithmType("AES");
175         request.getPrimaryAlgorithmProperties().put("aes-key-value", "recovery-secret-value");
176         WorkflowContextSnapshot result = createSnapshot("plan-1", "clarifying");
177         result.setRequest(request);
178         return result;
179     }
180     
181     private WorkflowContextSnapshot createDetailedSnapshot() {
182         EncryptWorkflowRequest request = new EncryptWorkflowRequest();
183         request.setDatabase("logic_db");
184         request.setSchema("public");
185         request.setTable("orders");
186         request.setColumn("phone");
187         request.getPrimaryAlgorithmProperties().put("aes-key-value", "123456");
188         WorkflowContextSnapshot result = createSnapshot("plan-1", "planned");
189         result.setRequest(request);
190         result.getPropertyRequirements().add(new AlgorithmPropertyRequirement("primary", "aes-key-value", true, true, "key", ""));
191         result.getRuleArtifacts().add(new RuleArtifact("create", "CREATE ENCRYPT RULE `orders` (PROPERTIES('aes-key-value'='123456'))"));
192         result.setInteractionPlan(createInteractionPlan());
193         return result;
194     }
195     
196     private InteractionPlan createInteractionPlan() {
197         InteractionPlan result = new InteractionPlan();
198         result.setCurrentStep("review");
199         result.setDeliveryMode("interactive");
200         result.setExecutionMode("review-then-execute");
201         return result;
202     }

204     private List<String> extractResourceUris(final List<?> resources) {
205         return resources.stream().map(each -> (String) ((Map<?, ?>) each).get("uri")).toList();
206     }
207     
208     private String findResourceKind(final List<?> resources, final String uri) {
209         for (Object each : resources) {
210             Map<?, ?> resource = (Map<?, ?>) each;
211             if (uri.equals(resource.get("uri"))) {
212                 return (String) resource.get("resource_kind");
213             }
214         }
215         return "";
216     }
