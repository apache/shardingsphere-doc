409     private long getDataQuotaValue(final DataQuotaValueContext ctx) {
410         if (null != ctx.NUMBER_()) {
411             return parseQuotaNumber(ctx.NUMBER_());
412         }
413         TerminalNode token = null == ctx.FILESIZE_LITERAL() ? ctx.IDENTIFIER_() : ctx.FILESIZE_LITERAL();
414         String text = token.getText();
415         int unitIndex = 0;
416         while (unitIndex < text.length() && Character.isDigit(text.charAt(unitIndex))) {
417             unitIndex++;
418         }
419         long unitBytes = 0 == unitIndex ? 0L : getDataQuotaUnitBytes(text.substring(unitIndex));
420         if (0L == unitBytes) {
421             throw new SQLParsingException(String.format("Expected data quota value, got: %s", text), text, token.getSymbol().getLine());
422         }
423         try {
424             return Math.multiplyExact(Long.parseLong(text.substring(0, unitIndex)), unitBytes);
425         } catch (final ArithmeticException | NumberFormatException ex) {
426             throw new SQLParsingException(String.format("Data quota value out of range: %s", text), text, token.getSymbol().getLine());
427         }
428     }

430     private long getDataQuotaUnitBytes(final String unit) {
431         switch (unit.toUpperCase()) {
432             case "":
433             case "B":
434                 return 1L;
435             case "K":
436             case "KB":
437                 return 1024L;
438             case "M":
439             case "MB":
440                 return 1024L * 1024L;
441             case "G":
442             case "GB":
443                 return 1024L * 1024L * 1024L;
444             case "T":
445             case "TB":
446                 return 1024L * 1024L * 1024L * 1024L;
447             case "P":
448             case "PB":
449                 return 1024L * 1024L * 1024L * 1024L * 1024L;
450             default:
451                 return 0L;
452         }
453     }

455     private long parseQuotaNumber(final TerminalNode token) {
456         String text = token.getText();
457         if (!text.chars().allMatch(Character::isDigit)) {
458             throw new SQLParsingException(String.format("Expected integer quota value, got: %s", text), text, token.getSymbol().getLine());
459         }
460         try {
461             return Long.parseLong(text);
462         } catch (final NumberFormatException ex) {
463             throw new SQLParsingException(String.format("Quota value out of range: %s", text), text, token.getSymbol().getLine());
464         }
465     }

668                         SQLUtils.getExactlyValue(ctx.timestampValue(0).getText())));

681     public ASTNode visitCreateTable(final CreateTableContext ctx) {
682         Collection<ColumnDefinitionSegment> columnDefinitions = new LinkedList<>();
683         Collection<ConstraintDefinitionSegment> constraintDefinitions = new LinkedList<>();
684         if (null != ctx.createDefinitionClause()) {
685             CollectionValue<CreateDefinitionSegment> createDefinitions = (CollectionValue<CreateDefinitionSegment>) visit(ctx.createDefinitionClause());
686             for (CreateDefinitionSegment each : createDefinitions.getValue()) {
687                 if (each instanceof ColumnDefinitionSegment) {
688                     columnDefinitions.add((ColumnDefinitionSegment) each);
689                 } else if (each instanceof ConstraintDefinitionSegment) {
690                     constraintDefinitions.add((ConstraintDefinitionSegment) each);
691                 }
692             }
693         }
694         SimpleTableSegment likeTable = null;
695         Collection<RollupSegment> rollups = new LinkedList<>();
696         if (null != ctx.createLikeClause()) {
697             likeTable = (SimpleTableSegment) visit(ctx.createLikeClause());
698             for (IdentifierContext each : ctx.createLikeClause().identifier()) {
699                 rollups.add(new RollupSegment(each.getStart().getStartIndex(), each.getStop().getStopIndex(), new IdentifierValue(each.getText())));
700             }
701         }
702         CreateTableOptionSegment createTableOption = null;
703         if (null != ctx.createTableOptions()) {
704             createTableOption = (CreateTableOptionSegment) visit(ctx.createTableOptions());
705         }
706         SelectStatement selectStatement = null;
707         // DORIS ADDED BEGIN
708         if (null != ctx.duplicateAsQueryExpression()) {
709             selectStatement = (SelectStatement) visit(ctx.duplicateAsQueryExpression().select());
710         }
711         // DORIS ADDED END
712         return CreateTableStatement.builder()
713                 .databaseType(getDatabaseType())
714                 .table((SimpleTableSegment) visit(ctx.tableName()))
715                 .ifNotExists(null != ctx.ifNotExists())
716                 .temporary(null != ctx.TEMPORARY())
717                 .likeTable(likeTable)
718                 .createTableOption(createTableOption)
719                 .selectStatement(selectStatement)
720                 .columnDefinitions(columnDefinitions)
721                 .constraintDefinitions(constraintDefinitions)
722                 .rollups(rollups)
723                 .build();
724     }

1039         }
1040         if (null == alterTableDrop.KEY() && null == alterTableDrop.keyOrIndex()) {
1041             ColumnSegment column = new ColumnSegment(alterTableDrop.columnInternalRef.start.getStartIndex(), alterTableDrop.columnInternalRef.stop.getStopIndex(),
1042                     (IdentifierValue) visit(alterTableDrop.columnInternalRef));

1044                     new DropColumnDefinitionSegment(alterTableDrop.getStart().getStartIndex(), alterTableDrop.getStop().getStopIndex(), Collections.singleton(column));

1058                     new DropIndexDefinitionSegment(alterListContext.getStart().getStartIndex(), alterListContext.getStop().getStopIndex(), (IndexSegment) visit(alterTableDrop.indexName())));
