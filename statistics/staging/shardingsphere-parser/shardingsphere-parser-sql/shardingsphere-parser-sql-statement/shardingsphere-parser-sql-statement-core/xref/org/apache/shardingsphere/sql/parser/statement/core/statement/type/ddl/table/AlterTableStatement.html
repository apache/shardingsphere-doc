54  /**
55   * Alter table statement.
56   */
57  @Getter
58  @Setter
59  public final class AlterTableStatement extends DDLStatement {
60      
61      private SimpleTableSegment table;
62      
63      private SimpleTableSegment renameTable;
64      
65      private ConvertTableDefinitionSegment convertTableDefinition;
66      
67      private ModifyCollectionRetrievalSegment modifyCollectionRetrieval;
68      
69      private AlgorithmTypeSegment algorithmSegment;
70      
71      private LockTableSegment lockTableSegment;
72      
73      private DropPrimaryKeyDefinitionSegment dropPrimaryKeyDefinition;
74      
75      private final Collection<AddColumnDefinitionSegment> addColumnDefinitions = new LinkedList<>();
76      
77      private final Collection<ModifyColumnDefinitionSegment> modifyColumnDefinitions = new LinkedList<>();
78      
79      private final Collection<ChangeColumnDefinitionSegment> changeColumnDefinitions = new LinkedList<>();
80      
81      private final Collection<DropColumnDefinitionSegment> dropColumnDefinitions = new LinkedList<>();
82      
83      private final Collection<AddConstraintDefinitionSegment> addConstraintDefinitions = new LinkedList<>();
84      
85      private final Collection<ValidateConstraintDefinitionSegment> validateConstraintDefinitions = new LinkedList<>();
86      
87      private final Collection<ModifyConstraintDefinitionSegment> modifyConstraintDefinitions = new LinkedList<>();
88      
89      private final Collection<DropConstraintDefinitionSegment> dropConstraintDefinitions = new LinkedList<>();
90      
91      private final Collection<DropIndexDefinitionSegment> dropIndexDefinitions = new LinkedList<>();
92      
93      private final Collection<RenameColumnSegment> renameColumnDefinitions = new LinkedList<>();
94      
95      private final Collection<RenameIndexDefinitionSegment> renameIndexDefinitions = new LinkedList<>();
96      
97      /**
98       * Get rename table.
99       *
100      * @return rename table
101      */
102     public Optional<SimpleTableSegment> getRenameTable() {
103         return Optional.ofNullable(renameTable);
104     }
105     
106     /**
107      * Get convert table definition.
108      *
109      * @return convert table definition
110      */
111     public Optional<ConvertTableDefinitionSegment> getConvertTableDefinition() {
112         return Optional.ofNullable(convertTableDefinition);
113     }
114     
115     /**
116      * Get modify collection retrieval.
117      *
118      * @return modify collection retrieval
119      */
120     public Optional<ModifyCollectionRetrievalSegment> getModifyCollectionRetrieval() {
121         return Optional.ofNullable(modifyCollectionRetrieval);
122     }
123     
124     /**
125      * Get algorithm segment.
126      *
127      * @return algorithm segment
128      */
129     public Optional<AlgorithmTypeSegment> getGetAlgorithmSegment() {
130         return Optional.ofNullable(algorithmSegment);
131     }
132     
133     /**
134      * Get lock table Segment.
135      *
136      * @return lock table segment
137      */
138     public Optional<LockTableSegment> getLockTableSegment() {
139         return Optional.ofNullable(lockTableSegment);
140     }
141     
142     /**
143      * Get drop primary key.
144      *
145      * @return drop primary key
146      */
147     public Optional<DropPrimaryKeyDefinitionSegment> getDropPrimaryKeyDefinition() {
148         return Optional.ofNullable(dropPrimaryKeyDefinition);
149     }
150     
151     @Override
152     public SQLStatementAttributes getAttributes() {
153         return new SQLStatementAttributes(new TableSQLStatementAttribute(getTables()), new AlterTableConstraintSQLStatementAttribute(), new AlterTableIndexSQLStatementAttribute());
154     }
155     
156     private Collection<SimpleTableSegment> getTables() {
157         Collection<SimpleTableSegment> result = new LinkedList<>();
158         result.add(table);
159         if (getRenameTable().isPresent()) {
160             result.add(getRenameTable().get());
161         }
162         for (AddColumnDefinitionSegment each : addColumnDefinitions) {
163             for (ColumnDefinitionSegment columnDefinition : each.getColumnDefinitions()) {
164                 result.addAll(columnDefinition.getReferencedTables());
165             }
166         }
167         for (ModifyColumnDefinitionSegment each : modifyColumnDefinitions) {
168             result.addAll(each.getColumnDefinition().getReferencedTables());
169         }
170         for (AddConstraintDefinitionSegment each : addConstraintDefinitions) {
171             each.getConstraintDefinition().getReferencedTable().ifPresent(result::add);
172         }
173         return result;
174     }
175     
176     private class AlterTableConstraintSQLStatementAttribute implements ConstraintSQLStatementAttribute {
177         
178         @Override
179         public Collection<ConstraintSegment> getConstraints() {
180             Collection<ConstraintSegment> result = new LinkedList<>();
181             for (AddConstraintDefinitionSegment each : addConstraintDefinitions) {
182                 each.getConstraintDefinition().getConstraintName().ifPresent(result::add);
183             }
184             validateConstraintDefinitions.stream().map(ValidateConstraintDefinitionSegment::getConstraintName).forEach(result::add);
185             dropConstraintDefinitions.stream().map(DropConstraintDefinitionSegment::getConstraintName).forEach(result::add);
186             return result;
187         }
188     }
189     
190     private class AlterTableIndexSQLStatementAttribute implements IndexSQLStatementAttribute {
191         
192         @Override
193         public Collection<IndexSegment> getIndexes() {
194             Collection<IndexSegment> result = new LinkedList<>();
195             for (AddConstraintDefinitionSegment each : addConstraintDefinitions) {
196                 each.getConstraintDefinition().getIndexName().ifPresent(result::add);
197             }
198             dropIndexDefinitions.stream().map(DropIndexDefinitionSegment::getIndexSegment).forEach(result::add);
199             for (RenameIndexDefinitionSegment each : getRenameIndexDefinitions()) {
200                 result.add(each.getIndexSegment());
201                 result.add(each.getRenameIndexSegment());
202             }
203             return result;
204         }
205         
206         @Override
207         public Collection<ColumnSegment> getIndexColumns() {
208             return addConstraintDefinitions.stream().flatMap(each -> each.getConstraintDefinition().getIndexColumns().stream()).collect(Collectors.toList());
209         }
210     }
211 }
