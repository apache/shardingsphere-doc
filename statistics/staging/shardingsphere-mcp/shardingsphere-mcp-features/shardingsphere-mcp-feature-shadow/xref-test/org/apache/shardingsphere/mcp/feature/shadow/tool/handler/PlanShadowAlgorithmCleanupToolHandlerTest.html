52  class PlanShadowAlgorithmCleanupToolHandlerTest {
53      
54      @Test
55      void assertHandle() {
56          try (
57                  MockedConstruction<ShadowWorkflowPlanningService> mocked = mockConstruction(ShadowWorkflowPlanningService.class,
58                          (mock, context) -> when(mock.planAlgorithmCleanup(any(), any(), any())).thenReturn(createSnapshot()))) {
59              WorkflowContextFixture fixture = createWorkflowContextFixture();
60              MCPSuccessPayload actual = new PlanShadowAlgorithmCleanupToolHandler()
61                      .handle(fixture.requestContext, Map.of("database", "logic_db", "algorithm_name", "unused_algorithm"));
62              ArgumentCaptor<ShadowAlgorithmCleanupWorkflowRequest> requestCaptor = ArgumentCaptor.forClass(ShadowAlgorithmCleanupWorkflowRequest.class);
63              verify(mocked.constructed().getFirst()).planAlgorithmCleanup(
64                      eq(fixture.workflowSessionContext), eq(fixture.queryFacade), requestCaptor.capture());
65              assertThat(requestCaptor.getValue().getAlgorithmName(), is("unused_algorithm"));
66              assertThat(actual.toPayload().get("workflow_kind"), is("shadow.cleanup"));
67          }
68      }
69      
70      private WorkflowContextSnapshot createSnapshot() {
71          ShadowAlgorithmCleanupWorkflowRequest request = new ShadowAlgorithmCleanupWorkflowRequest();
72          WorkflowContextSnapshot result = new WorkflowContextSnapshot();
73          result.setPlanId("plan-1");
74          result.setWorkflowKind(WorkflowKind.valueOf(ShadowFeatureDefinition.ALGORITHM_CLEANUP_WORKFLOW_KIND.getValue()));
75          result.setStatus(WorkflowLifecycle.STATUS_PLANNED);
76          result.setRequest(request);
77          ClarifiedIntent clarifiedIntent = new ClarifiedIntent();
78          clarifiedIntent.setOperationType("create");
79          result.setClarifiedIntent(clarifiedIntent);
80          result.setInteractionPlan(InteractionPlan.create("plan-1", request, "Shadow workflow plan.", List.of("review"), List.of("rules")));
81          result.getRuleArtifacts().add(new RuleArtifact("create", "CREATE SHADOW RULE `shadow_rule`(SOURCE=`demo_ds`, SHADOW=`demo_ds_shadow`, `t_order`(TYPE(NAME='sql_hint')))"));
82          return result;
83      }
84      
85      private WorkflowContextFixture createWorkflowContextFixture() {
86          MCPFeatureRequestContext result = mock(MCPFeatureRequestContext.class);
87          WorkflowSessionContext workflowSessionContext = new TestWorkflowSessionContext();
88          MCPFeatureQueryFacade queryFacade = mock(MCPFeatureQueryFacade.class);
89          when(result.getSessionIdentity()).thenReturn(new MCPSessionIdentity("session-1", "", "", Map.of()));
90          when(result.getWorkflowSessionContext()).thenReturn(workflowSessionContext);
91          when(result.getQueryFacade()).thenReturn(queryFacade);
92          when(result.getExecutionFacade()).thenReturn(mock(MCPFeatureExecutionFacade.class));
93          return new WorkflowContextFixture(result, workflowSessionContext, queryFacade);
94      }
95      
96      private record WorkflowContextFixture(MCPFeatureRequestContext requestContext, WorkflowSessionContext workflowSessionContext,
97                                            MCPFeatureQueryFacade queryFacade) {
98      }
