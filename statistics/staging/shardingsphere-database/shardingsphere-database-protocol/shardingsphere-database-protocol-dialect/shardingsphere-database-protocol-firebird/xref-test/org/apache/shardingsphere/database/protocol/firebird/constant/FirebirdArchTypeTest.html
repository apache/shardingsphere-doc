37  class FirebirdArchTypeTest {
38      
39      @ParameterizedTest(name = "{0}")
40      @MethodSource("valueOfCases")
41      void assertValueOf(final String name, final int code, final FirebirdArchType expected) {
42          assertThat(FirebirdArchType.valueOf(code), is(expected));
43      }
44      
45      @Test
46      void assertValueOfWithInvalidCode() {
47          assertThrows(NullPointerException.class, () -> FirebirdArchType.valueOf(999));
48      }
49      
50      @ParameterizedTest(name = "{0}")
51      @MethodSource("isValidCases")
52      void assertIsValid(final String name, final FirebirdArchType arch, final boolean expectedValid) {
53          assertThat(FirebirdArchType.isValid(arch), is(expectedValid));
54      }
55      
56      @ParameterizedTest(name = "{0}")
57      @MethodSource("architectureCases")
58      void assertArchitecture(final String name, final String osName, final String osArch, final String expectedArchitecture) throws ReflectiveOperationException, IOException {
59          assertThat(resolveArchitectureByEnvironment(osName, osArch), is(expectedArchitecture));
60      }
61      
62      private String resolveArchitectureByEnvironment(final String osName, final String osArch) throws ReflectiveOperationException, IOException {
63          String originalOsName = System.getProperty("os.name");
64          String originalOsArch = System.getProperty("os.arch");
65          try {
66              setOrClearProperty("os.name", osName);
67              setOrClearProperty("os.arch", osArch);
68              try (URLClassLoader classLoader = new URLClassLoader(classPathUrls(), null)) {
69                  Class<?> archTypeClass = Class.forName(FirebirdArchType.class.getName(), true, classLoader);
70                  Object architecture = archTypeClass.getField("ARCHITECTURE").get(null);
71                  return ((Enum<?>) architecture).name();
72              }
73          } finally {
74              setOrClearProperty("os.name", originalOsName);
75              setOrClearProperty("os.arch", originalOsArch);
76          }
77      }
78      
79      private void setOrClearProperty(final String key, final String value) {
80          if (null == value) {
81              System.clearProperty(key);
82              return;
83          }
84          System.setProperty(key, value);
85      }
86      
87      private URL[] classPathUrls() throws MalformedURLException {
88          String[] classPathEntries = System.getProperty("java.class.path").split(File.pathSeparator);
89          URL[] result = new URL[classPathEntries.length];
90          for (int i = 0; i < classPathEntries.length; i++) {
91              result[i] = Paths.get(classPathEntries[i]).toUri().toURL();
92          }
93          return result;
94      }

96      private static Stream<Arguments> valueOfCases() {
97          return Stream.of(
98                  Arguments.of("resolve ARCH_GENERIC by code", 1, FirebirdArchType.ARCH_GENERIC),
99                  Arguments.of("resolve ARCH_INTEL_32 by code", 29, FirebirdArchType.ARCH_INTEL_32),

108     }
