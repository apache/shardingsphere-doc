56  @ExtendWith(MockitoExtension.class)
57  class IncrementalTaskPositionManagerTest {
58      
59      private final DatabaseType databaseType = TypedSPILoader.getService(DatabaseType.class, "H2");
60      
61      @Mock
62      private DialectIncrementalPositionManager dialectPositionManager;
63      
64      private IncrementalTaskPositionManager incrementalTaskPositionManager;
65      
66      @SneakyThrows(ReflectiveOperationException.class)
67      @BeforeEach
68      void setUp() {
69          incrementalTaskPositionManager = new IncrementalTaskPositionManager(databaseType);
70          Plugins.getMemberAccessor().set(IncrementalTaskPositionManager.class.getDeclaredField("dialectPositionManager"), incrementalTaskPositionManager, dialectPositionManager);
71      }
72      
73      @Test
74      void assertGetPositionWithInitialProgress() throws SQLException {
75          JobItemIncrementalTasksProgress initialProgress = mock(JobItemIncrementalTasksProgress.class);
76          IngestPosition position = mock(IngestPosition.class);
77          when(initialProgress.getIncrementalPosition()).thenReturn(Optional.of(position));
78          IncrementalDumperContext dumperContext = mockIncrementalDumperContext();
79          assertThat(incrementalTaskPositionManager.getPosition(initialProgress, dumperContext, mock(PipelineDataSourceManager.class)), is(position));
80      }
81      
82      @Test
83      void assertGetPositionWithoutIncrementalProgress() throws SQLException {
84          JobItemIncrementalTasksProgress initialProgress = mock(JobItemIncrementalTasksProgress.class);
85          when(initialProgress.getIncrementalPosition()).thenReturn(Optional.empty());
86          IncrementalDumperContext dumperContext = mockIncrementalDumperContext();
87          PipelineDataSourceManager dataSourceManager = mock(PipelineDataSourceManager.class);
88          PipelineDataSource dataSource = mock(PipelineDataSource.class);
89          when(dataSourceManager.getDataSource(dumperContext.getCommonContext().getDataSourceConfig())).thenReturn(dataSource);
90          IngestPosition position = mock(IngestPosition.class);
91          when(dialectPositionManager.init(dataSource, dumperContext.getJobId())).thenReturn(position);
92          assertThat(incrementalTaskPositionManager.getPosition(initialProgress, dumperContext, dataSourceManager), is(position));
93      }
94      
95      @Test
96      void assertGetPositionWithoutInitialProgress() throws SQLException {
97          IncrementalDumperContext dumperContext = mockIncrementalDumperContext();
98          PipelineDataSourceManager dataSourceManager = mock(PipelineDataSourceManager.class);
99          PipelineDataSource dataSource = mock(PipelineDataSource.class);
100         when(dataSourceManager.getDataSource(dumperContext.getCommonContext().getDataSourceConfig())).thenReturn(dataSource);
101         IngestPosition position = mock(IngestPosition.class);
102         when(dialectPositionManager.init(dataSource, dumperContext.getJobId())).thenReturn(position);
103         assertThat(incrementalTaskPositionManager.getPosition(null, dumperContext, dataSourceManager), is(position));
104     }
105     
106     private IncrementalDumperContext mockIncrementalDumperContext() {
107         IncrementalDumperContext result = mock(IncrementalDumperContext.class, RETURNS_DEEP_STUBS);
108         PipelineDataSourceConfiguration dataSourceConfig = mock(PipelineDataSourceConfiguration.class);
109         when(result.getCommonContext().getDataSourceConfig()).thenReturn(dataSourceConfig);
110         return result;
111     }
112     
113     @Test
114     void assertDestroyPositionWithShardingSpherePipelineDataSourceConfiguration() throws SQLException {
115         YamlRootConfiguration rootConfig = new YamlRootConfiguration();
116         Map<String, Object> dataSourceProps = new HashMap<>(2, 1F);
117         dataSourceProps.put("dataSourceClassName", MockedDataSource.class.getName());
118         dataSourceProps.put("url", "jdbc:mock://127.0.0.1/foo_ds");
119         rootConfig.getDataSources().put("foo_ds", dataSourceProps);
120         ShardingSpherePipelineDataSourceConfiguration pipelineDataSourceConfig = new ShardingSpherePipelineDataSourceConfiguration(rootConfig);
121         incrementalTaskPositionManager.destroyPosition("foo_job", pipelineDataSourceConfig);
122         verify(dialectPositionManager).destroy(any(), eq("foo_job"));
123     }
124     
125     @Test
126     void assertDestroyPositionWithStandardPipelineDataSourceConfiguration() throws SQLException {
127         Map<String, Object> dataSourceProps = new HashMap<>(2, 1F);
128         dataSourceProps.put("dataSourceClassName", MockedDataSource.class.getName());
129         dataSourceProps.put("url", "jdbc:mock://127.0.0.1/foo_ds");
130         StandardPipelineDataSourceConfiguration pipelineDataSourceConfig = new StandardPipelineDataSourceConfiguration(dataSourceProps);
131         incrementalTaskPositionManager.destroyPosition("foo_job", pipelineDataSourceConfig);
132         verify(dialectPositionManager).destroy(any(), eq("foo_job"));
133     }
134     
135     @Test
136     void assertDestroyPositionWithUnknownPipelineDataSourceConfiguration() throws SQLException {
137         incrementalTaskPositionManager.destroyPosition("foo_job", mock(PipelineDataSourceConfiguration.class));
138         verify(dialectPositionManager, never()).destroy(any(), eq("foo_job"));
139     }
