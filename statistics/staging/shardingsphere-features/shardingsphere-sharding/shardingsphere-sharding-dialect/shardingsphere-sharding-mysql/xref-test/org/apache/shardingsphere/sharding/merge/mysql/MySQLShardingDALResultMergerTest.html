64  class MySQLShardingDALResultMergerTest {
65      
66      private final DatabaseType databaseType = TypedSPILoader.getService(DatabaseType.class, "MySQL");
67      
68      private final DialectShardingDALResultMerger resultMerger = DatabaseTypedSPILoader.getService(DialectShardingDALResultMerger.class, databaseType);
69      
70      private final List<QueryResult> queryResults = Collections.singletonList(mock());
71      
72      @Test
73      void assertMergeForShowDatabasesStatement() throws SQLException {
74          SQLStatementContext sqlStatementContext = mockSQLStatementContext(mock(MySQLShowDatabasesStatement.class));
75          Optional<MergedResult> actual = resultMerger.merge("foo_db", mock(), sqlStatementContext, mock(), queryResults);
76          assertTrue(actual.isPresent());
77          assertThat(actual.get(), isA(LocalDataMergedResult.class));
78      }
79      
80      @Test
81      void assertMergeForShowShowTablesStatement() throws SQLException {
82          SQLStatementContext sqlStatementContext = mockSQLStatementContext(new MySQLShowTablesStatement(databaseType, null, null, false));
83          Optional<MergedResult> actual = resultMerger.merge("foo_db", mock(), sqlStatementContext, mock(), queryResults);
84          assertTrue(actual.isPresent());
85          assertThat(actual.get(), isA(MySQLShardingLogicTablesMergedResult.class));
86      }
87      
88      @Test
89      void assertMergeForShowTableStatusStatement() throws SQLException {
90          SQLStatementContext sqlStatementContext = mockSQLStatementContext(new MySQLShowTableStatusStatement(databaseType, null, null));
91          Optional<MergedResult> actual = resultMerger.merge("foo_db", mock(), sqlStatementContext, mock(), queryResults);
92          assertTrue(actual.isPresent());
93          assertThat(actual.get(), isA(MySQLShardingShowTableStatusMergedResult.class));
94      }
95      
96      @Test
97      void assertMergeForShowIndexStatement() throws SQLException {
98          SQLStatementContext sqlStatementContext = mockSQLStatementContext(new MySQLShowIndexStatement(databaseType, null, null));
99          Optional<MergedResult> actual = resultMerger.merge("foo_db", mock(), sqlStatementContext, mock(), queryResults);
100         assertTrue(actual.isPresent());
101         assertThat(actual.get(), isA(MySQLShardingShowIndexMergedResult.class));
102     }
103     
104     @Test
105     void assertMergeForShowCreateTableStatement() throws SQLException {
106         SQLStatementContext sqlStatementContext = mockSQLStatementContext(new MySQLShowCreateTableStatement(databaseType, null));
107         Optional<MergedResult> actual = resultMerger.merge("foo_db", mock(), sqlStatementContext, mock(), queryResults);
108         assertTrue(actual.isPresent());
109         assertThat(actual.get(), isA(MySQLShardingShowCreateTableMergedResult.class));
110     }
111     
112     @ParameterizedTest(name = "{0}")
113     @MethodSource("getExplainSelectQueryResults")
114     void assertMergeForExplainSelect(final String name, final List<QueryResult> queryResults, final Collection<Object> expectedRows) throws SQLException {
115         ExplainStatement sqlStatement = mock(ExplainStatement.class);
116         when(sqlStatement.getExplainableSQLStatement()).thenReturn(mock(SelectStatement.class));
117         SQLStatementContext sqlStatementContext = mock(SQLStatementContext.class);
118         when(sqlStatementContext.getSqlStatement()).thenReturn(sqlStatement);
119         Optional<MergedResult> actual = resultMerger.merge("foo_db", mock(), sqlStatementContext, mock(), queryResults);
120         assertTrue(actual.isPresent());
121         Collection<Object> actualRows = new ArrayList<>(expectedRows.size());
122         while (actual.get().next()) {
123             actualRows.add(actual.get().getValue(1, Object.class));
124         }
125         assertThat(actualRows, is(expectedRows));
126     }
127     
128     private static Stream<Arguments> getExplainSelectQueryResults() throws SQLException {
129         return Stream.of(
130                 Arguments.of("single result", Collections.singletonList(mockQueryResult("foo_plan_0")), Collections.singletonList("foo_plan_0")),
131                 Arguments.of("multiple results", Arrays.asList(mockQueryResult("foo_plan_0"), mockQueryResult("foo_plan_1")), Arrays.asList("foo_plan_0", "foo_plan_1")),
132                 Arguments.of("empty middle result", Arrays.asList(mockQueryResult("foo_plan_0"), mock(QueryResult.class), mockQueryResult("foo_plan_2")), Arrays.asList("foo_plan_0", "foo_plan_2")));
133     }
134     
135     private static QueryResult mockQueryResult(final String value) throws SQLException {
136         QueryResult result = mock(QueryResult.class);
137         when(result.next()).thenReturn(true, false);
138         when(result.getValue(1, Object.class)).thenReturn(value);
139         return result;
140     }
141     
142     @Test
143     void assertMergeForExplainNonSelect() throws SQLException {
144         ExplainStatement sqlStatement = mock(ExplainStatement.class);
145         when(sqlStatement.getExplainableSQLStatement()).thenReturn(mock(InsertStatement.class));
146         SQLStatementContext sqlStatementContext = mock(SQLStatementContext.class);
147         when(sqlStatementContext.getSqlStatement()).thenReturn(sqlStatement);
148         assertFalse(resultMerger.merge("foo_db", mock(), sqlStatementContext, mock(), queryResults).isPresent());
149     }
150     
151     @Test
152     void assertMergeForOtherDALStatement() throws SQLException {
153         SQLStatementContext sqlStatementContext = mock(SQLStatementContext.class);
154         when(sqlStatementContext.getSqlStatement()).thenReturn(mock(DALStatement.class));
155         assertFalse(resultMerger.merge("foo_db", mock(), sqlStatementContext, mock(), queryResults).isPresent());
156     }
157     
158     private SQLStatementContext mockSQLStatementContext(final DALStatement sqlStatement) {
159         SQLStatementContext result = mock(SQLStatementContext.class, RETURNS_DEEP_STUBS);
160         when(result.getSqlStatement()).thenReturn(sqlStatement);
161         when(result.getTablesContext().getSchemaName()).thenReturn(Optional.empty());
162         return result;
163     }
