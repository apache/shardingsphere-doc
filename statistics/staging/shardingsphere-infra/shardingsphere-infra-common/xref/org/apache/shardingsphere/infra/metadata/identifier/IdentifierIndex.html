40  /**
41   * Identifier index for metadata objects.
42   *
43   * @param <T> metadata object type
44   */
45  @RequiredArgsConstructor
46  @Slf4j
47  public final class IdentifierIndex<T> {
48      
49      private final DatabaseIdentifierContext databaseIdentifierContext;
50      
51      private final IdentifierScope identifierScope;
52      
53      private volatile Snapshot<T> snapshot = Snapshot.empty();
54      
55      /**
56       * Rebuild identifier index by actual names.
57       *
58       * @param values actual name and metadata object map
59       */
60      public synchronized void rebuild(final Map<String, T> values) {
61          Map<String, T> newExactValues = new LinkedHashMap<>(values.size(), 1F);
62          Map<String, Collection<String>> newNormalizedIdentifiers = new LinkedHashMap<>(values.size(), 1F);
63          IdentifierCasePolicy policy = databaseIdentifierContext.getMetaDataPolicy(identifierScope);
64          for (Entry<String, T> entry : values.entrySet()) {
65              newExactValues.put(entry.getKey(), entry.getValue());
66              addNormalizedIdentifier(newNormalizedIdentifiers, policy, entry.getKey());
67          }
68          snapshot = createSnapshot(newExactValues, newNormalizedIdentifiers, policy);
69      }
70      
71      /**
72       * Get all metadata objects.
73       *
74       * @return all metadata objects
75       */
76      public Collection<T> getAll() {
77          return snapshot.getExactValues().values();
78      }
79      
80      /**
81       * Get all actual names.
82       *
83       * @return all actual names
84       */
85      public Collection<String> getAllNames() {
86          return snapshot.getExactValues().keySet();
87      }
88      
89      /**
90       * Put metadata object by actual name.
91       *
92       * @param name actual name
93       * @param value metadata object
94       */
95      public synchronized void put(final String name, final T value) {
96          Snapshot<T> currentSnapshot = snapshot;
97          Map<String, T> newExactValues = new LinkedHashMap<>(currentSnapshot.getExactValues());
98          Map<String, Collection<String>> newNormalizedIdentifiers = copyNormalizedIdentifiers(currentSnapshot.getNormalizedIdentifierNames());
99          IdentifierCasePolicy policy = databaseIdentifierContext.getMetaDataPolicy(identifierScope);
100         if (newExactValues.containsKey(name)) {
101             removeNormalizedIdentifier(newNormalizedIdentifiers, policy, name);
102         }
103         newExactValues.put(name, value);
104         addNormalizedIdentifier(newNormalizedIdentifiers, policy, name);
105         snapshot = createSnapshot(newExactValues, newNormalizedIdentifiers, policy);
106     }
107     
108     /**
109      * Remove metadata object by actual name.
110      *
111      * @param name actual name
112      * @return removed metadata object
113      */
114     public synchronized T remove(final String name) {
115         Snapshot<T> currentSnapshot = snapshot;
116         if (!currentSnapshot.getExactValues().containsKey(name)) {
117             return null;
118         }
119         Map<String, T> newExactValues = new LinkedHashMap<>(currentSnapshot.getExactValues());
120         Map<String, Collection<String>> newNormalizedIdentifiers = copyNormalizedIdentifiers(currentSnapshot.getNormalizedIdentifierNames());
121         IdentifierCasePolicy policy = databaseIdentifierContext.getMetaDataPolicy(identifierScope);
122         T result = newExactValues.remove(name);
123         removeNormalizedIdentifier(newNormalizedIdentifiers, policy, name);
124         snapshot = createSnapshot(newExactValues, newNormalizedIdentifiers, policy);
125         return result;
126     }
127     
128     /**
129      * Judge whether identifier index is empty or not.
130      *
131      * @return identifier index is empty or not
132      */
133     public boolean isEmpty() {
134         return snapshot.getExactValues().isEmpty();
135     }
136     
137     /**
138      * Get size of identifier index.
139      *
140      * @return size of identifier index
141      */
142     public int size() {
143         return snapshot.getExactValues().size();
144     }
145     
146     /**
147      * Judge whether contains metadata object by unquoted identifier or not.
148      *
149      * @param identifier unquoted identifier
150      * @return contains metadata object by unquoted identifier or not
151      */
152     public boolean contains(final String identifier) {
153         return null != get(identifier);
154     }
155     
156     /**
157      * Get metadata object by unquoted identifier.
158      *
159      * @param identifier unquoted identifier
160      * @return matched metadata object
161      */
162     public T get(final String identifier) {
163         Snapshot<T> currentSnapshot = snapshot;
164         IdentifierCasePolicy policy = databaseIdentifierContext.getMetaDataPolicy(identifierScope);
165         if (LookupMode.EXACT == policy.getLookupMode(QuoteCharacter.NONE)) {
166             return currentSnapshot.getExactValues().get(identifier);
167         }
168         return getByNormalizedIdentifier(currentSnapshot, policy, identifier);
169     }
170     
171     /**
172      * Find metadata object by identifier value.
173      *
174      * @param identifierValue identifier value
175      * @return matched metadata object
176      */
177     public Optional<T> find(final IdentifierValue identifierValue) {
178         Snapshot<T> currentSnapshot = snapshot;
179         IdentifierCasePolicy policy = databaseIdentifierContext.getMetaDataPolicy(identifierScope);
180         if (LookupMode.EXACT == policy.getLookupMode(identifierValue.getQuoteCharacter())) {
181             return Optional.ofNullable(currentSnapshot.getExactValues().get(identifierValue.getValue()));
182         }
183         return QuoteCharacter.NONE == identifierValue.getQuoteCharacter()
184                 ? Optional.ofNullable(getByNormalizedIdentifier(currentSnapshot, policy, identifierValue.getValue()))
185                 : findByQuotedNormalizedIdentifier(currentSnapshot, policy, identifierValue);
186     }
187     
188     private T getByNormalizedIdentifier(final Snapshot<T> currentSnapshot, final IdentifierCasePolicy policy, final String identifier) {
189         NormalizedBucket<T> normalizedBucket = currentSnapshot.getNormalizedBuckets().get(policy.normalizeForLookup(identifier));
190         if (null == normalizedBucket) {
191             return null;
192         }
193         return getByUnquotedNormalizedIdentifier(currentSnapshot, normalizedBucket, identifier);
194     }
195     
196     private T getByUnquotedNormalizedIdentifier(final Snapshot<T> currentSnapshot, final NormalizedBucket<T> normalizedBucket, final String identifier) {
197         if (!normalizedBucket.hasUnquotedIdentifier()) {
198             return null;
199         }
200         if (normalizedBucket.hasSingleUnquotedIdentifier()) {
201             return normalizedBucket.getSingleUnquotedValue();
202         }
203         T exactMatchedValue = findExactMatchedValue(currentSnapshot.getExactValues(), normalizedBucket.getUnquotedIdentifiers(), identifier);
204         if (null != exactMatchedValue) {
205             return exactMatchedValue;
206         }
207         throw new AmbiguousIdentifierException(identifier, normalizedBucket.getUnquotedIdentifiers());
208     }
209     
210     private Optional<T> findByQuotedNormalizedIdentifier(final Snapshot<T> currentSnapshot, final IdentifierCasePolicy policy, final IdentifierValue identifierValue) {
211         NormalizedBucket<T> normalizedBucket = currentSnapshot.getNormalizedBuckets().get(policy.normalizeForLookup(identifierValue.getValue()));
212         if (null == normalizedBucket) {
213             return Optional.empty();
214         }
215         if (normalizedBucket.hasSingleIdentifier()) {
216             return policy.matches(normalizedBucket.getSingleIdentifier(), identifierValue.getValue(), identifierValue.getQuoteCharacter())
217                     ? Optional.ofNullable(normalizedBucket.getSingleValue())
218                     : Optional.empty();
219         }
220         String matchedIdentifier = null;
221         Collection<String> ambiguousIdentifiers = null;
222         for (String each : normalizedBucket.getIdentifiers()) {
223             if (!policy.matches(each, identifierValue.getValue(), identifierValue.getQuoteCharacter())) {
224                 continue;
225             }
226             if (null == matchedIdentifier) {
227                 matchedIdentifier = each;
228                 continue;
229             }
230             if (null == ambiguousIdentifiers) {
231                 ambiguousIdentifiers = new LinkedList<>();
232                 ambiguousIdentifiers.add(matchedIdentifier);
233             }
234             ambiguousIdentifiers.add(each);
235         }
236         if (null == matchedIdentifier) {
237             return Optional.empty();
238         }
239         if (null == ambiguousIdentifiers) {
240             return Optional.ofNullable(currentSnapshot.getExactValues().get(matchedIdentifier));
241         }
242         T exactMatchedValue = findExactMatchedValue(currentSnapshot.getExactValues(), ambiguousIdentifiers, identifierValue.getValue());
243         if (null != exactMatchedValue) {
244             return Optional.of(exactMatchedValue);
245         }
246         throw new AmbiguousIdentifierException(identifierValue.getValue(), ambiguousIdentifiers);
247     }
248     
249     private T findExactMatchedValue(final Map<String, T> exactValues, final Collection<String> matchedIdentifiers, final String identifierValue) {
250         if (!matchedIdentifiers.contains(identifierValue)) {
251             return null;
252         }
253         log.warn("Identifier '{}' matched multiple actual identifiers {}. Fallback to exact identifier '{}'.", identifierValue, matchedIdentifiers, identifierValue);
254         return exactValues.get(identifierValue);
255     }
256     
257     private void addNormalizedIdentifier(final Map<String, Collection<String>> values, final IdentifierCasePolicy policy, final String name) {
258         String normalizedName = policy.normalizeForLookup(name);
259         values.computeIfAbsent(normalizedName, key -> new LinkedList<>()).add(name);
260     }
261     
262     private void removeNormalizedIdentifier(final Map<String, Collection<String>> values, final IdentifierCasePolicy policy, final String name) {
263         String normalizedName = policy.normalizeForLookup(name);
264         Collection<String> candidateIdentifiers = values.get(normalizedName);
265         if (null == candidateIdentifiers) {
266             return;
267         }
268         candidateIdentifiers.remove(name);
269         if (candidateIdentifiers.isEmpty()) {
270             values.remove(normalizedName);
271         }
272     }
273     
274     private Map<String, Collection<String>> copyNormalizedIdentifiers(final Map<String, Collection<String>> values) {
275         Map<String, Collection<String>> result = new LinkedHashMap<>(values.size(), 1F);
276         for (Entry<String, Collection<String>> entry : values.entrySet()) {
277             result.put(entry.getKey(), new LinkedList<>(entry.getValue()));
278         }
279         return result;
280     }
281     
282     private Snapshot<T> createSnapshot(final Map<String, T> exactValues, final Map<String, Collection<String>> normalizedIdentifiers, final IdentifierCasePolicy policy) {
283         Map<String, Collection<String>> immutableNormalizedIdentifiers = new LinkedHashMap<>(normalizedIdentifiers.size(), 1F);
284         Map<String, NormalizedBucket<T>> normalizedBuckets = new LinkedHashMap<>(normalizedIdentifiers.size(), 1F);
285         for (Entry<String, Collection<String>> entry : normalizedIdentifiers.entrySet()) {
286             Collection<String> identifiers = Collections.unmodifiableCollection(new LinkedList<>(entry.getValue()));
287             immutableNormalizedIdentifiers.put(entry.getKey(), identifiers);
288             normalizedBuckets.put(entry.getKey(), createNormalizedBucket(exactValues, identifiers, policy));
289         }
290         return new Snapshot<>(Collections.unmodifiableMap(new LinkedHashMap<>(exactValues)),
291                 Collections.unmodifiableMap(immutableNormalizedIdentifiers), Collections.unmodifiableMap(normalizedBuckets));
292     }
293     
294     private NormalizedBucket<T> createNormalizedBucket(final Map<String, T> exactValues, final Collection<String> identifiers, final IdentifierCasePolicy policy) {
295         String singleIdentifier = null;
296         T singleValue = null;
297         if (1 == identifiers.size()) {
298             singleIdentifier = identifiers.iterator().next();
299             singleValue = exactValues.get(singleIdentifier);
300         }
301         String singleUnquotedIdentifier = null;
302         T singleUnquotedValue = null;
303         Collection<String> unquotedIdentifiers = null;
304         for (String each : identifiers) {
305             if (!policy.matches(each, each, QuoteCharacter.NONE)) {
306                 continue;
307             }
308             if (null == singleUnquotedIdentifier) {
309                 singleUnquotedIdentifier = each;
310                 singleUnquotedValue = exactValues.get(each);
311                 continue;
312             }
313             if (null == unquotedIdentifiers) {
314                 unquotedIdentifiers = new LinkedList<>();
315                 unquotedIdentifiers.add(singleUnquotedIdentifier);
316             }
317             unquotedIdentifiers.add(each);
318         }
319         return new NormalizedBucket<>(singleIdentifier, singleValue, identifiers, singleUnquotedIdentifier,
320                 singleUnquotedValue, null == unquotedIdentifiers ? null : Collections.unmodifiableCollection(unquotedIdentifiers));
321     }
322     
323     @Override
324     public String toString() {
325         return snapshot.getExactValues().toString();
326     }
327     
328     @AllArgsConstructor(access = AccessLevel.PRIVATE)
329     @Getter(AccessLevel.PRIVATE)
330     private static final class Snapshot<T> {
331         
332         private static final Snapshot<?> EMPTY = new Snapshot<>(Collections.emptyMap(), Collections.emptyMap(), Collections.emptyMap());
333         
334         private final Map<String, T> exactValues;
335         
336         private final Map<String, Collection<String>> normalizedIdentifierNames;
337         
338         private final Map<String, NormalizedBucket<T>> normalizedBuckets;
339         
340         @SuppressWarnings("unchecked")
341         private static <T> Snapshot<T> empty() {
342             return (Snapshot<T>) EMPTY;
343         }
344     }
345     
346     @AllArgsConstructor(access = AccessLevel.PRIVATE)
347     private static final class NormalizedBucket<T> {
348         
349         @Getter(AccessLevel.PRIVATE)
350         private final String singleIdentifier;
351         
352         @Getter(AccessLevel.PRIVATE)
353         private final T singleValue;
354         
355         @Getter(AccessLevel.PRIVATE)
356         private final Collection<String> identifiers;
357         
358         private final String singleUnquotedIdentifier;
359         
360         @Getter(AccessLevel.PRIVATE)
361         private final T singleUnquotedValue;
362         
363         @Getter(AccessLevel.PRIVATE)
364         private final Collection<String> unquotedIdentifiers;
365         
366         private boolean hasSingleIdentifier() {
367             return null != singleIdentifier;
368         }
369         
370         private boolean hasUnquotedIdentifier() {
371             return null != singleUnquotedIdentifier;
372         }
373         
374         private boolean hasSingleUnquotedIdentifier() {
375             return null == unquotedIdentifiers && null != singleUnquotedIdentifier;
376         }
377     }
