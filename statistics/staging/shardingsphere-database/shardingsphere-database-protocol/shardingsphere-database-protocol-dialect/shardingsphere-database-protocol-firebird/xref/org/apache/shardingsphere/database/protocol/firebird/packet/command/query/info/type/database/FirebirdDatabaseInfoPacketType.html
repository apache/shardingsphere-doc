146     private static final Map<Integer, FirebirdDatabaseInfoPacketType> FIREBIRD_DATABASE_INFO_TYPE_CACHE = new HashMap<>();
147     
148     private final int code;
149     
150     static {
151         for (FirebirdDatabaseInfoPacketType each : values()) {
152             FIREBIRD_DATABASE_INFO_TYPE_CACHE.put(each.code, each);
153         }
154     }
155     
156     /**
157      * Value of.
158      *
159      * @param code arch type code
160      * @return Firebird dpb type
161      */
162     public static FirebirdDatabaseInfoPacketType valueOf(final int code) {
163         return FIREBIRD_DATABASE_INFO_TYPE_CACHE.get(code);
164     }
165     
166     /**
167      * Creates info packet of this type.
168      *
169      * @param payload Firebird packet payload
170      * @return Firebird database info packet
171      */
172     public static FirebirdInfoPacket createPacket(final FirebirdPacketPayload payload) {
173         return new FirebirdInfoPacket(payload, FirebirdDatabaseInfoPacketType::valueOf);
174     }
