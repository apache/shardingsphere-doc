176         request.setTable("orders");
177         request.setColumn("phone");
178         ClarifiedIntent clarifiedIntent = new ClarifiedIntent();
179         WorkflowContextSnapshot snapshot = new WorkflowContextSnapshot();
180         MCPMetadataQueryFacade metadataQueryFacade = mock(MCPMetadataQueryFacade.class);
181         when(metadataQueryFacade.queryDatabase("logic_db")).thenReturn(Optional.of(createDatabaseMetadata()));
