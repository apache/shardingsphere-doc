64  class MySQLDialectExceptionMapperTest {
65      
66      private final DatabaseType databaseType = TypedSPILoader.getService(DatabaseType.class, "MySQL");
67      
68      private final SQLDialectExceptionMapper mapper = DatabaseTypedSPILoader.getService(SQLDialectExceptionMapper.class, databaseType);
69      
70      @ParameterizedTest(name = "{0}")
71      @MethodSource("convertArguments")
72      void assertConvert(final String name, final Class<? extends SQLDialectException> sqlDialectExceptionClass, final VendorError vendorError) {
73          assertSQLException(mapper.convert(mock(sqlDialectExceptionClass)), vendorError);
74      }
75      
76      private void assertSQLException(final SQLException actual, final VendorError vendorError) {
77          assertThat(actual.getSQLState(), is(vendorError.getSqlState().getValue()));
78          assertThat(actual.getErrorCode(), is(vendorError.getVendorCode()));
79      }
80      
81      @Test
82      void assertConvertWithUnknownDatabaseName() {
83          assertSQLExceptionWithMessage(mapper.convert(new UnknownDatabaseException("logic_db")), MySQLVendorError.ER_BAD_DB_ERROR, "logic_db");
84      }
85      
86      @Test
87      void assertConvertWithUnknownCollation() {
88          assertSQLExceptionWithMessage(mapper.convert(new UnknownCollationException("foo_collation")), MySQLVendorError.ER_UNKNOWN_COLLATION, "foo_collation");
89      }
90      
91      @Test
92      void assertConvertWithInvalidParameterValue() {
93          assertSQLExceptionWithMessage(
94                  mapper.convert(new InvalidParameterValueException("character_set_client", "ucs2")), MySQLVendorError.ER_WRONG_VALUE_FOR_VAR, "character_set_client", "ucs2");
95      }
96      
97      @Test
98      void assertConvertWithCollationCharsetMismatch() {
99          assertSQLExceptionWithMessage(mapper.convert(new CollationCharsetMismatchException("utf8mb4_bin", "latin1")),
100                 MySQLVendorError.ER_COLLATION_CHARSET_MISMATCH, "utf8mb4_bin", "latin1");
101     }
102     
103     @Test
104     void assertConvertWithUsingPassword() {
105         assertSQLExceptionWithMessage(mapper.convert(new AccessDeniedException("root", "127.0.0.1", true)), MySQLVendorError.ER_ACCESS_DENIED_ERROR, "root", "127.0.0.1", "YES");
106     }
107     
108     @Test
109     void assertConvertWithColumnNotFound() {
110         assertSQLExceptionWithMessage(mapper.convert(new ColumnNotFoundException("t_order", "order_id")), MySQLVendorError.ER_BAD_FIELD_ERROR, "order_id", "field list");
111     }
112     
113     private void assertSQLExceptionWithMessage(final SQLException actual, final VendorError vendorError, final Object... messageArgs) {
114         assertSQLException(actual, vendorError);
115         assertThat(actual.getMessage(), is(String.format(vendorError.getReason(), messageArgs)));
116     }
117     
118     @Test
119     void assertConvertWithUnknownException() {
120         SQLDialectException sqlDialectException = mock(SQLDialectException.class);
121         SQLException expected = new UnknownSQLException(sqlDialectException).toSQLException();
122         SQLException actual = mapper.convert(sqlDialectException);
123         assertThat(actual.getMessage(), is(expected.getMessage()));
124         assertThat(actual.getSQLState(), is(expected.getSQLState()));
125         assertThat(actual.getErrorCode(), is(expected.getErrorCode()));
126         assertThat(actual.getCause(), is(expected.getCause()));
127     }
128     
129     private static Stream<Arguments> convertArguments() {
130         return Stream.of(
131                 Arguments.of("unknown_database_without_name", UnknownDatabaseException.class, MySQLVendorError.ER_NO_DB_ERROR),
132                 Arguments.of("no_database_selected", NoDatabaseSelectedException.class, MySQLVendorError.ER_NO_DB_ERROR),
133                 Arguments.of("database_create_exists", DatabaseCreateExistsException.class, MySQLVendorError.ER_DB_CREATE_EXISTS_ERROR),
134                 Arguments.of("database_drop_not_exists", DatabaseDropNotExistsException.class, MySQLVendorError.ER_DB_DROP_NOT_EXISTS_ERROR),
135                 Arguments.of("table_exists", TableExistsException.class, MySQLVendorError.ER_TABLE_EXISTS_ERROR),
136                 Arguments.of("dialect_sql_parsing", DialectSQLParsingException.class, MySQLVendorError.ER_PARSE_ERROR),
137                 Arguments.of("no_such_table", NoSuchTableException.class, MySQLVendorError.ER_NO_SUCH_TABLE),
138                 Arguments.of("insert_columns_and_values_mismatched", InsertColumnsAndValuesMismatchedException.class, MySQLVendorError.ER_WRONG_VALUE_COUNT_ON_ROW),
139                 Arguments.of("table_modify_in_transaction", TableModifyInTransactionException.class, MySQLVendorError.ER_ERROR_ON_MODIFYING_GTID_EXECUTED_TABLE),
140                 Arguments.of("too_many_connections", TooManyConnectionsException.class, MySQLVendorError.ER_CON_COUNT_ERROR),
141                 Arguments.of("unsupported_prepared_statement", UnsupportedPreparedStatementException.class, MySQLVendorError.ER_UNSUPPORTED_PS),
142                 Arguments.of("too_many_placeholders", TooManyPlaceholdersException.class, MySQLVendorError.ER_PS_MANY_PARAM),
143                 Arguments.of("unknown_charset", UnknownCharsetException.class, MySQLVendorError.ER_UNKNOWN_CHARACTER_SET),
144                 Arguments.of("handshake", HandshakeException.class, MySQLVendorError.ER_HANDSHAKE_ERROR),
145                 Arguments.of("access_denied_without_password", AccessDeniedException.class, MySQLVendorError.ER_ACCESS_DENIED_ERROR),
146                 Arguments.of("database_access_denied", DatabaseAccessDeniedException.class, MySQLVendorError.ER_DBACCESS_DENIED_ERROR),
147                 Arguments.of("unknown_system_variable", UnknownSystemVariableException.class, MySQLVendorError.ER_UNKNOWN_SYSTEM_VARIABLE),
148                 Arguments.of("error_local_variable", ErrorLocalVariableException.class, MySQLVendorError.ER_LOCAL_VARIABLE),
149                 Arguments.of("error_global_variable", ErrorGlobalVariableException.class, MySQLVendorError.ER_GLOBAL_VARIABLE),
150                 Arguments.of("incorrect_global_local_variable", IncorrectGlobalLocalVariableException.class, MySQLVendorError.ER_INCORRECT_GLOBAL_LOCAL_VAR),
151                 Arguments.of("column_not_found", ColumnNotFoundException.class, MySQLVendorError.ER_BAD_FIELD_ERROR));
152     }
