85  @ExtendWith(MockitoExtension.class)
86  class FirebirdCommandExecutorFactoryTest {
87      
88      @Mock(answer = Answers.RETURNS_DEEP_STUBS)
89      private ConnectionSession connectionSession;
90      
91      @ParameterizedTest(name = "{0}")
92      @MethodSource("newInstanceArguments")
93      void assertNewInstance(final String name, final FirebirdCommandPacketType commandPacketType, final Class<? extends CommandPacket> commandPacketClass,
94                             final Class<? extends CommandExecutor> expectedExecutorClass) {
95          assertThat(FirebirdCommandExecutorFactory.newInstance(commandPacketType, mock(commandPacketClass), connectionSession), isA(expectedExecutorClass));
96      }
