47  /**
48   * Metadata factory for PostgreSQL prepared statements.
49   */
50  @NoArgsConstructor(access = AccessLevel.PRIVATE)
51  public final class PostgreSQLPreparedStatementMetadataFactory {
52      
53      /**
54       * Load actual prepared statement for metadata access.
55       *
56       * @param connectionSession connection session
57       * @param preparedStatement prepared statement
58       * @param parameters parameters
59       * @return actual prepared statement
60       * @throws SQLException SQL exception
61       */
62      public static PreparedStatement load(final ConnectionSession connectionSession, final PostgreSQLServerPreparedStatement preparedStatement, final List<Object> parameters) throws SQLException {
63          ShardingSphereMetaData metaData = ProxyContext.getInstance().getContextManager().getMetaDataContexts().getMetaData();
64          SQLStatementContext sqlStatementContext = new SQLBindEngine(metaData, connectionSession.getCurrentDatabaseName(), preparedStatement.getHintValueContext())
65                  .bind(preparedStatement.getSqlStatementContext().getSqlStatement());
66          if (sqlStatementContext instanceof ParameterAware) {
67              ((ParameterAware) sqlStatementContext).bindParameters(parameters);
68          }
69          QueryContext queryContext = new QueryContext(
70                  sqlStatementContext, preparedStatement.getSql(), parameters, preparedStatement.getHintValueContext(), connectionSession.getConnectionContext(), metaData);
71          ExecutionContext executionContext = new KernelProcessor().generateExecutionContext(queryContext, metaData.getGlobalRuleMetaData(), metaData.getProps());
72          ShardingSpherePreconditions.checkNotEmpty(executionContext.getExecutionUnits(),
73                  () -> new PreparedStatementMetadataResolutionException("no execution unit was generated"));
74          ExecutionUnit executionUnit = executionContext.getExecutionUnits().iterator().next();
75          ProxyDatabaseConnectionManager databaseConnectionManager = connectionSession.getDatabaseConnectionManager();
76          List<Connection> connections =
77                  databaseConnectionManager.getConnections(connectionSession.getUsedDatabaseName(), executionUnit.getDataSourceName(), 0, 1, ConnectionMode.CONNECTION_STRICTLY);
78          ShardingSpherePreconditions.checkNotEmpty(connections,
79                  () -> new PreparedStatementMetadataResolutionException("no backend connection was acquired"));
80          String sql = executionUnit.getSqlUnit().getSql();
81          PreparedStatement result = connections.iterator().next().prepareStatement(sql);
82          try {
83              if (executionContext.getExecutionUnits().size() > 1) {
84                  Optional<DialectJDBCResultMetadataChecker> checker =
85                          DatabaseTypedSPILoader.findService(DialectJDBCResultMetadataChecker.class, connectionSession.getProtocolType());
86                  if (checker.isPresent()) {
87                      checker.get().check(executionContext.getExecutionUnits(), result, sql);
88                  }
89              }
90          } catch (final SQLException | ShardingSphereSQLException ex) {
91              QuietlyCloser.close(result);
92              throw ex;
93          }
94          return result;
95      }
96  }
