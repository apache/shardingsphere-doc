52  /**
53   * Pipeline data source configuration for ShardingSphere-JDBC.
54   */
55  @Getter
56  @EqualsAndHashCode(of = "parameter")
57  public final class ShardingSpherePipelineDataSourceConfiguration implements PipelineDataSourceConfiguration {
58      
59      public static final String TYPE = "ShardingSphereJDBC";
60      
61      private final String parameter;
62      
63      private final String databaseName;
64      
65      private final Map<String, DataSourcePoolProperties> dataSourcePoolPropertiesMap;
66      
67      @Getter(AccessLevel.NONE)
68      private final Map<String, String> jdbcUrlKeys;
69      
70      @Getter(AccessLevel.NONE)
71      private final Map<String, Collection<String>> customPoolPropertiesKeys;
72      
73      private final Collection<RuleConfiguration> ruleConfigurations;
74      
75      private final DatabaseType databaseType;
76      
77      public ShardingSpherePipelineDataSourceConfiguration(final String param) {
78          YamlRootConfiguration rootConfig = YamlEngine.unmarshal(param, YamlRootConfiguration.class, true);
79          // Need remove dataSourceProperties, because if the parameter at dataSourceProperties will override parameter at jdbcUrl
80          for (Map<String, Object> each : rootConfig.getDataSources().values()) {
81              each.remove("dataSourceProperties");
82          }
83          parameter = YamlEngine.marshal(rootConfig);
84          Map<String, Object> props = rootConfig.getDataSources().values().iterator().next();
85          databaseType = DatabaseTypeFactory.get(getJdbcUrl(props));
86          databaseName = rootConfig.getDatabaseName();
87          dataSourcePoolPropertiesMap = createDataSourcePoolPropertiesMap(rootConfig.getDataSources());
88          jdbcUrlKeys = new HashMap<>(rootConfig.getDataSources().size(), 1F);
89          customPoolPropertiesKeys = new HashMap<>(rootConfig.getDataSources().size(), 1F);
90          rootConfig.getDataSources().forEach((key, value) -> {
91              jdbcUrlKeys.put(key, value.containsKey("url") ? "url" : "jdbcUrl");
92              if (value.containsKey("customPoolProps")) {
93                  customPoolPropertiesKeys.put(key, new LinkedHashSet<>(((Map<String, Object>) value.get("customPoolProps")).keySet()));
94              }
95          });
96          ruleConfigurations = new YamlRuleConfigurationSwapperEngine().swapToRuleConfigurations(rootConfig.getRules());
97      }
98      
99      public ShardingSpherePipelineDataSourceConfiguration(final YamlRootConfiguration rootConfig) {
100         this(YamlEngine.marshal(getYamlParameterConfiguration(rootConfig)));
101     }
102     
103     private static YamlParameterConfiguration getYamlParameterConfiguration(final YamlRootConfiguration rootConfig) {
104         YamlParameterConfiguration result = new YamlParameterConfiguration();
105         result.setDatabaseName(rootConfig.getDatabaseName());
106         result.setDataSources(rootConfig.getDataSources());
107         result.setRules(rootConfig.getRules());
108         result.getProps().putAll(rootConfig.getProps());
109         return result;
110     }
111     
112     private String getJdbcUrl(final Map<String, Object> props) {
113         Object result = props.getOrDefault("url", props.get("jdbcUrl"));
114         Preconditions.checkNotNull(result, "url or jdbcUrl is required.");
115         return result.toString();
116     }
117     
118     private void appendJdbcQueryProperties(final Map<String, Object> dataSourceProps, final JdbcQueryPropertiesExtension extension) {
119         String jdbcUrlKey = dataSourceProps.containsKey("url") ? "url" : "jdbcUrl";
120         String jdbcUrl = dataSourceProps.get(jdbcUrlKey).toString();
121         Properties queryProps = new StandardJdbcUrlParser().parseQueryProperties(jdbcUrl.contains("?") ? jdbcUrl.substring(jdbcUrl.indexOf("?") + 1) : "");
122         extension.extendQueryProperties(queryProps);
123         dataSourceProps.put(jdbcUrlKey, new JdbcUrlAppender().appendQueryProperties(jdbcUrl, queryProps));
124     }
125     
126     private DataSourcePoolProperties adjustDataSourcePoolProperties(final DataSourcePoolProperties dataSourcePoolProps, final Optional<JdbcQueryPropertiesExtension> extension,
127                                                                     final Map<String, Object> customPoolProps) {
128         Map<String, Object> localProps = dataSourcePoolProps.getAllLocalProperties();
129         extension.ifPresent(optional -> appendJdbcQueryProperties(localProps, optional));
130         for (String each : Arrays.asList("minPoolSize", "minimumIdle")) {
131             localProps.put(each, "1");
132         }
133         localProps.putAll(customPoolProps);
134         return new DataSourcePoolProperties(dataSourcePoolProps.getPoolClassName(), localProps);
135     }
136     
137     @SuppressWarnings("unchecked")
138     private Map<String, DataSourcePoolProperties> createDataSourcePoolPropertiesMap(final Map<String, Map<String, Object>> dataSources) {
139         Map<String, DataSourcePoolProperties> result = new LinkedHashMap<>(dataSources.size(), 1F);
140         YamlDataSourceConfigurationSwapper swapper = new YamlDataSourceConfigurationSwapper();
141         Optional<JdbcQueryPropertiesExtension> extension = DatabaseTypedSPILoader.findService(JdbcQueryPropertiesExtension.class, databaseType);
142         dataSources.forEach((key, value) -> {
143             Map<String, Object> customPoolProps = (Map<String, Object>) value.get("customPoolProps");
144             result.put(key, adjustDataSourcePoolProperties(swapper.swapToDataSourcePoolProperties(value), extension, null == customPoolProps ? Collections.emptyMap() : customPoolProps));
145         });
146         return result;
147     }
148     
149     /**
150      * Get rule configurations for data source creation.
151      *
152      * @return independent rule configurations
153      */
154     public Collection<RuleConfiguration> getCreationRuleConfigurations() {
155         YamlRuleConfigurationSwapperEngine swapper = new YamlRuleConfigurationSwapperEngine();
156         YamlRootConfiguration yamlConfig = new YamlRootConfiguration();
157         yamlConfig.setRules(swapper.swapToYamlRuleConfigurations(ruleConfigurations));
158         return swapper.swapToRuleConfigurations(YamlEngine.unmarshal(YamlEngine.marshal(yamlConfig), YamlRootConfiguration.class, true).getRules());
159     }
160     
161     @Override
162     public Object getDataSourceConfiguration() {
163         return this;
164     }
165     
166     @Override
167     public String getType() {
168         return TYPE;
169     }
170     
171     /**
172      * Get actual data source configuration.
173      *
174      * @param actualDataSourceName actual data source name
175      * @return actual data source configuration
176      */
177     public StandardPipelineDataSourceConfiguration getActualDataSourceConfiguration(final String actualDataSourceName) {
178         DataSourcePoolProperties dataSourcePoolProps = dataSourcePoolPropertiesMap.get(actualDataSourceName);
179         Preconditions.checkNotNull(dataSourcePoolProps, "actualDataSourceName '{}' does not exist", actualDataSourceName);
180         Map<String, Object> dataSourceConfig = new YamlDataSourceConfigurationSwapper().swapToMap(dataSourcePoolProps);
181         if ("jdbcUrl".equals(jdbcUrlKeys.get(actualDataSourceName)) && dataSourceConfig.containsKey("url")) {
182             dataSourceConfig.put("jdbcUrl", dataSourceConfig.remove("url"));
183         }
184         Collection<String> customPoolPropsKeys = customPoolPropertiesKeys.get(actualDataSourceName);
185         if (null != customPoolPropsKeys) {
186             Map<String, Object> customPoolProps = new LinkedHashMap<>(customPoolPropsKeys.size(), 1F);
187             Map<String, Object> localProps = dataSourcePoolProps.getAllLocalProperties();
188             customPoolPropsKeys.forEach(each -> {
189                 if (localProps.containsKey(each)) {
190                     customPoolProps.put(each, localProps.get(each));
191                 } else if (dataSourceConfig.containsKey(each)) {
192                     customPoolProps.put(each, dataSourceConfig.get(each));
193                 }
194                 dataSourceConfig.remove(each);
195             });
196             dataSourceConfig.put("customPoolProps", customPoolProps);
197         }
198         for (String each : Arrays.asList("minPoolSize", "minimumIdle")) {
199             dataSourceConfig.put(each, "1");
200         }
201         return new StandardPipelineDataSourceConfiguration(dataSourceConfig);
202     }
