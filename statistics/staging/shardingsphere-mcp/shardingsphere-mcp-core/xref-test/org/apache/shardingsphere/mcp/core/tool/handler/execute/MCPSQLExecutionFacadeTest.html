178         MCPDatabaseCapabilityProvider capabilityProvider = mock(MCPDatabaseCapabilityProvider.class);
179         MCPSessionExecutionCoordinator coordinator = mock(MCPSessionExecutionCoordinator.class);
180         MCPJdbcTransactionStatementExecutor transactionExecutor = mock(MCPJdbcTransactionStatementExecutor.class);

186         MCPSQLExecutionFacade facade = createFacade(capabilityProvider, coordinator, transactionExecutor, statementExecutor);
187         try (
188                 MockedConstruction<ShardingSphereSQLParserEngine> ignored = mockConstruction(ShardingSphereSQLParserEngine.class,

190             MCPInvalidRequestException actual = assertThrows(MCPInvalidRequestException.class, () -> facade.execute(createExecutionRequest(sql)));
191             assertThat(actual.getMessage(), is(String.format("Cross-schema SQL is not supported for database `logic_db`: `%s`.", expectedObject)));
192             verifyNoInteractions(transactionExecutor, statementExecutor);
193         }
