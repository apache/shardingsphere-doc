18  package org.apache.shardingsphere.database.protocol.firebird.packet.generic;
19  
20  import lombok.Getter;
21  import lombok.RequiredArgsConstructor;
22  import org.apache.shardingsphere.database.protocol.firebird.err.FirebirdStatusVector;
23  import org.apache.shardingsphere.database.protocol.firebird.packet.FirebirdPacket;
24  import org.apache.shardingsphere.database.protocol.firebird.packet.command.FirebirdCommandPacketType;
25  import org.apache.shardingsphere.database.protocol.firebird.payload.FirebirdPacketPayload;
26  
27  import java.util.ArrayList;
28  import java.util.List;
29  
30  /**
31   * Batch completion state response (op_batch_cs) for Firebird.
32   *
33   * <p>Failures are reported as detailed errors (record number plus status vector). The simplified-errors list
34   * (p_batch_errors, record numbers without a status vector) is always empty: the proxy holds the full status vector for
35   * the failure, so it never needs the record-number-only fallback that Firebird uses to cap server-side memory.</p>
36   */
37  public class FirebirdBatchCompletionStateResponse extends FirebirdPacket {
38      
39      private int statementHandle;
40      
41      private long recordsCount;
42      
43      private int[] updateCounts = new int[0];
44      
45      private final List<DetailedError> detailedErrors = new ArrayList<>();
46      
47      /**
48       * Set statement handle.
49       *
50       * @param objectHandle statement handle
51       * @return this response
52       */
53      public FirebirdBatchCompletionStateResponse setHandle(final int objectHandle) {
54          statementHandle = objectHandle;
55          return this;
56      }
57      
58      /**
59       * Set records count.
60       *
61       * @param recordsCount records count
62       * @return this response
63       */
64      public FirebirdBatchCompletionStateResponse setRecordsCount(final long recordsCount) {
65          this.recordsCount = recordsCount;
66          return this;
67      }
68      
69      /**
70       * Set update counts.
71       *
72       * @param updateCounts update counts
73       * @return this response
74       */
75      public FirebirdBatchCompletionStateResponse setUpdateCounts(final int[] updateCounts) {
76          this.updateCounts = null == updateCounts ? new int[0] : updateCounts;
77          return this;
78      }
79      
80      /**
81       * Add detailed error for a failed batch message.
82       *
83       * @param element zero-based index of the failed batch message
84       * @param statusVector status vector describing the failure
85       * @return this response
86       */
87      public FirebirdBatchCompletionStateResponse addDetailedError(final int element, final FirebirdStatusVector statusVector) {
88          detailedErrors.add(new DetailedError(element, statusVector));
89          return this;
90      }
91      
92      @Override
93      protected void write(final FirebirdPacketPayload payload) {
94          payload.writeInt4(FirebirdCommandPacketType.BATCH_CS.getValue());
95          payload.writeInt4(statementHandle);
96          payload.writeInt4(Math.toIntExact(recordsCount));
97          payload.writeInt4(updateCounts.length);
98          payload.writeInt4(detailedErrors.size());
99          payload.writeInt4(0);
100         for (int c : updateCounts) {
101             payload.writeInt4(c);
102         }
103         for (DetailedError each : detailedErrors) {
104             payload.writeInt4(each.element);
105             each.statusVector.write(payload);
106         }
107     }
108     
109     /**
110      * Detailed error reporting a failed batch message.
111      */
112     @RequiredArgsConstructor
113     @Getter
114     public static final class DetailedError {
115         
116         private final int element;
117         
118         private final FirebirdStatusVector statusVector;
119     }
120 }
