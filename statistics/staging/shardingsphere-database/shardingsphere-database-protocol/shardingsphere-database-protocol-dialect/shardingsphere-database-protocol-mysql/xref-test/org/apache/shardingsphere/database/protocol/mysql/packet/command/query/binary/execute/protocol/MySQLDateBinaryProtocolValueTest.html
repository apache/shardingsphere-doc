49  @ExtendWith(MockitoExtension.class)
50  class MySQLDateBinaryProtocolValueTest {
51      
52      @Mock
53      private MySQLPacketPayload payload;
54      
55      @Test
56      void assertReadWithZeroByte() {
57          when(payload.readInt1()).thenReturn(0);
58          assertThrows(SQLFeatureNotSupportedException.class, () -> new MySQLDateBinaryProtocolValue().read(payload, false));
59      }
60      
61      @ParameterizedTest(name = "{0}")
62      @MethodSource("readArguments")
63      void assertRead(final String name, final int[] readInt1Values, final int year, final int microseconds, final LocalDateTime expected) throws SQLException {
64          Integer[] tailValues = new Integer[readInt1Values.length - 1];
65          for (int i = 1; i < readInt1Values.length; i++) {
66              tailValues[i - 1] = readInt1Values[i];
67          }
68          when(payload.readInt1()).thenReturn(readInt1Values[0], tailValues);
69          when(payload.readInt2()).thenReturn(year);
70          if (0 <= microseconds) {
71              when(payload.readInt4()).thenReturn(microseconds);
72          }
73          Timestamp actual = (Timestamp) new MySQLDateBinaryProtocolValue().read(payload, false);
74          assertThat(actual.toLocalDateTime(), is(expected));
75      }
76      
77      @Test
78      void assertReadWithIllegalArgument() {
79          when(payload.readInt1()).thenReturn(100);
80          assertThrows(SQLFeatureNotSupportedException.class, () -> new MySQLDateBinaryProtocolValue().read(payload, false));
81      }
82      
83      @ParameterizedTest(name = "{0}")
84      @MethodSource("writeArguments")
85      void assertWrite(final String name, final Object value, final int expectedLength, final int year, final int month,
86                       final int dayOfMonth, final int hour, final int minute, final int second, final Integer expectedMicroseconds) {
87          ByteBuf buffer = Unpooled.buffer();
88          try {
89              new MySQLDateBinaryProtocolValue().write(new MySQLPacketPayload(buffer, StandardCharsets.UTF_8), value);
90              assertThat(buffer.readUnsignedByte(), is((short) expectedLength));
91              assertThat(buffer.readUnsignedShortLE(), is(year));
92              assertThat(buffer.readUnsignedByte(), is((short) month));
93              assertThat(buffer.readUnsignedByte(), is((short) dayOfMonth));
94              if (4 != expectedLength) {
95                  assertThat(buffer.readUnsignedByte(), is((short) hour));
96                  assertThat(buffer.readUnsignedByte(), is((short) minute));
97                  assertThat(buffer.readUnsignedByte(), is((short) second));
98              }
99              if (11 == expectedLength) {
100                 assertThat(buffer.readUnsignedIntLE(), is(expectedMicroseconds.longValue()));
101             }
102             assertThat(buffer.readableBytes(), is(0));
103         } finally {
104             buffer.release();
105         }
106     }

108     @Test
109     void assertWriteWithNanoOverflowWithoutSecondCarry() {
110         LocalDateTime dateTime = mock(LocalDateTime.class);
111         when(dateTime.getYear()).thenReturn(2026);
112         when(dateTime.getMonthValue()).thenReturn(1);
113         when(dateTime.getDayOfMonth()).thenReturn(2);
114         when(dateTime.getHour()).thenReturn(3);
115         when(dateTime.getMinute()).thenReturn(4);
116         when(dateTime.getSecond()).thenReturn(10);
117         when(dateTime.getNano()).thenReturn(1_000_000_001);
118         new MySQLDateBinaryProtocolValue().write(payload, dateTime);
119         InOrder inOrder = inOrder(payload);
120         inOrder.verify(payload).writeInt1(11);
121         inOrder.verify(payload).writeInt2(2026);
122         inOrder.verify(payload).writeInt1(1);
123         inOrder.verify(payload).writeInt1(2);
124         inOrder.verify(payload).writeInt1(3);
125         inOrder.verify(payload).writeInt1(4);
126         inOrder.verify(payload).writeInt1(11);
127         inOrder.verify(payload).writeInt4(0);
128     }
129     
130     @Test
131     void assertWriteWithNanoOverflowAndSecondCarry() {
132         LocalDateTime dateTime = mock(LocalDateTime.class);
133         LocalDateTime normalized = LocalDateTime.of(2026, 1, 2, 3, 4, 1);
134         when(dateTime.getYear()).thenReturn(2026);
135         when(dateTime.getMonthValue()).thenReturn(1);
136         when(dateTime.getDayOfMonth()).thenReturn(2);
137         when(dateTime.getHour()).thenReturn(3);
138         when(dateTime.getMinute()).thenReturn(4);
139         when(dateTime.getSecond()).thenReturn(59);
140         when(dateTime.getNano()).thenReturn(2_000_000_000);
141         when(dateTime.plusSeconds(2)).thenReturn(normalized);
142         new MySQLDateBinaryProtocolValue().write(payload, dateTime);
143         verify(dateTime).plusSeconds(2);
144         InOrder inOrder = inOrder(payload);
145         inOrder.verify(payload).writeInt1(7);
146         inOrder.verify(payload).writeInt2(2026);
147         inOrder.verify(payload).writeInt1(1);
148         inOrder.verify(payload).writeInt1(2);
149         inOrder.verify(payload).writeInt1(3);
150         inOrder.verify(payload).writeInt1(4);
151         inOrder.verify(payload).writeInt1(1);
152     }
