59      @Test
60      void assertCreateStorageResourceWithStatementSetsFetchSizeOnlyInMemoryStrictly() throws SQLException {
61          ConnectionSession connectionSession = mock(ConnectionSession.class);
62          JDBCBackendStatement backendStatement = new JDBCBackendStatement(connectionSession);
63          Connection connection = mock(Connection.class);
64          Statement statement = mock(Statement.class);
65          when(connection.createStatement()).thenReturn(statement);
66          StatementMemoryStrictlyFetchSizeSetter fetchSizeSetter = mock(StatementMemoryStrictlyFetchSizeSetter.class);
67          when(DatabaseTypedSPILoader.findService(StatementMemoryStrictlyFetchSizeSetter.class, databaseType)).thenReturn(Optional.of(fetchSizeSetter));
68          StatementOption statementOption = new StatementOption(false);
69          Statement actualMemoryStrict = backendStatement.createStorageResource(connection, ConnectionMode.MEMORY_STRICTLY, statementOption, databaseType);
70          assertThat(actualMemoryStrict, is(statement));
71          verify(fetchSizeSetter).setFetchSize(statement);
72          assertThat(backendStatement.createStorageResource(connection, ConnectionMode.CONNECTION_STRICTLY, statementOption, databaseType), is(statement));
73          verifyNoMoreInteractions(fetchSizeSetter);
74      }

76      @Test
77      void assertCreateStorageResourceWithPreparedStatement() throws SQLException {
78          ConnectionSession connectionSession = mock(ConnectionSession.class);
79          when(connectionSession.getCurrentPreparedStatementCacheKey()).thenReturn(Optional.empty());
80          JDBCBackendStatement backendStatement = new JDBCBackendStatement(connectionSession);
81          Connection connection = mock(Connection.class);
82          PreparedStatement preparedStatementWithGeneratedKeys = mock(PreparedStatement.class);
83          PreparedStatement preparedStatement = mock(PreparedStatement.class);
84          String sql = "SELECT * FROM foo WHERE id = ? AND name = ?";
85          when(connection.prepareStatement(sql, Statement.RETURN_GENERATED_KEYS)).thenReturn(preparedStatementWithGeneratedKeys);
86          when(connection.prepareStatement(sql)).thenReturn(preparedStatement);
87          when(DatabaseTypedSPILoader.findService(StatementMemoryStrictlyFetchSizeSetter.class, databaseType)).thenReturn(Optional.empty());
88          Object typeUnspecifiedParam = mock(TypeUnspecifiedSQLParameter.class);
89          SQLUnit sqlUnitWithParams = new SQLUnit(sql, Arrays.asList(typeUnspecifiedParam, "foo_param"));
90          ExecutionUnit executionUnitWithParams = new ExecutionUnit("ds", sqlUnitWithParams);
91          StatementOption statementOption = new StatementOption(true);
92          Statement actualWithGeneratedKeys = backendStatement.createStorageResource(executionUnitWithParams, connection, 0, ConnectionMode.MEMORY_STRICTLY, statementOption, databaseType);
93          assertThat(actualWithGeneratedKeys, is(preparedStatementWithGeneratedKeys));
94          verify(preparedStatementWithGeneratedKeys).setObject(1, typeUnspecifiedParam, Types.OTHER);
95          verify(preparedStatementWithGeneratedKeys).setObject(2, "foo_param");
96          Statement actualWithoutGeneratedKeys = backendStatement.createStorageResource(
97                  new ExecutionUnit("ds", new SQLUnit(sql, Collections.emptyList())), connection, 0, ConnectionMode.CONNECTION_STRICTLY, new StatementOption(false), databaseType);
98          assertThat(actualWithoutGeneratedKeys, is(preparedStatement));
99      }

101     @Test
102     void assertCreateStorageResourceReusesPreparedStatementForSamePreparedStatementCacheKey() throws SQLException {
103         ConnectionSession connectionSession = mock(ConnectionSession.class);
104         PreparedStatementCacheContext cacheContext = new PreparedStatementCacheContext(8);
105         when(connectionSession.getPreparedStatementCacheContext()).thenReturn(cacheContext);
106         when(connectionSession.getCurrentPreparedStatementCacheKey()).thenReturn(Optional.of(new PreparedStatementCacheKey("statement-1")));
107         JDBCBackendStatement backendStatement = new JDBCBackendStatement(connectionSession);
108         Connection connection = mock(Connection.class);
109         PreparedStatement preparedStatement = mock(PreparedStatement.class);
110         String sql = "SELECT * FROM foo WHERE id = ?";
111         when(connection.prepareStatement(sql)).thenReturn(preparedStatement);
112         StatementOption statementOption = new StatementOption(false);
113         backendStatement.createStorageResource(
114                 new ExecutionUnit("ds", new SQLUnit(sql, Collections.singletonList(1))), connection, 0, ConnectionMode.CONNECTION_STRICTLY, statementOption, databaseType);
115         backendStatement.createStorageResource(
116                 new ExecutionUnit("ds", new SQLUnit(sql, Collections.singletonList(2))), connection, 0, ConnectionMode.CONNECTION_STRICTLY, statementOption, databaseType);
117         verify(connection).prepareStatement(sql);
118         verify(preparedStatement, times(2)).clearParameters();
119         verify(preparedStatement).setObject(1, 1);
120         verify(preparedStatement).setObject(1, 2);
121     }

160     @Test
161     void assertCreateNewPreparedStatementAfterCacheInvalidation() throws SQLException {
162         ConnectionSession connectionSession = mock(ConnectionSession.class);
163         PreparedStatementCacheContext cacheContext = new PreparedStatementCacheContext(8);
164         PreparedStatementCacheKey preparedStatementCacheKey = new PreparedStatementCacheKey("statement-1");
165         when(connectionSession.getPreparedStatementCacheContext()).thenReturn(cacheContext);
166         when(connectionSession.getCurrentPreparedStatementCacheKey()).thenReturn(Optional.of(preparedStatementCacheKey));
167         JDBCBackendStatement backendStatement = new JDBCBackendStatement(connectionSession);
168         Connection connection = mock(Connection.class);
169         PreparedStatement firstPreparedStatement = mock(PreparedStatement.class);
170         PreparedStatement secondPreparedStatement = mock(PreparedStatement.class);
171         String sql = "SELECT * FROM foo WHERE id = ?";
172         when(connection.prepareStatement(sql)).thenReturn(firstPreparedStatement, secondPreparedStatement);
173         StatementOption statementOption = new StatementOption(false);
174         backendStatement.createStorageResource(
175                 new ExecutionUnit("ds", new SQLUnit(sql, Collections.singletonList(1))), connection, 0, ConnectionMode.CONNECTION_STRICTLY, statementOption, databaseType);
176         cacheContext.invalidate(preparedStatementCacheKey);
177         backendStatement.createStorageResource(
178                 new ExecutionUnit("ds", new SQLUnit(sql, Collections.singletonList(2))), connection, 0, ConnectionMode.CONNECTION_STRICTLY, statementOption, databaseType);
179         verify(connection, times(2)).prepareStatement(sql);
180         verify(firstPreparedStatement).close();
181     }
