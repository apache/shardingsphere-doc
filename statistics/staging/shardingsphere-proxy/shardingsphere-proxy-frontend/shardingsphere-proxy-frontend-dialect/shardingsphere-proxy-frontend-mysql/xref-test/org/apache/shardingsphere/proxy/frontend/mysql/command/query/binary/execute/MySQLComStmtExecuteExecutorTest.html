162     private SelectStatement prepareSelectStatement() {
163         return SelectStatement.builder().databaseType(databaseType).projections(new ProjectionsSegment(0, 0)).build();
164     }

166     private UpdateStatement prepareUpdateStatement() {
167         ColumnSegment columnSegment = new ColumnSegment(0, 0, new IdentifierValue("col"));
168         ColumnAssignmentSegment columnAssignmentSegment = new ColumnAssignmentSegment(0, 0, Collections.singletonList(columnSegment), new ParameterMarkerExpressionSegment(0, 0, 0));
169         return UpdateStatement.builder()
170                 .databaseType(databaseType)
171                 .setAssignment(new SetAssignmentSegment(0, 0, Collections.singletonList(columnAssignmentSegment)))
172                 .build();
173     }

175     private MySQLServerPreparedStatement prepareLongDataStatement(final MySQLBinaryColumnType columnType) {
176         MySQLServerPreparedStatement result = new MySQLServerPreparedStatement("SELECT ?", prepareSelectStatementContext(), new HintValueContext());
177         result.getParameterTypes().add(new MySQLPreparedStatementParameterType(MySQLBinaryColumnType.STRING, 0));
178         result.getParameterColumnTypes().add(columnType);
179         return result;
180     }

191     @Test
192     void assertIsQueryResponse() throws SQLException {
193         MySQLComStmtExecutePacket packet = mock(MySQLComStmtExecutePacket.class);
194         when(packet.getStatementId()).thenReturn(1);
195         MySQLComStmtExecuteExecutor executor = new MySQLComStmtExecuteExecutor(packet, connectionSession);
196         QueryHeader queryHeader = mock(QueryHeader.class);
197         when(queryHeader.getColumnTypeName()).thenReturn("VARCHAR");
198         when(proxyBackendHandler.execute()).thenReturn(new QueryResponseHeader(Collections.singletonList(queryHeader)));
199         when(proxyBackendHandler.next()).thenReturn(true, false);
200         when(proxyBackendHandler.getRowData()).thenReturn(new QueryResponseRow(Collections.singletonList(new QueryResponseCell(Types.INTEGER, 1))));
201         when(ProxyBackendHandlerFactory.newInstance(eq(databaseType), any(QueryContext.class), eq(connectionSession), anyBoolean())).thenReturn(proxyBackendHandler);
202         Iterator<DatabasePacket> actual = executor.execute().iterator();
203         assertThat(executor.getResponseType(), is(ResponseType.QUERY));
204         assertThat(actual.next(), isA(MySQLFieldCountPacket.class));
205         assertThat(actual.next(), isA(MySQLColumnDefinition41Packet.class));
206         assertThat(actual.next(), isA(MySQLEofPacket.class));
207         assertFalse(actual.hasNext());
208         assertTrue(executor.next());
209         MySQLPacket actualQueryRowPacket = executor.getQueryRowPacket();
210         assertThat(actualQueryRowPacket, isA(MySQLBinaryResultSetRowPacket.class));
211         executor.close();
212         verify(proxyBackendHandler).close();
213     }
