61  class FirebirdDialectExceptionMapperTest {
62      
63      private final DatabaseType databaseType = TypedSPILoader.getService(DatabaseType.class, "Firebird");
64      
65      private final SQLDialectExceptionMapper mapper = DatabaseTypedSPILoader.getService(SQLDialectExceptionMapper.class, databaseType);
66      
67      @ParameterizedTest(name = "{0}")
68      @MethodSource("convertArguments")
69      void assertConvert(final String name, final SQLDialectException sqlDialectException, final VendorError vendorError, final Object[] messageArgs) {
70          SQLException actual = mapper.convert(sqlDialectException);
71          assertThat(actual.getSQLState(), is(vendorError.getSqlState().getValue()));
72          assertThat(actual.getErrorCode(), is(vendorError.getVendorCode()));
73          assertThat(actual.getMessage(), is(String.format(vendorError.getReason(), messageArgs)));
74      }
75      
76      @Test
77      void assertConvertWithLoginError() {
78          SQLException actual = mapper.convert(new LoginErrorException());
79          assertThat(actual.getSQLState(), is("28000"));
80          assertThat(actual.getErrorCode(), is(ISCConstants.isc_login_error));
81          assertThat(actual.getMessage(), is(""));
82      }
83      
84      @Test
85      void assertConvertWithUnknownException() {
86          assertThat(mapper.convert(mock(SQLDialectException.class)).getSQLState(), is("HY000"));
87      }
88
