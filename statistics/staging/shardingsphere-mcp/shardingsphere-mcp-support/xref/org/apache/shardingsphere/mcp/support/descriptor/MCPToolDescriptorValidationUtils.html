35  /**
36   * MCP tool descriptor validation utilities.
37   */
38  @NoArgsConstructor(access = AccessLevel.PRIVATE)
39  public final class MCPToolDescriptorValidationUtils {
40      
41      private static final Collection<String> REQUIRED_WORKFLOW_PLAN_OUTPUT_FIELDS = List.of(
42              "response_mode", MCPPayloadFieldNames.SUMMARY, WorkflowFieldNames.PLAN_ID, "workflow_kind", "status", "missing_required_inputs", "clarification_questions",
43              "elicitation_support", "fallback_reason", "issues", "global_steps", "current_step", "algorithm_recommendations", "property_requirements",
44              "validation_strategy", "delivery_mode", "execution_mode", "intent_inference", "argument_provenance", "review_focus", "proxy_topology_hint",
45              "distsql_artifacts", MCPPayloadFieldNames.RESOURCES_TO_READ, MCPPayloadFieldNames.NEXT_ACTIONS);
46      
47      private static final Collection<String> REQUIRED_WORKFLOW_PLAN_SCHEMA_FIELDS = List.of(
48              "response_mode", WorkflowFieldNames.PLAN_ID, "workflow_kind", "status", "missing_required_inputs",
49              MCPPayloadFieldNames.RESOURCES_TO_READ, MCPPayloadFieldNames.NEXT_ACTIONS);
50      
51      private static final Collection<String> REQUIRED_WORKFLOW_PLAN_META_FIELDS = List.of(
52              "org.apache.shardingsphere/workflow-kind", "org.apache.shardingsphere/artifact-categories", "org.apache.shardingsphere/side-effect-scope",
53              "org.apache.shardingsphere/related-resource-uris", "org.apache.shardingsphere/follow-up-tools");
54      
55      /**
56       * Validate required workflow planning output fields.
57       *
58       * @param descriptor tool descriptor
59       */
60      public static void validateRequiredWorkflowPlanOutputFields(final MCPToolDescriptor descriptor) {
61          validateRequiredOutputFields(descriptor, REQUIRED_WORKFLOW_PLAN_OUTPUT_FIELDS);
62          validateRequiredWorkflowPlanSchemaFields(descriptor);
63      }
64      
65      /**
66       * Validate required workflow planning output fields with additional fields.
67       *
68       * @param descriptor tool descriptor
69       * @param additionalFields additional required fields
70       */
71      public static void validateRequiredWorkflowPlanOutputFields(final MCPToolDescriptor descriptor, final Collection<String> additionalFields) {
72          Collection<String> requiredFields = new LinkedList<>(REQUIRED_WORKFLOW_PLAN_OUTPUT_FIELDS);
73          requiredFields.addAll(additionalFields);
74          validateRequiredOutputFields(descriptor, requiredFields);
75          validateRequiredWorkflowPlanSchemaFields(descriptor);
76      }
77      
78      private static void validateRequiredWorkflowPlanSchemaFields(final MCPToolDescriptor descriptor) {
79          Object required = descriptor.getOutputSchema().get("required");
80          ShardingSpherePreconditions.checkState(required instanceof Collection,
81                  () -> new IllegalStateException(String.format("Tool `%s` outputSchema must declare required workflow fields.", descriptor.getName())));
82          for (String each : REQUIRED_WORKFLOW_PLAN_SCHEMA_FIELDS) {
83              ShardingSpherePreconditions.checkState(((Collection<?>) required).contains(each),
84                      () -> new IllegalStateException(String.format("Tool `%s` outputSchema must require `%s`.", descriptor.getName(), each)));
85          }
86      }
87      
88      /**
89       * Validate required output fields.
90       *
91       * @param descriptor tool descriptor
92       * @param requiredFields required fields
93       */
94      public static void validateRequiredOutputFields(final MCPToolDescriptor descriptor, final Collection<String> requiredFields) {
95          Map<?, ?> properties = (Map<?, ?>) descriptor.getOutputSchema().get("properties");
96          for (String each : requiredFields) {
97              ShardingSpherePreconditions.checkState(properties.containsKey(each),
98                      () -> new IllegalStateException(String.format("Tool `%s` outputSchema must declare `%s`.", descriptor.getName(), each)));
99              Object property = properties.get(each);
100             ShardingSpherePreconditions.checkState(property instanceof Map,
101                     () -> new IllegalStateException(String.format("Tool `%s` outputSchema property `%s` must be an object.", descriptor.getName(), each)));
102             Object description = ((Map<?, ?>) property).get("description");
103             checkDescription(null == description ? "" : description.toString(), String.format("Tool output field `%s.%s` description", descriptor.getName(), each));
104         }
105     }

107     /**
108      * Find input schema property.
109      *
110      * @param descriptor tool descriptor
111      * @param fieldName field name
112      * @return found input schema property
113      */
114     public static Optional<Map<?, ?>> findToolInputProperty(final MCPToolDescriptor descriptor, final String fieldName) {
115         return findToolSchemaProperty(descriptor.getInputSchema(), fieldName);
116     }
117     
118     /**
119      * Find output schema property.
120      *
121      * @param descriptor tool descriptor
122      * @param fieldName field name
123      * @return found output schema property
124      */
125     public static Optional<Map<?, ?>> findToolOutputProperty(final MCPToolDescriptor descriptor, final String fieldName) {
126         return findToolSchemaProperty(descriptor.getOutputSchema(), fieldName);
127     }
128     
129     private static Optional<Map<?, ?>> findToolSchemaProperty(final Map<String, Object> schema, final String fieldName) {
130         Object properties = schema.get("properties");
131         if (!(properties instanceof Map)) {
132             return Optional.empty();
133         }
134         Object property = ((Map<?, ?>) properties).get(fieldName);
135         return property instanceof Map ? Optional.of((Map<?, ?>) property) : Optional.empty();
136     }
137     
138     /**
139      * Check whether input schema property is required.
140      *
141      * @param descriptor tool descriptor
142      * @param fieldName field name
143      * @return whether input schema property is required
144      */
145     public static boolean isRequiredToolInput(final MCPToolDescriptor descriptor, final String fieldName) {
146         Object required = descriptor.getInputSchema().get("required");
147         return required instanceof Collection && ((Collection<?>) required).contains(fieldName);
148     }
149     
150     /**
151      * Validate that a model-facing schema does not use removed field aliases.
152      *
153      * @param descriptor tool descriptor
154      * @param schema model-facing schema
155      */
156     static void validateModelFacingSchemaFields(final MCPToolDescriptor descriptor, final Map<String, Object> schema) {
157         validateNoRemovedModelFacingFields(descriptor, schema);
158     }
159     
160     private static void validateNoRemovedModelFacingFields(final MCPToolDescriptor descriptor, final Object value) {
161         if (value instanceof Map) {
162             validateNoRemovedModelFacingFieldMap(descriptor, (Map<?, ?>) value);
163         } else if (value instanceof Collection) {
164             for (Object each : (Collection<?>) value) {
165                 validateNoRemovedModelFacingFields(descriptor, each);
166             }
167         }
168     }
169     
170     private static void validateNoRemovedModelFacingFieldMap(final MCPToolDescriptor descriptor, final Map<?, ?> value) {
171         for (Entry<?, ?> entry : value.entrySet()) {
172             String key = String.valueOf(entry.getKey());
173             validateNoRemovedModelFacingField(descriptor, key);
174             if ("required".equals(key)) {
175                 validateNoRemovedModelFacingRequiredFields(descriptor, entry.getValue());
176             }
177             validateNoRemovedModelFacingFields(descriptor, entry.getValue());
178         }
179     }
180     
181     private static void validateNoRemovedModelFacingRequiredFields(final MCPToolDescriptor descriptor, final Object value) {
182         if (!(value instanceof Collection)) {
183             return;
184         }
185         for (Object each : (Collection<?>) value) {
186             validateNoRemovedModelFacingField(descriptor, String.valueOf(each));
187         }
188     }
189     
190     private static void validateNoRemovedModelFacingField(final MCPToolDescriptor descriptor, final String fieldName) {
191         ShardingSpherePreconditions.checkState(!MCPModelFacingPayloadContract.isRemovedFieldName(fieldName),
192                 () -> new IllegalStateException(String.format("Tool `%s` model-facing contract must use canonical fields instead of removed `%s`.", descriptor.getName(), fieldName)));
193     }
194     
195     /**
196      * Validate required workflow planning metadata fields.
197      *
198      * @param descriptor tool descriptor
199      */
200     public static void validateRequiredWorkflowPlanMetaFields(final MCPToolDescriptor descriptor) {
201         validateRequiredMetaFields(descriptor, REQUIRED_WORKFLOW_PLAN_META_FIELDS);
202     }
203     
204     /**
205      * Validate required metadata fields.
206      *
207      * @param descriptor tool descriptor
208      * @param requiredFields required fields
209      */
210     public static void validateRequiredMetaFields(final MCPToolDescriptor descriptor, final Collection<String> requiredFields) {
211         Map<String, Object> meta = descriptor.getMeta();
212         for (String each : requiredFields) {
213             ShardingSpherePreconditions.checkState(meta.containsKey(each),
214                     () -> new IllegalStateException(String.format("Tool `%s` metadata must declare `%s`.", descriptor.getName(), each)));
215         }
216     }
217     
218     /**
219      * Check description.
220      *
221      * @param value description value
222      * @param label description label
223      */
224     public static void checkDescription(final String value, final String label) {
225         ShardingSpherePreconditions.checkState(null != value && !value.isBlank(), () -> new IllegalStateException(String.format("%s is required.", label)));
226     }
