67      @ParameterizedTest(name = "{0}")
68      @MethodSource("createWithSystemDatabaseArguments")
69      void assertCreateWithSystemDatabase(final String name, final DatabaseType protocolType, final String databaseName, final String expectedDefaultSchemaName, final int expectedSchemaCount) {
70          ShardingSphereDatabase actual = ShardingSphereDatabaseFactory.create(databaseName, protocolType, props);
71          assertThat(actual.getName(), is(databaseName));
72          assertThat(actual.getProtocolType(), is(protocolType));
73          assertThat(actual.getAllSchemas().size(), is(expectedSchemaCount));
74          assertTrue(actual.containsSchema(expectedDefaultSchemaName));
75          assertTrue(actual.getRuleMetaData().getRules().isEmpty());
76      }

78      @Test
79      void assertCreateWithDatabaseConfiguration() throws SQLException {
80          DatabaseType protocolType = mock(DatabaseType.class);
81          Map<String, ShardingSphereSchema> systemSchemas = new LinkedHashMap<>(2, 1F);
82          systemSchemas.put("system_schema", new ShardingSphereSchema("system_schema", protocolType));
83          systemSchemas.put("metadata_schema", new ShardingSphereSchema("metadata_schema", protocolType));
84          try (
85                  MockedStatic<DefaultSchemaNameResolver> mockedDefaultSchemaNameResolver = mockStatic(DefaultSchemaNameResolver.class);
86                  MockedStatic<DatabaseRulesBuilder> mockedRulesBuilder = mockStatic(DatabaseRulesBuilder.class);
87                  MockedStatic<GenericSchemaBuilder> mockedGenericSchemaBuilder = mockStatic(GenericSchemaBuilder.class);
88                  MockedStatic<SystemSchemaBuilder> mockedSystemSchemaBuilder = mockStatic(SystemSchemaBuilder.class)) {
89              ComputeNodeInstanceContext computeNodeInstanceContext = mock(ComputeNodeInstanceContext.class);
90              mockedDefaultSchemaNameResolver.when(() -> DefaultSchemaNameResolver.resolveProtocol(protocolType, "foo_db")).thenReturn("foo_schema");
91              mockedRulesBuilder.when(() -> DatabaseRulesBuilder.build(eq("foo_db"), eq(protocolType), eq(databaseConfig), eq(computeNodeInstanceContext), any(ResourceMetaData.class)))
92                      .thenReturn(Collections.singleton(mock(ShardingSphereRule.class)));
93              mockedGenericSchemaBuilder.when(() -> GenericSchemaBuilder.buildWithInventoryCheck(eq(protocolType), any(GenericSchemaBuilderMaterial.class)))
94                      .thenReturn(Collections.singletonMap("foo_schema", new ShardingSphereSchema("foo_schema", protocolType)));
95              mockedSystemSchemaBuilder.when(() -> SystemSchemaBuilder.build("foo_db", protocolType, props)).thenReturn(systemSchemas);
96              ShardingSphereDatabase actual = ShardingSphereDatabaseFactory.create("foo_db", protocolType, databaseConfig, props, computeNodeInstanceContext);
97              mockedDefaultSchemaNameResolver.verify(() -> DefaultSchemaNameResolver.resolveProtocol(protocolType, "foo_db"));
98              assertThat(actual.getName(), is("foo_db"));
99              assertThat(actual.getProtocolType(), is(protocolType));
100             assertThat(actual.getAllSchemas().size(), is(3));
101             assertThat(actual.getRuleMetaData().getRules().size(), is(1));
102             assertTrue(actual.containsSchema("foo_schema"));
103             assertTrue(actual.containsSchema("system_schema"));
104             assertTrue(actual.containsSchema("metadata_schema"));
105         }
106     }

108     @Test
109     void assertCreateWithDatabaseConfigurationAndRevisionCandidateSchemas() throws SQLException {
110         DatabaseType protocolType = mock(DatabaseType.class);
111         ShardingSphereSchema candidateSchema = new ShardingSphereSchema("candidate_schema", protocolType);
112         try (
113                 MockedStatic<DefaultSchemaNameResolver> mockedDefaultSchemaNameResolver = mockStatic(DefaultSchemaNameResolver.class);
114                 MockedStatic<DatabaseRulesBuilder> mockedRulesBuilder = mockStatic(DatabaseRulesBuilder.class);
115                 MockedStatic<GenericSchemaBuilder> mockedGenericSchemaBuilder = mockStatic(GenericSchemaBuilder.class);
116                 MockedStatic<SystemSchemaBuilder> mockedSystemSchemaBuilder = mockStatic(SystemSchemaBuilder.class)) {
117             ComputeNodeInstanceContext computeNodeInstanceContext = mock(ComputeNodeInstanceContext.class);
118             mockedDefaultSchemaNameResolver.when(() -> DefaultSchemaNameResolver.resolveProtocol(protocolType, "foo_db")).thenReturn("foo_schema");
119             mockedRulesBuilder.when(() -> DatabaseRulesBuilder.build(eq("foo_db"), eq(protocolType), eq(databaseConfig), eq(computeNodeInstanceContext), any(ResourceMetaData.class)))
120                     .thenReturn(Collections.singleton(mock(ShardingSphereRule.class)));
121             mockedGenericSchemaBuilder.when(() -> GenericSchemaBuilder.buildWithInventoryCheck(eq(protocolType), argThat(material -> material.getRevisionCandidateSchemas().contains(candidateSchema))))
122                     .thenReturn(Collections.singletonMap("foo_schema", new ShardingSphereSchema("foo_schema", protocolType)));
123             mockedSystemSchemaBuilder.when(() -> SystemSchemaBuilder.build("foo_db", protocolType, props)).thenReturn(Collections.emptyMap());
124             ShardingSphereDatabase actual = ShardingSphereDatabaseFactory.createWithRevisionCandidateSchemas(
125                     "foo_db", protocolType, databaseConfig, props, computeNodeInstanceContext, Collections.singleton(candidateSchema));
126             mockedDefaultSchemaNameResolver.verify(() -> DefaultSchemaNameResolver.resolveProtocol(protocolType, "foo_db"));
127             assertTrue(actual.containsSchema("foo_schema"));
128         }
129     }
