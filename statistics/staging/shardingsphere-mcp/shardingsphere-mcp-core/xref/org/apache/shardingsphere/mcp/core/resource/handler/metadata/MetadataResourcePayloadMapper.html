18  package org.apache.shardingsphere.mcp.core.resource.handler.metadata;
19  
20  import lombok.RequiredArgsConstructor;
21  import org.apache.shardingsphere.database.connector.core.metadata.database.enums.TableType;
22  import org.apache.shardingsphere.infra.metadata.database.schema.model.ShardingSphereIndex;
23  import org.apache.shardingsphere.infra.metadata.database.schema.model.ShardingSphereSchema;
24  import org.apache.shardingsphere.infra.metadata.database.schema.model.ShardingSphereTable;
25  import org.apache.shardingsphere.mcp.api.capability.resource.MCPResourceURIVariables;
26  import org.apache.shardingsphere.mcp.support.database.capability.SupportedMCPMetadataObjectType;
27  import org.apache.shardingsphere.mcp.support.database.metadata.jdbc.RuntimeDatabaseProfile;
28  import org.apache.shardingsphere.mcp.support.database.metadata.model.MCPColumnMetadata;
29  import org.apache.shardingsphere.mcp.support.database.metadata.model.MCPSequenceMetadata;
30  import org.apache.shardingsphere.mcp.support.database.spi.MCPMetadataQueryFacade;
31  import org.apache.shardingsphere.mcp.support.descriptor.ShardingSphereMCPResourceMetadata;
32  
33  import java.util.Comparator;
34  import java.util.LinkedHashMap;
35  import java.util.List;
36  import java.util.Map;
37  
38  /**
39   * Metadata resource payload mapper.
40   */
41  @RequiredArgsConstructor
42  public final class MetadataResourcePayloadMapper {
43      
44      private final MCPMetadataQueryFacade metadataQueryFacade;
45      
46      private final MCPResourceURIVariables uriVariables;
47      
48      private final boolean detail;
49      
50      /**
51       * Map internal metadata objects to MCP resource payload objects.
52       *
53       * @param metadata resource metadata
54       * @param items internal metadata objects
55       * @return MCP resource payload objects
56       */
57      public List<?> map(final ShardingSphereMCPResourceMetadata metadata, final List<?> items) {
58          return items.stream().map(each -> mapItem(metadata, each)).toList();
59      }
60      
61      private Object mapItem(final ShardingSphereMCPResourceMetadata metadata, final Object item) {
62          if (item instanceof RuntimeDatabaseProfile databaseProfile) {
63              return createDatabasePayload(databaseProfile);
64          }
65          if (item instanceof ShardingSphereSchema schema) {
66              return createSchemaPayload(getDatabase(), schema, detail);
67          }
68          if (item instanceof ShardingSphereTable table) {
69              return "view".equals(metadata.getObjectScope()) ? createViewPayload(getDatabase(), getSchema(), table, detail) : createTablePayload(getDatabase(), getSchema(), table, detail);
70          }
71          if (item instanceof MCPColumnMetadata columnMetadata) {
72              return createColumnPayload(columnMetadata);
73          }
74          if (item instanceof ShardingSphereIndex index) {
75              return createIndexPayload(index);
76          }
77          if (item instanceof MCPSequenceMetadata sequenceMetadata) {
78              return createSequencePayload(sequenceMetadata);
79          }
80          return item;
81      }
82      
83      private Map<String, Object> createDatabasePayload(final RuntimeDatabaseProfile databaseProfile) {
84          Map<String, Object> result = new LinkedHashMap<>(4, 1F);
85          result.put("database", databaseProfile.getDatabase());
86          result.put("databaseType", databaseProfile.getDatabaseType());
87          result.put("databaseVersion", databaseProfile.getDatabaseVersion());
88          result.put("schemas", detail
89                  ? metadataQueryFacade.querySchemas(databaseProfile.getDatabase()).stream().map(each -> createSchemaPayload(databaseProfile.getDatabase(), each, true)).toList()
90                  : List.of());
91          return result;
92      }
93      
94      private Map<String, Object> createSchemaPayload(final String database, final ShardingSphereSchema schema, final boolean detail) {
95          Map<String, Object> result = new LinkedHashMap<>(5, 1F);
96          result.put("database", database);
97          result.put("schema", schema.getName());
98          result.put("tables", detail
99                  ? schema.getAllTables().stream().filter(each -> TableType.TABLE == each.getType()).sorted(Comparator.comparing(ShardingSphereTable::getName))
100                         .map(each -> createTablePayload(database, schema.getName(), each, false)).toList()
101                 : List.of());
102         result.put("views", detail
103                 ? schema.getAllTables().stream().filter(each -> TableType.VIEW == each.getType()).sorted(Comparator.comparing(ShardingSphereTable::getName))
104                         .map(each -> createViewPayload(database, schema.getName(), each, false)).toList()
105                 : List.of());
106         result.put("sequences", detail
107                 && metadataQueryFacade.isSupportedMetadataObjectType(database, SupportedMCPMetadataObjectType.SEQUENCE)
108                         ? metadataQueryFacade.querySequences(database, schema.getName()).stream().map(this::createSequencePayload).toList()
109                         : List.of());
110         return result;
111     }
112     
113     private Map<String, Object> createTablePayload(final String database, final String schema, final ShardingSphereTable table, final boolean detail) {
114         Map<String, Object> result = new LinkedHashMap<>(5, 1F);
115         result.put("database", database);
116         result.put("schema", schema);
117         result.put("table", table.getName());
118         result.put("columns", detail
119                 ? metadataQueryFacade.queryTableColumns(database, schema, table.getName()).stream()
120                         .map(each -> createColumnPayload(database, schema, table.getName(), "", each)).toList()
121                 : List.of());
122         result.put("indexes", detail
123                 ? metadataQueryFacade.queryIndexes(database, schema, table.getName()).stream().map(each -> createIndexPayload(database, schema, table.getName(), each)).toList()
124                 : List.of());
125         return result;
126     }
127     
128     private Map<String, Object> createViewPayload(final String database, final String schema, final ShardingSphereTable view, final boolean detail) {
129         Map<String, Object> result = new LinkedHashMap<>(4, 1F);
130         result.put("database", database);
131         result.put("schema", schema);
132         result.put("view", view.getName());
133         result.put("columns", detail
134                 ? metadataQueryFacade.queryViewColumns(database, schema, view.getName()).stream()
135                         .map(each -> createColumnPayload(database, schema, "", view.getName(), each)).toList()
136                 : List.of());
137         return result;
138     }
139     
140     private Map<String, Object> createColumnPayload(final MCPColumnMetadata column) {
141         return createColumnPayload(getDatabase(), getSchema(), getVariable("table"), getVariable("view"), column);
142     }
143     
144     private Map<String, Object> createColumnPayload(final String database, final String schema, final String table, final String view, final MCPColumnMetadata column) {
145         Map<String, Object> result = new LinkedHashMap<>(9, 1F);
146         result.put("database", database);
147         result.put("schema", schema);
148         result.put("table", table);
149         result.put("view", view);
150         result.put("column", column.getName());
151         result.put("ordinalPosition", column.getOrdinalPosition());
152         result.put("jdbcType", column.getJdbcType());
153         result.put("nativeTypeName", column.getNativeTypeName());
154         result.put("nullability", column.getNullability().getValue());
155         return result;
156     }
157     
158     private Map<String, Object> createIndexPayload(final ShardingSphereIndex index) {
159         return createIndexPayload(getDatabase(), getSchema(), getVariable("table"), index);
160     }
161     
162     private Map<String, Object> createIndexPayload(final String database, final String schema, final String table, final ShardingSphereIndex index) {
163         Map<String, Object> result = new LinkedHashMap<>(6, 1F);
164         result.put("database", database);
165         result.put("schema", schema);
166         result.put("table", table);
167         result.put("index", index.getName());
168         result.put("columns", index.getColumns());
169         result.put("unique", index.isUnique());
170         return result;
171     }
172     
173     private Map<String, Object> createSequencePayload(final MCPSequenceMetadata sequence) {
174         Map<String, Object> result = new LinkedHashMap<>(3, 1F);
175         result.put("database", sequence.getDatabase());
176         result.put("schema", sequence.getSchema());
177         result.put("sequence", sequence.getSequence());
178         return result;
179     }
180     
181     private String getDatabase() {
182         return getVariable("database");
183     }
184     
185     private String getSchema() {
186         return getVariable("schema");
187     }
188     
189     private String getVariable(final String variableName) {
190         return uriVariables.containsVariable(variableName) ? uriVariables.getValue(variableName) : "";
191     }
192 }
