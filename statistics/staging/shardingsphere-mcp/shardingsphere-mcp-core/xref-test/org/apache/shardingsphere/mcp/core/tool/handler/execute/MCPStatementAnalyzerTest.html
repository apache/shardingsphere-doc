47      private final MCPStatementAnalyzer analyzer = new MCPStatementAnalyzer();
48      
49      @ParameterizedTest(name = "{0}")
50      @MethodSource("supportedDatabaseTypes")
51      void assertAnalyzeWithSupportedDatabaseType(final String databaseType, final String sql) {
52          ClassificationResult actual = analyzer.analyze(sql, createCapability(databaseType));
53          assertThat(actual.getStatementClass(), is(SupportedMCPStatement.QUERY));
54          assertThat(actual.getStatementType(), is("SELECT"));
55      }
56      
57      @ParameterizedTest(name = "{0}")
58      @MethodSource("statementCases")
59      void assertAnalyze(final String name, final String databaseType, final String sql, final SupportedMCPStatement expectedStatementClass,
60                         final String expectedStatementType, final String expectedNormalizedSql, final String expectedTargetObjectName, final String expectedSavepointName) {
61          ClassificationResult actual = analyzer.analyze(sql, createCapability(databaseType));
62          assertThat(actual.getStatementClass(), is(expectedStatementClass));
63          assertThat(actual.getStatementType(), is(expectedStatementType));
64          assertThat(actual.getNormalizedSql(), is(expectedNormalizedSql));
65          assertThat(actual.getTargetObjectName().orElse(""), is(expectedTargetObjectName));
66          assertThat(actual.getSavepointName().orElse(""), is(expectedSavepointName));
67      }
68      
69      @ParameterizedTest(name = "{0}")
70      @MethodSource("invalidStatementCases")
71      void assertAnalyzeWithInvalidStatement(final String name, final String databaseType, final String sql, final Class<? extends RuntimeException> expectedExceptionClass,
72                                             final String expectedMessage) {
73          RuntimeException actual = assertThrows(expectedExceptionClass, () -> analyzer.analyze(sql, createCapability(databaseType)));
74          assertThat(actual.getMessage(), is(expectedMessage));
75      }
76      
77      @ParameterizedTest(name = "{0}")
78      @MethodSource("metadataStatementCases")
79      void assertAnalyzeMetadataStatement(final String name, final String sql, final String expectedStatementType) {
80          MetadataIntrospectionSQLStatementException actual = assertThrows(
81                  MetadataIntrospectionSQLStatementException.class, () -> analyzer.analyze(sql, createCapability("MySQL")));
82          assertThat(actual.getStatementType(), is(expectedStatementType));
83      }

85      @Test
86      void assertAnalyzeReferencedObjects() {
87          ClassificationResult actual = analyzer.analyze(
88                  "SELECT * FROM logic_db.orders JOIN other_db.order_items ON orders.order_id = order_items.order_id", createCapability("MySQL"));
89          assertThat(getObjectNames(actual), contains("logic_db.orders", "other_db.order_items"));
90      }

93      @MethodSource("referencedObjectCases")

95          assertThat(getObjectNames(analyzer.analyze(sql, createCapability(databaseType))), is(expected));
96      }

167     private List<String> getObjectNames(final ClassificationResult classificationResult) {
168         return classificationResult.getReferencedObjects().stream().map(SQLStatementObjectName::getObjectName).toList();
169     }
170     
171     @Test
172     void assertAnalyzeRuleDistSQL() {
173         ClassificationResult actual = analyzer.analyze(
174                 "CREATE SHARDING TABLE RULE t_order(DATANODES('ds_${0..1}.t_order_${0..1}'))", createCapability("MySQL"));
175         assertTrue(actual.isRuleDistSQL());
176         assertThat(actual.getSideEffectScope(), is("rule-metadata"));
177     }
178     
179     @Test
180     void assertAnalyzePhysicalDDL() {
181         ClassificationResult actual = analyzer.analyze("CREATE TABLE orders(order_id BIGINT)", createCapability("MySQL"));
182         assertFalse(actual.isRuleDistSQL());
183         assertThat(actual.getSideEffectScope(), is("physical-structure"));
184     }
