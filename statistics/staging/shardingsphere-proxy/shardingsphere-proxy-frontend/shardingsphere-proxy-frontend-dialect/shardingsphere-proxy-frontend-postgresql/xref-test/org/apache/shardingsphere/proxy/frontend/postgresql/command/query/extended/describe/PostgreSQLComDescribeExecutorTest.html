386     @ParameterizedTest(name = "{0}")
387     @MethodSource("provideReturningCases")
388     void assertDescribePreparedStatementInsertWithReturning(final String testName, final String statementId, final String sql,
389                                                             final List<PostgreSQLBinaryColumnType> expectedParamTypes, final List<PostgreSQLColumnDescription> expectedColumns) throws SQLException {
390         when(packet.getType()).thenReturn('S');
391         when(packet.getName()).thenReturn(statementId);
392         SQLStatement sqlStatement = SQL_PARSER_ENGINE.parse(sql, false);
393         List<PostgreSQLBinaryColumnType> parameterTypes = new ArrayList<>(sqlStatement.getParameterCount());
394         for (int i = 0; i < sqlStatement.getParameterCount(); i++) {
395             parameterTypes.add(PostgreSQLBinaryColumnType.UNSPECIFIED);
396         }
397         SQLStatementContext sqlStatementContext = mock(InsertStatementContext.class);
398         when(sqlStatementContext.getSqlStatement()).thenReturn(sqlStatement);
399         ContextManager contextManager = mockContextManager();
400         when(ProxyContext.getInstance().getContextManager()).thenReturn(contextManager);
401         prepareJDBCBackendConnectionForResultMetaData(sql, expectedColumns.size());
402         if ("returning complex columns".equals(testName)) {
403             columnTypeOIDs = new HashMap<>(2, 1F);
404             columnTypeOIDs.put(12, 2249);
405             columnTypeOIDs.put(13, 2249);
406         }
407         List<Integer> parameterIndexes = IntStream.range(0, sqlStatement.getParameterCount()).boxed().collect(Collectors.toList());
408         connectionSession.getServerPreparedStatementRegistry().addPreparedStatement(
409                 statementId, new PostgreSQLServerPreparedStatement(sql, sqlStatementContext, new HintValueContext(), parameterTypes, parameterIndexes));
410         Collection<DatabasePacket> actualPackets = executor.execute();
411         Iterator<DatabasePacket> actualIterator = actualPackets.iterator();
412         PostgreSQLParameterDescriptionPacket parameterDescription = (PostgreSQLParameterDescriptionPacket) actualIterator.next();
413         PostgreSQLPacketPayload mockPayload = mock(PostgreSQLPacketPayload.class);
414         parameterDescription.write(mockPayload);
415         verify(mockPayload).writeInt2(expectedParamTypes.size());
416         Map<Integer, Long> expectedTypeCounts = expectedParamTypes.stream()
417                 .collect(Collectors.groupingBy(PostgreSQLBinaryColumnType::getValue, Collectors.counting()));
418         for (Entry<Integer, Long> entry : expectedTypeCounts.entrySet()) {
419             verify(mockPayload, times(entry.getValue().intValue())).writeInt4(entry.getKey());
420         }
421         PostgreSQLRowDescriptionPacket rowDescriptionPacket = (PostgreSQLRowDescriptionPacket) actualIterator.next();
422         List<PostgreSQLColumnDescription> actualColumnDescriptions = getColumnDescriptions(rowDescriptionPacket);
423         assertThat(actualColumnDescriptions.size(), is(expectedColumns.size()));
424         for (int i = 0; i < expectedColumns.size(); i++) {
