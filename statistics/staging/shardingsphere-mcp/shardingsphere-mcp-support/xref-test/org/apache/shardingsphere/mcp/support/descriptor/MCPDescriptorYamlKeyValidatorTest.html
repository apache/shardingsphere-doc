31      @Test
32      void assertValidate() {
33          String yamlContent = """
34                  resourceTemplates:
35                    - uriTemplate: shardingsphere://workflows/{plan_id}
36                      name: workflow-plan
37                      title: Workflow Plan
38                      description: Workflow plan.
39                      mimeType: application/json
40                      annotations:
41                        audience:
42                          - assistant
43                        priority: 0.5
44                        lastModified: "2026-05-13T00:00:00Z"
45                      shardingSphereMetadata:
46                        uriVariables:
47                          - name: plan_id
48                            title: Plan ID
49                            description: Plan ID.
50                            required: true
51                            scope: workflow-plan
52                        resourceKind: detail
53                  tools:
54                    - name: database_gateway_search_metadata
55                      title: Search Metadata
56                      description: Search metadata.
57                      inputSchema:
58                        type: object
59                        properties:
60                          query:
61                            type: string
62                            description: Query.
63                        required:
64                          - query
65                        additionalProperties: false
66                      annotations:
67                        title: Search Metadata
68                        readOnlyHint: true
69                        destructiveHint: false
70                        idempotentHint: true
71                        openWorldHint: true
72                      outputSchema:
73                        type: object
74                      runtime:
75                        workflowRole: plan
76                  prompts:
77                    - name: inspect_metadata
78                      title: Inspect Metadata
79                      description: Inspect metadata.
80                      binding:
81                        templateResource: META-INF/shardingsphere-mcp/prompts/inspect-metadata.md
82                      arguments:
83                        - name: database
84                          title: Database
85                          description: Database.
86                          required: false
87                  completionTargets:
88                    - referenceType: prompt
89                      reference: inspect_metadata
90                      arguments:
91                        - database
92                      maxValues: 50
93                  resourceNavigation:
94                    - from: database_gateway_apply_workflow

97                        - plan_id
98                      carriedArguments:

102         assertDoesNotThrow(() -> MCPDescriptorYamlKeyValidator.validate("test.yaml", toBytes(yamlContent)));

263     private void assertUnknownKey(final String yamlContent, final String expectedKeyPath) {
264         IllegalArgumentException actual = assertThrows(IllegalArgumentException.class,
265                 () -> MCPDescriptorYamlKeyValidator.validate("test.yaml", toBytes(yamlContent)));
266         assertThat(actual.getMessage(), is(String.format("MCP descriptor resource `test.yaml` contains unknown key `%s`.", expectedKeyPath)));
267     }
268     
269     private void assertValidationError(final String yamlContent, final String expectedMessage) {
270         IllegalArgumentException actual = assertThrows(IllegalArgumentException.class,
271                 () -> MCPDescriptorYamlKeyValidator.validate("test.yaml", toBytes(yamlContent)));
272         assertThat(actual.getMessage(), is(expectedMessage));
273     }
274     
275     private byte[] toBytes(final String yamlContent) {
276         return yamlContent.getBytes(StandardCharsets.UTF_8);
277     }
