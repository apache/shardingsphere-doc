61      @ParameterizedTest(name = "{0}")
62      @MethodSource("getParameterFrameArguments")
63      void assertInsertValueParameterFrames(final String name, final List<ParameterMarkerSegment> markers, final List<ExpressionSegment> values, final List<Object> params,
64                                            final List<List<Object>> expectedRows, final List<Integer> expectedOffsets) {
65          InsertStatement statement = InsertStatement.builder().values(Collections.singletonList(new InsertValuesSegment(30, 60, values))).build();
66          statement.addParameterMarkers(markers);
67          InsertStatementBaseContext baseContext = mock(InsertStatementBaseContext.class, RETURNS_DEEP_STUBS);
68          when(baseContext.getSqlStatement()).thenReturn(statement);
69          when(baseContext.getValueExpressions()).thenReturn(Collections.singletonList(values));
70          InsertStatementBindingContext actual = new InsertStatementBindingContext(baseContext, params, mock(ShardingSphereMetaData.class), "foo_db");
71          List<List<Object>> actualRows = actual.getInsertValueContexts().stream().map(InsertValueContext::getParameters).collect(Collectors.toList());
72          List<Integer> actualOffsets = actual.getInsertValueContexts().stream().map(InsertValueContext::getParametersOffset).collect(Collectors.toList());
73          assertThat(actualRows, is(expectedRows));
74          assertThat(actualOffsets, is(expectedOffsets));
75      }

77      private static Stream<Arguments> getParameterFrameArguments() {
78          ParameterMarkerExpressionSegment first = new ParameterMarkerExpressionSegment(40, 40, 0);
79          ParameterMarkerExpressionSegment second = new ParameterMarkerExpressionSegment(50, 50, 1);
80          ParameterMarkerExpressionSegment prefixedFirst = new ParameterMarkerExpressionSegment(40, 40, 2);
81          ParameterMarkerExpressionSegment prefixedSecond = new ParameterMarkerExpressionSegment(50, 50, 3);
82          ParameterMarkerExpressionSegment laterWhen = new ParameterMarkerExpressionSegment(10, 10, 2);
83          ParameterMarkerExpressionSegment laterElseValue = new ParameterMarkerExpressionSegment(40, 40, 4);
84          List<Object> row = Arrays.asList("foo_value", "bar_value");
85          List<Object> paramsWithPrefix = Arrays.asList("foo_cte", "bar_cte", "foo_value", "bar_value");
86          List<ParameterMarkerSegment> prefixMarkers = Arrays.asList(new ParameterMarkerExpressionSegment(0, 0, 0), new ParameterMarkerExpressionSegment(10, 10, 1));
87          List<ExpressionSegment> literalValues = Arrays.asList(new LiteralExpressionSegment(40, 40, "foo_value"), new LiteralExpressionSegment(50, 50, "bar_value"));
88          return Stream.of(
89                  Arguments.of("ordinary values", Arrays.asList(first, second), Arrays.asList(first, second), row, Collections.singletonList(row), Collections.singletonList(0)),
90                  Arguments.of("WITH prefix", Arrays.asList(prefixMarkers.get(0), prefixMarkers.get(1), prefixedFirst, prefixedSecond), Arrays.asList(prefixedFirst, prefixedSecond),
91                          paramsWithPrefix, Collections.singletonList(row), Collections.singletonList(2)),
92                  Arguments.of("child global indexes", Arrays.asList(prefixedFirst, prefixedSecond), Arrays.asList(prefixedFirst, prefixedSecond),
93                          paramsWithPrefix, Collections.singletonList(row), Collections.singletonList(2)),
94                  Arguments.of("conditional ELSE marker gap", Arrays.asList(prefixMarkers.get(0), prefixedFirst), Collections.singletonList(prefixedFirst),
95                          Arrays.asList("foo_when", "foo_first", "foo_else"), Collections.singletonList(Collections.singletonList("foo_else")), Collections.singletonList(2)),
96                  Arguments.of("later WHEN marker gap", Arrays.asList(prefixMarkers.get(0), prefixedSecond), Collections.singletonList(prefixedSecond),
97                          Arrays.asList("foo_when", "foo_first", "bar_when", "foo_second"), Collections.singletonList(Collections.singletonList("foo_second")), Collections.singletonList(3)),
98                  Arguments.of("later ELSE marker gap", Arrays.asList(laterWhen, laterElseValue), Collections.singletonList(laterElseValue),
99                          Arrays.asList("foo_when", "foo_first", "bar_when", "foo_second", "foo_else"), Collections.singletonList(Collections.singletonList("foo_else")),
100                         Collections.singletonList(4)),
101                 Arguments.of("literal values after WITH", prefixMarkers, literalValues, Arrays.asList("foo_cte", "bar_cte"),
102                         Collections.singletonList(Collections.emptyList()), Collections.singletonList(2)),
103                 Arguments.of("literal values", Collections.emptyList(), literalValues, Collections.emptyList(),
104                         Collections.singletonList(Collections.emptyList()), Collections.singletonList(0)));
105     }

107     @Test
108     void assertInsertSelectWithClausePreservesAllValuesRows() {
109         LiteralExpressionSegment first = new LiteralExpressionSegment(30, 30, 1);
110         SelectStatement nested = SelectStatement.builder().projections(new ProjectionsSegment(44, 44)).build();
111         SubqueryExpressionSegment second = new SubqueryExpressionSegment(new SubquerySegment(36, 45, nested, "SELECT 2"));
112         List<InsertValuesSegment> rows = Arrays.asList(new InsertValuesSegment(29, 31, Collections.singletonList(first)), new InsertValuesSegment(35, 46, Collections.singletonList(second)));
113         SelectStatement source = SelectStatement.builder().projections(new ProjectionsSegment(29, 46)).values(rows).build();
114         WithSegment with = new WithSegment(0, 18, Collections.emptyList());
115         InsertStatement statement = InsertStatement.builder().table(new SimpleTableSegment(new TableNameSegment(20, 22, new IdentifierValue("foo"))))
116                 .with(with).insertSelect(new SubquerySegment(29, 46, source, "VALUES (1), ((SELECT 2))")).build();
117         InsertStatementBaseContext base = mock(InsertStatementBaseContext.class, RETURNS_DEEP_STUBS);
118         when(base.getSqlStatement()).thenReturn(statement);
119         try (
120                 MockedConstruction<SelectStatementContext> ignored = mockConstruction(SelectStatementContext.class,
121                         (context, construction) -> when(context.getSqlStatement()).thenReturn((SelectStatement) construction.arguments().get(0)))) {
122             InsertStatementBindingContext actual = new InsertStatementBindingContext(base, Collections.emptyList(), mock(ShardingSphereMetaData.class), "foo_db");
123             SelectStatement actualSelect = actual.getInsertSelectContext().getSelectStatementContext().getSqlStatement();
124             assertThat(actualSelect, not(sameInstance(source)));
125             assertThat(actualSelect.getWith().get(), sameInstance(with));
126             assertThat(actualSelect.getValues().size(), is(2));
127             assertThat(actualSelect.getValues(), contains(rows.get(0), rows.get(1)));
128             assertThat(source.getValues().size(), is(2));
129             assertFalse(source.getWith().isPresent());
130         }
131     }

133     @Test
134     void assertInsertSelectWithOuterAndInnerWithClauses() {
135         ParameterMarkerExpressionSegment outerParameter = new ParameterMarkerExpressionSegment(15, 15, 0);
136         ParameterMarkerExpressionSegment innerParameter = new ParameterMarkerExpressionSegment(45, 45, 1);
137         WithSegment outer = new WithSegment(0, 26, Collections.emptyList());
138         WithSegment inner = new WithSegment(38, 60, Collections.emptyList());
139         SelectStatement source = SelectStatement.builder().projections(new ProjectionsSegment(70, 75)).with(inner).build();
140         source.addParameterMarkers(Collections.singletonList(innerParameter));
141         InsertStatement statement = InsertStatement.builder().table(new SimpleTableSegment(new TableNameSegment(27, 29, new IdentifierValue("foo"))))
142                 .with(outer).insertSelect(new SubquerySegment(38, 75, source, "WITH foo_inner AS (SELECT ?) SELECT 1")).build();
143         statement.addParameterMarkers(Arrays.asList(outerParameter, innerParameter));
144         InsertStatementBaseContext base = mock(InsertStatementBaseContext.class, RETURNS_DEEP_STUBS);
145         when(base.getSqlStatement()).thenReturn(statement);
146         try (
147                 MockedConstruction<SelectStatementContext> ignored = mockConstruction(SelectStatementContext.class,
148                         (context, construction) -> when(context.getSqlStatement()).thenReturn((SelectStatement) construction.arguments().get(0)))) {
149             InsertStatementBindingContext actual = new InsertStatementBindingContext(base, Arrays.asList("foo_outer", "bar_inner"), mock(ShardingSphereMetaData.class), "foo_db");
150             SelectStatement actualSelect = actual.getInsertSelectContext().getSelectStatementContext().getSqlStatement();
151             assertThat(actualSelect.getWith().orElse(null), sameInstance(inner));
152             assertThat(actualSelect.getParameterMarkers(), contains(outerParameter, innerParameter));
153             assertThat(actual.getInsertSelectContext().getParameters(), is(Arrays.asList("foo_outer", "bar_inner")));
154         }
