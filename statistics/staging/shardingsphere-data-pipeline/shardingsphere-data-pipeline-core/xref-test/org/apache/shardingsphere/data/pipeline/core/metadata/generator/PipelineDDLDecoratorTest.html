76      @Test
77      void assertDecorateUsesSchemaSpecificCandidatesForSameNamedTables() {
78          String logicIndexName = "very_long_named_index_boundary_case_for_sharding_length_safety_validation";
79          String actualIndexName = IndexMetaDataUtils.getActualIndexName(logicIndexName, "t_order_0", databaseType);
80          assertTrue(actualIndexName.matches(".*_t[0-9a-z]{8}$"));
81          String actualSQL = String.format("CREATE INDEX %s ON t_order_0 (order_id)", actualIndexName);
82          String expected = String.format("CREATE INDEX %s ON t_order (order_id)", logicIndexName);
83          assertThat(
84                  new PipelineDDLDecorator(mockMetaData(logicIndexName)).decorate(databaseType, "foo_db", "schema_b", "t_order", createParserEngine(actualSQL, actualIndexName), actualSQL).orElse(""),
85                  is(expected));
86      }

88      private ShardingSphereMetaData mockMetaData(final String logicIndexName) {
89          ShardingSphereMetaData result = mock(ShardingSphereMetaData.class);
90          ShardingSphereDatabase database = mock(ShardingSphereDatabase.class);
91          ShardingSphereSchema schemaA = new ShardingSphereSchema("schema_a", databaseType, Collections.singleton(
92                  new ShardingSphereTable("t_order", Collections.emptyList(),
93                          Collections.singleton(new ShardingSphereIndex("another_very_long_named_index_for_schema_a_validation", Collections.emptyList(), false)), Collections.emptyList())),
94                  Collections.emptyList());
95          ShardingSphereSchema schemaB = new ShardingSphereSchema("schema_b", databaseType, Collections.singleton(
96                  new ShardingSphereTable("t_order", Collections.emptyList(),
97                          Collections.singleton(new ShardingSphereIndex(logicIndexName, Collections.emptyList(), false)), Collections.emptyList())),
98                  Collections.emptyList());
99          when(result.getDatabase("foo_db")).thenReturn(database);
100         when(database.containsSchema("schema_b")).thenReturn(true);
101         when(database.getSchema("schema_b")).thenReturn(schemaB);
102         when(database.getAllSchemas()).thenReturn(Arrays.asList(schemaA, schemaB));
103         return result;
104     }

106     private SQLParserEngine createParserEngine(final String actualSQL, final String actualIndexName) {
107         SQLParserEngine result = mock(SQLParserEngine.class);
108         when(result.parse(actualSQL, true)).thenReturn(CreateIndexStatement.builder()
109                 .databaseType(databaseType)
110                 .index(new IndexSegment(getIndexStartIndex(), getIndexStopIndex(actualIndexName),
111                         new IndexNameSegment(getIndexStartIndex(), getIndexStopIndex(actualIndexName), new IdentifierValue(actualIndexName))))
112                 .table(new SimpleTableSegment(new TableNameSegment(getTableStartIndex(actualSQL), getTableStopIndex(actualSQL), new IdentifierValue("t_order_0"))))
113                 .build());
114         return result;
115     }
116     
117     private int getIndexStartIndex() {
118         return "CREATE INDEX ".length();
119     }
120     
121     private int getIndexStopIndex(final String actualIndexName) {
122         return getIndexStartIndex() + actualIndexName.length() - 1;
123     }
124     
125     private int getTableStartIndex(final String actualSQL) {
126         return actualSQL.indexOf("t_order_0");
127     }
128     
129     private int getTableStopIndex(final String actualSQL) {
130         return getTableStartIndex(actualSQL) + "t_order_0".length() - 1;
131     }

133     private SimpleTableSegment createQualifiedTable(final String sql) {
134         int tableStartIndex = sql.indexOf("\"UPPER_TABLE\"");
135         SimpleTableSegment result = new SimpleTableSegment(new TableNameSegment(tableStartIndex, tableStartIndex + "\"UPPER_TABLE\"".length() - 1, new IdentifierValue("\"UPPER_TABLE\"")));
136         int schemaStartIndex = sql.indexOf("\"UPPER_SCHEMA\"");
137         result.setOwner(new OwnerSegment(schemaStartIndex, schemaStartIndex + "\"UPPER_SCHEMA\"".length() - 1, new IdentifierValue("\"UPPER_SCHEMA\"")));
138         return result;
139     }

167         private CreateTableStatement createTableStatement(final String sql, final String columnName) {
168             int primaryKeyStartIndex = sql.lastIndexOf('"' + columnName + '"');
169             ConstraintDefinitionSegment constraint = mock(ConstraintDefinitionSegment.class);
170             when(constraint.getPrimaryKeyColumns()).thenReturn(Collections.singleton(
171                     new ColumnSegment(primaryKeyStartIndex, primaryKeyStartIndex + columnName.length() + 1, new IdentifierValue('"' + columnName + '"'))));
172             return CreateTableStatement.builder().databaseType(databaseType).table(createQualifiedTable(sql))
173                     .columnDefinitions(Collections.singleton(mockColumnDefinition(sql, columnName)))
174                     .constraintDefinitions(Collections.singleton(constraint)).build();
175         }

181             return result;

236                 columnReferences.add(new ColumnSegment(columnStartIndex, columnStartIndex + "\"ORDER_ID\"".length() - 1, new IdentifierValue("\"ORDER_ID\"")));

238             }
239             ConstraintDefinitionSegment constraintDefinition = mock(ConstraintDefinitionSegment.class);
240             when(constraintDefinition.getColumnReferences()).thenReturn(columnReferences);
241             CreateTableStatement sqlStatement = CreateTableStatement.builder().databaseType(databaseType).table(createQualifiedTable(actualSQL))
242                     .columnDefinitions(Collections.singleton(mockColumnDefinition(actualSQL, "ORDER_ID"))).constraintDefinitions(Collections.singleton(constraintDefinition)).build();
243             assertThat(new PipelineDDLDecorator(mockTargetMetaData(new DatabaseIdentifierContext(IdentifierCasePolicyResolver.resolveProtocol(databaseType))))
244                     .decorate(databaseType, "foo_db", "upper_schema", "t_order", mockParserEngine(actualSQL, sqlStatement), actualSQL).orElse(""),
245                     is("CREATE TABLE \"upper_schema\".t_order (\"order_id\" bigint, " + expectedConstraint + ")"));
246         }

355         @CsvSource(delimiter = '|', quoteCharacter = '~', value = {
356                 "\"ORDER_ID\" integer, \"DERIVED\" integer GENERATED ALWAYS AS (\"ORDER_ID\" + 1) STORED)|\"order_id\" integer, \"derive"
357                         + "d\" integer GENERATED ALWAYS AS (\"order_id\" + 1) STORED)|ORDER_ID,DERIVED,ORDER_ID",
358                 "\"ORDER_ID\" integer) PARTITION BY RANGE (\"ORDER_ID\")|\"order_id\" integer) PARTITION BY RANGE (\"order_id\")|ORDER_"
359                         + "ID,ORDER_ID",
360                 "\"STATUS\" text DEFAULT '\"ORDER_ID\"')|\"status\" text DEFAULT '\"ORDER_ID\"')|STATUS"})
361         void assertDecorateTableReferences(final String tableDefinition, final String expectedDefinition, final String referenceNames) {
362             String actualSQL = "CREATE TABLE \"UPPER_SCHEMA\".\"UPPER_TABLE\" (" + tableDefinition;
363             CreateTableStatement sqlStatement = CreateTableStatement.builder().databaseType(databaseType).table(createQualifiedTable(actualSQL)).build();
364             int columnStartIndex = 0;
365             for (String each : referenceNames.split(",")) {
366                 columnStartIndex = actualSQL.indexOf('"' + each + '"', columnStartIndex);
367                 sqlStatement.getColumnReferences().add(new ColumnSegment(columnStartIndex, columnStartIndex + each.length() + 1, new IdentifierValue('"' + each + '"')));

407         @ParameterizedTest(name = "{0}.{1}")
408         @CsvSource(delimiter = '|', quoteCharacter = '~', value = {

413             String actualSQL = "CREATE TABLE \"UPPER_SCHEMA\".\"UPPER_TABLE\" (\"ORDER_ID\" integer, FOREIGN KEY (\"ORDER_ID\") REFERENCES \"" + referencedSchemaName

415             int tableStartIndex = actualSQL.lastIndexOf('"' + referencedTableName + '"');

417                     new IdentifierValue('"' + referencedTableName + '"')));
418             int schemaStartIndex = actualSQL.indexOf('"' + referencedSchemaName + '"', actualSQL.indexOf("REFERENCES"));
419             referencedTable.setOwner(new OwnerSegment(schemaStartIndex, schemaStartIndex + referencedSchemaName.length() + 1, new IdentifierValue('"' + referencedSchemaName + '"')));
420             int localColumnStartIndex = actualSQL.indexOf("\"ORDER_ID\"", actualSQL.indexOf("FOREIGN KEY"));

426             when(constraint.getReferencedColumns()).thenReturn(Collections.singleton(new ColumnSegment(referencedColumnStartIndex,

429             int columnStartIndex = actualSQL.indexOf("\"ORDER_ID\"");
430             when(columnDefinition.getColumnName()).thenReturn(new ColumnSegment(columnStartIndex, columnStartIndex + "\"ORDER_ID\"".length() - 1, new IdentifierValue("\"ORDER_ID\"")));
431             CreateTableStatement sqlStatement = CreateTableStatement.builder().databaseType(databaseType).table(createQualifiedTable(actualSQL))
432                     .columnDefinitions(Collections.singleton(columnDefinition)).constraintDefinitions(Collections.singleton(constraint)).build();
433             assertThat(new PipelineDDLDecorator(mockTargetMetaData(new DatabaseIdentifierContext(IdentifierCasePolicyResolver.resolveProtocol(databaseType))))
434                     .decorate(databaseType, "foo_db", "target_schema", "t_order", mockParserEngine(actualSQL, sqlStatement), actualSQL).orElse(""),
435                     is("CREATE TABLE \"target_schema\".t_order (\"order_id\" integer, FOREIGN KEY (\"order_id\") REFERENCES " + expectedReference + ")"));
436         }
