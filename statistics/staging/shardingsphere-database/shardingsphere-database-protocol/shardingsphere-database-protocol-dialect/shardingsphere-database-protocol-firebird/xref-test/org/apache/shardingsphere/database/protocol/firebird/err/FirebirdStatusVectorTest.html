18  package org.apache.shardingsphere.database.protocol.firebird.err;
19  
20  import org.apache.shardingsphere.database.protocol.firebird.payload.FirebirdPacketPayload;
21  import org.firebirdsql.gds.ISCConstants;
22  import org.junit.jupiter.api.Test;
23  import org.junit.jupiter.api.extension.ExtendWith;
24  import org.mockito.InOrder;
25  import org.mockito.Mock;
26  import org.mockito.junit.jupiter.MockitoExtension;
27  
28  import java.sql.SQLException;
29  
30  import static org.hamcrest.MatcherAssert.assertThat;
31  import static org.hamcrest.Matchers.is;
32  import static org.mockito.Mockito.inOrder;
33  import static org.mockito.Mockito.verifyNoMoreInteractions;
34  
35  @ExtendWith(MockitoExtension.class)
36  class FirebirdStatusVectorTest {
37      
38      @Mock
39      private FirebirdPacketPayload payload;
40      
41      @Test
42      void assertWriteKeepsNativeCodeAndStripsStateSuffix() {
43          FirebirdStatusVector vector = new FirebirdStatusVector(new SQLException("prefix; detail [SQLState:42000]", "42000", ISCConstants.isc_arith_except));
44          assertThat(vector.getGdsCode(), is(ISCConstants.isc_arith_except));
45          assertThat(vector.getErrorMessage(), is("prefix; detail"));
46          vector.write(payload);
47          InOrder inOrder = inOrder(payload);
48          inOrder.verify(payload).writeInt4(ISCConstants.isc_arg_gds);
49          inOrder.verify(payload).writeInt4(ISCConstants.isc_arith_except);
50          inOrder.verify(payload).writeInt4(ISCConstants.isc_arg_string);
51          inOrder.verify(payload).writeString("prefix; detail");
52          inOrder.verify(payload).writeInt4(ISCConstants.isc_arg_end);
53          verifyNoMoreInteractions(payload);
54      }
55      
56      @Test
57      void assertWriteRebuildsDuplicateKeyVectorAsSegments() {
58          String message = "violation of PRIMARY or UNIQUE KEY constraint \"INTEG_2\" on table \"MY_TABLE\"; Problematic key value is (\"COL1\" = 1) [SQLState:23000, ISC error code:335544665]";
59          FirebirdStatusVector vector = new FirebirdStatusVector(new SQLException(message, "23000", ISCConstants.isc_unique_key_violation));
60          assertThat(vector.getGdsCode(), is(ISCConstants.isc_unique_key_violation));
61          vector.write(payload);
62          InOrder inOrder = inOrder(payload);
63          inOrder.verify(payload).writeInt4(ISCConstants.isc_arg_gds);
64          inOrder.verify(payload).writeInt4(ISCConstants.isc_unique_key_violation);
65          inOrder.verify(payload).writeInt4(ISCConstants.isc_arg_string);
66          inOrder.verify(payload).writeString("INTEG_2");
67          inOrder.verify(payload).writeInt4(ISCConstants.isc_arg_string);
68          inOrder.verify(payload).writeString("MY_TABLE");
69          inOrder.verify(payload).writeInt4(ISCConstants.isc_arg_gds);
70          inOrder.verify(payload).writeInt4(ISCConstants.isc_random);
71          inOrder.verify(payload).writeInt4(ISCConstants.isc_arg_string);
72          inOrder.verify(payload).writeString("Problematic key value is (\"COL1\" = 1)");
73          inOrder.verify(payload).writeInt4(ISCConstants.isc_arg_end);
74          verifyNoMoreInteractions(payload);
75      }
76      
77      @Test
78      void assertWriteOmitsArgumentWhenMessageIsEmpty() {
79          FirebirdStatusVector vector = new FirebirdStatusVector(new SQLException("", "28000", ISCConstants.isc_login));
80          vector.write(payload);
81          InOrder inOrder = inOrder(payload);
82          inOrder.verify(payload).writeInt4(ISCConstants.isc_arg_gds);
83          inOrder.verify(payload).writeInt4(ISCConstants.isc_login);
84          inOrder.verify(payload).writeInt4(ISCConstants.isc_arg_end);
85          verifyNoMoreInteractions(payload);
86      }
87      
88      @Test
89      void assertWriteWithNullMessage() {
90          FirebirdStatusVector vector = new FirebirdStatusVector(new SQLException(null, "28000", ISCConstants.isc_login));
91          assertThat(vector.getErrorMessage(), is(""));
92          vector.write(payload);
93          InOrder inOrder = inOrder(payload);
94          inOrder.verify(payload).writeInt4(ISCConstants.isc_arg_gds);
95          inOrder.verify(payload).writeInt4(ISCConstants.isc_login);
96          inOrder.verify(payload).writeInt4(ISCConstants.isc_arg_end);
97          verifyNoMoreInteractions(payload);
98      }
99      
100     @Test
101     void assertWriteUsesRandomCodeWhenErrorCodeIsLowerThanArithExcept() {
102         FirebirdStatusVector vector = new FirebirdStatusVector(new SQLException("plain", "00000", ISCConstants.isc_arith_except - 1));
103         assertThat(vector.getGdsCode(), is(ISCConstants.isc_random));
104         vector.write(payload);
105         InOrder inOrder = inOrder(payload);
106         inOrder.verify(payload).writeInt4(ISCConstants.isc_arg_gds);
107         inOrder.verify(payload).writeInt4(ISCConstants.isc_random);
108         inOrder.verify(payload).writeInt4(ISCConstants.isc_arg_string);
109         inOrder.verify(payload).writeString("plain");
110         inOrder.verify(payload).writeInt4(ISCConstants.isc_arg_end);
111         verifyNoMoreInteractions(payload);
112     }
113     
114     @Test
115     void assertWriteFailSafeUsesRandomCodeAndKeepsSqlStateForUnmatchedParameterizedTemplate() {
116         FirebirdStatusVector vector = new FirebirdStatusVector(new SQLException("totally unrelated text", "23000", ISCConstants.isc_unique_key_violation));
117         assertThat(vector.getGdsCode(), is(ISCConstants.isc_random));
118         vector.write(payload);
119         InOrder inOrder = inOrder(payload);
120         inOrder.verify(payload).writeInt4(ISCConstants.isc_arg_gds);
121         inOrder.verify(payload).writeInt4(ISCConstants.isc_random);
122         inOrder.verify(payload).writeInt4(ISCConstants.isc_arg_string);
123         inOrder.verify(payload).writeString("totally unrelated text");
124         inOrder.verify(payload).writeInt4(ISCConstants.isc_arg_sql_state);
125         inOrder.verify(payload).writeString("23000");
126         inOrder.verify(payload).writeInt4(ISCConstants.isc_arg_end);
127         verifyNoMoreInteractions(payload);
128     }
129 }
