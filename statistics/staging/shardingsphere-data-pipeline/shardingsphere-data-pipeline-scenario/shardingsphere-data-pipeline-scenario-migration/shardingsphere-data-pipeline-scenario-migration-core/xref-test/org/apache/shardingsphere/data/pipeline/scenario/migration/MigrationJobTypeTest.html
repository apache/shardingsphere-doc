45  class MigrationJobTypeTest {
46      
47      @SuppressWarnings("rawtypes")
48      private final PipelineJobType jobType = TypedSPILoader.getService(PipelineJobType.class, "MIGRATION");
49      
50      @Test
51      void assertGetOption() {
52          PipelineJobOption actual = jobType.getOption();
53          assertThat(actual.getCode(), is("01"));
54          assertTrue(actual.isTransmissionJob());
55          assertThat(actual.getYamlJobConfigurationSwapper(), isA(YamlMigrationJobConfigurationSwapper.class));
56          assertThat(actual.getYamlJobItemProgressSwapper(), isA(YamlTransmissionJobItemProgressSwapper.class));
57          assertThat(actual.getJobClass(), is(MigrationJob.class));
58          assertFalse(actual.isIgnoreToStartDisabledJobWhenJobItemProgressIsFinished());
59          assertThat(actual.getToBeStartDisabledNextJobType(), is("CONSISTENCY_CHECK"));
60          assertThat(actual.getToBeStoppedPreviousJobType(), is("CONSISTENCY_CHECK"));
61          assertFalse(actual.isForceNoShardingWhenConvertToJobConfigurationPOJO());
62      }
63      
64      @SuppressWarnings("unchecked")
65      @Test
66      void assertGetJobTarget() {
67          JobDataNodeEntry dataNodeEntry1 = new JobDataNodeEntry("foo_tbl", Arrays.asList(new DataNode("db.foo_tbl_0"), new DataNode("db.foo_tbl_1")));
68          JobDataNodeEntry dataNodeEntry2 = new JobDataNodeEntry("bar_tbl", Arrays.asList(new DataNode("db.bar_tbl_0"), new DataNode("db.bar_tbl_1")));
69          MigrationJobConfiguration jobConfig = mock(MigrationJobConfiguration.class);
70          when(jobConfig.getJobShardingDataNodes()).thenReturn(Collections.singletonList(new JobDataNodeLine(Arrays.asList(dataNodeEntry1, dataNodeEntry2))));
71          PipelineJobTarget actual = jobType.getJobTarget(jobConfig);
72          assertNull(actual.getDatabaseName());
73          assertThat(actual.getTableName(), is("db.foo_tbl_0,db.foo_tbl_1,db.bar_tbl_0,db.bar_tbl_1"));
74      }
75      
76      @SuppressWarnings("unchecked")
77      @Test
78      void assertBuildDataConsistencyChecker() {
79          assertThat(jobType.buildDataConsistencyChecker(mock(MigrationJobConfiguration.class), mock(), mock()), isA(PipelineDataConsistencyChecker.class));
80      }
81  }
