18  package org.apache.shardingsphere.mcp.support.workflow.service;
19  
20  import org.apache.shardingsphere.mcp.support.workflow.model.WorkflowContextSnapshot;
21  import org.apache.shardingsphere.mcp.support.workflow.model.WorkflowKind;
22  import org.apache.shardingsphere.mcp.support.workflow.model.WorkflowRequest;
23  import org.junit.jupiter.api.Test;
24  
25  import java.util.List;
26  import java.util.Map;
27  
28  import static org.hamcrest.MatcherAssert.assertThat;
29  import static org.hamcrest.Matchers.is;
30  
31  class WorkflowGuidanceResourceHintProviderTest {
32      
33      @Test
34      void assertCreateResourcesToReadWithDescriptorResources() {
35          List<Map<String, Object>> actual = new WorkflowGuidanceResourceHintProvider().createResourcesToRead(createSnapshot("encrypt.rule", "logic_db", "", "t_order"));
36          assertThat(actual, is(List.of(Map.of(
37                  "uri", "shardingsphere://workflow/test-resource",
38                  "resource_kind", "detail",
39                  "purpose", "read_first",
40                  "reason", "Workflow descriptor resource used by descriptor catalog loader tests.",
41                  "source_field", "resources_to_read"))));
42      }
43      
44      @Test
45      void assertCreateResourcesToReadWithSnapshotResources() {
46          WorkflowContextSnapshot snapshot = createSnapshot("test.workflow", "logic_db", "", "t_order");
47          snapshot.getResourceUriTemplates().add("shardingsphere://workflow/test-resource");
48          List<Map<String, Object>> actual = new WorkflowGuidanceResourceHintProvider().createResourcesToRead(snapshot);
49          assertThat(actual.getFirst().get("uri"), is("shardingsphere://workflow/test-resource"));
50      }
51      
52      private WorkflowContextSnapshot createSnapshot(final String workflowKind, final String database, final String schema, final String table) {
53          WorkflowContextSnapshot result = new WorkflowContextSnapshot();
54          result.setWorkflowKind(WorkflowKind.valueOf(workflowKind));
55          WorkflowRequest request = new WorkflowRequest();
56          request.setDatabase(database);
57          request.setSchema(schema);
58          request.setTable(table);
59          result.setRequest(request);
60          return result;
61      }
62  }
