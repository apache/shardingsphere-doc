51  class DriverExecuteQueryExecutorTest {
52      
53      @Test
54      void assertExecuteQueryWithFederation() throws SQLException {
55          ShardingSphereMetaData metaData = mock(ShardingSphereMetaData.class, RETURNS_DEEP_STUBS);
56          SQLFederationEngine sqlFederationEngine = mock(SQLFederationEngine.class);
57          QueryContext queryContext = mock(QueryContext.class, RETURNS_DEEP_STUBS);
58          when(sqlFederationEngine.decide(queryContext, metaData.getGlobalRuleMetaData())).thenReturn(true);
59          ResultSet expected = mock(ResultSet.class);
60          when(sqlFederationEngine.executeQuery(any(), any(), any())).thenReturn(expected);
61          JDBCExecutor jdbcExecutor = mock(JDBCExecutor.class);
62          RawExecutor rawExecutor = mock(RawExecutor.class);
63          DriverExecuteQueryExecutor executor = new DriverExecuteQueryExecutor(mock(ShardingSphereConnection.class, RETURNS_DEEP_STUBS), metaData, jdbcExecutor, rawExecutor, sqlFederationEngine);
64          ResultSet actual = executor.executeQuery(mock(ShardingSphereDatabase.class), queryContext, mock(DriverExecutionPrepareEngine.class), mock(Statement.class), Collections.emptyMap(),
65                  mock(StatementAddCallback.class), mock(StatementReplayCallback.class));
66          assertThat(actual, sameInstance(expected));
67          verifyNoInteractions(jdbcExecutor, rawExecutor);
68      }
69      
70      @Test
71      void assertExecuteQueryWithJDBCPushDown() throws SQLException {
72          ShardingSphereDatabase database = mock(ShardingSphereDatabase.class, RETURNS_DEEP_STUBS);
73          when(database.getRuleMetaData().getAttributes(RawExecutionRuleAttribute.class)).thenReturn(Collections.emptyList());
74          QueryContext queryContext = mock(QueryContext.class, RETURNS_DEEP_STUBS);
75          Statement statement = mock(Statement.class);
76          DriverExecutionPrepareEngine prepareEngine = mock(DriverExecutionPrepareEngine.class);
77          StatementAddCallback addCallback = mock(StatementAddCallback.class);
78          StatementReplayCallback replayCallback = mock(StatementReplayCallback.class);
79          ResultSet expected = mock(ResultSet.class);
80          try (
81                  MockedConstruction<DriverJDBCPushDownExecuteQueryExecutor> mockedExecutors = mockConstruction(DriverJDBCPushDownExecuteQueryExecutor.class,
82                          (mock, context) -> when(mock.executeQuery(database, queryContext, prepareEngine, statement, Collections.emptyMap(), addCallback, replayCallback)).thenReturn(expected))) {
83              DriverExecuteQueryExecutor executor = new DriverExecuteQueryExecutor(mock(ShardingSphereConnection.class, RETURNS_DEEP_STUBS), mock(ShardingSphereMetaData.class, RETURNS_DEEP_STUBS),
84                      mock(JDBCExecutor.class), mock(RawExecutor.class), mock(SQLFederationEngine.class));
85              ResultSet actual = executor.executeQuery(database, queryContext, prepareEngine, statement, Collections.emptyMap(), addCallback, replayCallback);
86              assertThat(actual, sameInstance(expected));
87              verify(mockedExecutors.constructed().iterator().next()).executeQuery(database, queryContext, prepareEngine, statement, Collections.emptyMap(), addCallback, replayCallback);
88          }
89      }
90      
91      @Test
92      void assertExecuteQueryWithRawPushDown() throws SQLException {
93          ShardingSphereDatabase database = mock(ShardingSphereDatabase.class, RETURNS_DEEP_STUBS);
94          when(database.getRuleMetaData().getAttributes(RawExecutionRuleAttribute.class)).thenReturn(Collections.singleton(mock(RawExecutionRuleAttribute.class)));
95          QueryContext queryContext = mock(QueryContext.class, RETURNS_DEEP_STUBS);
96          Statement statement = mock(Statement.class);
97          ResultSet expected = mock(ResultSet.class);
98          try (
99                  MockedConstruction<DriverRawPushDownExecuteQueryExecutor> mockedExecutors = mockConstruction(DriverRawPushDownExecuteQueryExecutor.class,
100                         (mock, context) -> when(mock.executeQuery(database, queryContext, statement, Collections.emptyMap())).thenReturn(expected))) {
101             DriverExecuteQueryExecutor executor = new DriverExecuteQueryExecutor(mock(ShardingSphereConnection.class, RETURNS_DEEP_STUBS), mock(ShardingSphereMetaData.class, RETURNS_DEEP_STUBS),
102                     mock(JDBCExecutor.class), mock(RawExecutor.class), mock(SQLFederationEngine.class));
103             ResultSet actual = executor.executeQuery(database, queryContext, mock(DriverExecutionPrepareEngine.class), statement, Collections.emptyMap(), mock(StatementAddCallback.class),
104                     mock(StatementReplayCallback.class));
105             assertThat(actual, sameInstance(expected));
106             verify(mockedExecutors.constructed().iterator().next()).executeQuery(database, queryContext, statement, Collections.emptyMap());
107         }
