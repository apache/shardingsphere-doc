62      @Test
63      void assertReadUnknownDatabaseRecovery() throws IOException, InterruptedException {
64          try (MCPInteractionClient interactionClient = createOpenedInteractionClient()) {
65              Map<String, Object> actual = interactionClient.readResource("shardingsphere://databases/missing_db");
66              assertTrue(getPayloadItems(actual).isEmpty());
67              assertThat(getObjectOrEmpty(actual.get("empty_state")).get("category"), is("unknown_database"));
68              assertThat(getObjectOrEmpty(actual.get("recovery")).get("category"), is("unknown_database"));
69              Map<String, Object> nextAction = getRequiredObjectList(actual.get("next_actions")).getFirst();
70              assertThat(nextAction.get("type"), is("resource_read"));
71              assertThat(nextAction.get("resource_uri"), is("shardingsphere://databases"));
72          }
73      }

91      @Test
92      void assertReadTableDetail() throws IOException, InterruptedException {
93          try (MCPInteractionClient interactionClient = createOpenedInteractionClient()) {
94              List<Map<String, Object>> items = getPayloadItems(interactionClient.readResource(
95                      String.format("shardingsphere://databases/%s/schemas/%s/tables/orders", LOGICAL_DATABASE_NAME, LOGICAL_DATABASE_NAME)));
96              assertThat(items.size(), is(1));
97              Map<String, Object> actualItem = items.getFirst();
98              assertThat(String.valueOf(actualItem.get("table")), is("orders"));
99              assertThat(getNestedNames(actualItem, "columns", "column"), is(List.of("amount", "order_id", "status")));
100             assertThat(getNestedNames(actualItem, "indexes", "index"), containsInAnyOrder("PRIMARY", "idx_orders_status"));
101         }
102     }

105     void assertSearchMetadataTablesAndViews() throws IOException, InterruptedException {
106         try (MCPInteractionClient interactionClient = createOpenedInteractionClient()) {
107             List<Map<String, Object>> items = getPayloadItems(interactionClient.call("database_gateway_search_metadata",
108                     Map.of("database", LOGICAL_DATABASE_NAME, "schema", LOGICAL_DATABASE_NAME, "query", "order", "object_types", List.of("table", "view"))));
109             assertThat(items.stream().map(each -> String.valueOf(each.get("name"))).toList(), is(List.of("order_items", "orders", "active_orders")));

113     @Test
114     void assertValidateRuntimeDatabase() throws IOException, InterruptedException {
115         try (MCPInteractionClient interactionClient = createOpenedInteractionClient()) {
116             Map<String, Object> actual = interactionClient.call(
117                     "database_gateway_validate_runtime_database", Map.of("database", LOGICAL_DATABASE_NAME));
118             assertThat(actual.get("response_mode"), is("validation"));
119             assertThat(actual.get("status"), is("ready"));
120             assertThat(actual.get("database"), is(LOGICAL_DATABASE_NAME));
121             assertFalse(((List<?>) actual.get("checks")).isEmpty());
122             assertTrue(((List<?>) actual.get("checks")).stream().map(each -> (Map<?, ?>) each)
123                     .allMatch(each -> "passed".equals(each.get("status"))));
124         }
125     }

127     @Test
128     void assertSearchMetadataPagination() throws IOException, InterruptedException {
129         try (MCPInteractionClient interactionClient = createOpenedInteractionClient()) {
130             Map<String, Object> firstPage = interactionClient.call("database_gateway_search_metadata",
131                     Map.of("database", LOGICAL_DATABASE_NAME, "schema", LOGICAL_DATABASE_NAME, "query", "order",
132                             "object_types", List.of("table", "view"), "limit", 2, "offset", 0));
133             assertThat(getPayloadItems(firstPage).stream().map(each -> String.valueOf(each.get("name"))).toList(), is(List.of("order_items", "orders")));
134             assertTrue((Boolean) firstPage.get("has_more"));
135             assertThat(firstPage.get("next_offset"), is(2));
136             Map<String, Object> secondPage = interactionClient.call("database_gateway_search_metadata",
137                     Map.of("database", LOGICAL_DATABASE_NAME, "schema", LOGICAL_DATABASE_NAME, "query", "order",
138                             "object_types", List.of("table", "view"), "limit", 2, "offset", 2));
139             assertThat(getPayloadItems(secondPage).stream().map(each -> String.valueOf(each.get("name"))).toList(), is(List.of("active_orders")));
140             assertFalse((Boolean) secondPage.get("has_more"));
141             assertFalse(secondPage.containsKey("next_offset"));
142         }
143     }
