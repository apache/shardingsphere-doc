18  package org.apache.shardingsphere.database.connector.h2.jdbcurl;
19  
20  import org.apache.shardingsphere.database.connector.core.exception.UnrecognizedDatabaseURLException;
21  import org.apache.shardingsphere.database.connector.core.jdbcurl.parser.ConnectionProperties;
22  import org.apache.shardingsphere.database.connector.core.jdbcurl.parser.ConnectionPropertiesParser;
23  import org.junit.jupiter.params.ParameterizedTest;
24  import org.junit.jupiter.params.provider.Arguments;
25  import org.junit.jupiter.params.provider.MethodSource;
26  
27  import java.util.stream.Stream;
28  
29  import static org.hamcrest.CoreMatchers.is;
30  import static org.hamcrest.MatcherAssert.assertThat;
31  import static org.junit.jupiter.api.Assertions.assertThrows;
32  
33  class H2ConnectionPropertiesParserTest {
34      
35      private final ConnectionPropertiesParser parser = new H2ConnectionPropertiesParser();
36      
37      @ParameterizedTest(name = "{index}: {0}")
38      @MethodSource("provideArguments")
39      void assertParse(final String name, final String url, final String expectedHost, final int expectedPort, final String expectedCatalog, final String expectedModel, final boolean expectException) {
40          if (expectException) {
41              assertThrows(UnrecognizedDatabaseURLException.class, () -> parser.parse(url, null, null));
42          } else {
43              assertParseSuccess(url, expectedHost, expectedPort, expectedCatalog, expectedModel);
44          }
45      }
46      
47      private void assertParseSuccess(final String url, final String expectedHost, final int expectedPort, final String expectedCatalog, final String expectedModel) {
48          ConnectionProperties actual = parser.parse(url, null, null);
49          assertThat(actual.getHostname(), is(expectedHost));
50          assertThat(actual.getPort(), is(expectedPort));
51          assertThat(actual.getCatalog(), is(expectedCatalog));
52          assertThat(actual.getQueryProperties().getProperty("model"), is(expectedModel));
53      }
54      
55      private static Stream<Arguments> provideArguments() {
56          return Stream.of(
57                  Arguments.of("mem", "jdbc:h2:mem:ds_mem;MODE=MySQL", "", -1, "ds_mem", "mem", false),
58                  Arguments.of("tcp", "jdbc:h2:tcp://127.0.0.1:9092/~/demo/name;DB_CLOSE_DELAY=-1", "127.0.0.1", 9092, "name", "tcp:", false),
59                  Arguments.of("file", "jdbc:h2:file:/data/demo/db_file;MODE=MySQL", "", -1, "db_file", "file:", false),
60                  Arguments.of("invalid", "jdbc:invalid:h2", "", -1, "", "", true));
61      }
62  }
