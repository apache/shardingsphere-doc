28  import org.apache.shardingsphere.mcp.support.workflow.service.WorkflowPlanPayloadBuilder;

30  import org.apache.shardingsphere.mcp.support.workflow.service.WorkflowRequestBinder;
31  
32  import java.util.Map;
33  
34  /**
35   * Tool handler for shadow rule workflow planning.
36   */
37  public final class PlanShadowRuleToolHandler implements MCPToolHandler<MCPFeatureRequestContext> {
38      
39      private final ShadowWorkflowPlanningService planningService = new ShadowWorkflowPlanningService();
40      
41      @Override
42      public Class<MCPFeatureRequestContext> getContextType() {
43          return MCPFeatureRequestContext.class;
44      }
45      
46      @Override
47      public String getToolName() {
48          return ShadowFeatureDefinition.PLAN_RULE_TOOL_NAME;
49      }
50      
51      @Override
52      public MCPSuccessPayload handle(final MCPFeatureRequestContext requestContext, final Map<String, Object> arguments) {
53          ShadowRuleWorkflowRequest request = WorkflowRequestBinder.bindPlanningRequest(ShadowRuleWorkflowRequest::new, arguments, this::bindFeatureArguments);
54          WorkflowContextSnapshot snapshot = planningService.planRule(requestContext.getWorkflowSessionContext(), requestContext.getQueryFacade(), request);
55          return new MCPMapPayload(WorkflowPlanPayloadBuilder.buildWithArtifacts(snapshot, snapshot.getRequest()));
56      }
57      
58      private void bindFeatureArguments(final ShadowRuleWorkflowRequest request, final WorkflowPlanningArguments workflowPlanningArguments) {
59          workflowPlanningArguments.applyStringArgument(ShadowFeatureDefinition.RULE_FIELD, request::setRuleName);
60          workflowPlanningArguments.applyStringArgument(ShadowFeatureDefinition.SOURCE_STORAGE_UNIT_FIELD, request::setSourceStorageUnit);
61          workflowPlanningArguments.applyStringArgument(ShadowFeatureDefinition.SHADOW_STORAGE_UNIT_FIELD, request::setShadowStorageUnit);
62          workflowPlanningArguments.applyStringArgument(ShadowFeatureDefinition.TABLE_FIELD, request::setTableName);
63          workflowPlanningArguments.applyStringArgument(ShadowFeatureDefinition.ALGORITHM_TYPE_FIELD, request::setAlgorithmType);
64          request.putAlgorithmProperties(workflowPlanningArguments.getMapArgument(ShadowFeatureDefinition.ALGORITHM_PROPERTIES_FIELD));
65      }
66      
67  }
