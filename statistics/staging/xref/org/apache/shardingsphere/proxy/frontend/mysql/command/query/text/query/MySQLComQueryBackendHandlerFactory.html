52  /**
53   * Backend handler factory for MySQL COM_QUERY.
54   */
55  @NoArgsConstructor(access = AccessLevel.PRIVATE)
56  final class MySQLComQueryBackendHandlerFactory {
57      
58      private static final String BINARY_INTRODUCER = "_binary";
59      
60      private static final char MALFORMED_INPUT_REPLACEMENT = (char) 0xFFFD;
61      
62      static ProxyBackendHandler newInstance(final MySQLComQueryPacket packet, final ConnectionSession connectionSession) throws SQLException {
63          DatabaseType databaseType = TypedSPILoader.getService(DatabaseType.class, "MySQL");
64          String sql = packet.getSQL();
65          List<Object> binaryLiteralValues = Collections.emptyList();
66          SQLStatement sqlStatement;
67          Optional<ExtractionResult> extractionResult = tryExtractBinaryLiteral(sql, packet, connectionSession);
68          if (extractionResult.isPresent() && 1 == MultiSQLSplitter.split(extractionResult.get().getSql()).size()) {
69              String parameterizedSQL = SQLHintUtils.removeHint(extractionResult.get().getSql());
70              try {
71                  sqlStatement = ProxySQLComQueryParser.parse(parameterizedSQL, databaseType, connectionSession);
72                  sql = parameterizedSQL;
73                  binaryLiteralValues = extractionResult.get().getBinaryLiteralValues();
74              } catch (final DialectSQLParsingException ignored) {
75                  sqlStatement = ProxySQLComQueryParser.parse(sql, databaseType, connectionSession);
76              }
77          } else {
78              sqlStatement = ProxySQLComQueryParser.parse(sql, databaseType, connectionSession);
79          }
80          return binaryLiteralValues.isEmpty() && areMultiStatements(connectionSession, sqlStatement, sql)
81                  ? new MySQLMultiStatementsProxyBackendHandler(connectionSession, sqlStatement, sql)
82                  : createSingleStatementHandler(databaseType, sql, sqlStatement, binaryLiteralValues, packet, connectionSession);
83      }
84      
85      private static Optional<ExtractionResult> tryExtractBinaryLiteral(final String sql, final MySQLComQueryPacket packet, final ConnectionSession connectionSession) throws SQLException {
86          Optional<byte[]> originalSQLBytes = packet.findOriginalSQLBytes();
87          if (!originalSQLBytes.isPresent() || !requiresBinaryLiteralInspection(sql)) {
88              return Optional.empty();
89          }
90          ExtractionResult extractionResult = MySQLComQueryBinaryParameterExtractor.extract(
91                  originalSQLBytes.get(), connectionSession.getAttributeMap().attr(CommonConstants.CHARSET_ATTRIBUTE_KEY).get());
92          return extractionResult.getBinaryLiteralValues().isEmpty() ? Optional.empty() : Optional.of(extractionResult);
93      }
94      
95      private static ProxyBackendHandler createSingleStatementHandler(final DatabaseType databaseType, final String sql, final SQLStatement sqlStatement,
96                                                                      final List<Object> binaryLiteralValues, final MySQLComQueryPacket packet,
97                                                                      final ConnectionSession connectionSession) throws SQLException {
98          if (binaryLiteralValues.isEmpty()) {
99              return ProxyBackendHandlerFactory.newInstance(databaseType, sql, sqlStatement, connectionSession, packet.getHintValueContext());
100         }
101         ShardingSphereMetaData metaData = ProxyContext.getInstance().getContextManager().getMetaDataContexts().getMetaData();
102         SQLStatementContext sqlStatementContext = new SQLBindEngine(metaData, connectionSession.getCurrentDatabaseName(), packet.getHintValueContext()).bind(sqlStatement);
103         if (sqlStatementContext instanceof ParameterAware) {
104             ((ParameterAware) sqlStatementContext).bindParameters(binaryLiteralValues);
105         }
106         QueryContext queryContext = new QueryContext(sqlStatementContext, sql, binaryLiteralValues, packet.getHintValueContext(), connectionSession.getConnectionContext(), metaData, true);
107         return ProxyBackendHandlerFactory.newInstance(databaseType, queryContext, connectionSession, true);
108     }
109     
110     private static boolean requiresBinaryLiteralInspection(final String sql) {
111         for (int i = 0; i < sql.length(); i++) {
112             char current = sql.charAt(i);
113             if (MALFORMED_INPUT_REPLACEMENT == current || '_' == current && sql.regionMatches(true, i, BINARY_INTRODUCER, 0, BINARY_INTRODUCER.length())) {
114                 return true;
115             }
116         }
117         return false;
118     }
119     
120     private static boolean areMultiStatements(final ConnectionSession connectionSession, final SQLStatement sqlStatement, final String sql) {
121         return isMultiStatementsEnabled(connectionSession)
122                 && isSuitableMultiStatementsSQLStatement(sqlStatement) && MultiSQLSplitter.hasSameTypeMultiStatements(sqlStatement, MultiSQLSplitter.split(sql));
123     }
124     
125     private static boolean isMultiStatementsEnabled(final ConnectionSession connectionSession) {
126         return connectionSession.getAttributeMap().hasAttr(MySQLConstants.OPTION_MULTI_STATEMENTS_ATTRIBUTE_KEY)
127                 && MySQLComSetOptionPacket.MYSQL_OPTION_MULTI_STATEMENTS_ON == connectionSession.getAttributeMap().attr(MySQLConstants.OPTION_MULTI_STATEMENTS_ATTRIBUTE_KEY).get();
128     }
129     
130     private static boolean isSuitableMultiStatementsSQLStatement(final SQLStatement sqlStatement) {
131         return containsInsertOnDuplicateKey(sqlStatement) || sqlStatement instanceof UpdateStatement || sqlStatement instanceof DeleteStatement;
132     }
133     
134     private static boolean containsInsertOnDuplicateKey(final SQLStatement sqlStatement) {
135         return sqlStatement instanceof InsertStatement && ((InsertStatement) sqlStatement).getOnDuplicateKeyColumns().isPresent();
136     }
