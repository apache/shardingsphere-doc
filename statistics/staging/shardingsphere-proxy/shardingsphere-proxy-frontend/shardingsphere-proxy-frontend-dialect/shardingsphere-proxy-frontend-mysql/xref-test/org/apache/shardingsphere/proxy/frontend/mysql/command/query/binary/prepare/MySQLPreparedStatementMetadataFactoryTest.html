65  @ExtendWith(AutoMockExtension.class)
66  @StaticMockSettings(ProxyContext.class)
67  class MySQLPreparedStatementMetadataFactoryTest {
68      
69      @Mock
70      private ConnectionSession connectionSession;
71      
72      @Test
73      void assertLoad() throws SQLException {
74          PreparedStatement expected = mockPreparedStatement();
75          mockMetaDataWithDatabase();
76          mockDatabaseConnectionManager(Collections.singletonList(mockBackendConnection(expected, null)));
77          try (MockedConstruction<KernelProcessor> ignored = mockKernelProcessor(createExecutionContext(createExecutionUnits()))) {
78              assertThat(MySQLPreparedStatementMetadataFactory.load(connectionSession, createPreparedStatement()), is(expected));
79          }
80      }
81      
82      @Test
83      void assertLoadWithNoExecutionUnit() {
84          mockMetaData();
85          try (MockedConstruction<KernelProcessor> ignored = mockKernelProcessor(createExecutionContext(Collections.emptyList()))) {
86              ShardingSphereSQLException actual = assertThrows(PreparedStatementMetadataResolutionException.class,
87                      () -> MySQLPreparedStatementMetadataFactory.load(connectionSession, createPreparedStatement()));
88              assertThat(actual.getMessage(), is("Can not resolve prepared statement metadata because no execution unit was generated."));
89          }
90      }
91      
92      @Test
93      void assertLoadWithNoBackendConnection() throws SQLException {
94          mockMetaDataWithDatabase();
95          mockDatabaseConnectionManager(Collections.emptyList());
96          try (MockedConstruction<KernelProcessor> ignored = mockKernelProcessor(createExecutionContext(createExecutionUnits()))) {
97              ShardingSphereSQLException actual = assertThrows(PreparedStatementMetadataResolutionException.class,
98                      () -> MySQLPreparedStatementMetadataFactory.load(connectionSession, createPreparedStatement()));
99              assertThat(actual.getMessage(), is("Can not resolve prepared statement metadata because no backend connection was acquired."));
100         }
101     }
102     
103     @Test
104     void assertLoadWithSQLException() throws SQLException {
105         SQLException expected = new SQLException("expected");
106         mockMetaDataWithDatabase();
107         mockDatabaseConnectionManager(Collections.singletonList(mockBackendConnection(null, expected)));
108         try (MockedConstruction<KernelProcessor> ignored = mockKernelProcessor(createExecutionContext(createExecutionUnits()))) {
109             assertThat(assertThrows(SQLException.class, () -> MySQLPreparedStatementMetadataFactory.load(connectionSession, createPreparedStatement())), is(expected));
110         }
111     }
112     
113     private MySQLServerPreparedStatement createPreparedStatement() {
114         SQLStatementContext sqlStatementContext = mock(SQLStatementContext.class);
115         TablesContext tablesContext = mock(TablesContext.class);
116         when(tablesContext.getDatabaseNames()).thenReturn(Collections.singleton("foo_db"));
117         when(sqlStatementContext.getTablesContext()).thenReturn(tablesContext);
118         return new MySQLServerPreparedStatement("SELECT 1", sqlStatementContext, new HintValueContext());
119     }
120     
121     private void mockMetaData() {
122         ContextManager contextManager = mock(ContextManager.class, RETURNS_DEEP_STUBS);
123         ShardingSphereMetaData metaData = mock(ShardingSphereMetaData.class);
124         when(contextManager.getMetaDataContexts().getMetaData()).thenReturn(metaData);
125         when(metaData.getGlobalRuleMetaData()).thenReturn(mock(RuleMetaData.class));
126         when(metaData.getProps()).thenReturn(new ConfigurationProperties(new Properties()));
127         when(ProxyContext.getInstance().getContextManager()).thenReturn(contextManager);
128     }
129     
130     private void mockMetaDataWithDatabase() {
131         mockMetaData();
132         ShardingSphereMetaData metaData = ProxyContext.getInstance().getContextManager().getMetaDataContexts().getMetaData();
133         ShardingSphereDatabase database = mock(ShardingSphereDatabase.class);
134         when(metaData.getDatabase("foo_db")).thenReturn(database);
135         when(database.getName()).thenReturn("foo_db");
136     }
137     
138     private MockedConstruction<KernelProcessor> mockKernelProcessor(final ExecutionContext executionContext) {
139         return mockConstruction(KernelProcessor.class, (mock, context) -> when(mock.generateExecutionContext(any(), any(), any())).thenReturn(executionContext));
140     }
141     
142     private ExecutionContext createExecutionContext(final Collection<ExecutionUnit> executionUnits) {
143         return new ExecutionContext(mock(), executionUnits, mock(RouteContext.class));
144     }
145     
146     private Collection<ExecutionUnit> createExecutionUnits() {
147         return Collections.singletonList(new ExecutionUnit("ds_0", new SQLUnit("SELECT 1", Collections.emptyList())));
148     }
149     
150     private void mockDatabaseConnectionManager(final List<Connection> connections) throws SQLException {
151         ProxyDatabaseConnectionManager databaseConnectionManager = mock(ProxyDatabaseConnectionManager.class);
152         when(databaseConnectionManager.getConnections(anyString(), anyString(), anyInt(), anyInt(), any())).thenReturn(connections);
153         when(connectionSession.getDatabaseConnectionManager()).thenReturn(databaseConnectionManager);
154     }
155     
156     private Connection mockBackendConnection(final PreparedStatement preparedStatement, final SQLException ex) throws SQLException {
157         Connection result = mock(Connection.class);
158         if (null == ex) {
159             when(result.prepareStatement(anyString())).thenReturn(preparedStatement);
160         } else {
161             when(result.prepareStatement(anyString())).thenThrow(ex);
162         }
163         return result;
164     }
165     
166     private PreparedStatement mockPreparedStatement() {
167         return mock(PreparedStatement.class);
168     }
169 }
