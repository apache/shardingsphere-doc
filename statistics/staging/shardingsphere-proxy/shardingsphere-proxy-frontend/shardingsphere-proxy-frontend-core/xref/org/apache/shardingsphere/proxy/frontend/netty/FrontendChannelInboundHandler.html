18  package org.apache.shardingsphere.proxy.frontend.netty;
19

21  import io.netty.channel.Channel;
22  import io.netty.channel.ChannelHandlerContext;
23  import io.netty.channel.ChannelInboundHandlerAdapter;
24  import io.netty.handler.timeout.IdleStateEvent;
25  import lombok.extern.slf4j.Slf4j;
26  import org.apache.shardingsphere.authentication.result.AuthenticationResult;
27  import org.apache.shardingsphere.database.protocol.constant.CommonConstants;
28  import org.apache.shardingsphere.infra.executor.sql.process.Process;
29  import org.apache.shardingsphere.infra.executor.sql.process.ProcessEngine;
30  import org.apache.shardingsphere.infra.executor.sql.process.ProcessRegistry;
31  import org.apache.shardingsphere.infra.metadata.user.Grantee;
32  import org.apache.shardingsphere.infra.session.connection.ConnectionContext;
33  import org.apache.shardingsphere.proxy.backend.session.ConnectionSession;
34  import org.apache.shardingsphere.proxy.frontend.exception.ExpectedExceptions;
35  import org.apache.shardingsphere.proxy.frontend.executor.ConnectionThreadExecutorGroup;
36  import org.apache.shardingsphere.proxy.frontend.executor.UserExecutorGroup;
37  import org.apache.shardingsphere.proxy.frontend.spi.DatabaseProtocolFrontendEngine;
38  import org.apache.shardingsphere.proxy.frontend.state.ProxyStateContext;
39  
40  import java.sql.SQLException;
41  import java.util.Collection;
42  import java.util.Optional;
43  import java.util.concurrent.atomic.AtomicBoolean;
44  
45  /**
46   * Frontend channel inbound handler.
47   */
48  @Slf4j
49  public final class FrontendChannelInboundHandler extends ChannelInboundHandlerAdapter {
50      
51      private final DatabaseProtocolFrontendEngine databaseProtocolFrontendEngine;
52      
53      private final ConnectionSession connectionSession;
54      
55      private final ProcessEngine processEngine = new ProcessEngine();
56      
57      private final AtomicBoolean authenticated = new AtomicBoolean(false);
58      
59      public FrontendChannelInboundHandler(final DatabaseProtocolFrontendEngine databaseProtocolFrontendEngine, final Channel channel) {
60          this.databaseProtocolFrontendEngine = databaseProtocolFrontendEngine;
61          connectionSession = new ConnectionSession(databaseProtocolFrontendEngine.getType(), channel);
62      }
63      
64      @Override
65      public void channelActive(final ChannelHandlerContext context) {
66          int connectionId = databaseProtocolFrontendEngine.getAuthenticationEngine().handshake(context);
67          ConnectionThreadExecutorGroup.getInstance().register(connectionId);
68          connectionSession.setConnectionId(connectionId);
69      }
70      
71      @Override
72      public void channelRead(final ChannelHandlerContext context, final Object message) {
73          if (!authenticated.get()) {
74              authenticated.set(authenticate(context, (ByteBuf) message));
75              return;
76          }
77          ProxyStateContext.execute(context, message, databaseProtocolFrontendEngine, connectionSession);
78      }
79      
80      private boolean authenticate(final ChannelHandlerContext context, final ByteBuf message) {
81          try {
82              AuthenticationResult authResult = databaseProtocolFrontendEngine.getAuthenticationEngine().authenticate(context,
83                      databaseProtocolFrontendEngine.getCodecEngine().createPacketPayload(message, context.channel().attr(CommonConstants.CHARSET_ATTRIBUTE_KEY).get()));
84              if (authResult.isFinished()) {
85                  finishAuthentication(authResult);
86              }
87              return authResult.isFinished();
88              // CHECKSTYLE:OFF
89          } catch (final Exception ex) {
90              // CHECKSTYLE:ON
91              failAuthentication(context, ex);
92          } finally {
93              message.release();
94          }
95          return false;
96      }
97      
98      private void finishAuthentication(final AuthenticationResult authenticationResult) {
99          connectionSession.setGrantee(new Grantee(authenticationResult.getUsername(), authenticationResult.getHostname()));
100         connectionSession.setCurrentDatabaseName(authenticationResult.getDatabase());
101         connectionSession.setProcessId(processEngine.connect(connectionSession.getUsedDatabaseName(), connectionSession.getConnectionContext().getGrantee()));
102         connectionSession.setConnectionAttributes(authenticationResult.getConnectionAttributes());
103     }
104     
105     private void failAuthentication(final ChannelHandlerContext context, final Exception cause) {
106         if (ExpectedExceptions.isExpected(cause.getClass())) {
107             log.debug("Exception occur: ", cause);
108         } else {
109             log.error("Exception occur: ", cause);
110         }
111         context.writeAndFlush(databaseProtocolFrontendEngine.getCommandExecuteEngine().getErrorPacket(cause));
112         context.close();
113     }
114     
115     @Override
116     public void userEventTriggered(final ChannelHandlerContext ctx, final Object event) throws Exception {
117         if (event instanceof IdleStateEvent) {
118             if (isIdle()) {
119                 ConnectionContext connectionContext = connectionSession.getConnectionContext();
120                 Grantee grantee = null == connectionContext ? null : connectionContext.getGrantee();
121                 String databaseName = null == connectionContext ? null : connectionContext.getCurrentDatabaseName().orElse("NONE");
122                 log.info("Connection {} (processId: {}) will be closed due to receiving an IdleStateEvent as it is idle. Grantee: {}, database name: {}",
123                         connectionSession.getConnectionId(), connectionSession.getProcessId(), grantee, databaseName);
124                 ctx.close();
125             } else {
126                 log.info("Received IdleStateEvent, but connection {} (processId: {}) is not idle, ignore.", connectionSession.getConnectionId(), connectionSession.getProcessId());
127             }
128         }
129         super.userEventTriggered(ctx, event);
130     }
131     
132     private boolean isIdle() {
133         if (null == connectionSession.getProcessId()) {
134             return true;
135         }
136         Process process = ProcessRegistry.getInstance().get(connectionSession.getProcessId());
137         return null == process || process.isIdle();
138     }
139     
140     @Override
141     public void channelInactive(final ChannelHandlerContext context) {
142         context.fireChannelInactive();
143         UserExecutorGroup.getInstance().getExecutorService().execute(this::closeAllResources);
144     }
145     
146     private void closeAllResources() {
147         ConnectionThreadExecutorGroup.getInstance().unregisterAndAwaitTermination(connectionSession.getConnectionId());
148         processCloseExceptions(connectionSession.getDatabaseConnectionManager().closeAllResources());
149         Optional.ofNullable(connectionSession.getProcessId()).ifPresent(processEngine::disconnect);
150         databaseProtocolFrontendEngine.release(connectionSession);
151     }
152     
153     private void processCloseExceptions(final Collection<SQLException> exceptions) {
154         if (exceptions.isEmpty()) {
155             return;
156         }
157         SQLException ex = new SQLException("");
158         for (SQLException each : exceptions) {
159             ex.setNextException(each);
160         }
161         processException(ex);
162     }
163     
164     private void processException(final Exception cause) {
165         if (ExpectedExceptions.isExpected(cause.getClass())) {
166             log.debug("Exception occur: ", cause);
167         } else {
168             log.error("Exception occur: ", cause);
169         }
170     }
171     
172     @Override
173     public void channelWritabilityChanged(final ChannelHandlerContext context) {
174         if (context.channel().isWritable()) {
175             connectionSession.getDatabaseConnectionManager().getConnectionResourceLock().doNotify();
176         }
177     }
178 }
