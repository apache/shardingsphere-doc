52      @Test
53      void assertIsValidBindingTableConfigurationWithInlineAlgorithm() {
54          ShardingRuleConfiguration shardingRuleConfig = new ShardingRuleConfiguration();
55          shardingRuleConfig.getTables().add(createTableRuleConfigurationWithInlineAlgorithm("t_order", "ds_${0..1}.t_order_${0..1}", "ds_inline", "t_order_inline"));
56          shardingRuleConfig.getTables().add(createTableRuleConfigurationWithInlineAlgorithm("t_order_item", "ds_${0..1}.t_order_item_${0..1}", "ds_inline", "t_order_item_inline"));
57          shardingRuleConfig.getShardingAlgorithms().put("ds_inline",
58                  new AlgorithmConfiguration("INLINE", PropertiesBuilder.build(new Property("algorithm-expression", "ds_${user_id % 2}"))));
59          shardingRuleConfig.getShardingAlgorithms().put("t_order_inline",
60                  new AlgorithmConfiguration("INLINE", PropertiesBuilder.build(new Property("algorithm-expression", "t_order_${order_id % 2}"))));
61          shardingRuleConfig.getShardingAlgorithms().put("t_order_item_inline",
62                  new AlgorithmConfiguration("INLINE", PropertiesBuilder.build(new Property("algorithm-expression", "t_order_item_${order_id % 2}"))));
63          shardingRuleConfig.setDefaultDatabaseShardingStrategy(new NoneShardingStrategyConfiguration());
64          shardingRuleConfig.setDefaultTableShardingStrategy(new NoneShardingStrategyConfiguration());
65          ShardingRule shardingRule = new ShardingRule(shardingRuleConfig, createDataSourcesForInline(), mock(ComputeNodeInstanceContext.class, RETURNS_DEEP_STUBS), Collections.emptyList());
66          ShardingTableReferenceRuleConfiguration bindingTableGroup = new ShardingTableReferenceRuleConfiguration("", "t_order,t_order_item");
67          BindingTableCheckedConfiguration checkedConfig = new BindingTableCheckedConfiguration(
68                  shardingRule.getDataSourceNames(), shardingRule.getShardingAlgorithms(), shardingRuleConfig.getShardingAlgorithms(), Collections.singleton(bindingTableGroup),
69                  shardingRuleConfig.getDefaultDatabaseShardingStrategy(), shardingRuleConfig.getDefaultTableShardingStrategy(), shardingRule.getDefaultShardingColumn());
70          ShardingRuleChecker checker = new ShardingRuleChecker(shardingRule);
71          assertTrue(checker.isValidBindingTableConfiguration(shardingRule.getShardingTables(), checkedConfig));
72      }

96      @Test
97      void assertCreateShardingRuleWithDuplicatedBindingTableGroups() {
98          ShardingRuleConfiguration shardingRuleConfig = new ShardingRuleConfiguration();
99          shardingRuleConfig.getTables().add(createTableRuleConfiguration("t_order", "ds_${0..7}.t_order"));
100         shardingRuleConfig.getTables().add(createTableRuleConfiguration("t_order_item", "ds_${0..7}.t_order_item"));
101         shardingRuleConfig.getTables().add(createTableRuleConfiguration("t_product", "ds_${0..7}.t_product"));
102         shardingRuleConfig.getShardingAlgorithms().put("mod_hash", new AlgorithmConfiguration("MOD.HASH.FIXTURE", PropertiesBuilder.build(new Property("sharding-count", "8"))));
103         shardingRuleConfig.setDefaultDatabaseShardingStrategy(new NoneShardingStrategyConfiguration());
104         shardingRuleConfig.setDefaultTableShardingStrategy(new NoneShardingStrategyConfiguration());
105         shardingRuleConfig.getBindingTableGroups().add(new ShardingTableReferenceRuleConfiguration("foo_group_1", "t_order,t_order_item"));
106         shardingRuleConfig.getBindingTableGroups().add(new ShardingTableReferenceRuleConfiguration("foo_group_2", "t_order_item,t_product"));
107         SQLException actual = assertThrows(DuplicateBindingTableException.class, () -> new ShardingRule(shardingRuleConfig, createDataSources(), mock(ComputeNodeInstanceContext.class,
108                 RETURNS_DEEP_STUBS), Collections.emptyList())).toSQLException();
109         assertThat(actual.getErrorCode(), is(20007));
110         assertThat(actual.getSQLState(), is("42S01"));
111         assertThat(actual.getMessage(), is("Logic table 't_order_item' should not appear in more than one binding table group, found in 'foo_group_1' and 'foo_group_2'."));
112     }

147     private ShardingTableRuleConfiguration createTableRuleConfigurationWithInlineAlgorithm(final String tableName, final String actualDataNodes,
148                                                                                            final String databaseAlgorithmName, final String tableAlgorithmName) {
149         ShardingTableRuleConfiguration result = new ShardingTableRuleConfiguration(tableName, actualDataNodes);
150         result.setDatabaseShardingStrategy(new StandardShardingStrategyConfiguration("user_id", databaseAlgorithmName));
151         result.setTableShardingStrategy(new StandardShardingStrategyConfiguration("order_id", tableAlgorithmName));
152         return result;
153     }
154     
155     private ShardingTableRuleConfiguration createTableRuleConfiguration(final String tableName, final String actualDataNodes) {
156         ShardingTableRuleConfiguration result = new ShardingTableRuleConfiguration(tableName, actualDataNodes);
157         result.setDatabaseShardingStrategy(new StandardShardingStrategyConfiguration("route_group_id", "mod_hash"));
158         result.setTableShardingStrategy(new NoneShardingStrategyConfiguration());
159         return result;
160     }
