57  /**
58   * DDL statement visitor for Hive.
59   */
60  public final class HiveDDLStatementVisitor extends HiveStatementVisitor implements DDLStatementVisitor {
61      
62      public HiveDDLStatementVisitor(final DatabaseType databaseType) {
63          super(databaseType);
64      }
65      
66      @Override
67      public ASTNode visitCreateDatabase(final CreateDatabaseContext ctx) {
68          return new CreateDatabaseStatement(getDatabaseType(), new IdentifierValue(ctx.identifier().getText()).getValue(), null != ctx.ifNotExists());
69      }
70      
71      @Override
72      public ASTNode visitDropDatabase(final DropDatabaseContext ctx) {
73          return new DropDatabaseStatement(getDatabaseType(), new IdentifierValue(ctx.identifier().getText()).getValue(), null != ctx.ifExists());
74      }
75      
76      @Override
77      public ASTNode visitAlterDatabase(final AlterDatabaseContext ctx) {
78          return new AlterDatabaseStatement(getDatabaseType());
79      }
80      
81      @Override
82      public ASTNode visitAlterTable(final AlterTableContext ctx) {
83          AlterTableStatement result = new AlterTableStatement(getDatabaseType());
84          result.setTable((SimpleTableSegment) visit(ctx.alterTableCommonClause().tableName()));
85          if (null != ctx.COMPACT()) {
86              String compactionType = ctx.string_().getText().replace("'", "");
87              if (!isValidCompactionType(compactionType)) {
88                  throw new IllegalArgumentException("Invalid compaction type. Must be 'MAJOR', 'MINOR' or 'REBALANCE'");
89              }
90              if ((null != ctx.clusteredIntoClause() || null != ctx.orderByClause())
91                      && !"REBALANCE".equalsIgnoreCase(compactionType)) {
92                  throw new IllegalArgumentException("[CLUSTERED INTO n BUCKETS] and [ORDER BY col_list] clauses can only be used with REBALANCE compaction");
93              }
94          }
95          return result;
96      }
97      
98      private boolean isValidCompactionType(final String compactionType) {
99          return "MAJOR".equalsIgnoreCase(compactionType)
100                 || "MINOR".equalsIgnoreCase(compactionType)
101                 || "REBALANCE".equalsIgnoreCase(compactionType);
102     }
103     
104     @Override
105     public ASTNode visitMsckStatement(final MsckStatementContext ctx) {
106         AlterTableStatement result = new AlterTableStatement(getDatabaseType());
107         result.setTable((SimpleTableSegment) visit(ctx.tableName()));
108         return result;
109     }
110     
111     @SuppressWarnings("unchecked")
112     @Override
113     public ASTNode visitDropTable(final DropTableContext ctx) {
114         DropTableStatement result = new DropTableStatement(getDatabaseType());
115         result.setIfExists(null != ctx.ifExists());
116         result.getTables().add((SimpleTableSegment) visit(ctx.tableNameWithDb()));
117         return result;
118     }
119     
120     @SuppressWarnings("unchecked")
121     @Override
122     public ASTNode visitTruncateTable(final TruncateTableContext ctx) {
123         return new TruncateStatement(getDatabaseType(), Collections.singleton((SimpleTableSegment) visit(ctx.tableNameWithDb())));
124     }
125     
126     @SuppressWarnings("unchecked")
127     @Override
128     public ASTNode visitCreateTable(final CreateTableContext ctx) {
129         CreateTableStatement result = new CreateTableStatement(getDatabaseType());
130         result.setTable((SimpleTableSegment) visit(ctx.createTableCommonClause().tableNameWithDb()));
131         result.setIfNotExists(null != ctx.createTableCommonClause().ifNotExists());
132         if (null != ctx.createDefinitionClause()) {
133             CollectionValue<CreateDefinitionSegment> createDefinitions = (CollectionValue<CreateDefinitionSegment>) visit(ctx.createDefinitionClause());
134             for (CreateDefinitionSegment each : createDefinitions.getValue()) {
135                 if (each instanceof ColumnDefinitionSegment) {
136                     result.getColumnDefinitions().add((ColumnDefinitionSegment) each);
137                 } else if (each instanceof ConstraintDefinitionSegment) {
138                     result.getConstraintDefinitions().add((ConstraintDefinitionSegment) each);
139                 }
140             }
141         }
142         return result;
143     }
144     
145     @Override
146     public ASTNode visitCreateDefinitionClause(final CreateDefinitionClauseContext ctx) {
147         CollectionValue<CreateDefinitionSegment> result = new CollectionValue<>();
148         for (ColumnDefinitionContext each : ctx.columnDefinition()) {
149             result.getValue().add((ColumnDefinitionSegment) visit(each));
150         }
151         for (TableConstraintContext each : ctx.tableConstraint()) {
152             result.getValue().add((ConstraintDefinitionSegment) visit(each));
153         }
154         return result;
155     }
156     
157     @Override
158     public ASTNode visitColumnDefinition(final ColumnDefinitionContext ctx) {
159         ColumnSegment column = new ColumnSegment(ctx.columnName().getStart().getStartIndex(), ctx.columnName().getStop().getStopIndex(),
160                 new IdentifierValue(ctx.columnName().getText()));
161         DataTypeSegment dataType = (DataTypeSegment) visit(ctx.dataTypeClause());
162         return new ColumnDefinitionSegment(ctx.getStart().getStartIndex(), ctx.getStop().getStopIndex(), column, dataType, false, false, getText(ctx));
163     }
164     
165     private String getText(final ParserRuleContext ctx) {
166         return ctx.start.getInputStream().getText(new Interval(ctx.start.getStartIndex(), ctx.stop.getStopIndex()));
167     }
168     
169     @Override
170     public ASTNode visitDataTypeClause(final DataTypeClauseContext ctx) {
171         DataTypeSegment result = new DataTypeSegment();
172         result.setStartIndex(ctx.getStart().getStartIndex());
173         result.setStopIndex(ctx.getStop().getStopIndex());
174         if (null != ctx.primitiveType()) {
175             result.setDataTypeName(ctx.primitiveType().getText());
176         } else if (null != ctx.arrayType()) {
177             result.setDataTypeName(ctx.arrayType().getText());
178         } else if (null != ctx.mapType()) {
179             result.setDataTypeName(ctx.mapType().getText());
180         } else if (null != ctx.structType()) {
181             result.setDataTypeName(ctx.structType().getText());
182         } else if (null != ctx.unionType()) {
183             result.setDataTypeName(ctx.unionType().getText());
184         }
185         return result;
186     }
187     
188     @Override
189     public ASTNode visitTableNameWithDb(final TableNameWithDbContext ctx) {
190         if (1 == ctx.identifier().size()) {
191             return new SimpleTableSegment(new TableNameSegment(ctx.getStart().getStartIndex(), ctx.getStop().getStopIndex(),
192                     new IdentifierValue(ctx.identifier(0).getText())));
193         } else {
194             SimpleTableSegment result = new SimpleTableSegment(new TableNameSegment(ctx.identifier(1).getStart().getStartIndex(),
195                     ctx.identifier(1).getStop().getStopIndex(), new IdentifierValue(ctx.identifier(1).getText())));
196             result.setOwner(new org.apache.shardingsphere.sql.parser.statement.core.segment.generic.OwnerSegment(
197                     ctx.identifier(0).getStart().getStartIndex(), ctx.identifier(0).getStop().getStopIndex(),
198                     new IdentifierValue(ctx.identifier(0).getText())));
199             return result;
200         }
201     }
202     
203     @Override
204     public ASTNode visitTableConstraint(final TableConstraintContext ctx) {
205         return new ConstraintDefinitionSegment(ctx.getStart().getStartIndex(), ctx.getStop().getStopIndex());
206     }
