46  class SearchMetadataPayloadBuilderTest {
47      
48      private static final String TOOL_NAME = "database_gateway_search_metadata";
49      
50      @Test
51      void assertBuildsLargeResultGuidance() {
52          MetadataSearchResult searchResult = new MetadataSearchResult(List.of(createHit("logic_db", "public", "orders", List.of("name"))),
53                  Map.of("object_types", List.of("table")), 101, 100, true, 100);
54          Map<String, Object> actual = SearchMetadataPayloadBuilder.build(mock(MCPFeatureRequestContext.class), createRequest("", "", ""), searchResult, TOOL_NAME);
55          assertThat(actual.get("summary"), is("Metadata search returned 100 of 101 matches."));
56          assertTrue((Boolean) actual.get("truncated"));
57          Map<?, ?> actualLargeResultGuidance = (Map<?, ?>) actual.get("large_result_guidance");
58          assertThat(actualLargeResultGuidance.get("state"), is("metadata_search_result_truncated"));
59          assertThat(actualLargeResultGuidance.get("narrowing_arguments"), is(List.of("database", "schema", "query", "object_types")));
60      }
61      
62      @Test
63      void assertBuildsPaginationNavigation() {
64          MetadataSearchResult searchResult = new MetadataSearchResult(List.of(createHit("logic_db", "public", "orders", List.of("name"))),
65                  Map.of("object_types", List.of("table")), 3, 1, true, 100);
66          MetadataSearchRequest request = new MetadataSearchRequest("logic_db", "public", "order", Set.of(SupportedMCPMetadataObjectType.TABLE), 1, 1);
67          Map<String, Object> actual = SearchMetadataPayloadBuilder.build(mock(MCPFeatureRequestContext.class), request, searchResult, TOOL_NAME);
68          assertTrue((Boolean) actual.get("has_more"));
69          assertThat(actual.get("next_offset"), is(2));
70          assertThat(actual.get("continuation_mode"), is("pagination"));
71          assertFalse(actual.containsKey("large_result_guidance"));
72          Map<?, ?> actualNextAction = (Map<?, ?>) ((List<?>) actual.get("next_actions")).getFirst();
73          assertThat(actualNextAction.get("type"), is("tool_call"));
74          assertThat(actualNextAction.get("tool_name"), is(TOOL_NAME));
75          assertThat(actualNextAction.get("arguments"), is(Map.of(
76                  "database", "logic_db", "schema", "public", "query", "order", "object_types", List.of("table"), "limit", 1, "offset", 2)));
77      }
78      
79      @Test
80      void assertBuildsLastPage() {
81          MetadataSearchResult searchResult = new MetadataSearchResult(List.of(createHit("logic_db", "public", "orders", List.of("name"))),
82                  Map.of("object_types", List.of("table")), 101, 1, false, 100);
83          MetadataSearchRequest request = new MetadataSearchRequest("logic_db", "", "", Set.of(SupportedMCPMetadataObjectType.TABLE), 2, 100);
84          Map<String, Object> actual = SearchMetadataPayloadBuilder.build(mock(MCPFeatureRequestContext.class), request, searchResult, TOOL_NAME);
85          assertFalse((Boolean) actual.get("has_more"));
86          assertFalse(actual.containsKey("next_offset"));
87          assertFalse(actual.containsKey("large_result_guidance"));
88          assertFalse(actual.containsKey("empty_state"));
89          assertFalse(actual.containsKey("next_actions"));
90      }
91      
92      @Test
93      void assertBuildsEmptyStateWithBroadenedSearchAction() {
94          MCPFeatureRequestContext requestContext = createDatabaseContext();
95          MetadataSearchResult searchResult = new MetadataSearchResult(List.of(), Map.of(), 0, 0, false, 100);
96          Map<String, Object> actual = SearchMetadataPayloadBuilder.build(requestContext, createRequest("logic_db", "public", "missing"), searchResult, TOOL_NAME);
97          Map<?, ?> actualEmptyState = (Map<?, ?>) actual.get("empty_state");
98          assertThat(actualEmptyState.get("category"), is("object_not_visible"));
99          Map<?, ?> actualNextAction = (Map<?, ?>) ((List<?>) actual.get("next_actions")).getFirst();
100         assertThat(actualNextAction.get("type"), is("tool_call"));
101         assertThat(actualNextAction.get("tool_name"), is(TOOL_NAME));
102         assertThat(((Map<?, ?>) actualNextAction.get("arguments")).get("query"), is("missing"));
103         assertFalse(((Map<?, ?>) actualNextAction.get("arguments")).containsKey("database"));
104     }
105     
106     @Test
107     void assertBuildsAmbiguityState() {
108         MetadataSearchResult searchResult = new MetadataSearchResult(List.of(
109                 createHit("bar_db", "public", "orders", List.of("name")),
110                 createHit("baz_db", "public", "orders", List.of("name")),
111                 createHit("foo_db", "public", "orders_archive", List.of("name"))),
112                 Map.of(), 3, 3, false, 100);
113         Map<String, Object> actual = SearchMetadataPayloadBuilder.build(mock(MCPFeatureRequestContext.class), createRequest("", "", "orders"), searchResult, TOOL_NAME);
114         Map<?, ?> actualAmbiguityState = (Map<?, ?>) actual.get("ambiguity_state");
115         assertTrue((Boolean) actualAmbiguityState.get("ambiguous"));
116         assertThat(actualAmbiguityState.get("ambiguous_by"), is(List.of("name", "database")));
117         assertThat(actualAmbiguityState.get("candidate_count"), is(2));
118         assertThat(actualAmbiguityState.get("duplicated_names"), is(List.of("orders")));
119         Map<?, ?> actualNextAction = (Map<?, ?>) ((List<?>) actual.get("next_actions")).getFirst();
120         assertThat(actualNextAction.get("order"), is(1));
121     }
122     
123     private MetadataSearchRequest createRequest(final String database, final String schema, final String query) {
124         return new MetadataSearchRequest(database, schema, query, Set.of(SupportedMCPMetadataObjectType.TABLE), 100, 0);
125     }
126     
127     private MetadataSearchHit createHit(final String database, final String schema, final String name, final List<String> matchedFields) {
128         return MetadataSearchHit.builder()
129                 .database(database).schema(schema).objectType("table").table(name).view("").name(name)
130                 .resource(Map.of()).parentResource(Map.of()).nextResources(List.of()).derivationStatus("complete").derivationReason("")
131                 .matchKind("exact").matchedFields(matchedFields).matchedValue(name).build();
132     }
133     
134     private MCPFeatureRequestContext createDatabaseContext() {
135         MCPFeatureRequestContext result = mock(MCPFeatureRequestContext.class);
136         MCPMetadataQueryFacade metadataQueryFacade = mock(MCPMetadataQueryFacade.class);
137         MCPFeatureCapabilityFacade capabilityFacade = mock(MCPFeatureCapabilityFacade.class);
138         when(result.getMetadataQueryFacade()).thenReturn(metadataQueryFacade);
139         when(result.getCapabilityFacade()).thenReturn(capabilityFacade);
140         when(metadataQueryFacade.queryDatabases()).thenReturn(
141                 List.of(new RuntimeDatabaseProfile("logic_db", "FixtureDB", "1.0", TransactionCapability.LOCAL_WITH_SAVEPOINT,
142                         new DatabaseIdentifierContext(IdentifierCasePolicyFactory.newInsensitivePolicySet()))));
143         when(metadataQueryFacade.querySchema("logic_db", "public")).thenReturn(Optional.of(mock(ShardingSphereSchema.class)));
144         when(capabilityFacade.findDatabaseProfile("logic_db")).thenReturn(Optional.of(mock(RuntimeDatabaseProfile.class)));
145         return result;
146     }
