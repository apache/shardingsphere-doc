42  /**
43   * MCP tool specification factory.
44   */
45  public final class MCPToolSpecificationFactory {
46      
47      private final List<MCPToolDescriptor> descriptors;
48      
49      private final MCPToolController controller;
50      
51      private final MCPToolElicitationHandler elicitationHandler;
52      
53      private final MCPCallToolResultFactory callToolResultFactory;
54      
55      public MCPToolSpecificationFactory(final MCPRuntimeContext runtimeContext) {
56          descriptors = ToolDefinitionRegistry.getSupportedToolDescriptors();
57          controller = new MCPToolController(runtimeContext);
58          elicitationHandler = new MCPToolElicitationHandler(controller, runtimeContext.getActiveTransport(), Clock.systemUTC());
59          callToolResultFactory = new MCPCallToolResultFactory();
60      }
61      
62      /**
63       * Create MCP tool specifications.
64       *
65       * @return tool specifications
66       */
67      public List<SyncToolSpecification> createToolSpecifications() {
68          return descriptors.stream().map(each -> new SyncToolSpecification(createTool(each), this::handle)).toList();
69      }
70      
71      private McpSchema.Tool createTool(final MCPToolDescriptor descriptor) {
72          McpSchema.Tool.Builder result = McpSchema.Tool.builder().name(descriptor.getName()).title(descriptor.getTitle()).description(descriptor.getDescription())
73                  .inputSchema(createInputSchema(descriptor.getInputSchema()));
74          if (!descriptor.getOutputSchema().isEmpty()) {
75              result.outputSchema(descriptor.getOutputSchema());
76          }
77          MCPToolAnnotations annotations = descriptor.getAnnotations();
78          result.annotations(
79                  new ToolAnnotations(annotations.getTitle(), annotations.isReadOnlyHint(), annotations.isDestructiveHint(), annotations.isIdempotentHint(), annotations.isOpenWorldHint(), null));
80          if (!descriptor.getMeta().isEmpty()) {
81              result.meta(descriptor.getMeta());
82          }
83          return result.build();
84      }
85      
86      @SuppressWarnings("unchecked")
87      private McpSchema.JsonSchema createInputSchema(final Map<String, Object> inputSchema) {
88          String type = String.valueOf(inputSchema.get("type"));
89          Map<String, Object> props = (Map<String, Object>) inputSchema.get("properties");
90          List<String> required = (List<String>) inputSchema.get("required");
91          Boolean additionalProps = (Boolean) inputSchema.get("additionalProperties");
92          return new McpSchema.JsonSchema(type, props, required, additionalProps, Collections.emptyMap(), Collections.emptyMap());
93      }
94      
95      private McpSchema.CallToolResult handle(final McpSyncServerExchange exchange, final McpSchema.CallToolRequest request) {
96          try {
97              Map<String, Object> arguments = Optional.ofNullable(request.arguments()).orElse(Collections.emptyMap());
98              MCPToolDefinition definition = ToolDefinitionRegistry.getToolDefinition(request.name());
99              MCPSuccessPayload payload = controller.handle(exchange.sessionId(), definition, arguments);
100             return callToolResultFactory.create(definition.getDescriptor(), getEffectivePayload(exchange, payload, definition, arguments));
101         } catch (final ShardingSphereMCPException | RuntimeDatabaseConnectionException ex) {
102             return callToolResultFactory.create(MCPErrorConverter.convert(ex));
103             // CHECKSTYLE:OFF
104         } catch (final RuntimeException ex) {
105             // CHECKSTYLE:ON
106             throw MCPTransportErrorFactory.createError(ex);
107         }
108     }
109     
110     private MCPSuccessPayload getEffectivePayload(final McpSyncServerExchange exchange, final MCPSuccessPayload successPayload, final MCPToolDefinition definition,
111                                                   final Map<String, Object> arguments) {
112         Map<String, Object> payload = successPayload.toPayload();
113         return elicitationHandler.shouldHandle(definition.getDescriptor(), payload)
114                 ? elicitationHandler.handle(exchange, definition, arguments, payload).orElse(successPayload)
115                 : successPayload;
116     }
117 }
