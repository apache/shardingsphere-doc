33  import static org.hamcrest.MatcherAssert.assertThat;
34  import static org.hamcrest.Matchers.is;
35  import static org.junit.jupiter.api.Assertions.assertThrows;
36  
37  class MCPCompletionTargetDescriptorValidatorTest {
38      
39      @ParameterizedTest(name = "{0}")
40      @MethodSource("invalidCompletionTargets")
41      void assertValidateWithInvalidCompletionTarget(final String name, final Collection<MCPCompletionTargetDescriptor> descriptors, final Collection<MCPResourceDescriptor> resources,
42                                                     final String expectedMessage) {
43          MCPPromptDescriptor prompt = new MCPPromptDescriptor("test_prompt", "Test Prompt", "Guide the model through a test prompt.",
44                  List.of(new MCPPromptArgumentDescriptor("database", "Database", "Logical database.", false)), Map.of());
45          IllegalStateException actual = assertThrows(IllegalStateException.class, () -> MCPCompletionTargetDescriptorValidator.validate(descriptors, List.of(prompt), resources));
46          assertThat(actual.getMessage(), is(expectedMessage));
47      }
48      
49      private static Collection<Arguments> invalidCompletionTargets() {
50          MCPCompletionTargetDescriptor completion = new MCPCompletionTargetDescriptor("prompt", "test_prompt", List.of("database"), 50, Map.of());
51          MCPResourceDescriptor fixedResource = new MCPResourceDescriptor("shardingsphere://capabilities", "foo_fixed", "Capabilities", "Read capabilities.", "application/json",
52                  MCPResourceAnnotations.EMPTY, Map.of());
53          MCPResourceDescriptor template = new MCPResourceDescriptor("shardingsphere://databases/{database}", "foo_template", "Database", "Read database.", "application/json",
54                  MCPResourceAnnotations.EMPTY, Map.of());
55          return List.of(
56                  Arguments.of("unknown context argument",
57                          List.of(new MCPCompletionTargetDescriptor("prompt", "test_prompt", List.of("database"), 50,
58                                  Map.of(MCPShardingSphereMetadataKeys.REQUIRED_CONTEXT_ARGUMENTS, Map.of("database", List.of("tenant"))))),
59                          List.of(), "Completion target `prompt:test_prompt` context argument `tenant` for `database` is not declared by the target."),
60                  Arguments.of("duplicate arguments", List.of(new MCPCompletionTargetDescriptor("prompt", "test_prompt", List.of("database", "database"), 50, Map.of())), List.of(),
61                          "Completion target `prompt:test_prompt` must not contain duplicate arguments."),
62                  Arguments.of("duplicate target", List.of(completion, completion), List.of(), "Duplicate MCP completion target `prompt:test_prompt`."),
63                  Arguments.of("unknown prompt", List.of(new MCPCompletionTargetDescriptor("prompt", "foo_unknown", List.of("database"), 50, Map.of())), List.of(),
64                          "Completion target references unknown prompt `foo_unknown`."),
65                  Arguments.of("unknown resource", List.of(new MCPCompletionTargetDescriptor("resource", "shardingsphere://foo_unknown", List.of("database"), 50, Map.of())), List.of(),
66                          "Completion target references unknown resource `shardingsphere://foo_unknown`."),
67                  Arguments.of("unsupported reference", List.of(new MCPCompletionTargetDescriptor("tool", "test_prompt", List.of("database"), 50, Map.of())), List.of(),
68                          "Unsupported completion reference type `tool`."),
69                  Arguments.of("fixed resource target", List.of(new MCPCompletionTargetDescriptor("resource", "shardingsphere://capabilities", List.of(), 50, Map.of())), List.of(fixedResource),
70                          "Completion target `resource:shardingsphere://capabilities` must reference a resource template."),
71                  Arguments.of("unknown resource argument", List.of(new MCPCompletionTargetDescriptor("resource", "shardingsphere://databases/{database}", List.of("schema"), 50, Map.of())),
72                          List.of(template), "Completion target `resource:shardingsphere://databases/{database}` argument `schema` is not a URI template variable."),
73                  Arguments.of("context metadata is not object",
74                          List.of(new MCPCompletionTargetDescriptor("prompt", "test_prompt", List.of("database"), 50, Map.of(MCPShardingSphereMetadataKeys.REQUIRED_CONTEXT_ARGUMENTS, "database"))),
75                          List.of(), "Completion target `prompt:test_prompt` metadata `org.apache.shardingsphere/required-context-arguments` must be an object."),
76                  Arguments.of("unknown context metadata key",
77                          List.of(new MCPCompletionTargetDescriptor("prompt", "test_prompt", List.of("database"), 50,
78                                  Map.of(MCPShardingSphereMetadataKeys.REQUIRED_CONTEXT_ARGUMENTS, Map.of("schema", List.of())))),
79                          List.of(), "Completion target `prompt:test_prompt` required context argument `schema` is not declared by the target."),
80                  Arguments.of("context value is not list",
81                          List.of(new MCPCompletionTargetDescriptor("prompt", "test_prompt", List.of("database"), 50,
82                                  Map.of(MCPShardingSphereMetadataKeys.REQUIRED_CONTEXT_ARGUMENTS, Map.of("database", "schema")))),
83                          List.of(), "Completion target `prompt:test_prompt` required context for `database` must be a list."));
84      }
