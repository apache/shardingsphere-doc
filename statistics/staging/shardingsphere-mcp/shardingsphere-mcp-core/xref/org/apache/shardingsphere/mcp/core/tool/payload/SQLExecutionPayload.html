18  package org.apache.shardingsphere.mcp.core.tool.payload;
19  
20  import lombok.AccessLevel;
21  import lombok.RequiredArgsConstructor;
22  import org.apache.shardingsphere.mcp.api.payload.MCPSuccessPayload;
23  import org.apache.shardingsphere.mcp.support.database.tool.result.SQLExecutionColumnDefinition;
24  import org.apache.shardingsphere.mcp.support.database.tool.result.SQLExecutionResult;
25  import org.apache.shardingsphere.mcp.support.database.tool.result.SQLExecutionResultKind;
26  import org.apache.shardingsphere.mcp.support.protocol.MCPNextActionUtils;
27  import org.apache.shardingsphere.mcp.support.protocol.MCPPayloadFieldNames;
28  import org.apache.shardingsphere.mcp.support.protocol.MCPResponseMode;
29  
30  import java.util.LinkedHashMap;
31  import java.util.LinkedHashSet;
32  import java.util.List;
33  import java.util.Locale;
34  import java.util.Map;
35  import java.util.Set;
36  
37  /**
38   * SQL execution payload.
39   */
40  @RequiredArgsConstructor(access = AccessLevel.PRIVATE)
41  public final class SQLExecutionPayload implements MCPSuccessPayload {
42      
43      private static final int ROW_OBJECT_LIMIT = 100;
44      
45      private static final String EXECUTED_TRUNCATION_SUMMARY = " Returned rows were truncated; do not replay the statement automatically.";
46      
47      private static final String EXECUTED_TRUNCATION_ACTION = "The side-effecting statement already executed and returned truncated rows; do not replay it automatically. "
48              + "Use a separate read-only query if more data is needed.";
49      
50      private final SQLExecutionResult executionResult;
51      
52      private final String responseMode;
53      
54      /**
55       * Create a payload for a read-only query result.
56       *
57       * @param executionResult SQL execution result
58       * @return query payload
59       */
60      public static SQLExecutionPayload query(final SQLExecutionResult executionResult) {
61          return new SQLExecutionPayload(executionResult, MCPResponseMode.QUERY);
62      }
63      
64      /**
65       * Create a payload for an executed side-effecting statement result.
66       *
67       * @param executionResult SQL execution result
68       * @return executed payload
69       */
70      public static SQLExecutionPayload executed(final SQLExecutionResult executionResult) {
71          return new SQLExecutionPayload(executionResult, MCPResponseMode.EXECUTED);
72      }
73      
74      @Override
75      public Map<String, Object> toPayload() {
76          Map<String, Object> result = new LinkedHashMap<>(33, 1F);
77          result.put("response_mode", responseMode);
78          result.put("result_kind", executionResult.getResultKind().name().toLowerCase(Locale.ENGLISH));
79          if (isExecuted()) {
80              result.put(MCPPayloadFieldNames.EXECUTION_MODE, "execute");
81          }
82          result.put("statement_class", executionResult.getStatementClass().name().toLowerCase(Locale.ENGLISH));
83          result.put("statement_type", executionResult.getStatementType());
84          result.put("status", "OK");
85          result.put(MCPPayloadFieldNames.SUMMARY, createSummary());
86          if (!executionResult.getNormalizedSql().isEmpty()) {
87              result.put("normalized_sql", executionResult.getNormalizedSql());
88          }
89          if (SQLExecutionResultKind.RESULT_SET == executionResult.getResultKind()) {
90              result.put("columns", executionResult.getColumns());
91              result.put("rows", executionResult.getRows());
92              appendRowObjects(result);
93              result.put("returned_row_count", executionResult.getRows().size());
94          }
95          if (SQLExecutionResultKind.UPDATE_COUNT == executionResult.getResultKind()) {
96              result.put("affected_rows", executionResult.getAffectedRows());
97          }
98          result.put("applied_max_rows", executionResult.getAppliedMaxRows());
99          result.put("applied_timeout_ms", executionResult.getAppliedTimeoutMs());
100         result.put("truncated", executionResult.isTruncated());
101         result.put(MCPPayloadFieldNames.NEXT_ACTIONS, createNextActions());
102         return result;
103     }
104     
105     private boolean isExecuted() {
106         return MCPResponseMode.EXECUTED.equals(responseMode);
107     }
108     
109     private String createSummary() {
110         return switch (executionResult.getResultKind()) {
111             case RESULT_SET -> createResultSetSummary();
112             case UPDATE_COUNT -> String.format("Executed %s statement and affected %d row(s).", executionResult.getStatementType(), executionResult.getAffectedRows());
113             case STATEMENT_ACK -> createStatementAcknowledgement();
114         };
115     }
116     
117     private String createResultSetSummary() {
118         String result = isExecuted()
119                 ? String.format("Executed side-effecting SQL (statement type %s) and returned %d row(s).", executionResult.getStatementType(), executionResult.getRows().size())
120                 : String.format("Executed %s statement and returned %d row(s).", executionResult.getStatementType(), executionResult.getRows().size());
121         if (!executionResult.isTruncated()) {
122             return result;
123         }
124         return result + (isExecuted() ? EXECUTED_TRUNCATION_SUMMARY : " Result was truncated.");
125     }
126     
127     private String createStatementAcknowledgement() {
128         switch (executionResult.getStatementType()) {
129             case "BEGIN", "START TRANSACTION":
130                 return "Transaction started.";
131             case "COMMIT":
132                 return "Transaction committed.";
133             case "ROLLBACK":
134                 return "Transaction rolled back.";
135             case "SAVEPOINT":
136                 return "Savepoint created.";
137             case "ROLLBACK TO SAVEPOINT":
138                 return "Savepoint rolled back.";
139             case "RELEASE SAVEPOINT":
140                 return "Savepoint released.";
141             default:
142                 return "Statement executed.";
143         }
144     }
145     
146     private void appendRowObjects(final Map<String, Object> payload) {
147         String rowObjectStatus = getRowObjectStatus();
148         payload.put("row_object_status", rowObjectStatus);
149         if ("available".equals(rowObjectStatus)) {
150             payload.put("row_objects", createRowObjects());
151         }
152     }
153     
154     private String getRowObjectStatus() {
155         String unavailableStatus = getUnavailableRowObjectStatus();
156         if (!unavailableStatus.isEmpty()) {
157             return unavailableStatus;
158         }
159         return ROW_OBJECT_LIMIT < executionResult.getRows().size() ? "omitted_large_result" : "available";
160     }
161     
162     private String getUnavailableRowObjectStatus() {
163         if (executionResult.getColumns().isEmpty()) {
164             return executionResult.getRows().isEmpty() ? "" : "unnamed_columns";
165         }
166         Set<String> columnNames = new LinkedHashSet<>(executionResult.getColumns().size());
167         for (SQLExecutionColumnDefinition each : executionResult.getColumns()) {
168             if (null == each.getColumnName() || each.getColumnName().isBlank()) {
169                 return "unnamed_columns";
170             }
171             if (!columnNames.add(each.getColumnName())) {
172                 return "duplicate_column_labels";
173             }
174         }
175         return "";
176     }
177     
178     private List<Map<String, Object>> createRowObjects() {
179         return executionResult.getRows().stream().map(this::createRowObject).toList();
180     }
181     
182     private Map<String, Object> createRowObject(final List<Object> row) {
183         Map<String, Object> result = new LinkedHashMap<>(executionResult.getColumns().size(), 1F);
184         for (int i = 0; i < executionResult.getColumns().size(); i++) {
185             result.put(executionResult.getColumns().get(i).getColumnName(), i < row.size() ? row.get(i) : null);
186         }
187         return result;
188     }
189     
190     private List<Map<String, Object>> createNextActions() {
191         if (SQLExecutionResultKind.RESULT_SET == executionResult.getResultKind() && executionResult.isTruncated()) {
192             return isExecuted()
193                     ? List.of(MCPNextActionUtils.stop(EXECUTED_TRUNCATION_ACTION))
194                     : List.of(MCPNextActionUtils.askUser(
195                             "The result was truncated by max_rows. Ask for a narrower SELECT, stronger WHERE clause, or smaller projection before retrying.", List.of("sql")));
196         }
197         return List.of(MCPNextActionUtils.stop(SQLExecutionResultKind.RESULT_SET == executionResult.getResultKind()
198                 ? "Return the result rows to the user or ask a follow-up question if the user requested more analysis."
199                 : "Report the execution status to the user and stop unless the user asks for another operation."));
200     }
