67      private static final String EXCLUDED_TABLE_NAME = "excluded";

69      private static final ColumnSegmentInfo UNBOUND_COLUMN_INFO = new ColumnSegmentInfo(null, TableSourceType.TEMPORARY_TABLE);

71      private static final Set<SegmentType> COLUMN_ONLY_SEGMENT_TYPES = EnumSet.of(
72              SegmentType.LOCK, SegmentType.SET_ASSIGNMENT_COLUMNS, SegmentType.COPY, SegmentType.INSERT_COLUMNS, SegmentType.DEFINITION_COLUMNS);

74      private static final Map<SegmentType, String> SEGMENT_TYPE_MESSAGES = ImmutableMap.<SegmentType, String>builder()
75              .put(SegmentType.PROJECTION, "field list")
76              .put(SegmentType.JOIN_ON, "on clause")
77              .put(SegmentType.JOIN_USING, "from clause")
78              .put(SegmentType.PREDICATE, "where clause")
79              .put(SegmentType.HAVING, "having clause")
80              .put(SegmentType.ORDER_BY, "order clause")
81              .put(SegmentType.GROUP_BY, "group statement")
82              .put(SegmentType.INSERT_COLUMNS, "field list")
83              .build();

85      private static final String UNKNOWN_SEGMENT_TYPE_MESSAGE = "unknown clause";

87      /**
88       * Bind column segment.
89       *
90       * @param segment column segment
91       * @param parentSegmentType parent segment type
92       * @param binderContext statement binder context
93       * @param tableBinderContexts table binder contexts
94       * @param outerTableBinderContexts outer table binder contexts
95       * @return bound column segment
96       */
97      public static ColumnSegment bind(final ColumnSegment segment, final SegmentType parentSegmentType, final SQLStatementBinderContext binderContext,
98                                       final Multimap<CaseInsensitiveString, TableSegmentBinderContext> tableBinderContexts,
99                                       final Multimap<CaseInsensitiveString, TableSegmentBinderContext> outerTableBinderContexts) {
100         if (isExcludedColumn(segment, parentSegmentType)) {
101             return segment;
102         }
103         ColumnSegment columnSegment = createNestedObjectColumnSegment(segment, parentSegmentType, binderContext, tableBinderContexts, outerTableBinderContexts).orElse(segment);
104         Collection<TableSegmentBinderContext> tableSegmentBinderContexts =
105                 getTableSegmentBinderContexts(columnSegment, parentSegmentType, binderContext, tableBinderContexts, outerTableBinderContexts);
106         ColumnSegmentInfo columnSegmentInfo = getColumnSegmentInfo(columnSegment, parentSegmentType, tableSegmentBinderContexts, outerTableBinderContexts, binderContext);
107         Optional<ColumnSegment> inputColumnSegment = columnSegmentInfo.getInputColumnSegment();
108         if (!inputColumnSegment.isPresent() && isUnparenthesizedFunction(segment, parentSegmentType, binderContext)) {
109             return segment;
110         }
111         ShardingSpherePreconditions.checkState(inputColumnSegment.isPresent() || isSkipColumnBind(tableSegmentBinderContexts, outerTableBinderContexts.values()),
112                 () -> new ColumnNotFoundException(segment.getExpression(), SEGMENT_TYPE_MESSAGES.getOrDefault(parentSegmentType, UNKNOWN_SEGMENT_TYPE_MESSAGE)));
113         ColumnSegment result = copy(columnSegment);
114         inputColumnSegment.ifPresent(optional -> result.setVariable(optional.isVariable()));
115         columnSegment.getOwner().ifPresent(optional -> result.setOwner(bindOwnerTableContext(optional, inputColumnSegment.orElse(null))));
116         result.setColumnBoundInfo(createColumnSegmentBoundInfo(columnSegment, inputColumnSegment.orElse(null), columnSegmentInfo.getTableSourceType()));
117         return result;
118     }
119     
120     private static Optional<ColumnSegment> createNestedObjectColumnSegment(final ColumnSegment segment, final SegmentType parentSegmentType,
121                                                                            final SQLStatementBinderContext binderContext,
122                                                                            final Multimap<CaseInsensitiveString, TableSegmentBinderContext> tableBinderContexts,
123                                                                            final Multimap<CaseInsensitiveString, TableSegmentBinderContext> outerTableBinderContexts) {
124         if (!segment.getOwner().isPresent()) {
125             return Optional.empty();
126         }
127         OwnerSegment owner = segment.getOwner().get();
128         if (!getTableBinderContextByOwner(owner.getIdentifier().getValue(), tableBinderContexts, outerTableBinderContexts, binderContext, parentSegmentType).isEmpty()) {
129             return Optional.empty();
130         }
131         ColumnSegment candidate = createNestedObjectColumnSegment(segment, owner);
132         Collection<TableSegmentBinderContext> candidateTableBinderContexts = owner.getOwner()
133                 .map(optional -> getTableBinderContextByOwner(optional.getIdentifier().getValue(), tableBinderContexts, outerTableBinderContexts, binderContext, parentSegmentType))
134                 .orElseGet(tableBinderContexts::values);
135         return getInputInfoFromTableBinderContexts(candidateTableBinderContexts, candidate, parentSegmentType).getInputColumnSegment().map(optional -> candidate);
136     }
137     
138     private static ColumnSegment createNestedObjectColumnSegment(final ColumnSegment segment, final OwnerSegment owner) {
139         int startIndex = owner.getOwner().map(OwnerSegment::getStartIndex).orElse(owner.getStartIndex());
140         ColumnSegment result = new ColumnSegment(startIndex, segment.getStopIndex(), owner.getIdentifier());
141         result.setNestedObjectAttributes(Collections.singletonList(segment.getIdentifier()));
142         owner.getOwner().ifPresent(result::setOwner);
143         return result;
144     }

146     private static boolean isExcludedColumn(final ColumnSegment segment, final SegmentType parentSegmentType) {
147         return SegmentType.SET_ASSIGNMENT == parentSegmentType && segment.getOwner().isPresent() && EXCLUDED_TABLE_NAME.equalsIgnoreCase(segment.getOwner().get().getIdentifier().getValue());
148     }
149     
150     private static ColumnSegment copy(final ColumnSegment segment) {
151         ColumnSegment result = new ColumnSegment(segment.getStartIndex(), segment.getStopIndex(), segment.getIdentifier());
152         result.setNestedObjectAttributes(segment.getNestedObjectAttributes());
153         result.setVariable(segment.isVariable());
154         segment.getLeftParentheses().ifPresent(result::setLeftParentheses);
155         segment.getRightParentheses().ifPresent(result::setRightParentheses);
156         return result;
157     }

159     private static boolean isUnparenthesizedFunction(final ColumnSegment segment, final SegmentType parentSegmentType, final SQLStatementBinderContext binderContext) {
160         if (QuoteCharacter.NONE != segment.getIdentifier().getQuoteCharacter() || COLUMN_ONLY_SEGMENT_TYPES.contains(parentSegmentType)) {
161             return false;
162         }
163         DialectFunctionOption functionOption = new DatabaseTypeRegistry(binderContext.getSqlStatement().getDatabaseType()).getDialectDatabaseMetaData().getFunctionOption();
164         return functionOption.getUnparenthesizedFunctionNames().contains(segment.getIdentifier().getValue()) || isUnparenthesizedQualifiedFunction(segment, functionOption);
165     }
166     
167     private static boolean isUnparenthesizedQualifiedFunction(final ColumnSegment segment, final DialectFunctionOption functionOption) {
168         if (!segment.getOwner().isPresent() || segment.getOwner().get().getOwner().isPresent()
169                 || QuoteCharacter.NONE != segment.getOwner().get().getIdentifier().getQuoteCharacter()) {
170             return false;
171         }
172         Collection<String> functionNames = functionOption.getUnparenthesizedQualifiedFunctionNames();
173         return !functionNames.isEmpty() && functionNames.contains(segment.getOwner().get().getIdentifier().getValue() + '.' + segment.getIdentifier().getValue());
174     }
175     
176     private static OwnerSegment bindOwnerTableContext(final OwnerSegment owner, final ColumnSegment inputColumnSegment) {
177         IdentifierValue originalDatabase = null == inputColumnSegment ? null : inputColumnSegment.getColumnBoundInfo().getOriginalDatabase();
178         IdentifierValue originalSchema = null == inputColumnSegment ? null : inputColumnSegment.getColumnBoundInfo().getOriginalSchema();
179         if (null != originalDatabase && null != originalSchema) {
180             owner.setTableBoundInfo(new TableSegmentBoundInfo(originalDatabase, originalSchema));
181         }
182         return owner;
183     }

185     private static Collection<TableSegmentBinderContext> getTableSegmentBinderContexts(final ColumnSegment segment, final SegmentType parentSegmentType,
186                                                                                        final SQLStatementBinderContext binderContext,
187                                                                                        final Multimap<CaseInsensitiveString, TableSegmentBinderContext> tableBinderContexts,
188                                                                                        final Multimap<CaseInsensitiveString, TableSegmentBinderContext> outerTableBinderContexts) {
189         if (segment.getOwner().isPresent()) {
190             String owner = segment.getOwner().get().getIdentifier().getValue();
191             return getTableBinderContextByOwner(owner, tableBinderContexts, outerTableBinderContexts, binderContext, parentSegmentType);
192         }
193         if (!binderContext.getJoinTableProjectionSegments().isEmpty() && isNeedUseJoinTableProjectionBind(segment, parentSegmentType, binderContext)) {
194             return Collections.singleton(new SimpleTableSegmentBinderContext(binderContext.getJoinTableProjectionSegments(), TableSourceType.MIXED_TABLE));
195         }
196         return tableBinderContexts.values();
197     }
198     
199     private static Collection<TableSegmentBinderContext> getTableBinderContextByOwner(final String owner, final Multimap<CaseInsensitiveString, TableSegmentBinderContext> tableBinderContexts,
200                                                                                       final Multimap<CaseInsensitiveString, TableSegmentBinderContext> outerTableBinderContexts,
201                                                                                       final SQLStatementBinderContext binderContext, final SegmentType parentSegmentType) {
202         if (null == owner) {
203             return Collections.emptyList();
204         }
205         CaseInsensitiveString caseInsensitiveOwner = CaseInsensitiveString.of(owner);
206         if (tableBinderContexts.containsKey(caseInsensitiveOwner)) {
207             return tableBinderContexts.get(caseInsensitiveOwner);
208         }
209         if (SegmentType.ORDER_BY == parentSegmentType && null != binderContext.getOrderByFromTableBinderContexts()
210                 && binderContext.getOrderByFromTableBinderContexts().containsKey(caseInsensitiveOwner)) {
211             return binderContext.getOrderByFromTableBinderContexts().get(caseInsensitiveOwner);
212         }
213         if (outerTableBinderContexts.containsKey(caseInsensitiveOwner)) {
214             return outerTableBinderContexts.get(caseInsensitiveOwner);
215         }
216         if (binderContext.getExternalTableBinderContexts().containsKey(caseInsensitiveOwner)) {
217             return binderContext.getExternalTableBinderContexts().get(caseInsensitiveOwner);
218         }
219         return Collections.emptyList();
220     }

222     private static boolean isNeedUseJoinTableProjectionBind(final ColumnSegment segment, final SegmentType parentSegmentType, final SQLStatementBinderContext binderContext) {
223         return SegmentType.PROJECTION == parentSegmentType
224                 || SegmentType.PREDICATE == parentSegmentType && binderContext.getUsingColumnNames().contains(segment.getIdentifier().getValue());
225     }

227     private static ColumnSegmentInfo getColumnSegmentInfo(final ColumnSegment segment, final SegmentType parentSegmentType, final Collection<TableSegmentBinderContext> tableBinderContexts,
228                                                           final Multimap<CaseInsensitiveString, TableSegmentBinderContext> outerTableBinderContexts,
229                                                           final SQLStatementBinderContext binderContext) {
230         Optional<ColumnSegment> valueVariableColumnSegment = findInputColumnSegmentByValueVariables(segment, parentSegmentType, binderContext.getSqlStatement().getVariableNames());
231         if (valueVariableColumnSegment.isPresent()) {
232             return new ColumnSegmentInfo(valueVariableColumnSegment.get(), TableSourceType.TEMPORARY_TABLE);
233         }
234         ColumnSegmentInfo result = isModelProjectionColumn(segment, parentSegmentType, binderContext.getModelColumnNames())
235                 ? new ColumnSegmentInfo(findInputColumnSegmentByModelColumns(segment, binderContext.getModelColumnNames()).orElse(null), TableSourceType.TEMPORARY_TABLE)
236                 : getInputInfoFromTableBinderContexts(tableBinderContexts, segment, parentSegmentType);
237         if (isNotFoundInputColumn(result, segment) && SegmentType.ORDER_BY == parentSegmentType && null != binderContext.getOrderByFromTableBinderContexts()) {
238             result = getInputInfoFromTableBinderContexts(binderContext.getOrderByFromTableBinderContexts().values(), segment, parentSegmentType);
239         }
240         if (isNotFoundInputColumn(result, segment)) {
241             ColumnSegment inputColumnSegment = findInputColumnSegmentFromOuterTable(segment, outerTableBinderContexts).orElse(null);
242             result = new ColumnSegmentInfo(inputColumnSegment, null == inputColumnSegment ? TableSourceType.TEMPORARY_TABLE : inputColumnSegment.getColumnBoundInfo().getTableSourceType());
243         }
244         if (isNotFoundInputColumn(result, segment)) {
245             ColumnSegment inputColumnSegment = findInputColumnSegmentFromExternalTables(segment, binderContext.getExternalTableBinderContexts()).orElse(null);
246             result = new ColumnSegmentInfo(inputColumnSegment, null == inputColumnSegment ? TableSourceType.TEMPORARY_TABLE : inputColumnSegment.getColumnBoundInfo().getTableSourceType());
247         }
248         if (isNeedFindInputColumnByVariables(result, segment, binderContext.getSqlStatement().getVariableNames())) {
249             result = new ColumnSegmentInfo(findInputColumnSegmentByVariables(segment, binderContext.getSqlStatement().getVariableNames()).orElse(null), TableSourceType.TEMPORARY_TABLE);
250         }
251         if (isNotFoundInputColumn(result, segment)) {
252             result = new ColumnSegmentInfo(findInputColumnSegmentByPivotColumns(segment, binderContext.getPivotColumnNames()).orElse(null), TableSourceType.TEMPORARY_TABLE);
253         }
254         if (isNotFoundInputColumn(result, segment)) {
255             result = new ColumnSegmentInfo(findInputColumnSegmentByModelColumns(segment, binderContext.getModelColumnNames()).orElse(null), TableSourceType.TEMPORARY_TABLE);
256         }
257         return result;
258     }

260     private static Optional<ColumnSegment> findInputColumnSegmentByValueVariables(final ColumnSegment segment, final SegmentType parentSegmentType,
261                                                                                   final Collection<String> variableNames) {
262         return SegmentType.VALUES == parentSegmentType && !segment.getOwner().isPresent() ? findInputColumnSegmentByVariables(segment, variableNames) : Optional.empty();
263     }

265     private static boolean isModelProjectionColumn(final ColumnSegment segment, final SegmentType parentSegmentType, final Collection<String> modelColumnNames) {
266         return SegmentType.PROJECTION == parentSegmentType && !segment.getOwner().isPresent() && modelColumnNames.contains(segment.getIdentifier().getValue());
267     }

269     private static boolean isNotFoundInputColumn(final ColumnSegmentInfo segmentInfo, final ColumnSegment segment) {
270         return !segmentInfo.getInputColumnSegment().isPresent() && !segment.getOwner().isPresent();
271     }
272     
273     private static boolean isNeedFindInputColumnByVariables(final ColumnSegmentInfo segmentInfo, final ColumnSegment segment, final Collection<String> variableNames) {
274         return !segmentInfo.getInputColumnSegment().isPresent() && (!segment.getOwner().isPresent() || isVariableOwner(segment, variableNames));
275     }

277     private static ColumnSegmentInfo getInputInfoFromTableBinderContexts(final Collection<TableSegmentBinderContext> tableBinderContexts,
278                                                                          final ColumnSegment segment, final SegmentType parentSegmentType) {
279         ColumnSegment inputColumnSegment = null;
280         TableSourceType tableSourceType = TableSourceType.TEMPORARY_TABLE;
281         for (TableSegmentBinderContext each : tableBinderContexts) {
282             Optional<ProjectionSegment> projectionSegment = each.findProjectionSegmentByColumnLabel(segment.getIdentifier().getValue());
283             if (!projectionSegment.isPresent()) {
284                 continue;
285             }
286             if (projectionSegment.get() instanceof ColumnProjectionSegment) {
287                 ShardingSpherePreconditions.checkState(null == inputColumnSegment,
288                         () -> new AmbiguousColumnException(segment.getExpression(), SEGMENT_TYPE_MESSAGES.getOrDefault(parentSegmentType, UNKNOWN_SEGMENT_TYPE_MESSAGE)));
289             }
290             inputColumnSegment = getColumnSegment(projectionSegment.get());
291             tableSourceType = TableSourceType.MIXED_TABLE == each.getTableSourceType() ? getTableSourceTypeFromInputColumn(inputColumnSegment) : each.getTableSourceType();
292             if (each instanceof SimpleTableSegmentBinderContext && ((SimpleTableSegmentBinderContext) each).isFromWithSegment()) {
293                 break;
294             }
295         }
296         return null == inputColumnSegment ? UNBOUND_COLUMN_INFO : new ColumnSegmentInfo(inputColumnSegment, tableSourceType);
297     }
298     
299     private static TableSourceType getTableSourceTypeFromInputColumn(final ColumnSegment inputColumnSegment) {
300         return null == inputColumnSegment ? TableSourceType.TEMPORARY_TABLE : inputColumnSegment.getColumnBoundInfo().getTableSourceType();
301     }
302     
303     private static ColumnSegment getColumnSegment(final ProjectionSegment projectionSegment) {
304         if (projectionSegment instanceof ColumnProjectionSegment) {
305             return ((ColumnProjectionSegment) projectionSegment).getColumn();
306         }
307         if (projectionSegment instanceof ExpressionProjectionSegment) {
308             Collection<ColumnSegment> returnedColumns = ColumnExtractor.extractReturnedColumns(((ExpressionProjectionSegment) projectionSegment).getExpr());
309             if (1 == returnedColumns.size()) {
310                 return returnedColumns.iterator().next();
311             }
312         }
313         if (projectionSegment instanceof SubqueryProjectionSegment && 1 == ((SubqueryProjectionSegment) projectionSegment).getSubquery().getSelect().getProjections().getProjections().size()) {
314             return getColumnSegment(((SubqueryProjectionSegment) projectionSegment).getSubquery().getSelect().getProjections().getProjections().iterator().next());
315         }
316         return new ColumnSegment(0, 0, new IdentifierValue(projectionSegment.getColumnLabel()));
317     }

319     private static Optional<ColumnSegment> findInputColumnSegmentByPivotColumns(final ColumnSegment segment, final Collection<String> pivotColumnNames) {
320         return pivotColumnNames.isEmpty() || !pivotColumnNames.contains(segment.getIdentifier().getValue()) ? Optional.empty() : Optional.of(new ColumnSegment(0, 0, segment.getIdentifier()));
321     }
322     
323     private static Optional<ColumnSegment> findInputColumnSegmentByModelColumns(final ColumnSegment segment, final Collection<String> modelColumnNames) {
324         return modelColumnNames.isEmpty() || !modelColumnNames.contains(segment.getIdentifier().getValue()) ? Optional.empty() : Optional.of(new ColumnSegment(0, 0, segment.getIdentifier()));
325     }

327     private static Optional<ColumnSegment> findInputColumnSegmentFromOuterTable(final ColumnSegment segment,
328                                                                                 final Multimap<CaseInsensitiveString, TableSegmentBinderContext> outerTableBinderContexts) {
329         TableSegmentBinderContext[] tableBinderContexts = outerTableBinderContexts.values().toArray(new TableSegmentBinderContext[outerTableBinderContexts.size()]);
330         for (int i = tableBinderContexts.length - 1; i >= 0; i--) {
331             Optional<ProjectionSegment> result = tableBinderContexts[i].findProjectionSegmentByColumnLabel(segment.getIdentifier().getValue());
332             if (result.isPresent()) {
333                 return Optional.of(createColumnSegment(result.get()));
334             }
335         }
336         return Optional.empty();
337     }
338     
339     private static ColumnSegment createColumnSegment(final ProjectionSegment projectionSegment) {
340         if (projectionSegment instanceof ColumnProjectionSegment) {
341             return ((ColumnProjectionSegment) projectionSegment).getColumn();
342         }
343         if (projectionSegment instanceof ParameterMarkerExpressionSegment) {
344             ParameterMarkerExpressionSegment parameterMarker = (ParameterMarkerExpressionSegment) projectionSegment;
345             ColumnSegment result = new ColumnSegment(0, 0, parameterMarker.getAlias().orElseGet(() -> new IdentifierValue(parameterMarker.getColumnLabel())));
346             if (null != parameterMarker.getBoundInfo()) {
347                 result.setColumnBoundInfo(parameterMarker.getBoundInfo());
348             }
349             return result;
350         }
351         return new ColumnSegment(0, 0, new IdentifierValue(projectionSegment.getColumnLabel()));
352     }
353     
354     private static Optional<ColumnSegment> findInputColumnSegmentFromExternalTables(final ColumnSegment segment,
355                                                                                     final Multimap<CaseInsensitiveString, TableSegmentBinderContext> externalTableBinderContexts) {
356         for (TableSegmentBinderContext each : externalTableBinderContexts.values()) {
357             Optional<ProjectionSegment> result = each.findProjectionSegmentByColumnLabel(segment.getIdentifier().getValue());
358             if (result.isPresent()) {
359                 return Optional.of(createColumnSegment(result.get()));
360             }
361         }
362         return Optional.empty();
363     }

365     private static Optional<ColumnSegment> findInputColumnSegmentByVariables(final ColumnSegment segment, final Collection<String> variableNames) {
366         if (variableNames.isEmpty()) {
367             return Optional.empty();
368         }
369         if (!segment.getOwner().isPresent() && containsVariableName(variableNames, segment.getIdentifier().getValue()) || isVariableOwner(segment, variableNames)) {
370             return Optional.of(createVariableColumnSegment(segment));
371         }
372         return Optional.empty();
373     }
374     
375     private static boolean isVariableOwner(final ColumnSegment segment, final Collection<String> variableNames) {
376         return segment.getOwner().isPresent() && containsVariableName(variableNames, segment.getOwner().get().getIdentifier().getValue());
377     }
378     
379     private static boolean containsVariableName(final Collection<String> variableNames, final String variableName) {
380         for (String each : variableNames) {
381             if (each.equalsIgnoreCase(variableName)) {
382                 return true;
383             }
384         }
385         return false;
386     }
387     
388     private static ColumnSegment createVariableColumnSegment(final ColumnSegment segment) {
389         ColumnSegment result = copy(segment);
390         segment.getOwner().ifPresent(result::setOwner);
391         result.setVariable(true);
392         return result;
393     }
394     
395     private static boolean isSkipColumnBind(final Collection<TableSegmentBinderContext> tableBinderContexts, final Collection<TableSegmentBinderContext> outerBinderContexts) {
396         for (TableSegmentBinderContext each : tableBinderContexts) {
397             if (each instanceof FunctionTableSegmentBinderContext) {
398                 return true;
399             }
400             if (each instanceof SimpleTableSegmentBinderContext) {
401                 SimpleTableSegmentBinderContext binderContext = (SimpleTableSegmentBinderContext) each;
402                 return binderContext.isContainsDBLink() || binderContext.isSkipColumnBind();
403             }
404         }
405         for (TableSegmentBinderContext each : outerBinderContexts) {
406             if (each instanceof FunctionTableSegmentBinderContext) {
407                 return true;
408             }
409             if (each instanceof SimpleTableSegmentBinderContext) {
410                 SimpleTableSegmentBinderContext binderContext = (SimpleTableSegmentBinderContext) each;
411                 return binderContext.isContainsDBLink() || binderContext.isSkipColumnBind();
412             }
413         }
414         return false;
415     }

417     /**
418      * Bind using column segment.
419      *
420      * @param segment using column segment
421      * @param parentSegmentType parent segment type
422      * @param tableBinderContexts table binder contexts
423      * @return bound using column segment
424      */
425     public static ColumnSegment bindUsingColumn(final ColumnSegment segment, final SegmentType parentSegmentType,
426                                                 final Multimap<CaseInsensitiveString, TableSegmentBinderContext> tableBinderContexts) {
427         ColumnSegment result = copy(segment);
428         List<ColumnSegmentInfo> usingColumnSegmentInfos = findUsingColumnSegmentInfos(tableBinderContexts.values(), segment.getIdentifier().getValue());
429         ShardingSpherePreconditions.checkState(usingColumnSegmentInfos.size() >= 2,
430                 () -> new ColumnNotFoundException(segment.getExpression(), SEGMENT_TYPE_MESSAGES.getOrDefault(parentSegmentType, UNKNOWN_SEGMENT_TYPE_MESSAGE)));
431         ColumnSegmentInfo usingColumnInputInfo = usingColumnSegmentInfos.get(0);
432         ColumnSegmentInfo otherUsingColumnInputInfo = usingColumnSegmentInfos.get(1);
433         result.setColumnBoundInfo(createColumnSegmentBoundInfo(segment, usingColumnInputInfo.getInputColumnSegment().orElse(null), usingColumnInputInfo.getTableSourceType()));
434         result.setOtherUsingColumnBoundInfo(createColumnSegmentBoundInfo(segment, otherUsingColumnInputInfo.getInputColumnSegment().orElse(null), otherUsingColumnInputInfo.getTableSourceType()));
435         return result;
436     }
437     
438     private static List<ColumnSegmentInfo> findUsingColumnSegmentInfos(final Collection<TableSegmentBinderContext> tableBinderContexts, final String columnName) {
439         List<ColumnSegmentInfo> result = new ArrayList<>(tableBinderContexts.size());
440         for (TableSegmentBinderContext each : tableBinderContexts) {
441             Optional<ProjectionSegment> projectionSegment = each.findProjectionSegmentByColumnLabel(columnName);
442             if (!projectionSegment.isPresent()) {
443                 continue;
444             }
445             ColumnSegment columnSegment = projectionSegment.get() instanceof ColumnProjectionSegment ? ((ColumnProjectionSegment) projectionSegment.get()).getColumn() : null;
446             result.add(new ColumnSegmentInfo(columnSegment, each.getTableSourceType()));
447         }
448         return result;
449     }
450     
451     /**
452      * Create column segment bound info.
453      *
454      * @param segment column segment
455      * @param inputColumnSegment input column segment
456      * @param tableSourceType table source type
457      * @return created column segment bound info
458      */
459     public static ColumnSegmentBoundInfo createColumnSegmentBoundInfo(final ColumnSegment segment, final ColumnSegment inputColumnSegment, final TableSourceType tableSourceType) {
460         ColumnSegmentBoundInfo inputBoundInfo = null == inputColumnSegment ? null : inputColumnSegment.getColumnBoundInfo();
461         IdentifierValue segmentOriginalTable = segment.getColumnBoundInfo().getOriginalTable();
462         IdentifierValue originalTable = null != inputColumnSegment && Strings.isNullOrEmpty(segmentOriginalTable.getValue()) ? inputBoundInfo.getOriginalTable() : segmentOriginalTable;
463         IdentifierValue originalColumn = null == inputColumnSegment ? segment.getColumnBoundInfo().getOriginalColumn() : inputBoundInfo.getOriginalColumn();
464         TableSegmentBoundInfo tableBoundInfo = null == inputColumnSegment ? new TableSegmentBoundInfo(null, null) : inputBoundInfo.getTableBoundInfo();
465         ColumnSegmentBoundInfo result = new ColumnSegmentBoundInfo(tableBoundInfo, originalTable, originalColumn, tableSourceType);
466         if (null != inputColumnSegment) {
467             inputColumnSegment.getOwner().ifPresent(result::setOwner);
468         }
469         return result;
470     }

473     @Getter
474     private static class ColumnSegmentInfo {
475         
476         private final ColumnSegment inputColumnSegment;
477         
478         private final TableSourceType tableSourceType;
479         
480         Optional<ColumnSegment> getInputColumnSegment() {
481             return Optional.ofNullable(inputColumnSegment);
482         }
483     }
