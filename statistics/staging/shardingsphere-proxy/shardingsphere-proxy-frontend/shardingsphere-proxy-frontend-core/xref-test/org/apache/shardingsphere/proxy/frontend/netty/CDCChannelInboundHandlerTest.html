172     @Test
173     void assertExceptionCaughtWithWrapperClosesChannel(final LogCaptureAssertion logCaptureAssertion) {
174         ChannelHandlerContext context = mock(ChannelHandlerContext.class, RETURNS_DEEP_STUBS);
175         ChannelFuture channelFuture = mock(ChannelFuture.class);
176         when(context.writeAndFlush(any())).thenReturn(channelFuture);
177         handler.exceptionCaught(context, new CDCExceptionWrapper("request-id", new PipelineInvalidParameterException("invalid")));
178         verify(context).writeAndFlush(argThat(argument -> "request-id".equals(((CDCResponse) argument).getRequestId())));
179         verify(channelFuture).addListener(ChannelFutureListener.CLOSE);
180         assertExpectedError(logCaptureAssertion, new PipelineInvalidParameterException("invalid"));
181     }

183     private void assertExpectedError(final LogCaptureAssertion logCaptureAssertion, final Throwable expectedException) {
184         logCaptureAssertion.assertErrorLog(actualException -> {
185             IThrowableProxy actualCause = CDCExceptionWrapper.class.getName().equals(actualException.getClassName()) ? actualException.getCause() : actualException;
186             assertThat(actualCause.getClassName(), is(expectedException.getClass().getName()));
187             assertThat(actualCause.getMessage(), is(expectedException.getMessage()));
188         });
189     }
