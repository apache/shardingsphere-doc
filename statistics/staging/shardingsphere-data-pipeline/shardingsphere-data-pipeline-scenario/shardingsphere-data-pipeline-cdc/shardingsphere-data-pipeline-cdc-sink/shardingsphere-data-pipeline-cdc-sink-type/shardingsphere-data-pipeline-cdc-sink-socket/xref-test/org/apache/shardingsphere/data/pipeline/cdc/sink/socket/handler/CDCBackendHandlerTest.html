128     @Test
129     void assertStreamDataWhenSchemaAvailable() {
130         when(sinkFactory.create(any())).thenReturn(sink);
131         ShardingSphereDatabase database = mockDatabase();
132         mockDialectMetaData(false);
133         Map<String, Set<String>> schemaTableNames = Collections.singletonMap("foo_schema", Collections.singleton("foo_tbl"));
134         when(CDCSchemaTableUtils.parseTableExpressions(eq(database), anyCollection())).thenReturn(schemaTableNames);
135         DataNode fooDataNode = new DataNode("foo_ds", "foo_schema", "foo_tbl");
136         when(PipelineDataNodeUtils.buildTableAndDataNodesMap(database, schemaTableNames)).thenReturn(Collections.singletonMap("foo_tbl", Collections.singletonList(fooDataNode)));
137         when(jobAPI.create(any(StreamDataParameter.class), eq(CDCSinkType.SOCKET), any(Properties.class))).thenReturn("foo_job");
138         when(jobConfigManager.getJobConfiguration("foo_job")).thenReturn(createJobConfiguration());
139         StreamDataRequestBody requestBody = StreamDataRequestBody.newBuilder().setDatabase("foo_db")
140                 .addSourceSchemaTable(SchemaTable.newBuilder().setSchema("foo_schema").setTable("foo_tbl")).build();
141         CDCResponse actualResponse = backendHandler.streamData("foo_req", requestBody, createConnectionContext(), mock());
142         assertThat(actualResponse.getStatus(), is(CDCResponse.Status.SUCCEED));
143         assertThat(actualResponse.getResponseCase(), is(CDCResponse.ResponseCase.STREAM_DATA_RESULT));
144         assertThat(actualResponse.getStreamDataResult().getStreamingId(), is("foo_job"));
145         ArgumentCaptor<StreamDataParameter> parameterCaptor = ArgumentCaptor.forClass(StreamDataParameter.class);
146         verify(jobAPI).create(parameterCaptor.capture(), eq(CDCSinkType.SOCKET), any(Properties.class));
147         assertThat(parameterCaptor.getValue().getSchemaTableNames(), is(Collections.singletonList("foo_schema.foo_tbl")));
148         assertThat(parameterCaptor.getValue().getTableAndDataNodesMap().get("foo_tbl"), is(Collections.singletonList(fooDataNode)));
149     }

208     @Test
209     void assertStartStreamingSuccess() {
210         when(sinkFactory.create(any())).thenReturn(sink);
211         CDCJobConfiguration jobConfig = createJobConfiguration();
212         when(jobConfigManager.getJobConfiguration("foo_job")).thenReturn(jobConfig);
213         ShardingSphereDatabase database = mock(ShardingSphereDatabase.class);
214         mockProxyContext(database);
215         Channel channel = mock(Channel.class);
216         CDCConnectionContext connectionContext = createConnectionContext();
217         try (MockedStatic<PipelineJobRegistry> jobRegistryMocked = mockStatic(PipelineJobRegistry.class)) {
218             jobRegistryMocked.when(() -> PipelineJobRegistry.stop("foo_job")).thenThrow(new IllegalStateException("stop should be owned by CDCJobAPI"));
219             backendHandler.startStreaming("foo_job", connectionContext, channel);
220             ArgumentCaptor<PipelineSink> sinkCaptor = ArgumentCaptor.forClass(PipelineSink.class);
221             verify(jobAPI).start(eq("foo_job"), sinkCaptor.capture());
222             assertThat(sinkCaptor.getValue(), sameInstance(sink));
223             ArgumentCaptor<CDCSocketSinkCreationContext> contextCaptor = ArgumentCaptor.forClass(CDCSocketSinkCreationContext.class);
224             verify(sinkFactory).create(contextCaptor.capture());
225             assertThat(contextCaptor.getValue().getChannel(), sameInstance(channel));
226             assertThat(contextCaptor.getValue().getDatabase(), sameInstance(database));
227             assertThat(contextCaptor.getValue().getSchemaTableNames(), sameInstance(jobConfig.getSchemaTableNames()));
228             assertThat(contextCaptor.getValue().getProperties(), sameInstance(jobConfig.getSinkConfig().getProps()));
229             assertThat(connectionContext.getJobId(), is("foo_job"));
230             jobRegistryMocked.verifyNoInteractions();
231         }
232     }

281     @Test
282     void assertStopStreamingWhenPreviousConnectionClosesAfterReconnect() {
283         CDCJobConfiguration jobConfig = createJobConfiguration();
284         when(jobConfigManager.getJobConfiguration("foo_job")).thenReturn(jobConfig);
285         ShardingSphereDatabase database = mock(ShardingSphereDatabase.class);
286         mockProxyContext(database);
287         Channel previousChannel = mockChannel(DefaultChannelId.newInstance());
288         Channel currentChannel = mockChannel(DefaultChannelId.newInstance());
289         PipelineCDCSocketSink previousSink = new PipelineCDCSocketSink(previousChannel, database, jobConfig.getSchemaTableNames());
290         PipelineCDCSocketSink currentSink = new PipelineCDCSocketSink(currentChannel, database, jobConfig.getSchemaTableNames());
291         when(sinkFactory.create(any())).thenReturn(previousSink, currentSink);
292         CDCJob job = mock(CDCJob.class);
293         doAnswer(invocation -> {
294             when(job.getSink()).thenReturn(invocation.getArgument(1));
295             return null;
296         }).when(jobAPI).start(eq("foo_job"), any(PipelineSink.class));
297         try (MockedStatic<PipelineJobRegistry> jobRegistryMocked = mockStatic(PipelineJobRegistry.class)) {
298             jobRegistryMocked.when(() -> PipelineJobRegistry.get("foo_job")).thenReturn(job);
299             backendHandler.startStreaming("foo_job", createConnectionContext(), previousChannel);
300             backendHandler.startStreaming("foo_job", createConnectionContext(), currentChannel);
301             backendHandler.stopStreaming("foo_job", previousChannel.id());
302             assertThat(job.getSink(), sameInstance(currentSink));
303             jobRegistryMocked.verify(() -> PipelineJobRegistry.stop("foo_job"), never());
304             verify(jobAPI, never()).disable("foo_job");
305         }
306     }
