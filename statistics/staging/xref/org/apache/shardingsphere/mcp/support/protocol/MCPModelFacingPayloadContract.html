18  package org.apache.shardingsphere.mcp.support.protocol;
19  
20  import lombok.AccessLevel;
21  import lombok.NoArgsConstructor;
22  
23  import java.util.Collection;
24  import java.util.Collections;
25  import java.util.LinkedHashSet;
26  import java.util.List;
27  import java.util.Map;
28  import java.util.Set;
29  
30  /**
31   * MCP model-facing payload contract.
32   */
33  @NoArgsConstructor(access = AccessLevel.PRIVATE)
34  public final class MCPModelFacingPayloadContract {
35      
36      private static final Collection<String> REMOVED_MODEL_FACING_FIELDS = Set.of(
37              "target_tool", "target_resource", "required_arguments", "action_kind", "suggested_next_tool", "suggested_next_tools", "recommended_next_tool",
38              "recommended_recovery", "suggested_next_action", "approved_by_user", "requires_user_approval", "approval_required", "user_overrides",
39              "structured_intent_evidence", "field_semantics", "reasoning_notes");
40      
41      private static final Map<String, Collection<String>> NEXT_ACTION_REQUIRED_FIELDS = Map.of(
42              "resource_read", List.of("order", "type", "title", "resource_uri"),
43              "tool_call", List.of("order", "type", "title", "tool_name", "arguments"),
44              "completion", List.of("order", "type", "title", "ref", "argument"),
45              "ask_user", List.of("order", "type", "title", "question"),
46              "terminal", List.of("order", "type", "title"));
47      
48      private static final Map<String, Collection<String>> NEXT_ACTION_ALLOWED_FIELDS = Map.of(
49              "resource_read", List.of("order", "type", "title", "resource_uri", "reason", "depends_on"),
50              "tool_call", List.of("order", "type", "title", "tool_name", "arguments", "reason", "depends_on"),
51              "completion", List.of("order", "type", "title", "ref", "argument", "context", "missing_context_arguments", "resume_ref", "resume_arguments", "reason", "depends_on"),
52              "ask_user", List.of("order", "type", "title", "question", "required_inputs", "depends_on"),
53              "terminal", List.of("order", "type", "title", "reason", "depends_on"));
54      
55      private static final Collection<String> NEXT_ACTION_SCHEMA_ALLOWED_FIELDS = createNextActionSchemaAllowedFields();
56      
57      private static final Collection<String> MODEL_CRITICAL_FIELD_NAMES = List.of(
58              MCPPayloadFieldNames.SUMMARY, MCPPayloadFieldNames.NEXT_ACTIONS, MCPPayloadFieldNames.RESOURCES_TO_READ, MCPPayloadFieldNames.RESOURCE,
59              MCPPayloadFieldNames.SELF_RESOURCE, MCPPayloadFieldNames.PARENT_RESOURCE, MCPPayloadFieldNames.NEXT_RESOURCES, "manual_artifact_summary",
60              "empty_state", "ambiguity_state", MCPPayloadFieldNames.RECOVERY, "recovery_guidance", "remediation");
61      
62      private static Collection<String> createNextActionSchemaAllowedFields() {
63          Set<String> result = new LinkedHashSet<>();
64          for (Collection<String> each : NEXT_ACTION_ALLOWED_FIELDS.values()) {
65              result.addAll(each);
66          }
67          return Collections.unmodifiableSet(result);
68      }
69      
70      /**
71       * Check whether a field name is a removed model-facing alias.
72       *
73       * @param fieldName field name
74       * @return true if removed, otherwise false
75       */
76      public static boolean isRemovedFieldName(final String fieldName) {
77          return REMOVED_MODEL_FACING_FIELDS.contains(fieldName);
78      }
79      
80      /**
81       * Get required fields for a canonical next_actions item type.
82       *
83       * @param actionType action type
84       * @return required field names
85       */
86      public static Collection<String> getNextActionRequiredFields(final String actionType) {
87          return NEXT_ACTION_REQUIRED_FIELDS.getOrDefault(actionType, List.of());
88      }
89      
90      /**
91       * Get allowed fields for a canonical next_actions item type.
92       *
93       * @param actionType action type
94       * @return allowed field names
95       */
96      public static Collection<String> getNextActionAllowedFields(final String actionType) {
97          return NEXT_ACTION_ALLOWED_FIELDS.getOrDefault(actionType, List.of());
98      }
99      
100     /**
101      * Get fields allowed by the next_actions item output schema.
102      *
103      * @return allowed schema field names
104      */
105     public static Collection<String> getNextActionSchemaAllowedFields() {
106         return NEXT_ACTION_SCHEMA_ALLOWED_FIELDS;
107     }
108     
109     /**
110      * Get model-critical field names that must stay visible to MCP clients and E2E model prompts.
111      *
112      * @return model-critical field names
113      */
114     public static Collection<String> getModelCriticalFieldNames() {
115         return MODEL_CRITICAL_FIELD_NAMES;
116     }
117 }
