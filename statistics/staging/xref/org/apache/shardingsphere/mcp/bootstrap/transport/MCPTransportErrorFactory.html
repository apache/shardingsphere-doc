18  package org.apache.shardingsphere.mcp.bootstrap.transport;
19  
20  import io.modelcontextprotocol.spec.McpError;
21  import io.modelcontextprotocol.spec.McpSchema;
22  import lombok.AccessLevel;
23  import lombok.NoArgsConstructor;
24  import lombok.extern.slf4j.Slf4j;
25  import org.apache.shardingsphere.mcp.api.exception.MCPInvalidRequestException;
26  import org.apache.shardingsphere.mcp.api.exception.MCPNotFoundException;
27  import org.apache.shardingsphere.mcp.api.exception.MCPUnsupportedException;
28  import org.apache.shardingsphere.mcp.api.exception.ShardingSphereMCPException;
29  import org.apache.shardingsphere.mcp.bootstrap.transport.server.http.validator.MCPTransportSecurityException;
30  import org.apache.shardingsphere.mcp.core.protocol.error.MCPErrorConverter;
31  import org.apache.shardingsphere.mcp.core.protocol.error.MCPErrorPayload;
32  import org.apache.shardingsphere.mcp.support.database.metadata.jdbc.RuntimeDatabaseConnectionException;
33  
34  import java.sql.SQLException;
35  import java.util.LinkedHashMap;
36  import java.util.List;
37  import java.util.Map;
38  
39  /**
40   * MCP transport error factory.
41   */
42  @NoArgsConstructor(access = AccessLevel.PRIVATE)
43  @Slf4j
44  public final class MCPTransportErrorFactory {
45      
46      /**
47       * Create MCP transport error.
48       *
49       * @param cause error cause
50       * @return MCP transport error
51       */
52      public static McpError createError(final Throwable cause) {
53          if (cause instanceof MCPTransportSecurityException ex) {
54              return createProtocolError(new MCPErrorPayload(cause.getMessage(), createTransportSecurityRecovery(ex)), getProtocolErrorCode(cause));
55          }
56          if (!isApplicationError(cause)) {
57              return createInternalError(cause);
58          }
59          return createProtocolError(MCPErrorConverter.convert(cause), getProtocolErrorCode(cause));
60      }
61      
62      /**
63       * Create MCP resource error.
64       *
65       * @param cause error cause
66       * @return MCP resource error
67       */
68      public static McpError createResourceError(final Throwable cause) {
69          if (!isApplicationError(cause)) {
70              return createInternalError(cause);
71          }
72          return createProtocolError(MCPErrorConverter.convert(cause), getResourceProtocolErrorCode(cause));
73      }
74      
75      private static McpError createInternalError(final Throwable cause) {
76          MCPErrorPayload errorPayload = new MCPErrorPayload("Service is temporarily unavailable.");
77          Map<String, Object> payload = errorPayload.toPayload();
78          log.error("Unexpected MCP request failure, error ID: {}.", payload.get("error_id"), cause);
79          return createProtocolError(errorPayload, payload, McpSchema.ErrorCodes.INTERNAL_ERROR);
80      }
81      
82      private static Map<String, Object> createTransportSecurityRecovery(final MCPTransportSecurityException cause) {
83          Map<String, Object> result = new LinkedHashMap<>(5, 1F);
84          result.put("response_mode", "recovery");
85          result.put("recovery_category", "transport_security");
86          result.put("recoverable", false);
87          result.put("category", cause.getCategory());
88          result.put("model_action", createTransportSecurityModelAction(cause.getCategory()));
89          result.put("next_actions", List.of(Map.of("order", 1, "type", "terminal", "title", "Stop", "reason", "Transport security policy rejected the request.")));
90          return result;
91      }
92      
93      private static String createTransportSecurityModelAction(final String category) {
94          if (MCPTransportSecurityException.CATEGORY_ORIGIN_NOT_ALLOWED.equals(category)) {
95              return "Use an allowed loopback Origin or omit Origin for trusted local MCP HTTP requests.";
96          }
97          return "Start a new MCP session or align trusted session attribution headers before retrying.";
98      }
99      
100     private static int getProtocolErrorCode(final Throwable cause) {
101         if (cause instanceof MCPTransportSecurityException || cause instanceof MCPInvalidRequestException || cause instanceof MCPUnsupportedException) {
102             return McpSchema.ErrorCodes.INVALID_PARAMS;
103         }
104         return McpSchema.ErrorCodes.INTERNAL_ERROR;
105     }
106     
107     private static int getResourceProtocolErrorCode(final Throwable cause) {
108         if (cause instanceof MCPNotFoundException) {
109             return McpSchema.ErrorCodes.RESOURCE_NOT_FOUND;
110         }
111         if (cause instanceof MCPInvalidRequestException || cause instanceof MCPUnsupportedException) {
112             return McpSchema.ErrorCodes.INVALID_PARAMS;
113         }
114         return McpSchema.ErrorCodes.INTERNAL_ERROR;
115     }
116     
117     private static boolean isApplicationError(final Throwable cause) {
118         return cause instanceof ShardingSphereMCPException || cause instanceof RuntimeDatabaseConnectionException || cause instanceof SQLException;
119     }
120     
121     private static McpError createProtocolError(final MCPErrorPayload errorPayload, final int code) {
122         return createProtocolError(errorPayload, errorPayload.toPayload(), code);
123     }
124     
125     private static McpError createProtocolError(final MCPErrorPayload errorPayload, final Map<String, Object> payload, final int code) {
126         return McpError.builder(code).message(errorPayload.getMessage()).data(payload).build();
127     }
128 }
