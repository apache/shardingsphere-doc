102     void assertBuildJobItemContextWithTransformedDataSourceConfigurations() {
103         try (
104                 MockedStatic<DatabaseTypeFactory> databaseTypeFactory = mockStatic(DatabaseTypeFactory.class);
105                 MockedStatic<PipelineAPIFactory> pipelineAPIFactory = mockStatic(PipelineAPIFactory.class);
106                 MockedStatic<PipelineContextManager> pipelineContextManager = mockStatic(PipelineContextManager.class)) {
107             mockDatabaseTypeFactory(databaseTypeFactory);
108             mockGovernanceFacade(pipelineAPIFactory, createSourceDataSourceYaml(10, 20));
109             mockProxyContext(pipelineContextManager);
110             MigrationJobItemContext actual = new MigrationJobExecutorCallback().buildJobItemContext(createJobConfiguration(), 0, null, mockProcessContext(),

112             MigrationTaskConfiguration actualTaskConfig = actual.getTaskConfig();
113             assertThat(getMaxPoolSize((StandardPipelineDataSourceConfiguration) actualTaskConfig.getDumperContext().getCommonContext().getDataSourceConfig()), is("10"));
114             ImporterConfiguration actualImporterConfig = actualTaskConfig.getImporterConfig();
115             assertThat(getRootDataSourceMaxPoolSize((ShardingSpherePipelineDataSourceConfiguration) actualImporterConfig.getDataSourceConfig()), is("30"));
116             CreateTableConfiguration actualCreateTableConfig = actualTaskConfig.getCreateTableConfigurations().iterator().next();
117             assertThat(getMaxPoolSize((StandardPipelineDataSourceConfiguration) actualCreateTableConfig.getSourceDataSourceConfig()), is("10"));
118             assertThat(getRootDataSourceMaxPoolSize((ShardingSpherePipelineDataSourceConfiguration) actualCreateTableConfig.getTargetDataSourceConfig()), is("30"));
119         }
120     }

123     @Test
124     void assertBuildJobItemContextWithRuleConfigurations() {
125         RuleConfiguration ruleConfig = mock(RuleConfiguration.class);
126         PipelineRequiredColumnsExtractor extractor = mock(PipelineRequiredColumnsExtractor.class);
127         Map<ShardingSphereIdentifier, Collection<String>> requiredColumns = Collections.singletonMap(new ShardingSphereIdentifier("t_order"), Collections.singleton("id"));
128         when(extractor.getTableAndRequiredColumnsMap(eq(ruleConfig), anyCollection())).thenReturn(requiredColumns);
129         AtomicReference<Collection<RuleConfiguration>> capturedRuleConfigs = new AtomicReference<>();
130         try (
131                 MockedStatic<OrderedSPILoader> orderedSPILoader = mockStatic(OrderedSPILoader.class);
132                 MockedStatic<DatabaseTypeFactory> databaseTypeFactory = mockStatic(DatabaseTypeFactory.class);
133                 MockedStatic<PipelineAPIFactory> pipelineAPIFactory = mockStatic(PipelineAPIFactory.class);
134                 MockedStatic<PipelineContextManager> pipelineContextManager = mockStatic(PipelineContextManager.class)) {
135             orderedSPILoader.when(() -> OrderedSPILoader.getServices(eq(PipelineRequiredColumnsExtractor.class), anyCollection())).thenAnswer(invocation -> {
136                 capturedRuleConfigs.set(invocation.getArgument(1));
137                 return Collections.singletonMap(ruleConfig, extractor);
138             });
139             mockDatabaseTypeFactory(databaseTypeFactory);
140             MigrationJobConfiguration jobConfig = createJobConfiguration();
141             Collection<RuleConfiguration> ruleConfigs = ((ShardingSpherePipelineDataSourceConfiguration) jobConfig.getTarget()).getRuleConfigurations();
142             ruleConfigs.add(ruleConfig);
143             mockGovernanceFacade(pipelineAPIFactory, createSourceDataSourceYaml(10, 20));
144             mockProxyContext(pipelineContextManager);
145             MigrationJobItemContext actual = new MigrationJobExecutorCallback().buildJobItemContext(
146                     jobConfig, 0, null, mockProcessContext(), mock(PipelineDataSourceManager.class));
147             assertThat(capturedRuleConfigs.get(), sameInstance(ruleConfigs));
148             assertThat(actual.getTaskConfig().getImporterConfig().getShardingColumns("t_order"), is(Collections.singleton("id")));
149         }
150     }

177     void assertBuildJobItemContextWithPostgreSQLTargetIdentifiers() {
178         try (
179                 MockedConstruction<DatabaseTypeRegistry> ignored = mockConstruction(DatabaseTypeRegistry.class, withSettings().defaultAnswer(RETURNS_DEEP_STUBS),
180                         (mock, context) -> when(mock.getDialectDatabaseMetaData().getSchemaOption().isSchemaAvailable()).thenReturn(true));

182                 MockedStatic<PipelineAPIFactory> pipelineAPIFactory = mockStatic(PipelineAPIFactory.class);

193                     is(Collections.singletonList(new QualifiedTable("upper_schema", "t_order"))));

197     @Test
198     void assertBuildJobItemContextWithCaseSensitiveMySQLTargetIdentifier() {
199         PipelineDataSource targetDataSource = mock(PipelineDataSource.class);
200         PipelineDataSourceManager dataSourceManager = mock(PipelineDataSourceManager.class);
201         when(dataSourceManager.getDataSource(any())).thenReturn(targetDataSource);
202         try (
203                 MockedStatic<DatabaseTypeFactory> databaseTypeFactory = mockStatic(DatabaseTypeFactory.class);
204                 MockedStatic<PipelineAPIFactory> pipelineAPIFactory = mockStatic(PipelineAPIFactory.class);
205                 MockedStatic<PipelineContextManager> pipelineContextManager = mockStatic(PipelineContextManager.class);
206                 MockedStatic<IdentifierNormalizeEngine> normalizeEngine = mockStatic(IdentifierNormalizeEngine.class, withSettings().defaultAnswer(CALLS_REAL_METHODS))) {
207             mockDatabaseTypeFactory(databaseTypeFactory);
208             mockGovernanceFacade(pipelineAPIFactory, createSourceDataSourceYaml(10, 20));
209             mockProxyContext(pipelineContextManager);
210             normalizeEngine.when(() -> IdentifierNormalizeEngine.resolvePolicy(any(DatabaseType.class), isNull(), eq(IdentifierScope.TABLE)))
211                     .thenReturn(IdentifierCasePolicyFactory.newLowerCaseInsensitivePolicySet().getPolicy(IdentifierScope.TABLE));
212             normalizeEngine.when(() -> IdentifierNormalizeEngine.resolvePolicy(any(DatabaseType.class), same(targetDataSource), eq(IdentifierScope.TABLE)))
213                     .thenReturn(IdentifierCasePolicyFactory.newSensitivePolicySet().getPolicy(IdentifierScope.TABLE));
214             MigrationJobItemContext actual = new MigrationJobExecutorCallback().buildJobItemContext(
215                     createMySQLJobConfiguration(), 0, null, mockProcessContext(), dataSourceManager);
216             CreateTableConfiguration actualCreateTableConfig = actual.getTaskConfig().getCreateTableConfigurations().iterator().next();
217             assertThat(actualCreateTableConfig.getTargetName(), is(new QualifiedTable(null, "T_ORDER")));
218             normalizeEngine.verify(() -> IdentifierNormalizeEngine.resolvePolicy(any(DatabaseType.class), same(targetDataSource), eq(IdentifierScope.TABLE)));
219         }
220     }

319     private String getMaxPoolSize(final StandardPipelineDataSourceConfiguration dataSourceConfig) {
320         DataSourcePoolProperties poolProps = (DataSourcePoolProperties) dataSourceConfig.getDataSourceConfiguration();
321         return String.valueOf(poolProps.getPoolPropertySynonyms().getStandardProperties().get("maxPoolSize"));
322     }

324     private String getRootDataSourceMaxPoolSize(final ShardingSpherePipelineDataSourceConfiguration dataSourceConfig) {
325         Collection<DataSourcePoolProperties> dataSources = dataSourceConfig.getDataSourcePoolPropertiesMap().values();
326         return String.valueOf(dataSources.iterator().next().getPoolPropertySynonyms().getStandardProperties().get("maxPoolSize"));
327     }
