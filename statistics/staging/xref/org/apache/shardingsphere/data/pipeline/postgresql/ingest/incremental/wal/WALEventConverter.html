48  /**
49   * WAL event converter.
50   */
51  public final class WALEventConverter {
52      
53      private final IncrementalDumperContext dumperContext;
54      
55      private final PipelineTableMetaDataLoader metaDataLoader;
56      
57      private final IdentifierCasePolicy schemaIdentifierCasePolicy;
58      
59      private final IdentifierCasePolicy tableIdentifierCasePolicy;
60      
61      private final Map<String, ShardingSphereIdentifier> quotedTableNameMap;
62      
63      private final Map<String, ShardingSphereIdentifier> unquotedTableNameMap;
64      
65      /**
66       * Create a WAL event converter.
67       *
68       * @param dumperContext incremental dumper context
69       * @param metaDataLoader pipeline table metadata loader
70       */
71      public WALEventConverter(final IncrementalDumperContext dumperContext, final PipelineTableMetaDataLoader metaDataLoader) {
72          this.dumperContext = dumperContext;
73          this.metaDataLoader = metaDataLoader;
74          schemaIdentifierCasePolicy = IdentifierNormalizeEngine.resolvePolicy(
75                  dumperContext.getCommonContext().getDataSourceConfig().getDatabaseType(), null, IdentifierScope.SCHEMA);
76          tableIdentifierCasePolicy = IdentifierNormalizeEngine.resolvePolicy(
77                  dumperContext.getCommonContext().getDataSourceConfig().getDatabaseType(), null, IdentifierScope.TABLE);
78          Map<ShardingSphereIdentifier, ShardingSphereIdentifier> tableNameMap = dumperContext.getCommonContext().getTableNameMapper().getTableNameMap();
79          quotedTableNameMap = new HashMap<>(tableNameMap.size(), 1F);
80          unquotedTableNameMap = new HashMap<>(tableNameMap.size(), 1F);
81          for (Entry<ShardingSphereIdentifier, ShardingSphereIdentifier> entry : tableNameMap.entrySet()) {
82              String actualTableName = entry.getKey().getValue();
83              quotedTableNameMap.put(getLookupKey(actualTableName, QuoteCharacter.QUOTE), entry.getValue());
84              if (tableIdentifierCasePolicy.matches(actualTableName, actualTableName, QuoteCharacter.NONE)) {
85                  unquotedTableNameMap.put(getLookupKey(actualTableName, QuoteCharacter.NONE), entry.getValue());
86              }
87          }
88      }
89      
90      /**
91       * Convert WAL event to {@code Record}.
92       *
93       * @param event WAL event
94       * @return record
95       * @throws UnsupportedSQLOperationException unsupported SQL operation exception
96       */
97      public Record convert(final AbstractWALEvent event) {
98          if (!(event instanceof AbstractRowEvent)) {
99              return createPlaceholderRecord(event);
100         }
101         AbstractRowEvent rowEvent = (AbstractRowEvent) event;
102         QuoteCharacter tableQuoteCharacter = QuoteCharacter.getQuoteCharacter(rowEvent.getTableName());
103         String actualTableName = tableQuoteCharacter.unwrap(rowEvent.getTableName());
104         ShardingSphereIdentifier logicTableName = getLogicTableName(actualTableName, tableQuoteCharacter);
105         if (null == logicTableName) {
106             return createPlaceholderRecord(event);
107         }
108         String expectedSchemaName = dumperContext.getCommonContext().getTableAndSchemaNameMapper().getSchemaName(logicTableName);
109         if (isDifferentSchemaName(rowEvent, expectedSchemaName)) {
110             return createPlaceholderRecord(event);
111         }
112         PipelineTableMetaData tableMetaData = metaDataLoader.getTableMetaData(expectedSchemaName, actualTableName);
113         if (event instanceof WriteRowEvent writeRowEvent) {
114             return handleWriteRowEvent(writeRowEvent, tableMetaData, logicTableName);
115         }
116         if (event instanceof UpdateRowEvent updateRowEvent) {
117             return handleUpdateRowEvent(updateRowEvent, tableMetaData, logicTableName);
118         }
119         if (event instanceof DeleteRowEvent deleteRowEvent) {
120             return handleDeleteRowEvent(deleteRowEvent, tableMetaData, logicTableName);
121         }
122         throw new UnsupportedSQLOperationException("");
123     }
124     
125     private ShardingSphereIdentifier getLogicTableName(final String actualTableName, final QuoteCharacter quoteCharacter) {
126         Map<String, ShardingSphereIdentifier> tableNameMap = QuoteCharacter.NONE == quoteCharacter ? unquotedTableNameMap : quotedTableNameMap;
127         return tableNameMap.get(getLookupKey(actualTableName, quoteCharacter));
128     }
129     
130     private String getLookupKey(final String identifier, final QuoteCharacter quoteCharacter) {
131         return LookupMode.EXACT == tableIdentifierCasePolicy.getLookupMode(quoteCharacter) ? identifier : tableIdentifierCasePolicy.normalizeForLookup(identifier);
132     }
133     
134     private boolean isDifferentSchemaName(final AbstractRowEvent event, final String expectedSchemaName) {
135         if (!isConcreteSchemaName(expectedSchemaName)) {
136             return false;
137         }
138         String actualSchemaName = event.getSchemaName();
139         if (!isConcreteSchemaName(actualSchemaName)) {
140             return false;
141         }
142         // The mapper stores a canonical schema name, while the WAL decoders preserve the event identifier's quote characters.
143         QuoteCharacter quoteCharacter = QuoteCharacter.getQuoteCharacter(actualSchemaName);
144         return !schemaIdentifierCasePolicy.matches(expectedSchemaName, quoteCharacter.unwrap(actualSchemaName), quoteCharacter);
145     }
146     
147     private static boolean isConcreteSchemaName(final String schemaName) {
148         return null != schemaName && !schemaName.isEmpty() && !"*".equals(schemaName);
149     }
150     
151     private PlaceholderRecord createPlaceholderRecord(final AbstractWALEvent event) {
152         return new PlaceholderRecord(new WALPosition(event.getLogSequenceNumber()));
153     }
154     
155     private DataRecord handleWriteRowEvent(final WriteRowEvent writeRowEvent, final PipelineTableMetaData tableMetaData, final ShardingSphereIdentifier logicTableName) {
156         DataRecord result = createDataRecord(PipelineSQLOperationType.INSERT, writeRowEvent, logicTableName, writeRowEvent.getAfterRow().size());
157         putColumnsIntoDataRecord(result, tableMetaData, writeRowEvent.getAfterRow());
158         return result;
159     }
160     
161     private DataRecord handleUpdateRowEvent(final UpdateRowEvent updateRowEvent, final PipelineTableMetaData tableMetaData, final ShardingSphereIdentifier logicTableName) {
162         DataRecord result = createDataRecord(PipelineSQLOperationType.UPDATE, updateRowEvent, logicTableName, updateRowEvent.getAfterRow().size());
163         putColumnsIntoDataRecord(result, tableMetaData, updateRowEvent.getAfterRow());
164         return result;
165     }
166     
167     private DataRecord handleDeleteRowEvent(final DeleteRowEvent event, final PipelineTableMetaData tableMetaData, final ShardingSphereIdentifier logicTableName) {
168         // TODO completion columns
169         DataRecord result = createDataRecord(PipelineSQLOperationType.DELETE, event, logicTableName, event.getPrimaryKeys().size());
170         // TODO Unique key may be a column within unique index
171         List<String> primaryKeyColumns = tableMetaData.getPrimaryKeyColumns();
172         for (int i = 0; i < event.getPrimaryKeys().size(); i++) {
173             result.addColumn(new NormalColumn(primaryKeyColumns.get(i), event.getPrimaryKeys().get(i), null, true, true));
174         }
175         return result;
176     }
177     
178     private DataRecord createDataRecord(final PipelineSQLOperationType type, final AbstractRowEvent rowsEvent, final ShardingSphereIdentifier logicTableName, final int columnCount) {
179         DataRecord result = new DataRecord(type, rowsEvent.getSchemaName(), logicTableName.toString(), new WALPosition(rowsEvent.getLogSequenceNumber()), columnCount);
180         result.setActualTableName(rowsEvent.getTableName());
181         result.setCsn(rowsEvent.getCsn());
182         return result;
183     }
184     
185     private void putColumnsIntoDataRecord(final DataRecord dataRecord, final PipelineTableMetaData tableMetaData, final List<Object> values) {
186         for (int i = 0, count = values.size(); i < count; i++) {
187             PipelineColumnMetaData columnMetaData = tableMetaData.getColumnMetaData(i + 1);
188             boolean isUniqueKey = columnMetaData.isUniqueKey();
189             Object uniqueKeyOldValue = isUniqueKey && PipelineSQLOperationType.UPDATE == dataRecord.getType() ? values.get(i) : null;
190             Column column = new NormalColumn(columnMetaData.getName(), uniqueKeyOldValue, values.get(i), true, isUniqueKey);
191             dataRecord.addColumn(column);
192         }
193     }
