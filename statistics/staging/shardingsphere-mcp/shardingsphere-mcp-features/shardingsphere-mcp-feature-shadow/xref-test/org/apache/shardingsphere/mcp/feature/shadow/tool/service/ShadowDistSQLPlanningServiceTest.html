18  package org.apache.shardingsphere.mcp.feature.shadow.tool.service;
19  
20  import org.apache.shardingsphere.mcp.feature.shadow.tool.model.ShadowDefaultAlgorithmWorkflowRequest;
21  import org.apache.shardingsphere.mcp.feature.shadow.tool.model.ShadowRuleWorkflowRequest;
22  import org.apache.shardingsphere.mcp.support.workflow.model.RuleArtifact;
23  import org.junit.jupiter.api.Test;
24  
25  import java.util.Map;
26  
27  import static org.hamcrest.MatcherAssert.assertThat;
28  import static org.hamcrest.Matchers.is;
29  import static org.junit.jupiter.api.Assertions.assertFalse;
30  import static org.junit.jupiter.api.Assertions.assertTrue;
31  
32  class ShadowDistSQLPlanningServiceTest {
33      
34      @Test
35      void assertPlanCreateRule() {
36          RuleArtifact actual = new ShadowDistSQLPlanningService().planCreateRule(createRuleRequest());
37          assertThat(actual.getSql(), is("CREATE SHADOW RULE `shadow_rule`(SOURCE=`demo_ds`, SHADOW=`demo_ds_shadow`, "
38                  + "`t_order`(TYPE(NAME='value_match', PROPERTIES('column'='user_id', 'operation'='insert', 'value'='1'))))"));
39          assertNoForbiddenArtifacts(actual.getSql());
40      }
41      
42      @Test
43      void assertPlanCreateRuleFormatsReservedIdentifiers() {
44          ShadowRuleWorkflowRequest request = createRuleRequest();
45          request.setRuleName("type");
46          request.setSourceStorageUnit("from");
47          request.setShadowStorageUnit("table");
48          request.setTableName("order");
49          RuleArtifact actual = new ShadowDistSQLPlanningService().planCreateRule(request);
50          assertThat(actual.getSql(), is("CREATE SHADOW RULE `type`(SOURCE=`from`, SHADOW=`table`, "
51                  + "`order`(TYPE(NAME='value_match', PROPERTIES('column'='user_id', 'operation'='insert', 'value'='1'))))"));
52      }
53      
54      @Test
55      void assertPlanAlterRule() {
56          assertTrue(new ShadowDistSQLPlanningService().planAlterRule(createRuleRequest()).getSql().startsWith("ALTER SHADOW RULE `shadow_rule`"));
57      }
58      
59      @Test
60      void assertPlanDropRule() {
61          assertThat(new ShadowDistSQLPlanningService().planDropRule("shadow_rule").getSql(), is("DROP SHADOW RULE `shadow_rule`"));
62      }
63      
64      @Test
65      void assertPlanDefaultAlgorithm() {
66          ShadowDefaultAlgorithmWorkflowRequest request = new ShadowDefaultAlgorithmWorkflowRequest();
67          request.setAlgorithmType("SQL_HINT");
68          String actual = new ShadowDistSQLPlanningService().planCreateDefaultAlgorithm(request).getSql();
69          assertThat(actual, is("CREATE DEFAULT SHADOW ALGORITHM TYPE(NAME='SQL_HINT')"));
70          assertFalse(actual.contains(" FROM "));
71      }
72      
73      @Test
74      void assertPlanAlterDefaultAlgorithm() {
75          ShadowDefaultAlgorithmWorkflowRequest request = new ShadowDefaultAlgorithmWorkflowRequest();
76          request.setAlgorithmType("SQL_HINT");
77          assertThat(new ShadowDistSQLPlanningService().planAlterDefaultAlgorithm(request).getSql(),
78                  is("ALTER DEFAULT SHADOW ALGORITHM TYPE(NAME='SQL_HINT')"));
79      }
80      
81      @Test
82      void assertPlanCleanup() {
83          assertThat(new ShadowDistSQLPlanningService().planDropAlgorithm("unused_algorithm").getSql(), is("DROP SHADOW ALGORITHM `unused_algorithm`"));
84      }
85      
86      private ShadowRuleWorkflowRequest createRuleRequest() {
87          ShadowRuleWorkflowRequest result = new ShadowRuleWorkflowRequest();
88          result.setRuleName("shadow_rule");
89          result.setSourceStorageUnit("demo_ds");
90          result.setShadowStorageUnit("demo_ds_shadow");
91          result.setTableName("t_order");
92          result.setAlgorithmType("VALUE_MATCH");
93          result.putAlgorithmProperties(Map.of("operation", "insert", "column", "user_id", "value", "1"));
94          return result;
95      }
96      
97      private void assertNoForbiddenArtifacts(final String sql) {
98          assertFalse(sql.contains("CREATE TABLE"));
99          assertFalse(sql.contains("CREATE INDEX"));
100         assertFalse(sql.contains("REGISTER STORAGE UNIT"));
101         assertFalse(sql.contains("MIGRATE"));
102     }
103 }
