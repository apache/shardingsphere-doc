114     @Test
115     void assertInitializeTwice() {
116         ClusterPersistRepositoryConfiguration repositoryConfig = mockRepositoryConfiguration("FIXTURE");
117         CoordinatorRegistryCenter registryCenter = mock(CoordinatorRegistryCenter.class);
118         serviceLoader.when(() -> TypedSPILoader.findService(CoordinatorRegistryCenterProvider.class, "FIXTURE")).thenReturn(Optional.of(provider));
119         when(provider.create(repositoryConfig, STATISTICS_JOB_PATH)).thenReturn(registryCenter);
120         when(contextManager.getMetaDataContexts().getMetaData().getTemporaryProps()).thenReturn(
121                 new TemporaryConfigurationProperties(PropertiesBuilder.build(new Property(TemporaryConfigurationPropertyKey.PROXY_META_DATA_COLLECTOR_CRON.getKey(), "0 0/5 * * * ?"))));
122         AtomicReference<JobConfiguration> jobConfigRef = new AtomicReference<>();
123         try (
124                 MockedConstruction<ScheduleJobBootstrap> scheduleJobBootstrapConstruction = mockConstruction(ScheduleJobBootstrap.class,
125                         (mock, context) -> jobConfigRef.set((JobConfiguration) context.arguments().get(2)));
126                 MockedConstruction<JobOperateAPIImpl> jobOperateAPIConstruction = mockConstruction(JobOperateAPIImpl.class)) {
127             jobWorker.initialize(contextManager);
128             jobWorker.initialize(contextManager);
129             verify(provider).create(repositoryConfig, STATISTICS_JOB_PATH);
130             verify(scheduleJobBootstrapConstruction.constructed().get(0)).schedule();
131             assertThat(jobConfigRef.get().getCron(), is("0 0/5 * * * ?"));
132             verify(jobOperateAPIConstruction.constructed().get(0)).trigger("statistics-collect");
133             jobWorker.destroy();
134             verify(scheduleJobBootstrapConstruction.constructed().get(0)).shutdown();
135             verify(registryCenter).close();
136         }
137     }

149     @Test
150     void assertInitializeFailureReleasesResourcesAndCanRetry() {
151         ClusterPersistRepositoryConfiguration repositoryConfig = mockRepositoryConfiguration("FIXTURE");
152         CoordinatorRegistryCenter firstRegistryCenter = mock(CoordinatorRegistryCenter.class);
153         CoordinatorRegistryCenter secondRegistryCenter = mock(CoordinatorRegistryCenter.class);
154         serviceLoader.when(() -> TypedSPILoader.findService(CoordinatorRegistryCenterProvider.class, "FIXTURE")).thenReturn(Optional.of(provider));
155         when(provider.create(repositoryConfig, STATISTICS_JOB_PATH)).thenReturn(firstRegistryCenter, secondRegistryCenter);
156         when(contextManager.getMetaDataContexts().getMetaData().getTemporaryProps()).thenReturn(new TemporaryConfigurationProperties(new Properties()));
157         RuntimeException expected = new RuntimeException("expected");
158         AtomicBoolean failOnSchedule = new AtomicBoolean(true);
159         try (
160                 MockedConstruction<ScheduleJobBootstrap> scheduleJobBootstrapConstruction = mockConstruction(ScheduleJobBootstrap.class, (mock, context) -> {
161                     if (failOnSchedule.compareAndSet(true, false)) {
162                         doThrow(expected).when(mock).schedule();
163                     }
164                 });
165                 MockedConstruction<JobOperateAPIImpl> jobOperateAPIConstruction = mockConstruction(JobOperateAPIImpl.class)) {
166             assertThat(assertThrows(RuntimeException.class, () -> jobWorker.initialize(contextManager)), sameInstance(expected));
167             assertFalse(workerInitialized.get());
168             verify(scheduleJobBootstrapConstruction.constructed().get(0)).shutdown();
169             verify(firstRegistryCenter).close();
170             jobWorker.initialize(contextManager);
171             assertTrue(workerInitialized.get());
172             verify(scheduleJobBootstrapConstruction.constructed().get(1)).schedule();
173             verify(jobOperateAPIConstruction.constructed().get(0)).trigger("statistics-collect");
174             jobWorker.destroy();
175             verify(scheduleJobBootstrapConstruction.constructed().get(1)).shutdown();
176             verify(secondRegistryCenter).close();
177         }
178     }

180     @Test
181     void assertPreserveInitializationFailureWhenCleanupFailed() {
182         ClusterPersistRepositoryConfiguration repositoryConfig = mockRepositoryConfiguration("FIXTURE");
183         CoordinatorRegistryCenter registryCenter = mock(CoordinatorRegistryCenter.class);
184         serviceLoader.when(() -> TypedSPILoader.findService(CoordinatorRegistryCenterProvider.class, "FIXTURE")).thenReturn(Optional.of(provider));
185         when(provider.create(repositoryConfig, STATISTICS_JOB_PATH)).thenReturn(registryCenter);
186         when(contextManager.getMetaDataContexts().getMetaData().getTemporaryProps()).thenReturn(new TemporaryConfigurationProperties(new Properties()));
187         RuntimeException expected = new RuntimeException("expected");
188         RuntimeException cleanupException = new RuntimeException("cleanup failed");
189         try (MockedConstruction<ScheduleJobBootstrap> scheduleJobBootstrapConstruction = mockConstruction(ScheduleJobBootstrap.class, (mock, context) -> {
190             doThrow(expected).when(mock).schedule();
191             doThrow(cleanupException).when(mock).shutdown();
192         })) {
193             RuntimeException actual = assertThrows(RuntimeException.class, () -> jobWorker.initialize(contextManager));
194             assertThat(actual, sameInstance(expected));
195             assertThat(actual.getSuppressed(), arrayContaining(cleanupException));
196             assertFalse(workerInitialized.get());
197             verify(scheduleJobBootstrapConstruction.constructed().get(0)).shutdown();
198             verify(registryCenter).close();
199         }
200     }

213     @Test
214     void assertUpdateJobConfiguration() {
215         setStaticField("contextManager", contextManager);
216         CoordinatorRegistryCenter registryCenter = mock(CoordinatorRegistryCenter.class);
217         setStaticField("registryCenter", registryCenter);
218         when(contextManager.getMetaDataContexts().getMetaData().getTemporaryProps()).thenReturn(
219                 new TemporaryConfigurationProperties(PropertiesBuilder.build(new Property(TemporaryConfigurationPropertyKey.PROXY_META_DATA_COLLECTOR_CRON.getKey(), "invalid"))));
220         AtomicReference<Object> constructorRegistryCenter = new AtomicReference<>();
221         try (
222                 MockedConstruction<JobConfigurationAPIImpl> jobConfigurationAPIConstruction = mockConstruction(JobConfigurationAPIImpl.class,
223                         (mock, context) -> constructorRegistryCenter.set(context.arguments().get(0)))) {
224             jobWorker.updateJobConfiguration();
225             assertThat(constructorRegistryCenter.get(), is(registryCenter));
226             ArgumentCaptor<JobConfigurationPOJO> argumentCaptor = ArgumentCaptor.forClass(JobConfigurationPOJO.class);
227             verify(jobConfigurationAPIConstruction.constructed().get(0)).updateJobConfiguration(argumentCaptor.capture());
228             JobConfiguration jobConfiguration = argumentCaptor.getValue().toJobConfiguration();
229             assertThat(jobConfiguration.getCron(), is(TemporaryConfigurationPropertyKey.PROXY_META_DATA_COLLECTOR_CRON.getDefaultValue()));
230         }
231     }
