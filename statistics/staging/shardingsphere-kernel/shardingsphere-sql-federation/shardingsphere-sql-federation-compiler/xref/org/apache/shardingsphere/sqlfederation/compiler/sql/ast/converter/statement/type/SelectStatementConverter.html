52  /**
53   * Select statement converter.
54   */
55  public final class SelectStatementConverter implements SQLStatementConverter<SelectStatement, SqlNode> {
56      
57      @Override
58      public SqlNode convert(final SelectStatement selectStatement) {
59          checkUnsupportedOpenGaussFeatures(selectStatement);
60          SqlSelect sqlSelect = convertSelect(selectStatement);
61          SqlNode sqlWith = convertWith(sqlSelect, selectStatement);
62          SqlNode sqlCombine = convertCombine(null == sqlWith ? sqlSelect : sqlWith, selectStatement);
63          SqlNodeList orderBy = selectStatement.getOrderBy().flatMap(OrderByConverter::convert).orElse(SqlNodeList.EMPTY);
64          Optional<LimitSegment> limit = selectStatement.getLimit();
65          if (limit.isPresent()) {
66              ShardingSpherePreconditions.checkState(!limit.get().isImplicitRowCount(), () -> new UnsupportedSQLOperationException("Implicit FETCH row count in SQL Federation"));
67              SqlNode offset = limit.get().getOffset().flatMap(PaginationValueSQLConverter::convert).orElse(null);
68              SqlNode rowCount = limit.get().getRowCount().flatMap(PaginationValueSQLConverter::convert).orElse(null);
69              return new SqlOrderBy(SqlParserPos.ZERO, sqlCombine, orderBy, offset, rowCount);
70          }
71          return orderBy.isEmpty() ? sqlCombine : new SqlOrderBy(SqlParserPos.ZERO, sqlCombine, orderBy, null, null);
72      }
73      
74      private void checkUnsupportedOpenGaussFeatures(final SelectStatement selectStatement) {
75          if (!"openGauss".equals(selectStatement.getDatabaseType().getType())) {
76              return;
77          }
78          ShardingSpherePreconditions.checkState(!containsCube(selectStatement),
79                  () -> new UnsupportedSQLOperationException("openGauss CUBE query in SQL Federation"));
80          ShardingSpherePreconditions.checkState(!containsAggregationWithNamedWindow(selectStatement),
81                  () -> new UnsupportedSQLOperationException("openGauss aggregate query with a named window in SQL Federation"));
82      }
83      
84      private boolean containsCube(final SelectStatement selectStatement) {
85          if (!selectStatement.getGroupBy().isPresent() || !selectStatement.getGroupBy().get().isContainsGroupingExtension()) {
86              return false;
87          }
88          GroupBySegment groupBy = selectStatement.getGroupBy().get();
89          for (OrderByItemSegment each : groupBy.getGroupByItems()) {
90              if (each instanceof ExpressionOrderByItemSegment && isCubeExpression(((ExpressionOrderByItemSegment) each).getExpression())) {
91                  return true;
92              }
93          }
94          return false;
95      }
96      
97      private boolean isCubeExpression(final String expression) {
98          int leftParenthesisIndex = expression.indexOf('(');
99          return leftParenthesisIndex > 0 && "CUBE".equalsIgnoreCase(expression.substring(0, leftParenthesisIndex).trim());
100     }
101     
102     private boolean containsAggregationWithNamedWindow(final SelectStatement selectStatement) {
103         for (ProjectionSegment each : selectStatement.getProjections().getProjections()) {
104             if (each instanceof AggregationProjectionSegment && ((AggregationProjectionSegment) each).getWindow().isPresent()
105                     && null != ((AggregationProjectionSegment) each).getWindow().get().getWindowName()) {
106                 return true;
107             }
108         }
109         return false;
110     }
111     
112     private SqlNode convertWith(final SqlNode sqlSelect, final SelectStatement selectStatement) {
113         return selectStatement.getWith().flatMap(segment -> WithConverter.convert(segment, sqlSelect)).orElse(null);
114     }
115     
116     private SqlSelect convertSelect(final SelectStatement selectStatement) {
117         SqlNodeList distinct = DistinctConverter.convert(selectStatement.getProjections()).orElse(null);
118         SqlNodeList projection = ProjectionsConverter.convert(selectStatement.getProjections()).orElseThrow(IllegalStateException::new);
119         SqlNode from = selectStatement.getFrom().flatMap(TableConverter::convert).orElse(null);
120         SqlNode where = selectStatement.getWhere().flatMap(WhereConverter::convert).orElse(null);
121         SqlNodeList groupBy = selectStatement.getGroupBy().flatMap(GroupByConverter::convert).orElse(null);
122         SqlNode having = selectStatement.getHaving().flatMap(HavingConverter::convert).orElse(null);
123         SqlNodeList window = selectStatement.getWindow().flatMap(WindowConverter::convert).orElse(SqlNodeList.EMPTY);
124         return new SqlSelect(SqlParserPos.ZERO, distinct, projection, from, where, groupBy, having, window, null, null, null, null, SqlNodeList.EMPTY);
125     }
126     
127     private SqlNode convertCombine(final SqlNode sqlNode, final SelectStatement selectStatement) {
128         if (selectStatement.getCombine().isPresent()) {
129             CombineSegment combineSegment = selectStatement.getCombine().get();
130             return new SqlBasicCall(CombineOperatorConverter.convert(combineSegment.getCombineType()),
131                     Arrays.asList(convert(combineSegment.getLeft().getSelect()), convert(combineSegment.getRight().getSelect())), SqlParserPos.ZERO);
132         }
133         return sqlNode;
134     }
