56  /**
57   * Select statement binder.
58   */
59  @RequiredArgsConstructor
60  public final class SelectStatementBinder implements SQLStatementBinder<SelectStatement> {
61      
62      private final Multimap<CaseInsensitiveString, TableSegmentBinderContext> outerTableBinderContexts;
63      
64      public SelectStatementBinder() {
65          outerTableBinderContexts = LinkedHashMultimap.create();
66      }
67      
68      @Override
69      public SelectStatement bind(final SelectStatement sqlStatement, final SQLStatementBinderContext binderContext) {
70          Multimap<CaseInsensitiveString, TableSegmentBinderContext> tableBinderContexts = LinkedHashMultimap.create();
71          WithSegment boundWith = sqlStatement.getWith().map(optional -> WithSegmentBinder.bind(optional, binderContext, binderContext.getExternalTableBinderContexts())).orElse(null);
72          Optional<TableSegment> boundFrom = sqlStatement.getFrom().map(optional -> TableSegmentBinder.bind(optional, binderContext, tableBinderContexts, outerTableBinderContexts));
73          ProjectionsSegment boundProjections = ProjectionsSegmentBinder.bind(sqlStatement.getProjections(), binderContext, boundFrom.orElse(null), tableBinderContexts, outerTableBinderContexts);
74          WhereSegment boundWhere = sqlStatement.getWhere().map(optional -> WhereSegmentBinder.bind(optional, binderContext, tableBinderContexts, outerTableBinderContexts)).orElse(null);
75          CombineSegment boundCombine = sqlStatement.getCombine().map(optional -> CombineSegmentBinder.bind(optional, binderContext, outerTableBinderContexts)).orElse(null);
76          LockSegment boundLock = sqlStatement.getLock().map(optional -> LockSegmentBinder.bind(optional, binderContext, tableBinderContexts, outerTableBinderContexts)).orElse(null);
77          Multimap<CaseInsensitiveString, TableSegmentBinderContext> currentTableBinderContexts = createCurrentTableBinderContexts(binderContext, boundProjections);
78          GroupBySegment boundGroupBy =
79                  sqlStatement.getGroupBy().map(optional -> GroupBySegmentBinder.bind(optional, binderContext, currentTableBinderContexts, tableBinderContexts, outerTableBinderContexts)).orElse(null);
80          OrderBySegment boundOrderBy =
81                  sqlStatement.getOrderBy().map(optional -> OrderBySegmentBinder.bind(optional, binderContext, currentTableBinderContexts, tableBinderContexts, outerTableBinderContexts)).orElse(null);
82          HavingSegment boundHaving =
83                  sqlStatement.getHaving().map(optional -> HavingSegmentBinder.bind(optional, binderContext, currentTableBinderContexts, tableBinderContexts, outerTableBinderContexts)).orElse(null);
84          return copy(sqlStatement, boundWith, boundFrom.orElse(null), boundProjections, boundWhere, boundCombine, boundLock, boundGroupBy, boundOrderBy, boundHaving);
85      }
86      
87      private Multimap<CaseInsensitiveString, TableSegmentBinderContext> createCurrentTableBinderContexts(final SQLStatementBinderContext binderContext, final ProjectionsSegment boundProjections) {
88          Multimap<CaseInsensitiveString, TableSegmentBinderContext> result = LinkedHashMultimap.create();
89          Collection<ProjectionSegment> subqueryProjections = SubqueryTableBindUtils.createSubqueryProjections(
90                  boundProjections.getProjections(), new IdentifierValue(""), binderContext.getSqlStatement().getDatabaseType(), TableSourceType.MIXED_TABLE);
91          result.put(new CaseInsensitiveString(""), new SimpleTableSegmentBinderContext(subqueryProjections, TableSourceType.MIXED_TABLE));
92          return result;
93      }
94      
95      private SelectStatement copy(final SelectStatement sqlStatement, final WithSegment boundWith, final TableSegment boundFrom, final ProjectionsSegment boundProjections,
96                                   final WhereSegment boundWhere, final CombineSegment boundCombine, final LockSegment boundLock,
97                                   final GroupBySegment boundGroupBy, final OrderBySegment boundOrderBy, final HavingSegment boundHaving) {
98          SelectStatement result = new SelectStatement(sqlStatement.getDatabaseType());
99          result.setWith(boundWith);
100         result.setFrom(boundFrom);
101         result.setProjections(boundProjections);
102         result.setWhere(boundWhere);
103         result.setCombine(boundCombine);
104         result.setLock(boundLock);
105         result.setGroupBy(boundGroupBy);
106         result.setOrderBy(boundOrderBy);
107         result.setHaving(boundHaving);
108         sqlStatement.getLimit().ifPresent(result::setLimit);
109         sqlStatement.getWindow().ifPresent(result::setWindow);
110         sqlStatement.getModel().ifPresent(result::setModel);
111         sqlStatement.getSubqueryType().ifPresent(result::setSubqueryType);
112         SQLStatementCopyUtils.copyAttributes(sqlStatement, result);
113         return result;
114     }
