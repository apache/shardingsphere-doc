36  import static org.hamcrest.MatcherAssert.assertThat;
37  import static org.hamcrest.Matchers.is;
38  import static org.mockito.Mockito.mock;
39  import static org.mockito.Mockito.when;
40  
41  class ShardingRuleConfigurationToDistSQLConverterTest {
42      
43      @SuppressWarnings("unchecked")
44      private final RuleConfigurationToDistSQLConverter<ShardingRuleConfiguration> converter = TypedSPILoader.getService(RuleConfigurationToDistSQLConverter.class, ShardingRuleConfiguration.class);
45      
46      @Test
47      void assertConvertWithEmptyTables() {
48          ShardingRuleConfiguration shardingRuleConfig = mock(ShardingRuleConfiguration.class);
49          when(shardingRuleConfig.getTables()).thenReturn(Collections.emptyList());
50          when(shardingRuleConfig.getAutoTables()).thenReturn(Collections.emptyList());
51          assertThat(converter.convert(shardingRuleConfig), is(""));
52      }
53      
54      @Test
55      void assertConvert() {
56          ShardingRuleConfiguration shardingRuleConfig = new ShardingRuleConfiguration();
57          shardingRuleConfig.getTables().add(createShardingTableRuleConfiguration());
58          shardingRuleConfig.setDefaultDatabaseShardingStrategy(new StandardShardingStrategyConfiguration("user_id", "database_inline"));
59          shardingRuleConfig.setDefaultTableShardingStrategy(new NoneShardingStrategyConfiguration());
60          shardingRuleConfig.getShardingAlgorithms().put("database_inline", createShardingInlineAlgorithmConfiguration("ds_${user_id % 2}"));
61          shardingRuleConfig.getShardingAlgorithms().put("t_order_inline", createShardingInlineAlgorithmConfiguration("t_order_${order_id % 2}"));
62          shardingRuleConfig.getKeyGenerators().put("snowflake", createKeyGeneratorConfiguration());
63          shardingRuleConfig.getKeyGenerateStrategies().put("t_order_order_id", new ColumnKeyGenerateStrategiesRuleConfiguration("snowflake", "t_order", "order_id"));
64          shardingRuleConfig.getAuditors().put("sharding_key_required_auditor", createAuditorConfiguration());
65          assertThat(converter.convert(shardingRuleConfig),
66                  is("CREATE SHARDING TABLE RULE t_order (" + System.lineSeparator() + "DATANODES('ds_${0..1}.t_order_${0..1}')," + System.lineSeparator()
67                          + "TABLE_STRATEGY(TYPE='standard', SHARDING_COLUMN=order_id, SHARDING_ALGORITHM(TYPE(NAME='inline', PROPERTIES('algorithm-expression'='t_order_${order_id % 2}')))),"
68                          + System.lineSeparator() + "KEY_GENERATE_STRATEGY(COLUMN=order_id, TYPE(NAME='snowflake'))," + System.lineSeparator()
69                          + "AUDIT_STRATEGY(TYPE(NAME='dml_sharding_conditions'), ALLOW_HINT_DISABLE=true)" + System.lineSeparator() + ");" + System.lineSeparator() + System.lineSeparator()
70                          + "CREATE DEFAULT SHARDING DATABASE STRATEGY(TYPE='standard', SHARDING_COLUMN=user_id, "
71                          + "SHARDING_ALGORITHM(TYPE(NAME='inline', PROPERTIES('algorithm-expression'='ds_${user_id % 2}'))));" + System.lineSeparator() + System.lineSeparator()
72                          + "CREATE DEFAULT SHARDING TABLE STRATEGY(TYPE='none');"));
73      }
74      
75      private ShardingTableRuleConfiguration createShardingTableRuleConfiguration() {
76          ShardingTableRuleConfiguration result = new ShardingTableRuleConfiguration("t_order", "ds_${0..1}.t_order_${0..1}");
77          result.setTableShardingStrategy(new StandardShardingStrategyConfiguration("order_id", "t_order_inline"));
78          result.setAuditStrategy(new ShardingAuditStrategyConfiguration(Collections.singleton("sharding_key_required_auditor"), true));
79          return result;
80      }
81      
82      private AlgorithmConfiguration createShardingInlineAlgorithmConfiguration(final String algorithmExpression) {
83          return new AlgorithmConfiguration("INLINE", PropertiesBuilder.build(new Property("algorithm-expression", algorithmExpression)));
84      }
85      
86      private AlgorithmConfiguration createKeyGeneratorConfiguration() {
87          return new AlgorithmConfiguration("SNOWFLAKE", new Properties());
88      }
89      
90      private AlgorithmConfiguration createAuditorConfiguration() {
91          return new AlgorithmConfiguration("DML_SHARDING_CONDITIONS", new Properties());
92      }
93  }
