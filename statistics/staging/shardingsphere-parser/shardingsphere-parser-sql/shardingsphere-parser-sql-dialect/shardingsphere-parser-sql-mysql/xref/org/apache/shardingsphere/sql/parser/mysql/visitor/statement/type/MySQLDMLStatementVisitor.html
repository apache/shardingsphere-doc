63  /**
64   * DML statement visitor for MySQL.
65   */
66  public final class MySQLDMLStatementVisitor extends MySQLStatementVisitor implements DMLStatementVisitor {
67      
68      @Override
69      public ASTNode visitCall(final CallContext ctx) {
70          MySQLCallStatement result = new MySQLCallStatement();
71          String procedureName = null == ctx.owner() ? ctx.identifier().getText() : ctx.owner().getText() + "." + ctx.identifier().getText();
72          result.setProcedureName(procedureName);
73          List<ExpressionSegment> params = ctx.expr().stream().map(each -> (ExpressionSegment) visit(each)).collect(Collectors.toList());
74          result.getParameters().addAll(params);
75          return result;
76      }
77      
78      @Override
79      public ASTNode visitDoStatement(final DoStatementContext ctx) {
80          MySQLDoStatement result = new MySQLDoStatement();
81          result.getParameters().addAll(ctx.expr().stream().map(each -> (ExpressionSegment) visit(each)).collect(Collectors.toList()));
82          return result;
83      }
84      
85      @Override
86      public ASTNode visitHandlerStatement(final HandlerStatementContext ctx) {
87          return new MySQLHandlerStatement();
88      }
89      
90      @Override
91      public ASTNode visitImportStatement(final ImportStatementContext ctx) {
92          return new MySQLImportStatement();
93      }
94      
95      @Override
96      public ASTNode visitLoadStatement(final LoadStatementContext ctx) {
97          return null == ctx.loadDataStatement() ? visit(ctx.loadXmlStatement()) : visit(ctx.loadDataStatement());
98      }
99      
100     @Override
101     public ASTNode visitLoadDataStatement(final LoadDataStatementContext ctx) {
102         MySQLLoadDataStatement result = new MySQLLoadDataStatement();
103         result.setTableSegment((SimpleTableSegment) visit(ctx.tableName()));
104         return result;
105     }
106     
107     @Override
108     public ASTNode visitLoadXmlStatement(final LoadXmlStatementContext ctx) {
109         MySQLLoadXMLStatement result = new MySQLLoadXMLStatement();
110         result.setTableSegment((SimpleTableSegment) visit(ctx.tableName()));
111         return result;
112     }
113     
114     @Override
115     public ASTNode visitIndexHint(final IndexHintContext ctx) {
116         Collection<String> indexNames = new LinkedList<>();
117         if (null != ctx.indexNameList()) {
118             ctx.indexNameList().indexName().forEach(each -> indexNames.add(each.getText()));
119         }
120         String useType;
121         if (null != ctx.USE()) {
122             useType = ctx.USE().getText();
123         } else if (null != ctx.IGNORE()) {
124             useType = ctx.IGNORE().getText();
125         } else {
126             useType = ctx.FORCE().getText();
127         }
128         IndexHintSegment result = new IndexHintSegment(ctx.start.getStartIndex(), ctx.stop.getStopIndex(), indexNames, useType,
129                 null == ctx.INDEX() ? ctx.KEY().getText() : ctx.INDEX().getText(), getOriginalText(ctx));
130         if (null != ctx.indexHintClause().FOR()) {
131             String hintScope;
132             if (null != ctx.indexHintClause().JOIN()) {
133                 hintScope = "JOIN";
134             } else if (null != ctx.indexHintClause().ORDER()) {
135                 hintScope = "ORDER BY";
136             } else {
137                 hintScope = "GROUP BY";
138             }
139             result.setHintScope(hintScope);
140         }
141         return result;
142     }
143     
144     @Override
145     public ASTNode visitWindowClause(final WindowClauseContext ctx) {
146         WindowSegment result = new WindowSegment(ctx.getStart().getStartIndex(), ctx.getStop().getStopIndex());
147         for (WindowItemContext each : ctx.windowItem()) {
148             result.getItemSegments().add((WindowItemSegment) visit(each));
149         }
150         return result;
151     }
152     
153     @Override
154     public ASTNode visitWindowItem(final WindowItemContext ctx) {
155         WindowItemSegment result = new WindowItemSegment(ctx.getStart().getStartIndex(), ctx.getStop().getStopIndex());
156         result.setWindowName(new IdentifierValue(ctx.identifier().getText()));
157         WindowItemSegment windowItemSegment = (WindowItemSegment) visit(ctx.windowSpecification());
158         result.setPartitionListSegments(windowItemSegment.getPartitionListSegments());
159         result.setOrderBySegment(windowItemSegment.getOrderBySegment());
160         result.setFrameClause(windowItemSegment.getFrameClause());
161         return result;
162     }
163     
164     @Override
165     public ASTNode visitWindowSpecification(final WindowSpecificationContext ctx) {
166         WindowItemSegment result = new WindowItemSegment(ctx.start.getStartIndex(), ctx.stop.getStopIndex());
167         if (null != ctx.PARTITION()) {
168             result.setPartitionListSegments(getExpressions(ctx.expr()));
169         }
170         if (null != ctx.orderByClause()) {
171             result.setOrderBySegment((OrderBySegment) visit(ctx.orderByClause()));
172         }
173         if (null != ctx.frameClause()) {
174             result.setFrameClause(new CommonExpressionSegment(ctx.frameClause().start.getStartIndex(), ctx.frameClause().stop.getStopIndex(), ctx.frameClause().getText()));
175         }
176         if (null != ctx.identifier()) {
177             result.setWindowName(new IdentifierValue(ctx.identifier().getText()));
178         }
179         return result;
180     }
181     
182     @Override
183     public ASTNode visitWindowFunction(final WindowFunctionContext ctx) {
184         super.visitWindowFunction(ctx);
185         FunctionSegment result = new FunctionSegment(ctx.getStart().getStartIndex(), ctx.getStop().getStopIndex(), ctx.funcName.getText(), getOriginalText(ctx));
186         if (null != ctx.NTILE()) {
187             result.getParameters().add((ExpressionSegment) visit(ctx.simpleExpr()));
188         }
189         if (null != ctx.LEAD() || null != ctx.LAG() || null != ctx.FIRST_VALUE() || null != ctx.LAST_VALUE()) {
190             result.getParameters().add((ExpressionSegment) visit(ctx.expr()));
191         }
192         if (null != ctx.NTH_VALUE()) {
193             result.getParameters().add((ExpressionSegment) visit(ctx.expr()));
194             result.getParameters().add((ExpressionSegment) visit(ctx.simpleExpr()));
195         }
196         result.setWindow((WindowItemSegment) visit(ctx.windowingClause()));
197         return result;
198     }
199     
200     @Override
201     public ASTNode visitWindowingClause(final WindowingClauseContext ctx) {
202         WindowItemSegment result = new WindowItemSegment(ctx.getStart().getStartIndex(), ctx.getStop().getStopIndex());
203         if (null != ctx.windowName) {
204             result.setWindowName((IdentifierValue) visit(ctx.windowName));
205         }
206         if (null != ctx.windowSpecification()) {
207             WindowItemSegment windowItemSegment = (WindowItemSegment) visit(ctx.windowSpecification());
208             result.setPartitionListSegments(windowItemSegment.getPartitionListSegments());
209             result.setOrderBySegment(windowItemSegment.getOrderBySegment());
210             result.setFrameClause(windowItemSegment.getFrameClause());
211         }
212         return result;
213     }
214     
215     @Override
216     public ASTNode visitReturningClause(final ReturningClauseContext ctx) {
217         TargetListContext targetList = ctx.targetList();
218         ProjectionsSegment projectionsSegment = new ProjectionsSegment(targetList.getStart().getStartIndex(), targetList.getStop().getStopIndex());
219         for (ProjectionContext projectionContext : targetList.projection()) {
220             projectionsSegment.getProjections().add((ProjectionSegment) visit(projectionContext));
221         }
222         return new ReturningSegment(ctx.getStart().getStartIndex(), ctx.getStop().getStopIndex(), projectionsSegment);
223     }
