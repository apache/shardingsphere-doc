27  import java.io.BufferedReader;
28  import java.io.BufferedWriter;
29  import java.io.IOException;
30  import java.io.InputStreamReader;
31  import java.io.OutputStreamWriter;
32  import java.nio.charset.StandardCharsets;
33  import java.util.List;
34  import java.util.Map;
35  import java.util.concurrent.CopyOnWriteArrayList;
36  import java.util.concurrent.ExecutionException;
37  import java.util.concurrent.ExecutorService;
38  import java.util.concurrent.Executors;
39  import java.util.concurrent.Future;
40  import java.util.concurrent.TimeUnit;
41  import java.util.concurrent.TimeoutException;

43  /**
44   * STDIO MCP interaction client backed by one child process.
45   */
46  abstract class AbstractProcessMCPStdioInteractionClient extends AbstractMCPInteractionClient {
47      
48      private static final long PROCESS_STOP_TIMEOUT_SECONDS = 5L;
49      
50      private static final long RESPONSE_TIMEOUT_SECONDS = 30L;
51      
52      private static final int STDERR_DIAGNOSTIC_MAX_CHARS = 4096;
53      
54      private static final String INITIALIZE_REQUEST_ID = "init-1";
55      
56      private final List<String> stdErrorMessages = new CopyOnWriteArrayList<>();
57      
58      private Process process;
59      
60      private Thread stdErrorCollector;
61      
62      private BufferedWriter writer;
63      
64      private BufferedReader reader;
65      
66      @Override
67      public final void open() throws IOException, InterruptedException {
68          if (null != process) {
69              return;
70          }
71          try {
72              process = createProcessBuilder().start();
73              stdErrorCollector = startStdErrorCollector(process, stdErrorMessages);
74              writer = new BufferedWriter(new OutputStreamWriter(process.getOutputStream(), StandardCharsets.UTF_8));
75              reader = new BufferedReader(new InputStreamReader(process.getInputStream(), StandardCharsets.UTF_8));
76              initializeSession();
77          } catch (final IOException | InterruptedException | IllegalStateException ex) {
78              closeQuietly();
79              throw ex;
80          }
81      }
82      
83      @Override
84      public final void close() throws IOException {
85          if (null == process) {
86              return;
87          }
88          try {
89              writer.close();
90              waitForNormalExit();
91          } catch (final InterruptedException ex) {
92              Thread.currentThread().interrupt();
93              throw new IOException("STDIO MCP process was interrupted during shutdown.", ex);
94          } finally {
95              closeQuietly();
96          }
97      }
98      
99      protected abstract ProcessBuilder createProcessBuilder() throws IOException;
100     
101     protected abstract String getClientName();
102     
103     @Override
104     protected final void ensureOpened() {
105         if (null == process) {
106             throw new IllegalStateException("MCP session is not initialized.");
107         }
108     }
109     
110     @Override
111     protected final Map<String, Object> sendRequest(final String requestId, final String method,
112                                                     final Map<String, Object> params) throws IOException, InterruptedException {
113         writeJsonRpcMessage(MCPInteractionProtocolSupport.createJsonRpcRequest(requestId, method, params));
114         return readResponse(requestId);
115     }
116     
117     private void initializeSession() throws IOException, InterruptedException {
118         Map<String, Object> initializePayload = sendRequest(INITIALIZE_REQUEST_ID, "initialize",
119                 MCPInteractionProtocolSupport.createInitializeRequestParams(getClientName()));
120         if (MCPInteractionPayloads.hasJsonRpcError(initializePayload)) {
121             throw createRuntimeFailureException("Failed to initialize STDIO MCP session: "
122                     + MCPInteractionPayloads.getJsonRpcErrorPayload(initializePayload).get("message")
123                     + ".");
124         }
125         Map<String, Object> initializeResult = MCPInteractionPayloads.getRequiredJsonRpcResult(initializePayload);
126         if (!MCPInteractionProtocolSupport.PROTOCOL_VERSION.equals(initializeResult.get("protocolVersion"))) {
127             throw createRuntimeFailureException("Unexpected STDIO MCP protocol version: " + initializeResult + ".");
128         }
129         sendNotification("notifications/initialized", Map.of());
130     }
131     
132     private Thread startStdErrorCollector(final Process process, final List<String> stdErrorMessages) {
133         Thread result = new Thread(() -> collectStdError(process, stdErrorMessages), getClientName() + "-stderr");
134         result.setDaemon(true);
135         result.start();
136         return result;
137     }
138     
139     private void collectStdError(final Process process, final List<String> stdErrorMessages) {
140         try (BufferedReader actualReader = new BufferedReader(new InputStreamReader(process.getErrorStream(), StandardCharsets.UTF_8))) {
141             String line;
142             while (null != (line = actualReader.readLine())) {
143                 stdErrorMessages.add(line);
144             }
145         } catch (final IOException ignored) {
146         }
147     }
148     
149     @Override
150     protected final void sendNotification(final String method, final Map<String, Object> params) throws IOException {
151         writeJsonRpcMessage(MCPInteractionProtocolSupport.createJsonRpcNotification(method, params));
152     }
153     
154     private void writeJsonRpcMessage(final Map<String, Object> payload) throws IOException {
155         try {
156             writer.write(JsonEngine.marshal(payload));
157         } catch (final JsonException ex) {
158             throw new IOException(ex.getMessage(), ex);
159         }
160         writer.newLine();
161         writer.flush();
162     }
163     
164     private Map<String, Object> readResponse(final String requestId) throws IOException, InterruptedException {
165         long deadlineNanos = System.nanoTime() + TimeUnit.SECONDS.toNanos(RESPONSE_TIMEOUT_SECONDS);
166         String line;
167         while (null != (line = readLine(requestId, deadlineNanos))) {
168             if (line.isBlank()) {
169                 continue;
170             }
171             Map<String, Object> result;
172             try {
173                 result = JsonEngine.unmarshal(line, new JsonTypeReference<Map<String, Object>>() {
174                 });
175             } catch (final JsonException ex) {
176                 throw new IOException(ex.getMessage(), ex);
177             }
178             if (requestId.equals(String.valueOf(result.get("id")))) {
179                 return result;
180             }
181         }
182         throw createRuntimeFailureException("STDIO MCP runtime did not return a response.");
183     }
184     
185     private String readLine(final String requestId, final long deadlineNanos) throws IOException, InterruptedException {
186         long remainingNanos = deadlineNanos - System.nanoTime();
187         if (0L >= remainingNanos) {
188             throw createResponseTimeoutException(requestId);
189         }
190         ExecutorService executor = Executors.newVirtualThreadPerTaskExecutor();
191         Future<String> responseLine = executor.submit(reader::readLine);
192         try {
193             return responseLine.get(remainingNanos, TimeUnit.NANOSECONDS);
194         } catch (final TimeoutException ex) {
195             responseLine.cancel(true);
196             throw createResponseTimeoutException(requestId);
197         } catch (final ExecutionException ex) {
198             if (ex.getCause() instanceof IOException) {
199                 throw (IOException) ex.getCause();
200             }
201             throw new IOException("Failed to read STDIO MCP response.", ex.getCause());
202         } finally {
203             executor.shutdownNow();
204         }
205     }
206     
207     private IllegalStateException createResponseTimeoutException(final String requestId) throws InterruptedException {
208         destroyProcess();
209         return createRuntimeFailureException(String.format(
210                 "STDIO MCP runtime did not return response `%s` within %d seconds.", requestId, RESPONSE_TIMEOUT_SECONDS));
211     }
212     
213     private void waitForNormalExit() throws InterruptedException {
214         if (!process.waitFor(PROCESS_STOP_TIMEOUT_SECONDS, TimeUnit.SECONDS)) {
215             destroyProcess();
216             throw createRuntimeFailureException("STDIO MCP process did not exit after stdin closed.");
217         }
218         if (0 != process.exitValue()) {
219             throw createRuntimeFailureException("STDIO MCP process exited with code " + process.exitValue() + ".");
220         }
221     }
222     
223     private void destroyProcess() throws InterruptedException {
224         if (!process.isAlive()) {
225             return;
226         }
227         process.destroy();
228         if (process.waitFor(PROCESS_STOP_TIMEOUT_SECONDS, TimeUnit.SECONDS)) {
229             return;
230         }
231         process.destroyForcibly();
232         process.waitFor(PROCESS_STOP_TIMEOUT_SECONDS, TimeUnit.SECONDS);
233     }
234     
235     private String getStdErrorOutput() {
236         if (stdErrorMessages.isEmpty()) {
237             return "<empty>";
238         }
239         String result = String.join(System.lineSeparator(), stdErrorMessages);
240         return result.length() <= STDERR_DIAGNOSTIC_MAX_CHARS ? result : result.substring(0, STDERR_DIAGNOSTIC_MAX_CHARS) + "...<truncated>";
241     }
242     
243     private IllegalStateException createRuntimeFailureException(final String defaultMessage) {
244         return new IllegalStateException(createRuntimeFailureMessage(defaultMessage));
245     }
246     
247     private String createRuntimeFailureMessage(final String defaultMessage) {
248         String actualMessage = getProcessFailureMessage();
249         return (null == actualMessage ? defaultMessage : defaultMessage + " Process failure: " + appendSentenceTerminator(actualMessage)) + " stderr: " + getStdErrorOutput();
250     }
251     
252     private String appendSentenceTerminator(final String value) {
253         return value.endsWith(".") ? value : value + ".";
254     }
255     
256     private String getProcessFailureMessage() {
257         return stdErrorMessages.stream().filter(each -> each.startsWith("Exception in thread "))
258                 .map(this::extractFailureMessage).findFirst().orElse(null);
259     }
260     
261     private String extractFailureMessage(final String errorLine) {
262         int separatorIndex = errorLine.lastIndexOf(": ");
263         return -1 == separatorIndex ? errorLine : errorLine.substring(separatorIndex + 2);
264     }
265     
266     private void closeQuietly() {
267         closeReaderQuietly();
268         closeWriterQuietly();
269         destroyProcessQuietly();
270         if (null != stdErrorCollector) {
271             try {
272                 stdErrorCollector.join(TimeUnit.SECONDS.toMillis(PROCESS_STOP_TIMEOUT_SECONDS));
273             } catch (final InterruptedException ignored) {
274                 Thread.currentThread().interrupt();
275             }
276         }
277         process = null;
278         reader = null;
279         writer = null;
280         stdErrorCollector = null;
281         MCPArtifactUtils.writeRuntimeLogIfConfigured(getClientName() + "-", stdErrorMessages);
282         stdErrorMessages.clear();
283     }
284     
285     private void closeReaderQuietly() {
286         try {
287             if (null != reader) {
288                 reader.close();
289             }
290         } catch (final IOException ignored) {
291         }
292     }
293     
294     private void closeWriterQuietly() {
295         try {
296             if (null != writer) {
297                 writer.close();
298             }
299         } catch (final IOException ignored) {
300         }
301     }
302     
303     private void destroyProcessQuietly() {
304         if (null == process || !process.isAlive()) {
305             return;
306         }
307         process.destroy();
308         try {
309             if (process.waitFor(PROCESS_STOP_TIMEOUT_SECONDS, TimeUnit.SECONDS)) {
310                 return;
311             }
312             process.destroyForcibly();
313             process.waitFor(PROCESS_STOP_TIMEOUT_SECONDS, TimeUnit.SECONDS);
314         } catch (final InterruptedException ignored) {
315             Thread.currentThread().interrupt();
316         }
317     }
