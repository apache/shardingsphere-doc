53  class ShardingSphereResultSetFactoryTest {
54      
55      @ParameterizedTest(name = "{0}")
56      @MethodSource("columnMappingArguments")
57      void assertNewInstance(final String name, final Map<String, Integer> columnLabelAndIndexMap, final Map<String, Integer> expected) throws SQLException {
58          ResultSet resultSet = mock(ResultSet.class);
59          ResultSetMetaData resultSetMetaData = mock(ResultSetMetaData.class);
60          when(resultSet.getMetaData()).thenReturn(resultSetMetaData);
61          QueryResult queryResult = mock(QueryResult.class);
62          when(queryResult.getJDBCResultSet()).thenReturn(Optional.of(resultSet));
63          List<QueryResult> queryResults = Collections.singletonList(queryResult);
64          QueryContext queryContext = mock(QueryContext.class);
65          SQLStatementContext sqlStatementContext = mock(SQLStatementContext.class);
66          when(queryContext.getSqlStatementContext()).thenReturn(sqlStatementContext);
67          MergedResult mergedResult = mock(MergedResult.class);
68          ShardingSphereResultSetFactory factory = new ShardingSphereResultSetFactory(mock(ConnectionContext.class), mock(ShardingSphereMetaData.class), mock(ConfigurationProperties.class),
69                  Collections.emptyList());
70          try (
71                  MockedStatic<ShardingSphereResultSetUtils> mockedResultSetUtils = mockStatic(ShardingSphereResultSetUtils.class);
72                  MockedConstruction<MergeEngine> ignored = mockConstruction(MergeEngine.class, (mock, context) -> when(mock.merge(queryResults, queryContext)).thenReturn(mergedResult))) {
73              if (null == columnLabelAndIndexMap) {
74                  mockedResultSetUtils.when(() -> ShardingSphereResultSetUtils.createColumnLabelAndIndexMap(sqlStatementContext, resultSetMetaData)).thenReturn(expected);
75              }
76              ShardingSphereResultSet actual =
77                      (ShardingSphereResultSet) factory.newInstance(mock(ShardingSphereDatabase.class), queryContext, queryResults, mock(Statement.class), columnLabelAndIndexMap);
78              assertThat(actual.getColumnLabelAndIndexMap(), sameInstance(expected));
79              verify(resultSet).getMetaData();
80          }
81      }
82      
83      private static Stream<Arguments> columnMappingArguments() {
84          Map<String, Integer> supplied = Collections.singletonMap("foo_label", 2);
85          Map<String, Integer> empty = Collections.emptyMap();
86          return Stream.of(
87                  Arguments.of("absent mapping", null, Collections.singletonMap("foo_label", 1)),
88                  Arguments.of("supplied mapping", supplied, supplied),
89                  Arguments.of("empty mapping", empty, empty));
90      }
