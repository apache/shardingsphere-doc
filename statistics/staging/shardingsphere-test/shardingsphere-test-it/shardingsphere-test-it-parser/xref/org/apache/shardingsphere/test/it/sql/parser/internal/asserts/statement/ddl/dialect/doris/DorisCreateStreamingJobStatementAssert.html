33  import static org.hamcrest.MatcherAssert.assertThat;
34  import static org.hamcrest.Matchers.is;
35  import static org.junit.jupiter.api.Assertions.assertNotNull;
36  import static org.junit.jupiter.api.Assertions.assertTrue;
37  
38  /**
39   * Create streaming job statement assert for Doris.
40   */
41  @NoArgsConstructor(access = AccessLevel.PRIVATE)
42  public final class DorisCreateStreamingJobStatementAssert {
43      
44      /**
45       * Assert create streaming job statement is correct with expected parser result.
46       *
47       * @param assertContext assert context
48       * @param actual actual create streaming job statement
49       * @param expected expected create streaming job statement test case
50       */
51      public static void assertIs(final SQLCaseAssertContext assertContext, final DorisCreateStreamingJobStatement actual, final DorisCreateStreamingJobStatementTestCase expected) {
52          assertJobName(assertContext, actual, expected);
53          assertComment(assertContext, actual, expected);
54          assertSourceType(assertContext, actual, expected);
55          assertTargetDatabase(assertContext, actual, expected);
56          assertSourceProperties(assertContext, actual, expected);
57          assertTargetProperties(assertContext, actual, expected);
58          assertJobProperties(assertContext, actual, expected);
59          assertInsertStatement(assertContext, actual, expected);
60      }
61      
62      private static void assertJobName(final SQLCaseAssertContext assertContext, final DorisCreateStreamingJobStatement actual, final DorisCreateStreamingJobStatementTestCase expected) {
63          assertNotNull(actual.getJobName(), assertContext.getText("Job name should exist."));
64          if (null != expected.getJobName()) {
65              assertThat(assertContext.getText("Job name does not match: "), actual.getJobName().getIdentifier().getValue(), is(expected.getJobName().getName()));
66              SQLSegmentAssert.assertIs(assertContext, actual.getJobName(), expected.getJobName());
67          }
68      }
69      
70      private static void assertComment(final SQLCaseAssertContext assertContext, final DorisCreateStreamingJobStatement actual, final DorisCreateStreamingJobStatementTestCase expected) {
71          if (null != expected.getComment()) {
72              assertTrue(actual.getComment().isPresent(), assertContext.getText("Comment should exist."));
73              assertThat(assertContext.getText("Comment value does not match: "), actual.getComment().get().getValue(), is(expected.getComment().getValue()));
74              SQLSegmentAssert.assertIs(assertContext, actual.getComment().get(), expected.getComment());
75          }
76      }
77      
78      private static void assertSourceType(final SQLCaseAssertContext assertContext, final DorisCreateStreamingJobStatement actual, final DorisCreateStreamingJobStatementTestCase expected) {
79          if (null != expected.getSourceType()) {
80              assertTrue(actual.getSourceType().isPresent(), assertContext.getText("Source type should exist."));
81              assertThat(assertContext.getText("Source type does not match: "), actual.getSourceType().get(), is(expected.getSourceType()));
82          }
83      }
84      
85      private static void assertTargetDatabase(final SQLCaseAssertContext assertContext, final DorisCreateStreamingJobStatement actual, final DorisCreateStreamingJobStatementTestCase expected) {
86          if (null != expected.getTargetDatabase()) {
87              assertTrue(actual.getTargetDatabase().isPresent(), assertContext.getText("Target database should exist."));
88              assertThat(assertContext.getText("Target database does not match: "), actual.getTargetDatabase().get(), is(expected.getTargetDatabase()));
89          }
90      }
91      
92      private static void assertSourceProperties(final SQLCaseAssertContext assertContext, final DorisCreateStreamingJobStatement actual, final DorisCreateStreamingJobStatementTestCase expected) {
93          if (!expected.getSourceProperties().isEmpty()) {
94              assertTrue(actual.getSourceProperties().isPresent(), assertContext.getText("Source properties should exist."));
95              assertProperties(assertContext, actual.getSourceProperties().get().getProperties(), expected.getSourceProperties(), "Source");
96          }
97      }
98      
99      private static void assertTargetProperties(final SQLCaseAssertContext assertContext, final DorisCreateStreamingJobStatement actual, final DorisCreateStreamingJobStatementTestCase expected) {
100         if (!expected.getTargetProperties().isEmpty()) {
101             assertTrue(actual.getTargetProperties().isPresent(), assertContext.getText("Target properties should exist."));
102             assertProperties(assertContext, actual.getTargetProperties().get().getProperties(), expected.getTargetProperties(), "Target");
103         }
104     }
105     
106     private static void assertJobProperties(final SQLCaseAssertContext assertContext, final DorisCreateStreamingJobStatement actual, final DorisCreateStreamingJobStatementTestCase expected) {
107         if (!expected.getJobProperties().isEmpty()) {
108             assertTrue(actual.getJobProperties().isPresent(), assertContext.getText("Job properties should exist."));
109             assertProperties(assertContext, actual.getJobProperties().get().getProperties(), expected.getJobProperties(), "Job");
110         }
111     }
112     
113     private static void assertInsertStatement(final SQLCaseAssertContext assertContext, final DorisCreateStreamingJobStatement actual, final DorisCreateStreamingJobStatementTestCase expected) {
114         if (null != expected.getInsertTable()) {
115             assertNotNull(actual.getInsertStatement(), assertContext.getText("Insert statement should exist."));
116             assertTrue(actual.getInsertStatement().getTable().isPresent(), assertContext.getText("Insert target table should exist."));
117             TableAssert.assertIs(assertContext, actual.getInsertStatement().getTable().get(), expected.getInsertTable());
118         }
119         if (null != expected.getInsertSelectTable()) {
120             assertNotNull(actual.getInsertStatement(), assertContext.getText("Insert statement should exist."));
121             assertTrue(actual.getInsertStatement().getInsertSelect().isPresent(), assertContext.getText("Insert select should exist."));
122             SimpleTableSegment fromTable = (SimpleTableSegment) actual.getInsertStatement().getInsertSelect().get().getSelect().getFrom().orElse(null);
123             assertNotNull(fromTable, assertContext.getText("Insert select from table should exist."));
124             TableAssert.assertIs(assertContext, fromTable, expected.getInsertSelectTable());
125         }
126     }
127     
128     private static void assertProperties(final SQLCaseAssertContext assertContext, final List<PropertySegment> actual, final List<PropertyTestCase> expected, final String type) {
129         assertThat(assertContext.getText(type + " properties size does not match: "), actual.size(), is(expected.size()));
130         for (int i = 0; i < expected.size(); i++) {
131             assertProperty(assertContext, actual.get(i), expected.get(i));
132         }
133     }
134     
135     private static void assertProperty(final SQLCaseAssertContext assertContext, final PropertySegment actual, final PropertyTestCase expected) {
136         assertThat(assertContext.getText(String.format("Property key '%s' assertion error: ", expected.getKey())), actual.getKey(), is(expected.getKey()));
137         assertThat(assertContext.getText(String.format("Property value for key '%s' assertion error: ", expected.getKey())), actual.getValue(), is(expected.getValue()));
138         SQLSegmentAssert.assertIs(assertContext, actual, expected);
139     }
