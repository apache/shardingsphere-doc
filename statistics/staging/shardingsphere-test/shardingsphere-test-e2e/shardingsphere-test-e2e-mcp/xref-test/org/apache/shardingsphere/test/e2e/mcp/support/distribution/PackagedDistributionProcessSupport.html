18  package org.apache.shardingsphere.test.e2e.mcp.support.distribution;
19  
20  import lombok.Getter;
21  import org.apache.shardingsphere.test.e2e.mcp.support.artifact.MCPArtifactUtils;
22  import org.apache.shardingsphere.test.e2e.mcp.support.distribution.PackagedDistributionTestSupport.PreparedPackagedDistribution;
23  
24  import java.io.BufferedReader;
25  import java.io.IOException;
26  import java.io.InputStreamReader;
27  import java.nio.charset.StandardCharsets;
28  import java.nio.file.Files;
29  import java.nio.file.Path;
30  import java.util.List;
31  import java.util.Locale;
32  import java.util.concurrent.CopyOnWriteArrayList;
33  import java.util.concurrent.TimeUnit;
34  
35  /**
36   * Packaged distribution process support.
37   */
38  public final class PackagedDistributionProcessSupport implements AutoCloseable {
39      
40      private static final long PROCESS_STOP_TIMEOUT_SECONDS = 5L;
41      
42      private static final String WINDOWS_OS_NAME_PREFIX = "windows";
43      
44      private final Path distributionHome;
45      
46      private final Path configFile;
47      
48      @SuppressWarnings("UseOfProcessBuilder")
49      private final ProcessBuilder processBuilder;
50      
51      private final String outputCollectorThreadName;
52      
53      @Getter
54      private final List<String> outputMessages = new CopyOnWriteArrayList<>();
55      
56      private Process process;
57      
58      private Thread outputCollector;
59      
60      public PackagedDistributionProcessSupport(final PreparedPackagedDistribution distribution, final String outputCollectorThreadName) {
61          distributionHome = distribution.home();
62          configFile = distribution.configFile();
63          processBuilder = createProcessBuilder(distributionHome, configFile);
64          processBuilder.redirectErrorStream(true);
65          this.outputCollectorThreadName = outputCollectorThreadName;
66      }
67      
68      /**
69       * Create a process builder for starting packaged distribution scripts.
70       *
71       * @param distributionHome packaged distribution home
72       * @param configFile runtime config file
73       * @return process builder
74       */
75      public static ProcessBuilder createProcessBuilder(final Path distributionHome, final Path configFile) {
76          @SuppressWarnings("UseOfProcessBuilder")
77          ProcessBuilder result = new ProcessBuilder(createCommand(distributionHome, configFile, System.getProperty("os.name", "")));
78          result.directory(distributionHome.toFile());
79          return result;
80      }
81      
82      private static List<String> createCommand(final Path distributionHome, final Path configFile, final String osName) {
83          Path startScript = resolveStartScript(distributionHome, osName);
84          return isWindows(osName) ? List.of("cmd", "/c", startScript.toString(), configFile.toString()) : List.of(startScript.toString(), configFile.toString());
85      }
86      
87      private static Path resolveStartScript(final Path distributionHome, final String osName) {
88          return distributionHome.resolve(isWindows(osName) ? "bin/start.bat" : "bin/start.sh");
89      }
90      
91      /**
92       * Start the packaged distribution process when needed.
93       *
94       * @throws IOException IO exception
95       */
96      public void startIfNeeded() throws IOException {
97          if (null == process) {
98              prepareRuntimeLayout(distributionHome, configFile);
99              process = processBuilder.start();
100             outputCollector = startOutputCollector(process, outputCollectorThreadName, outputMessages);
101         }
102     }
103     
104     public boolean isAlive() {
105         return null != process && process.isAlive();
106     }
107     
108     @Override
109     public void close() {
110         if (null != process) {
111             process.destroy();
112             try {
113                 if (!process.waitFor(PROCESS_STOP_TIMEOUT_SECONDS, TimeUnit.SECONDS)) {
114                     process.destroyForcibly();
115                     process.waitFor(PROCESS_STOP_TIMEOUT_SECONDS, TimeUnit.SECONDS);
116                 }
117                 if (null != outputCollector) {
118                     outputCollector.join(TimeUnit.SECONDS.toMillis(PROCESS_STOP_TIMEOUT_SECONDS));
119                 }
120             } catch (final InterruptedException ignored) {
121                 Thread.currentThread().interrupt();
122             } finally {
123                 MCPArtifactUtils.writeRuntimeLogIfConfigured(outputCollectorThreadName + "-process-", outputMessages);
124                 MCPArtifactUtils.copyRuntimeLogIfConfigured(outputCollectorThreadName + "-server-", distributionHome.resolve("logs/mcp.log"));
125                 process = null;
126                 outputCollector = null;
127                 outputMessages.clear();
128             }
129         }
130     }
131     
132     private Thread startOutputCollector(final Process process, final String threadName, final List<String> outputMessages) {
133         Thread result = new Thread(() -> collectOutput(process, outputMessages), threadName);
134         result.setDaemon(true);
135         result.start();
136         return result;
137     }
138     
139     /**
140      * Prepare packaged distribution runtime layout before starting a process.
141      *
142      * @param distributionHome packaged distribution home
143      * @param configFile runtime config file
144      * @throws IOException I/O exception
145      */
146     public static void prepareRuntimeLayout(final Path distributionHome, final Path configFile) throws IOException {
147         verifyConfigurationFile(configFile);
148         verifyRuntimeLibraries(distributionHome.resolve("lib"));
149         Files.createDirectories(distributionHome.resolve("data"));
150         Files.createDirectories(distributionHome.resolve("logs"));
151         Files.createDirectories(distributionHome.resolve("plugins"));
152     }
153     
154     private static void verifyConfigurationFile(final Path configFile) throws IOException {
155         if (Files.isRegularFile(configFile)) {
156             return;
157         }
158         throw new IOException("MCP configuration file `" + configFile + "` does not exist.");
159     }
160     
161     private static void verifyRuntimeLibraries(final Path libraryDirectory) throws IOException {
162         if (Files.isDirectory(libraryDirectory)) {
163             return;
164         }
165         throw new IOException("MCP runtime libraries are missing under `" + libraryDirectory + "`.");
166     }
167     
168     private static boolean isWindows(final String osName) {
169         return osName.toLowerCase(Locale.ENGLISH).startsWith(WINDOWS_OS_NAME_PREFIX);
170     }
171     
172     private void collectOutput(final Process process, final List<String> outputMessages) {
173         try (BufferedReader reader = new BufferedReader(new InputStreamReader(process.getInputStream(), StandardCharsets.UTF_8))) {
174             String line;
175             while (null != (line = reader.readLine())) {
176                 outputMessages.add(line);
177             }
178         } catch (final IOException ignored) {
179         }
180     }
181 }
