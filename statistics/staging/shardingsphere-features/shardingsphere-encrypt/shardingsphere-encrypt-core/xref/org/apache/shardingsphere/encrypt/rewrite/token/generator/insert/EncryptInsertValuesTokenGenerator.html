62  /**
63   * Insert values token generator for encrypt.
64   */
65  @RequiredArgsConstructor
66  @Setter
67  public final class EncryptInsertValuesTokenGenerator implements OptionalSQLTokenGenerator<InsertStatementContext>, PreviousSQLTokensAware {
68      
69      private static final Pattern TEMPORAL_LITERAL_PATTERN = Pattern.compile("^\\s*(?:\\{\\s*)?(?:DATE|TIME|TIMESTAMP)\\s*'([^']*)'\\s*}?\\s*$", Pattern.CASE_INSENSITIVE);
70      
71      private static final String DEFAULT_EXPRESSION = "DEFAULT";
72      
73      private final EncryptRule rule;
74      
75      private final ShardingSphereDatabase database;
76      
77      private final SQLRewriteContext sqlRewriteContext;
78      
79      private List<SQLToken> previousSQLTokens;
80      
81      @Override
82      public boolean isGenerateSQLToken(final SQLStatementContext sqlStatementContext) {
83          return sqlStatementContext instanceof InsertStatementContext && !((InsertStatementContext) sqlStatementContext).getSqlStatement().getValues().isEmpty();
84      }
85      
86      @Override
87      public InsertValuesToken generateSQLToken(final InsertStatementContext insertStatementContext) {
88          Optional<SQLToken> insertValuesToken = findPreviousSQLToken(InsertValuesToken.class);
89          if (insertValuesToken.isPresent()) {
90              processPreviousSQLToken(insertStatementContext, (InsertValuesToken) insertValuesToken.get());
91              return (InsertValuesToken) insertValuesToken.get();
92          }
93          return generateNewSQLToken(insertStatementContext);
94      }
95      
96      private Optional<SQLToken> findPreviousSQLToken(final Class<?> sqlToken) {
97          for (SQLToken each : previousSQLTokens) {
98              if (sqlToken.isAssignableFrom(each.getClass())) {
99                  return Optional.of(each);
100             }
101         }
102         return Optional.empty();
103     }
104     
105     private void processPreviousSQLToken(final InsertStatementContext insertStatementContext, final InsertValuesToken insertValuesToken) {
106         String tableName = insertStatementContext.getSqlStatement().getTable().map(optional -> optional.getTableName().getIdentifier().getValue()).orElse("");
107         EncryptTable encryptTable = rule.getEncryptTable(tableName);
108         int count = 0;
109         String schemaName = insertStatementContext.getTablesContext().getSchemaName().orElseGet(database::getDefaultSchemaName);
110         for (InsertValueContext each : insertStatementContext.getInsertValueContexts()) {
111             applyInsertColumnTokens(insertValuesToken.getInsertValues().get(count), encryptToken(schemaName, encryptTable, insertStatementContext, each));
112             count++;
113         }
114     }
115     
116     private InsertValuesToken generateNewSQLToken(final InsertStatementContext insertStatementContext) {
117         String tableName = insertStatementContext.getSqlStatement().getTable().map(optional -> optional.getTableName().getIdentifier().getValue()).orElse("");
118         Collection<InsertValuesSegment> insertValuesSegments = insertStatementContext.getSqlStatement().getValues();
119         InsertValuesToken result = new EncryptInsertValuesToken(getStartIndex(insertValuesSegments), getStopIndex(insertValuesSegments));
120         EncryptTable encryptTable = rule.getEncryptTable(tableName);
121         String schemaName = insertStatementContext.getTablesContext().getSchemaName().orElseGet(database::getDefaultSchemaName);
122         for (InsertValueContext each : insertStatementContext.getInsertValueContexts()) {
123             InsertValue insertValueToken = new InsertValue(new LinkedList<>(each.getValueExpressions()));
124             applyInsertColumnTokens(insertValueToken, encryptToken(schemaName, encryptTable, insertStatementContext, each));
125             result.getInsertValues().add(insertValueToken);
126         }
127         return result;
128     }
129     
130     private int getStartIndex(final Collection<InsertValuesSegment> segments) {
131         int result = segments.iterator().next().getStartIndex();
132         for (InsertValuesSegment each : segments) {
133             result = Math.min(result, each.getStartIndex());
134         }
135         return result;
136     }
137     
138     private int getStopIndex(final Collection<InsertValuesSegment> segments) {
139         int result = segments.iterator().next().getStopIndex();
140         for (InsertValuesSegment each : segments) {
141             result = Math.max(result, each.getStopIndex());
142         }
143         return result;
144     }
145     
146     private Collection<EncryptInsertColumnToken> encryptToken(final String schemaName, final EncryptTable encryptTable,
147                                                               final InsertStatementContext insertStatementContext, final InsertValueContext insertValueContext) {
148         Collection<EncryptInsertColumnToken> result = new LinkedList<>();
149         String tableName = encryptTable.getTable();
150         Optional<SQLToken> useDefaultInsertColumnsToken = findPreviousSQLToken(UseDefaultInsertColumnsToken.class);
151         Iterator<String> descendingColumnNames = insertStatementContext.getDescendingColumnNames();
152         while (descendingColumnNames.hasNext()) {
153             String columnName = descendingColumnNames.next();
154             if (!encryptTable.isEncryptColumn(columnName)) {
155                 continue;
156             }
157             EncryptColumn encryptColumn = rule.getEncryptTable(tableName).getEncryptColumn(columnName);
158             int columnIndex = useDefaultInsertColumnsToken
159                     .map(optional -> ((UseDefaultInsertColumnsToken) optional).getColumns().indexOf(columnName)).orElseGet(() -> insertStatementContext.getColumnNames().indexOf(columnName));
160             Object literalValue = insertValueContext.getLiteralValue(columnIndex).orElse(null);
161             ExpressionSegment valueExpression = insertValueContext.getValueExpressions().get(columnIndex);
162             int parameterIndexCount = getParameterIndexCount(insertValueContext.getValueExpressions());
163             generateCipherColumnToken(schemaName, tableName, encryptColumn, valueExpression, columnIndex, literalValue).ifPresent(result::add);
164             if (encryptColumn.getAssistedQuery().isPresent()) {
165                 addAssistedQueryColumn(schemaName, tableName, encryptColumn, valueExpression, columnIndex, literalValue, parameterIndexCount).ifPresent(result::add);
166             }
167             if (encryptColumn.getLikeQuery().isPresent()) {
168                 addLikeQueryColumn(schemaName, tableName, encryptColumn, valueExpression, columnIndex, literalValue, parameterIndexCount).ifPresent(result::add);
169             }
170         }
171         return result;
172     }
173     
174     private void applyInsertColumnTokens(final InsertValue insertValueToken, final Collection<EncryptInsertColumnToken> insertColumnTokens) {
175         for (EncryptInsertColumnToken each : insertColumnTokens) {
176             if (each instanceof Substitutable) {
177                 insertValueToken.putSubstitutedSQLToken(each.getColumnIndex(), each);
178                 continue;
179             }
180             if (each instanceof Attachable) {
181                 insertValueToken.addAddedSQLToken(each.getColumnIndex(), each);
182             }
183         }
184     }
185     
186     private Optional<EncryptInsertColumnToken> generateCipherColumnToken(final String schemaName, final String tableName, final EncryptColumn encryptColumn,
187                                                                          final ExpressionSegment valueExpression, final int columnIndex, final Object originalValue) {
188         if (isDefaultExpression(valueExpression)) {
189             return Optional.of(new EncryptInsertSubstitutableColumnToken(columnIndex, createDefaultExpression(valueExpression)));
190         }
191         if (isLiteralExpression(valueExpression)) {
192             return Optional.of(new EncryptInsertSubstitutableColumnToken(columnIndex, new LiteralExpressionSegment(
193                     valueExpression.getStartIndex(), valueExpression.getStopIndex(),
194                     encryptColumn.getCipher().encrypt(database.getName(), schemaName, tableName, encryptColumn.getName(), originalValue))));
195         }
196         if (valueExpression instanceof ColumnSegment) {
197             return Optional.of(new EncryptInsertSubstitutableColumnToken(columnIndex, createColumnSegment((ColumnSegment) valueExpression, encryptColumn.getCipher().getName())));
198         }
199         return Optional.empty();
200     }
201     
202     private Optional<String> getOriginalSQLExpressionText(final ExpressionSegment valueExpression) {
203         if (null == sqlRewriteContext || valueExpression.getStartIndex() < 0 || valueExpression.getStopIndex() >= sqlRewriteContext.getSql().length()) {
204             return Optional.empty();
205         }
206         return Optional.of(sqlRewriteContext.getSql().substring(valueExpression.getStartIndex(), valueExpression.getStopIndex() + 1));
207     }
208     
209     private Optional<EncryptInsertColumnToken> addAssistedQueryColumn(final String schemaName, final String tableName, final EncryptColumn encryptColumn,
210                                                                       final ExpressionSegment valueExpression, final int columnIndex, final Object originalValue, final int parameterIndexCount) {
211         Optional<AssistedQueryColumnItem> assistedQueryColumnItem = encryptColumn.getAssistedQuery();
212         Preconditions.checkState(assistedQueryColumnItem.isPresent());
213         Object derivedValue = assistedQueryColumnItem.get().encrypt(database.getName(), schemaName, tableName, encryptColumn.getName(), originalValue);
214         return addDerivedColumn(valueExpression, columnIndex, derivedValue, assistedQueryColumnItem.get().getName(), parameterIndexCount);
215     }
216     
217     private Optional<EncryptInsertColumnToken> addLikeQueryColumn(final String schemaName, final String tableName, final EncryptColumn encryptColumn,
218                                                                   final ExpressionSegment valueExpression, final int columnIndex, final Object originalValue, final int parameterIndexCount) {
219         Optional<LikeQueryColumnItem> likeQueryColumnItem = encryptColumn.getLikeQuery();
220         Preconditions.checkState(likeQueryColumnItem.isPresent());
221         Object derivedValue = likeQueryColumnItem.get().encrypt(database.getName(), schemaName, tableName, encryptColumn.getName(), originalValue);
222         return addDerivedColumn(valueExpression, columnIndex, derivedValue, likeQueryColumnItem.get().getName(), parameterIndexCount);
223     }
224     
225     private Optional<EncryptInsertColumnToken> addDerivedColumn(final ExpressionSegment valueExpression, final int columnIndex, final Object derivedValue,
226                                                                 final String derivedColumnName, final int parameterIndexCount) {
227         if (valueExpression instanceof LiteralExpressionSegment) {
228             return Optional.of(new EncryptInsertAttachableColumnToken(columnIndex, new LiteralExpressionSegment(0, 0, derivedValue)));
229         }
230         if (valueExpression instanceof ParameterMarkerExpressionSegment) {
231             return Optional.of(new EncryptInsertAttachableColumnToken(columnIndex, new ParameterMarkerExpressionSegment(0, 0, parameterIndexCount)));
232         }
233         if (valueExpression instanceof ColumnSegment) {
234             return Optional.of(new EncryptInsertAttachableColumnToken(columnIndex, createColumnSegment((ColumnSegment) valueExpression, derivedColumnName)));
235         }
236         return Optional.of(new EncryptInsertAttachableColumnToken(columnIndex, valueExpression));
237     }
238     
239     private boolean isDefaultExpression(final ExpressionSegment valueExpression) {
240         return null != valueExpression.getText() && DEFAULT_EXPRESSION.equalsIgnoreCase(valueExpression.getText().trim());
241     }
242     
243     private ExpressionSegment createDefaultExpression(final ExpressionSegment valueExpression) {
244         return new CommonExpressionSegment(valueExpression.getStartIndex(), valueExpression.getStopIndex(), DEFAULT_EXPRESSION, Collections.emptyList());
245     }
246     
247     private boolean isLiteralExpression(final ExpressionSegment valueExpression) {
248         return valueExpression instanceof LiteralExpressionSegment || isTemporalLiteral(valueExpression);
249     }
250     
251     private boolean isTemporalLiteral(final ExpressionSegment valueExpression) {
252         return getTemporalLiteralValue(valueExpression).isPresent();
253     }
254     
255     private Optional<String> getTemporalLiteralValue(final ExpressionSegment valueExpression) {
256         return getTemporalLiteralExpressionText(valueExpression).flatMap(this::getTemporalLiteralValue);
257     }
258     
259     private Optional<String> getTemporalLiteralValue(final String expressionText) {
260         if (null == expressionText) {
261             return Optional.empty();
262         }
263         Matcher matcher = TEMPORAL_LITERAL_PATTERN.matcher(expressionText);
264         return matcher.matches() ? Optional.of(matcher.group(1)) : Optional.empty();
265     }
266     
267     private Optional<String> getTemporalLiteralExpressionText(final ExpressionSegment valueExpression) {
268         if (getTemporalLiteralValue(valueExpression.getText()).isPresent()) {
269             return Optional.of(valueExpression.getText());
270         }
271         return getOriginalSQLExpressionText(valueExpression).filter(optional -> getTemporalLiteralValue(optional).isPresent());
272     }
273     
274     private int getParameterIndexCount(final Collection<ExpressionSegment> expressionSegments) {
275         int result = 0;
276         for (ExpressionSegment each : expressionSegments) {
277             if (each instanceof ParameterMarkerExpressionSegment) {
278                 result++;
279             }
280         }
281         return result;
282     }
283     
284     private ColumnSegment createColumnSegment(final ColumnSegment originalColumn, final String columnName) {
285         ColumnSegment result = new ColumnSegment(originalColumn.getStartIndex(), originalColumn.getStopIndex(), new IdentifierValue(columnName, originalColumn.getIdentifier().getQuoteCharacter()));
286         result.setNestedObjectAttributes(originalColumn.getNestedObjectAttributes());
287         originalColumn.getOwner().ifPresent(result::setOwner);
288         result.setColumnBoundInfo(originalColumn.getColumnBoundInfo());
289         result.setOtherUsingColumnBoundInfo(originalColumn.getOtherUsingColumnBoundInfo());
290         result.setVariable(originalColumn.isVariable());
291         return result;
292     }
