99  import java.util.function.Consumer;

101 @RequiredArgsConstructor
102 final class SQLStatementTreeWalker {
103     
104     private final Consumer<SQLStatement> statementConsumer;
105     
106     private final Consumer<ExpressionSegment> expressionConsumer;
107     
108     void walk(final SQLStatement sqlStatement) {
109         if (null == sqlStatement) {
110             return;
111         }
112         statementConsumer.accept(sqlStatement);
113         if (sqlStatement instanceof SelectStatement select) {
114             walkSelect(select);
115         } else if (sqlStatement instanceof InsertStatement insert) {
116             walkInsert(insert);
117         } else if (sqlStatement instanceof UpdateStatement update) {
118             walkUpdate(update);
119         } else if (sqlStatement instanceof DeleteStatement delete) {
120             walkDelete(delete);
121         } else if (sqlStatement instanceof MergeStatement merge) {
122             walkMerge(merge);
123         } else {
124             walkDDL(sqlStatement);
125         }
126     }
127     
128     private void walkDDL(final SQLStatement sqlStatement) {
129         if (sqlStatement instanceof CreateTableStatement createTableStatement) {
130             createTableStatement.getSelectStatement().ifPresent(this::walk);
131         } else if (sqlStatement instanceof CreateViewStatement createViewStatement) {
132             walk(createViewStatement.getSelect());
133         } else if (sqlStatement instanceof AlterViewStatement alterViewStatement) {
134             alterViewStatement.getSelect().ifPresent(this::walk);
135         } else if (sqlStatement instanceof CreateMaterializedViewStatement createMaterializedViewStatement) {
136             walk(createMaterializedViewStatement.getSelect());
137         } else if (sqlStatement instanceof CreateFunctionStatement createFunctionStatement) {
138             walkRoutine(createFunctionStatement.getSqlStatements(), createFunctionStatement.getDynamicSqlStatementExpressions());
139         } else if (sqlStatement instanceof CreateProcedureStatement createProcedureStatement) {
140             walkRoutine(createProcedureStatement.getSqlStatements(), createProcedureStatement.getDynamicSqlStatementExpressions());
141         } else if (sqlStatement instanceof CreateTriggerStatement createTriggerStatement) {
142             walkRoutine(createTriggerStatement.getSqlStatements(), createTriggerStatement.getDynamicSqlStatementExpressions());
143         }
144     }
145     
146     private void walkRoutine(final Iterable<SQLStatementSegment> sqlStatements, final Iterable<ExpressionSegment> expressions) {
147         sqlStatements.forEach(each -> walk(each.getSqlStatement()));
148         expressions.forEach(this::walkExpression);
149     }
150     
151     private void walkSelect(final SelectStatement sqlStatement) {
152         walkProjections(sqlStatement.getProjections());
153         sqlStatement.getWhere().ifPresent(each -> walkExpression(each.getExpr()));
154         sqlStatement.getHaving().ifPresent(each -> walkExpression(each.getExpr()));
155         sqlStatement.getGroupBy().ifPresent(each -> walkOrderByItems(each.getGroupByItems()));
156         sqlStatement.getOrderBy().ifPresent(each -> walkOrderByItems(each.getOrderByItems()));
157         sqlStatement.getFrom().ifPresent(this::walkTable);
158         sqlStatement.getHierarchicalQuery().ifPresent(each -> {
159             walkExpression(each.getStartWith());
160             walkExpression(each.getConnectBy());
161         });
162         sqlStatement.getCombine().ifPresent(this::walkCombine);
163         sqlStatement.getWith().ifPresent(this::walkWith);
164         sqlStatement.getLimit().ifPresent(this::walkLimit);
165         sqlStatement.getWindow().ifPresent(each -> each.getItemSegments().forEach(this::walkWindow));
166         sqlStatement.getModel().ifPresent(this::walkModel);
167         sqlStatement.getTransformSegments().forEach(this::walkExpression);
168     }
169     
170     private void walkInsert(final InsertStatement sqlStatement) {
171         sqlStatement.getInsertSelect().ifPresent(each -> walk(each.getSelect()));
172         sqlStatement.getSetAssignment().ifPresent(each -> each.getAssignments().forEach(assignment -> walkExpression(assignment.getValue())));
173         sqlStatement.getOnDuplicateKeyColumns().ifPresent(each -> each.getColumns().forEach(assignment -> walkExpression(assignment.getValue())));
174         sqlStatement.getValues().forEach(each -> each.getValues().forEach(this::walkExpression));
175         sqlStatement.getWhere().ifPresent(each -> walkExpression(each.getExpr()));
176         sqlStatement.getExec().ifPresent(each -> each.getExpressionSegments().forEach(this::walkExpression));
177         sqlStatement.getRowSetFunction().ifPresent(this::walkExpression);
178         sqlStatement.getMultiTableInsertInto().ifPresent(each -> each.getInsertStatements().forEach(this::walk));
179         sqlStatement.getMultiTableConditionalInto().ifPresent(this::walkMultiTableConditionalInto);
180         sqlStatement.getWith().ifPresent(this::walkWith);
181         sqlStatement.getReturning().ifPresent(each -> walkProjections(each.getProjections()));
182         sqlStatement.getOutput().ifPresent(this::walkOutput);
183     }
184     
185     private void walkMultiTableConditionalInto(final MultiTableConditionalIntoSegment segment) {
186         for (MultiTableConditionalIntoWhenThenSegment each : segment.getWhenThenSegments()) {
187             walkExpression(each.getWhenSegment());
188             each.getThenSegment().getInsertStatements().forEach(this::walk);
189         }
190         segment.getElseSegment().ifPresent(each -> each.getInsertStatements().forEach(this::walk));
191     }
192     
193     private void walkUpdate(final UpdateStatement sqlStatement) {
194         walkTable(sqlStatement.getTable());
195         sqlStatement.getAssignment().ifPresent(each -> each.getAssignments().forEach(assignment -> walkExpression(assignment.getValue())));
196         sqlStatement.getWhere().ifPresent(each -> walkExpression(each.getExpr()));
197         sqlStatement.getDeleteWhere().ifPresent(each -> walkExpression(each.getExpr()));
198         sqlStatement.getOrderBy().ifPresent(each -> walkOrderByItems(each.getOrderByItems()));
199         sqlStatement.getLimit().ifPresent(this::walkLimit);
200         sqlStatement.getFrom().ifPresent(this::walkTable);
201         sqlStatement.getWith().ifPresent(this::walkWith);
202         sqlStatement.getReturning().ifPresent(each -> walkProjections(each.getProjections()));
203         sqlStatement.getOutput().ifPresent(this::walkOutput);
204     }
205     
206     private void walkDelete(final DeleteStatement sqlStatement) {
207         walkTable(sqlStatement.getTable());
208         sqlStatement.getWhere().ifPresent(each -> walkExpression(each.getExpr()));
209         sqlStatement.getOrderBy().ifPresent(each -> walkOrderByItems(each.getOrderByItems()));
210         sqlStatement.getLimit().ifPresent(this::walkLimit);
211         sqlStatement.getWith().ifPresent(this::walkWith);
212         sqlStatement.getReturning().ifPresent(each -> walkProjections(each.getProjections()));
213         sqlStatement.getOutput().ifPresent(this::walkOutput);
214     }
215     
216     private void walkMerge(final MergeStatement sqlStatement) {
217         walkTable(sqlStatement.getTarget());
218         walkTable(sqlStatement.getSource());
219         if (null != sqlStatement.getExpression()) {
220             walkExpression(sqlStatement.getExpression().getExpr());
221         }
222         sqlStatement.getUpdate().ifPresent(this::walk);
223         sqlStatement.getInsert().ifPresent(this::walk);
224         for (MergeWhenAndThenSegment each : sqlStatement.getWhenAndThens()) {
225             walkExpression(each.getAndExpr());
226             if (null != each.getUpdate()) {
227                 walk(each.getUpdate());
228             }
229             if (null != each.getInsert()) {
230                 walk(each.getInsert());
231             }
232         }
233         sqlStatement.getWith().ifPresent(this::walkWith);
234         sqlStatement.getOutput().ifPresent(this::walkOutput);
235     }
236     
237     private void walkWith(final WithSegment with) {
238         with.getCommonTableExpressions().forEach(each -> {
239             ShardingSpherePreconditions.checkState(null == each.getDataModification(), MCPUnsupportedSQLStatementException::new);
240             walk(each.getSubquery().getSelect());
241         });
242     }
243     
244     private void walkCombine(final CombineSegment combine) {
245         walk(combine.getLeft().getSelect());
246         walk(combine.getRight().getSelect());
247     }
248     
249     private void walkProjections(final ProjectionsSegment projections) {
250         if (null == projections) {
251             return;
252         }
253         for (ProjectionSegment each : projections.getProjections()) {
254             if (each instanceof ExpressionProjectionSegment expressionProjection) {
255                 walkExpression(expressionProjection.getExpr());
256             } else if (each instanceof ExpressionSegment expression) {
257                 walkExpression(expression);
258             } else if (each instanceof ColumnProjectionSegment columnProjection) {
259                 walkExpression(columnProjection.getColumn());
260             } else if (each instanceof SubqueryProjectionSegment subqueryProjection) {
261                 walk(subqueryProjection.getSubquery().getSelect());
262             } else if (each instanceof DatetimeProjectionSegment datetimeProjection) {
263                 walkExpression(datetimeProjection.getLeft());
264                 walkExpression(datetimeProjection.getRight());
265             }
266         }
267     }
268     
269     private void walkOrderByItems(final Iterable<OrderByItemSegment> items) {
270         for (OrderByItemSegment each : items) {
271             if (each instanceof ExpressionOrderByItemSegment expressionOrderByItem) {
272                 walkExpression(expressionOrderByItem.getExpr());
273             }
274         }
275     }
276     
277     private void walkLimit(final LimitSegment limit) {
278         limit.getOffset().ifPresent(this::walkPaginationValue);
279         limit.getRowCount().ifPresent(this::walkPaginationValue);
280     }
281     
282     private void walkPaginationValue(final PaginationValueSegment value) {
283         if (value instanceof ExpressionPaginationValueSegment expressionPaginationValue) {
284             walkExpression(expressionPaginationValue.getExpression());
285         }
286     }
287     
288     private void walkOutput(final OutputSegment output) {
289         walkProjections(output.getOutputColumns());
290     }
291     
292     private void walkWindow(final WindowItemSegment window) {
293         if (null != window.getPartitionListSegments()) {
294             window.getPartitionListSegments().forEach(this::walkExpression);
295         }
296         if (null != window.getOrderBySegment()) {
297             walkOrderByItems(window.getOrderBySegment().getOrderByItems());
298         }
299         walkExpression(window.getFrameClause());
300     }
301     
302     private void walkModel(final ModelSegment model) {
303         model.getReferenceModelSelects().forEach(each -> walk(each.getSelect()));
304         model.getCellAssignmentSelects().forEach(each -> walk(each.getSelect()));
305         model.getOrderBySegments().forEach(each -> walkOrderByItems(each.getOrderByItems()));
306         model.getPartitionColumns().forEach(this::walkModelColumn);
307         model.getDimensionColumns().forEach(this::walkModelColumn);
308         model.getMeasureColumns().forEach(this::walkModelColumn);
309         model.getCellAssignmentColumns().forEach(this::walkExpression);
310     }
311     
312     private void walkModelColumn(final ModelColumnSegment modelColumn) {
313         walkExpression(modelColumn.getExpression());
314     }
315     
316     private void walkTable(final TableSegment table) {
317         if (table instanceof SimpleTableSegment simpleTable) {
318             simpleTable.getTableSampleExpression().ifPresent(this::walkExpression);
319         } else if (table instanceof FunctionTableSegment functionTable) {
320             walkExpression(functionTable.getTableFunction());
321         } else if (table instanceof JoinTableSegment join) {
322             walkTable(join.getLeft());
323             walkTable(join.getRight());
324             walkExpression(join.getCondition());
325             join.getLeftQueryPartitionListSegments().forEach(this::walkExpression);
326             join.getRightQueryPartitionListSegments().forEach(this::walkExpression);
327         } else if (table instanceof SubqueryTableSegment subqueryTable) {
328             walk(subqueryTable.getSubquery().getSelect());
329         } else if (table instanceof CollectionTableSegment collectionTable) {
330             walkExpression(collectionTable.getExpressionSegment());
331         } else if (table instanceof DeleteMultiTableSegment deleteMultiTable) {
332             walkTable(deleteMultiTable.getRelationTable());
333         }
334     }
335     
336     private void walkExpression(final ExpressionSegment expression) {
337         if (null == expression) {
338             return;
339         }
340         expressionConsumer.accept(expression);
341         switch (expression) {
342             case FunctionSegment function -> walkFunction(function);
343             case CommonExpressionSegment common -> common.getChildren().forEach(this::walkExpression);
344             case AggregationProjectionSegment aggregation -> {
345                 aggregation.getParameters().forEach(this::walkExpression);
346                 aggregation.getWindow().ifPresent(this::walkWindow);
347             }
348             case BinaryOperationExpression binary -> {
349                 walkExpression(binary.getLeft());
350                 walkExpression(binary.getRight());
351             }
352             case ColumnAssignmentSegment assignment -> walkExpression(assignment.getValue());
353             case ListExpression list -> list.getItems().forEach(this::walkExpression);
354             case BetweenExpression between -> {
355                 walkExpression(between.getLeft());
356                 walkExpression(between.getBetweenExpr());
357                 walkExpression(between.getAndExpr());
358             }
359             case InExpression in -> {
360                 walkExpression(in.getLeft());
361                 walkExpression(in.getRight());
362             }
363             case CaseWhenExpression caseWhen -> walkCaseWhen(caseWhen);
364             case NotExpression not -> walkExpression(not.getExpression());
365             case TypeCastExpression typeCast -> walkExpression(typeCast.getExpression());
366             case CollateExpression collate -> collate.getExpr().ifPresent(this::walkExpression);
367             case DatetimeExpression datetime -> {
368                 walkExpression(datetime.getLeft());
369                 walkExpression(datetime.getRight());
370             }
371             case IntervalExpression interval -> walkExpression(interval.getValue());
372             case IntervalExpressionProjection intervalProjection -> walkIntervalProjection(intervalProjection);
373             case KeyValueSegment keyValue -> {
374                 walkExpression(keyValue.getKey());
375                 walkExpression(keyValue.getValue());
376             }
377             case RowExpression row -> row.getItems().forEach(this::walkExpression);
378             case UnaryOperationExpression unary -> walkExpression(unary.getExpression());
379             case ValuesExpression values -> values.getRowConstructorList().forEach(each -> each.getValues().forEach(this::walkExpression));
380             case MatchAgainstExpression match -> walkMatchAgainst(match);
381             case MultisetExpression multiset -> {
382                 walkExpression(multiset.getLeft());
383                 walkExpression(multiset.getRight());
384             }
385             case OuterJoinExpression outerJoin -> walkExpression(outerJoin.getColumnName());
386             case SubquerySegment subquery -> walk(subquery.getSelect());
387             case SubqueryExpressionSegment subqueryExpression -> walk(subqueryExpression.getSubquery().getSelect());
388             case ExistsSubqueryExpression exists -> walk(exists.getSubquery().getSelect());
389             case QuantifySubqueryExpression quantify -> walk(quantify.getSubquery().getSelect());
390             default -> walkXMLExpression(expression);
391         }
392     }
393     
394     private void walkFunction(final FunctionSegment function) {
395         function.getParameters().forEach(this::walkExpression);
396         function.getWindow().ifPresent(this::walkWindow);
397     }
398     
399     private void walkCaseWhen(final CaseWhenExpression caseWhen) {
400         walkExpression(caseWhen.getCaseExpr());
401         caseWhen.getWhenExprs().forEach(this::walkExpression);
402         caseWhen.getThenExprs().forEach(this::walkExpression);
403         walkExpression(caseWhen.getElseExpr());
404     }
405     
406     private void walkIntervalProjection(final IntervalExpressionProjection interval) {
407         walkExpression(interval.getLeft());
408         walkExpression(interval.getMinus());
409         walkExpression(interval.getRight());
410     }
411     
412     private void walkMatchAgainst(final MatchAgainstExpression match) {
413         match.getColumns().forEach(this::walkExpression);
414         walkExpression(match.getExpr());
415     }
416     
417     private void walkXMLExpression(final ExpressionSegment expression) {
418         if (expression instanceof XmlElementFunctionSegment xmlElement) {
419             xmlElement.getXmlAttributes().forEach(this::walkExpression);
420             xmlElement.getParameters().forEach(this::walkExpression);
421         } else if (expression instanceof XmlPiFunctionSegment xmlPi) {
422             walkExpression(xmlPi.getEvalNameValueExpr());
423             walkExpression(xmlPi.getValueExpr());
424         } else if (expression instanceof XmlQueryAndExistsFunctionSegment xmlQueryAndExists) {
425             xmlQueryAndExists.getParameters().forEach(this::walkExpression);
426         } else if (expression instanceof XmlSerializeFunctionSegment xmlSerialize) {
427             walkExpression(xmlSerialize.getParameter());
428         } else if (expression instanceof XmlTableFunctionSegment xmlTable) {
429             walkXmlTableOptions(xmlTable.getXmlTableOption());
430         } else if (expression instanceof XmlTableOptionsSegment xmlTableOptions) {
431             walkXmlTableOptions(xmlTableOptions);
432         } else if (expression instanceof XmlTableColumnSegment xmlTableColumn) {
433             walkExpression(xmlTableColumn.getDefaultExpr());
434         }
435     }
436     
437     private void walkXmlTableOptions(final XmlTableOptionsSegment options) {
438         if (null == options) {
439             return;
440         }
441         options.getParameters().forEach(this::walkExpression);
442         options.getXmlTableColumnSegments().forEach(each -> walkExpression(each.getDefaultExpr()));
443     }
