53      private FirebirdPacketPayload payload;

56      private ByteBuf byteBuf;

58      @BeforeEach
59      void setUp() {
60          lenient().when(byteBuf.duplicate()).thenReturn(byteBuf);
61      }

63      @ParameterizedTest(name = "{0}")
64      @MethodSource("assertExecuteStatementPacketArguments")
65      void assertExecuteStatementPacket(final String name, final short blrType, final FirebirdBinaryColumnType expectedParameterType, final Object expectedParameterValue) {
66          when(payload.readInt4()).thenReturn(FirebirdCommandPacketType.EXECUTE.getValue(), 1, 2, 0, 1, 123);
67          when(payload.readInt1()).thenReturn(0);
68          if (FirebirdBinaryColumnType.BLOB == expectedParameterType) {
69              when(payload.readInt8()).thenReturn(123L);
70          } else if (FirebirdBinaryColumnType.BOOLEAN == expectedParameterType) {
71              when(payload.readInt1Unsigned()).thenReturn(1);
72          }
73          when(payload.readBuffer()).thenReturn(byteBuf);
74          when(byteBuf.isReadable()).thenReturn(true);
75          when(byteBuf.readUnsignedByte()).thenReturn((short) 5, (short) 0, blrType, (short) BlrConstants.blr_end);
76          FirebirdExecuteStatementPacket packet = new FirebirdExecuteStatementPacket(payload, FirebirdProtocolVersion.PROTOCOL_VERSION13);
77          assertThat(packet.getStatementId(), is(1));
78          assertThat(packet.getTransactionId(), is(2));
79          assertThat(packet.getParameterTypes(), is(Collections.singletonList(expectedParameterType)));
80          assertThat(packet.getParameterValues(), is(Collections.singletonList(expectedParameterValue)));
81      }
82      
83      @Test
84      void assertExecuteStatementPacketWithTimestampWithTimeZoneParameter() {
85          when(payload.readInt4()).thenReturn(FirebirdCommandPacketType.EXECUTE.getValue(), 1, 2, 0, 1, 60310, 801301234, 1559);
86          when(payload.readBuffer()).thenReturn(byteBuf);
87          when(byteBuf.isReadable()).thenReturn(true);
88          when(byteBuf.readUnsignedByte()).thenReturn((short) 5, (short) 0, (short) BlrConstants.blr_timestamp_tz, (short) BlrConstants.blr_end);
89          when(byteBuf.skipBytes(anyInt())).thenReturn(byteBuf);
90          FirebirdExecuteStatementPacket actual = new FirebirdExecuteStatementPacket(payload, FirebirdProtocolVersion.PROTOCOL_VERSION13);
91          assertThat(actual.getParameterTypes(), is(Collections.singletonList(FirebirdBinaryColumnType.TIMESTAMP_TZ)));
92          assertThat(actual.getParameterValues(), is(Collections.singletonList(OffsetDateTime.parse("2024-01-02T00:15:30.123400+02:00"))));
93      }

116     @Test
117     void assertExecuteStatementPacketForStoredProcedure() {
118         when(payload.readInt1()).thenReturn(0);
119         when(payload.readInt4()).thenReturn(FirebirdCommandPacketType.EXECUTE2.getValue(), 1, 2, 0, 1, 123, 9);
120         ByteBuf returnBlr = mock(ByteBuf.class);
121         when(payload.readBuffer()).thenReturn(byteBuf, returnBlr);
122         when(byteBuf.isReadable()).thenReturn(true);
123         when(byteBuf.readUnsignedByte()).thenReturn((short) 5, (short) 0, (short) BlrConstants.blr_long, (short) BlrConstants.blr_end);
124         when(returnBlr.isReadable()).thenReturn(true);
125         when(returnBlr.duplicate()).thenReturn(returnBlr);
126         when(returnBlr.readUnsignedByte()).thenReturn((short) 5, (short) 0, (short) BlrConstants.blr_long, (short) BlrConstants.blr_end);
127         when(payload.readInt4Unsigned()).thenReturn(30L, 1L, 1024L);
128         FirebirdExecuteStatementPacket packet = new FirebirdExecuteStatementPacket(payload, FirebirdProtocolVersion.PROTOCOL_VERSION19);
129         assertThat(packet.getReturnColumns(), is(Collections.singletonList(FirebirdBinaryColumnType.LONG)));
130         assertThat(packet.getOutputMessageNumber(), is(9));
131         assertThat(packet.getParameterValues(), is(Collections.singletonList(123)));
132         assertThat(packet.getStatementTimeout(), is(30L));
133         assertThat(packet.getCursorFlags(), is(1L));
134         assertThat(packet.getMaxBlobSize(), is(1024L));
135     }

156     @Test
157     void assertGetLength() {
158         when(payload.getByteBuf()).thenReturn(byteBuf);
159         when(payload.readInt4()).thenReturn(FirebirdCommandPacketType.EXECUTE.getValue(), 1, 2, 0, 1, 123);
160         when(payload.readInt1()).thenReturn(0);
161         when(payload.readBuffer()).thenReturn(byteBuf);
162         when(byteBuf.isReadable()).thenReturn(true);
163         when(byteBuf.readUnsignedByte()).thenReturn((short) 5, (short) 0, (short) BlrConstants.blr_long, (short) BlrConstants.blr_end);
164         when(byteBuf.readerIndex()).thenReturn(42);
165         assertThat(FirebirdExecuteStatementPacket.getLength(payload, FirebirdProtocolVersion.PROTOCOL_VERSION13), is(42));
166         verify(byteBuf).resetReaderIndex();
167     }
