18  package org.apache.shardingsphere.test.e2e.operation.transaction.cases.statement;
19  
20  import org.apache.shardingsphere.test.e2e.operation.transaction.engine.base.TransactionContainerComposer;
21  import org.apache.shardingsphere.test.e2e.operation.transaction.engine.base.TransactionTestCase;
22  import org.apache.shardingsphere.test.e2e.operation.transaction.engine.constants.TransactionTestConstants;
23  
24  import java.sql.Connection;
25  import java.sql.SQLException;
26  
27  import static org.junit.jupiter.api.Assertions.assertTrue;
28  
29  /**
30   * MySQL TCL statement transaction test case.
31   */
32  @TransactionTestCase(adapters = TransactionTestConstants.PROXY, dbTypes = TransactionTestConstants.MYSQL)
33  public final class MySQLTCLStatementTestCase extends BaseTCLStatementTransactionTestCase {
34      
35      public MySQLTCLStatementTestCase(final TransactionTestCaseParameter testCaseParam) {
36          super(testCaseParam);
37      }
38      
39      @Override
40      protected void executeTest(final TransactionContainerComposer containerComposer) throws SQLException {
41          try (
42                  Connection connection = getDataSource().getConnection();
43                  Connection queryConnection = getDataSource().getConnection()) {
44              assertRollback(connection, queryConnection);
45              assertCommit(connection, queryConnection);
46              assertBegin(connection, queryConnection);
47              assertStartTransaction(connection, queryConnection);
48              assertSetAutoCommit(connection, queryConnection);
49          }
50      }
51      
52      private void assertBegin(final Connection connection, final Connection queryConnection) throws SQLException {
53          executeWithLog(connection, "BEGIN");
54          executeWithLog(connection, "INSERT INTO account (id, balance, transaction_id) VALUES (3, 3, 3), (4, 4, 4)");
55          assertAccountBalances(queryConnection, 1, 2);
56          executeWithLog(connection, "BEGIN");
57          assertAccountBalances(queryConnection, 1, 2, 3, 4);
58          executeWithLog(connection, "INSERT INTO account (id, balance, transaction_id) VALUES (5, 5, 5), (6, 6, 6)");
59          executeWithLog(connection, "ROLLBACK");
60          assertAccountBalances(queryConnection, 1, 2, 3, 4);
61      }
62      
63      private void assertStartTransaction(final Connection connection, final Connection queryConnection) throws SQLException {
64          executeWithLog(connection, "DELETE FROM account");
65          assertAccountRowCount(queryConnection, 0);
66          executeWithLog(connection, "START TRANSACTION");
67          executeWithLog(connection, "INSERT INTO account (id, balance, transaction_id) VALUES (1, 1, 1), (2, 2, 2)");
68          assertAccountBalances(queryConnection);
69          executeWithLog(connection, "COMMIT");
70          assertAccountBalances(queryConnection, 1, 2);
71      }
72      
73      private void assertSetAutoCommit(final Connection connection, final Connection queryConnection) throws SQLException {
74          executeWithLog(connection, "DELETE FROM account");
75          assertAccountBalances(queryConnection);
76          executeWithLog(connection, "SET AUTOCOMMIT=0");
77          executeWithLog(connection, "INSERT INTO account (id, balance, transaction_id) VALUES (1, 1, 1), (2, 2, 2)");
78          assertAccountBalances(queryConnection);
79          executeWithLog(connection, "COMMIT");
80          assertAccountBalances(queryConnection, 1, 2);
81          executeWithLog(connection, "INSERT INTO account (id, balance, transaction_id) VALUES (3, 3, 3), (4, 4, 4)");
82          assertAccountBalances(queryConnection, 1, 2);
83          executeWithLog(connection, "BEGIN");
84          assertAccountBalances(queryConnection, 1, 2, 3, 4);
85          assertTrue(connection.getAutoCommit());
86          executeWithLog(connection, "INSERT INTO account (id, balance, transaction_id) VALUES (5, 5, 5), (6, 6, 6)");
87          assertAccountBalances(queryConnection, 1, 2, 3, 4);
88          executeWithLog(connection, "SET AUTOCOMMIT=1");
89          assertAccountBalances(queryConnection, 1, 2, 3, 4, 5, 6);
90          assertTrue(connection.getAutoCommit());
91      }
92  }
