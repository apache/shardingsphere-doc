23  import org.apache.shardingsphere.mcp.api.exception.MCPInvalidRequestException;
24  import org.apache.shardingsphere.mcp.core.protocol.exception.MCPWorkflowStateException;
25  import org.apache.shardingsphere.mcp.support.workflow.model.WorkflowContextSnapshot;
26  import org.apache.shardingsphere.mcp.support.workflow.model.WorkflowKind;
27  import org.apache.shardingsphere.mcp.support.workflow.spi.MCPWorkflowDefinitionProvider;
28  import org.apache.shardingsphere.mcp.support.workflow.spi.WorkflowRuntimeDefinition;
29  
30  import java.util.Collection;
31  import java.util.LinkedHashMap;
32  import java.util.LinkedList;
33  import java.util.Map;
34  import java.util.Objects;
35  import java.util.Optional;
36  
37  /**
38   * Workflow runtime definition registry.
39   */
40  public final class WorkflowRuntimeDefinitionRegistry {
41      
42      private final Map<WorkflowKind, WorkflowRuntimeDefinition> registeredDefinitions;
43      
44      public WorkflowRuntimeDefinitionRegistry(final Collection<WorkflowRuntimeDefinition> definitions) {
45          registeredDefinitions = createRegisteredDefinitions(definitions);
46      }
47      
48      private static Map<WorkflowKind, WorkflowRuntimeDefinition> createRegisteredDefinitions(final Collection<WorkflowRuntimeDefinition> definitions) {
49          Map<WorkflowKind, WorkflowRuntimeDefinition> result = new LinkedHashMap<>(Math.max(definitions.size(), 1), 1F);
50          for (WorkflowRuntimeDefinition each : definitions) {
51              Objects.requireNonNull(each, "Workflow definition is required.");
52              WorkflowKind workflowKind = Objects.requireNonNull(each.getWorkflowKind(),
53                      () -> String.format("Workflow kind is required for `%s`.", each.getClass().getName()));
54              ShardingSpherePreconditions.checkNotNull(each.getRuntimeHandler(),
55                      () -> new IllegalArgumentException(String.format("Workflow runtime handler is required for `%s`.", workflowKind.getValue())));
56              ShardingSpherePreconditions.checkNotNull(each.getApplyArtifactValidator(),
57                      () -> new IllegalArgumentException(String.format("Workflow apply artifact validator is required for `%s`.", workflowKind.getValue())));
58              WorkflowRuntimeDefinition previousDefinition = result.putIfAbsent(workflowKind, each);
59              ShardingSpherePreconditions.checkState(null == previousDefinition,
60                      () -> new IllegalArgumentException(String.format("Duplicate workflow kind `%s` with `%s` and `%s`.",
61                              workflowKind.getValue(), previousDefinition.getClass().getName(), each.getClass().getName())));
62          }
63          return result;
64      }
65      
66      /**
67       * Load workflow runtime definitions from providers.
68       *
69       * @return workflow runtime definition registry
70       */
71      public static WorkflowRuntimeDefinitionRegistry load() {
72          return new WorkflowRuntimeDefinitionRegistry(loadDefinitions());
73      }
74      
75      private static Collection<WorkflowRuntimeDefinition> loadDefinitions() {
76          Collection<WorkflowRuntimeDefinition> result = new LinkedList<>();
77          for (MCPHandlerProvider each : ShardingSphereServiceLoader.getServiceInstances(MCPHandlerProvider.class)) {
78              if (each instanceof MCPWorkflowDefinitionProvider workflowDefinitionProvider) {
79                  result.addAll(createDefinitions(workflowDefinitionProvider));
80              }
81          }
82          return result;
83      }
84      
85      private static Collection<WorkflowRuntimeDefinition> createDefinitions(final MCPWorkflowDefinitionProvider workflowDefinitionProvider) {
86          Collection<WorkflowRuntimeDefinition> result = Objects.requireNonNull(workflowDefinitionProvider.getWorkflowDefinitions(),
87                  () -> String.format("Workflow definitions are required for `%s`.", workflowDefinitionProvider.getClass().getName()));
88          result.forEach(each -> Objects.requireNonNull(each, () -> String.format("Workflow definition is required for `%s`.", workflowDefinitionProvider.getClass().getName())));
89          return result;
90      }
91      
92      /**
93       * Find registered workflow runtime definition.
94       *
95       * @param workflowKind workflow kind
96       * @return workflow runtime definition
97       */
98      public Optional<WorkflowRuntimeDefinition> findRegisteredDefinition(final WorkflowKind workflowKind) {
99          return Optional.ofNullable(registeredDefinitions.get(workflowKind));
100     }
101     
102     /**
103      * Get required workflow runtime definition.
104      *
105      * @param workflowKind workflow kind
106      * @return workflow runtime definition
107      */
108     public WorkflowRuntimeDefinition getRequired(final WorkflowKind workflowKind) {
109         return findRegisteredDefinition(workflowKind).orElseThrow(() -> new MCPInvalidRequestException(String.format("Unknown workflow_kind `%s`.", workflowKind.getValue())));
110     }
111     
112     /**
113      * Get the workflow runtime definition required by a snapshot.
114      *
115      * @param snapshot workflow snapshot
116      * @return workflow runtime definition
117      */
118     public WorkflowRuntimeDefinition getRequired(final WorkflowContextSnapshot snapshot) {
119         ShardingSpherePreconditions.checkNotNull(snapshot.getWorkflowKind(),
120                 () -> new MCPWorkflowStateException(String.format("Workflow kind is required for plan_id `%s`.", snapshot.getPlanId()), snapshot.getPlanId()));
121         return getRequired(snapshot.getWorkflowKind());
122     }
123 }
