47  @NoArgsConstructor(access = AccessLevel.PRIVATE)

50      /**
51       * Append model-facing next action guidance to a planning response.
52       *
53       * @param payload response payload
54       * @param snapshot workflow snapshot
55       */
56      public static void appendPlanningGuidance(final Map<String, Object> payload, final WorkflowContextSnapshot snapshot) {
57          List<String> missingRequiredInputs = createMissingRequiredInputs(snapshot);
58          List<Map<String, Object>> clarificationQuestions = createClarificationQuestions(snapshot, missingRequiredInputs);
59          payload.put("missing_required_inputs", missingRequiredInputs);
60          payload.put(MCPPayloadFieldNames.CLARIFICATION_QUESTIONS, clarificationQuestions);
61          payload.put(MCPPayloadFieldNames.RESOURCES_TO_READ, new WorkflowGuidanceResourceHintProvider().createResourcesToRead(snapshot));
62          payload.put("proxy_topology_hint", createProxyTopologyHint(snapshot));
63          payload.put(MCPPayloadFieldNames.NEXT_ACTIONS, createPlanningNextActions(snapshot, missingRequiredInputs));
64      }
65      
66      private static Map<String, Object> createProxyTopologyHint(final WorkflowContextSnapshot snapshot) {
67          Map<String, Object> result = new LinkedHashMap<>(4, 1F);
68          result.put("expected_runtime_view", "proxy_rule_distsql");
69          result.put("workflow_kind", resolveWorkflowKind(snapshot));
70          result.put(MCPPayloadFieldNames.REASON, "Rule DistSQL workflow planning must use Proxy DistSQL-visible rule state.");
71          result.put("safe_recovery", "Read the feature algorithm and rule resources from ShardingSphere Proxy before retrying.");
72          return result;
73      }

75      /**
76       * Append model-facing next action guidance to an apply response.
77       *
78       * @param payload response payload
79       * @param status apply status
80       */
81      public static void appendApplyGuidance(final Map<String, Object> payload, final String status) {
82          List<Map<String, Object>> nextActions = new LinkedList<>();
83          if (WorkflowLifecycle.STATUS_COMPLETED.equals(status)) {
84              nextActions.add(MCPNextActionUtils.callTool(WorkflowToolDescriptors.VALIDATE_TOOL_NAME, "Validate the runtime state after workflow artifacts are applied or exported.",
85                      Map.of(WorkflowFieldNames.PLAN_ID, Objects.toString(payload.get(WorkflowFieldNames.PLAN_ID), ""))));
86          }
87          if (WorkflowLifecycle.STATUS_AWAITING_MANUAL_EXECUTION.equals(status)) {
88              nextActions.add(MCPNextActionUtils.askUser("Confirm the manual artifacts were executed outside MCP before validation.", List.of("manual_artifacts_executed")));
89          }
90          if (WorkflowLifecycle.STATUS_FAILED.equals(status) && isSecretReferenceRecovery(payload)) {
91              nextActions.add(MCPNextActionUtils.askUser("Review the manual artifacts, replace neutral secret placeholders outside MCP, and execute them through the normal operational channel.",
92                      List.of("manual_artifacts_executed")));
93          } else if (WorkflowLifecycle.STATUS_FAILED.equals(status)) {
94              nextActions.add(MCPNextActionUtils.askUser("Inspect issues and retry database_gateway_apply_workflow only after the failed artifact is corrected.", List.of("issues")));
95          }
96          payload.put(MCPPayloadFieldNames.NEXT_ACTIONS, nextActions);
97      }
98      
99      private static boolean isSecretReferenceRecovery(final Map<String, Object> payload) {
100         Object category = payload.get("category");
101         return MCPDiagnosticCategory.SECRET_REFERENCE_MALFORMED.equals(category)
102                 || MCPDiagnosticCategory.SECRET_REFERENCE_MISSING.equals(category)
103                 || MCPDiagnosticCategory.SECRET_REFERENCE_MANUAL_EXECUTION_REQUIRED.equals(category);
104     }

106     /**
107      * Append model-facing next action guidance to a validation response.
108      *
109      * @param payload response payload
110      * @param snapshot workflow snapshot
111      * @param validationReport validation report
112      */
113     public static void appendValidationGuidance(final Map<String, Object> payload, final WorkflowContextSnapshot snapshot, final ValidationReport validationReport) {
114         boolean failed = WorkflowLifecycle.STATUS_FAILED.equals(validationReport.getOverallStatus());
115         payload.put("recovery_guidance", failed ? createValidationRecovery(snapshot) : "");
116         payload.put(MCPPayloadFieldNames.NEXT_ACTIONS, failed ? createValidationFailureActions(snapshot) : List.of(createStopAction()));
117     }
118     
119     private static String createValidationRecovery(final WorkflowContextSnapshot snapshot) {
120         return isManualOnlyWorkflow(snapshot)
121                 ? "Manual-only artifacts are exported but not executed by MCP. Execute them manually, then run database_gateway_validate_workflow again."
122                 : "Inspect mismatches, adjust the plan or runtime state, then run database_gateway_validate_workflow again.";
123     }
124     
125     private static List<Map<String, Object>> createValidationFailureActions(final WorkflowContextSnapshot snapshot) {
126         if (isManualOnlyWorkflow(snapshot)) {
127             return List.of(MCPNextActionUtils.askUser(
128                     "Confirm the manual artifacts were executed outside MCP, then run database_gateway_validate_workflow again.", List.of("manual_artifacts_executed")));
129         }
130         String planningTool = resolvePlanningTool(snapshot);
131         return planningTool.isEmpty()
132                 ? List.of(MCPNextActionUtils.askUser("Confirm the workflow kind before re-planning with the existing plan_id.", List.of("workflow_kind", "mismatches")))
133                 : List.of(MCPNextActionUtils.callTool(
134                         planningTool, "Re-plan with corrected metadata or algorithm choices.", Map.of(WorkflowFieldNames.PLAN_ID, snapshot.getPlanId())));
135     }
136     
137     private static boolean isManualOnlyWorkflow(final WorkflowContextSnapshot snapshot) {
138         return null != snapshot.getRequest() && WorkflowLifecycle.EXECUTION_MODE_MANUAL_ONLY.equalsIgnoreCase(snapshot.getRequest().getExecutionMode());
139     }
140     
141     private static List<String> createMissingRequiredInputs(final WorkflowContextSnapshot snapshot) {
142         List<String> result = new LinkedList<>();
143         ClarifiedIntent clarifiedIntent = snapshot.getClarifiedIntent();
144         for (String each : clarifiedIntent.getUnresolvedFields()) {
145             String missingInput = normalizeMissingInput(snapshot, each);
146             if (!result.contains(missingInput)) {
147                 result.add(missingInput);
148             }
149         }
150         for (WorkflowIssue each : snapshot.getIssues()) {
151             addMissingInputsFromIssue(result, snapshot, each);
152         }
153         if (result.isEmpty() && !clarifiedIntent.getClarificationMessages().isEmpty()) {
154             result.add("user_clarification");
155         }
156         return result;
157     }
158     
159     private static List<Map<String, Object>> createClarificationQuestions(final WorkflowContextSnapshot snapshot, final List<String> missingRequiredInputs) {
160         List<Map<String, Object>> result = new LinkedList<>();
161         List<String> clarificationMessages = snapshot.getClarifiedIntent().getClarificationMessages();
162         for (int i = 0; i < missingRequiredInputs.size(); i++) {
163             String fieldName = missingRequiredInputs.get(i);
164             result.add(createClarificationQuestion(snapshot, fieldName, i < clarificationMessages.size() ? clarificationMessages.get(i) : ""));
165         }
166         return result;
167     }
168     
169     private static Map<String, Object> createClarificationQuestion(final WorkflowContextSnapshot snapshot, final String fieldName, final String clarificationMessage) {
170         Map<String, Object> result = new LinkedHashMap<>(6, 1F);
171         boolean secret = isSecretClarificationField(snapshot, fieldName);
172         String inputType = resolveClarificationInputType(fieldName, secret);
173         result.put(MCPPayloadFieldNames.FIELD, fieldName);
174         result.put("question_key", fieldName.replace('.', '_'));
175         result.put(MCPPayloadFieldNames.INPUT_TYPE, inputType);
176         if ("boolean".equals(inputType)) {
177             result.put(MCPPayloadFieldNames.ALLOWED_VALUES, List.of(true, false));
178         }
179         result.put(MCPPayloadFieldNames.SECRET, secret);
180         result.put(MCPPayloadFieldNames.DISPLAY_MESSAGE, clarificationMessage.isBlank() ? String.format("Please provide `%s`.", fieldName) : clarificationMessage);
181         return result;
182     }
183     
184     private static String resolveClarificationInputType(final String fieldName, final boolean secret) {
185         if (secret) {
186             return "secret";
187         }
188         return fieldName.startsWith("requires_") ? "boolean" : "string";
189     }
190     
191     private static boolean isSecretClarificationField(final WorkflowContextSnapshot snapshot, final String fieldName) {
192         String propertyKey = resolveAlgorithmPropertyKey(fieldName);
193         if (propertyKey.isEmpty()) {
194             return false;
195         }
196         for (AlgorithmPropertyRequirement each : snapshot.getPropertyRequirements()) {
197             if (propertyKey.equals(each.getPropertyKey())) {
198                 return each.isSecret();
199             }
200         }
201         return true;
202     }
203     
204     private static void addMissingInputsFromIssue(final Collection<String> missingInputs, final WorkflowContextSnapshot snapshot, final WorkflowIssue issue) {
205         addRequiredIdentifierInput(missingInputs, issue, WorkflowIssueCode.DATABASE_REQUIRED, WorkflowFieldNames.DATABASE);
206         addRequiredIdentifierInput(missingInputs, issue, WorkflowIssueCode.TABLE_REQUIRED, WorkflowFieldNames.TABLE);
207         addRequiredIdentifierInput(missingInputs, issue, WorkflowIssueCode.COLUMN_REQUIRED, WorkflowFieldNames.COLUMN);
208         Object missingRuleInputs = issue.getDetails().get("missing_inputs");
209         if (missingRuleInputs instanceof Collection) {
210             for (Object each : (Collection<?>) missingRuleInputs) {
211                 String missingInput = String.valueOf(each);
212                 if (!missingInputs.contains(missingInput)) {
213                     missingInputs.add(missingInput);
214                 }
215             }
216         }
217         Object missingProperties = issue.getDetails().get("missing_properties");
218         if (missingProperties instanceof Collection) {
219             for (Object each : (Collection<?>) missingProperties) {
220                 String missingInput = resolveAlgorithmPropertyInput(snapshot, String.valueOf(each));
221                 if (!missingInputs.contains(missingInput)) {
222                     missingInputs.add(missingInput);
223                 }
224             }
225         }
226     }
227     
228     private static void addRequiredIdentifierInput(final Collection<String> missingInputs, final WorkflowIssue issue, final String issueCode, final String fieldName) {
229         if (issueCode.equals(issue.getCode()) && !missingInputs.contains(fieldName)) {
230             missingInputs.add(fieldName);
231         }
232     }
233     
234     private static String normalizeMissingInput(final WorkflowContextSnapshot snapshot, final String fieldName) {
235         if (!fieldName.startsWith("algorithm_properties.")) {
236             return fieldName;
237         }
238         return resolveAlgorithmPropertyInput(snapshot, fieldName.substring("algorithm_properties.".length()));
239     }
240     
241     private static String resolveAlgorithmPropertyInput(final WorkflowContextSnapshot snapshot, final String propertyKey) {
242         for (AlgorithmPropertyRequirement each : snapshot.getPropertyRequirements()) {
243             if (propertyKey.equals(each.getPropertyKey())) {
244                 return String.format("%s.%s", resolveAlgorithmPropertiesArgument(each.getAlgorithmRole()), propertyKey);
245             }
246         }
247         return String.format("%s.%s", WorkflowFieldNames.PRIMARY_ALGORITHM_PROPERTIES, propertyKey);
248     }
249     
250     private static String resolveAlgorithmPropertiesArgument(final String algorithmRole) {
251         if ("assisted_query".equals(algorithmRole)) {
252             return WorkflowFieldNames.ASSISTED_QUERY_ALGORITHM_PROPERTIES;
253         }
254         if ("like_query".equals(algorithmRole)) {
255             return WorkflowFieldNames.LIKE_QUERY_ALGORITHM_PROPERTIES;
256         }
257         return WorkflowFieldNames.PRIMARY_ALGORITHM_PROPERTIES;
258     }
259     
260     private static String resolveAlgorithmPropertyKey(final String fieldName) {
261         int separatorIndex = fieldName.indexOf('.');
262         if (-1 == separatorIndex) {
263             return "";
264         }
265         String argumentName = fieldName.substring(0, separatorIndex);
266         if ("algorithm_properties".equals(argumentName) || argumentName.endsWith("_algorithm_properties")) {
267             return fieldName.substring(separatorIndex + 1);
268         }
269         return "";
270     }

272     private static List<Map<String, Object>> createPlanningNextActions(final WorkflowContextSnapshot snapshot, final List<String> missingRequiredInputs) {
273         if (WorkflowLifecycle.STATUS_CLARIFYING.equals(snapshot.getStatus())) {
274             return List.of(MCPNextActionUtils.askUser("Ask for the missing inputs, then call the same planning tool with the existing plan_id.", missingRequiredInputs));
275         }
276         if (WorkflowLifecycle.STATUS_PLANNED.equals(snapshot.getStatus())) {
277             return List.of(MCPNextActionUtils.callTool(WorkflowToolDescriptors.APPLY_TOOL_NAME, "Preview workflow artifacts before execution.",
278                     Map.of(WorkflowFieldNames.PLAN_ID, snapshot.getPlanId(), WorkflowFieldNames.EXECUTION_MODE, WorkflowLifecycle.EXECUTION_MODE_PREVIEW)));
279         }
280         if (WorkflowLifecycle.STATUS_FAILED.equals(snapshot.getStatus())) {
281             return createRecoveryPlanningActions(snapshot);
282         }
283         return List.of();
284     }
285     
286     private static List<Map<String, Object>> createRecoveryPlanningActions(final WorkflowContextSnapshot snapshot) {
287         if (hasIssue(snapshot, WorkflowIssueCode.CLUSTER_MODE_REQUIRED)) {
288             return List.of(MCPNextActionUtils.stop("Connect to a Cluster-mode ShardingSphere Proxy, then start a new workflow plan."));
289         }
290         String planningTool = resolvePlanningTool(snapshot);
291         if (hasIssue(snapshot, WorkflowIssueCode.RULE_INPUT_CONFLICT)) {
292             return planningTool.isEmpty()
293                     ? List.of(MCPNextActionUtils.askUser(
294                             "Choose one input mode, remove conflicting inputs, and start a new plan without plan_id.", List.of("workflow_kind", "conflicting_inputs")))
295                     : List.of(MCPNextActionUtils.callTool(planningTool, "Choose one input mode, remove conflicting inputs, and start a new plan without plan_id.", Map.of()));
296         }
297         return planningTool.isEmpty()
298                 ? List.of(MCPNextActionUtils.askUser(
299                         "Confirm the workflow kind, then call the matching planning tool with the existing plan_id.", List.of("workflow_kind", "issues")))
300                 : List.of(MCPNextActionUtils.callTool(
301                         planningTool, "Re-plan after resolving the reported issues.", Map.of(WorkflowFieldNames.PLAN_ID, snapshot.getPlanId())));
302     }
303     
304     private static boolean hasIssue(final WorkflowContextSnapshot snapshot, final String issueCode) {
305         return snapshot.getIssues().stream().anyMatch(each -> issueCode.equals(each.getCode()));
306     }
307     
308     private static Map<String, Object> createStopAction() {
309         return MCPNextActionUtils.stop("Validation passed. Report the confirmed workflow result to the user.");
310     }
311     
312     private static String resolvePlanningTool(final WorkflowContextSnapshot snapshot) {
313         return MCPDescriptorCatalogIndex.findPlanningToolNameByWorkflowKind(resolveWorkflowKind(snapshot)).orElse("");
314     }
315     
316     private static String resolveWorkflowKind(final WorkflowContextSnapshot snapshot) {
317         WorkflowKind workflowKind = snapshot.getWorkflowKind();
318         return null == workflowKind ? "" : workflowKind.getValue();
319     }
