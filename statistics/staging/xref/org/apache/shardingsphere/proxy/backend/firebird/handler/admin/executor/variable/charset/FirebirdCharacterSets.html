85      
86      private static final Map<String, FirebirdCharacterSets> CHARACTER_SETS_MAP;
87      
88      static {
89          Map<String, FirebirdCharacterSets> map = new HashMap<>(64, 1F);
90          for (FirebirdCharacterSets each : values()) {
91              map.put(each.name(), each);
92              for (String eachAlias : each.aliases) {
93                  map.put(eachAlias.toUpperCase(), each);
94              }
95          }
96          CHARACTER_SETS_MAP = map;
97      }
98      
99      private final Charset charset;
100     
101     private final String[] aliases;
102     
103     FirebirdCharacterSets(final Supplier<Charset> charsetSupplier, final String... aliases) {
104         Charset result = null;
105         try {
106             result = charsetSupplier.get();
107         } catch (final UnsupportedCharsetException ignored) {
108         }
109         charset = result;
110         this.aliases = aliases;
111     }
112     
113     /**
114      * Find corresponding {@link Charset} by charset name defined in Firebird.
115      *
116      * @param charsetName charset name defined in Firebird
117      * @return corresponding {@link Charset}
118      */
119     public static Charset findCharacterSet(final String charsetName) {
120         String formattedCharsetName = formatValue(charsetName);
121         FirebirdCharacterSets result = CHARACTER_SETS_MAP.get(formattedCharsetName.toUpperCase());
122         return null == result || null == result.charset ? Charset.forName(formattedCharsetName) : result.charset;
123     }
124     
125     private static String formatValue(final String value) {
126         return QuoteCharacter.SINGLE_QUOTE.isWrapped(value) || QuoteCharacter.QUOTE.isWrapped(value) ? value.substring(1, value.length() - 1) : value.trim();
127     }
