18  package org.apache.shardingsphere.mcp.core.protocol.error;
19  
20  import lombok.AccessLevel;
21  import lombok.NoArgsConstructor;
22  import org.apache.shardingsphere.mcp.support.database.metadata.jdbc.RuntimeDatabaseConnectionException;
23  import org.apache.shardingsphere.mcp.support.protocol.MCPPayloadFieldNames;
24  import org.apache.shardingsphere.mcp.support.protocol.MCPResourceHintUtils;
25  import org.apache.shardingsphere.mcp.support.protocol.MCPResponseMode;
26  
27  import java.util.LinkedHashMap;
28  import java.util.List;
29  import java.util.Map;
30  
31  /**
32   * MCP recovery payload support.
33   */
34  @NoArgsConstructor(access = AccessLevel.PRIVATE)
35  final class MCPRecoveryPayloadSupport {
36      
37      static Map<String, Object> createBaseRecovery(final String category, final String modelAction) {
38          Map<String, Object> result = new LinkedHashMap<>(8, 1F);
39          result.put("response_mode", MCPResponseMode.RECOVERY);
40          result.put("recovery_category", normalizeRecoveryCategory(category));
41          result.put("recoverable", true);
42          result.put("category", category);
43          result.put("model_action", modelAction);
44          return result;
45      }
46      
47      static Map<String, Object> getSuggestedArguments(final Map<String, Object> suggestedArguments, final Map<String, Object> defaultArguments) {
48          return suggestedArguments.isEmpty() ? defaultArguments : suggestedArguments;
49      }
50      
51      static List<Map<String, Object>> createResourceHintList(final String uri, final String resourceKind, final String reason) {
52          return List.of(MCPResourceHintUtils.create(uri, resourceKind, "read_first", reason, MCPPayloadFieldNames.RESOURCES_TO_READ));
53      }
54      
55      private static String normalizeRecoveryCategory(final String category) {
56          if (isRuntimeRecoveryCategory(category)) {
57              return "unavailable_runtime";
58          }
59          if ("banned_sql_statement".equals(category)) {
60              return "terminal_operator_action";
61          }
62          if (category.startsWith("missing_")) {
63              return "missing_context";
64          }
65          if (category.startsWith("unsupported_") || "read_only_sql_sent_to_update_tool".equals(category)) {
66              return "unsupported_target";
67          }
68          if ("invalid_enum_value".equals(category)) {
69              return "invalid_enum";
70          }
71          if ("rule_distsql_execution_failed".equals(category)) {
72              return "rule_metadata_execution";
73          }
74          if ("unsafe_sql_attempted".equals(category) || "multiple_sql_statements".equals(category)) {
75              return "unsafe_sql";
76          }
77          if ("stale_workflow".equals(category) || "workflow_state_error".equals(category)) {
78              return "stale_workflow";
79          }
80          if (category.endsWith("_not_found")) {
81              return "not_found";
82          }
83          if (category.contains("conflict")) {
84              return "ambiguous";
85          }
86          return "validation";
87      }
88      
89      private static boolean isRuntimeRecoveryCategory(final String category) {
90          switch (category) {
91              case RuntimeDatabaseConnectionException.CATEGORY_MISSING_JDBC_DRIVER, RuntimeDatabaseConnectionException.CATEGORY_AUTHENTICATION_FAILED,
92                      RuntimeDatabaseConnectionException.CATEGORY_AUTHORIZATION_FAILED, RuntimeDatabaseConnectionException.CATEGORY_CONNECTION_TIMEOUT,
93                      RuntimeDatabaseConnectionException.CATEGORY_INVALID_CONFIGURATION, RuntimeDatabaseConnectionException.CATEGORY_DATABASE_UNAVAILABLE,
94                      RuntimeDatabaseConnectionException.CATEGORY_CONNECTION_FAILED:
95                  return true;
96              default:
97                  return false;
98          }
99      }
100 }
