18  package org.apache.shardingsphere.driver.executor.engine.transaction;
19  
20  import org.apache.shardingsphere.driver.jdbc.core.connection.ShardingSphereConnection;
21  import org.apache.shardingsphere.driver.jdbc.core.savepoint.ShardingSphereSavepoint;
22  import org.apache.shardingsphere.infra.session.query.QueryContext;
23  import org.apache.shardingsphere.sql.parser.statement.core.statement.tcl.ReleaseSavepointStatement;
24  import org.apache.shardingsphere.sql.parser.statement.core.statement.tcl.SavepointStatement;
25  import org.apache.shardingsphere.sql.parser.statement.core.statement.tcl.TCLStatement;
26  import org.apache.shardingsphere.transaction.core.TransactionOperationType;
27  
28  import java.sql.SQLException;
29  
30  /**
31   * Driver transaction statement executor.
32   */
33  public final class DriverTransactionSQLStatementExecutor {
34      
35      private final ShardingSphereConnection connection;
36      
37      private TransactionOperationType operationType;
38      
39      public DriverTransactionSQLStatementExecutor(final ShardingSphereConnection connection) {
40          this.connection = connection;
41      }
42      
43      /**
44       * Decide whether to execute TCL statement.
45       *
46       * @param queryContext query context
47       * @return whether to execute TCL statement or not
48       */
49      public boolean decide(final QueryContext queryContext) {
50          if (!(queryContext.getSqlStatementContext().getSqlStatement() instanceof TCLStatement)) {
51              return false;
52          }
53          TCLStatement tclStatement = (TCLStatement) queryContext.getSqlStatementContext().getSqlStatement();
54          if (tclStatement instanceof SavepointStatement) {
55              operationType = TransactionOperationType.SAVEPOINT;
56              return true;
57          }
58          if (tclStatement instanceof ReleaseSavepointStatement) {
59              operationType = TransactionOperationType.RELEASE_SAVEPOINT;
60              return true;
61          }
62          // TODO support more TCL statements
63          return false;
64      }
65      
66      /**
67       * Execute TCL statement.
68       *
69       * @param tclStatement SQL statement
70       * @return whether to execute TCL statement or not
71       * @throws SQLException SQL exception
72       */
73      public boolean execute(final TCLStatement tclStatement) throws SQLException {
74          if (TransactionOperationType.SAVEPOINT == operationType) {
75              connection.setSavepoint(((SavepointStatement) tclStatement).getSavepointName());
76              return true;
77          }
78          if (TransactionOperationType.RELEASE_SAVEPOINT == operationType) {
79              ShardingSphereSavepoint savepoint = new ShardingSphereSavepoint(((ReleaseSavepointStatement) tclStatement).getSavepointName());
80              connection.releaseSavepoint(savepoint);
81              return true;
82          }
83          return false;
84      }
85  }
