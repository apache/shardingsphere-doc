35  import java.util.ArrayList;
36  import java.util.Collection;
37  import java.util.HashMap;

44  /**
45   * Resource definition registry.
46   */
47  @NoArgsConstructor(access = AccessLevel.PRIVATE)
48  public final class ResourceDefinitionRegistry {
49      
50      private static final List<ResourceDefinition> REGISTERED_RESOURCE_DEFINITIONS;
51      
52      static {
53          Map<MCPUriTemplate, MCPResourceHandler<?>> resourceHandlers = createRegisteredResourceHandlers(
54                  ShardingSphereServiceLoader.getServiceInstances(MCPHandlerProvider.class).stream().flatMap(each -> each.getResourceHandlers().stream()).toList());
55          validateRegisteredResourceHandlers(resourceHandlers);
56          REGISTERED_RESOURCE_DEFINITIONS = createRegisteredResourceDefinitions(resourceHandlers);
57          validateRegisteredResourceDescriptors();
58      }
59      
60      private static Map<MCPUriTemplate, MCPResourceHandler<?>> createRegisteredResourceHandlers(final Collection<MCPResourceHandler<?>> handlers) {
61          ShardingSpherePreconditions.checkNotEmpty(handlers, () -> new IllegalStateException("No resource handlers are registered."));
62          Map<MCPUriTemplate, MCPResourceHandler<?>> result = new LinkedHashMap<>(handlers.size(), 1F);
63          for (MCPResourceHandler<?> each : handlers) {
64              String uriTemplate = each.getResourceUriTemplate();
65              ShardingSpherePreconditions.checkState(null != uriTemplate && !uriTemplate.isBlank(),
66                      () -> new IllegalArgumentException(String.format("Resource URI template is required for `%s`.", each.getClass().getName())));
67              MCPRequestContextTypes.validateContextType(each.getContextType(), each.getClass());
68              result.put(new MCPUriTemplate(uriTemplate), each);
69          }
70          return result;
71      }
72      
73      private static void validateRegisteredResourceHandlers(final Map<MCPUriTemplate, MCPResourceHandler<?>> resourceHandlers) {
74          Map<String, Class<?>> registeredPatterns = new HashMap<>(resourceHandlers.size(), 1F);
75          for (Entry<MCPUriTemplate, MCPResourceHandler<?>> entry : resourceHandlers.entrySet()) {
76              String pattern = entry.getValue().getResourceUriTemplate();
77              Class<?> previousPatternClass = registeredPatterns.putIfAbsent(pattern, entry.getValue().getClass());
78              ShardingSpherePreconditions.checkState(null == previousPatternClass,
79                      () -> new IllegalArgumentException(String.format("Duplicate resource URI template `%s` with `%s` and `%s`.",
80                              pattern, previousPatternClass.getName(), entry.getValue().getClass().getName())));
81          }
82          List<Entry<MCPUriTemplate, MCPResourceHandler<?>>> entries = new ArrayList<>(resourceHandlers.entrySet());
83          for (int i = 0; i < entries.size(); i++) {
84              Entry<MCPUriTemplate, MCPResourceHandler<?>> current = entries.get(i);
85              for (int j = i + 1; j < entries.size(); j++) {
86                  Entry<MCPUriTemplate, MCPResourceHandler<?>> other = entries.get(j);
87                  ShardingSpherePreconditions.checkState(!current.getKey().overlaps(other.getKey()), () -> new IllegalArgumentException(
88                          String.format("Overlapping resource URI templates `%s` with `%s` and `%s`.",
89                                  current.getValue().getResourceUriTemplate(), current.getValue().getClass().getName(), other.getValue().getClass().getName())));
90              }
91          }
92      }
93      
94      private static List<ResourceDefinition> createRegisteredResourceDefinitions(final Map<MCPUriTemplate, MCPResourceHandler<?>> handlers) {
95          return handlers.entrySet().stream().map(entry -> new ResourceDefinition(entry.getKey(),
96                  MCPDescriptorCatalogIndex.getRequiredResourceDescriptor(entry.getValue().getResourceUriTemplate()), entry.getValue())).toList();
97      }
98      
99      private static void validateRegisteredResourceDescriptors() {
100         for (MCPResourceDescriptor each : MCPDescriptorCatalogIndex.getResourceDescriptors()) {
101             ShardingSpherePreconditions.checkState(isRegisteredResourceDescriptor(each),
102                     () -> new IllegalStateException(String.format("MCP resource descriptor `%s` has no registered handler.", each.getUriTemplate())));
103         }
104     }
105     
106     private static boolean isRegisteredResourceDescriptor(final MCPResourceDescriptor descriptor) {
107         for (ResourceDefinition each : REGISTERED_RESOURCE_DEFINITIONS) {
108             if (descriptor.getUriTemplate().equals(each.handler().getResourceUriTemplate())) {
109                 return true;
110             }
111         }
112         return false;
113     }
114     
115     /**
116      * Dispatch resource URI to registered resource.
117      *
118      * @param requestContext request context
119      * @param resourceUri resource URI
120      * @return handled payload
121      */
122     public static Optional<MCPSuccessPayload> dispatch(final MCPFeatureRuntimeRequestContext requestContext, final String resourceUri) {
123         for (ResourceDefinition each : REGISTERED_RESOURCE_DEFINITIONS) {
124             Optional<MCPResourceURIVariables> matchedUriVariables = each.uriTemplate().parse(resourceUri);
125             if (matchedUriVariables.isPresent()) {
126                 return Optional.of(dispatch(requestContext, each.handler(), matchedUriVariables.get()));
127             }
128         }
129         return Optional.empty();
130     }
131     
132     private static <T extends MCPRequestContext> MCPSuccessPayload dispatch(final MCPFeatureRuntimeRequestContext requestContext, final MCPResourceHandler<T> resourceHandler,
133                                                                             final MCPResourceURIVariables uriVariables) {
134         return resourceHandler.handle(resourceHandler.getContextType().cast(requestContext), uriVariables);
135     }
136     
137     /**
138      * Get supported resource descriptors.
139      *
140      * @return supported resource descriptors
141      */
142     public static Collection<MCPResourceDescriptor> getSupportedResourceDescriptors() {
143         return REGISTERED_RESOURCE_DEFINITIONS.stream().map(ResourceDefinition::descriptor).toList();
144     }
145     
146     private record ResourceDefinition(MCPUriTemplate uriTemplate, MCPResourceDescriptor descriptor, MCPResourceHandler<?> handler) {
147     }
148 }
