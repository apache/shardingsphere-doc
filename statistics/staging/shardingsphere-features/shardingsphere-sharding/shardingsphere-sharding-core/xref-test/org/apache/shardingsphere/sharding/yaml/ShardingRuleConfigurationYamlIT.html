52      private static ShardingRuleConfiguration getExpectedRuleConfiguration() {
53          ShardingRuleConfiguration result = new ShardingRuleConfiguration();
54          ShardingTableRuleConfiguration userTableRuleConfig = new ShardingTableRuleConfiguration("t_user", "ds_${0..1}.t_user_${0..15}");
55          userTableRuleConfig.setDatabaseShardingStrategy(new ComplexShardingStrategyConfiguration("region_id, user_id", "core_complex_fixture"));
56          userTableRuleConfig.setTableShardingStrategy(new ComplexShardingStrategyConfiguration("region_id, user_id", "core_complex_fixture"));
57          result.getTables().add(userTableRuleConfig);
58          ShardingTableRuleConfiguration stockTableRuleConfig = new ShardingTableRuleConfiguration("t_stock", "ds_${0..1}.t_stock{0..8}");
59          stockTableRuleConfig.setDatabaseShardingStrategy(new HintShardingStrategyConfiguration("core_hint_fixture"));
60          stockTableRuleConfig.setTableShardingStrategy(new HintShardingStrategyConfiguration("core_hint_fixture"));
61          result.getTables().add(stockTableRuleConfig);
62          ShardingTableRuleConfiguration orderTableRuleConfig = new ShardingTableRuleConfiguration("t_order", "ds_${0..1}.t_order_${0..1}");
63          orderTableRuleConfig.setTableShardingStrategy(new StandardShardingStrategyConfiguration("order_id", "table_inline"));
64          result.getTables().add(orderTableRuleConfig);
65          ShardingTableRuleConfiguration orderItemTableRuleConfig = new ShardingTableRuleConfiguration("t_order_item", "ds_${0..1}.t_order_item_${0..1}");
66          orderItemTableRuleConfig.setTableShardingStrategy(new StandardShardingStrategyConfiguration("order_id", "core_standard_fixture"));
67          result.getTables().add(orderItemTableRuleConfig);
68          result.getBindingTableGroups().add(new ShardingTableReferenceRuleConfiguration("foo", "t_order, t_order_item"));
69          result.setDefaultDatabaseShardingStrategy(new StandardShardingStrategyConfiguration("order_id", "database_inline"));
70          result.setDefaultTableShardingStrategy(new NoneShardingStrategyConfiguration());
71          result.setDefaultShardingColumn("order_id");
72          result.setDefaultKeyGenerateStrategy(new KeyGenerateStrategyConfiguration("id", "snowflake"));
73          result.setDefaultAuditStrategy(new ShardingAuditStrategyConfiguration(Collections.singletonList("sharding_key_required_auditor"), true));
74          result.getKeyGenerateStrategies().put("t_order", createColumnKeyGenerateStrategyRuleConfiguration());
75          result.getKeyGenerateStrategies().put("id_sequence", createSequenceKeyGenerateStrategyRuleConfiguration());
76          result.getShardingAlgorithms().put("core_standard_fixture", new AlgorithmConfiguration("CORE.STANDARD.FIXTURE", new Properties()));
77          result.getShardingAlgorithms().put("core_complex_fixture", new AlgorithmConfiguration("CORE.COMPLEX.FIXTURE", new Properties()));
78          result.getShardingAlgorithms().put("core_hint_fixture", new AlgorithmConfiguration("CORE.HINT.FIXTURE", new Properties()));
79          result.getShardingAlgorithms().put("database_inline", new AlgorithmConfiguration("INLINE", PropertiesBuilder.build(new Property("algorithm-expression", "ds_${order_id % 2}"))));
80          result.getShardingAlgorithms().put("table_inline", new AlgorithmConfiguration("INLINE", PropertiesBuilder.build(new Property("algorithm-expression", "t_order_${order_id % 2}"))));
81          result.getKeyGenerators().put("snowflake", new AlgorithmConfiguration("SNOWFLAKE", new Properties()));
82          result.getAuditors().put("sharding_key_required_auditor", new AlgorithmConfiguration("DML_SHARDING_CONDITIONS", new Properties()));
83          return result;
84      }
85      
86      @Override
87      protected boolean assertYamlConfiguration(final YamlRuleConfiguration actual) {
88          assertShardingRule((YamlShardingRuleConfiguration) actual);
89          return true;
90      }
91      
92      private void assertShardingRule(final YamlShardingRuleConfiguration actual) {
93          assertThat(actual.getTables().size(), is(4));
94          assertTUser(actual);
95          assertTStock(actual);
96          assertTOrder(actual);
97          assertTOrderItem(actual);
98          assertBindingTable(actual);
99          assertKeyGenerateStrategies(actual);
100         assertThat(actual.getDefaultShardingColumn(), is("order_id"));
101     }
102     
103     private void assertTUser(final YamlShardingRuleConfiguration actual) {
104         assertThat(actual.getTables().get("t_user").getActualDataNodes(), is("ds_${0..1}.t_user_${0..15}"));
105         assertThat(actual.getTables().get("t_user").getDatabaseStrategy().getComplex().getShardingColumns(), is("region_id, user_id"));
106         assertThat(actual.getTables().get("t_user").getDatabaseStrategy().getComplex().getShardingAlgorithmName(), is("core_complex_fixture"));
107         assertThat(actual.getTables().get("t_user").getTableStrategy().getComplex().getShardingColumns(), is("region_id, user_id"));
108         assertThat(actual.getTables().get("t_user").getTableStrategy().getComplex().getShardingAlgorithmName(), is("core_complex_fixture"));
109     }
110     
111     private void assertTStock(final YamlShardingRuleConfiguration actual) {
112         assertThat(actual.getTables().get("t_stock").getActualDataNodes(), is("ds_${0..1}.t_stock{0..8}"));
113         assertThat(actual.getTables().get("t_stock").getDatabaseStrategy().getHint().getShardingAlgorithmName(), is("core_hint_fixture"));
114         assertThat(actual.getTables().get("t_stock").getTableStrategy().getHint().getShardingAlgorithmName(), is("core_hint_fixture"));
115     }
116     
117     private void assertTOrder(final YamlShardingRuleConfiguration actual) {
118         assertThat(actual.getTables().get("t_order").getActualDataNodes(), is("ds_${0..1}.t_order_${0..1}"));
119         assertThat(actual.getTables().get("t_order").getTableStrategy().getStandard().getShardingColumn(), is("order_id"));
120         assertThat(actual.getTables().get("t_order").getTableStrategy().getStandard().getShardingAlgorithmName(), is("table_inline"));
121     }
122     
123     private void assertTOrderItem(final YamlShardingRuleConfiguration actual) {
124         assertThat(actual.getTables().get("t_order_item").getActualDataNodes(), is("ds_${0..1}.t_order_item_${0..1}"));
125         assertThat(actual.getTables().get("t_order_item").getTableStrategy().getStandard().getShardingColumn(), is("order_id"));
126         assertThat(actual.getTables().get("t_order_item").getTableStrategy().getStandard().getShardingAlgorithmName(), is("core_standard_fixture"));
127         assertThat(actual.getTables().get("t_order_item").getTableStrategy().getStandard().getShardingAlgorithmName(), is("core_standard_fixture"));
128     }
129     
130     private void assertBindingTable(final YamlShardingRuleConfiguration actual) {
131         assertThat(actual.getBindingTables().size(), is(1));
132         assertThat(new ArrayList<>(actual.getBindingTables()).get(0), is("foo:t_order, t_order_item"));
133     }
134     
135     private void assertKeyGenerateStrategies(final YamlShardingRuleConfiguration actual) {
136         assertThat(actual.getKeyGenerateStrategies().size(), is(2));
137         assertThat(actual.getKeyGenerateStrategies().get("t_order").getKeyGenerateType(), is("column"));
138         assertThat(actual.getKeyGenerateStrategies().get("t_order").getKeyGeneratorName(), is("snowflake"));
139         assertThat(actual.getKeyGenerateStrategies().get("t_order").getLogicTable(), is("t_order"));
140         assertThat(actual.getKeyGenerateStrategies().get("t_order").getKeyGenerateColumn(), is("id"));
141         assertThat(actual.getKeyGenerateStrategies().get("id_sequence").getKeyGenerateType(), is("sequence"));
142         assertThat(actual.getKeyGenerateStrategies().get("id_sequence").getKeyGeneratorName(), is("snowflake"));
143         assertThat(actual.getKeyGenerateStrategies().get("id_sequence").getKeyGenerateSequence(), is("sequence_name"));
144     }
145     
146     private static KeyGenerateStrategiesConfiguration createColumnKeyGenerateStrategyRuleConfiguration() {
147         return new ColumnKeyGenerateStrategiesRuleConfiguration("snowflake", "t_order", "id");
148     }
149     
150     private static KeyGenerateStrategiesConfiguration createSequenceKeyGenerateStrategyRuleConfiguration() {
151         return new SequenceKeyGenerateStrategiesRuleConfiguration("snowflake", "sequence_name");
152     }
