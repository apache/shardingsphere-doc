18  package org.apache.shardingsphere.data.pipeline.core.consistencycheck;
19  
20  import com.google.common.collect.ImmutableMap;
21  import org.apache.commons.lang3.builder.EqualsBuilder;
22  import org.junit.jupiter.api.Test;
23  import org.junit.jupiter.params.ParameterizedTest;
24  import org.junit.jupiter.params.provider.Arguments;
25  import org.junit.jupiter.params.provider.MethodSource;
26  
27  import java.math.BigDecimal;
28  import java.sql.Timestamp;
29  import java.util.Arrays;
30  import java.util.List;
31  import java.util.Map;
32  import java.util.stream.Stream;
33  
34  import static org.junit.jupiter.api.Assertions.assertFalse;
35  import static org.junit.jupiter.api.Assertions.assertTrue;
36  
37  class DataConsistencyCheckUtilsTest {
38      
39      private static final List<String> UNIQUE_KEYS_NAMES = Arrays.asList("order_id", "user_id");
40      
41      @Test
42      void assertIsIntegerEquals() {
43          EqualsBuilder equalsBuilder = new EqualsBuilder();
44          String value = "123";
45          Long longValue = Long.parseLong(value);
46          assertTrue(DataConsistencyCheckUtils.isMatched(equalsBuilder, longValue, Integer.parseInt(value)));
47          assertTrue(DataConsistencyCheckUtils.isMatched(equalsBuilder, longValue, Short.parseShort(value)));
48          assertTrue(DataConsistencyCheckUtils.isMatched(equalsBuilder, longValue, Byte.parseByte(value)));
49      }
50      
51      @Test
52      void assertIsBigDecimalEquals() {
53          BigDecimal one = BigDecimal.valueOf(3322L, 1);
54          BigDecimal another = BigDecimal.valueOf(33220L, 2);
55          assertTrue(DataConsistencyCheckUtils.isBigDecimalEquals(one, another));
56      }
57      
58      @Test
59      void assertTimestampEquals() {
60          EqualsBuilder equalsBuilder = new EqualsBuilder();
61          long time = System.currentTimeMillis();
62          assertTrue(DataConsistencyCheckUtils.isMatched(equalsBuilder, new Timestamp(time), new Timestamp(time / 10L * 10L + 1L)));
63          assertFalse(DataConsistencyCheckUtils.isMatched(equalsBuilder, new Timestamp(time), new Timestamp(time + 1000L)));
64      }
65      
66      @ParameterizedTest(name = "{0}")
67      @MethodSource("matchedPreEpochTimestampCases")
68      void assertTimestampMatchedBeforeEpoch(final String scenario, final long thisMillis, final long thatMillis) {
69          assertTrue(DataConsistencyCheckUtils.isMatched(new EqualsBuilder(), new Timestamp(thisMillis), new Timestamp(thatMillis)));
70      }
71      
72      private static Stream<Arguments> matchedPreEpochTimestampCases() {
73          return Stream.of(
74                  Arguments.of("same second before epoch", -1000L, -999L),
75                  Arguments.of("sub second tolerance before epoch", -1999L, -1001L));
76      }
77      
78      @ParameterizedTest(name = "{0}")
79      @MethodSource("notMatchedPreEpochTimestampCases")
80      void assertTimestampNotMatchedBeforeEpoch(final String scenario, final long thisMillis, final long thatMillis) {
81          assertFalse(DataConsistencyCheckUtils.isMatched(new EqualsBuilder(), new Timestamp(thisMillis), new Timestamp(thatMillis)));
82      }
83      
84      private static Stream<Arguments> notMatchedPreEpochTimestampCases() {
85          return Stream.of(
86                  Arguments.of("different seconds before epoch", -1500L, -1000L),
87                  Arguments.of("different seconds across epoch", -500L, 400L));
88      }
89      
90      @Test
91      void assertIsFirstUniqueKeysValueMatched() {
92          Map<String, Object> record1 = ImmutableMap.of("order_id", 101, "user_id", 1, "status", "ok");
93          Map<String, Object> record2 = ImmutableMap.of("order_id", 102, "user_id", 2, "status", "ok");
94          EqualsBuilder equalsBuilder = new EqualsBuilder();
95          assertTrue(DataConsistencyCheckUtils.isFirstUniqueKeyValueMatched(record1, record1, UNIQUE_KEYS_NAMES.get(0), equalsBuilder));
96          assertFalse(DataConsistencyCheckUtils.isFirstUniqueKeyValueMatched(record1, record2, UNIQUE_KEYS_NAMES.get(0), equalsBuilder));
97      }
98  }
