18  package org.apache.shardingsphere.mcp.feature.sharding.tool.handler;
19  
20  import org.apache.shardingsphere.mcp.feature.sharding.tool.model.ShardingWorkflowRequest;
21  import org.apache.shardingsphere.mcp.support.workflow.service.WorkflowPlanningArguments;
22  import org.apache.shardingsphere.mcp.support.workflow.service.WorkflowRequestBinder;
23  
24  import java.util.Collection;
25  import java.util.Map;
26  import java.util.stream.Stream;
27  
28  /**
29   * Sharding planning request binder.
30   */
31  public final class ShardingPlanningRequestBinder {
32      
33      /**
34       * Bind sharding table rule workflow request.
35       *
36       * @param arguments raw MCP arguments
37       * @return sharding table rule workflow request
38       */
39      public ShardingWorkflowRequest bindTableRule(final Map<String, Object> arguments) {
40          return WorkflowRequestBinder.bindPlanningRequest(ShardingWorkflowRequest::new, arguments, this::bindTableRuleArguments);
41      }
42      
43      /**
44       * Bind sharding table reference rule workflow request.
45       *
46       * @param arguments raw MCP arguments
47       * @return sharding table reference rule workflow request
48       */
49      public ShardingWorkflowRequest bindTableReferenceRule(final Map<String, Object> arguments) {
50          return WorkflowRequestBinder.bindPlanningRequest(ShardingWorkflowRequest::new, arguments, this::bindTableReferenceRuleArguments);
51      }
52      
53      /**
54       * Bind sharding default strategy workflow request.
55       *
56       * @param arguments raw MCP arguments
57       * @return sharding default strategy workflow request
58       */
59      public ShardingWorkflowRequest bindDefaultStrategy(final Map<String, Object> arguments) {
60          return WorkflowRequestBinder.bindPlanningRequest(ShardingWorkflowRequest::new, arguments, this::bindDefaultStrategyArguments);
61      }
62      
63      /**
64       * Bind sharding key generator workflow request.
65       *
66       * @param arguments raw MCP arguments
67       * @return sharding key generator workflow request
68       */
69      public ShardingWorkflowRequest bindKeyGenerator(final Map<String, Object> arguments) {
70          return WorkflowRequestBinder.bindPlanningRequest(ShardingWorkflowRequest::new, arguments, this::bindKeyGeneratorArguments);
71      }
72      
73      /**
74       * Bind sharding key generate strategy workflow request.
75       *
76       * @param arguments raw MCP arguments
77       * @return sharding key generate strategy workflow request
78       */
79      public ShardingWorkflowRequest bindKeyGenerateStrategy(final Map<String, Object> arguments) {
80          return WorkflowRequestBinder.bindPlanningRequest(ShardingWorkflowRequest::new, arguments, this::bindKeyGenerateStrategyArguments);
81      }
82      
83      /**
84       * Bind sharding rule component cleanup workflow request.
85       *
86       * @param arguments raw MCP arguments
87       * @return sharding rule component cleanup workflow request
88       */
89      public ShardingWorkflowRequest bindRuleComponentCleanup(final Map<String, Object> arguments) {
90          return WorkflowRequestBinder.bindPlanningRequest(ShardingWorkflowRequest::new, arguments, this::bindRuleComponentCleanupArguments);
91      }
92      
93      private void bindTableRuleArguments(final ShardingWorkflowRequest request, final WorkflowPlanningArguments workflowPlanningArguments) {
94          workflowPlanningArguments.applyStringArgument("rule", request::setRuleName);
95          workflowPlanningArguments.applyStringArgument("data_nodes", request::setDataNodes);
96          workflowPlanningArguments.applyStringArgument("storage_units", request::setStorageUnits);
97          workflowPlanningArguments.applyStringArgument("strategy_type", request::setStrategyType);
98          workflowPlanningArguments.applyStringArgument("sharding_columns", request::setShardingColumns);
99          workflowPlanningArguments.applyStringArgument("algorithm_type", request::setAlgorithmType);
100         workflowPlanningArguments.applyStringArgument("key_generate_column", request::setKeyGenerateColumn);
101         workflowPlanningArguments.applyStringArgument("key_generator", request::setKeyGeneratorName);
102         workflowPlanningArguments.applyStringArgument("key_generator_type", request::setKeyGeneratorType);
103         workflowPlanningArguments.applyStringArgument("allow_hint_disable", request::setAllowHintDisable);
104         request.putAlgorithmProperties(workflowPlanningArguments.getMapArgument("algorithm_properties"));
105         request.putKeyGeneratorProperties(workflowPlanningArguments.getMapArgument("key_generator_properties"));
106         replaceValues(request.getAuditorNames(), workflowPlanningArguments.getStringArgument("auditors"));
107     }
108     
109     private void bindTableReferenceRuleArguments(final ShardingWorkflowRequest request, final WorkflowPlanningArguments workflowPlanningArguments) {
110         workflowPlanningArguments.applyStringArgument("rule", request::setRuleName);
111         replaceValues(request.getReferenceTables(), workflowPlanningArguments.getStringArgument("reference_tables"));
112     }
113     
114     private void bindDefaultStrategyArguments(final ShardingWorkflowRequest request, final WorkflowPlanningArguments workflowPlanningArguments) {
115         workflowPlanningArguments.applyStringArgument("default_strategy_type", request::setDefaultStrategyType);
116         workflowPlanningArguments.applyStringArgument("strategy_type", request::setStrategyType);
117         workflowPlanningArguments.applyStringArgument("sharding_columns", request::setShardingColumns);
118         workflowPlanningArguments.applyStringArgument("algorithm_type", request::setAlgorithmType);
119         request.putAlgorithmProperties(workflowPlanningArguments.getMapArgument("algorithm_properties"));
120     }
121     
122     private void bindKeyGeneratorArguments(final ShardingWorkflowRequest request, final WorkflowPlanningArguments workflowPlanningArguments) {
123         workflowPlanningArguments.applyStringArgument("key_generator", request::setKeyGeneratorName);
124         workflowPlanningArguments.applyStringArgument("key_generator_type", request::setKeyGeneratorType);
125         request.putKeyGeneratorProperties(workflowPlanningArguments.getMapArgument("key_generator_properties"));
126     }
127     
128     private void bindKeyGenerateStrategyArguments(final ShardingWorkflowRequest request, final WorkflowPlanningArguments workflowPlanningArguments) {
129         workflowPlanningArguments.applyStringArgument("key_generate_strategy", request::setKeyGenerateStrategyName);
130         workflowPlanningArguments.applyStringArgument("sequence", request::setSequenceName);
131         workflowPlanningArguments.applyStringArgument("key_generator", request::setKeyGeneratorName);
132         workflowPlanningArguments.applyStringArgument("key_generator_type", request::setKeyGeneratorType);
133         request.putKeyGeneratorProperties(workflowPlanningArguments.getMapArgument("key_generator_properties"));
134     }
135     
136     private void bindRuleComponentCleanupArguments(final ShardingWorkflowRequest request, final WorkflowPlanningArguments workflowPlanningArguments) {
137         workflowPlanningArguments.applyStringArgument("component_type", request::setComponentType);
138         workflowPlanningArguments.applyStringArgument("component_name", request::setComponentName);
139     }
140     
141     private void replaceValues(final Collection<String> target, final String value) {
142         Collection<String> values = createStringList(value);
143         if (!values.isEmpty()) {
144             target.clear();
145             target.addAll(values);
146         }
147     }
148     
149     private Collection<String> createStringList(final String value) {
150         return Stream.of(value.split(",")).map(String::trim).filter(each -> !each.isEmpty()).toList();
151     }

153 }
