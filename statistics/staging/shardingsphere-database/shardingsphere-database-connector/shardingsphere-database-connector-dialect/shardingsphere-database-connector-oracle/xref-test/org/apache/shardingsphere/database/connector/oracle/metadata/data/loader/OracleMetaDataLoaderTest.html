95      @ParameterizedTest(name = "{0}")
96      @MethodSource("assertLoadArguments")
97      void assertLoad(final String name, final int majorVersion, final int minorVersion,
98                      final boolean withPrimaryKey, final String collation, final boolean expectedCaseSensitive) throws SQLException {
99          DataSource dataSource = mockDataSource();
100         ResultSet tableMetaDataResultSet = mockTableMetaDataResultSet(collation);
101         ResultSet indexMetaDataResultSet = mockIndexMetaDataResultSet();
102         ResultSet primaryKeyConstraintsResultSet = withPrimaryKey ? mockPrimaryKeyConstraintsMetaDataResultSet() : mock(ResultSet.class);
103         ResultSet primaryKeyColumnsResultSet = mockPrimaryKeyColumnsMetaDataResultSet();
104         when(dataSource.getConnection().prepareStatement(getTableMetaDataSQL(majorVersion, minorVersion)).executeQuery()).thenReturn(tableMetaDataResultSet);
105         when(dataSource.getConnection().prepareStatement(ALL_INDEXES_SQL).executeQuery()).thenReturn(indexMetaDataResultSet);
106         when(dataSource.getConnection().prepareStatement(ALL_PRIMARY_KEY_CONSTRAINTS_SQL).executeQuery()).thenReturn(primaryKeyConstraintsResultSet);
107         when(dataSource.getConnection().prepareStatement(ALL_PRIMARY_KEY_COLUMNS_SQL).executeQuery()).thenReturn(primaryKeyColumnsResultSet);
108         when(dataSource.getConnection().getMetaData().getUserName()).thenReturn("TEST");
109         when(dataSource.getConnection().getMetaData().getDatabaseMajorVersion()).thenReturn(majorVersion);
110         when(dataSource.getConnection().getMetaData().getDatabaseMinorVersion()).thenReturn(minorVersion);
111         TableMetaData actualTableMetaData = assertAndGetSingleTableMetaData(loadMetaData(dataSource));
112         assertThat(actualTableMetaData.getIndexes().size(), is(1));
113         List<ColumnMetaData> columnMetaDataList = new ArrayList<>(actualTableMetaData.getColumns());
114         assertColumnMetaData(columnMetaDataList.get(0), getExpectedFirstColumnMetaData(majorVersion, minorVersion, withPrimaryKey));
115         assertColumnMetaData(columnMetaDataList.get(1), new ColumnMetaData("name", Types.VARCHAR, false, false, expectedCaseSensitive, false, false, true));
116         assertColumnMetaData(columnMetaDataList.get(2), new ColumnMetaData("creation_time", Types.TIMESTAMP, false, false, false, true, false, true));
117     }

122     void assertLoadPrimaryKeys(final String name, final Collection<String> tableNames, final Collection<String> constraintNames,
123                                final Collection<String> primaryKeyTableNames, final Collection<String> primaryKeyColumnNames) throws SQLException {
124         DataSource dataSource = mockDataSource();
125         Connection connection = dataSource.getConnection();
126         ResultSet tableMetaDataResultSet = mockMultipleTableMetaDataResultSet(tableNames);
127         ResultSet primaryKeyConstraintsResultSet = mockPrimaryKeyConstraintRows(constraintNames);
128         ResultSet primaryKeyColumnsResultSet = mockPrimaryKeyColumnRows(primaryKeyTableNames, primaryKeyColumnNames);
129         PreparedStatement primaryKeyColumnsStatement = mock(PreparedStatement.class);
130         when(connection.prepareStatement(getMultipleTableMetaDataSQL(tableNames)).executeQuery()).thenReturn(tableMetaDataResultSet);
131         when(connection.prepareStatement(getPrimaryKeyConstraintsSQL(tableNames)).executeQuery()).thenReturn(primaryKeyConstraintsResultSet);
132         when(connection.prepareStatement(getPrimaryKeyColumnsSQL(tableNames, constraintNames))).thenReturn(primaryKeyColumnsStatement);
133         when(primaryKeyColumnsStatement.executeQuery()).thenReturn(primaryKeyColumnsResultSet);
134         when(connection.getMetaData().getUserName()).thenReturn("TEST");
135         when(connection.getMetaData().getDatabaseMajorVersion()).thenReturn(12);
136         when(connection.getMetaData().getDatabaseMinorVersion()).thenReturn(2);
137         clearInvocations(connection, primaryKeyColumnsStatement);
138         Collection<SchemaMetaData> actualSchemaMetaData = loadMetaData(dataSource, tableNames);
139         assertPrimaryKeys(actualSchemaMetaData.iterator().next().getTables(), primaryKeyTableNames, primaryKeyColumnNames);
140         verify(connection).prepareStatement(getPrimaryKeyConstraintsSQL(tableNames));
141         verify(connection).prepareStatement(getPrimaryKeyColumnsSQL(tableNames, constraintNames));
142         verify(primaryKeyColumnsStatement).setString(1, "TEST");
143         int parameterIndex = 2;
144         for (String each : constraintNames) {
145             verify(primaryKeyColumnsStatement).setString(parameterIndex++, each);
146         }
147     }

328     private String getMultipleTableMetaDataSQL(final Collection<String> tableNames) {
329         return "SELECT OWNER AS TABLE_SCHEMA, TABLE_NAME, COLUMN_NAME, NULLABLE, DATA_TYPE, COLUMN_ID, HIDDEN_COLUMN , IDENTITY_COLUMN, COLLATION"
330                 + " FROM ALL_TAB_COLS WHERE OWNER = ? AND TABLE_NAME IN (" + quote(tableNames) + ") ORDER BY COLUMN_ID";
331     }

333     private String getPrimaryKeyConstraintsSQL(final Collection<String> tableNames) {
334         return "SELECT OWNER AS TABLE_SCHEMA, TABLE_NAME, CONSTRAINT_NAME FROM ALL_CONSTRAINTS WHERE CONSTRAINT_TYPE = 'P' AND OWNER = ? AND TABLE_NAME IN ("
335                 + quote(tableNames) + ")";
336     }

338     private String getPrimaryKeyColumnsSQL(final Collection<String> tableNames, final Collection<String> constraintNames) {
339         return "SELECT OWNER AS TABLE_SCHEMA, TABLE_NAME, COLUMN_NAME FROM ALL_CONS_COLUMNS WHERE OWNER = ? AND TABLE_NAME IN (" + quote(tableNames)
340                 + ") AND CONSTRAINT_NAME IN (" + String.join(",", Collections.nCopies(constraintNames.size(), "?")) + ")";
341     }

343     private String quote(final Collection<String> values) {
344         return values.stream().map(each -> String.format("'%s'", each)).collect(Collectors.joining(","));
345     }

347     private String getTableMetaDataSQL(final int majorVersion, final int minorVersion) {
348         if (majorVersion > 12 || majorVersion == 12 && minorVersion >= 2) {
349             return ALL_TAB_COLUMNS_SQL_WITH_IDENTITY_AND_COLLATION;
350         }
351         if (majorVersion == 12 && minorVersion == 1) {
352             return ALL_TAB_COLUMNS_SQL_WITH_IDENTITY;
353         }
354         return ALL_TAB_COLUMNS_SQL_WITHOUT_IDENTITY_AND_COLLATION;
355     }

397     private void assertPrimaryKeys(final Collection<TableMetaData> actualTables, final Collection<String> primaryKeyTableNames,
398                                    final Collection<String> primaryKeyColumnNames) {
399         List<String> tables = new ArrayList<>(primaryKeyTableNames);
400         List<String> columns = new ArrayList<>(primaryKeyColumnNames);
401         Collection<String> expectedPrimaryKeys = new LinkedList<>();
402         for (int i = 0; i < columns.size(); i++) {
403             expectedPrimaryKeys.add(tables.get(i) + "." + columns.get(i));
404         }
405         for (TableMetaData each : actualTables) {
406             for (ColumnMetaData column : each.getColumns()) {
407                 boolean expectedPrimaryKey = expectedPrimaryKeys.contains(each.getName() + "." + column.getName());
408                 assertThat(column.isPrimaryKey(), is(expectedPrimaryKey));
409             }
410         }
411     }
