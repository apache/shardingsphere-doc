113     void assertScheduleThrowsForConflictingSourceTableIdentity() {
114         Collection<MigrationSourceTargetEntry> entries = Arrays.asList(
115                 new MigrationSourceTargetEntry(new DataNode("foo_ds", "foo_schema", "foo_tbl"), "foo_target"),
116                 new MigrationSourceTargetEntry(new DataNode("foo_ds", "bar_schema", "FOO_TBL"), "bar_target"));
117         PipelineInvalidParameterException actual = assertThrows(PipelineInvalidParameterException.class, () -> jobAPI.schedule(contextKey, entries, "foo_db"));

120         verify(jobManager, never()).start(any());
121     }

148     @Test
149     void assertScheduleBuildsJobConfiguration() {
150         Map<String, DataSourcePoolProperties> sources = new LinkedHashMap<>(2, 1F);
151         sources.put("bar_ds", createDataSourcePoolProperties("jdbc:h2:mem:bar_db"));
152         sources.put("foo_ds", createDataSourcePoolProperties("jdbc:h2:mem:foo_db"));
153         when(dataSourcePersistService.load(contextKey, "MIGRATION")).thenReturn(sources);
154         ContextManager contextManager = mock(ContextManager.class, Answers.RETURNS_DEEP_STUBS);
155         ShardingSphereDatabase database = contextManager.getMetaDataContexts().getMetaData().getDatabase("foo_db");
156         when(database.getName()).thenReturn("foo_db");
157         StorageUnit storageUnit = mock(StorageUnit.class);
158         when(storageUnit.getDataSourcePoolProperties()).thenReturn(createDataSourcePoolProperties("jdbc:h2:mem:target_db"));
159         when(database.getResourceMetaData().getStorageUnits()).thenReturn(Collections.singletonMap("target_ds", storageUnit));
160         when(database.getRuleMetaData().getConfigurations()).thenReturn(Collections.singleton(new SingleRuleConfiguration(Collections.singletonList("target_ds.foo_target"), "target_ds")));
161         MigrationSourceTargetEntry entry = new MigrationSourceTargetEntry(new DataNode("bar_ds", "bar_schema", "bar_tbl"), "foo_target");
162         Collection<MigrationSourceTargetEntry> entries = Arrays.asList(entry,
163                 new MigrationSourceTargetEntry(new DataNode("foo_ds", (String) null, "foo_tbl_item"), "foo_item_target"),
164                 new MigrationSourceTargetEntry(new DataNode("foo_ds", "foo_schema", "foo_tbl"), "bar_target"), entry);
165         try (MockedStatic<PipelineContextManager> ignored = mockStatic(PipelineContextManager.class)) {
166             when(PipelineContextManager.getProxyContext()).thenReturn(contextManager);
167             String actualJobId = jobAPI.schedule(contextKey, entries, "foo_db");
168             ArgumentCaptor<MigrationJobConfiguration> jobConfigCaptor = ArgumentCaptor.forClass(MigrationJobConfiguration.class);
169             verify(jobManager).start(jobConfigCaptor.capture());
170             MigrationJobConfiguration actual = jobConfigCaptor.getValue();
171             List<String> expectedShardingDataNodes = Arrays.asList("foo_target:bar_ds.bar_schema.bar_tbl", "bar_target:foo_ds.foo_schema.foo_tbl|foo_item_target:foo_ds.foo_tbl_item");
172             assertThat(actualJobId, is(PipelineJobIdUtils.marshal(new MigrationJobId(contextKey, expectedShardingDataNodes))));
173             assertThat(actual.getJobId(), is(actualJobId));
174             assertThat(actual.getTargetDatabaseName(), is("foo_db"));
175             assertThat(actual.getSourceDatabaseType().getType(), is("H2"));
176             assertThat(actual.getTargetDatabaseType().getType(), is("H2"));
177             assertThat(actual.getTargetTableNames(), is(Arrays.asList("bar_target", "foo_item_target", "foo_target")));
178             assertThat(actual.getSources().keySet(), contains("foo_ds", "bar_ds"));
179             assertThat(actual.getTargetTableSchemaMap().get("bar_target"), is("foo_schema"));
180             assertNull(actual.getTargetTableSchemaMap().get("foo_item_target"));
181             assertThat(actual.getTargetTableSchemaMap().get("foo_target"), is("bar_schema"));
182             assertThat(actual.getConcurrency(), is(3));
183             assertThat(actual.getRetryTimes(), is(3));
184             YamlMigrationJobConfiguration actualYaml = new YamlMigrationJobConfigurationSwapper().swapToYamlConfiguration(actual);
185             assertThat(actualYaml.getSources().get("foo_ds").getType(), is("JDBC"));
186             Map<String, Object> expectedDataSourceProps = new LinkedHashMap<>(5, 1F);
187             expectedDataSourceProps.put("url", "jdbc:h2:mem:foo_db");
188             expectedDataSourceProps.put("username", "foo_user");
189             expectedDataSourceProps.put("password", "foo_password");
190             expectedDataSourceProps.put("minPoolSize", 5);
191             expectedDataSourceProps.put("dataSourceClassName", "com.zaxxer.hikari.HikariDataSource");
192             assertThat(YamlEngine.unmarshal(actualYaml.getSources().get("foo_ds").getParameter(), Map.class), is(expectedDataSourceProps));
193             expectedDataSourceProps.put("url", "jdbc:h2:mem:bar_db");
194             assertThat(YamlEngine.unmarshal(actualYaml.getSources().get("bar_ds").getParameter(), Map.class), is(expectedDataSourceProps));
195             assertThat(actualYaml.getTarget().getType(), is("ShardingSphereJDBC"));
196             YamlRootConfiguration actualTarget = YamlEngine.unmarshal(actualYaml.getTarget().getParameter(), YamlRootConfiguration.class, true);
197             assertThat(actualTarget.getDatabaseName(), is("foo_db"));
198             expectedDataSourceProps.put("url", "jdbc:h2:mem:target_db");
199             assertThat(actualTarget.getDataSources(), is(Collections.singletonMap("target_ds", expectedDataSourceProps)));
200             assertThat(actualYaml.getTablesFirstDataNodes(), is("bar_target:foo_ds.foo_schema.foo_tbl|foo_item_target:foo_ds.foo_tbl_item|foo_target:bar_ds.bar_schema.bar_tbl"));
201             assertThat(actualYaml.getJobShardingDataNodes(), is(expectedShardingDataNodes));
202         }
203     }

223     private static Stream<Arguments> provideInvalidSources() {
224         Map<String, DataSourcePoolProperties> mixedSources = new LinkedHashMap<>(2, 1F);
225         mixedSources.put("foo_ds", createDataSourcePoolProperties("jdbc:h2:mem:foo_db"));
226         mixedSources.put("bar_ds", createDataSourcePoolProperties("jdbc:mysql://localhost:3306/bar_db"));
227         return Stream.of(
228                 Arguments.of("missing source storage unit", Collections.singletonList(new MigrationSourceTargetEntry(new DataNode("foo_ds.foo_tbl"), "foo_target")), Collections.emptyMap(),
229                         "There is invalid parameter value. foo_ds doesn't exist. Run `SHOW MIGRATION SOURCE STORAGE UNITS;` to verify it."),
230                 Arguments.of("duplicate target table", Arrays.asList(new MigrationSourceTargetEntry(new DataNode("foo_ds.foo_tbl"), "foo_target"),
231                         new MigrationSourceTargetEntry(new DataNode("bar_ds.bar_tbl"), "foo_target")), Collections.singletonMap("bar_ds", createDataSourcePoolProperties("jdbc:h2:mem:bar_db")),
232                         "There is invalid parameter value. More than one source table for foo_target"),
233                 Arguments.of("mixed source database types", Arrays.asList(new MigrationSourceTargetEntry(new DataNode("foo_ds.foo_tbl"), "bar_target"),
234                         new MigrationSourceTargetEntry(new DataNode("bar_ds.bar_tbl"), "foo_target")), mixedSources,
235                         "There is invalid parameter value. Source storage units have different database types"));
236     }

238     @Test
239     void assertCommit() {
240         String jobId = PipelineJobIdUtils.marshal(new MigrationJobId(contextKey, Collections.emptyList()));
241         String checkJobId = "foo_check_job";
242         MigrationJobConfiguration jobConfig = mock(MigrationJobConfiguration.class);
243         when(jobConfig.getTargetDatabaseName()).thenReturn("foo_db");
244         when(jobConfigManager.getJobConfiguration(jobId)).thenReturn(jobConfig);
245         ContextManager contextManager = mock(ContextManager.class);
246         ShardingSphereDatabase database = mock(ShardingSphereDatabase.class);
247         when(contextManager.getDatabase("foo_db")).thenReturn(database);
248         try (
249                 MockedStatic<PipelineAPIFactory> ignoredFactory = mockStatic(PipelineAPIFactory.class, Answers.RETURNS_DEEP_STUBS);
250                 MockedStatic<PipelineContextManager> ignoredContext = mockStatic(PipelineContextManager.class)) {
251             when(PipelineAPIFactory.getPipelineGovernanceFacade(contextKey).getJobFacade().getCheck().listCheckJobIds(jobId)).thenReturn(Collections.singleton(checkJobId));
252             when(PipelineContextManager.getContext(contextKey)).thenReturn(contextManager);
253             jobAPI.commit(jobId);
254             InOrder inOrder = inOrder(jobManager, contextManager);
255             inOrder.verify(jobManager).stop(jobId);
256             inOrder.verify(jobManager).drop(checkJobId);
257             inOrder.verify(contextManager).getDatabase("foo_db");
258             inOrder.verify(contextManager).reloadDatabase(database);
259             inOrder.verify(jobManager).drop(jobId);
260         }
261     }

284     @ParameterizedTest(name = "{0}")
285     @MethodSource("provideRollbackSchemas")
286     void assertRollbackNormalizesSchemaAndPreservesActualTable(final String name, final String schemaName, final String expectedSchemaName) throws SQLException {
287         String jobId = PipelineJobIdUtils.marshal(new MigrationJobId(contextKey, Collections.emptyList()));
288         MigrationJobConfiguration jobConfig = mock(MigrationJobConfiguration.class);
289         when(jobConfig.getTargetTableNames()).thenReturn(Collections.singletonList("T_Order"));
290         when(jobConfig.getTargetTableSchemaMap()).thenReturn(Collections.singletonMap("T_Order", schemaName));
291         Connection connection = mock(Connection.class);
292         Statement statement = mock(Statement.class);
293         when(connection.createStatement()).thenReturn(statement);
294         try (
295                 MockedStatic<PipelineAPIFactory> ignoredFactory = mockStatic(PipelineAPIFactory.class, Answers.RETURNS_DEEP_STUBS);
296                 MockedConstruction<PipelineJobConfigurationManager> ignoredManager = mockConstruction(PipelineJobConfigurationManager.class,
297                         (mock, context) -> when(mock.getJobConfiguration(jobId)).thenReturn(jobConfig));
298                 MockedConstruction<PipelineDataSource> ignoredDataSource = mockConstruction(PipelineDataSource.class, (mock, context) -> {
299                     when(mock.getConnection()).thenReturn(connection);
300                     when(mock.getIdentifierContext()).thenReturn(new DatabaseIdentifierContext(IdentifierCasePolicyFactory.newLowerCasePolicySet()));
301                 });
302                 MockedConstruction<PipelinePrepareSQLBuilder> builders = mockConstruction(PipelinePrepareSQLBuilder.class,
303                         (mock, context) -> when(mock.buildDropSQL(expectedSchemaName, "T_Order")).thenReturn("DROP TABLE IF EXISTS target_table"))) {
304             jobAPI.rollback(jobId);
305             verify(builders.constructed().get(0)).buildDropSQL(expectedSchemaName, "T_Order");
306             verify(statement).execute("DROP TABLE IF EXISTS target_table");
307             verify(jobManager).drop(jobId);
308         }
309     }
