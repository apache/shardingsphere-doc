18  package org.apache.shardingsphere.mode.manager.cluster.dispatch.handler.global.type;
19  
20  import com.google.common.base.Strings;
21  import org.apache.shardingsphere.infra.util.yaml.YamlEngine;
22  import org.apache.shardingsphere.infra.yaml.data.pojo.YamlRowStatistics;
23  import org.apache.shardingsphere.mode.event.DataChangedEvent;
24  import org.apache.shardingsphere.mode.event.DataChangedEvent.Type;
25  import org.apache.shardingsphere.mode.manager.ContextManager;
26  import org.apache.shardingsphere.mode.manager.cluster.dispatch.handler.global.GlobalDataChangedEventHandler;
27  import org.apache.shardingsphere.mode.metadata.manager.statistics.StatisticsManager;
28  import org.apache.shardingsphere.mode.node.path.statistics.database.StatisticsDatabaseNodePath;
29  import org.apache.shardingsphere.mode.node.path.statistics.StatisticsNodePathParser;
30  
31  import java.util.Arrays;
32  import java.util.Collection;
33  import java.util.Optional;
34  
35  /**
36   * Statistics changed handler.
37   */
38  public final class StatisticsChangedHandler implements GlobalDataChangedEventHandler {
39      
40      @Override
41      public String getSubscribedKey() {
42          return new StatisticsDatabaseNodePath().getRootPath();
43      }
44      
45      @Override
46      public Collection<Type> getSubscribedTypes() {
47          return Arrays.asList(Type.ADDED, Type.UPDATED, Type.DELETED);
48      }
49      
50      @Override
51      public void handle(final ContextManager contextManager, final DataChangedEvent event) {
52          StatisticsManager databaseManager = contextManager.getMetaDataContextManager().getStatisticsManager();
53          Optional<String> databaseName = StatisticsNodePathParser.findDatabaseName(event.getKey(), false);
54          if (databaseName.isPresent()) {
55              handleDatabaseChanged(databaseManager, event.getType(), databaseName.get());
56              return;
57          }
58          databaseName = StatisticsNodePathParser.findDatabaseName(event.getKey(), true);
59          if (!databaseName.isPresent()) {
60              return;
61          }
62          Optional<String> schemaName = StatisticsNodePathParser.findSchemaName(event.getKey(), false);
63          if (schemaName.isPresent()) {
64              handleSchemaChanged(databaseManager, event.getType(), databaseName.get(), schemaName.get());
65              return;
66          }
67          schemaName = StatisticsNodePathParser.findSchemaName(event.getKey(), true);
68          if (!schemaName.isPresent()) {
69              return;
70          }
71          Optional<String> tableName = StatisticsNodePathParser.findTableName(event.getKey(), false);
72          if (tableName.isPresent()) {
73              handleTableChanged(databaseManager, event.getType(), databaseName.get(), schemaName.get(), tableName.get());
74              return;
75          }
76          tableName = StatisticsNodePathParser.findTableName(event.getKey(), true);
77          if (!tableName.isPresent()) {
78              return;
79          }
80          Optional<String> uniqueKey = StatisticsNodePathParser.findRowUniqueKey(event.getKey());
81          if (uniqueKey.isPresent()) {
82              handleRowDataChanged(databaseManager, event.getType(), event.getValue(), databaseName.get(), schemaName.get(), tableName.get(), uniqueKey.get());
83          }
84      }
85      
86      private void handleDatabaseChanged(final StatisticsManager databaseManager, final Type type, final String databaseName) {
87          switch (type) {
88              case ADDED:
89              case UPDATED:
90                  databaseManager.addDatabaseStatistics(databaseName);
91                  return;
92              case DELETED:
93                  databaseManager.dropDatabaseStatistics(databaseName);
94                  return;
95              default:
96          }
97      }
98      
99      private void handleSchemaChanged(final StatisticsManager databaseManager, final Type type, final String databaseName, final String schemaName) {
100         switch (type) {
101             case ADDED:
102             case UPDATED:
103                 databaseManager.addSchemaStatistics(databaseName, schemaName);
104                 return;
105             case DELETED:
106                 databaseManager.dropSchemaStatistics(databaseName, schemaName);
107                 return;
108             default:
109         }
110     }
111     
112     private void handleTableChanged(final StatisticsManager databaseManager, final Type type, final String databaseName, final String schemaName, final String tableName) {
113         switch (type) {
114             case ADDED:
115             case UPDATED:
116                 databaseManager.addTableStatistics(databaseName, schemaName, tableName);
117                 return;
118             case DELETED:
119                 databaseManager.dropTableStatistics(databaseName, schemaName, tableName);
120                 return;
121             default:
122         }
123     }
124     
125     private void handleRowDataChanged(final StatisticsManager databaseManager, final Type type, final String eventValue,
126                                       final String databaseName, final String schemaName, final String tableName, final String uniqueKey) {
127         if ((Type.ADDED == type || Type.UPDATED == type) && !Strings.isNullOrEmpty(eventValue)) {
128             databaseManager.alterRowStatistics(databaseName, schemaName, tableName, YamlEngine.unmarshal(eventValue, YamlRowStatistics.class));
129         } else if (Type.DELETED == type) {
130             databaseManager.deleteRowStatistics(databaseName, schemaName, tableName, uniqueKey);
131         }
132     }
133 }
