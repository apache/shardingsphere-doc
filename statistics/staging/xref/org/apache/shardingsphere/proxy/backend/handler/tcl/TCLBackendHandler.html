18  package org.apache.shardingsphere.proxy.backend.handler.tcl;
19  
20  import org.apache.shardingsphere.infra.database.core.metadata.database.metadata.DialectDatabaseMetaData;
21  import org.apache.shardingsphere.infra.database.core.type.DatabaseTypeRegistry;
22  import org.apache.shardingsphere.infra.exception.core.ShardingSpherePreconditions;
23  import org.apache.shardingsphere.infra.exception.dialect.exception.transaction.InTransactionException;
24  import org.apache.shardingsphere.proxy.backend.connector.TransactionManager;
25  import org.apache.shardingsphere.proxy.backend.connector.jdbc.transaction.BackendTransactionManager;
26  import org.apache.shardingsphere.proxy.backend.handler.ProxyBackendHandler;
27  import org.apache.shardingsphere.proxy.backend.response.header.ResponseHeader;
28  import org.apache.shardingsphere.proxy.backend.response.header.update.UpdateResponseHeader;
29  import org.apache.shardingsphere.proxy.backend.session.ConnectionSession;
30  import org.apache.shardingsphere.sql.parser.statement.core.statement.SQLStatement;
31  import org.apache.shardingsphere.sql.parser.statement.core.statement.tcl.ReleaseSavepointStatement;
32  import org.apache.shardingsphere.sql.parser.statement.core.statement.tcl.RollbackStatement;
33  import org.apache.shardingsphere.sql.parser.statement.core.statement.tcl.SavepointStatement;
34  import org.apache.shardingsphere.sql.parser.statement.core.statement.tcl.SetAutoCommitStatement;
35  import org.apache.shardingsphere.sql.parser.statement.core.statement.tcl.TCLStatement;
36  import org.apache.shardingsphere.transaction.core.TransactionOperationType;
37  
38  import java.sql.SQLException;
39  import java.sql.SQLFeatureNotSupportedException;
40  
41  /**
42   * TCL backend handler.
43   */
44  public final class TCLBackendHandler implements ProxyBackendHandler {
45      
46      private final TCLStatement tclStatement;
47      
48      private final TransactionOperationType operationType;
49      
50      private final ConnectionSession connectionSession;
51      
52      private final DialectDatabaseMetaData dialectDatabaseMetaData;
53      
54      private final TransactionManager backendTransactionManager;
55      
56      public TCLBackendHandler(final TCLStatement tclStatement, final TransactionOperationType operationType, final ConnectionSession connectionSession) {
57          this.tclStatement = tclStatement;
58          this.operationType = operationType;
59          this.connectionSession = connectionSession;
60          dialectDatabaseMetaData = new DatabaseTypeRegistry(connectionSession.getProtocolType()).getDialectDatabaseMetaData();
61          backendTransactionManager = new BackendTransactionManager(connectionSession.getDatabaseConnectionManager());
62      }
63      
64      @Override
65      public ResponseHeader execute() throws SQLException {
66          switch (operationType) {
67              case BEGIN:
68                  handleBegin();
69                  break;
70              case SAVEPOINT:
71                  handleSavepoint();
72                  break;
73              case ROLLBACK_TO_SAVEPOINT:
74                  handleRollbackToSavepoint();
75                  break;
76              case RELEASE_SAVEPOINT:
77                  handleReleaseSavepoint();
78                  break;
79              case COMMIT:
80                  SQLStatement sqlStatement = getSQLStatementByCommit();
81                  backendTransactionManager.commit();
82                  return new UpdateResponseHeader(sqlStatement);
83              case ROLLBACK:
84                  backendTransactionManager.rollback();
85                  break;
86              case SET_AUTOCOMMIT:
87                  handleSetAutoCommit();
88                  break;
89              default:
90                  throw new SQLFeatureNotSupportedException(operationType.name());
91          }
92          return new UpdateResponseHeader(tclStatement);
93      }
94      
95      private void handleBegin() throws SQLException {
96          if (connectionSession.getTransactionStatus().isInTransaction()) {
97              if (dialectDatabaseMetaData.getTransactionOption().isSupportAutoCommitInNestedTransaction()) {
98                  backendTransactionManager.commit();
99              } else if (dialectDatabaseMetaData.getSchemaOption().getDefaultSchema().isPresent()) {
100                 throw new InTransactionException();
101             }
102         }
103         backendTransactionManager.begin();
104     }
105     
106     private void handleSavepoint() throws SQLException {
107         ShardingSpherePreconditions.checkState(connectionSession.getTransactionStatus().isInTransaction() || !dialectDatabaseMetaData.getSchemaOption().getDefaultSchema().isPresent(),
108                 () -> new SQLFeatureNotSupportedException("SAVEPOINT can only be used in transaction blocks"));
109         backendTransactionManager.setSavepoint(((SavepointStatement) tclStatement).getSavepointName());
110     }
111     
112     private void handleRollbackToSavepoint() throws SQLException {
113         ShardingSpherePreconditions.checkState(connectionSession.getTransactionStatus().isInTransaction() || !dialectDatabaseMetaData.getSchemaOption().getDefaultSchema().isPresent(),
114                 () -> new SQLFeatureNotSupportedException("ROLLBACK TO SAVEPOINT can only be used in transaction blocks"));
115         backendTransactionManager.rollbackTo(((RollbackStatement) tclStatement).getSavepointName().get());
116     }
117     
118     private void handleReleaseSavepoint() throws SQLException {
119         ShardingSpherePreconditions.checkState(connectionSession.getTransactionStatus().isInTransaction() || !dialectDatabaseMetaData.getSchemaOption().getDefaultSchema().isPresent(),
120                 () -> new SQLFeatureNotSupportedException("RELEASE SAVEPOINT can only be used in transaction blocks"));
121         backendTransactionManager.releaseSavepoint(((ReleaseSavepointStatement) tclStatement).getSavepointName());
122     }
123     
124     private SQLStatement getSQLStatementByCommit() {
125         return connectionSession.getConnectionContext().getTransactionContext().isExceptionOccur() && dialectDatabaseMetaData.getTransactionOption().isReturnRollbackStatementWhenCommitFailed()
126                 ? new RollbackStatement()
127                 : tclStatement;
128     }
129     
130     private void handleSetAutoCommit() throws SQLException {
131         if (dialectDatabaseMetaData.getTransactionOption().isSupportAutoCommitInNestedTransaction() && connectionSession.getTransactionStatus().isInTransaction()
132                 && ((SetAutoCommitStatement) tclStatement).isAutoCommit()) {
133             backendTransactionManager.commit();
134         }
135         connectionSession.setAutoCommit(((SetAutoCommitStatement) tclStatement).isAutoCommit());
136     }
137 }
