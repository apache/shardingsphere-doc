82      private static final String CREATE_SHARDING_RULE_SQL = String.format("CREATE SHARDING TABLE RULE t_order("
83              + "STORAGE_UNITS(%s,%s),"
84              + "SHARDING_COLUMN=user_id,"
85              + "TYPE(NAME='hash_mod',PROPERTIES('sharding-count'='4')),"
86              + "KEY_GENERATE_STRATEGY(COLUMN=order_id,TYPE(NAME='snowflake'))"
87              + ")", PipelineContainerComposer.DS_0, PipelineContainerComposer.DS_1);
88      
89      private static final String SOURCE_TABLE_NAME = "t_order";
90      
91      @ParameterizedTest(name = "{0}")
92      @EnabledIf("isEnabled")
93      @ArgumentsSource(PipelineE2ETestCaseArgumentsProvider.class)
94      void assertCDCDataImportSuccess(final PipelineTestParameter testParam) throws SQLException {
95          // If run on NATIVE mode and database timezone is UTC, then: 1) set `e2e.timezone=UTC` in e2e-env.properties, 2) add this line of code in proxy Bootstrap main method.
96          // TimeZone.setDefault(TimeZone.getTimeZone("UTC"));
97          try (PipelineContainerComposer containerComposer = new PipelineContainerComposer(testParam)) {
98              PipelineE2EDistSQLFacade distSQLFacade = new PipelineE2EDistSQLFacade(containerComposer, new CDCJobType());
99              distSQLFacade.alterPipelineRule();
100             for (String each : Arrays.asList(PipelineContainerComposer.DS_0, PipelineContainerComposer.DS_1)) {
101                 distSQLFacade.registerStorageUnit(each);
102             }
103             createOrderTableRule(containerComposer);
104             distSQLFacade.createBroadcastRule("t_address");
105             QualifiedTable orderQualifiedTable = containerComposer.createQualifiedTableWithSchema(SOURCE_TABLE_NAME);
106             initSchemaAndTable(containerComposer, containerComposer.getProxyDataSource(), orderQualifiedTable, 3);
107             PipelineDataSource jdbcDataSource = new PipelineDataSource(containerComposer.generateShardingSphereDataSourceFromProxy(), containerComposer.getDatabaseType());
108             log.info("init data begin: {}", LocalDateTime.now());
109             IntPkLargeOrderDAO orderDAO = new IntPkLargeOrderDAO(jdbcDataSource, containerComposer.getDatabaseType(), orderQualifiedTable.format());
110             orderDAO.batchInsert(PipelineContainerComposer.TABLE_INIT_ROW_COUNT);
111             DataSourceExecuteUtils.executeBatch(jdbcDataSource, "INSERT INTO t_address(id, address_name) VALUES (?,?)", Arrays.asList(new Object[]{1, "a"}, new Object[]{2, "b"}));
112             DataSourceExecuteUtils.executeBatch(jdbcDataSource, "INSERT INTO t_single(id) VALUES (?)", Arrays.asList(new Object[]{1}, new Object[]{2}, new Object[]{3}));
113             log.info("init data end: {}", LocalDateTime.now());
114             PipelineDataSource targetDataSource = DataSourceTestUtils.createStandardDataSource(containerComposer.getActualJdbcUrlTemplate(PipelineContainerComposer.DS_4, false),
115                     containerComposer.getUsername(), containerComposer.getPassword());
116             initSchemaAndTable(containerComposer, targetDataSource, orderQualifiedTable, 0);
117             String jobId = runCDCIncrementalTaskAndGetJobId(containerComposer, distSQLFacade, jdbcDataSource, targetDataSource, orderQualifiedTable, orderDAO);
118             Awaitility.waitAtMost(10L, TimeUnit.SECONDS).pollInterval(500L, TimeUnit.MILLISECONDS)
119                     .until(() -> distSQLFacade.listJobs().stream().noneMatch(each -> Boolean.parseBoolean(each.get("active").toString())));
120             distSQLFacade.drop(jobId);
121             assertTrue(distSQLFacade.listJobs().isEmpty());
122         }
123     }
124     
125     private String runCDCIncrementalTaskAndGetJobId(final PipelineContainerComposer containerComposer, final PipelineE2EDistSQLFacade distSQLFacade, final PipelineDataSource jdbcDataSource,
126                                                     final PipelineDataSource targetDataSource, final QualifiedTable orderQualifiedTable, final IntPkLargeOrderDAO orderDAO) throws SQLException {
127         try (CDCClient ignored = buildCDCClientAndStart(targetDataSource, containerComposer)) {
128             Awaitility.waitAtMost(10L, TimeUnit.SECONDS).pollInterval(1L, TimeUnit.SECONDS).until(() -> !distSQLFacade.listJobIds().isEmpty());
129             String jobId = distSQLFacade.listJobIds().getFirst();
130             distSQLFacade.waitJobIncrementalStageFinished(jobId);
131             String orderTableName = orderQualifiedTable.format();
132             new E2EIncrementalTask(jdbcDataSource, orderTableName, new SnowflakeKeyGenerateAlgorithm(), containerComposer.getDatabaseType(), 20).run();
133             distSQLFacade.waitJobIncrementalStageFinished(jobId);
134             for (int i = 1; i <= 4; i++) {
135                 int orderId = 10000 + i;
136                 orderDAO.insert(orderId, i, "OK");
137                 containerComposer.assertRecordExists(targetDataSource, orderTableName, orderId);
138             }
139             assertDataMatched(jdbcDataSource, targetDataSource, orderQualifiedTable);
140             assertDataMatched(jdbcDataSource, targetDataSource, new QualifiedTable(null, "t_address"));
141             assertDataMatched(jdbcDataSource, targetDataSource, new QualifiedTable(null, "t_single"));
142             return jobId;
143         }
144     }

146     private void createOrderTableRule(final PipelineContainerComposer containerComposer) throws SQLException {
147         containerComposer.proxyExecuteWithLog(CREATE_SHARDING_RULE_SQL, 0);
148         Awaitility.waitAtMost(20L, TimeUnit.SECONDS).pollInterval(2L, TimeUnit.SECONDS).until(() -> !containerComposer.queryForListWithLog("SHOW SHARDING TABLE RULE t_order").isEmpty());
149     }

151     private void initSchemaAndTable(final PipelineContainerComposer containerComposer, final DataSource dataSource, final QualifiedTable orderQualifiedTable, final int seconds) throws SQLException {
152         containerComposer.createSchema(dataSource, seconds);
153         new IntPkLargeOrderDAO(dataSource, containerComposer.getDatabaseType(), orderQualifiedTable.format()).createTable();
154         try (
155                 Connection connection = dataSource.getConnection();
156                 Statement statement = connection.createStatement()) {
157             statement.execute("CREATE TABLE t_address(id integer primary key, address_name varchar(255))");
158             statement.execute("CREATE TABLE t_single(id integer primary key)");
159         }
160         containerComposer.sleepSeconds(seconds);
161     }
162     
163     private CDCClient buildCDCClientAndStart(final PipelineDataSource dataSource, final PipelineContainerComposer containerComposer) {
164         DataSourceRecordConsumer recordConsumer = new DataSourceRecordConsumer(dataSource, containerComposer.getDatabaseType());
165         CDCClient result = new CDCClient(new CDCClientConfiguration("localhost", containerComposer.getContainerComposer().getProxyCDCPort(), 10000));
166         result.connect(recordConsumer, new RetryStreamingExceptionHandler(result, 5, 5000), (ctx, serverErrorResult) -> log.error("Server error: {}", serverErrorResult.getErrorMessage()));
167         result.login(new CDCLoginParameter(ProxyContainerConstants.USER, ProxyContainerConstants.PASSWORD));
168         // TODO add full=false test case later
169         result.startStreaming(new StartStreamingParameter("sharding_db", Collections.singleton(SchemaTable.newBuilder().setTable("*").setSchema("*").build()), true));
170         return result;
171     }

173     private void assertDataMatched(final PipelineDataSource sourceDataSource, final PipelineDataSource targetDataSource, final QualifiedTable qualifiedTable) {
174         StandardPipelineTableMetaDataLoader metaDataLoader = new StandardPipelineTableMetaDataLoader(targetDataSource);
175         PipelineTableMetaData tableMetaData = metaDataLoader.getTableMetaData(qualifiedTable.getSchemaName(), qualifiedTable.getTableName());
176         List<PipelineColumnMetaData> uniqueKeys = Collections.singletonList(tableMetaData.getColumnMetaData(tableMetaData.getPrimaryKeyColumns().getFirst()));
177         ConsistencyCheckJobItemProgressContext progressContext = new ConsistencyCheckJobItemProgressContext("", 0, sourceDataSource.getDatabaseType().getType());
178         progressContext.getTableCheckRangePositions().add(new TableCheckRangePosition(0, null, qualifiedTable.getTableName(),
179                 UniqueKeyIngestPosition.ofUnsplit(), UniqueKeyIngestPosition.ofUnsplit(), null));
180         TableInventoryCheckParameter param = new TableInventoryCheckParameter("", sourceDataSource, targetDataSource, qualifiedTable, qualifiedTable,
181                 tableMetaData.getColumnNames(), uniqueKeys, null, progressContext);
182         TableDataConsistencyChecker tableChecker = TypedSPILoader.getService(TableDataConsistencyChecker.class, "DATA_MATCH", new Properties());
183         TableDataConsistencyCheckResult checkResult = tableChecker.buildTableInventoryChecker(param).checkSingleTableInventoryData();
184         assertTrue(checkResult.isMatched());
185     }
