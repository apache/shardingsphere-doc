18  package org.apache.shardingsphere.test.infra.framework.extension.log;
19  
20  import ch.qos.logback.classic.Level;
21  import ch.qos.logback.classic.Logger;
22  import ch.qos.logback.classic.spi.ILoggingEvent;
23  import ch.qos.logback.core.read.ListAppender;
24  import org.hamcrest.MatcherAssert;
25  import org.junit.jupiter.api.extension.AfterEachCallback;
26  import org.junit.jupiter.api.extension.BeforeEachCallback;
27  import org.junit.jupiter.api.extension.ExtensionContext;
28  import org.junit.jupiter.api.extension.ParameterContext;
29  import org.junit.jupiter.api.extension.ParameterResolver;
30  import org.slf4j.LoggerFactory;
31  
32  import static org.hamcrest.Matchers.is;
33  
34  /**
35   * Log capture extension.
36   */
37  public final class LogCaptureExtension implements BeforeEachCallback, AfterEachCallback, ParameterResolver {
38      
39      private ListAppender<ILoggingEvent> listAppender;
40      
41      private Logger logger;
42      
43      @Override
44      public void beforeEach(final ExtensionContext context) {
45          String targetClassName = context.getRequiredTestClass().getName().substring(0, context.getRequiredTestClass().getName().lastIndexOf("Test"));
46          logger = (Logger) LoggerFactory.getLogger(targetClassName);
47          listAppender = new ListAppender<>();
48          listAppender.start();
49          logger.addAppender(listAppender);
50      }
51      
52      @Override
53      public void afterEach(final ExtensionContext context) {
54          if (null != logger && null != listAppender) {
55              logger.detachAppender(listAppender);
56          }
57      }
58      
59      @Override
60      public boolean supportsParameter(final ParameterContext parameterContext, final ExtensionContext extensionContext) {
61          return parameterContext.getParameter().getType() == LogCaptureExtension.class;
62      }
63      
64      @Override
65      public Object resolveParameter(final ParameterContext parameterContext, final ExtensionContext extensionContext) {
66          return this;
67      }
68      
69      /**
70       * Assert the number of captured logs.
71       *
72       * @param expectedCount expected log count
73       */
74      public void assertLogCount(final int expectedCount) {
75          MatcherAssert.assertThat(listAppender.list.size(), is(expectedCount));
76      }
77      
78      /**
79       * Assert the number of captured logs with specific level and message.
80       *
81       * @param actualLogEventIndex actual log event index
82       * @param expectedLevel expected log level
83       * @param expectedMessage expected log message
84       */
85      public void assertLogContent(final int actualLogEventIndex, final Level expectedLevel, final String expectedMessage) {
86          MatcherAssert.assertThat(listAppender.list.get(actualLogEventIndex).getLevel(), is(expectedLevel));
87          MatcherAssert.assertThat(listAppender.list.get(actualLogEventIndex).getFormattedMessage(), is(expectedMessage));
88      }
89  }
