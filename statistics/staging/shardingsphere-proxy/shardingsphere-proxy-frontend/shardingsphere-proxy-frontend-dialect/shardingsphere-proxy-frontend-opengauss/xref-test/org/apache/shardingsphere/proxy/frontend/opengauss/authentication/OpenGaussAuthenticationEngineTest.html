399     private String createMd5Digest(final String username, final String password, final byte[] md5Salt) {
400         String passwordHash = new String(Hex.encodeHex(DigestUtils.md5(password + username), true));
401         MessageDigest messageDigest = DigestUtils.getMd5Digest();
402         messageDigest.update(passwordHash.getBytes());
403         messageDigest.update(md5Salt);
404         return "md5" + new String(Hex.encodeHex(messageDigest.digest(), true));
405     }

407     @SneakyThrows({GeneralSecurityException.class, DecoderException.class})
408     private String createScramDigest(final String password, final OpenGaussAuthenticationHexData authHexData, final int serverIteration) {
409         PBEKeySpec keySpec = new PBEKeySpec(password.toCharArray(), Hex.decodeHex(authHexData.getSalt()), serverIteration, 32 * 8);
410         byte[] secretKey = SecretKeyFactory.getInstance("PBKDF2WithHmacSHA1").generateSecret(keySpec).getEncoded();
411         Mac mac = Mac.getInstance("HmacSHA256");
412         mac.init(new SecretKeySpec(secretKey, "HmacSHA256"));
413         byte[] clientKey = mac.doFinal("Client Key".getBytes(StandardCharsets.UTF_8));
414         MessageDigest sha256 = MessageDigest.getInstance("SHA-256");
415         byte[] storedKey = sha256.digest(clientKey);
416         Mac clientSignatureMac = Mac.getInstance("HmacSHA256");
417         clientSignatureMac.init(new SecretKeySpec(storedKey, "HmacSHA256"));
418         byte[] clientSignature = clientSignatureMac.doFinal(Hex.decodeHex(authHexData.getNonce()));
419         byte[] clientProof = xor(clientKey, clientSignature);
420         return HexStringUtils.toHexString(clientProof);
421     }

423     private byte[] xor(final byte[] left, final byte[] right) {
424         byte[] result = new byte[left.length];
425         for (int index = 0; index < left.length; index++) {
426             result[index] = (byte) (left[index] ^ right[index]);
427         }
428         return result;
429     }
