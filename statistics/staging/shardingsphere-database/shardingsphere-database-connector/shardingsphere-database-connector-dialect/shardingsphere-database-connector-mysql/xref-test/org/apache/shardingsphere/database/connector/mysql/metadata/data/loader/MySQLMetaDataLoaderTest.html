98      @SuppressWarnings({"JDBCResourceOpenedButNotSafelyClosed", "resource"})
99      @ParameterizedTest(name = "{0}")
100     @MethodSource("loadArguments")
101     void assertLoad(final String name, final Collection<String> actualTableNames, final String tableMetaDataSQL,
102                     final String catalog, final String expectedDatabaseName, final boolean includeView, final boolean includeConstraints, final boolean includeCompositeIndex) throws SQLException {
103         DataSource dataSource = mockDataSource();
104         ResultSet tableMetaDataResultSet = mockTableMetaDataResultSet();
105         when(dataSource.getConnection().prepareStatement(tableMetaDataSQL).executeQuery()).thenReturn(tableMetaDataResultSet);
106         if (includeCompositeIndex) {
107             ResultSet compositeIndexMetaDataResultSet = mockCompositeIndexMetaDataResultSet();
108             when(dataSource.getConnection().prepareStatement(INDEX_METADATA_SQL).executeQuery()).thenReturn(compositeIndexMetaDataResultSet);
109         } else {
110             ResultSet simpleIndexMetaDataResultSet = mockSimpleIndexMetaDataResultSet();
111             when(dataSource.getConnection().prepareStatement(INDEX_METADATA_SQL).executeQuery()).thenReturn(simpleIndexMetaDataResultSet);
112         }
113         ResultSet viewResultSet = includeView ? mockSingleViewResultSet() : mock(ResultSet.class);
114         when(dataSource.getConnection().prepareStatement(VIEW_METADATA_SQL).executeQuery()).thenReturn(viewResultSet);
115         ResultSet constraintMetaDataResultSet = includeConstraints ? mockConstraintMetaDataResultSet() : mock(ResultSet.class);
116         when(dataSource.getConnection().prepareStatement(CONSTRAINT_METADATA_SQL).executeQuery()).thenReturn(constraintMetaDataResultSet);
117         Map<String, String> cachedDatabaseTables = GlobalDataSourceRegistry.getInstance().getCachedDatabaseTables();
118         String previous = cachedDatabaseTables.put("tbl", "fallback_db");
119         try {
120             when(dataSource.getConnection().getCatalog()).thenReturn(catalog);
121             DataTypeRegistry.load(dataSource, "MySQL");
122             Collection<SchemaMetaData> actual = dialectMetaDataLoader.load(new MetaDataLoaderMaterial(actualTableNames, "foo_ds", dataSource, databaseType, "sharding_db"));
123             assertTableMetaData(actual, includeConstraints, includeCompositeIndex);
124             verify(dataSource.getConnection().prepareStatement(tableMetaDataSQL)).setString(1, expectedDatabaseName);
125         } finally {
126             if (null == previous) {
127                 cachedDatabaseTables.remove("tbl");
128             } else {
129                 cachedDatabaseTables.put("tbl", previous);
130             }
131         }
132     }

200     private void assertTableMetaData(final Collection<SchemaMetaData> schemaMetaDataList, final boolean expectedConstraintExists, final boolean expectedCompositeIndexExists) {
201         assertThat(schemaMetaDataList.size(), is(1));
202         Collection<TableMetaData> tables = schemaMetaDataList.iterator().next().getTables();
203         assertThat(tables.size(), is(1));
204         TableMetaData actualTableMetaData = tables.iterator().next();
205         assertThat(actualTableMetaData.getName(), is("tbl"));
206         assertThat(actualTableMetaData.getColumns().size(), is(9));
207         Iterator<ColumnMetaData> columnsIterator = actualTableMetaData.getColumns().iterator();
208         assertColumnMetaData(columnsIterator.next(), new ColumnMetaData("id", Types.INTEGER, !expectedCompositeIndexExists, true, true, false, false, false));
209         assertColumnMetaData(columnsIterator.next(), new ColumnMetaData("name", Types.VARCHAR, false, false, false, false, false, true));
210         assertColumnMetaData(columnsIterator.next(), new ColumnMetaData("doc", Types.LONGVARCHAR, false, false, false, true, false, true));
211         assertColumnMetaData(columnsIterator.next(), new ColumnMetaData("geo", Types.BINARY, false, false, false, true, false, true));
212         assertColumnMetaData(columnsIterator.next(), new ColumnMetaData("t_year", Types.DATE, false, false, false, true, false, true));
213         assertColumnMetaData(columnsIterator.next(), new ColumnMetaData("pg", Types.BINARY, false, false, false, true, false, true));
214         assertColumnMetaData(columnsIterator.next(), new ColumnMetaData("mpg", Types.BINARY, false, false, false, true, false, true));
215         assertColumnMetaData(columnsIterator.next(), new ColumnMetaData("pt", Types.BINARY, false, false, false, true, false, true));
216         assertColumnMetaData(columnsIterator.next(), new ColumnMetaData("mpt", Types.BINARY, false, false, false, true, false, true));
217         assertIndexMetaData(actualTableMetaData.getIndexes(), expectedCompositeIndexExists);
218         assertConstraintMetaData(actualTableMetaData.getConstraints(), expectedConstraintExists);
219     }
220     
221     private void assertColumnMetaData(final ColumnMetaData actual, final ColumnMetaData expected) {
222         assertThat(actual.getName(), is(expected.getName()));
223         assertThat(actual.getDataType(), is(expected.getDataType()));
224         assertThat(actual.isPrimaryKey(), is(expected.isPrimaryKey()));
225         assertThat(actual.isGenerated(), is(expected.isGenerated()));
226         assertThat(actual.isCaseSensitive(), is(expected.isCaseSensitive()));
227         assertThat(actual.isVisible(), is(expected.isVisible()));
228         assertThat(actual.isUnsigned(), is(expected.isUnsigned()));
229         assertThat(actual.isNullable(), is(expected.isNullable()));
230     }
231     
232     private void assertIndexMetaData(final Collection<IndexMetaData> actualIndexMetaData, final boolean expectedCompositeIndexExists) {
233         assertThat(actualIndexMetaData.size(), is(1));
234         IndexMetaData actualIndexMetaDataItem = actualIndexMetaData.iterator().next();
235         assertThat(actualIndexMetaDataItem.getName(), is(expectedCompositeIndexExists ? "idx_composite" : "PRIMARY"));
236         assertThat(actualIndexMetaDataItem.getColumns(), is(expectedCompositeIndexExists ? Arrays.asList("id", "name") : Collections.singletonList("id")));
237         assertTrue(actualIndexMetaDataItem.isUnique());
238     }
239     
240     private void assertConstraintMetaData(final Collection<ConstraintMetaData> actualConstraintMetaData, final boolean expectedConstraintExists) {
241         if (expectedConstraintExists) {
242             assertThat(actualConstraintMetaData.size(), is(2));
243             Iterator<ConstraintMetaData> constraints = actualConstraintMetaData.iterator();
244             assertThat(constraints.next().getName(), is("fk_order"));
245             assertThat(constraints.next().getName(), is("fk_item"));
246         } else {
247             assertTrue(actualConstraintMetaData.isEmpty());
248         }
249     }

251     private static Stream<Arguments> loadArguments() {
252         return Stream.of(
253                 Arguments.of("load without requested table names", Collections.emptyList(), TABLE_METADATA_SQL, "sharding_db", "sharding_db", false, false, false),
254                 Arguments.of("load with requested table names", Collections.singletonList("tbl"), TABLE_METADATA_SQL_WITH_TABLE, "sharding_db", "sharding_db", false, false, false),
255                 Arguments.of("load with empty catalog fallback and view constraints", Collections.singletonList("tbl"), TABLE_METADATA_SQL_WITH_TABLE, "", "fallback_db", true, true, true),
256                 Arguments.of("load with null catalog fallback", Collections.singletonList("tbl"), TABLE_METADATA_SQL_WITH_TABLE, null, "fallback_db", true, true, true));
257     }
