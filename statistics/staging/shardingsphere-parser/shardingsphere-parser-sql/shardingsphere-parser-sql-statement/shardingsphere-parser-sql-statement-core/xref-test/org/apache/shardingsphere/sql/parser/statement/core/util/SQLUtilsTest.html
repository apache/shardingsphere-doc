21  import org.apache.shardingsphere.sql.parser.statement.core.value.literal.impl.DateTimeLiteralValue;
22  import org.apache.shardingsphere.sql.parser.statement.core.value.literal.impl.TemporalLiteralValue;
23  import org.junit.jupiter.api.Test;
24  
25  import java.math.BigDecimal;
26  import java.math.BigInteger;
27  
28  import static org.hamcrest.MatcherAssert.assertThat;
29  import static org.hamcrest.Matchers.is;
30  import static org.junit.jupiter.api.Assertions.assertNull;
31  
32  class SQLUtilsTest {
33      
34      @Test
35      void assertGetExactlyNumberForInteger() {
36          assertThat(SQLUtils.getExactlyNumber("100000", 10), is(100000));
37          assertThat(SQLUtils.getExactlyNumber("100000", 16), is(1048576));
38          assertThat(SQLUtils.getExactlyNumber(String.valueOf(Integer.MIN_VALUE), 10), is(Integer.MIN_VALUE));
39          assertThat(SQLUtils.getExactlyNumber(String.valueOf(Integer.MAX_VALUE), 10), is(Integer.MAX_VALUE));
40      }
41      
42      @Test
43      void assertGetExactlyNumberForLong() {
44          assertThat(SQLUtils.getExactlyNumber("100000000000", 10), is(100000000000L));
45          assertThat(SQLUtils.getExactlyNumber("100000000000", 16), is(17592186044416L));
46          assertThat(SQLUtils.getExactlyNumber(String.valueOf(Long.MIN_VALUE), 10), is(Long.MIN_VALUE));
47          assertThat(SQLUtils.getExactlyNumber(String.valueOf(Long.MAX_VALUE), 10), is(Long.MAX_VALUE));
48      }
49      
50      @Test
51      void assertGetExactlyNumberForBigInteger() {
52          assertThat(SQLUtils.getExactlyNumber("10000000000000000000", 10), is(new BigInteger("10000000000000000000")));
53          assertThat(SQLUtils.getExactlyNumber("10000000000000000000", 16), is(new BigInteger("75557863725914323419136")));
54          assertThat(SQLUtils.getExactlyNumber(String.valueOf(Long.MIN_VALUE + 1L), 10), is(Long.MIN_VALUE + 1L));
55          assertThat(SQLUtils.getExactlyNumber(String.valueOf(Long.MAX_VALUE - 1L), 10), is(Long.MAX_VALUE - 1L));
56      }
57      
58      @Test
59      void assertGetExactlyNumberForBigDecimal() {
60          assertThat(SQLUtils.getExactlyNumber("1.1", 10), is(new BigDecimal("1.1")));
61      }
62      
63      @Test
64      void assertGetExactlyValue() {
65          assertThat(SQLUtils.getExactlyValue("`xxx`"), is("xxx"));
66          assertThat(SQLUtils.getExactlyValue("[xxx]"), is("xxx"));
67          assertThat(SQLUtils.getExactlyValue("\"xxx\""), is("xxx"));
68          assertThat(SQLUtils.getExactlyValue("'xxx'"), is("xxx"));
69          assertThat(SQLUtils.getExactlyValue("`[xxx`"), is("xxx"));
70          assertThat(SQLUtils.getExactlyValue("```[xxx```"), is("`xxx`"));
71      }
72      
73      @Test
74      void assertGetExactlyValueWithReservedCharacters() {
75          assertThat(SQLUtils.getExactlyValue("`xxx`", "`"), is("`xxx`"));
76          assertThat(SQLUtils.getExactlyValue("[xxx]", "[]"), is("[xxx]"));
77          assertThat(SQLUtils.getExactlyValue("\"xxx\"", "\""), is("\"xxx\""));
78          assertThat(SQLUtils.getExactlyValue("'xxx'", "'"), is("'xxx'"));
79      }
80      
81      @Test
82      void assertGetExactlyValueUsingNull() {
83          assertNull(SQLUtils.getExactlyValue(null));
84      }
85      
86      @Test
87      void assertGetExactlyValueWithReservedCharactersUsingNull() {
88          assertNull(SQLUtils.getExactlyValue(null, "`"));
89      }
90      
91      @Test
92      void assertGetExactlyExpressionUsingAndReturningNull() {
93          assertNull(SQLUtils.getExactlyExpression(null));
94      }
95      
96      @Test
97      void assertGetExactlyExpressionUsingAndReturningEmptyString() {
98          assertThat(SQLUtils.getExactlyExpression(""), is(""));
99      }
100     
101     @Test
102     void assertGetExactlyExpression() {
103         assertThat(SQLUtils.getExactlyExpression("((a + b*c))"), is("((a+b*c))"));
104     }
105     
106     @Test
107     void assertCreateLiteralExpressionForTemporalLiteral() {
108         TemporalLiteralExpressionSegment actual = (TemporalLiteralExpressionSegment) SQLUtils.createLiteralExpression(
109                 new TemporalLiteralValue("DATE", "'2024-03-01'"), 0, 16, "DATE '2024-03-01'");
110         assertThat(actual.getLiterals(), is("2024-03-01"));
111         assertThat(actual.getText(), is("DATE '2024-03-01'"));
112     }
113     
114     @Test
115     void assertCreateLiteralExpressionForDateTimeLiteral() {
116         TemporalLiteralExpressionSegment actual = (TemporalLiteralExpressionSegment) SQLUtils.createLiteralExpression(
117                 new DateTimeLiteralValue("DATE", "'2024-03-01'", false), 0, 16, "DATE '2024-03-01'");
118         assertThat(actual.getLiterals(), is("2024-03-01"));
119         assertThat(actual.getText(), is("DATE '2024-03-01'"));
120     }
121     
122     @Test
123     void assertCreateLiteralExpressionForBraceDateTimeLiteral() {
124         TemporalLiteralExpressionSegment actual = (TemporalLiteralExpressionSegment) SQLUtils.createLiteralExpression(
125                 new DateTimeLiteralValue("ts", "'2024-03-01 10:00:00'", true), 0, 25, "{ts '2024-03-01 10:00:00'}");
126         assertThat(actual.getLiterals(), is("2024-03-01 10:00:00"));
127         assertThat(actual.getText(), is("{ts '2024-03-01 10:00:00'}"));
128     }
129     
130     @Test
131     void assertGetExpressionWithoutOutsideParentheses() {
132         assertThat(SQLUtils.getExpressionWithoutOutsideParentheses("((a + b*c))"), is("a + b*c"));
133         assertThat(SQLUtils.getExpressionWithoutOutsideParentheses(""), is(""));
134     }
135     
136     @Test
137     void assertTrimSemiColon() {
138         assertThat(SQLUtils.trimSemicolon("SHOW DATABASES;"), is("SHOW DATABASES"));
139         assertThat(SQLUtils.trimSemicolon("SHOW DATABASES"), is("SHOW DATABASES"));
140     }
141     
142     @Test
143     void assertTrimComment() {
144         assertThat(SQLUtils.trimComment("/* This is a comment */ SHOW DATABASES"), is("SHOW DATABASES"));
145         assertThat(SQLUtils.trimComment("/* This is a query with a semicolon */ SHOW DATABASES;"), is("SHOW DATABASES"));
146         assertThat(SQLUtils.trimComment("/* This is a query with spaces */    SHOW DATABASES   "), is("SHOW DATABASES"));
147     }
148     
149     @Test
150     void assertTryGetRealContentInBackticks() {
151         assertThat(SQLUtils.tryGetRealContentInBackticks("`"), is("`"));
152         assertThat(SQLUtils.tryGetRealContentInBackticks("``"), is("``"));
153         assertThat(SQLUtils.tryGetRealContentInBackticks("```"), is("```"));
154         assertThat(SQLUtils.tryGetRealContentInBackticks("t_order"), is("t_order"));
155         assertThat(SQLUtils.tryGetRealContentInBackticks("````"), is("`"));
156         assertThat(SQLUtils.tryGetRealContentInBackticks("`t_order`"), is("t_order"));
157         assertThat(SQLUtils.tryGetRealContentInBackticks("```t_order`"), is("`t_order"));
158         assertThat(SQLUtils.tryGetRealContentInBackticks("```````t_order```````"), is("```t_order```"));
159         assertThat(SQLUtils.tryGetRealContentInBackticks("``t_order`"), is("``t_order`"));
160         assertThat(SQLUtils.tryGetRealContentInBackticks("`````t_o``r``d``e``r```"), is("``t_o`r`d`e`r`"));
161     }
