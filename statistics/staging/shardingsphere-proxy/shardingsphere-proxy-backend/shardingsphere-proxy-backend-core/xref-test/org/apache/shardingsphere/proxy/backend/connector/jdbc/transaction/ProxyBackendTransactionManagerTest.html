132     @ParameterizedTest(name = "{0}")
133     @MethodSource("provideBeginScenarios")
134     void assertBeginScenarios(final String name, final TransactionType defaultType, final boolean initialInTransaction, final boolean hasDistributedManager,
135                               final boolean expectLocalBegin, final boolean expectDistributedBegin, final boolean expectCloseResources,
136                               final boolean expectSetInTransaction, final boolean expectTransactionTypeSet, final boolean hasHook) {
137         when(transactionStatus.isInTransaction()).thenReturn(initialInTransaction);
138         TransactionHook transactionHook = hasHook ? mock(TransactionHook.class) : null;
139         Map<ShardingSphereRule, TransactionHook> transactionHooks = hasHook ? Collections.singletonMap(mock(ShardingSphereRule.class), transactionHook) : Collections.emptyMap();
140         ShardingSphereTransactionManagerEngine transactionManagerEngine = hasDistributedManager ? mock(ShardingSphereTransactionManagerEngine.class) : null;
141         if (hasDistributedManager) {
142             when(transactionManagerEngine.getTransactionManager(TransactionType.XA)).thenReturn(distributedTransactionManager);
143         }
144         mockProxyContext(defaultType, transactionManagerEngine, transactionHooks);
145         ProxyBackendTransactionManager transactionManager = new ProxyBackendTransactionManager(databaseConnectionManager);
146         setLocalTransactionManager(transactionManager);
147         transactionManager.begin();
148         if (expectCloseResources) {
149             verify(databaseConnectionManager).closeHandlers(true);
150             verify(databaseConnectionManager).closeConnections(false);
151         } else {
152             verify(databaseConnectionManager, never()).closeHandlers(true);
153             verify(databaseConnectionManager, never()).closeConnections(false);
154         }
155         if (expectSetInTransaction) {
156             verify(transactionStatus).setInTransaction(true);
157         } else {
158             verify(transactionStatus, never()).setInTransaction(true);
159         }
160         if (hasHook) {
161             verify(transactionHook).beforeBegin(any(), any(DatabaseType.class), eq(transactionContext));
162             verify(transactionHook).afterBegin(any(), any(DatabaseType.class), eq(transactionContext));
163         }
164         if (expectLocalBegin) {
165             verify(localTransactionManager).begin();
166         } else {
167             verify(localTransactionManager, never()).begin();
168         }
169         if (expectDistributedBegin) {
170             verify(distributedTransactionManager).begin();
171         } else {
172             verify(distributedTransactionManager, never()).begin();
173         }
174         if (expectTransactionTypeSet) {
175             assertTrue(transactionContext.isTransactionStarted());
176             assertThat(transactionContext.getTransactionType().get(), is(defaultType.name()));
177         } else {
178             assertFalse(transactionContext.isTransactionStarted());
179         }
180     }

192     @ParameterizedTest(name = "{0}")
193     @MethodSource("provideCommitScenarios")
194     void assertCommitScenarios(final String name, final TransactionType transactionType, final boolean needLock, final boolean hasDistributedManager,
195                                final boolean exceptionOccurred, final boolean hasHook, final boolean expectDistributedCommit) throws SQLException {
196         when(transactionStatus.isInTransaction()).thenReturn(true);
197         transactionContext.setExceptionOccur(exceptionOccurred);
198         ConnectionSavepointManager savepointManager = mock(ConnectionSavepointManager.class);
199         when(ConnectionSavepointManager.getInstance()).thenReturn(savepointManager);
200         TransactionHook transactionHook = hasHook ? mock(TransactionHook.class) : null;
201         if (hasHook) {
202             when(transactionHook.isNeedLockWhenCommit(any())).thenReturn(needLock);
203         }
204         Map<ShardingSphereRule, TransactionHook> transactionHooks = hasHook ? Collections.singletonMap(mock(ShardingSphereRule.class), transactionHook) : Collections.emptyMap();
205         ShardingSphereTransactionManagerEngine transactionManagerEngine = hasDistributedManager ? mock(ShardingSphereTransactionManagerEngine.class) : null;
206         if (hasDistributedManager) {
207             when(transactionManagerEngine.getTransactionManager(TransactionType.XA)).thenReturn(distributedTransactionManager);
208         }
209         if (needLock) {
210             doAnswer(invocation -> {
211                 ExclusiveOperationVoidCallback callback = invocation.getArgument(2);
212                 callback.execute();
213                 return null;
214             }).when(exclusiveOperatorEngine).operate(any(), anyLong(), any());
215         }
216         mockProxyContext(transactionType, transactionManagerEngine, transactionHooks);
217         ProxyBackendTransactionManager transactionManager = new ProxyBackendTransactionManager(databaseConnectionManager);
218         setLocalTransactionManager(transactionManager);
219         transactionManager.commit();
220         if (needLock) {
221             verify(exclusiveOperatorEngine).operate(any(), anyLong(), any());
222         } else {
223             verify(exclusiveOperatorEngine, never()).operate(any(), anyLong(), any());
224         }
225         if (hasHook) {
226             verify(transactionHook).beforeCommit(any(), any(DatabaseType.class), anyCollection(), eq(transactionContext));
227         }
228         if (expectDistributedCommit) {
229             verify(distributedTransactionManager).commit(exceptionOccurred);
230         } else {
231             verify(localTransactionManager).commit();
232             verify(distributedTransactionManager, never()).commit(anyBoolean());
233         }
234         verify(savepointManager).transactionFinished(connection);
235         verify(transactionStatus).setInTransaction(false);
236         verify(connectionContext).close();
237     }

249     @ParameterizedTest(name = "{0}")
250     @MethodSource("provideRollbackScenarios")
251     void assertRollbackScenarios(final String name, final TransactionType transactionType, final boolean secondInTransaction,
252                                  final boolean hasDistributedManager, final boolean expectDistributedRollback, final boolean hasHook,
253                                  final boolean expectAfterHook, final boolean expectClear) throws SQLException {
254         when(transactionStatus.isInTransaction()).thenReturn(true, secondInTransaction);
255         ConnectionSavepointManager savepointManager = mock(ConnectionSavepointManager.class);
256         when(ConnectionSavepointManager.getInstance()).thenReturn(savepointManager);
257         TransactionHook transactionHook = hasHook ? mock(TransactionHook.class) : null;
258         Map<ShardingSphereRule, TransactionHook> transactionHooks = hasHook ? Collections.singletonMap(mock(ShardingSphereRule.class), transactionHook) : Collections.emptyMap();
259         ShardingSphereTransactionManagerEngine transactionManagerEngine = hasDistributedManager ? mock(ShardingSphereTransactionManagerEngine.class) : null;
260         if (hasDistributedManager) {
261             when(transactionManagerEngine.getTransactionManager(TransactionType.XA)).thenReturn(distributedTransactionManager);
262         }
263         mockProxyContext(transactionType, transactionManagerEngine, transactionHooks);
264         ProxyBackendTransactionManager transactionManager = new ProxyBackendTransactionManager(databaseConnectionManager);
265         setLocalTransactionManager(transactionManager);
266         transactionManager.rollback();
267         if (hasHook) {
268             verify(transactionHook).beforeRollback(any(), any(DatabaseType.class), anyCollection(), eq(transactionContext));
269         }
270         if (expectDistributedRollback) {
271             verify(distributedTransactionManager).rollback();
272         } else if (expectClear) {
273             verify(localTransactionManager).rollback();
274             verify(distributedTransactionManager, never()).rollback();
275         } else {
276             verify(localTransactionManager, never()).rollback();
277             verify(distributedTransactionManager, never()).rollback();
278         }
279         if (hasHook) {
280             if (expectAfterHook) {
281                 verify(transactionHook).afterRollback(any(), any(DatabaseType.class), anyCollection(), eq(transactionContext));
282             } else {
283                 verify(transactionHook, never()).afterRollback(any(), any(DatabaseType.class), anyCollection(), eq(transactionContext));
284             }
285         }
286         if (expectClear) {
287             verify(savepointManager).transactionFinished(connection);
288             verify(transactionStatus).setInTransaction(false);
289             verify(connectionContext).close();
290         } else {
291             verify(savepointManager, never()).transactionFinished(any());
292             verify(transactionStatus, never()).setInTransaction(false);
293             verify(connectionContext, never()).close();
294         }
295     }

312     @ParameterizedTest(name = "{0}")
313     @MethodSource("provideRollbackToScenarios")
314     void assertRollbackToScenarios(final String name, final boolean initialExceptionFlag, final int exceptionCount, final boolean expectFlagCleared,
315                                    final boolean expectThrows) throws SQLException {
316         ConnectionSavepointManager savepointManager = mock(ConnectionSavepointManager.class);
317         when(ConnectionSavepointManager.getInstance()).thenReturn(savepointManager);
318         transactionContext.setExceptionOccur(initialExceptionFlag);
319         if (exceptionCount == 2) {
320             Connection anotherConnection = mock(Connection.class);
321             mockCachedConnections(connection, anotherConnection);
322             doThrow(new SQLException("first")).when(savepointManager).rollbackToSavepoint(connection, "sp");
323             doThrow(new SQLException("second")).when(savepointManager).rollbackToSavepoint(anotherConnection, "sp");
324         } else {
325             mockCachedConnections(connection);
326         }
327         mockProxyContext(TransactionType.LOCAL, null, Collections.emptyMap());
328         ProxyBackendTransactionManager transactionManager = new ProxyBackendTransactionManager(databaseConnectionManager);
329         if (expectThrows) {
330             SQLException actualException = assertThrows(SQLException.class, () -> transactionManager.rollbackTo("sp"));
331             assertThat(actualException.getMessage(), is("first"));
332             assertThat(actualException.getNextException().getMessage(), is("second"));
333         } else {
334             transactionManager.rollbackTo("sp");
335             verify(savepointManager).rollbackToSavepoint(connection, "sp");
336             if (exceptionCount == 0) {
337                 assertFalse(transactionContext.isExceptionOccur());
338             }
339             if (expectFlagCleared) {
340                 assertFalse(transactionContext.isExceptionOccur());
341             }
342         }
343     }

376     private void mockProxyContext(final TransactionType defaultType, final ShardingSphereTransactionManagerEngine engine, final Map<ShardingSphereRule, TransactionHook> transactionHooks) {
377         TransactionRule transactionRule = mock(TransactionRule.class);
378         when(transactionRule.getDefaultType()).thenReturn(defaultType);
379         when(transactionRule.getResource()).thenReturn(engine);
380         ContextManager contextManager = mock(ContextManager.class, Answers.RETURNS_DEEP_STUBS);
381         when(contextManager.getMetaDataContexts().getMetaData().getGlobalRuleMetaData()).thenReturn(new RuleMetaData(Collections.singleton(transactionRule)));
382         when(contextManager.getExclusiveOperatorEngine()).thenReturn(exclusiveOperatorEngine);
383         when(ProxyContext.getInstance().getContextManager()).thenReturn(contextManager);
384         when(OrderedSPILoader.<ShardingSphereRule, TransactionHook>getServices(eq(TransactionHook.class), anyCollection())).thenReturn(transactionHooks);
385     }
