79      @ParameterizedTest(name = "{0}")
80      @MethodSource("shorthandScopeCases")
81      void assertBindShorthandScope(final String name, final String localAlias, final boolean qualified, final String expectedLabel) {
82          ShorthandProjectionSegment shorthand = new ShorthandProjectionSegment(7, 18);
83          if (qualified) {
84              shorthand.setOwner(new OwnerSegment(7, 16, new IdentifierValue("foo_source")));
85          }
86          ProjectionsSegment projections = new ProjectionsSegment(7, 18);
87          projections.getProjections().add(shorthand);
88          Multimap<CaseInsensitiveString, TableSegmentBinderContext> local = LinkedHashMultimap.create();
89          if (!localAlias.isEmpty()) {
90              local.put(CaseInsensitiveString.of(localAlias), createSourceContext("foo_local"));
91          }
92          Multimap<CaseInsensitiveString, TableSegmentBinderContext> outer = LinkedHashMultimap.create();
93          outer.put(CaseInsensitiveString.of("foo_source"), createSourceContext("foo_outer"));
94          SQLStatementBinderContext context = mock(SQLStatementBinderContext.class);
95          when(context.getSqlStatement()).thenReturn(SelectStatement.builder().databaseType(TypedSPILoader.getService(DatabaseType.class, "FIXTURE")).projections(projections).build());
96          SimpleTableSegment boundTable = localAlias.isEmpty() ? null : mock(SimpleTableSegment.class);
97          if (null != boundTable) {
98              TableNameSegment tableName = mock(TableNameSegment.class);
99              when(tableName.getIdentifier()).thenReturn(new IdentifierValue(localAlias));
100             when(boundTable.getTableName()).thenReturn(tableName);
101         }
102         ProjectionsSegment actual = ProjectionsSegmentBinder.bind(projections, context, boundTable, local, outer);
103         ShorthandProjectionSegment actualShorthand = (ShorthandProjectionSegment) actual.getProjections().iterator().next();
104         assertThat(actualShorthand.getActualProjectionSegments().size(), is(1));
105         assertThat(actualShorthand.getActualProjectionSegments().iterator().next().getColumnLabel(), is(expectedLabel));
106     }

109         return Stream.of(
110                 Arguments.of("correlated star without local FROM", "", true, "foo_outer"),

112                 Arguments.of("local alias shadows outer", "foo_source", true, "foo_local"),
113                 Arguments.of("unqualified star stays local", "foo_source", false, "foo_local"));
114     }

122     @Test
123     void assertBindValuesRowsPreservesProjectionShape() {
124         ProjectionsSegment projections = new ProjectionsSegment(0, 15);
125         LiteralExpressionSegment firstValue = new LiteralExpressionSegment(8, 8, 1);
126         projections.getProjections().add(new ExpressionProjectionSegment(8, 8, "1", firstValue));
127         RowExpression first = new RowExpression(7, 9, "(1)");
128         first.getItems().add(firstValue);
129         projections.getValuesRows().add(first);
130         RowExpression second = new RowExpression(12, 14, "(2)");
131         LiteralExpressionSegment secondValue = new LiteralExpressionSegment(13, 13, 2);
132         second.getItems().add(secondValue);
133         projections.getValuesRows().add(second);
134         SQLStatementBinderContext context = mock(SQLStatementBinderContext.class);
135         when(context.getSqlStatement()).thenReturn(SelectStatement.builder().databaseType(TypedSPILoader.getService(DatabaseType.class, "FIXTURE")).projections(projections).build());
136         ProjectionsSegment actual = ProjectionsSegmentBinder.bind(projections, context, null, LinkedHashMultimap.create(), LinkedHashMultimap.create());
137         assertThat(actual.getProjections().size(), is(1));
138         assertThat(actual.getValuesRows().size(), is(2));
139         assertThat(actual.getValuesRows().get(1), not(sameInstance(second)));
140         assertThat(actual.getValuesRows().get(1).getText(), is("(2)"));
141         assertThat(actual.getValuesRows().get(1).getStartIndex(), is(12));
142         assertThat(actual.getValuesRows().get(1).getStopIndex(), is(14));
143         assertThat(actual.getValuesRows().get(1).getItems().iterator().next(), sameInstance(secondValue));
144     }

146     @ParameterizedTest(name = "local={0}, expected local={1}")
147     @CsvSource({"foo,true", "bar,false", "'',false"})
148     void assertBindParenthesizedRowAlias(final String localAlias, final boolean expectedLocal) {
149         ColumnSegment column = new ColumnSegment(1, 3, new IdentifierValue("foo"));
150         CommonExpressionSegment expression = new CommonExpressionSegment(0, 6, "(foo).*", Collections.singletonList(column));
151         expression.setRowExpansion(true);
152         ExpressionProjectionSegment projection = new ExpressionProjectionSegment(0, 6, "(foo).*", expression);
153         AliasSegment alias = new AliasSegment(8, 16, new IdentifierValue("foo_alias"));
154         projection.setAlias(alias);
155         ProjectionsSegment projections = new ProjectionsSegment(0, 16);
156         projections.getProjections().add(projection);
157         Multimap<CaseInsensitiveString, TableSegmentBinderContext> tables = LinkedHashMultimap.create();
158         if (!localAlias.isEmpty()) {
159             tables.put(CaseInsensitiveString.of(localAlias), mock(TableSegmentBinderContext.class));
160         }
161         Multimap<CaseInsensitiveString, TableSegmentBinderContext> outer = LinkedHashMultimap.create();
162         outer.put(CaseInsensitiveString.of("foo"), mock(TableSegmentBinderContext.class));
163         Multimap<CaseInsensitiveString, TableSegmentBinderContext> expectedContexts = expectedLocal ? tables : outer;
164         SQLStatementBinderContext context = mock(SQLStatementBinderContext.class, RETURNS_DEEP_STUBS);
165         when(context.getSqlStatement().getDatabaseType()).thenReturn(TypedSPILoader.getService(DatabaseType.class, "FIXTURE"));
166         DialectDatabaseMetaData metadata = mock(DialectDatabaseMetaData.class, RETURNS_DEEP_STUBS);
167         when(metadata.getSQLOption().isSupportWholeRowProjection()).thenReturn(true);
168         try (
169                 MockedStatic<ExpressionSegmentBinder> expressions = mockStatic(ExpressionSegmentBinder.class);
170                 MockedStatic<ShorthandProjectionSegmentBinder> shorthands = mockStatic(ShorthandProjectionSegmentBinder.class);
171                 MockedConstruction<DatabaseTypeRegistry> ignored = mockConstruction(DatabaseTypeRegistry.class,
172                         (registry, construction) -> when(registry.getDialectDatabaseMetaData()).thenReturn(metadata))) {
173             expressions.when(() -> ExpressionSegmentBinder.bind(same(expression), eq(SegmentType.PROJECTION), same(context), any(), same(outer)))
174                     .thenThrow(new ColumnNotFoundException("foo", "field list"));
175             shorthands.when(() -> ShorthandProjectionSegmentBinder.bind(any(), eq(null), same(expectedContexts))).thenAnswer(invocation -> invocation.getArgument(0));
176             ProjectionsSegment actual = ProjectionsSegmentBinder.bind(projections, context, null, tables, outer);
177             ShorthandProjectionSegment actualProjection = (ShorthandProjectionSegment) actual.getProjections().iterator().next();
178             assertThat(actualProjection.getStartIndex(), is(0));
179             assertThat(actualProjection.getStopIndex(), is(16));
180             assertThat(actualProjection.getOwner().get().getIdentifier(), is(new IdentifierValue("foo")));
181             assertThat(actualProjection.getAliasSegment().get(), sameInstance(alias));
182             assertThat(projection.getExpr(), sameInstance(expression));
183             assertTrue(expression.isRowExpansion());
184         }
185     }

187     @Test
188     void assertBindCompositeColumnBeforeRowAlias() {
189         ColumnSegment column = new ColumnSegment(1, 3, new IdentifierValue("foo"));
190         CommonExpressionSegment expression = new CommonExpressionSegment(0, 6, "(foo).*", Collections.singletonList(column));
191         expression.setRowExpansion(true);
192         ExpressionProjectionSegment projection = new ExpressionProjectionSegment(0, 6, "(foo).*", expression);
193         ProjectionsSegment projections = new ProjectionsSegment(0, 6);
194         projections.getProjections().add(projection);
195         SQLStatementBinderContext context = mock(SQLStatementBinderContext.class, RETURNS_DEEP_STUBS);
196         when(context.getSqlStatement().getDatabaseType()).thenReturn(TypedSPILoader.getService(DatabaseType.class, "FIXTURE"));
197         Multimap<CaseInsensitiveString, TableSegmentBinderContext> tables = LinkedHashMultimap.create();
198         tables.put(CaseInsensitiveString.of("foo"), mock(TableSegmentBinderContext.class));
199         Multimap<CaseInsensitiveString, TableSegmentBinderContext> outer = LinkedHashMultimap.create();
200         CommonExpressionSegment boundExpression = new CommonExpressionSegment(0, 6, "(foo).*", Collections.singletonList(column));
201         boundExpression.setRowExpansion(true);
202         try (MockedStatic<ExpressionSegmentBinder> expressions = mockStatic(ExpressionSegmentBinder.class)) {
203             expressions.when(() -> ExpressionSegmentBinder.bind(expression, SegmentType.PROJECTION, context, tables, outer)).thenReturn(boundExpression);
204             ProjectionsSegment actual = ProjectionsSegmentBinder.bind(projections, context, null, tables, outer);
205             assertThat(((ExpressionProjectionSegment) actual.getProjections().iterator().next()).getExpr(), sameInstance(boundExpression));
206         }
207     }

209     @ParameterizedTest(name = "{0}")
210     @MethodSource("rowExpansionErrorArguments")
211     void assertBindRowExpansionPreservesBindingFailure(final String caseName, final ProjectionSegment projection, final boolean supportWholeRow, final boolean visibleAlias) {
212         ProjectionsSegment projections = new ProjectionsSegment(0, 20);
213         projections.getProjections().add(projection);
214         SQLStatementBinderContext context = mock(SQLStatementBinderContext.class, RETURNS_DEEP_STUBS);
215         when(context.getSqlStatement().getDatabaseType()).thenReturn(TypedSPILoader.getService(DatabaseType.class, "FIXTURE"));
216         Multimap<CaseInsensitiveString, TableSegmentBinderContext> tables = LinkedHashMultimap.create();
217         if (visibleAlias) {
218             tables.put(CaseInsensitiveString.of("foo"), mock(TableSegmentBinderContext.class));
219         }
220         Multimap<CaseInsensitiveString, TableSegmentBinderContext> outer = LinkedHashMultimap.create();
221         DialectDatabaseMetaData metadata = mock(DialectDatabaseMetaData.class, RETURNS_DEEP_STUBS);
222         when(metadata.getSQLOption().isSupportWholeRowProjection()).thenReturn(supportWholeRow);
223         ColumnNotFoundException failure = new ColumnNotFoundException("foo", "field list");
224         try (
225                 MockedStatic<ExpressionSegmentBinder> expressions = mockStatic(ExpressionSegmentBinder.class);
226                 MockedStatic<ColumnProjectionSegmentBinder> columns = mockStatic(ColumnProjectionSegmentBinder.class);
227                 MockedConstruction<DatabaseTypeRegistry> ignored = mockConstruction(DatabaseTypeRegistry.class,
228                         (registry, construction) -> when(registry.getDialectDatabaseMetaData()).thenReturn(metadata))) {
229             expressions.when(() -> ExpressionSegmentBinder.bind(any(), eq(SegmentType.PROJECTION), same(context), any(), same(outer))).thenThrow(failure);
230             columns.when(() -> ColumnProjectionSegmentBinder.bind(any(), same(context), any(), same(outer))).thenThrow(failure);
231             ColumnNotFoundException actual = assertThrows(ColumnNotFoundException.class, () -> ProjectionsSegmentBinder.bind(projections, context, null, tables, outer));
232             assertThat(actual, sameInstance(failure));
233         }
234     }

236     private static Stream<Arguments> rowExpansionErrorArguments() {
237         ColumnSegment column = new ColumnSegment(1, 3, new IdentifierValue("foo"));
238         ColumnSegment qualified = new ColumnSegment(11, 13, new IdentifierValue("foo"));
239         qualified.setOwner(new OwnerSegment(1, 9, new IdentifierValue("foo_table")));
240         FunctionSegment function = new FunctionSegment(0, 13, "LENGTH", "LENGTH(foo)");
241         function.getParameters().add(column);
242         RowExpression row = new RowExpression(1, 8, "ROW(foo)");
243         row.getItems().add(column);
244         return Stream.of(
245                 Arguments.of("bare column", new ColumnProjectionSegment(column), false, false),
246                 Arguments.of("function expression", new ExpressionProjectionSegment(0, 13, function.getText(), function), true, true),
247                 Arguments.of("parenthesized column without star", createRowProjection(column, false), true, true),
248                 Arguments.of("dialect without whole row projection", createRowProjection(column, true), false, true),
249                 Arguments.of("qualified composite column", createRowProjection(qualified, true), true, true),
250                 Arguments.of("row constructor with missing column", createRowProjection(row, true), true, true),
251                 Arguments.of("invisible alias", createRowProjection(column, true), true, false));
252     }

254     private static ExpressionProjectionSegment createRowProjection(final ExpressionSegment child, final boolean rowExpansion) {
255         CommonExpressionSegment expression = new CommonExpressionSegment(0, 20, "foo_expression", Collections.singletonList(child));
256         expression.setRowExpansion(rowExpansion);
257         return new ExpressionProjectionSegment(0, 20, expression.getText(), expression);
258     }
