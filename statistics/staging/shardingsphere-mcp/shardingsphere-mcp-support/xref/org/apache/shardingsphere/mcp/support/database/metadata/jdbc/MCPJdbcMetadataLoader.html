57      /**
58       * Load database metadata.
59       *
60       * @param databaseName database name
61       * @param runtimeDatabaseConfig runtime database configuration
62       * @param databaseProfile runtime database profile
63       * @return database metadata snapshot
64       * @throws RuntimeDatabaseConnectionException when metadata loading fails
65       */
66      public MCPMetadataSnapshot load(final String databaseName, final RuntimeDatabaseConfiguration runtimeDatabaseConfig, final RuntimeDatabaseProfile databaseProfile) {
67          try (Connection connection = runtimeDatabaseConfig.openConnection(databaseName)) {
68              return loadMetadata(databaseName, databaseProfile, connection, connection.getMetaData());
69          } catch (final SQLException ex) {
70              throw RuntimeDatabaseConnectionException.connectionFailed(databaseName, databaseProfile.getDatabaseType(), ex);
71          }
72      }
73      
74      /**
75       * Load columns for one relation.
76       *
77       * @param databaseName database name
78       * @param runtimeDatabaseConfig runtime database configuration
79       * @param databaseProfile runtime database profile
80       * @param schemaName schema name
81       * @param relationName table or view name
82       * @return column metadata
83       * @throws RuntimeDatabaseConnectionException when metadata loading fails
84       */
85      public List<MCPColumnMetadata> loadColumns(final String databaseName, final RuntimeDatabaseConfiguration runtimeDatabaseConfig,
86                                                 final RuntimeDatabaseProfile databaseProfile, final String schemaName, final String relationName) {
87          try (Connection connection = runtimeDatabaseConfig.openConnection(databaseName)) {
88              DatabaseMetaData databaseMetaData = connection.getMetaData();
89              return loadColumnMetadata(connection, databaseMetaData, databaseProfile, schemaName, escapePattern(databaseMetaData, relationName));
90          } catch (final SQLException ex) {
91              throw RuntimeDatabaseConnectionException.connectionFailed(databaseName, databaseProfile.getDatabaseType(), ex);
92          }
93      }

95      /**
96       * Load all columns in one schema.
97       *
98       * @param databaseName database name
99       * @param runtimeDatabaseConfig runtime database configuration
100      * @param databaseProfile runtime database profile
101      * @param schemaName schema name
102      * @return column metadata
103      * @throws RuntimeDatabaseConnectionException when metadata loading fails
104      */
105     public List<MCPColumnMetadata> loadSchemaColumns(final String databaseName, final RuntimeDatabaseConfiguration runtimeDatabaseConfig,
106                                                      final RuntimeDatabaseProfile databaseProfile, final String schemaName) {
107         try (Connection connection = runtimeDatabaseConfig.openConnection(databaseName)) {
108             return loadColumnMetadata(connection, connection.getMetaData(), databaseProfile, schemaName, "%");
109         } catch (final SQLException ex) {
110             throw RuntimeDatabaseConnectionException.connectionFailed(databaseName, databaseProfile.getDatabaseType(), ex);
111         }
112     }

115      * Load indexes for one table.
116      *
117      * @param databaseName database name
118      * @param runtimeDatabaseConfig runtime database configuration
119      * @param databaseProfile runtime database profile
120      * @param schemaName schema name
121      * @param tableName table name
122      * @return index metadata

126                                                  final RuntimeDatabaseProfile databaseProfile, final String schemaName, final String tableName) {
127         try (Connection connection = runtimeDatabaseConfig.openConnection(databaseName)) {
128             return loadIndexMetadata(connection, connection.getMetaData(), databaseProfile, schemaName, tableName);

134     private MCPMetadataSnapshot loadMetadata(final String databaseName, final RuntimeDatabaseProfile databaseProfile,
135                                              final Connection connection, final DatabaseMetaData databaseMetaData) throws SQLException {
136         DatabaseType protocolType = TypedSPILoader.getService(DatabaseType.class, databaseProfile.getDatabaseType());
137         MCPDatabaseDialect databaseDialect = MCPDatabaseDialect.of(databaseProfile.getDatabaseType());
138         DialectSchemaSemantics defaultSchemaSemantics = databaseDialect.getDefaultSchemaSemantics();
139         DatabaseMetadataAccumulator accumulator = new DatabaseMetadataAccumulator(databaseName, protocolType);
140         loadTables(databaseName, defaultSchemaSemantics, databaseDialect, accumulator, databaseMetaData);
141         loadViews(databaseName, defaultSchemaSemantics, databaseDialect, accumulator, databaseMetaData);
142         loadSequences(databaseName, defaultSchemaSemantics, databaseDialect, accumulator, connection);
143         return accumulator.build();
144     }
145     
146     private void loadTables(final String databaseName, final DialectSchemaSemantics defaultSchemaSemantics, final MCPDatabaseDialect databaseDialect,
147                             final DatabaseMetadataAccumulator accumulator, final DatabaseMetaData databaseMetaData) throws SQLException {
148         try (ResultSet tables = databaseMetaData.getTables(null, null, "%", new String[]{"TABLE"})) {
149             while (tables.next()) {
150                 String schemaName = Objects.toString(tables.getString("TABLE_SCHEM"), "").trim();
151                 String catalogName = Objects.toString(tables.getString("TABLE_CAT"), "").trim();
152                 if (databaseDialect.isSystemSchema(schemaName, catalogName, defaultSchemaSemantics)) {
153                     continue;
154                 }
155                 String tableName = Objects.toString(tables.getString("TABLE_NAME"), "").trim();
156                 if (tableName.isEmpty()) {
157                     continue;
158                 }
159                 accumulator.getSchemaAccumulator(normalizeSchemaName(databaseName, defaultSchemaSemantics, schemaName)).addRelation(tableName, TableType.TABLE);
160             }
161         }
162     }
163     
164     private void loadViews(final String databaseName, final DialectSchemaSemantics defaultSchemaSemantics, final MCPDatabaseDialect databaseDialect,
165                            final DatabaseMetadataAccumulator accumulator, final DatabaseMetaData databaseMetaData) throws SQLException {
166         try (ResultSet views = databaseMetaData.getTables(null, null, "%", new String[]{"VIEW"})) {
167             while (views.next()) {
168                 String schemaName = Objects.toString(views.getString("TABLE_SCHEM"), "").trim();
169                 String catalogName = Objects.toString(views.getString("TABLE_CAT"), "").trim();
170                 if (databaseDialect.isSystemSchema(schemaName, catalogName, defaultSchemaSemantics)) {
171                     continue;
172                 }
173                 String viewName = Objects.toString(views.getString("TABLE_NAME"), "").trim();
174                 if (viewName.isEmpty()) {
175                     continue;
176                 }
177                 accumulator.getSchemaAccumulator(normalizeSchemaName(databaseName, defaultSchemaSemantics, schemaName)).addRelation(viewName, TableType.VIEW);
178             }
179         }
180     }
181     
182     private void loadSequences(final String databaseName, final DialectSchemaSemantics defaultSchemaSemantics, final MCPDatabaseDialect databaseDialect,
183                                final DatabaseMetadataAccumulator accumulator, final Connection connection) throws SQLException {
184         Optional<String> sequenceQuery = databaseDialect.getSequenceQuery();
185         if (sequenceQuery.isEmpty()) {
186             return;
187         }
188         try (Statement statement = connection.createStatement(); ResultSet sequences = statement.executeQuery(sequenceQuery.get())) {
189             while (sequences.next()) {
190                 String schemaName = Objects.toString(sequences.getString("SEQUENCE_SCHEMA"), "").trim();
191                 if (databaseDialect.isSystemSchema(schemaName)) {
192                     continue;
193                 }
194                 String sequenceName = Objects.toString(sequences.getString("SEQUENCE_NAME"), "");
195                 if (!sequenceName.isEmpty()) {
196                     accumulator.getSchemaAccumulator(normalizeSchemaName(databaseName, defaultSchemaSemantics, schemaName)).addSequence(sequenceName);
197                 }
198             }
199         }
200     }

202     private List<MCPColumnMetadata> loadColumnMetadata(final Connection connection, final DatabaseMetaData databaseMetaData, final RuntimeDatabaseProfile databaseProfile,
203                                                        final String schemaName, final String relationNamePattern) throws SQLException {
204         DialectSchemaSemantics schemaSemantics = MCPDatabaseDialect.of(databaseProfile.getDatabaseType()).getDefaultSchemaSemantics();
205         String catalogName = DialectSchemaSemantics.DATABASE_AS_SCHEMA == schemaSemantics ? resolveCatalogName(connection, schemaName) : null;
206         String schemaNamePattern = DialectSchemaSemantics.NATIVE_SCHEMA == schemaSemantics ? escapePattern(databaseMetaData, schemaName) : null;
207         List<MCPColumnMetadata> result = queryColumnMetadata(databaseMetaData, catalogName, schemaNamePattern, relationNamePattern);
208         if (result.isEmpty() && null != catalogName) {
209             result = queryColumnMetadata(databaseMetaData, null, schemaNamePattern, relationNamePattern);
210         }
211         result.sort(Comparator.comparing(MCPColumnMetadata::getRelationName)
212                 .thenComparingInt(MCPColumnMetadata::getOrdinalPosition).thenComparing(MCPColumnMetadata::getName));
213         return result;
214     }
215     
216     private List<MCPColumnMetadata> queryColumnMetadata(final DatabaseMetaData databaseMetaData, final String catalogName,
217                                                         final String schemaNamePattern, final String relationNamePattern) throws SQLException {
218         List<MCPColumnMetadata> result = new LinkedList<>();
219         try (ResultSet columns = databaseMetaData.getColumns(catalogName, schemaNamePattern, relationNamePattern, "%")) {
220             while (columns.next()) {
221                 String columnName = Objects.toString(columns.getString("COLUMN_NAME"), "").trim();
222                 if (!columnName.isEmpty()) {
223                     result.add(new MCPColumnMetadata(
224                             Objects.toString(columns.getString("TABLE_NAME"), "").trim(), columnName, columns.getInt("ORDINAL_POSITION"),
225                             columns.getInt("DATA_TYPE"), Objects.toString(columns.getString("TYPE_NAME"), "").trim(),
226                             Nullability.fromJdbcValue(columns.getInt("NULLABLE"))));
227                 }
228             }
229         }
230         return result;
231     }

233     private List<ShardingSphereIndex> loadIndexMetadata(final Connection connection, final DatabaseMetaData databaseMetaData, final RuntimeDatabaseProfile databaseProfile,
234                                                         final String schemaName, final String tableName) throws SQLException {
235         Map<String, IndexMetadataAccumulator> result = new LinkedHashMap<>(16, 1F);
236         DialectSchemaSemantics schemaSemantics = MCPDatabaseDialect.of(databaseProfile.getDatabaseType()).getDefaultSchemaSemantics();
237         String catalogName = DialectSchemaSemantics.DATABASE_AS_SCHEMA == schemaSemantics ? resolveCatalogName(connection, schemaName) : null;
238         String jdbcSchemaName = DialectSchemaSemantics.NATIVE_SCHEMA == schemaSemantics ? trimToNull(schemaName) : null;
239         try (ResultSet indexes = databaseMetaData.getIndexInfo(catalogName, jdbcSchemaName, tableName, false, false)) {
240             while (indexes.next()) {
241                 String indexName = Objects.toString(indexes.getString("INDEX_NAME"), "").trim();
242                 if (indexName.isEmpty() || DatabaseMetaData.tableIndexStatistic == indexes.getShort("TYPE")) {
243                     continue;
244                 }
245                 boolean unique = !indexes.getBoolean("NON_UNIQUE");
246                 IndexMetadataAccumulator accumulator = result.computeIfAbsent(indexName, ignored -> new IndexMetadataAccumulator(unique));
247                 accumulator.addColumn(indexes.getInt("ORDINAL_POSITION"), Objects.toString(indexes.getString("COLUMN_NAME"), "").trim());
248             }
249         } catch (final SQLFeatureNotSupportedException ignored) {
250             return Collections.emptyList();
251         }
252         return result.entrySet().stream().map(each -> each.getValue().build(each.getKey())).toList();
253     }
254     
255     private String resolveCatalogName(final Connection connection, final String schemaName) throws SQLException {
256         String result = trimToNull(connection.getCatalog());
257         return null == result ? trimToNull(schemaName) : result;
258     }
259     
260     private String escapePattern(final DatabaseMetaData databaseMetaData, final String value) throws SQLException {
261         String result = Objects.toString(value, "").trim();
262         if (result.isEmpty()) {
263             return null;
264         }
265         String escape = Objects.toString(databaseMetaData.getSearchStringEscape(), "");
266         if (escape.isEmpty()) {
267             return result;
268         }
269         return result.replace(escape, escape + escape).replace("%", escape + "%").replace("_", escape + "_");
270     }
271     
272     private String trimToNull(final String value) {
273         String result = Objects.toString(value, "").trim();
274         return result.isEmpty() ? null : result;
275     }
276     
277     private String normalizeSchemaName(final String databaseName, final DialectSchemaSemantics defaultSchemaSemantics, final String schemaName) {
278         String result = Objects.toString(schemaName, "").trim();
279         if (!result.isEmpty()) {
280             return result;
281         }
282         return DialectSchemaSemantics.DATABASE_AS_SCHEMA == defaultSchemaSemantics ? databaseName : result;
283     }

285     @RequiredArgsConstructor(access = AccessLevel.PRIVATE)
286     private static final class DatabaseMetadataAccumulator {
287         
288         private final String database;
289         
290         private final DatabaseType protocolType;
291         
292         private final Map<String, SchemaMetadataAccumulator> schemaAccumulators = new LinkedHashMap<>(16, 1F);
293         
294         private SchemaMetadataAccumulator getSchemaAccumulator(final String schema) {
295             SchemaMetadataAccumulator result = schemaAccumulators.get(schema);
296             if (null == result) {
297                 result = new SchemaMetadataAccumulator(database, schema, protocolType);
298                 schemaAccumulators.put(schema, result);
299             }
300             return result;
301         }
302         
303         private MCPMetadataSnapshot build() {
304             List<ShardingSphereSchema> schemas = new LinkedList<>();
305             Map<String, Collection<MCPSequenceMetadata>> sequences = new LinkedHashMap<>(schemaAccumulators.size(), 1F);
306             for (SchemaMetadataAccumulator each : schemaAccumulators.values()) {
307                 schemas.add(each.buildSchema());
308                 sequences.put(each.schema, new LinkedList<>(each.sequences.values()));
309             }
310             return new MCPMetadataSnapshot(schemas, sequences);
311         }
312     }
313     
314     @RequiredArgsConstructor(access = AccessLevel.PRIVATE)
315     private static final class SchemaMetadataAccumulator {
316         
317         private final String database;
318         
319         private final String schema;
320         
321         private final DatabaseType protocolType;
322         
323         private final Map<String, ShardingSphereTable> relations = new LinkedHashMap<>(16, 1F);
324         
325         private final Map<String, MCPSequenceMetadata> sequences = new LinkedHashMap<>(16, 1F);
326         
327         private void addRelation(final String name, final TableType type) {
328             relations.putIfAbsent(name, new ShardingSphereTable(name, Collections.emptyList(), Collections.emptyList(), Collections.emptyList(), type));
329         }
330         
331         private void addSequence(final String sequence) {
332             sequences.putIfAbsent(sequence, new MCPSequenceMetadata(database, schema, sequence));
333         }
334         
335         private ShardingSphereSchema buildSchema() {
336             return new ShardingSphereSchema(schema, protocolType, relations.values(), Collections.emptyList());
337         }
338     }
339     
340     @RequiredArgsConstructor(access = AccessLevel.PRIVATE)
341     private static final class IndexMetadataAccumulator {
342         
343         private final boolean unique;
344         
345         private final Map<Integer, String> columns = new LinkedHashMap<>(4, 1F);
346         
347         private void addColumn(final int ordinalPosition, final String columnName) {
348             if (!columnName.isEmpty()) {
349                 columns.putIfAbsent(ordinalPosition, columnName);
350             }
351         }
352         
353         private ShardingSphereIndex build(final String name) {
354             return new ShardingSphereIndex(name, columns.entrySet().stream().sorted(Entry.comparingByKey()).map(Entry::getValue).toList(), unique);
355         }
356     }
