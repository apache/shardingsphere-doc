29  import org.apache.shardingsphere.infra.util.props.PropertiesBuilder;

33  import java.util.Collection;

35  import java.util.Optional;
36  import java.util.Properties;

38  /**
39   * MD5 assisted encrypt algorithm.
40   */
41  public final class MD5AssistedEncryptAlgorithm implements EncryptAlgorithm {
42      
43      private static final String SALT_KEY = "salt";
44      
45      private static final String MD5_ENCODER = "md5-encoder";
46      
47      private static final String DEFAULT_MD5_ENCODER = "HEX";
48      
49      private static final String BASE64 = "BASE64";
50      
51      private static final Collection<String> SUPPORTED_ENCODERS = Arrays.asList(BASE64, DEFAULT_MD5_ENCODER);
52      
53      @Getter
54      private final EncryptAlgorithmMetaData metaData = new EncryptAlgorithmMetaData(false, true, false, byte[].class);
55      
56      private Properties props;
57      
58      private MessageDigestAlgorithm digestAlgorithm;
59      
60      private String md5Encoder;
61      
62      @Override
63      public void init(final Properties props) {
64          this.props = props;
65          digestAlgorithm = TypedSPILoader.getService(MessageDigestAlgorithm.class, getType(), props);
66          md5Encoder = getEncoder(props);
67      }
68      
69      private String getEncoder(final Properties props) {
70          String result = props.getProperty(MD5_ENCODER, DEFAULT_MD5_ENCODER).trim().toUpperCase(Locale.ENGLISH);
71          ShardingSpherePreconditions.checkContains(SUPPORTED_ENCODERS, result, () -> new AlgorithmInitializationException(this, String.format("Encoder must be `%s`", SUPPORTED_ENCODERS), MD5_ENCODER));
72          return result;
73      }
74      
75      @Override
76      public Optional<String> getEncoder() {
77          return Optional.of(md5Encoder);
78      }
79      
80      @Override
81      public byte[] encrypt(final Object plainValue, final AlgorithmSQLContext algorithmSQLContext) {
82          return digestAlgorithm.digest(plainValue);
83      }
84      
85      @Override
86      public Object decrypt(final Object cipherValue, final AlgorithmSQLContext algorithmSQLContext) {
87          throw new UnsupportedOperationException(String.format("Algorithm `%s` is unsupported to decrypt", getType()));
88      }
89      
90      @Override
91      public AlgorithmConfiguration toConfiguration() {
92          Properties result = PropertiesBuilder.build(new Property(SALT_KEY, props.getProperty(SALT_KEY, "")));
93          result.setProperty(MD5_ENCODER, md5Encoder);
94          return new AlgorithmConfiguration(getType(), result);
95      }
96      
97      @Override
98      public String getType() {
99          return "MD5";
100     }
101 }
