150         try (MockedConstruction<Bootstrap> ignored = mockConstruction(Bootstrap.class, (mock, context) -> {

326     @Test
327     void assertReconnectAfterTransientFailure() throws Exception {
328         client = createClientMock();
329         prepareClientChannel();
330         setServerVersion("5.5.0");
331         doReturn(true).when(client).execute(anyString());
332         when(channel.writeAndFlush(any(MySQLComRegisterSlaveCommandPacket.class))).thenAnswer(invocation -> {
333             Promise<Object> callback = new DefaultPromise<>(eventLoopGroup.next());
334             callback.setSuccess(new MySQLOKPacket(0));
335             setResponseCallback(callback);
336             return null;
337         });
338         client.subscribe("binlog-000004", 16L);
339         ArgumentCaptor<ChannelHandler> handlerCaptor = ArgumentCaptor.forClass(ChannelHandler.class);
340         verify(pipeline, times(2)).addLast(handlerCaptor.capture());
341         ChannelInboundHandlerAdapter handler = handlerCaptor.getAllValues().stream().filter(each -> "MySQLBinlogEventHandler".equals(each.getClass().getSimpleName()))
342                 .map(each -> (ChannelInboundHandlerAdapter) each).findFirst().orElseThrow(IllegalStateException::new);
343         ChannelHandlerContext context = mock(ChannelHandlerContext.class);
344         handler.channelRead(context, new PlaceholderBinlogEvent("binlog-000005", 32L, 1L));
345         doThrow(new PipelineInternalException("connect failed")).doNothing().when(client).connect();
346         doNothing().when(client).subscribe("binlog-000005", 32L);
347         CompletableFuture<Void> reconnectFuture = new CompletableFuture<>();
348         ArgumentCaptor<Runnable> reconnectCaptor = ArgumentCaptor.forClass(Runnable.class);
349         when(CompletableFuture.runAsync(reconnectCaptor.capture())).thenReturn(reconnectFuture);
350         handler.channelInactive(context);
351         handler.channelInactive(context);
352         assertThat(reconnectCaptor.getAllValues().size(), is(1));
353         reconnectCaptor.getValue().run();
354         verify(client, times(2)).connect();
355         verify(channel).close();
356         verify(client).subscribe("binlog-000005", 32L);
357         reconnectFuture.complete(null);
358         handler.channelInactive(context);
359         assertThat(reconnectCaptor.getAllValues().size(), is(2));
360     }

406     @Test
407     void assertMySQLCommandResponseHandlerBranches(final LogCaptureAssertion logCaptureAssertion) throws Exception {
408         final AtomicReference<ChannelInitializer<SocketChannel>> initializer = new AtomicReference<>();
409         SocketChannel socketChannel = mock(SocketChannel.class, RETURNS_DEEP_STUBS);
410         when(socketChannel.pipeline()).thenReturn(pipeline);
411         when(channelFuture.channel()).thenReturn(channel);
412         try (MockedConstruction<Bootstrap> ignored = mockConstruction(Bootstrap.class, (mock, context) -> {
413             when(mock.group(any())).thenReturn(mock);
414             when(mock.channel(any())).thenReturn(mock);
415             when(mock.option(any(), any())).thenReturn(mock);
416             when(mock.handler(any())).thenAnswer(invocation -> {
417                 initializer.set(invocation.getArgument(0));
418                 return mock;
419             });
420             when(mock.connect(anyString(), anyInt())).thenAnswer(mockChannelResponse(new MySQLServerVersion("5.5.0-log")));
421         })) {
422             client.connect();
423         }
424         ChannelHandlerContext context = mock(ChannelHandlerContext.class);
425         when(context.executor()).thenReturn(eventLoopGroup.next());
426         when(context.channel()).thenReturn(socketChannel);
427         when(context.pipeline()).thenReturn(pipeline);
428         initializer.get().channelRegistered(context);
429         ArgumentCaptor<ChannelHandler> captor = ArgumentCaptor.forClass(ChannelHandler.class);
430         verify(pipeline, times(7)).addLast(captor.capture());
431         ChannelInboundHandlerAdapter handler = captor.getAllValues().stream().filter(each -> "MySQLCommandResponseHandler".equals(each.getClass().getSimpleName()))
432                 .map(ChannelInboundHandlerAdapter.class::cast).findFirst().orElseThrow(IllegalStateException::new);
433         Promise<Object> callback = new DefaultPromise<>(eventLoopGroup.next());
434         setResponseCallback(callback);
435         handler.channelRead(mock(ChannelHandlerContext.class), new Object());
436         assertTrue(callback.isSuccess());
437         callback = new DefaultPromise<>(eventLoopGroup.next());
438         setResponseCallback(callback);
439         RuntimeException expectedException = new RuntimeException("ex");
440         handler.exceptionCaught(mock(ChannelHandlerContext.class), expectedException);
441         assertTrue(callback.isDone());
442         setResponseCallback(null);
443         handler.channelRead(mock(ChannelHandlerContext.class), new Object());
444         handler.exceptionCaught(mock(ChannelHandlerContext.class), new RuntimeException("ex"));
445         client.closeChannel(true);
446         logCaptureAssertion.assertErrorLog(actualException -> assertThat(((ThrowableProxy) actualException).getThrowable(), sameInstance(expectedException)));
447     }
