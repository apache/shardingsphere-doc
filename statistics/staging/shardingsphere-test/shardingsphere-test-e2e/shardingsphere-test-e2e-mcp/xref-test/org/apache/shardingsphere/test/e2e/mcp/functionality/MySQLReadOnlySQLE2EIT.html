97      @ParameterizedTest(name = "{0}")
98      @MethodSource("allTransportCases")
99      void assertAiNativeDeterministicInteractionLoop(final String name, final RuntimeTransport transport) throws IOException, InterruptedException {
100         useTransport(transport);
101         try (MCPInteractionClient interactionClient = createOpenedInteractionClient()) {
102             assertAiNativeGuidance(interactionClient.readResource("shardingsphere://guidance"));
103             assertAiNativeDiscovery(interactionClient);
104             Map<String, Object> searchMetadataPayload = interactionClient.call("database_gateway_search_metadata",
105                     Map.of("database", LOGICAL_DATABASE_NAME, "schema", LOGICAL_DATABASE_NAME, "query", "orders", "object_types", List.of("table")));
106             Map<String, Object> tableHit = MCPPayloadAssertions.findItem(searchMetadataPayload, "name", "orders");
107             String tableResourceUri = String.valueOf(getObjectOrEmpty(tableHit.get("resource")).get("uri"));
108             assertThat(tableResourceUri, is("shardingsphere://databases/logic_db/schemas/logic_db/tables/orders"));
109             assertFalse(getRequiredObjectList(tableHit.get("next_resources")).isEmpty());
110             MCPPayloadAssertions.assertSingleItemValue(interactionClient.readResource(tableResourceUri), "table", "orders");
111             assertAiNativeSqlPreview(interactionClient);
112             assertAiNativeSqlResult(interactionClient);
113         }
114     }

116     private void assertAiNativeGuidance(final Map<String, Object> guidance) {
117         assertTrue(guidance.containsKey("discovery"));
118         assertTrue(guidance.containsKey("model_contract"));
119         assertTrue(guidance.containsKey("next_action_contract"));
120         assertTrue(guidance.containsKey("common_flows"));
121         assertTrue(guidance.containsKey("security_hints"));
122         assertFalse(guidance.containsKey("model_first_summary"));
123         assertFalse(guidance.containsKey("surface_summary"));
124         assertFalse(guidance.containsKey("fingerprints"));
125         assertFalse(((List<?>) guidance.get("common_flows")).isEmpty());
126         Map<String, Object> discovery = getObjectOrEmpty(guidance.get("discovery"));
127         assertThat(getObjectOrEmpty(discovery.get("official_discovery_methods")).get("tools"), is("tools/list"));
128         assertThat(discovery.get("argument_completion_method"), is("completion/complete"));
129         Map<String, Object> modelContract = getObjectOrEmpty(guidance.get("model_contract"));
130         assertTrue(String.valueOf(modelContract.get("preflight_rule")).contains("database_gateway_validate_runtime_database"));
131         assertTrue(String.valueOf(getObjectOrEmpty(modelContract.get("sql_tool_selection")).get("read_only")).contains("database_gateway_execute_query"));
132         assertThat(modelContract.get("recovery_rule"), is("When a call fails, follow top-level next_actions before inventing a new call."));
133     }

135     private void assertAiNativeDiscovery(final MCPInteractionClient interactionClient) throws IOException, InterruptedException {
136         Map<String, Object> runtimeStatus = interactionClient.readResource("shardingsphere://runtime");
137         assertThat(String.valueOf(runtimeStatus.get("status")), is("available"));
138         assertThat(String.valueOf(runtimeStatus.get("configured_database_count")), is("1"));
139         assertTrue(MCPInteractionPayloads.getRequiredObjectList(interactionClient.listResources(), "resources").stream()
140                 .anyMatch(each -> "shardingsphere://runtime".equals(each.get("uri"))));
141         assertTrue(MCPInteractionPayloads.getRequiredObjectList(interactionClient.listResourceTemplates(), "resourceTemplates").stream()
142                 .anyMatch(each -> "shardingsphere://databases/{database}/schemas/{schema}/tables/{table}".equals(each.get("uriTemplate"))));
143         assertTrue(interactionClient.listTools().stream()
144                 .anyMatch(each -> "database_gateway_execute_update".equals(each.get("name")) && getObjectOrEmpty(each.get("outputSchema")).containsKey("properties")));
145         Map<String, Object> promptPayload = interactionClient.getPrompt("inspect_metadata",
146                 Map.of("database", LOGICAL_DATABASE_NAME, "schema", LOGICAL_DATABASE_NAME, "query", "orders"));
147         assertTrue(String.valueOf(promptPayload).contains("Stop conditions"));
148         Map<String, Object> completionPayload = interactionClient.complete(Map.of("type", "ref/prompt", "name", "inspect_metadata"),
149                 "schema", "log", Map.of("database", LOGICAL_DATABASE_NAME));
150         assertTrue(((List<?>) getObjectOrEmpty(completionPayload.get("completion")).get("values")).contains(LOGICAL_DATABASE_NAME));
151     }

153     private void assertAiNativeSqlPreview(final MCPInteractionClient interactionClient) throws IOException, InterruptedException {
154         Map<String, Object> actual = interactionClient.call("database_gateway_execute_update",
155                 Map.of("database", LOGICAL_DATABASE_NAME, "schema", LOGICAL_DATABASE_NAME, "sql", "UPDATE orders SET status = status WHERE order_id = -1", "execution_mode", "preview"));
156         assertThat(String.valueOf(actual.get("response_mode")), is("preview"));
157         assertThat(String.valueOf(actual.get("result_kind")), is("preview"));
158         assertThat(String.valueOf(actual.get("preview_semantics")), is("classification_only"));
159         assertFalse((Boolean) actual.get("would_execute"));
160         List<Map<String, Object>> nextActions = getRequiredObjectList(actual.get("next_actions"));
161         assertThat(nextActions.stream().map(each -> String.valueOf(each.get("type"))).toList(), is(List.of("ask_user", "tool_call")));
162         Map<String, Object> askUserAction = nextActions.getFirst();
163         assertThat(askUserAction.get("order"), is(1));
164         assertThat(askUserAction.get("required_inputs"), is(List.of("execution_approved")));
165         Map<String, Object> toolCallAction = nextActions.get(1);
166         assertThat(toolCallAction.get("order"), is(2));
167         assertThat(toolCallAction.get("tool_name"), is("database_gateway_execute_update"));
168         assertThat(toolCallAction.get("depends_on"), is(List.of(1)));
169         assertThat(getObjectOrEmpty(toolCallAction.get("arguments")).get("execution_mode"), is("execute"));
170     }

173         Map<String, Object> actual = interactionClient.call("database_gateway_execute_query",
174                 Map.of("database", LOGICAL_DATABASE_NAME, "schema", LOGICAL_DATABASE_NAME, "sql", "SELECT order_id, status FROM orders ORDER BY order_id", "max_rows", 1));
175         assertThat(String.valueOf(actual.get("result_kind")), is("result_set"));
176         assertThat(String.valueOf(actual.get("row_object_status")), is("available"));
177         assertThat(((List<?>) actual.get("row_objects")).size(), is(1));
178         assertThat(String.valueOf(actual.get("truncated")), is("true"));
179         assertThat(String.valueOf(getRequiredObjectList(actual.get("next_actions")).getFirst().get("type")), is("ask_user"));
180     }
