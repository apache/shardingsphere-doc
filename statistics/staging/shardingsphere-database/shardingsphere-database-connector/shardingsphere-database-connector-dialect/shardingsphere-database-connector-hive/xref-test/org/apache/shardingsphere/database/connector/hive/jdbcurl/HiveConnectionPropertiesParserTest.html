59      private final DatabaseType databaseType = TypedSPILoader.getService(DatabaseType.class, "Hive");

61      private final ConnectionPropertiesParser parser = DatabaseTypedSPILoader.getService(ConnectionPropertiesParser.class, databaseType);

63      @Test
64      void assertLoadWithoutHiveJdbcDriver() throws IOException, ReflectiveOperationException {
65          String[] classPathEntries = System.getProperty("surefire.test.class.path", System.getProperty("java.class.path")).split(File.pathSeparator);
66          URL[] classPathUrls = new URL[classPathEntries.length];
67          for (int i = 0; i < classPathEntries.length; i++) {
68              classPathUrls[i] = new File(classPathEntries[i]).toURI().toURL();
69          }
70          ClassLoader originalClassLoader = Thread.currentThread().getContextClassLoader();
71          try (URLClassLoader classLoader = new URLClassLoader(classPathUrls, ClassLoader.getSystemClassLoader().getParent()) {
72              
73              @Override
74              protected Class<?> loadClass(final String name, final boolean resolve) throws ClassNotFoundException {
75                  if (name.startsWith("org.apache.hive.jdbc.")) {
76                      throw new ClassNotFoundException(name);
77                  }
78                  return super.loadClass(name, resolve);
79              }
80          }) {
81              Thread.currentThread().setContextClassLoader(classLoader);
82              assertThrows(ClassNotFoundException.class, () -> classLoader.loadClass("org.apache.hive.jdbc.Utils"));
83              Class<?> databaseTypeClass = classLoader.loadClass(DatabaseType.class.getName());
84              Class<?> parserClass = classLoader.loadClass(ConnectionPropertiesParser.class.getName());
85              Method getServiceMethod = classLoader.loadClass(TypedSPILoader.class.getName()).getMethod("getService", Class.class, Object.class);
86              Object hiveDatabaseType = getServiceMethod.invoke(null, databaseTypeClass, "Hive");
87              Object actual = getServiceMethod.invoke(null, parserClass, hiveDatabaseType);
88              assertThat(actual.getClass().getName(), is(HiveConnectionPropertiesParser.class.getName()));
89              assertThat(actual.getClass().getClassLoader(), sameInstance(classLoader));
90          } finally {
91              Thread.currentThread().setContextClassLoader(originalClassLoader);
92          }
93      }

95      @ParameterizedTest(name = "{0}")
96      @MethodSource("parseSuccessArguments")
97      void assertParseWithValidUrl(final String name, final String url, final String expectedHostname, final int expectedPort, final String expectedCatalog, final Properties expectedQueryProps) {
98          ConnectionProperties actual = parser.parse(url, null, null);
99          assertThat(actual.getHostname(), is(expectedHostname));
100         assertThat(actual.getPort(), is(expectedPort));
101         assertThat(actual.getCatalog(), is(expectedCatalog));
102         assertNull(actual.getSchema());
103         assertThat(actual.getQueryProperties(), is(expectedQueryProps));
104     }

108     void assertParseWithInvalidUrl(final String name, final String url) {
109         assertThrows(JdbcUriParseException.class, () -> parser.parse(url, null, null));
110     }

112     @Test
113     void assertParseWithEmbeddedMode() {
114         assertThrows(RuntimeException.class, () -> parser.parse("jdbc:hive2://", null, null));
115     }
116     
117     @Test
118     void assertParseWithNullHostAndNonZeroPort() {
119         JdbcConnectionParams jdbcConnectionParams = mock(JdbcConnectionParams.class);
120         when(jdbcConnectionParams.getHost()).thenReturn(null);
121         when(jdbcConnectionParams.getPort()).thenReturn(10000);
122         when(jdbcConnectionParams.getDbName()).thenReturn("default");
123         when(jdbcConnectionParams.getSessionVars()).thenReturn(Collections.singletonMap("sessionKey", "sessionValue"));
124         when(jdbcConnectionParams.getHiveConfs()).thenReturn(Collections.singletonMap("hiveConfKey", "hiveConfValue"));
125         when(jdbcConnectionParams.getHiveVars()).thenReturn(Collections.singletonMap("hiveVarKey", "hiveVarValue"));
126         try (MockedStatic<Utils> mockedUtils = mockStatic(Utils.class, CALLS_REAL_METHODS)) {
127             mockedUtils.when(() -> Utils.parseURL(anyString(), any(Properties.class))).thenReturn(jdbcConnectionParams);
128             ConnectionProperties actualConnectionProperties = parser.parse("jdbc:hive2://branch/null-host-port/default", null, null);
129             assertNull(actualConnectionProperties.getHostname());
130             assertThat(actualConnectionProperties.getPort(), is(10000));
131             assertThat(actualConnectionProperties.getCatalog(), is("default"));
132             assertNull(actualConnectionProperties.getSchema());
133             assertThat(actualConnectionProperties.getQueryProperties(),
134                     is(PropertiesBuilder.build(new Property("sessionKey", "sessionValue"), new Property("hiveConfKey", "hiveConfValue"), new Property("hiveVarKey", "hiveVarValue"))));
135         }
136     }

143                 Arguments.of("complex", "jdbc:hive2://server:10002/db;user=foo;password=bar?transportMode=http;httpPath=hs2", "server", 10002, "db",
144                         PropertiesBuilder.build(new Property("user", "foo"), new Property("password", "bar"), new Property("transportMode", "http"), new Property("httpPath", "hs2"))));
145     }
