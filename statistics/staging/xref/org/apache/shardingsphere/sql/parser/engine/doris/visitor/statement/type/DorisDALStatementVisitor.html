1028     private String getPropertyValue(final PropertyValueContext ctx) {
1029         if (null != ctx.identifier()) {
1030             return ((IdentifierValue) visit(ctx.identifier())).getValue();
1031         }
1032         ASTNode result = visit(ctx.literals());
1033         if (result instanceof LiteralValue) {

1035         }
1036         return result.toString();

1039     private String getLiteralValueAsString(final LiteralValue<?> literalValue) {
1040         if (literalValue instanceof StringLiteralValue) {
1041             return ((StringLiteralValue) literalValue).getValue();
1042         }
1043         if (literalValue instanceof NumberLiteralValue) {
1044             return ((NumberLiteralValue) literalValue).getValue().toString();
1045         }
1046         if (literalValue instanceof BooleanLiteralValue) {
1047             return String.valueOf(((BooleanLiteralValue) literalValue).getValue());
1048         }
1049         if (literalValue instanceof NullLiteralValue) {
1050             return "NULL";
1051         }
1052         if (literalValue instanceof DateTimeLiteralValue) {
1053             return ((DateTimeLiteralValue) literalValue).getValue();
1054         }
1055         if (literalValue instanceof TemporalLiteralValue) {
1056             return ((TemporalLiteralValue) literalValue).getValue();
1057         }
1058         if (literalValue instanceof OtherLiteralValue) {
1059             return String.valueOf(((OtherLiteralValue) literalValue).getValue());
1060         }
1061         return String.valueOf(literalValue.getValue());
1062     }
