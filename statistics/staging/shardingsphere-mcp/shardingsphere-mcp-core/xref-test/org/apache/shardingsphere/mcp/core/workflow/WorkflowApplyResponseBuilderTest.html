30  import static org.hamcrest.MatcherAssert.assertThat;
31  import static org.hamcrest.Matchers.is;
32  import static org.junit.jupiter.api.Assertions.assertFalse;

34  class WorkflowApplyResponseBuilderTest {
35      
36      @Test
37      void assertBuildPreviewResponse() {
38          List<WorkflowArtifactBundle.ExecutableWorkflowArtifact> executableArtifacts = List.of(
39                  new WorkflowArtifactBundle.ExecutableWorkflowArtifact("CREATE MASK RULE orders", "CREATE MASK RULE orders"),
40                  new WorkflowArtifactBundle.ExecutableWorkflowArtifact("ALTER MASK RULE orders", "ALTER MASK RULE orders"));
41          Map<String, Object> actual = new WorkflowApplyResponseBuilder().buildPreviewResponse(createSnapshot("mask.rule"), executableArtifacts, "review-then-execute");
42          assertThat(actual.get("response_mode"), is("preview"));
43          assertThat(actual.get("summary"), is("Previewed 2 workflow artifacts with side-effect scope rule-metadata. Nothing has been applied."));
44          assertThat(actual.get("plan_id"), is("plan-1"));
45          assertThat(actual.get("status"), is("preview"));
46          assertThat(actual.get("execution_mode"), is("preview"));
47          assertFalse(actual.containsKey(WorkflowArtifactPayloadUtils.PAYLOAD_KEY_MANUAL_ARTIFACT_PACKAGE));
48          assertThat(((List<?>) actual.get("preview_artifacts")).size(), is(2));
49          Map<?, ?> actualReviewFocus = (Map<?, ?>) actual.get("review_focus");
50          assertThat(actualReviewFocus.get("artifact_categories"), is(List.of("rule_distsql")));
51          assertThat(actualReviewFocus.get("side_effect_scope"), is(List.of("rule-metadata")));
52          assertThat(actualReviewFocus.get("approval_values"), is(List.of("rule_distsql")));
53          assertThat(((List<?>) actual.get("next_actions")).size(), is(2));
54      }
55      
56      @Test
57      void assertBuildCompletedResponse() {
58          Map<String, Object> actual = new WorkflowApplyResponseBuilder().build(createSnapshot("encrypt.rule"), WorkflowLifecycle.STATUS_COMPLETED, "review-then-execute",
59                  List.of(), List.of(Map.of("artifact_type", "rule_distsql", "status", WorkflowLifecycle.STATUS_PASSED, "sql", "CREATE ENCRYPT RULE orders")), Map.of());
60          assertThat(actual.get("response_mode"), is("executed"));
61          assertThat(actual.get("summary"), is("Workflow apply completed for plan `plan-1` with 1 applied artifact(s)."));
62          assertThat(((Map<?, ?>) ((List<?>) actual.get("step_results")).getFirst()).get("sql"), is("CREATE ENCRYPT RULE orders"));
63          assertThat(((List<?>) actual.get("next_actions")).size(), is(1));
64      }
65      
66      @Test
67      void assertBuildManualResponse() {
68          Map<String, Object> manualArtifactPackage = Map.of(
69                  WorkflowArtifactPayloadUtils.PAYLOAD_KEY_DISTSQL_ARTIFACTS, List.of(Map.of("sql", "CREATE ENCRYPT RULE orders")));
70          Map<String, Object> actual = new WorkflowApplyResponseBuilder().build(createSnapshot("encrypt.rule"), WorkflowLifecycle.STATUS_AWAITING_MANUAL_EXECUTION, "manual-only",
71                  List.of(), List.of(), manualArtifactPackage);
72          assertThat(actual.get("response_mode"), is("manual_only"));
73          assertThat(actual.get("summary"), is("Workflow apply exported manual artifacts for plan `plan-1`; external execution is required before validation."));
74          assertThat(actual.get(WorkflowArtifactPayloadUtils.PAYLOAD_KEY_MANUAL_ARTIFACT_PACKAGE), is(manualArtifactPackage));
75          Map<?, ?> actualSummary = (Map<?, ?>) actual.get("manual_artifact_summary");
76          assertThat(actualSummary.get("distsql_artifact_count"), is(1));
77          assertThat(actualSummary.get("safe_independent_read_only_checks"),
78                  is("database_gateway_execute_query may run before manual execution confirmation when the user asked for read-only verification."));
79      }
80      
81      private WorkflowContextSnapshot createSnapshot(final String workflowKind) {
82          WorkflowContextSnapshot result = new WorkflowContextSnapshot();
83          result.setPlanId("plan-1");
84          result.setWorkflowKind(WorkflowKind.valueOf(workflowKind));
85          return result;
86      }
87  }
