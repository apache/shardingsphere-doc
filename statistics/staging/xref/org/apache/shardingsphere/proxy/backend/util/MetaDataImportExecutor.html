18  package org.apache.shardingsphere.proxy.backend.util;
19  
20  import lombok.RequiredArgsConstructor;
21  import org.apache.shardingsphere.infra.config.rule.RuleConfiguration;
22  import org.apache.shardingsphere.infra.util.yaml.YamlEngine;
23  import org.apache.shardingsphere.infra.yaml.config.swapper.rule.YamlRuleConfigurationSwapperEngine;
24  import org.apache.shardingsphere.mode.manager.ContextManager;
25  import org.apache.shardingsphere.proxy.backend.config.yaml.YamlProxyDatabaseConfiguration;
26  import org.apache.shardingsphere.proxy.backend.config.yaml.YamlProxyServerConfiguration;
27  import org.apache.shardingsphere.proxy.backend.distsql.export.ExportedMetaData;
28  
29  import java.sql.SQLException;
30  import java.util.Collection;
31  import java.util.LinkedList;
32  
33  /**
34   * Metadata import executor.
35   */
36  @RequiredArgsConstructor
37  public final class MetaDataImportExecutor {
38      
39      private final YamlRuleConfigurationSwapperEngine ruleConfigSwapperEngine = new YamlRuleConfigurationSwapperEngine();
40      
41      private final YamlDatabaseConfigurationImportExecutor databaseConfigImportExecutor;
42      
43      private final ContextManager contextManager;
44      
45      public MetaDataImportExecutor(final ContextManager contextManager) {
46          this.contextManager = contextManager;
47          this.databaseConfigImportExecutor = new YamlDatabaseConfigurationImportExecutor(contextManager);
48      }
49      
50      /**
51       * Import cluster configurations.
52       *
53       * @param exportedMetaData exported metadata
54       * @throws SQLException SQL exception
55       */
56      public void importClusterConfigurations(final ExportedMetaData exportedMetaData) throws SQLException {
57          Collection<YamlProxyDatabaseConfiguration> databaseConfigs = getYamlProxyDatabaseConfigurations(exportedMetaData);
58          importServerConfiguration(exportedMetaData);
59          importDatabaseConfigurations(databaseConfigs);
60      }
61      
62      private void importServerConfiguration(final ExportedMetaData exportedMetaData) throws SQLException {
63          YamlProxyServerConfiguration yamlServerConfig = YamlEngine.unmarshal(exportedMetaData.getRules() + System.lineSeparator() + exportedMetaData.getProps(), YamlProxyServerConfiguration.class);
64          if (null == yamlServerConfig) {
65              return;
66          }
67          importGlobalRules(yamlServerConfig);
68          importProps(yamlServerConfig);
69      }
70      
71      private void importGlobalRules(final YamlProxyServerConfiguration yamlServerConfig) {
72          Collection<RuleConfiguration> rules = ruleConfigSwapperEngine.swapToRuleConfigurations(yamlServerConfig.getRules());
73          for (RuleConfiguration each : rules) {
74              contextManager.getPersistServiceFacade().getModePersistServiceFacade().getMetaDataManagerPersistService().alterGlobalRuleConfiguration(each);
75          }
76      }
77      
78      private void importProps(final YamlProxyServerConfiguration yamlServerConfig) {
79          contextManager.getPersistServiceFacade().getModePersistServiceFacade().getMetaDataManagerPersistService().alterProperties(yamlServerConfig.getProps());
80      }
81      
82      private Collection<YamlProxyDatabaseConfiguration> getYamlProxyDatabaseConfigurations(final ExportedMetaData exportedMetaData) {
83          Collection<YamlProxyDatabaseConfiguration> result = new LinkedList<>();
84          for (String each : exportedMetaData.getDatabases().values()) {
85              result.add(YamlEngine.unmarshal(each, YamlProxyDatabaseConfiguration.class));
86          }
87          return result;
88      }
89      
90      /**
91       * Import database configurations.
92       *
93       * @param databaseConfigs YAML proxy database configuration
94       */
95      public void importDatabaseConfigurations(final Collection<YamlProxyDatabaseConfiguration> databaseConfigs) {
96          for (YamlProxyDatabaseConfiguration each : databaseConfigs) {
97              databaseConfigImportExecutor.importDatabaseConfiguration(each);
98          }
99      }
100 }
