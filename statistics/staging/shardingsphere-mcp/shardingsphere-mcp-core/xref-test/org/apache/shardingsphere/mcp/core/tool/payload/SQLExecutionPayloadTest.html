36  class SQLExecutionPayloadTest {
37      
38      @Test
39      void assertQueryResultSet() {
40          SQLExecutionColumnDefinition column = new SQLExecutionColumnDefinition("order_id", "INT", "INT", false);
41          Map<String, Object> actual = SQLExecutionPayload.query(SQLExecutionResult.resultSet(
42                  SupportedMCPStatement.QUERY, "SELECT", List.of(column), List.of(List.of(1)), false, 10, 5000, "SELECT order_id FROM orders")).toPayload();
43          assertThat(actual, is(Map.ofEntries(
44                  Map.entry("response_mode", "query"), Map.entry("result_kind", "result_set"), Map.entry("statement_class", "query"), Map.entry("statement_type", "SELECT"),
45                  Map.entry("status", "OK"), Map.entry("summary", "Executed SELECT statement and returned 1 row(s)."),
46                  Map.entry("normalized_sql", "SELECT order_id FROM orders"), Map.entry("columns", List.of(column)), Map.entry("rows", List.of(List.of(1))),
47                  Map.entry("row_object_status", "available"), Map.entry("row_objects", List.of(Map.of("order_id", 1))), Map.entry("returned_row_count", 1),
48                  Map.entry("applied_max_rows", 10), Map.entry("applied_timeout_ms", 5000), Map.entry("truncated", false),
49                  Map.entry("next_actions", List.of(MCPNextActionUtils.stop(
50                          "Return the result rows to the user or ask a follow-up question if the user requested more analysis."))))));
51      }
52      
53      @Test
54      void assertTruncatedQueryResultSet() {
55          Map<String, Object> actual = SQLExecutionPayload.query(SQLExecutionResult.resultSet(
56                  SupportedMCPStatement.QUERY, "SELECT", List.of(), List.of(), true, 10, 0, "SELECT * FROM orders")).toPayload();
57          assertThat(actual.get("summary"), is("Executed SELECT statement and returned 0 row(s). Result was truncated."));
58          assertThat(actual.get("next_actions"), is(List.of(MCPNextActionUtils.askUser(
59                  "The result was truncated by max_rows. Ask for a narrower SELECT, stronger WHERE clause, or smaller projection before retrying.", List.of("sql")))));
60      }
61      
62      @Test
63      void assertExecutedUpdateCount() {
64          Map<String, Object> actual = SQLExecutionPayload.executed(SQLExecutionResult.updateCount(
65                  SupportedMCPStatement.DML, "UPDATE", 2, 100, 0, "UPDATE orders SET status = 'PAID'")).toPayload();
66          assertThat(actual.get("response_mode"), is("executed"));
67          assertThat(actual.get("execution_mode"), is("execute"));
68          assertThat(actual.get("result_kind"), is("update_count"));
69          assertThat(actual.get("affected_rows"), is(2));
70          assertThat(actual.get("summary"), is("Executed UPDATE statement and affected 2 row(s)."));
71      }
72      
73      @Test
74      void assertExecutedResultSet() {
75          Map<String, Object> actual = SQLExecutionPayload.executed(SQLExecutionResult.resultSet(
76                  SupportedMCPStatement.DML, "SELECT", List.of(), List.of(List.of(1)), false, 100, 0, "WITH changed AS (...) SELECT * FROM changed")).toPayload();
77          assertThat(actual.get("summary"), is("Executed side-effecting SQL (statement type SELECT) and returned 1 row(s)."));
78          assertThat(actual.get("response_mode"), is("executed"));
79          assertThat(actual.get("execution_mode"), is("execute"));
80      }
81      
82      @Test
83      void assertTruncatedExecutedResultSet() {
84          Map<String, Object> actual = SQLExecutionPayload.executed(SQLExecutionResult.resultSet(
85                  SupportedMCPStatement.DML, "SELECT", List.of(), List.of(List.of(1)), true, 1, 0, "WITH changed AS (...) SELECT * FROM changed")).toPayload();
86          assertThat(actual.get("summary"), is("Executed side-effecting SQL (statement type SELECT) and returned 1 row(s). "
87                  + "Returned rows were truncated; do not replay the statement automatically."));
88          assertThat(actual.get("next_actions"), is(List.of(MCPNextActionUtils.stop(
89                  "The side-effecting statement already executed and returned truncated rows; do not replay it automatically. "
90                          + "Use a separate read-only query if more data is needed."))));
91      }

93      @ParameterizedTest(name = "{0}")
94      @CsvSource({
95              "begin, TRANSACTION_CONTROL, BEGIN, Transaction started.",
96              "start transaction, TRANSACTION_CONTROL, START TRANSACTION, Transaction started.",
97              "commit, TRANSACTION_CONTROL, COMMIT, Transaction committed.",
98              "rollback, TRANSACTION_CONTROL, ROLLBACK, Transaction rolled back.",
99              "savepoint, SAVEPOINT, SAVEPOINT, Savepoint created.",
100             "rollback to savepoint, SAVEPOINT, ROLLBACK TO SAVEPOINT, Savepoint rolled back.",
101             "release savepoint, SAVEPOINT, RELEASE SAVEPOINT, Savepoint released.",
102             "create, DDL, CREATE, Statement executed."
103     })
104     void assertStatementAcknowledgement(final String name, final SupportedMCPStatement statementClass, final String statementType, final String expectedSummary) {
105         Map<String, Object> actual = SQLExecutionPayload.executed(SQLExecutionResult.statementAck(
106                 statementClass, statementType, 100, 0, statementType)).toPayload();
107         assertThat(actual.get("summary"), is(expectedSummary));
108         assertFalse(actual.containsKey("message"));
109     }

113         Map<String, Object> actual = SQLExecutionPayload.query(SQLExecutionResult.resultSet(
114                 SupportedMCPStatement.QUERY, "SELECT", List.of(), List.of(List.of(1)), false, 100, 0, "SELECT 1")).toPayload();
115         assertThat(actual.get("row_object_status"), is("unnamed_columns"));
116         assertFalse(actual.containsKey("row_objects"));

127     @Test
128     void assertDuplicateColumnLabels() {
129         List<SQLExecutionColumnDefinition> columns = List.of(
130                 new SQLExecutionColumnDefinition("order_id", "INT", "INT", false), new SQLExecutionColumnDefinition("order_id", "INT", "INT", false));
131         Map<String, Object> actual = SQLExecutionPayload.query(SQLExecutionResult.resultSet(
132                 SupportedMCPStatement.QUERY, "SELECT", columns, List.of(List.of(1, 2)), false, 100, 0, "SELECT order_id, order_id FROM orders")).toPayload();
133         assertThat(actual.get("row_object_status"), is("duplicate_column_labels"));
134         assertFalse(actual.containsKey("row_objects"));
135     }

137     @Test
138     void assertLargeRowObjectsOmitted() {
139         List<List<Object>> rows = new LinkedList<>();
140         for (int i = 0; i < 101; i++) {
141             rows.add(List.of(i));
142         }
143         Map<String, Object> actual = SQLExecutionPayload.query(SQLExecutionResult.resultSet(SupportedMCPStatement.QUERY, "SELECT",
144                 List.of(new SQLExecutionColumnDefinition("order_id", "INT", "INT", false)), rows, false, 100, 0, "SELECT order_id FROM orders")).toPayload();
145         assertThat(actual.get("row_object_status"), is("omitted_large_result"));
146         assertFalse(actual.containsKey("row_objects"));
147     }
