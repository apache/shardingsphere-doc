45      private static final int COMPLETION_MAX_TOKENS = 512;

47      private static final int READINESS_MAX_TOKENS = 64;

49      private final LLME2EConfiguration config;

53      /**
54       * Wait until ready.
55       *
56       * @throws InterruptedException interrupted exception
57       * @throws IllegalStateException model service is not ready
58       */
59      public void waitUntilReady() throws InterruptedException {
60          new LLMChatReadinessProbe(config, this).waitUntilReady();
61      }
62      
63      String createHttpReadinessFailure(final String requestKind, final HttpResponse<String> response) {
64          return String.format("%s readiness request returned HTTP %d%s.", requestKind, response.statusCode(), createErrorCodeSuffix(response.body()));
65      }
66      
67      private String createErrorCodeSuffix(final String responseBody) {
68          try {
69              Map<String, Object> payload = parseJsonObject(responseBody, "Failed to parse readiness error response.");
70              String errorCode = Objects.toString(castToMap(payload.get("error")).get("code"), "").trim();
71              return errorCode.isEmpty() ? "" : String.format(" with error code `%s`", errorCode);
72          } catch (final IllegalStateException ignored) {
73              return "";
74          }
75      }
76      
77      /**
78       * Complete.
79       *
80       * @param messages messages
81       * @param tools tools
82       * @param toolChoice tool choice
83       * @param responseFormat response format
84       * @return LLM chat completion
85       * @throws IOException IO exception
86       * @throws InterruptedException interrupted exception
87       * @throws IllegalStateException model completion response is invalid
88       */
89      public LLMChatCompletion complete(final List<LLMChatMessage> messages, final List<Map<String, Object>> tools,
90                                        final String toolChoice, final Map<String, Object> responseFormat) throws IOException, InterruptedException {
91          HttpResponse<String> response = sendCompletionRequest(
92                  createCompletionRequestPayload(messages, tools, toolChoice, responseFormat, COMPLETION_MAX_TOKENS), config.getRequestTimeoutSeconds());
93          if (200 != response.statusCode()) {
94              throw new IllegalStateException(String.format("Model completion request failed with status %d%s.", response.statusCode(), createErrorCodeSuffix(response.body())));
95          }
96          Map<String, Object> payload = parseJsonObject(response.body(), "Failed to parse model completion response.");
97          List<Map<String, Object>> choices = castToList(payload.get("choices"));
98          Preconditions.checkState(!choices.isEmpty(), "Model completion response does not contain choices.");
99          Map<String, Object> message = castToMap(choices.get(0).get("message"));
100         Preconditions.checkState(!message.isEmpty(), "Model completion response does not contain one assistant message.");
101         return new LLMChatCompletion(Objects.toString(message.get("content"), "").trim(), createToolCalls(message.get("tool_calls")), response.body());
102     }
103     
104     private Map<String, Object> createCompletionRequestPayload(final List<LLMChatMessage> messages, final List<Map<String, Object>> tools,
105                                                                final String toolChoice, final Map<String, Object> responseFormat, final int maxTokens) {
106         Map<String, Object> requestPayload = new LinkedHashMap<>(16, 1F);
107         requestPayload.put("model", config.getModelName());
108         requestPayload.put("messages", createMessages(messages));
109         requestPayload.put("stream", false);
110         requestPayload.put("temperature", 0);
111         requestPayload.put("seed", 1);
112         requestPayload.put("reasoning_effort", "none");
113         requestPayload.put("max_tokens", maxTokens);
114         if (!tools.isEmpty()) {
115             requestPayload.put("tools", tools);
116         }
117         if (!toolChoice.isEmpty()) {
118             requestPayload.put("tool_choice", toolChoice);
119         }
120         if (!responseFormat.isEmpty()) {
121             requestPayload.put("response_format", responseFormat);
122         }
123         return requestPayload;
124     }
125     
126     private HttpResponse<String> sendCompletionRequest(final Map<String, Object> requestPayload, final int timeoutSeconds) throws IOException, InterruptedException {
127         HttpRequest request = HttpRequest.newBuilder(URI.create(config.getChatCompletionsUrl()))
128                 .timeout(Duration.ofSeconds(timeoutSeconds))
129                 .header("Authorization", "Bearer " + config.getApiKey())
130                 .header("Content-Type", "application/json")
131                 .POST(HttpRequest.BodyPublishers.ofString(JsonEngine.marshal(requestPayload)))
132                 .build();
133         return httpClient.send(request, HttpResponse.BodyHandlers.ofString());
134     }
135     
136     HttpResponse<String> sendModelListRequest() throws IOException, InterruptedException {
137         HttpRequest request = HttpRequest.newBuilder(URI.create(config.getModelsUrl()))
138                 .timeout(Duration.ofSeconds(Math.min(config.getRequestTimeoutSeconds(), 30)))
139                 .header("Authorization", "Bearer " + config.getApiKey())
140                 .GET()
141                 .build();
142         return httpClient.send(request, HttpResponse.BodyHandlers.ofString());
143     }
144     
145     HttpResponse<String> sendReadinessCompletionRequest(final List<LLMChatMessage> messages, final List<Map<String, Object>> tools,
146                                                         final String toolChoice, final boolean jsonResponse) throws IOException, InterruptedException {
147         return sendCompletionRequest(createCompletionRequestPayload(messages, tools, toolChoice, jsonResponse ? Map.of("type", "json_object") : Map.of(), READINESS_MAX_TOKENS),
148                 Math.min(config.getRequestTimeoutSeconds(), config.getReadyTimeoutSeconds()));
149     }
150     
151     boolean containsModel(final String responseBody) {
152         Map<String, Object> payload = parseJsonObject(responseBody, "Failed to parse model-list response.");
153         return castToList(payload.get("data")).stream().anyMatch(each -> config.getModelName().equals(Objects.toString(each.get("id"), "").trim()));
154     }
155     
156     boolean hasToolCallChoice(final String responseBody) {
157         return !createToolCalls(getFirstAssistantMessage(responseBody, "Failed to parse readiness tool-call response.").get("tool_calls")).isEmpty();
158     }
159     
160     private Map<String, Object> getFirstAssistantMessage(final String responseBody, final String errorMessage) {
161         List<Map<String, Object>> choices = castToList(parseJsonObject(responseBody, errorMessage).get("choices"));
162         return choices.isEmpty() ? Map.of() : castToMap(choices.get(0).get("message"));
163     }

165     List<Map<String, Object>> createReadinessTools() {
166         return List.of(Map.of(
167                 "type", "function",
168                 "function", Map.of(
169                         "name", "mcp_read_resource",
170                         "description", "Read one MCP resource.",
171                         "parameters", Map.of(
172                                 "type", "object",
173                                 "properties", Map.of("uri", Map.of("type", "string")),
174                                 "required", List.of("uri")))));
175     }

177     private List<Map<String, Object>> createMessages(final List<LLMChatMessage> messages) {
178         List<Map<String, Object>> result = new LinkedList<>();
179         for (LLMChatMessage each : messages) {
180             Map<String, Object> message = new LinkedHashMap<>(8, 1F);
181             message.put("role", each.getRole());
182             if (!each.getToolCallId().isEmpty()) {
183                 message.put("tool_call_id", each.getToolCallId());
184             }
185             if (!each.getContent().isEmpty()) {
186                 message.put("content", each.getContent());
187             } else if (each.getToolCalls().isEmpty()) {
188                 message.put("content", "");
189             }
190             if (!each.getToolCalls().isEmpty()) {
191                 message.put("tool_calls", createToolCalls(each.getToolCalls()));
192             }
193             result.add(message);
194         }
195         return result;
196     }
197     
198     private List<Map<String, Object>> createToolCalls(final List<LLMToolCall> toolCalls) {
199         return toolCalls.stream()
200                 .map(each -> Map.of("id", each.getId(), "type", "function", "function", Map.of("name", each.getName(), "arguments", each.getArgumentsJson()))).collect(Collectors.toList());
201     }
202     
203     private List<LLMToolCall> createToolCalls(final Object rawValue) {
204         List<LLMToolCall> result = new LinkedList<>();
205         for (Map<String, Object> each : castToList(rawValue)) {
206             Map<String, Object> function = castToMap(each.get("function"));
207             result.add(new LLMToolCall(
208                     Objects.toString(each.get("id"), "").trim(), Objects.toString(function.get("name"), "").trim(), Objects.toString(function.get("arguments"), "").trim()));
209         }
210         return result;
211     }
212     
213     private List<Map<String, Object>> castToList(final Object value) {
214         if (null == value) {
215             return List.of();
216         }
217         Preconditions.checkState(value instanceof List, "Expected a JSON array, but got `%s`.", value.getClass().getSimpleName());
218         List<?> values = (List<?>) value;
219         Preconditions.checkState(values.stream().allMatch(Map.class::isInstance), "Expected a JSON array of objects.");
220         return values.stream().map(this::castToMap).toList();
221     }
222     
223     @SuppressWarnings("unchecked")
224     private Map<String, Object> castToMap(final Object value) {
225         if (null == value) {
226             return Map.of();
227         }
228         Preconditions.checkState(value instanceof Map, "Expected a JSON object, but got `%s`.", value.getClass().getSimpleName());
229         Map<?, ?> result = (Map<?, ?>) value;
230         Preconditions.checkState(result.keySet().stream().allMatch(String.class::isInstance), "Expected a JSON object with string keys.");
231         return (Map<String, Object>) result;
232     }
233     
234     private Map<String, Object> parseJsonObject(final String responseBody, final String errorMessage) {
235         try {
236             return JsonEngine.unmarshal(responseBody, new JsonTypeReference<>() {
237             });
238             // CHECKSTYLE:OFF
239         } catch (final Exception ex) {
240             // CHECKSTYLE:ON
241             throw new IllegalStateException(errorMessage, ex);
242         }
243     }
