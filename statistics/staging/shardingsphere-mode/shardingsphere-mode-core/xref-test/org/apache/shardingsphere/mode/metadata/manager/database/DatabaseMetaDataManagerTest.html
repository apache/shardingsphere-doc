62      @Mock(answer = Answers.RETURNS_DEEP_STUBS)
63      private MetaDataContexts metaDataContexts;
64      
65      @Mock
66      private GlobalRule globalRule;
67      
68      private DatabaseMetaDataManager databaseMetaDataManager;
69      
70      @BeforeEach
71      void setUp() {
72          ShardingSphereDatabase database = mockDatabase();
73          lenient().when(metaDataContexts.getMetaData().containsDatabase("foo_db")).thenReturn(true);
74          lenient().when(metaDataContexts.getMetaData().getDatabase("foo_db")).thenReturn(database);
75          lenient().when(metaDataContexts.getMetaData().getAllDatabases()).thenReturn(Collections.singleton(database));
76          lenient().when(metaDataContexts.getMetaData().getGlobalRuleMetaData().getRules()).thenReturn(Collections.singleton(globalRule));
77          databaseMetaDataManager = new DatabaseMetaDataManager(metaDataContexts, mock(MetaDataPersistFacade.class));
78      }
79      
80      private ShardingSphereDatabase mockDatabase() {
81          ShardingSphereDatabase result = mock(ShardingSphereDatabase.class, RETURNS_DEEP_STUBS);
82          lenient().when(result.getName()).thenReturn("foo_db");
83          lenient().when(result.getProtocolType()).thenReturn(TypedSPILoader.getService(DatabaseType.class, "FIXTURE"));
84          lenient().when(result.containsSchema("foo_schema")).thenReturn(true);
85          return result;
86      }

88      @Test
89      void assertAddNotExistedDatabase() {
90          databaseMetaDataManager.addDatabase("new_db");
91          verify(metaDataContexts.getMetaData()).addDatabase(eq("new_db"), any(DatabaseType.class), any(ConfigurationProperties.class));
92      }

95      void assertAddExistedDatabase() {

97          databaseMetaDataManager.addDatabase("foo_db");
98          verify(metaDataContexts.getMetaData(), never()).addDatabase(eq("foo_db"), any(DatabaseType.class), any(ConfigurationProperties.class));
99      }

141     @Test
142     void assertDropSchemaWithSingleTableRefreshRules() {
143         ShardingSphereSchema schema = mock(ShardingSphereSchema.class);
144         ShardingSphereTable singleTable = new ShardingSphereTable("foo_tbl", Collections.emptyList(), Collections.emptyList(), Collections.emptyList());
145         when(schema.getAllTables()).thenReturn(Collections.singleton(singleTable));
146         when(metaDataContexts.getMetaData().getDatabase("foo_db").getSchema("foo_schema")).thenReturn(schema);
147         when(TableRefreshUtils.isSingleTable("foo_tbl", metaDataContexts.getMetaData().getDatabase("foo_db"))).thenReturn(true);
148         databaseMetaDataManager.dropSchema("foo_db", "foo_schema");
149         verify(metaDataContexts.getMetaData().getDatabase("foo_db")).reloadRules();

152     @Test
153     void assertRenameSchema() {
154         ShardingSphereSchema schema = createToBeAlteredSchema();
155         when(metaDataContexts.getMetaData().getDatabase("foo_db").getSchema("foo_schema")).thenReturn(schema);
156         databaseMetaDataManager.renameSchema("foo_db", "foo_schema", "bar_schema");
157         verify(metaDataContexts.getMetaData().getDatabase("foo_db")).addSchema(any(ShardingSphereSchema.class));
158         verify(metaDataContexts.getMetaData().getDatabase("foo_db")).dropSchema("foo_schema");
159         verify(metaDataContexts.getMetaData().getDatabase("foo_db")).reloadRules();

171     @Test
172     void assertAlterTable() {
173         ShardingSphereSchema toBeAlteredSchema = createToBeAlteredSchema();
174         when(metaDataContexts.getMetaData().getDatabase("foo_db").getAllSchemas()).thenReturn(Collections.singleton(toBeAlteredSchema));
175         when(metaDataContexts.getMetaData().getDatabase("foo_db").getSchema("foo_schema")).thenReturn(toBeAlteredSchema);
176         ShardingSphereColumn toBeChangedColumn = new ShardingSphereColumn("foo_col", Types.VARCHAR, false, false, false, true, false, false);
177         ShardingSphereTable toBeChangedTable = new ShardingSphereTable("foo_tbl", Collections.singleton(toBeChangedColumn), Collections.emptyList(), Collections.emptyList());
178         databaseMetaDataManager.alterTable("foo_db", "foo_schema", toBeChangedTable);
179         ShardingSphereTable table = metaDataContexts.getMetaData().getDatabase("foo_db").getSchema("foo_schema").getTable("foo_tbl");
180         assertThat(table.getAllColumns().size(), is(1));
181         assertTrue(table.containsColumn("foo_col"));
182     }

184     @Test
185     void assertAlterTableWithSingleTableReloadRules() {
186         ShardingSphereSchema toBeAlteredSchema = createToBeAlteredSchema();
187         when(metaDataContexts.getMetaData().getDatabase("foo_db").containsSchema("foo_schema")).thenReturn(true);
188         when(metaDataContexts.getMetaData().getDatabase("foo_db").getSchema("foo_schema")).thenReturn(toBeAlteredSchema);
189         when(TableRefreshUtils.isSingleTable("foo_tbl", metaDataContexts.getMetaData().getDatabase("foo_db"))).thenReturn(true);
190         databaseMetaDataManager.alterTable("foo_db", "foo_schema", new ShardingSphereTable("foo_tbl", Collections.emptyList(), Collections.emptyList(), Collections.emptyList()));
191         verify(metaDataContexts.getMetaData().getDatabase("foo_db")).reloadRules();
192     }

194     @Test
195     void assertAlterViewWithSingleTableReloadRules() {
196         ShardingSphereSchema toBeAlteredSchema = createToBeAlteredSchema();
197         when(metaDataContexts.getMetaData().getDatabase("foo_db").containsSchema("foo_schema")).thenReturn(true);
198         when(metaDataContexts.getMetaData().getDatabase("foo_db").getSchema("foo_schema")).thenReturn(toBeAlteredSchema);
199         when(TableRefreshUtils.isSingleTable("foo_view", metaDataContexts.getMetaData().getDatabase("foo_db"))).thenReturn(true);
200         databaseMetaDataManager.alterView("foo_db", "foo_schema", new ShardingSphereView("foo_view", ""));
201         verify(metaDataContexts.getMetaData().getDatabase("foo_db")).reloadRules();
202     }
203     
204     @Test
205     void assertAlterView() {
206         ShardingSphereSchema toBeAlteredSchema = createToBeAlteredSchema();
207         when(metaDataContexts.getMetaData().getDatabase("foo_db").getAllSchemas()).thenReturn(Collections.singleton(toBeAlteredSchema));
208         when(metaDataContexts.getMetaData().getDatabase("foo_db").getSchema("foo_schema")).thenReturn(toBeAlteredSchema);
209         ShardingSphereView toBeChangedView = new ShardingSphereView("foo_view", "SELECT `foo_view`.`foo_view`.`id` AS `id` FROM `foo_view`.`foo_view`");
210         databaseMetaDataManager.alterView("foo_db", "foo_schema", toBeChangedView);
211         ShardingSphereView view = metaDataContexts.getMetaData().getDatabase("foo_db").getSchema("foo_schema").getView("foo_view");
212         assertThat(view.getName(), is("foo_view"));
213         assertThat(view.getViewDefinition(), is("SELECT `foo_view`.`foo_view`.`id` AS `id` FROM `foo_view`.`foo_view`"));
214     }

251     private ShardingSphereSchema createToBeAlteredSchema() {
252         ShardingSphereTable beforeChangedTable = new ShardingSphereTable("foo_tbl", Collections.emptyList(), Collections.emptyList(), Collections.emptyList());
253         ShardingSphereView beforeChangedView = new ShardingSphereView("foo_view", "");
254         return new ShardingSphereSchema("foo_schema", Collections.singleton(beforeChangedTable), Collections.singleton(beforeChangedView));
