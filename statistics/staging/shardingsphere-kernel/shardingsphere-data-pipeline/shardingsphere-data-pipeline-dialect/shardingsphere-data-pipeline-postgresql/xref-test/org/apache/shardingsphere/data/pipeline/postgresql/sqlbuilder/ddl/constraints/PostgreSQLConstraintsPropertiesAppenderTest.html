49      @Test
50      void assertAppend() {
51          constraintsCallCounter.set(0);
52          PostgreSQLDDLTemplateExecutor templateExecutor = mock(PostgreSQLDDLTemplateExecutor.class);
53          when(templateExecutor.getMajorVersion()).thenReturn(11);
54          when(templateExecutor.executeByTemplate(anyMap(), anyString())).thenAnswer(invocation -> mockExecuteByTemplateFullCoverage(invocation.getArgument(0), invocation.getArgument(1)));
55          when(templateExecutor.executeByTemplateForSingleRow(anyMap(), anyString())).thenReturn(createParentTable());
56          Map<String, Object> context = createContext(true);
57          createAppender(templateExecutor).append(context);
58          Collection<Map<String, Object>> primaryKeys = (Collection<Map<String, Object>>) context.get("primary_key");
59          assertThat(primaryKeys.size(), is(1));
60          Map<String, Object> primaryKey = primaryKeys.iterator().next();
61          Collection<String> actualPrimaryColumns = ((Collection<Map<String, Object>>) primaryKey.get("columns")).stream().map(each -> (String) each.get("column")).toList();
62          assertThat(actualPrimaryColumns, is(Arrays.asList("id", "name")));
63          assertThat(primaryKey.get("include"), is(Collections.singletonList("include_col")));
64          Collection<Map<String, Object>> uniqueConstraints = (Collection<Map<String, Object>>) context.get("unique_constraint");
65          assertThat(uniqueConstraints.size(), is(1));
66          Collection<Map<String, Object>> checkConstraints = (Collection<Map<String, Object>>) context.get("check_constraint");
67          assertThat(checkConstraints.size(), is(1));
68          assertFalse((boolean) checkConstraints.iterator().next().get("conislocal"));
69          List<Map<String, Object>> foreignKeys = new LinkedList<>((Collection<Map<String, Object>>) context.get("foreign_key"));
70          assertThat(foreignKeys.size(), is(2));
71          Map<String, Object> foreignKeyWithIndex = foreignKeys.get(0);
72          assertThat(foreignKeyWithIndex.get("coveringindex"), is("idx_fk_col1"));
73          assertFalse((boolean) foreignKeyWithIndex.get("autoindex"));
74          assertTrue((boolean) foreignKeyWithIndex.get("hasindex"));
75          assertThat(foreignKeyWithIndex.get("remote_schema"), is("remote_schema"));
76          assertThat(foreignKeyWithIndex.get("remote_table"), is("remote_table"));
77          Map<String, Object> foreignKeyColumn = ((Collection<Map<String, Object>>) foreignKeyWithIndex.get("columns")).iterator().next();
78          assertThat(foreignKeyColumn.get("local_column"), is("fk_col1"));
79          assertThat(foreignKeyColumn.get("referenced"), is("ref_col1"));
80          assertThat(foreignKeyColumn.get("references_table_name"), is("public.ref_table"));
81          Map<String, Object> foreignKeyWithoutIndex = foreignKeys.get(1);
82          assertThat(foreignKeyWithoutIndex.get("coveringindex"), is((Object) null));
83          assertTrue((boolean) foreignKeyWithoutIndex.get("autoindex"));
84          assertFalse((boolean) foreignKeyWithoutIndex.get("hasindex"));
85          Collection<Map<String, Object>> exclusionConstraints = (Collection<Map<String, Object>>) context.get("exclude_constraint");
86          assertThat(exclusionConstraints.size(), is(1));
87          Map<String, Object> exclusionConstraint = exclusionConstraints.iterator().next();
88          Collection<Boolean> actualOrders = ((Collection<Map<String, Object>>) exclusionConstraint.get("columns")).stream().map(each -> (Boolean) each.get("order")).toList();
89          Collection<Boolean> actualNullsOrders = ((Collection<Map<String, Object>>) exclusionConstraint.get("columns")).stream().map(each -> (Boolean) each.get("nulls_order")).toList();
90          assertThat(actualOrders, is(Arrays.asList(true, false)));
91          assertThat(actualNullsOrders, is(Arrays.asList(false, true)));
92          assertThat(exclusionConstraint.get("include"), is(Collections.singletonList("excl_inc")));
93      }

95      @Test
96      void assertAppendWithoutIncludeWhenVersionLowerThan11() {
97          PostgreSQLDDLTemplateExecutor templateExecutor = mock(PostgreSQLDDLTemplateExecutor.class);
98          when(templateExecutor.getMajorVersion()).thenReturn(10);
99          when(templateExecutor.executeByTemplate(anyMap(), anyString())).thenAnswer(invocation -> mockExecuteByTemplateWithoutInclude(invocation.getArgument(1)));
100         Map<String, Object> context = createContext(false);
101         createAppender(templateExecutor).append(context);
102         Collection<Map<String, Object>> primaryKeys = (Collection<Map<String, Object>>) context.get("primary_key");
103         assertThat(primaryKeys.size(), is(1));
104         assertThat(primaryKeys.iterator().next().get("include"), is(Collections.emptyList()));
105         Collection<Map<String, Object>> uniqueConstraints = (Collection<Map<String, Object>>) context.get("unique_constraint");
106         assertThat(uniqueConstraints.size(), is(1));
107         Collection<Map<String, Object>> exclusionConstraints = (Collection<Map<String, Object>>) context.get("exclude_constraint");
108         assertThat(exclusionConstraints.size(), is(1));
109         assertThat(exclusionConstraints.iterator().next().get("include"), is(Collections.emptyList()));
110     }

135     private Collection<Map<String, Object>> mockExecuteByTemplateFullCoverage(final Map<String, Object> params, final String path) {
136         switch (path) {
137             case "component/index_constraint/%s/properties.ftl":
138                 return createIndexConstraintsProps();
139             case "component/index_constraint/%s/get_costraint_cols.ftl":
140                 return Arrays.asList(Collections.singletonMap("column", "\"id\""), Collections.singletonMap("column", "name"));
141             case "component/index_constraint/%s/get_constraint_include.ftl":
142                 return Collections.singletonList(Collections.singletonMap("colname", "include_col"));
143             case "component/exclusion_constraint/%s/properties.ftl":
144                 return Collections.singletonList(createExclusionConstraintProps());
145             case "component/exclusion_constraint/%s/get_constraint_cols.ftl":
146                 return createExclusionConstraintColumns();
147             case "exclusion_constraint/%s/get_constraint_include.ftl":
148                 return Collections.singletonList(Collections.singletonMap("colname", "excl_inc"));
149             case "component/foreign_key/%s/properties.ftl":
150                 return createForeignKeyProps();
151             case "component/foreign_key/%s/get_constraint_cols.ftl":
152                 return createForeignKeyColumns(params);
153             case "component/foreign_key/%s/get_constraints.ftl":
154                 int callCount = constraintsCallCounter.incrementAndGet();
155                 if (1 == callCount) {
156                     return Arrays.asList(createCoveringIndex(7L, "idx_other"), createCoveringIndex(6L, "idx_fk_col1"));
157                 }
158                 if (2 == callCount) {
159                     return Collections.singletonList(createCoveringIndex(8L, "idx_not_match"));
160                 }
161                 return Collections.emptyList();
162             case "component/foreign_key/%s/get_cols.ftl":
163                 return createCoveringIndexColumns(params);
164             case "component/check_constraint/%s/properties.ftl":
165                 return Arrays.asList(Collections.singletonMap("conislocal", true), Collections.singletonMap("conislocal", false));
166             default:
167                 return Collections.emptyList();
168         }
169     }

233     private Collection<Map<String, Object>> createForeignKeyColumns(final Map<String, Object> params) {
234         Collection<Map<String, Object>> keys = (Collection<Map<String, Object>>) params.get("keys");
235         String confKey = keys.iterator().next().get("confkey").toString();
236         Map<String, Object> column = new HashMap<>(2, 1F);
237         if ("1".equals(confKey)) {
238             column.put("conattname", "fk_col1");
239             column.put("confattname", "ref_col1");
240         } else if ("3".equals(confKey)) {
241             column.put("conattname", "fk_col3");
242             column.put("confattname", "ref_col3");
243         } else {
244             column.put("conattname", "fk_col2");
245             column.put("confattname", "ref_col2");
246         }
247         return Collections.singleton(column);
248     }
