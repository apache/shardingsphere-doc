37  /**
38   * Integration tests for shorthand projection (SELECT T.*) with HintManager on sharded table t_order.
39   *
40   * <p>These tests reproduce issue #28841: when HintManager routes to a single datasource,
41   * SQL bind is skipped and shorthand projections remain unexpanded, causing column count
42   * and label mismatches in the ResultSet.</p>
43   */
44  final class ShardingSphereResultSetShorthandProjectionIT {
45      
46      private static final String SHARDING_JDBC_URL = "jdbc:shardingsphere:classpath:config/driver/driver-fixture-h2-mysql.yaml";
47      
48      private static final String H2_DS_0_URL = "jdbc:h2:mem:mysql_ds_0;DB_CLOSE_DELAY=-1;DATABASE_TO_UPPER=false;MODE=MySQL";
49      
50      private static final String H2_DS_1_URL = "jdbc:h2:mem:mysql_ds_1;DB_CLOSE_DELAY=-1;DATABASE_TO_UPPER=false;MODE=MySQL";
51      
52      private static Connection shardingSphereConnection;
53      
54      @BeforeAll
55      static void setUp() throws SQLException {
56          shardingSphereConnection = DriverManager.getConnection(SHARDING_JDBC_URL);
57          initDataSource(H2_DS_0_URL);
58          initDataSource(H2_DS_1_URL);
59      }
60      
61      private static void initDataSource(final String url) throws SQLException {
62          try (Connection connection = DriverManager.getConnection(url, "sa", ""); Statement statement = connection.createStatement()) {
63              statement.execute("DROP TABLE IF EXISTS t_order");
64              statement.execute("CREATE TABLE t_order (order_id INT PRIMARY KEY, user_id INT, status VARCHAR(50))");
65              statement.execute("INSERT INTO t_order VALUES (1000, 10, 'ACTIVE')");
66              statement.execute("INSERT INTO t_order VALUES (1001, 11, 'PENDING')");
67              statement.execute("DROP TABLE IF EXISTS t_order_item");
68              statement.execute("CREATE TABLE t_order_item (item_id INT PRIMARY KEY, order_id INT, item_name VARCHAR(50))");
69              statement.execute("INSERT INTO t_order_item VALUES (1, 1000, 'ITEM_A')");
70              statement.execute("INSERT INTO t_order_item VALUES (2, 1001, 'ITEM_B')");
71          }
72      }
73      
74      @AfterAll
75      static void tearDown() throws SQLException {
76          if (null != shardingSphereConnection) {
77              shardingSphereConnection.close();
78          }
79      }
80      
81      /**
82       * Case 1: SELECT T.*, T.status status_new FROM t_order T WHERE T.order_id = 1000
83       * Expected columns: order_id, user_id, status, status_new (4 columns).
84       */
85      @Test
86      void assertShorthandWithAlias() throws SQLException {
87          try (HintManager hintManager = HintManager.getInstance()) {
88              hintManager.setDataSourceName("ds_0");
89              try (
90                      Statement statement = shardingSphereConnection.createStatement();
91                      ResultSet resultSet = statement.executeQuery("SELECT T.*, T.status status_new FROM t_order T WHERE T.order_id = 1000")) {
92                  ResultSetMetaData actualMetaData = resultSet.getMetaData();
93                  assertThat("column count should be 4", actualMetaData.getColumnCount(), is(4));
94                  assertThat(actualMetaData.getColumnLabel(1), is("order_id"));
95                  assertThat(actualMetaData.getColumnLabel(2), is("user_id"));
96                  assertThat(actualMetaData.getColumnLabel(3), is("status"));
97                  assertThat(actualMetaData.getColumnLabel(4), is("status_new"));
98                  assertTrue(resultSet.next());
99                  assertThat(resultSet.getString("order_id"), is("1000"));
100                 assertThat(resultSet.getString("status_new"), is("ACTIVE"));
101                 assertFalse(resultSet.next());
102             }
103         }
104     }

189     /**
190      * Case 6: SELECT T.*, T.status status_new FROM t_order T ORDER BY T.user_id
191      * Shorthand + ORDER BY derived column coexisting. The shorthand remains unexpanded,
192      * while ProjectionsContextEngine appends ORDER_BY_DERIVED_0 for the absent order-by item.
193      * Expected: 4 visible columns (order_id, user_id, status, status_new); ORDER_BY_DERIVED hidden.
194      */
195     @Test
196     void assertShorthandWithOrderByDerived() throws SQLException {
197         try (HintManager hintManager = HintManager.getInstance()) {
198             hintManager.setDataSourceName("ds_0");
199             String sql = "SELECT T.*, T.status status_new FROM t_order T ORDER BY T.user_id";
200             try (Statement statement = shardingSphereConnection.createStatement(); ResultSet resultSet = statement.executeQuery(sql)) {
201                 ResultSetMetaData actualMetaData = resultSet.getMetaData();
202                 assertThat("shorthand columns visible, ORDER_BY_DERIVED hidden", actualMetaData.getColumnCount(), is(4));
203                 assertThat(actualMetaData.getColumnLabel(1), is("order_id"));
204                 assertThat(actualMetaData.getColumnLabel(2), is("user_id"));
205                 assertThat(actualMetaData.getColumnLabel(3), is("status"));
206                 assertThat(actualMetaData.getColumnLabel(4), is("status_new"));
207                 assertTrue(resultSet.next());
208                 assertThat(resultSet.getString("order_id"), is("1000"));
209                 assertThat(resultSet.getString("status_new"), is("ACTIVE"));
210             }
211         }
212     }

234     /**
235      * Case 8: SELECT o.order_id, i.* FROM t_order o JOIN t_order_item i ON o.order_id = i.order_id WHERE o.order_id = 1000
236      * Reproduces the original issue #28841: explicit column + shorthand from another table in a JOIN.
237      * Expected columns: order_id, item_id, order_id, item_name (4 columns).
238      */
239     @Test
240     void assertJoinWithShorthandProjection() throws SQLException {
241         try (HintManager hintManager = HintManager.getInstance()) {
242             hintManager.setDataSourceName("ds_0");
243             try (
244                     Statement statement = shardingSphereConnection.createStatement();
245                     ResultSet resultSet = statement.executeQuery("SELECT o.order_id, i.* FROM t_order o JOIN t_order_item i ON o.order_id = i.order_id WHERE o.order_id = 1000")) {
246                 ResultSetMetaData actualMetaData = resultSet.getMetaData();
247                 assertThat("explicit column + shorthand from joined table", actualMetaData.getColumnCount(), is(4));
248                 assertTrue(resultSet.next());
249                 assertThat(resultSet.getString("item_name"), is("ITEM_A"));
250                 assertFalse(resultSet.next());
251             }
252         }
