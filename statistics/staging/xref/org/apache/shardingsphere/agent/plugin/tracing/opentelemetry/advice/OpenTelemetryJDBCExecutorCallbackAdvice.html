22  import io.opentelemetry.api.trace.SpanBuilder;
23  import io.opentelemetry.api.trace.StatusCode;
24  import io.opentelemetry.api.trace.Tracer;
25  import io.opentelemetry.context.Context;
26  import lombok.Getter;
27  import lombok.RequiredArgsConstructor;
28  import lombok.Setter;
29  import org.apache.shardingsphere.agent.api.advice.TargetAdviceMethod;
30  import org.apache.shardingsphere.agent.api.advice.TargetAdviceObject;
31  import org.apache.shardingsphere.agent.plugin.tracing.core.advice.TracingJDBCExecutorCallbackAdvice;
32  import org.apache.shardingsphere.agent.plugin.tracing.core.constant.AttributeConstants;
33  import org.apache.shardingsphere.agent.plugin.tracing.opentelemetry.constant.OpenTelemetryConstants;
34  import org.apache.shardingsphere.database.connector.core.jdbcurl.parser.ConnectionProperties;
35  import org.apache.shardingsphere.database.connector.core.type.DatabaseType;
36  import org.apache.shardingsphere.infra.annotation.HighFrequencyInvocation;
37  import org.apache.shardingsphere.infra.executor.sql.execute.engine.driver.jdbc.JDBCExecutionUnit;
38  
39  import java.util.Map;
40  import java.util.Map.Entry;
41  import java.util.concurrent.ConcurrentHashMap;
42  
43  /**
44   * OpenTelemetry JDBC executor callback advice executor.
45   */
46  public final class OpenTelemetryJDBCExecutorCallbackAdvice extends TracingJDBCExecutorCallbackAdvice<Span> {
47      
48      /**
49       * Spans that have been started but not completed yet, keyed by the execution unit of their own invocation.
50       */
51      private static final Map<JDBCExecutionUnit, PendingSpan> PENDING_SPANS = new ConcurrentHashMap<>();
52      
53      /**
54       * Number of spans that the current thread has started but not completed yet.
55       */
56      private static final ThreadLocal<Integer> PENDING_SPAN_COUNT = new ThreadLocal<>();
57      
58      /**
59       * Keeps the completion hooks of a pending invocation runnable while the plugin is disabled.
60       *
61       * @return true when the plugin is enabled or the current thread owns a pending span
62       */
63      @Override
64      @HighFrequencyInvocation
65      public boolean isPluginEnabled() {
66          return super.isPluginEnabled() || getPendingSpanCount() > 0;
67      }
68      
69      private static int getPendingSpanCount() {
70          Integer result = PENDING_SPAN_COUNT.get();
71          return null == result ? 0 : result;
72      }
73      
74      @Override
75      protected void recordExecuteInfo(final Span parentSpan, final TargetAdviceObject target, final JDBCExecutionUnit executionUnit, final boolean isTrunkThread,
76                                       final ConnectionProperties connectionProps, final DatabaseType databaseType) {
77          if (!super.isPluginEnabled()) {
78              return;
79          }
80          Tracer tracer = GlobalOpenTelemetry.getTracer(OpenTelemetryConstants.TRACER_NAME);
81          SpanBuilder spanBuilder = tracer.spanBuilder(OPERATION_NAME);
82          if (null != parentSpan) {
83              spanBuilder.setParent(Context.current().with(parentSpan));
84          }
85          spanBuilder.setAttribute(AttributeConstants.COMPONENT, AttributeConstants.COMPONENT_NAME);
86          spanBuilder.setAttribute(AttributeConstants.DB_TYPE, databaseType.getType());
87          spanBuilder.setAttribute(AttributeConstants.DB_INSTANCE, executionUnit.getExecutionUnit().getDataSourceName())
88                  .setAttribute(AttributeConstants.PEER_HOSTNAME, connectionProps.getHostname())
89                  .setAttribute(AttributeConstants.PEER_PORT, String.valueOf(connectionProps.getPort()))
90                  .setAttribute(AttributeConstants.DB_STATEMENT, executionUnit.getExecutionUnit().getSqlUnit().getSql())
91                  .setAttribute(AttributeConstants.DB_BIND_VARIABLES, executionUnit.getExecutionUnit().getSqlUnit().getParameters().toString())
92                  .setAttribute(AttributeConstants.SPAN_KIND, AttributeConstants.SPAN_KIND_CLIENT);
93          PENDING_SPANS.put(executionUnit, new PendingSpan(spanBuilder.startSpan()));
94          PENDING_SPAN_COUNT.set(getPendingSpanCount() + 1);
95      }
96      
97      @Override
98      @HighFrequencyInvocation
99      public void afterMethod(final TargetAdviceObject target, final TargetAdviceMethod method, final Object[] args, final Object result, final String pluginType) {
100         PendingSpan pending = PENDING_SPANS.remove((JDBCExecutionUnit) args[0]);
101         if (null != pending) {
102             decreasePendingSpanCount();
103             endSpan(pending);
104         }
105     }
106     
107     private static void decreasePendingSpanCount() {
108         int count = getPendingSpanCount() - 1;
109         if (count > 0) {
110             PENDING_SPAN_COUNT.set(count);
111         } else {
112             PENDING_SPAN_COUNT.remove();
113         }
114     }
115     
116     private static void endSpan(final PendingSpan pending) {
117         Span span = pending.getSpan();
118         if (null != pending.getThrowable()) {
119             span.setStatus(StatusCode.ERROR).recordException(pending.getThrowable());
120         } else {
121             span.setStatus(StatusCode.OK);
122         }
123         span.end();
124     }
125     
126     @Override
127     public void onThrowing(final TargetAdviceObject target, final TargetAdviceMethod method, final Object[] args, final Throwable throwable, final String pluginType) {
128         PendingSpan pending = PENDING_SPANS.get((JDBCExecutionUnit) args[0]);
129         if (null != pending) {
130             pending.setThrowable(throwable);
131         }
132     }
133     
134     /**
135      * Ends and releases every span that is still pending, and clears the caller thread's pending span count.
136      */
137     public static void endAllPendingSpans() {
138         for (Entry<JDBCExecutionUnit, PendingSpan> entry : PENDING_SPANS.entrySet()) {
139             PendingSpan pending = PENDING_SPANS.remove(entry.getKey());
140             if (null != pending) {
141                 endSpan(pending);
142             }
143         }
144         PENDING_SPAN_COUNT.remove();
145     }
146     
147     /**
148      * A span started for one invocation, together with the throwable recorded by a preceding {@code onThrowing} call.
149      */
150     @RequiredArgsConstructor
151     @Getter
152     private static final class PendingSpan {
153         
154         private final Span span;
155         
156         @Setter
157         private Throwable throwable;
158     }
159 }
