20  import lombok.AccessLevel;
21  import lombok.NoArgsConstructor;
22  import org.apache.shardingsphere.infra.merge.result.impl.local.LocalDataQueryResultRow;
23  
24  import javax.sql.rowset.CachedRowSet;
25  import javax.sql.rowset.RowSetMetaDataImpl;
26  import javax.sql.rowset.RowSetProvider;
27  import java.lang.reflect.InvocationHandler;
28  import java.lang.reflect.InvocationTargetException;
29  import java.lang.reflect.Method;
30  import java.lang.reflect.Proxy;
31  import java.sql.ResultSet;
32  import java.sql.ResultSetMetaData;
33  import java.sql.SQLException;
34  import java.sql.Statement;
35  import java.sql.Types;
36  import java.util.ArrayList;
37  import java.util.Collection;
38  import java.util.List;
39  
40  /**
41   * Creates JDBC result sets for DistSQL local rows.
42   */
43  @NoArgsConstructor(access = AccessLevel.PRIVATE)
44  public final class DistSQLResultSet {
45      
46      /**
47       * Create a result set from DistSQL rows.
48       *
49       * @param columnNames column names
50       * @param rows local rows
51       * @param statement statement that produced the result set
52       * @return JDBC result set
53       * @throws SQLException SQL exception
54       */
55      public static ResultSet create(final Collection<String> columnNames, final Collection<LocalDataQueryResultRow> rows, final Statement statement) throws SQLException {
56          RowSetMetaDataImpl metaData = new RowSetMetaDataImpl();
57          metaData.setColumnCount(columnNames.size());
58          int columnIndex = 1;
59          for (String each : columnNames) {
60              metaData.setColumnName(columnIndex, each);
61              metaData.setColumnLabel(columnIndex, each);
62              metaData.setColumnType(columnIndex, Types.CHAR);
63              metaData.setColumnTypeName(columnIndex, "CHAR");
64              metaData.setPrecision(columnIndex, 255);
65              metaData.setNullable(columnIndex, ResultSetMetaData.columnNullable);
66              columnIndex++;
67          }
68          CachedRowSet result = RowSetProvider.newFactory().createCachedRowSet();
69          result.setMetaData(metaData);
70          List<LocalDataQueryResultRow> orderedRows = new ArrayList<>(rows);
71          for (int rowIndex = orderedRows.size() - 1; rowIndex >= 0; rowIndex--) {
72              result.moveToInsertRow();
73              for (int i = 1; i <= columnNames.size(); i++) {
74                  result.updateObject(i, orderedRows.get(rowIndex).getCell(i));
75              }
76              result.insertRow();
77              result.moveToCurrentRow();
78          }
79          result.beforeFirst();
80          result.setType(statement.getResultSetType());
81          result.setConcurrency(statement.getResultSetConcurrency());
82          return (ResultSet) Proxy.newProxyInstance(DistSQLResultSet.class.getClassLoader(), new Class<?>[]{ResultSet.class}, new LifecycleHandler(result, statement));
83      }
84      
85      private static final class LifecycleHandler implements InvocationHandler {
86          
87          private final CachedRowSet delegate;
88          
89          private final Statement statement;
90          
91          private boolean closed;
92          
93          private LifecycleHandler(final CachedRowSet delegate, final Statement statement) {
94              this.delegate = delegate;
95              this.statement = statement;
96          }
97          
98          @Override
99          public Object invoke(final Object proxy, final Method method, final Object[] args) throws Throwable {
100             if ("equals".equals(method.getName())) {
101                 return proxy == args[0];
102             }
103             if ("hashCode".equals(method.getName())) {
104                 return System.identityHashCode(proxy);
105             }
106             if ("toString".equals(method.getName())) {
107                 return "DistSQLResultSet";
108             }
109             if ("isClosed".equals(method.getName())) {
110                 return closed;
111             }
112             if ("close".equals(method.getName())) {
113                 closed = true;
114                 delegate.close();
115                 return null;
116             }
117             if (closed) {
118                 throw new SQLException("Result set is closed.");
119             }
120             if ("getStatement".equals(method.getName())) {
121                 return statement;
122             }
123             try {
124                 return method.invoke(delegate, args);
125             } catch (final InvocationTargetException ex) {
126                 throw ex.getCause();
127             }
128         }
129     }
130 }
