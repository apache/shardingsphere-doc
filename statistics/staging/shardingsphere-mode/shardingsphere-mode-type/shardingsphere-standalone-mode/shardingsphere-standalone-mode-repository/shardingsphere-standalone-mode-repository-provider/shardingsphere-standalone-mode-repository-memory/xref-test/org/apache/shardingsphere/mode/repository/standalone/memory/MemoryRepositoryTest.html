85      @ParameterizedTest(name = "{0}, parentExists={1}")
86      @CsvSource({"/metadata/foo_db/schemas/foo_db/tables/foo_tbl,true", "/metadata/foo_db/schemas/foo_db/tables/foo_tbl,false", "/metadata/foo_db/schemas/foo_db/tables/foo_tbl/,true"})
87      void assertDeleteSubtree(final String key, final boolean parentExists) {
88          if (parentExists) {
89              memoryRepository.persist(key, "foo_tbl");
90          }
91          String tableKey = "/metadata/foo_db/schemas/foo_db/tables/foo_tbl";
92          memoryRepository.persist(tableKey + "/active_version", "1");
93          memoryRepository.persist(tableKey + "/versions/0", "foo_version_0");
94          memoryRepository.persist(tableKey + "/versions/1", "foo_version_1");
95          memoryRepository.persist(tableKey + "_extra/versions/0", "foo_sibling");
96          memoryRepository.delete(key);
97          assertNull(memoryRepository.query(key));
98          assertNull(memoryRepository.query(tableKey + "/active_version"));
99          assertNull(memoryRepository.query(tableKey + "/versions/0"));
100         assertNull(memoryRepository.query(tableKey + "/versions/1"));
101         assertTrue(memoryRepository.getChildrenKeys(tableKey + "/").isEmpty());
102         assertThat(memoryRepository.getChildrenKeys("/metadata/foo_db/schemas/foo_db/tables"), is(Collections.singletonList("foo_tbl_extra")));
103         assertThat(memoryRepository.query(tableKey + "_extra/versions/0"), is("foo_sibling"));
104     }

114     @Test
115     void assertConcurrentDelete() throws ReflectiveOperationException, InterruptedException, ExecutionException {
116         SortedMap<String, String> metaDataSortedMap = spy(new TreeMap<>());
117         metaDataSortedMap.put("/foo/child", "foo_child");
118         metaDataSortedMap.put("/bar", "bar_parent");
119         metaDataSortedMap.put("/bar/child", "bar_child");
120         metaDataSortedMap.put("/foo_extra/child", "foo_sibling");
121         Plugins.getMemberAccessor().set(MemoryRepository.class.getDeclaredField("metaDataSortedMap"), memoryRepository, metaDataSortedMap);
122         SortedMap<String, String> subtree = metaDataSortedMap.tailMap("/foo/");
123         SortedMap<?, ?> mockedSubtree = mock(SortedMap.class, RETURNS_DEEP_STUBS);
124         CountDownLatch deleteStarted = new CountDownLatch(1);
125         CountDownLatch continueDelete = new CountDownLatch(1);
126         when(mockedSubtree.keySet().iterator()).thenAnswer(invocation -> {
127             Iterator<String> result = subtree.keySet().iterator();
128             deleteStarted.countDown();
129             assertTrue(continueDelete.await(10, TimeUnit.SECONDS));
130             return result;
131         });
132         doReturn(mockedSubtree).when(metaDataSortedMap).tailMap("/foo/");
133         FutureTask<Void> deleteTask = new FutureTask<>(() -> memoryRepository.delete("/foo"), null);
134         Thread deleteThread = new Thread(deleteTask);
135         FutureTask<Void> anotherDeleteTask = new FutureTask<>(() -> memoryRepository.delete("/bar"), null);
136         Thread anotherDeleteThread = new Thread(anotherDeleteTask);
137         ThreadMXBean threadMXBean = ManagementFactory.getThreadMXBean();
138         try {
139             deleteThread.start();
140             assertTrue(deleteStarted.await(5, TimeUnit.SECONDS));
141             anotherDeleteThread.start();
142             assertTimeoutPreemptively(Duration.ofSeconds(5), () -> {
143                 while (!anotherDeleteTask.isDone()) {
144                     ThreadInfo threadInfo = threadMXBean.getThreadInfo(anotherDeleteThread.getId());
145                     if (null != threadInfo && deleteThread.getId() == threadInfo.getLockOwnerId()) {
146                         break;
147                     }
148                     Thread.yield();
149                 }
150             });
151         } finally {
152             continueDelete.countDown();
153             deleteThread.join(5000L);
154             anotherDeleteThread.join(5000L);
155         }
156         assertFalse(deleteThread.isAlive());
157         assertFalse(anotherDeleteThread.isAlive());
158         deleteTask.get();
159         anotherDeleteTask.get();
160         assertTrue(memoryRepository.getChildrenKeys("/foo/").isEmpty());
161         assertNull(memoryRepository.query("/bar"));
162         assertTrue(memoryRepository.getChildrenKeys("/bar/").isEmpty());
163         assertThat(memoryRepository.query("/foo_extra/child"), is("foo_sibling"));
164     }

172     }
