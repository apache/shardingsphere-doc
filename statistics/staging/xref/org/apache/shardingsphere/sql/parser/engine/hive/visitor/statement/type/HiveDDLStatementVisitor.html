123     @Override
124     public ASTNode visitAlterTable(final AlterTableContext ctx) {
125         AlterTableStatement result = new AlterTableStatement(getDatabaseType());
126         result.setTable((SimpleTableSegment) visit(ctx.alterTableCommonClause().tableName()));
127         if (null != ctx.changeColumn()) {
128             ChangeColumnDefinitionSegment changeColumnSegment = (ChangeColumnDefinitionSegment) visit(ctx.changeColumn());
129             result.getChangeColumnDefinitions().add(changeColumnSegment);
130         }
131         if (null != ctx.addColumns()) {
132             AddColumnDefinitionSegment addSeg = (AddColumnDefinitionSegment) visit(ctx.addColumns());
133             result.getAddColumnDefinitions().add(addSeg);
134         }
135         if (null != ctx.replaceColumns()) {
136             ReplaceColumnDefinitionSegment repSeg = (ReplaceColumnDefinitionSegment) visit(ctx.replaceColumns());
137             result.getReplaceColumnDefinitions().add(repSeg);
138         }
139         if (null != ctx.COMPACT()) {
140             String compactionType = ctx.string_().getText().replace("'", "");
141             if (!isValidCompactionType(compactionType)) {
142                 throw new IllegalArgumentException("Invalid compaction type. Must be 'MAJOR', 'MINOR' or 'REBALANCE'");
143             }
144             if ((null != ctx.clusteredIntoClause() || null != ctx.orderByClause())
145                     && !"REBALANCE".equalsIgnoreCase(compactionType)) {
146                 throw new IllegalArgumentException("[CLUSTERED INTO n BUCKETS] and [ORDER BY col_list] clauses can only be used with REBALANCE compaction");
147             }
148         }
149         return result;
150     }
151     
152     private boolean isValidCompactionType(final String compactionType) {
153         return "MAJOR".equalsIgnoreCase(compactionType)
154                 || "MINOR".equalsIgnoreCase(compactionType)
155                 || "REBALANCE".equalsIgnoreCase(compactionType);
156     }

178     @SuppressWarnings("unchecked")
179     @Override
180     public ASTNode visitCreateTable(final CreateTableContext ctx) {
181         CreateTableStatement result = new CreateTableStatement(getDatabaseType());
182         result.setTable((SimpleTableSegment) visit(ctx.createTableCommonClause().tableNameWithDb()));
183         result.setIfNotExists(null != ctx.createTableCommonClause().ifNotExists());
184         if (null != ctx.createDefinitionClause()) {
185             CollectionValue<CreateDefinitionSegment> createDefinitions = (CollectionValue<CreateDefinitionSegment>) visit(ctx.createDefinitionClause());
186             for (CreateDefinitionSegment each : createDefinitions.getValue()) {
187                 if (each instanceof ColumnDefinitionSegment) {
188                     result.getColumnDefinitions().add((ColumnDefinitionSegment) each);
189                 } else if (each instanceof ConstraintDefinitionSegment) {
190                     result.getConstraintDefinitions().add((ConstraintDefinitionSegment) each);
191                 }
192             }
193         }
194         return result;
195     }

197     @Override
198     public ASTNode visitCreateDefinitionClause(final CreateDefinitionClauseContext ctx) {
199         CollectionValue<CreateDefinitionSegment> result = new CollectionValue<>();
200         for (ColumnDefinitionContext each : ctx.columnDefinition()) {
201             result.getValue().add((ColumnDefinitionSegment) visit(each));
202         }
203         for (TableConstraintContext each : ctx.tableConstraint()) {
204             result.getValue().add((ConstraintDefinitionSegment) visit(each));
205         }
206         return result;
207     }

209     @Override
210     public ASTNode visitColumnDefinition(final ColumnDefinitionContext ctx) {
211         ColumnSegment column = new ColumnSegment(ctx.columnName().getStart().getStartIndex(), ctx.columnName().getStop().getStopIndex(),
212                 new IdentifierValue(ctx.columnName().getText()));
213         DataTypeSegment dataType = (DataTypeSegment) visit(ctx.dataTypeClause());
214         return new ColumnDefinitionSegment(ctx.getStart().getStartIndex(), ctx.getStop().getStopIndex(), column, dataType, false, false, getText(ctx));
215     }

217     @Override
218     public ASTNode visitChangeColumn(final ChangeColumnContext ctx) {
219         ColumnSegment oldColumn = new ColumnSegment(ctx.columnName(0).getStart().getStartIndex(), ctx.columnName(0).getStop().getStopIndex(),
220                 new IdentifierValue(ctx.columnName(0).getText()));
221         ColumnSegment newColumn = new ColumnSegment(ctx.columnName(1).getStart().getStartIndex(), ctx.columnName(1).getStop().getStopIndex(),
222                 new IdentifierValue(ctx.columnName(1).getText()));
223         DataTypeSegment dataType = (DataTypeSegment) visit(ctx.dataTypeClause());
224         ColumnDefinitionSegment columnDefinition = new ColumnDefinitionSegment(ctx.columnName(1).getStart().getStartIndex(),
225                 ctx.dataTypeClause().getStop().getStopIndex(), newColumn, dataType, false, false, getText(ctx));
226         ChangeColumnDefinitionSegment result = new ChangeColumnDefinitionSegment(ctx.getStart().getStartIndex(), ctx.getStop().getStopIndex(), columnDefinition);
227         result.setPreviousColumn(oldColumn);
228         if (null != ctx.FIRST()) {
229             ColumnFirstPositionSegment firstPos = new ColumnFirstPositionSegment(ctx.FIRST().getSymbol().getStartIndex(), ctx.FIRST().getSymbol().getStopIndex(), null);
230             result.setColumnPosition(firstPos);
231         } else if (null != ctx.AFTER()) {
232             ColumnNameContext afterCtx = ctx.columnName(ctx.columnName().size() - 1);
233             ColumnSegment afterColumn = new ColumnSegment(afterCtx.getStart().getStartIndex(), afterCtx.getStop().getStopIndex(),
234                     new IdentifierValue(afterCtx.getText()));
235             ColumnAfterPositionSegment afterPos = new ColumnAfterPositionSegment(afterCtx.getStart().getStartIndex(), afterCtx.getStop().getStopIndex(), afterColumn);
236             result.setColumnPosition(afterPos);
237         }
238         return result;
239     }

243         java.util.Collection<ColumnDefinitionSegment> cols = new java.util.LinkedList<>();
244         for (ColumnDefinitionContext each : ctx.columnDefinition()) {

246         }
247         return new AddColumnDefinitionSegment(ctx.getStart().getStartIndex(), ctx.getStop().getStopIndex(), cols);
248     }

251     public ASTNode visitReplaceColumns(final ReplaceColumnsContext ctx) {
252         java.util.Collection<ColumnDefinitionSegment> cols = new java.util.LinkedList<>();
253         for (ColumnDefinitionContext each : ctx.columnDefinition()) {
254             cols.add((ColumnDefinitionSegment) visit(each));
255         }
256         return new ReplaceColumnDefinitionSegment(ctx.getStart().getStartIndex(), ctx.getStop().getStopIndex(), cols);
257     }

259     private String getText(final ParserRuleContext ctx) {
260         return ctx.start.getInputStream().getText(new Interval(ctx.start.getStartIndex(), ctx.stop.getStopIndex()));
261     }

263     @Override
264     public ASTNode visitDataTypeClause(final DataTypeClauseContext ctx) {
265         DataTypeSegment result = new DataTypeSegment();
266         result.setStartIndex(ctx.getStart().getStartIndex());
267         result.setStopIndex(ctx.getStop().getStopIndex());
268         if (null != ctx.primitiveType()) {
269             result.setDataTypeName(ctx.primitiveType().getText());
270         } else if (null != ctx.arrayType()) {
271             result.setDataTypeName(ctx.arrayType().getText());
272         } else if (null != ctx.mapType()) {
273             result.setDataTypeName(ctx.mapType().getText());
274         } else if (null != ctx.structType()) {
275             result.setDataTypeName(ctx.structType().getText());
276         } else if (null != ctx.unionType()) {
277             result.setDataTypeName(ctx.unionType().getText());
278         }
279         return result;
280     }

282     @Override
283     public ASTNode visitTableNameWithDb(final TableNameWithDbContext ctx) {
284         if (1 == ctx.identifier().size()) {
285             return new SimpleTableSegment(new TableNameSegment(ctx.getStart().getStartIndex(), ctx.getStop().getStopIndex(),
286                     new IdentifierValue(ctx.identifier(0).getText())));
287         } else {
288             SimpleTableSegment result = new SimpleTableSegment(new TableNameSegment(ctx.identifier(1).getStart().getStartIndex(),
289                     ctx.identifier(1).getStop().getStopIndex(), new IdentifierValue(ctx.identifier(1).getText())));
290             result.setOwner(new org.apache.shardingsphere.sql.parser.statement.core.segment.generic.OwnerSegment(
291                     ctx.identifier(0).getStart().getStartIndex(), ctx.identifier(0).getStop().getStopIndex(),
292                     new IdentifierValue(ctx.identifier(0).getText())));
293             return result;
294         }
295     }

302     @Override
303     public ASTNode visitCreateView(final CreateViewContext ctx) {
304         CreateViewStatement result = new CreateViewStatement(getDatabaseType());
305         result.setView((SimpleTableSegment) visit(ctx.viewNameWithDb()));
306         if (null != ctx.commentClause() && !ctx.commentClause().isEmpty()) {
307             result.setViewDefinition(ctx.commentClause(ctx.commentClause().size() - 1).string_().getText().replace("'", ""));
308         }
309         if (null != ctx.tblProperties()) {
310             result.setViewDefinition(getText(ctx.tblProperties()));
311         }
312         HiveDMLStatementVisitor dmlVisitor = new HiveDMLStatementVisitor(getDatabaseType());
313         ASTNode selectNode = dmlVisitor.visit(ctx.select());
314         if (selectNode instanceof SelectStatement) {
315             result.setSelect((SelectStatement) selectNode);
316         }
317         result.setViewDefinition(getText(ctx));
318         return result;
319     }

346         AlterViewStatement result = new AlterViewStatement(getDatabaseType());

348         if (null != ctx.select()) {
349             HiveDMLStatementVisitor dmlVisitor = new HiveDMLStatementVisitor(getDatabaseType());

354             result.setViewDefinition(getText(ctx.select()));
355         }
356         return result;
