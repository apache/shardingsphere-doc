102     private DataSource mockDataSource(final String dataSourceName, final String schemaName, final String url, final List<String> tableNames) throws SQLException {
103         Connection connection = mock(Connection.class, RETURNS_DEEP_STUBS);
104         when(connection.getCatalog()).thenReturn(dataSourceName);
105         when(connection.getMetaData().getURL()).thenReturn(url);
106         if (null == schemaName) {
107             ResultSet tableResultSet = mockResultSet(tableNames);
108             when(connection.getMetaData().getTables(dataSourceName, null, null, new String[]{"TABLE", "PARTITIONED TABLE", "VIEW", "SYSTEM TABLE", "SYSTEM VIEW"}))
109                     .thenReturn(tableResultSet);
110         } else {
111             ResultSet schemaResultSet = mockSchemaResultSet(schemaName);
112             ResultSet tableResultSet = mockResultSet(tableNames);
113             when(connection.getMetaData().getSchemas()).thenReturn(schemaResultSet);
114             when(connection.getMetaData().getTables(dataSourceName, schemaName, null, new String[]{"TABLE", "PARTITIONED TABLE", "VIEW", "SYSTEM TABLE", "SYSTEM VIEW"}))
115                     .thenReturn(tableResultSet);
116         }
117         return new MockedDataSource(connection);
118     }

120     private ResultSet mockResultSet(final List<String> tableNames) throws SQLException {
121         ResultSet result = mock(ResultSet.class);
122         String firstTableName = tableNames.get(0);
123         Collection<String> remainTableNames = tableNames.subList(1, tableNames.size());
124         Collection<Boolean> remainNextResults = remainTableNames.stream().map(each -> true).collect(Collectors.toList());
125         remainNextResults.add(false);
126         when(result.next()).thenReturn(true, remainNextResults.toArray(new Boolean[tableNames.size()]));
127         when(result.getString("TABLE_NAME")).thenReturn(firstTableName, remainTableNames.toArray(new String[tableNames.size() - 1]));
128         return result;
129     }

132         ResultSet result = mock(ResultSet.class);
133         when(result.next()).thenReturn(true, false);
134         when(result.getString("TABLE_SCHEM")).thenReturn(schemaName);
135         return result;
136     }
