177     @Test
178     void assertForeignKeyColumnMappingQuery() {
179         String actual = PostgreSQLPipelineFreemarkerManager.getSQLByVersion(Collections.singletonMap("cid", 12345L), "component/foreign_key/%s/get_constraint_cols.ftl", 17, 0);
180         String expected = "SELECT a1.attname as conattname, a2.attname as confattname FROM ( SELECT conrelid, confrelid, conkey, confkey, generate_subscripts(conkey, 1) AS ordinal"
181                 + " FROM pg_catalog.pg_constraint WHERE oid = 12345::oid ) mapping"
182                 + " JOIN pg_catalog.pg_attribute a1 ON a1.attrelid = mapping.conrelid AND a1.attnum = mapping.conkey[mapping.ordinal]"
183                 + " JOIN pg_catalog.pg_attribute a2 ON a2.attrelid = mapping.confrelid AND a2.attnum = mapping.confkey[mapping.ordinal] ORDER BY mapping.ordinal";
184         assertThat(PATTERN.matcher(actual).replaceAll(" ").trim(), is(expected));
185     }

187     @ParameterizedTest(name = "{0} index columns")
188     @ValueSource(ints = {1, 2, 3})
189     void assertForeignKeyCoveringIndexColumns(final int columnCount) {
190         String actual = PostgreSQLPipelineFreemarkerManager.getSQLByVersion(Map.of("cid", 12345L, "colcnt", columnCount), "component/foreign_key/%s/get_cols.ftl", 17, 0);
191         StringBuilder expected = new StringBuilder("SELECT pg_catalog.pg_get_indexdef(12345, 1, true) AS column");
192         for (int i = 2; i <= columnCount; i++) {
193             expected.append(" UNION SELECT pg_catalog.pg_get_indexdef(12345, ").append(i).append(", true) AS column");
194         }
195         assertThat(PATTERN.matcher(actual).replaceAll(""), is(PATTERN.matcher(expected).replaceAll("")));
196     }
