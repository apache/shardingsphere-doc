72      void assertBindInsertValues() {
73          ErrorLoggingSegment errorLogging = new ErrorLoggingSegment(0, 0, null, null, "UNLIMITED");
74          InsertStatement insertStatement = InsertStatement.builder().databaseType(databaseType)
75                  .table(new SimpleTableSegment(new TableNameSegment(0, 0, new IdentifierValue("t_order"))))
76                  .insertColumns(new InsertColumnsSegment(0, 0, Arrays.asList(new ColumnSegment(0, 0, new IdentifierValue("order_id")),
77                          new ColumnSegment(0, 0, new IdentifierValue("user_id")), new ColumnSegment(0, 0, new IdentifierValue("status")))))
78                  .values(Collections.singletonList(new InsertValuesSegment(0, 0, Arrays.asList(new LiteralExpressionSegment(0, 0, 1),
79                          new LiteralExpressionSegment(0, 0, 1), new LiteralExpressionSegment(0, 0, "OK")))))
80                  .overwrite(true)
81                  .errorLogging(errorLogging)
82                  .build();
83          InsertStatement actual = new InsertStatementBinder().bind(insertStatement, new SQLStatementBinderContext(createMetaData(), "foo_db", new HintValueContext(), insertStatement));
84          assertThat(actual, not(insertStatement));
85          assertTrue(actual.isOverwrite());
86          assertThat(actual.getErrorLogging().get(), is(errorLogging));
87          assertTrue(actual.getTable().isPresent());
88          assertTrue(insertStatement.getTable().isPresent());
89          assertThat(actual.getTable().get().getTableName(), not(insertStatement.getTable().get().getTableName()));
90          assertTrue(actual.getInsertColumns().isPresent());
91          assertInsertColumns(actual.getInsertColumns().get().getColumns());
92      }

184     @Test
185     void assertBindInsertValuesSubqueryExpression() {
186         ProjectionsSegment projections = new ProjectionsSegment(8, 14);
187         projections.getProjections().add(new ColumnProjectionSegment(new ColumnSegment(8, 14, new IdentifierValue("status"))));
188         ColumnSegment predicateColumn = new ColumnSegment(34, 41, new IdentifierValue("order_id"));
189         BinaryOperationExpression predicate = new BinaryOperationExpression(34, 48, predicateColumn, new LiteralExpressionSegment(45, 48, 1000), "=", "order_id = 1000");
190         SelectStatement subquerySelect = SelectStatement.builder().databaseType(databaseType).projections(projections)
191                 .from(new SimpleTableSegment(new TableNameSegment(22, 28, new IdentifierValue("t_order"))))
192                 .where(new WhereSegment(34, 48, predicate)).build();
193         SubqueryExpressionSegment subqueryExpression = new SubqueryExpressionSegment(new SubquerySegment(0, 49, subquerySelect, "(SELECT status FROM t_order WHERE order_id = 1000)"));
194         InsertStatement insertStatement = InsertStatement.builder().databaseType(databaseType)
195                 .table(new SimpleTableSegment(new TableNameSegment(0, 0, new IdentifierValue("t_order"))))
196                 .insertColumns(new InsertColumnsSegment(0, 0, Collections.singletonList(new ColumnSegment(0, 0, new IdentifierValue("status")))))
197                 .values(Collections.singletonList(new InsertValuesSegment(0, 49, Collections.singletonList(subqueryExpression)))).build();
198         InsertStatement actual = new InsertStatementBinder().bind(insertStatement, new SQLStatementBinderContext(createMetaData(), "foo_db", new HintValueContext(), insertStatement));
199         SubqueryExpressionSegment actualSubqueryExpression = (SubqueryExpressionSegment) actual.getValues().iterator().next().getValues().iterator().next();
200         ProjectionSegment projectionSegment = actualSubqueryExpression.getSubquery().getSelect().getProjections().getProjections().iterator().next();
201         assertThat(((ColumnProjectionSegment) projectionSegment).getColumn().getColumnBoundInfo().getOriginalTable().getValue(), is("t_order"));
202         BinaryOperationExpression actualPredicate = (BinaryOperationExpression) actualSubqueryExpression.getSubquery().getSelect().getWhere().get().getExpr();
203         assertThat(((ColumnSegment) actualPredicate.getLeft()).getColumnBoundInfo().getOriginalTable().getValue(), is("t_order"));
204     }

222     @Test
223     void assertBindInsertValuesWithOnDuplicateKeyColumnsAndExcludedValue() {
224         ColumnSegment excludedUserIdColumn = new ColumnSegment(0, 0, new IdentifierValue("user_id"));
225         excludedUserIdColumn.setOwner(new OwnerSegment(0, 0, new IdentifierValue("EXCLUDED")));
226         ColumnAssignmentSegment onDuplicateAssignment = new ColumnAssignmentSegment(0, 0, Collections.singletonList(new ColumnSegment(0, 0, new IdentifierValue("user_id"))), excludedUserIdColumn);
227         InsertStatement insertStatement = InsertStatement.builder().databaseType(databaseType)
228                 .table(new SimpleTableSegment(new TableNameSegment(0, 0, new IdentifierValue("t_order"))))
229                 .insertColumns(new InsertColumnsSegment(0, 0, Arrays.asList(new ColumnSegment(0, 0, new IdentifierValue("order_id")),
230                         new ColumnSegment(0, 0, new IdentifierValue("user_id")), new ColumnSegment(0, 0, new IdentifierValue("status")))))
231                 .values(Collections.singletonList(new InsertValuesSegment(0, 0, Arrays.asList(new LiteralExpressionSegment(0, 0, 1),
232                         new LiteralExpressionSegment(0, 0, 1), new LiteralExpressionSegment(0, 0, "OK")))))
233                 .onDuplicateKeyColumns(new OnDuplicateKeyColumnsSegment(0, 0, Collections.singletonList(onDuplicateAssignment))).build();
234         InsertStatement actual = new InsertStatementBinder().bind(insertStatement, new SQLStatementBinderContext(createMetaData(), "foo_db", new HintValueContext(), insertStatement));
235         assertTrue(actual.getOnDuplicateKeyColumns().isPresent());
236         ColumnAssignmentSegment actualAssignment = actual.getOnDuplicateKeyColumns().get().getColumns().iterator().next();
237         assertThat(actualAssignment.getColumns().iterator().next().getColumnBoundInfo().getOriginalTable().getValue(), is("t_order"));
238         assertThat(actualAssignment.getValue(), isA(ColumnSegment.class));
239         ColumnSegment actualValue = (ColumnSegment) actualAssignment.getValue();
240         assertThat(excludedUserIdColumn, is(actualValue));
241         assertTrue(actualValue.getOwner().isPresent());
242         assertThat(actualValue.getOwner().get().getIdentifier().getValue(), is("EXCLUDED"));
243         assertThat(actualValue.getIdentifier().getValue(), is("user_id"));
244     }

271     private ShardingSphereMetaData createMetaData() {
272         ShardingSphereSchema schema = mock(ShardingSphereSchema.class, RETURNS_DEEP_STUBS);
273         IdentifierValue fooDatabase = new IdentifierValue("foo_db");
274         IdentifierValue tOrder = new IdentifierValue("t_order");
275         when(schema.getTable("t_order").getAllColumns()).thenReturn(Arrays.asList(
276                 new ShardingSphereColumn("order_id", Types.INTEGER, true, false, false, true, false, false),
277                 new ShardingSphereColumn("user_id", Types.INTEGER, false, false, false, true, false, false),
278                 new ShardingSphereColumn("status", Types.INTEGER, false, false, false, true, false, false)));
279         when(schema.getTable(tOrder).getAllColumns()).thenReturn(Arrays.asList(
280                 new ShardingSphereColumn("order_id", Types.INTEGER, true, false, false, true, false, false),
281                 new ShardingSphereColumn("user_id", Types.INTEGER, false, false, false, true, false, false),
282                 new ShardingSphereColumn("status", Types.INTEGER, false, false, false, true, false, false)));
283         ShardingSphereMetaData result = mock(ShardingSphereMetaData.class, RETURNS_DEEP_STUBS);
284         when(result.getDatabase("foo_db").getSchema("foo_db")).thenReturn(schema);
285         when(result.getDatabase(fooDatabase).getSchema(fooDatabase)).thenReturn(schema);
286         when(result.containsDatabase("foo_db")).thenReturn(true);
287         when(result.containsDatabase(fooDatabase)).thenReturn(true);
288         when(result.getDatabase("foo_db").getDefaultSchemaName()).thenReturn("foo_db");
289         when(result.getDatabase(fooDatabase).getDefaultSchemaName()).thenReturn("foo_db");
290         when(result.getDatabase("foo_db").containsSchema("foo_db")).thenReturn(true);
291         when(result.getDatabase(fooDatabase).containsSchema(fooDatabase)).thenReturn(true);
292         when(result.getDatabase("foo_db").getSchema("foo_db").containsTable("t_order")).thenReturn(true);
293         when(result.getDatabase(fooDatabase).getSchema(fooDatabase).containsTable(tOrder)).thenReturn(true);
294         return result;
295     }
