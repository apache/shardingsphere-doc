88      @Test
89      void assertCreateWithPersistedSchemas() throws SQLException {
90          when(GlobalRulesBuilder.buildRules(anyCollection(), anyCollection(), any(ConfigurationProperties.class))).thenReturn(Collections.emptyList());
91          when(ShardingSphereStatisticsFactory.create(any(), any())).thenReturn(new ShardingSphereStatistics());
92          ComputeNodeInstanceContext instanceContext = mockComputeNodeInstanceContext(mock(ProxyInstanceMetaData.class));
93          when(ShardingSphereDatabasesFactory.create(anyMap(), anyMap(), any(ConfigurationProperties.class), eq(instanceContext))).thenReturn(
94                  Arrays.asList(createDatabase("with_units", Collections.emptyList()), createDatabase("without_units", Collections.emptyList())));
95          Map<String, DatabaseConfiguration> databaseConfigs = createDatabaseConfigsWithAndWithoutStorageUnits();
96          Collection<String> databaseNames = Arrays.asList("with_units", "without_units", "missing_config");
97          try (
98                  MockedConstruction<PropertiesPersistService> ignoredService = mockConstruction(PropertiesPersistService.class, (mock, context) -> when(mock.load()).thenReturn(new Properties()));
99                  MockedConstruction<MetaDataPersistFacade> ignoredFacade = mockConstruction(MetaDataPersistFacade.class, withSettings().defaultAnswer(RETURNS_DEEP_STUBS),
100                         (mock, context) -> {
101                             when(mock.getPropsService().load()).thenReturn(new Properties());
102                             when(mock.loadDataSourceConfigurations(anyString())).thenReturn(Collections.emptyMap());
103                             when(mock.getDatabaseMetaDataFacade().getDatabase().loadAllDatabaseNames()).thenReturn(databaseNames);
104                             when(mock.getStatisticsService().load(any())).thenReturn(new ShardingSphereStatistics());
105                         });
106                 MockedConstruction<DataSourcePoolDestroyer> destroyerMocked = mockConstruction(DataSourcePoolDestroyer.class)) {
107             MetaDataContexts actual = new RegisterCenterMetaDataContextsInitFactory(repository, instanceContext).create(createContextManagerBuilderParameter(databaseConfigs));
108             assertThat(actual.getMetaData().getAllDatabases(), hasSize(2));
109             assertThat(destroyerMocked.constructed(), hasSize(1));
110             verify(destroyerMocked.constructed().get(0)).asyncDestroy();
111         }
112     }
113     
114     @Test
115     void assertCreateMergesViewsWhenSchemasNotPersisted() throws SQLException {
116         when(GlobalRulesBuilder.buildRules(anyCollection(), anyCollection(), any(ConfigurationProperties.class))).thenReturn(Collections.emptyList());
117         when(ShardingSphereStatisticsFactory.create(any(), any())).thenReturn(new ShardingSphereStatistics());
118         ShardingSphereDatabase fooDatabase = createDatabase("foo_db",
119                 Collections.singleton(new ShardingSphereSchema("foo_schema", Collections.emptyList(), Collections.singleton(new ShardingSphereView("local_view", "select 1")))));
120         ShardingSphereDatabase barDatabase = createDatabase("bar_db", Collections.emptyList());
121         ComputeNodeInstanceContext instanceContext = mockComputeNodeInstanceContext(mock(JDBCInstanceMetaData.class));
122         when(ShardingSphereDatabasesFactory.create(anyMap(), any(ConfigurationProperties.class), eq(instanceContext))).thenReturn(Arrays.asList(fooDatabase, barDatabase));
123         Map<String, DatabaseConfiguration> databaseConfigs = createDatabaseConfigsWithoutStorageUnits();
124         Properties props = PropertiesBuilder.build(new Property(ConfigurationPropertyKey.PERSIST_SCHEMAS_TO_REPOSITORY_ENABLED.getKey(), Boolean.FALSE.toString()));
125         Collection<ShardingSphereSchema> persistedSchemas = Arrays.asList(
126                 new ShardingSphereSchema("foo_schema", Collections.emptyList(), Collections.singleton(new ShardingSphereView("persisted_view", "select 2"))),
127                 new ShardingSphereSchema("missing_schema", Collections.emptyList(), Collections.singleton(new ShardingSphereView("ignored_view", "select 3"))));
128         try (
129                 MockedConstruction<PropertiesPersistService> ignoredService = mockConstruction(PropertiesPersistService.class, (mock, context) -> when(mock.load()).thenReturn(props));
130                 MockedConstruction<MetaDataPersistFacade> ignoredFacade = mockConstruction(MetaDataPersistFacade.class, withSettings().defaultAnswer(RETURNS_DEEP_STUBS),
131                         (mock, context) -> {
132                             when(mock.getPropsService().load()).thenReturn(new Properties());
133                             when(mock.loadDataSourceConfigurations(anyString())).thenReturn(Collections.emptyMap());
134                             when(mock.getDatabaseMetaDataFacade().getSchema().load("foo_db")).thenReturn(persistedSchemas);
135                             when(mock.getStatisticsService().load(any())).thenReturn(new ShardingSphereStatistics());
136                         });
137                 MockedConstruction<DataSourcePoolDestroyer> destroyerMocked = mockConstruction(DataSourcePoolDestroyer.class)) {
138             MetaDataContexts actual = new RegisterCenterMetaDataContextsInitFactory(repository, instanceContext).create(createContextManagerBuilderParameter(databaseConfigs));
139             assertThat(actual.getMetaData().getAllDatabases(), hasSize(2));
140             assertTrue(fooDatabase.getSchema("foo_schema").containsView("persisted_view"));
141             assertTrue(fooDatabase.getSchema("foo_schema").containsView("local_view"));
142             assertThat(destroyerMocked.constructed(), hasSize(0));
143         }
144     }

146     private ComputeNodeInstanceContext mockComputeNodeInstanceContext(final InstanceMetaData metaData) {
147         ComputeNodeInstanceContext result = mock(ComputeNodeInstanceContext.class, RETURNS_DEEP_STUBS);
148         when(result.getInstance().getMetaData()).thenReturn(metaData);
149         return result;
150     }

152     private Map<String, DatabaseConfiguration> createDatabaseConfigsWithAndWithoutStorageUnits() {
153         DatabaseConfiguration withUnits = mock(DatabaseConfiguration.class, RETURNS_DEEP_STUBS);
154         when(withUnits.getStorageUnits()).thenReturn(Collections.singletonMap("with_units", mock(StorageUnit.class)));
155         when(withUnits.getDataSources()).thenReturn(Collections.singletonMap(new StorageNode("with_units"), new MockedDataSource()));
156         when(withUnits.getRuleConfigurations()).thenReturn(Collections.emptyList());
157         Map<String, DatabaseConfiguration> result = new LinkedHashMap<>(2, 1F);
158         result.put("with_units", withUnits);
159         result.put("without_units", createDatabaseConfigWithoutStorageUnits());
160         return result;
161     }
162     
163     private Map<String, DatabaseConfiguration> createDatabaseConfigsWithoutStorageUnits() {
164         Map<String, DatabaseConfiguration> result = new LinkedHashMap<>(2, 1F);
165         result.put("foo_db", createDatabaseConfigWithoutStorageUnits());
166         result.put("bar_db", createDatabaseConfigWithoutStorageUnits());
167         return result;
168     }
169     
170     private DatabaseConfiguration createDatabaseConfigWithoutStorageUnits() {
171         DatabaseConfiguration result = mock(DatabaseConfiguration.class);
172         when(result.getStorageUnits()).thenReturn(Collections.emptyMap());
173         when(result.getDataSources()).thenReturn(Collections.emptyMap());
174         when(result.getRuleConfigurations()).thenReturn(Collections.emptyList());
175         return result;
176     }

180                 mock(DatabaseType.class), new ResourceMetaData(Collections.emptyMap(), Collections.emptyMap()), new RuleMetaData(Collections.emptyList()), schemas);
181     }
