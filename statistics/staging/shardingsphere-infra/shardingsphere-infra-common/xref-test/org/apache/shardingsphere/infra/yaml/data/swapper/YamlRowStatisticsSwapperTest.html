18  package org.apache.shardingsphere.infra.yaml.data.swapper;
19  
20  import org.apache.shardingsphere.infra.metadata.database.schema.model.ShardingSphereColumn;
21  import org.apache.shardingsphere.infra.metadata.statistics.RowStatistics;
22  import org.apache.shardingsphere.infra.yaml.data.pojo.YamlRowStatistics;
23  import org.junit.jupiter.api.Test;
24  import org.junit.jupiter.params.ParameterizedTest;
25  import org.junit.jupiter.params.provider.Arguments;
26  import org.junit.jupiter.params.provider.MethodSource;
27  
28  import java.math.BigDecimal;
29  import java.sql.Types;
30  import java.util.Collections;
31  import java.util.stream.Stream;
32  
33  import static org.hamcrest.MatcherAssert.assertThat;
34  import static org.hamcrest.Matchers.contains;
35  import static org.hamcrest.Matchers.is;
36  import static org.junit.jupiter.api.Assertions.assertTrue;
37  
38  class YamlRowStatisticsSwapperTest {
39      
40      @Test
41      void assertSwapToYamlConfigurationWithNullRows() {
42          YamlRowStatisticsSwapper swapper = new YamlRowStatisticsSwapper(Collections.emptyList());
43          YamlRowStatistics actual = swapper.swapToYamlConfiguration(new RowStatistics("foo_unique_key", null));
44          assertTrue(actual.getRows().isEmpty());
45          assertThat(actual.getUniqueKey(), is("foo_unique_key"));
46      }
47      
48      @ParameterizedTest(name = "{0}")
49      @MethodSource("getSwapToYamlConfigurationArguments")
50      void assertSwapToYamlConfigurationWithDataType(final String name, final int dataType, final Object input, final Object expected) {
51          ShardingSphereColumn column = new ShardingSphereColumn("foo_col", dataType, false, false, false, true, false, true);
52          YamlRowStatistics actual = new YamlRowStatisticsSwapper(Collections.singletonList(column))
53                  .swapToYamlConfiguration(new RowStatistics("foo_unique_key", Collections.singletonList(input)));
54          assertThat(actual.getRows(), contains(expected));
55      }
56      
57      private static Stream<Arguments> getSwapToYamlConfigurationArguments() {
58          return Stream.of(
59                  Arguments.of("decimal", Types.DECIMAL, new BigDecimal("7.5"), "7.5"),
60                  Arguments.of("numeric", Types.NUMERIC, new BigDecimal("8.25"), "8.25"),
61                  Arguments.of("null decimal", Types.DECIMAL, null, null),
62                  Arguments.of("bigint", Types.BIGINT, 5L, 5L),
63                  Arguments.of("varchar", Types.VARCHAR, "foo_value", "foo_value"));
64      }
65      
66      @Test
67      void assertSwapToObjectWithNullRows() {
68          YamlRowStatistics yamlConfig = new YamlRowStatistics();
69          yamlConfig.setUniqueKey("foo_unique_key");
70          yamlConfig.setRows(null);
71          RowStatistics actual = new YamlRowStatisticsSwapper(Collections.emptyList()).swapToObject(yamlConfig);
72          assertTrue(actual.getRows().isEmpty());
73          assertThat(actual.getUniqueKey(), is("foo_unique_key"));
74      }
75      
76      @ParameterizedTest(name = "{0}")
77      @MethodSource("getSwapToObjectArguments")
78      void assertSwapToObjectWithDataType(final String name, final int dataType, final Object input, final Object expected) {
79          ShardingSphereColumn column = new ShardingSphereColumn("foo_col", dataType, false, false, false, true, false, true);
80          YamlRowStatistics yamlConfig = new YamlRowStatistics();
81          yamlConfig.setRows(Collections.singletonList(input));
82          RowStatistics actual = new YamlRowStatisticsSwapper(Collections.singletonList(column)).swapToObject(yamlConfig);
83          assertThat(actual.getRows(), contains(expected));
84      }
85      
86      private static Stream<Arguments> getSwapToObjectArguments() {
87          return Stream.of(
88                  Arguments.of("decimal string", Types.DECIMAL, "1.5", new BigDecimal("1.5")),
89                  Arguments.of("numeric string", Types.NUMERIC, "2.5", new BigDecimal("2.5")),
90                  Arguments.of("numeric BigDecimal", Types.NUMERIC, new BigDecimal("3.5"), new BigDecimal("3.5")),
91                  Arguments.of("bigint integer", Types.BIGINT, 3, 3L),
92                  Arguments.of("bigint long", Types.BIGINT, 4L, 4L),
93                  Arguments.of("real double", Types.REAL, 5.5D, 5.5F),
94                  Arguments.of("real float", Types.REAL, 6.5F, 6.5F),
95                  Arguments.of("float double", Types.FLOAT, 7.5D, 7.5D),
96                  Arguments.of("double string", Types.DOUBLE, "8.5", 8.5D),
97                  Arguments.of("varchar", Types.VARCHAR, "foo_text", "foo_text"),
98                  Arguments.of("null decimal", Types.DECIMAL, null, null));
99      }
