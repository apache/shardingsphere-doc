127     @Override
128     public ASTNode visitInsert(final InsertContext ctx) {
129         InsertStatement result = (InsertStatement) visit(ctx.insertValuesClause());
130         result.setTable((SimpleTableSegment) visit(ctx.tableName()));
131         result.addParameterMarkers(getParameterMarkerSegments());
132         if (null != ctx.returningClause()) {
133             result.setReturning((ReturningSegment) visit(ctx.returningClause()));
134         }
135         return result;
136     }
137     
138     @SuppressWarnings("unchecked")
139     @Override
140     public ASTNode visitInsertValuesClause(final InsertValuesClauseContext ctx) {
141         InsertStatement result = new InsertStatement(getDatabaseType());
142         if (null != ctx.columnNames()) {
143             ColumnNamesContext columnNames = ctx.columnNames();
144             CollectionValue<ColumnSegment> columnSegments = (CollectionValue<ColumnSegment>) visit(columnNames);
145             result.setInsertColumns(new InsertColumnsSegment(columnNames.start.getStartIndex(), columnNames.stop.getStopIndex(), columnSegments.getValue()));
146         } else {
147             result.setInsertColumns(new InsertColumnsSegment(ctx.start.getStartIndex() - 1, ctx.start.getStartIndex() - 1, Collections.emptyList()));
148         }
149         result.getValues().addAll(createInsertValuesSegments(ctx.assignmentValues()));
150         return result;
151     }
152     
153     private Collection<InsertValuesSegment> createInsertValuesSegments(final Collection<AssignmentValuesContext> assignmentValuesContexts) {
154         Collection<InsertValuesSegment> result = new LinkedList<>();
155         for (AssignmentValuesContext each : assignmentValuesContexts) {
156             result.add((InsertValuesSegment) visit(each));
157         }
158         return result;
159     }

161     @Override
162     public ASTNode visitReturningClause(final ReturningClauseContext ctx) {
163         return new ReturningSegment(ctx.getStart().getStartIndex(), ctx.getStop().getStopIndex(), (ProjectionsSegment) visit(ctx.projections()));
164     }

166     @Override
167     public ASTNode visitUpdate(final UpdateContext ctx) {
168         UpdateStatement result = new UpdateStatement(getDatabaseType());
169         result.setTable((TableSegment) visit(ctx.tableReferences()));
170         result.setSetAssignment((SetAssignmentSegment) visit(ctx.setAssignmentsClause()));
171         if (null != ctx.whereClause()) {
172             result.setWhere((WhereSegment) visit(ctx.whereClause()));
173         }
174         result.addParameterMarkers(getParameterMarkerSegments());
175         if (null != ctx.returningClause()) {
176             result.setReturning((ReturningSegment) visit(ctx.returningClause()));
177         }
178         return result;
179     }

184         for (AssignmentContext each : ctx.assignment()) {
185             assignments.add((ColumnAssignmentSegment) visit(each));

201         ColumnSegment column = (ColumnSegment) visitColumnName(ctx.columnName());

256     @Override
257     public ASTNode visitSelectClause(final SelectClauseContext ctx) {
258         SelectStatement result = new SelectStatement(getDatabaseType());
259         result.setProjections((ProjectionsSegment) visit(ctx.projections()));
260         if (null != ctx.firstSkipClause()) {
261             result.setLimit((LimitSegment) visit(ctx.firstSkipClause()));
262         }
263         if (!ctx.selectSpecification().isEmpty()) {
264             result.getProjections().setDistinctRow(isDistinct(ctx.selectSpecification().get(0)));
265         }
266         if (null != ctx.fromClause()) {
267             TableSegment tableSegment = (TableSegment) visit(ctx.fromClause());
268             result.setFrom(tableSegment);
269         }
270         if (null != ctx.whereClause()) {
271             result.setWhere((WhereSegment) visit(ctx.whereClause()));
272         }
273         if (null != ctx.groupByClause()) {
274             result.setGroupBy((GroupBySegment) visit(ctx.groupByClause()));
275         }
276         if (null != ctx.orderByClause()) {
277             result.setOrderBy((OrderBySegment) visit(ctx.orderByClause()));
278         }
279         if (null != ctx.havingClause()) {
280             result.setHaving((HavingSegment) visit(ctx.havingClause()));
281         }
282         return result;
283     }

285     @Override
286     public ASTNode visitFirstSkipClause(final FirstSkipClauseContext ctx) {
287         PaginationValueSegment rowCount = null;
288         PaginationValueSegment offset = null;
289         if (null != ctx.FIRST()) {
290             rowCount = (PaginationValueSegment) visit(ctx.firstValue());
291             if (null != ctx.SKIP_()) {
292                 offset = (PaginationValueSegment) visit(ctx.skipValue());
293             }
294         } else if (null != ctx.SKIP_()) {
295             offset = (PaginationValueSegment) visit(ctx.skipValue());
296         }
297         return new LimitSegment(ctx.getStart().getStartIndex(), ctx.getStop().getStopIndex(), offset, rowCount);
298     }

300     @Override
301     public ASTNode visitFirstValue(final FirstValueContext ctx) {
302         // TODO Add expression support for firstSkipClause
303         if (null != ctx.numberLiterals()) {
304             return new NumberLiteralLimitValueSegment(ctx.getStart().getStartIndex(), ctx.getStop().getStopIndex(), ((NumberLiteralValue) visit(ctx.numberLiterals())).getValue().longValue());
305         }
306         ParameterMarkerSegment result = new ParameterMarkerLimitValueSegment(ctx.getStart().getStartIndex(), ctx.getStop().getStopIndex(),
307                 ((ParameterMarkerValue) visit(ctx.parameterMarker())).getValue());
308         getParameterMarkerSegments().add(result);
309         return result;
310     }

312     @Override
313     public ASTNode visitSkipValue(final SkipValueContext ctx) {
314         // TODO Add expression support for firstSkipClause
315         if (null != ctx.numberLiterals()) {
316             return new NumberLiteralLimitValueSegment(ctx.getStart().getStartIndex(), ctx.getStop().getStopIndex(), ((NumberLiteralValue) visit(ctx.numberLiterals())).getValue().longValue());
317         }
318         ParameterMarkerSegment result = new ParameterMarkerLimitValueSegment(ctx.getStart().getStartIndex(), ctx.getStop().getStopIndex(),
319                 ((ParameterMarkerValue) visit(ctx.parameterMarker())).getValue());
320         getParameterMarkerSegments().add(result);
321         return result;
322     }

339     @Override
340     public ASTNode visitProjections(final ProjectionsContext ctx) {
341         Collection<ProjectionSegment> projections = new LinkedList<>();
342         if (null != ctx.unqualifiedShorthand()) {
343             projections.add(new ShorthandProjectionSegment(ctx.unqualifiedShorthand().getStart().getStartIndex(), ctx.unqualifiedShorthand().getStop().getStopIndex()));
344         }
345         for (ProjectionContext each : ctx.projection()) {
346             projections.add((ProjectionSegment) visit(each));
347         }
348         ProjectionsSegment result = new ProjectionsSegment(ctx.getStart().getStartIndex(), ctx.getStop().getStopIndex());
349         result.getProjections().addAll(projections);
350         return result;
351     }

353     @Override
354     public ASTNode visitProjection(final ProjectionContext ctx) {
355         // FIXME :The stop index of project is the stop index of projection, instead of alias.
356         if (null != ctx.qualifiedShorthand()) {
357             QualifiedShorthandContext shorthand = ctx.qualifiedShorthand();
358             ShorthandProjectionSegment result = new ShorthandProjectionSegment(shorthand.getStart().getStartIndex(), shorthand.getStop().getStopIndex());
359             IdentifierValue identifier = new IdentifierValue(shorthand.identifier().getText());
360             result.setOwner(new OwnerSegment(shorthand.identifier().getStart().getStartIndex(), shorthand.identifier().getStop().getStopIndex(), identifier));
361             return result;
362         }
363         AliasSegment alias = null == ctx.alias() ? null : (AliasSegment) visit(ctx.alias());
364         if (null != ctx.columnName()) {
365             ColumnSegment column = (ColumnSegment) visit(ctx.columnName());
366             ColumnProjectionSegment result = new ColumnProjectionSegment(column);
367             result.setAlias(alias);
368             return result;
369         }
370         return createProjection(ctx, alias);
371     }

373     @Override
374     public ASTNode visitAlias(final AliasContext ctx) {
375         return null == ctx.identifier()
376                 ? new AliasSegment(ctx.start.getStartIndex(), ctx.stop.getStopIndex(), new IdentifierValue(ctx.STRING_().getText()))
377                 : new AliasSegment(ctx.start.getStartIndex(), ctx.stop.getStopIndex(), (IdentifierValue) visit(ctx.identifier()));
378     }

380     private ASTNode createProjection(final ProjectionContext ctx, final AliasSegment alias) {
381         ASTNode projection = visit(ctx.expr());
382         if (projection instanceof AggregationProjectionSegment) {
383             ((AggregationProjectionSegment) projection).setAlias(alias);
384             return projection;
385         }
386         if (projection instanceof ExpressionProjectionSegment) {
387             ((ExpressionProjectionSegment) projection).setAlias(alias);
388             return projection;
389         }
390         if (projection instanceof FunctionSegment) {
391             FunctionSegment segment = (FunctionSegment) projection;
392             ExpressionProjectionSegment result = new ExpressionProjectionSegment(segment.getStartIndex(), segment.getStopIndex(), segment.getText(), segment);
393             result.setAlias(alias);
394             return result;
395         }
396         if (projection instanceof CommonExpressionSegment) {
397             CommonExpressionSegment segment = (CommonExpressionSegment) projection;
398             ExpressionProjectionSegment result = new ExpressionProjectionSegment(segment.getStartIndex(), segment.getStopIndex(), segment.getText(), segment);
399             result.setAlias(alias);
400             return result;
401         }
402         // FIXME :For DISTINCT()
403         if (projection instanceof ColumnSegment) {
404             ExpressionProjectionSegment result = new ExpressionProjectionSegment(ctx.start.getStartIndex(), ctx.stop.getStopIndex(), getOriginalText(ctx), (ColumnSegment) projection);
405             result.setAlias(alias);
406             return result;
407         }
408         if (projection instanceof SubqueryExpressionSegment) {
409             SubqueryExpressionSegment subqueryExpressionSegment = (SubqueryExpressionSegment) projection;
410             String text = ctx.start.getInputStream().getText(new Interval(subqueryExpressionSegment.getStartIndex(), subqueryExpressionSegment.getStopIndex()));
411             SubqueryProjectionSegment result = new SubqueryProjectionSegment(((SubqueryExpressionSegment) projection).getSubquery(), text);
412             result.setAlias(alias);
413             return result;
414         }
415         if (projection instanceof BinaryOperationExpression) {
416             BinaryOperationExpression binaryExpression = (BinaryOperationExpression) projection;
417             int startIndex = binaryExpression.getStartIndex();
418             int stopIndex = null == alias ? binaryExpression.getStopIndex() : alias.getStopIndex();
419             ExpressionProjectionSegment result = new ExpressionProjectionSegment(startIndex, stopIndex, binaryExpression.getText(), binaryExpression);
420             result.setAlias(alias);
421             return result;
422         }
423         if (projection instanceof ParameterMarkerExpressionSegment) {
424             ParameterMarkerExpressionSegment result = (ParameterMarkerExpressionSegment) projection;
425             result.setAlias(alias);
426             return projection;
427         }
428         ExpressionSegment column = (ExpressionSegment) projection;
429         ExpressionProjectionSegment result = null == alias ? new ExpressionProjectionSegment(column.getStartIndex(), column.getStopIndex(), String.valueOf(column.getText()), column)
430                 : new ExpressionProjectionSegment(column.getStartIndex(), ctx.alias().stop.getStopIndex(), String.valueOf(column.getText()), column);
431         result.setAlias(alias);
432         return result;
433     }

440     @Override
441     public ASTNode visitTableReferences(final TableReferencesContext ctx) {
442         TableSegment result = (TableSegment) visit(ctx.escapedTableReference(0));
443         if (ctx.escapedTableReference().size() > 1) {
444             for (int i = 1; i < ctx.escapedTableReference().size(); i++) {
445                 result = generateJoinTableSourceFromEscapedTableReference(ctx.escapedTableReference(i), result);
446             }
447         }
448         return result;
449     }
450     
451     private JoinTableSegment generateJoinTableSourceFromEscapedTableReference(final EscapedTableReferenceContext ctx, final TableSegment tableSegment) {
452         JoinTableSegment result = new JoinTableSegment();
453         result.setStartIndex(tableSegment.getStartIndex());
454         result.setStopIndex(ctx.stop.getStopIndex());
455         result.setLeft(tableSegment);
456         result.setRight((TableSegment) visit(ctx));
457         result.setJoinType(JoinType.COMMA.name());
458         return result;
459     }

466     @Override
467     public ASTNode visitTableReference(final TableReferenceContext ctx) {
468         TableSegment result;
469         TableSegment left;
470         left = (TableSegment) visit(ctx.tableFactor());
471         if (!ctx.joinedTable().isEmpty()) {
472             for (JoinedTableContext each : ctx.joinedTable()) {
473                 left = visitJoinedTable(each, left);
474             }
475         }
476         result = left;
477         return result;
478     }

480     @Override
481     public ASTNode visitTableFactor(final TableFactorContext ctx) {
482         if (null != ctx.subquery()) {
483             SelectStatement subquery = (SelectStatement) visit(ctx.subquery());
484             SubquerySegment subquerySegment = new SubquerySegment(ctx.subquery().start.getStartIndex(), ctx.subquery().stop.getStopIndex(), subquery, getOriginalText(ctx.subquery()));
485             SubqueryTableSegment result = new SubqueryTableSegment(ctx.start.getStartIndex(), ctx.stop.getStopIndex(), subquerySegment);
486             if (null != ctx.alias()) {
487                 result.setAlias((AliasSegment) visit(ctx.alias()));
488             }
489             return result;
490         }
491         if (null != ctx.tableName()) {
492             SimpleTableSegment result = (SimpleTableSegment) visit(ctx.tableName());
493             if (null != ctx.alias()) {
494                 result.setAlias((AliasSegment) visit(ctx.alias()));
495             }
496             return result;
497         }
498         if (null != ctx.expr()) {
499             ExpressionSegment exprSegment = (ExpressionSegment) visit(ctx.expr());
500             FunctionTableSegment result = new FunctionTableSegment(exprSegment.getStartIndex(), exprSegment.getStopIndex(), exprSegment);
501             if (null != ctx.alias()) {
502                 result.setAlias((AliasSegment) visit(ctx.alias()));
503             }
504             return result;
505         }
506         return visit(ctx.tableReferences());
507     }
508     
509     private JoinTableSegment visitJoinedTable(final JoinedTableContext ctx, final TableSegment tableSegment) {
510         JoinTableSegment result = new JoinTableSegment();
511         result.setLeft(tableSegment);
512         result.setStartIndex(tableSegment.getStartIndex());
513         result.setStopIndex(ctx.stop.getStopIndex());
514         TableSegment right = (TableSegment) visit(ctx.tableFactor());
515         result.setRight(right);
516         result.setJoinType(getJoinType(ctx));
517         if (null != ctx.joinSpecification()) {
518             visitJoinSpecification(ctx.joinSpecification(), result);
519         }
520         return result;
521     }
522     
523     private String getJoinType(final JoinedTableContext ctx) {
524         if (null != ctx.LEFT()) {
525             return JoinType.LEFT.name();
526         } else if (null != ctx.RIGHT()) {
527             return JoinType.RIGHT.name();
528         } else if (null != ctx.INNER()) {
529             return JoinType.INNER.name();
530         } else if (null != ctx.CROSS()) {
531             return JoinType.CROSS.name();
532         }
533         return JoinType.INNER.name();
534     }
535     
536     private void visitJoinSpecification(final JoinSpecificationContext ctx, final JoinTableSegment joinTableSource) {
537         if (null != ctx.expr()) {
538             ExpressionSegment condition = (ExpressionSegment) visit(ctx.expr());
539             joinTableSource.setCondition(condition);
540         }
541         if (null != ctx.USING()) {
542             joinTableSource.setUsing(ctx.columnNames().columnName().stream().map(each -> (ColumnSegment) visit(each)).collect(Collectors.toList()));
543         }
544     }

547     public ASTNode visitWhereClause(final WhereClauseContext ctx) {
548         ExpressionSegment segment = (ExpressionSegment) visit(ctx.expr());
549         return new WhereSegment(ctx.getStart().getStartIndex(), ctx.getStop().getStopIndex(), segment);

553     public ASTNode visitGroupByClause(final GroupByClauseContext ctx) {

555         for (OrderByItemContext each : ctx.orderByItem()) {
