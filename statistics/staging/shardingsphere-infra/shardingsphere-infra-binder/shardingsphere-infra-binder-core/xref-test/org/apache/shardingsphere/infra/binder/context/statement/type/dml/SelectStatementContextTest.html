297     private OrderByItemSegment createOrderByItemSegment(final String type) {
298         switch (type) {
299             case INDEX_ORDER_BY:
300                 return new IndexOrderByItemSegment(0, 0, 4, OrderDirection.ASC, NullsOrderType.FIRST);
301             case COLUMN_ORDER_BY_WITH_OWNER:
302                 ColumnSegment columnSegment = new ColumnSegment(0, 0, new IdentifierValue("name"));
303                 OwnerSegment owner = new OwnerSegment(0, 0, new IdentifierValue("table"));
304                 owner.setTableBoundInfo(new TableSegmentBoundInfo(new IdentifierValue("foo_db"), new IdentifierValue("foo_schema")));
305                 columnSegment.setOwner(owner);
306                 return new ColumnOrderByItemSegment(columnSegment, OrderDirection.ASC, NullsOrderType.FIRST);
307             case COLUMN_ORDER_BY_WITH_ALIAS:
308                 return new ColumnOrderByItemSegment(new ColumnSegment(0, 0, new IdentifierValue("n")), OrderDirection.ASC, NullsOrderType.FIRST);
309             default:
310                 return new ColumnOrderByItemSegment(new ColumnSegment(0, 0, new IdentifierValue("id")), OrderDirection.ASC, NullsOrderType.FIRST);
311         }
312     }

314     private ProjectionsSegment createProjectionsSegment() {
315         ProjectionsSegment result = new ProjectionsSegment(0, 0);
316         result.setDistinctRow(true);
317         result.getProjections().addAll(Arrays.asList(getColumnProjectionSegmentWithoutOwner(), getColumnProjectionSegmentWithoutOwner(true), getColumnProjectionSegmentWithoutOwner(false)));
318         return result;
319     }
320     
321     private ProjectionSegment getColumnProjectionSegmentWithoutOwner() {
322         ColumnSegment columnSegment = new ColumnSegment(0, 0, new IdentifierValue("name"));
323         OwnerSegment owner = new OwnerSegment(0, 0, new IdentifierValue("table"));
324         owner.setTableBoundInfo(new TableSegmentBoundInfo(new IdentifierValue("foo_db"), new IdentifierValue("foo_schema")));
325         columnSegment.setOwner(owner);
326         return new ColumnProjectionSegment(columnSegment);
327     }

329     private ProjectionSegment getColumnProjectionSegmentWithoutOwner(final boolean hasAlias) {
330         ColumnSegment columnSegment = new ColumnSegment(0, 0, new IdentifierValue(hasAlias ? "name" : "id"));
331         ColumnProjectionSegment columnProjectionSegment = new ColumnProjectionSegment(columnSegment);
332         columnProjectionSegment.setAlias(new AliasSegment(0, 0, new IdentifierValue(hasAlias ? "n" : null)));
333         return columnProjectionSegment;
