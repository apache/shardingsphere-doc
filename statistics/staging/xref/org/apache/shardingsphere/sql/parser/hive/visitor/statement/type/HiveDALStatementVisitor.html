53  /**
54   * DAL statement visitor for Hive.
55   */
56  public final class HiveDALStatementVisitor extends HiveStatementVisitor implements DALStatementVisitor {
57      
58      public HiveDALStatementVisitor(final DatabaseType databaseType) {
59          super(databaseType);
60      }
61      
62      @Override
63      public ASTNode visitUse(final UseContext ctx) {
64          String database = null == ctx.DEFAULT() ? new IdentifierValue(ctx.identifier().getText()).getValue() : "default";
65          return new MySQLUseStatement(getDatabaseType(), database);
66      }
67      
68      @Override
69      public ASTNode visitShowDatabases(final ShowDatabasesContext ctx) {
70          ShowFilterSegment filter = null;
71          if (null != ctx.showLike()) {
72              filter = new ShowFilterSegment(ctx.showLike().getStart().getStartIndex(), ctx.showLike().getStop().getStopIndex());
73              filter.setLike((ShowLikeSegment) visit(ctx.showLike()));
74          }
75          MySQLShowDatabasesStatement result = new MySQLShowDatabasesStatement(getDatabaseType(), filter);
76          result.addParameterMarkers(getParameterMarkerSegments());
77          return result;
78      }
79      
80      @Override
81      public ASTNode visitShowLike(final ShowLikeContext ctx) {
82          StringLiteralValue literalValue = (StringLiteralValue) visit(ctx.stringLiterals());
83          return new ShowLikeSegment(ctx.getStart().getStartIndex(), ctx.getStop().getStopIndex(), literalValue.getValue());
84      }
85      
86      @Override
87      public ASTNode visitShowConnectors(final ShowConnectorsContext ctx) {
88          return new HiveShowConnectorsStatement(getDatabaseType());
89      }
90      
91      @Override
92      public ASTNode visitShowTables(final ShowTablesContext ctx) {
93          FromDatabaseSegment fromDatabase = null;
94          if (null != ctx.databaseName()) {
95              DatabaseSegment databaseSegment = (DatabaseSegment) visit(ctx.databaseName());
96              fromDatabase = new FromDatabaseSegment(ctx.databaseName().getStart().getStartIndex(), databaseSegment);
97          }
98          ShowFilterSegment filter = null;
99          if (null != ctx.stringLiterals()) {
100             StringLiteralValue literalValue = (StringLiteralValue) visit(ctx.stringLiterals());
101             ShowLikeSegment likeSegment = new ShowLikeSegment(ctx.stringLiterals().getStart().getStartIndex(), ctx.stringLiterals().getStop().getStopIndex(), literalValue.getValue());
102             filter = new ShowFilterSegment(ctx.stringLiterals().getStart().getStartIndex(), ctx.stringLiterals().getStop().getStopIndex());
103             filter.setLike(likeSegment);
104         }
105         MySQLShowTablesStatement result = new MySQLShowTablesStatement(getDatabaseType(), fromDatabase, filter, false);
106         result.addParameterMarkers(getParameterMarkerSegments());
107         return result;
108     }
109     
110     @Override
111     public ASTNode visitShowViews(final ShowViewsContext ctx) {
112         return new HiveShowViewsStatement(getDatabaseType());
113     }
114     
115     @Override
116     public ASTNode visitShowMaterializedViews(final ShowMaterializedViewsContext ctx) {
117         return new HiveShowMaterializedViewsStatement(getDatabaseType());
118     }
119     
120     @Override
121     public ASTNode visitShowPartitions(final ShowPartitionsContext ctx) {
122         return new HiveShowPartitionsStatement(getDatabaseType());
123     }
124     
125     @Override
126     public ASTNode visitShowTablesExtended(final ShowTablesExtendedContext ctx) {
127         return new HiveShowTablesExtendedStatement(getDatabaseType());
128     }
129     
130     @Override
131     public ASTNode visitShowTblproperties(final ShowTblpropertiesContext ctx) {
132         return new HiveShowTblpropertiesStatement(getDatabaseType());
133     }
134     
135     @Override
136     public ASTNode visitShowCreateTable(final ShowCreateTableContext ctx) {
137         return new MySQLShowCreateTableStatement(getDatabaseType(), (SimpleTableSegment) visit(ctx.tableName()));
138     }
139 }
