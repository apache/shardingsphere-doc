250     @ParameterizedTest(name = "{0}")
251     @MethodSource("assertValidateFailsDuringProfileLoadCases")
252     void assertValidateFailsDuringProfileLoad(final String name, final RuntimeDatabaseConnectionException cause, final String expectedDriverStatus,
253                                               final String expectedConnectivityStatus, final String expectedCategory) {
254         RuntimeDatabaseValidationService service = new RuntimeDatabaseValidationService();
255         MCPJdbcDatabaseProfileLoader profileLoader = getProfileLoader();
256         RuntimeDatabaseConfiguration runtimeDatabaseConfig = createRuntimeDatabaseConfiguration();
257         when(profileLoader.load(any(), any(RuntimeDatabaseConfiguration.class))).thenThrow(cause);
258         RuntimeDatabaseValidationResult actual = service.validate(new RuntimeDatabaseValidationRequest("logic_db"), ignored -> Optional.of(runtimeDatabaseConfig));
259         assertThat(actual.getStatus(), is("failed"));
260         assertThat(actual.getCategory(), is(expectedCategory));
261         assertThat(actual.getChecks().get(1).getStatus(), is(expectedDriverStatus));
262         assertThat(actual.getChecks().get(2).getStatus(), is(expectedConnectivityStatus));
263         assertThat(actual.getChecks().get(3).getStatus(), is("skipped"));
264         assertThat(actual.getChecks().get(4).getStatus(), is("skipped"));
265         verifyNoInteractions(getMetadataLoader());
266     }

303     private static final class InvisibleDatabaseDriver implements Driver {
304         
305         private static final String JDBC_URL = "jdbc:runtime-validation:invisible";
306         
307         private static final InvisibleDatabaseDriver INSTANCE = new InvisibleDatabaseDriver();
308         
309         static {
310             try {
311                 DriverManager.registerDriver(INSTANCE);
312             } catch (final SQLException ex) {
313                 throw new ExceptionInInitializerError(ex);
314             }
315         }
316         
317         @Override
318         public Connection connect(final String url, final Properties info) throws SQLException {
319             if (!acceptsURL(url)) {
320                 return null;
321             }
322             Connection result = mock(Connection.class);
323             DatabaseMetaData databaseMetaData = mock(DatabaseMetaData.class);
324             ResultSet catalogs = mock(ResultSet.class);
325             ResultSet schemas = mock(ResultSet.class);
326             when(result.getCatalog()).thenReturn("");
327             when(result.getSchema()).thenReturn("");
328             when(result.getMetaData()).thenReturn(databaseMetaData);
329             when(databaseMetaData.getCatalogs()).thenReturn(catalogs);
330             when(databaseMetaData.getSchemas()).thenReturn(schemas);
331             when(catalogs.next()).thenReturn(false);
332             when(schemas.next()).thenReturn(false);
333             return result;
334         }
335         
336         @Override
337         public boolean acceptsURL(final String url) {
338             return JDBC_URL.equals(url);
339         }
340         
341         @Override
342         public DriverPropertyInfo[] getPropertyInfo(final String url, final Properties info) {
343             return new DriverPropertyInfo[0];
344         }
345         
346         @Override
347         public int getMajorVersion() {
348             return 1;
349         }
350         
351         @Override
352         public int getMinorVersion() {
353             return 0;
354         }
355         
356         @Override
357         public boolean jdbcCompliant() {
358             return false;
359         }
360         
361         @Override
362         public Logger getParentLogger() throws SQLFeatureNotSupportedException {
363             throw new SQLFeatureNotSupportedException("Invisible database driver does not expose a parent logger.");
364         }
