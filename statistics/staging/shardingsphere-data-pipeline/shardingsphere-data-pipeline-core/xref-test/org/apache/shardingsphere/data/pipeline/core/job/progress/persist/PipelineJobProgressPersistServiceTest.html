72      @SuppressWarnings("unchecked")
73      @SneakyThrows(ReflectiveOperationException.class)
74      private Map<String, Map<Integer, PipelineJobProgressPersistContext>> getJobProgressPersistMap() {
75          Field field = PipelineJobProgressPersistService.class.getDeclaredField("JOB_PROGRESS_PERSIST_MAP");
76          return (Map<String, Map<Integer, PipelineJobProgressPersistContext>>) Plugins.getMemberAccessor().get(field, PipelineJobProgressPersistService.class);
77      }

79      @Test
80      void assertAdd() {
81          PipelineJobProgressPersistService.add("foo_id", 1);
82          Map<String, Map<Integer, PipelineJobProgressPersistContext>> jobProgressPersistMap = getJobProgressPersistMap();
83          assertThat(jobProgressPersistMap.get("foo_id").get(1).getUnhandledEventCount().get(), is(0L));
84      }

108     @SuppressWarnings("rawtypes")
109     @SneakyThrows(ReflectiveOperationException.class)
110     @Test
111     void assertPersistJobContextRunnableIteratesEntries() {
112         String jobId = "foo_scheduler_job";
113         int shardingItem = 2;
114         PipelineJobItemContext jobItemContext = mock(PipelineJobItemContext.class);
115         when(PipelineJobRegistry.getItemContext(jobId, shardingItem)).thenReturn(Optional.of(jobItemContext));
116         PipelineJobType<?> jobType = mock(PipelineJobType.class);
117         PipelineJobOption jobOption = mock(PipelineJobOption.class);
118         when(jobOption.getYamlJobItemProgressSwapper()).thenReturn(null);
119         when(jobType.getOption()).thenReturn(jobOption);
120         when(jobType.getType()).thenReturn("TEST");
121         try (
122                 MockedStatic<PipelineJobIdUtils> jobIdUtilsMock = mockStatic(PipelineJobIdUtils.class);
123                 MockedStatic<TypedSPILoader> typedSpiLoaderStatic = mockStatic(TypedSPILoader.class);
124                 MockedConstruction<PipelineJobItemManager> mockedConstruction = mockConstruction(PipelineJobItemManager.class,
125                         (mock, context) -> doNothing().when(mock).updateProgress(jobItemContext))) {
126             jobIdUtilsMock.when(() -> PipelineJobIdUtils.parseJobType(jobId)).thenReturn(jobType);
127             typedSpiLoaderStatic.when(() -> TypedSPILoader.getService(PipelineJobType.class, "TEST")).thenReturn(jobType);
128             PipelineJobProgressPersistService.add(jobId, shardingItem);
129             PipelineJobProgressPersistService.notifyPersist(jobId, shardingItem);
130             Class<?> runnableClass = Class.forName(PipelineJobProgressPersistService.class.getName() + "$PersistJobContextRunnable");
131             Constructor<?> constructor = runnableClass.getDeclaredConstructor();
132             constructor.setAccessible(true);
133             Runnable runnable = (Runnable) constructor.newInstance();
134             assertDoesNotThrow(runnable::run);
135             assertThat(getJobProgressPersistMap().get(jobId).get(shardingItem).getUnhandledEventCount().get(), is(0L));
136             verify(mockedConstruction.constructed().get(0)).updateProgress(jobItemContext);
137         } finally {
138             PipelineJobProgressPersistService.remove(jobId);
139         }
140     }

180     @Test
181     void assertPersistNowHandlesSubsequentExceptionLoggingBranches(final LogCaptureAssertion logCaptureAssertion) throws ClassNotFoundException {
182         synchronized (Class.forName(PipelineJobProgressPersistService.class.getName() + "$PersistJobContextRunnable")) {
183             String jobId = "foo_id_exception_follow";
184             int shardingItem = 1;
185             PipelineJobProgressPersistService.add(jobId, shardingItem);
186             PipelineJobProgressPersistContext persistContext = getJobProgressPersistMap().get(jobId).get(shardingItem);
187             try {
188                 persistContext.getUnhandledEventCount().set(-1L);
189                 when(PipelineJobRegistry.getItemContext(jobId, shardingItem)).thenReturn(Optional.empty());
190                 assertDoesNotThrow(() -> PipelineJobProgressPersistService.persistNow(jobId, shardingItem));
191                 logCaptureAssertion.assertErrorLog(actualException -> {
192                     assertThat(actualException.getClassName(), is(IllegalStateException.class.getName()));
193                     assertThat(actualException.getMessage(), is("Current unhandled event count must be greater than or equal to 0"));
194                 });
195                 assertTrue(persistContext.getFirstExceptionLogged().get());
196                 try (MockedStatic<ThreadLocalRandom> ignored = mockStatic(ThreadLocalRandom.class)) {
197                     ThreadLocalRandom randomMock = mock(ThreadLocalRandom.class);
198                     when(ThreadLocalRandom.current()).thenReturn(randomMock);
199                     when(randomMock.nextInt(60)).thenReturn(5, 4);
200                     assertDoesNotThrow(() -> PipelineJobProgressPersistService.persistNow(jobId, shardingItem));
201                     logCaptureAssertion.assertErrorLog(actualException -> {
202                         assertThat(actualException.getClassName(), is(IllegalStateException.class.getName()));
203                         assertThat(actualException.getMessage(), is("Current unhandled event count must be greater than or equal to 0"));
204                     });
205                     assertDoesNotThrow(() -> PipelineJobProgressPersistService.persistNow(jobId, shardingItem));
206                     verify(randomMock, times(2)).nextInt(60);
207                 }
208             } finally {
209                 persistContext.getUnhandledEventCount().set(0L);
210             }
211         }
