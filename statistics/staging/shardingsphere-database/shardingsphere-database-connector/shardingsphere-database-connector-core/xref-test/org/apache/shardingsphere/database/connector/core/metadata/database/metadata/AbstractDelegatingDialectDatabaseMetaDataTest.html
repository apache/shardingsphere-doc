42  import static org.hamcrest.MatcherAssert.assertThat;
43  import static org.hamcrest.Matchers.is;
44  import static org.mockito.Mockito.mock;
45  import static org.mockito.Mockito.verifyNoInteractions;

48  class AbstractDelegatingDialectDatabaseMetaDataTest {
49      
50      private final DialectDatabaseMetaData delegate = mock(DialectDatabaseMetaData.class);
51      
52      private final DialectDatabaseMetaData metaData = new FixtureDelegatingDialectDatabaseMetaData(delegate);
53      
54      @Test
55      void assertDelegate() {
56          when(delegate.getQuoteCharacter()).thenReturn(QuoteCharacter.BACK_QUOTE);
57          assertThat(metaData.getQuoteCharacter(), is(delegate.getQuoteCharacter()));
58          when(delegate.getIdentifierPatternType()).thenReturn(IdentifierPatternType.KEEP_ORIGIN);
59          assertThat(metaData.getIdentifierPatternType(), is(delegate.getIdentifierPatternType()));
60          when(delegate.getDefaultNullsOrderType()).thenReturn(NullsOrderType.HIGH);
61          assertThat(metaData.getDefaultNullsOrderType(), is(delegate.getDefaultNullsOrderType()));
62          when(delegate.getDataTypeOption()).thenReturn(mock(DialectDataTypeOption.class));
63          assertThat(metaData.getDataTypeOption(), is(delegate.getDataTypeOption()));
64          when(delegate.getDriverQuerySystemCatalogOption()).thenReturn(Optional.of(mock(DialectDriverQuerySystemCatalogOption.class)));
65          assertThat(metaData.getDriverQuerySystemCatalogOption(), is(delegate.getDriverQuerySystemCatalogOption()));
66          when(delegate.getSchemaOption()).thenReturn(mock(DialectSchemaOption.class));
67          assertThat(metaData.getSchemaOption(), is(delegate.getSchemaOption()));
68          when(delegate.getColumnOption()).thenReturn(mock(DialectColumnOption.class));
69          assertThat(metaData.getColumnOption(), is(delegate.getColumnOption()));
70          when(delegate.getIndexOption()).thenReturn(mock(DialectIndexOption.class));
71          assertThat(metaData.getIndexOption(), is(delegate.getIndexOption()));
72          when(delegate.getConnectionOption()).thenReturn(mock(DialectConnectionOption.class));
73          assertThat(metaData.getConnectionOption(), is(delegate.getConnectionOption()));
74          when(delegate.getTransactionOption()).thenReturn(mock(DialectTransactionOption.class));
75          assertThat(metaData.getTransactionOption(), is(delegate.getTransactionOption()));
76          when(delegate.getJoinOption()).thenReturn(mock(DialectJoinOption.class));
77          assertThat(metaData.getJoinOption(), is(delegate.getJoinOption()));
78          when(delegate.getPaginationOption()).thenReturn(mock(DialectPaginationOption.class));
79          assertThat(metaData.getPaginationOption(), is(delegate.getPaginationOption()));
80          when(delegate.getGeneratedKeyOption()).thenReturn(Optional.of(mock(DialectGeneratedKeyOption.class)));
81          assertThat(metaData.getGeneratedKeyOption(), is(delegate.getGeneratedKeyOption()));
82          when(delegate.getAlterTableOption()).thenReturn(Optional.of(mock(DialectAlterTableOption.class)));
83          assertThat(metaData.getAlterTableOption(), is(delegate.getAlterTableOption()));
84          when(delegate.getSQLBatchOption()).thenReturn(mock(DialectSQLBatchOption.class));
85          assertThat(metaData.getSQLBatchOption(), is(delegate.getSQLBatchOption()));
86          when(delegate.getSQLOption()).thenReturn(mock(DialectSQLOption.class));
87          assertThat(metaData.getSQLOption(), is(delegate.getSQLOption()));
88          when(delegate.getProtocolVersionOption()).thenReturn(mock(DialectProtocolVersionOption.class));
89          assertThat(metaData.getProtocolVersionOption(), is(delegate.getProtocolVersionOption()));
90          when(delegate.getFunctionOption()).thenReturn(mock(DialectFunctionOption.class));
91          assertThat(metaData.getFunctionOption(), is(delegate.getFunctionOption()));
92      }
93      
94      @Test
95      void assertBranchIdentity() {
96          assertThat(metaData.getDatabaseType(), is("BRANCH"));
97          assertThat(metaData.getType().getType(), is("BRANCH"));
98          verifyNoInteractions(delegate);
99      }
100     
101     private static final class FixtureDelegatingDialectDatabaseMetaData extends AbstractDelegatingDialectDatabaseMetaData {
102         
103         private FixtureDelegatingDialectDatabaseMetaData(final DialectDatabaseMetaData delegate) {
104             super(delegate);
105         }
106         
107         @Override
108         public String getDatabaseType() {
109             return "BRANCH";
110         }
111     }
112 }
