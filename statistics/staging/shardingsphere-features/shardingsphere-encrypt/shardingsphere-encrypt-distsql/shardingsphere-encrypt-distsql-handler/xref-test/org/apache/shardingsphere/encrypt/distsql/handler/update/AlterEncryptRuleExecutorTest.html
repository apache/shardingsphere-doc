62      @Test
63      void assertExecuteUpdateWithoutToBeAlteredRules() {
64          EncryptRule rule = mock(EncryptRule.class);
65          when(rule.getConfiguration()).thenReturn(new EncryptRuleConfiguration(Collections.emptyList(), Collections.emptyMap()));
66          assertThrows(MissingRequiredRuleException.class,
67                  () -> new DistSQLUpdateExecuteEngine(createSQLStatementWithAssistQueryAndLikeColumns(), "foo_db", mockContextManager(rule), null).executeUpdate());
68      }

70      @Test
71      void assertExecuteUpdateWithAssistQueryAndLikeColumns() throws SQLException {
72          EncryptRule rule = mock(EncryptRule.class);
73          when(rule.getAllTableNames()).thenReturn(Collections.singleton("t_encrypt"));
74          when(rule.getConfiguration()).thenReturn(createCurrentRuleConfiguration());
75          ContextManager contextManager = mockContextManager(rule);
76          MetaDataManagerPersistService metaDataManagerPersistService = contextManager.getPersistServiceFacade().getModeFacade().getMetaDataManagerService();
77          new DistSQLUpdateExecuteEngine(createSQLStatementWithAssistQueryAndLikeColumns(), "foo_db", contextManager, null).executeUpdate();
78          metaDataManagerPersistService.removeRuleConfigurationItem(any(), argThat(this::assertToBeDroppedRuleConfiguration));
79          assertDoesNotThrow(() -> metaDataManagerPersistService.alterRuleConfiguration(any(), argThat(this::assertToBeAlteredRuleConfiguration)));
80      }

94      @Test
95      void assertBuildToBeDroppedRuleConfigurationWhenAssistedAndLikeQueriesRemoved() {
96          EncryptRule rule = mock(EncryptRule.class);
97          when(rule.getConfiguration()).thenReturn(createCurrentRuleConfigurationWithGeneratedEncryptors());
98          AlterEncryptRuleExecutor executor = new AlterEncryptRuleExecutor();
99          executor.setRule(rule);
100         EncryptColumnRuleConfiguration alteredColumn = new EncryptColumnRuleConfiguration("user_id", new EncryptColumnItemRuleConfiguration("user_cipher", "foo_cipher"));
101         EncryptRuleConfiguration toBeAlteredRuleConfig = new EncryptRuleConfiguration(
102                 Collections.singleton(new EncryptTableRuleConfiguration("t_encrypt", Collections.singleton(alteredColumn))), Collections.emptyMap());
103         EncryptRuleConfiguration actual = executor.buildToBeDroppedRuleConfiguration(toBeAlteredRuleConfig);
104         assertTrue(actual.getTables().isEmpty());
105         assertThat(actual.getEncryptors().size(), is(2));
106         assertTrue(actual.getEncryptors().containsKey("foo_assisted"));
107         assertTrue(actual.getEncryptors().containsKey("foo_like"));
108     }

126     @Test
127     void assertBuildToBeDroppedRuleConfigurationWhenColumnDefinitionUnchanged() {
128         EncryptRule rule = mock(EncryptRule.class);
129         when(rule.getConfiguration()).thenReturn(createCurrentRuleConfigurationWithGeneratedEncryptors());
130         AlterEncryptRuleExecutor executor = new AlterEncryptRuleExecutor();
131         executor.setRule(rule);
132         EncryptColumnRuleConfiguration alteredColumn = new EncryptColumnRuleConfiguration("user_id", new EncryptColumnItemRuleConfiguration("user_cipher", "foo_cipher"));
133         alteredColumn.setAssistedQuery(new EncryptColumnItemRuleConfiguration("user_assisted", "foo_assisted"));
134         alteredColumn.setLikeQuery(new EncryptColumnItemRuleConfiguration("user_like", "foo_like"));
135         EncryptRuleConfiguration toBeAlteredRuleConfig = new EncryptRuleConfiguration(
136                 Collections.singleton(new EncryptTableRuleConfiguration("t_encrypt", Collections.singleton(alteredColumn))), Collections.emptyMap());
137         EncryptRuleConfiguration actual = executor.buildToBeDroppedRuleConfiguration(toBeAlteredRuleConfig);
138         assertTrue(actual.getTables().isEmpty());
139         assertTrue(actual.getEncryptors().isEmpty());
140     }

142     private ContextManager mockContextManager(final EncryptRule rule) {
143         DatabaseType databaseType = mock(DatabaseType.class);
144         ResourceMetaData resourceMetaData = mock(ResourceMetaData.class);
145         ShardingSphereDatabase database =
146                 new ShardingSphereDatabase("foo_db", databaseType, resourceMetaData, new RuleMetaData(Collections.singleton(rule)), Collections.emptyList(),
147                         new ConfigurationProperties(new Properties()));
148         ContextManager result = mock(ContextManager.class, RETURNS_DEEP_STUBS);
149         when(result.getDatabase("foo_db")).thenReturn(database);
150         return result;
151     }

153     private AlterEncryptRuleStatement createSQLStatementWithAssistQueryAndLikeColumns() {
154         EncryptColumnSegment columnSegment = new EncryptColumnSegment("user_id",
155                 new EncryptColumnItemSegment("user_cipher", new AlgorithmSegment("MD5", new Properties())),
156                 new EncryptColumnItemSegment("assisted_column", new AlgorithmSegment("MD5", new Properties())),
157                 new EncryptColumnItemSegment("like_column", new AlgorithmSegment("MD5", new Properties())));
158         EncryptRuleSegment ruleSegment = new EncryptRuleSegment("t_encrypt", Collections.singleton(columnSegment));
159         return new AlterEncryptRuleStatement(Collections.singleton(ruleSegment));
160     }
161     
162     private AlterEncryptRuleStatement createSQLStatementWithoutAssistQueryAndLikeColumns() {
163         EncryptColumnSegment columnSegment = new EncryptColumnSegment("user_id", new EncryptColumnItemSegment("user_cipher", new AlgorithmSegment("MD5", new Properties())), null, null);
164         EncryptRuleSegment ruleSegment = new EncryptRuleSegment("t_encrypt", Collections.singleton(columnSegment));
165         return new AlterEncryptRuleStatement(Collections.singleton(ruleSegment));
166     }
167     
168     private EncryptRuleConfiguration createCurrentRuleConfiguration() {
169         EncryptTableRuleConfiguration tableRuleConfig = new EncryptTableRuleConfiguration("t_encrypt", Collections.emptyList());
170         return new EncryptRuleConfiguration(new LinkedList<>(Collections.singleton(tableRuleConfig)), Collections.emptyMap());
171     }
172     
173     private EncryptRuleConfiguration createCurrentRuleConfigurationWithGeneratedEncryptors() {
174         EncryptColumnRuleConfiguration columnRuleConfig = new EncryptColumnRuleConfiguration("user_id", new EncryptColumnItemRuleConfiguration("user_cipher", "foo_cipher"));
175         columnRuleConfig.setAssistedQuery(new EncryptColumnItemRuleConfiguration("user_assisted", "foo_assisted"));
176         columnRuleConfig.setLikeQuery(new EncryptColumnItemRuleConfiguration("user_like", "foo_like"));
177         Map<String, AlgorithmConfiguration> encryptors = new HashMap<>(3, 1F);
178         encryptors.put("foo_cipher", new AlgorithmConfiguration("MD5", new Properties()));
179         encryptors.put("foo_assisted", new AlgorithmConfiguration("MD5", new Properties()));
180         encryptors.put("foo_like", new AlgorithmConfiguration("MD5", new Properties()));
181         EncryptTableRuleConfiguration tableRuleConfig = new EncryptTableRuleConfiguration("t_encrypt", Collections.singleton(columnRuleConfig));
182         return new EncryptRuleConfiguration(new LinkedList<>(Collections.singleton(tableRuleConfig)), encryptors);
183     }
184     
185     private boolean assertToBeDroppedRuleConfiguration(final EncryptRuleConfiguration actual) {
186         assertTrue(actual.getTables().isEmpty());
187         assertTrue(actual.getEncryptors().isEmpty());
188         return true;
189     }
190     
191     private boolean assertToBeAlteredRuleConfiguration(final EncryptRuleConfiguration actual) {
192         assertThat(actual.getTables().size(), is(1));
193         assertThat(actual.getTables().iterator().next().getName(), is("t_encrypt"));
194         assertThat(actual.getTables().iterator().next().getColumns().iterator().next().getName(), is("user_id"));
195         assertThat(actual.getEncryptors().size(), is(3));
196         return true;
197     }
