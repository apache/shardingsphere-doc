185     private static final Map<Integer, FirebirdCommandPacketType> FIREBIRD_COMMAND_PACKET_TYPE_CACHE = new HashMap<>();
186     
187     private final int value;
188     
189     static {
190         for (FirebirdCommandPacketType each : values()) {
191             FIREBIRD_COMMAND_PACKET_TYPE_CACHE.put(each.value, each);
192         }
193     }
194     
195     /**
196      * Value of integer.
197      *
198      * @param value integer value
199      * @return command packet type enum
200      */
201     public static FirebirdCommandPacketType valueOf(final int value) {
202         FirebirdCommandPacketType result = FIREBIRD_COMMAND_PACKET_TYPE_CACHE.get(value);
203         Preconditions.checkNotNull(result, "Cannot find '%s' in command packet type", value);
204         return result;
205     }
