20  import org.apache.shardingsphere.infra.metadata.database.schema.QualifiedDataSource;
21  import org.apache.shardingsphere.infra.spi.ShardingSphereServiceLoader;
22  import org.apache.shardingsphere.infra.util.eventbus.EventBusContext;
23  import org.apache.shardingsphere.mode.deliver.DeliverEventSubscriber;
24  import org.apache.shardingsphere.mode.spi.repository.PersistRepository;
25  import org.junit.jupiter.api.BeforeEach;
26  import org.junit.jupiter.api.Test;
27  import org.junit.jupiter.api.extension.ExtendWith;
28  import org.mockito.Mock;
29  import org.mockito.junit.jupiter.MockitoExtension;
30  
31  import static org.hamcrest.MatcherAssert.assertThat;
32  import static org.hamcrest.Matchers.not;
33  import static org.hamcrest.Matchers.sameInstance;
34  import static org.mockito.Mockito.mock;
35  import static org.mockito.Mockito.never;
36  import static org.mockito.Mockito.verify;
37  
38  @ExtendWith(MockitoExtension.class)
39  class ReadwriteSplittingQualifiedDataSourceChangedSubscriberTest {
40      
41      private ReadwriteSplittingQualifiedDataSourceChangedSubscriber subscriber;
42      
43      @Mock
44      private PersistRepository repository;
45      
46      @BeforeEach
47      void setUp() {
48          subscriber = new ReadwriteSplittingQualifiedDataSourceChangedSubscriber();
49          subscriber.setRepository(repository);
50      }
51      
52      @Test
53      void assertDeleteStorageNodeDataSourceDataSourceState() {
54          subscriber.delete(new QualifiedDataSourceDeletedEvent(new QualifiedDataSource("foo_db", "foo_group", "foo_ds")));
55          verify(repository).delete("/nodes/qualified_data_sources/foo_db.foo_group.foo_ds");
56      }
57      
58      @Test
59      void assertRepositoryIsolationAcrossEventBusContexts() {
60          PersistRepository anotherRepository = mock(PersistRepository.class);
61          EventBusContext firstEventBusContext = new EventBusContext();
62          EventBusContext secondEventBusContext = new EventBusContext();
63          DeliverEventSubscriber actualFirstSubscriber = ShardingSphereServiceLoader.getServiceInstances(DeliverEventSubscriber.class).iterator().next();
64          actualFirstSubscriber.setRepository(repository);
65          firstEventBusContext.register(actualFirstSubscriber);
66          DeliverEventSubscriber actualSecondSubscriber = ShardingSphereServiceLoader.getServiceInstances(DeliverEventSubscriber.class).iterator().next();
67          actualSecondSubscriber.setRepository(anotherRepository);
68          secondEventBusContext.register(actualSecondSubscriber);
69          assertThat(actualFirstSubscriber, not(sameInstance(actualSecondSubscriber)));
70          firstEventBusContext.post(new QualifiedDataSourceDeletedEvent(new QualifiedDataSource("foo_db", "foo_group", "foo_ds")));
71          verify(repository).delete("/nodes/qualified_data_sources/foo_db.foo_group.foo_ds");
72          verify(anotherRepository, never()).delete("/nodes/qualified_data_sources/foo_db.foo_group.foo_ds");
73          secondEventBusContext.post(new QualifiedDataSourceDeletedEvent(new QualifiedDataSource("bar_db", "bar_group", "bar_ds")));
74          verify(anotherRepository).delete("/nodes/qualified_data_sources/bar_db.bar_group.bar_ds");
75          verify(repository, never()).delete("/nodes/qualified_data_sources/bar_db.bar_group.bar_ds");
76      }
77  }
