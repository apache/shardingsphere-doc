54      @Test
55      void assertHandlePlanBroadcastRule() {
56          try (
57                  MockedConstruction<BroadcastWorkflowPlanningService> mocked = mockConstruction(BroadcastWorkflowPlanningService.class,
58                          (mock, context) -> when(mock.plan(any(), any(), any())).thenReturn(createSnapshot("planned")))) {
59              WorkflowContextFixture fixture = createWorkflowContextFixture();
60              MCPSuccessPayload actual = new PlanBroadcastRuleToolHandler().handle(fixture.requestContext, Map.of(
61                      "database", "logic_db",
62                      "tables", "t_order"));
63              assertThat(actual.toPayload().get("plan_id"), is("plan-1"));
64              ArgumentCaptor<BroadcastWorkflowRequest> requestCaptor = ArgumentCaptor.forClass(BroadcastWorkflowRequest.class);
65              verify(mocked.constructed().getFirst()).plan(eq(fixture.workflowSessionContext), eq(fixture.queryFacade), requestCaptor.capture());
66              assertThat(requestCaptor.getValue().getTargetTables(), is(List.of("t_order")));
67          }
68      }
69      
70      @Test
71      void assertHandlePlanBroadcastRuleWithArtifacts() {
72          try (
73                  MockedConstruction<BroadcastWorkflowPlanningService> ignored = mockConstruction(BroadcastWorkflowPlanningService.class,
74                          (mock, context) -> when(mock.plan(any(), any(), any())).thenReturn(createSnapshot("planned")))) {
75              MCPSuccessPayload actual = new PlanBroadcastRuleToolHandler().handle(createWorkflowContextFixture().requestContext,
76                      Map.of("database", "logic_db", "table", "t_order"));
77              Map<String, Object> actualPayload = actual.toPayload();
78              assertTrue(String.valueOf(((Map<?, ?>) ((List<?>) actualPayload.get("distsql_artifacts")).getFirst()).get("sql")).contains("CREATE BROADCAST TABLE RULE"));
79              List<?> actualResourcesToRead = (List<?>) actualPayload.get("resources_to_read");
80              List<String> actualResourceUris = extractResourceUris(actualResourcesToRead);
81              assertTrue(actualResourceUris.contains("shardingsphere://features/broadcast/databases/logic_db/rules"));
82              assertThat(findResourceKind(actualResourcesToRead, "shardingsphere://features/broadcast/databases/logic_db/rules"), is("rule"));
83              assertFalse(actualResourceUris.contains("shardingsphere://databases/logic_db/schemas/public/tables/t_order/columns"));
84              assertThat(((Map<?, ?>) actualPayload.get("proxy_topology_hint")).get("expected_runtime_view"), is("proxy_rule_distsql"));
85              Map<?, ?> actualNextAction = (Map<?, ?>) ((List<?>) actualPayload.get("next_actions")).getFirst();
86              assertThat(actualNextAction.get("tool_name"), is("database_gateway_apply_workflow"));
87              assertThat(((Map<?, ?>) actualNextAction.get("arguments")).get("execution_mode"), is("preview"));
88          }
89      }

91      private WorkflowContextSnapshot createSnapshot(final String status) {
92          BroadcastWorkflowRequest request = new BroadcastWorkflowRequest();
93          request.setDatabase("logic_db");
94          request.setTable("t_order");
95          WorkflowContextSnapshot result = new WorkflowContextSnapshot();
96          result.setPlanId("plan-1");
97          result.setWorkflowKind(BroadcastFeatureDefinition.WORKFLOW_KIND);
98          result.setStatus(status);
99          result.setRequest(request);
100         ClarifiedIntent clarifiedIntent = new ClarifiedIntent();
101         clarifiedIntent.setOperationType("create");
102         result.setClarifiedIntent(clarifiedIntent);
103         result.setInteractionPlan(InteractionPlan.create("plan-1", request, "Broadcast workflow plan.", List.of("review"), List.of("rules")));
104         result.getRuleArtifacts().add(new RuleArtifact("create", "CREATE BROADCAST TABLE RULE `t_order`"));
105         return result;
106     }
107     
108     private List<String> extractResourceUris(final List<?> resources) {
109         return resources.stream().map(each -> (String) ((Map<?, ?>) each).get("uri")).toList();
110     }
111     
112     private String findResourceKind(final List<?> resources, final String uri) {
113         for (Object each : resources) {
114             Map<?, ?> resource = (Map<?, ?>) each;
115             if (uri.equals(resource.get("uri"))) {
116                 return (String) resource.get("resource_kind");
117             }
118         }
119         return "";
120     }
121     
122     private WorkflowContextFixture createWorkflowContextFixture() {
123         MCPFeatureRequestContext result = mock(MCPFeatureRequestContext.class);
124         WorkflowSessionContext workflowSessionContext = new TestWorkflowSessionContext();
125         MCPFeatureQueryFacade queryFacade = mock(MCPFeatureQueryFacade.class);
126         MCPFeatureExecutionFacade executionFacade = mock(MCPFeatureExecutionFacade.class);
127         when(result.getSessionIdentity()).thenReturn(new MCPSessionIdentity("session-1", "", "", Map.of()));
128         when(result.getWorkflowSessionContext()).thenReturn(workflowSessionContext);
129         when(result.getQueryFacade()).thenReturn(queryFacade);
130         when(result.getExecutionFacade()).thenReturn(executionFacade);
131         return new WorkflowContextFixture(result, workflowSessionContext, queryFacade, executionFacade);
132     }

136     }
