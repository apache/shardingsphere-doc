38  /**
39   * Alter transmission rule executor.
40   */
41  public final class AlterTransmissionRuleExecutor implements DistSQLUpdateExecutor<AlterTransmissionRuleStatement> {
42      
43      private final PipelineProcessConfigurationPersistService processConfigPersistService = new PipelineProcessConfigurationPersistService();
44      
45      @Override
46      public void executeUpdate(final AlterTransmissionRuleStatement sqlStatement, final ContextManager contextManager) {
47          PipelineProcessConfiguration processConfig = convertToProcessConfiguration(sqlStatement.getProcessConfigSegment());
48          AlgorithmConfiguration streamChannel = processConfig.getStreamChannel();
49          if (null != streamChannel && !TypedSPILoader.findService(PipelineChannelCreator.class, streamChannel.getType()).isPresent()) {
50              throw new PipelineInvalidParameterException("Unknown stream channel type `" + streamChannel.getType() + "`.");
51          }
52          String jobType = TypedSPILoader.getService(PipelineJobType.class, sqlStatement.getJobTypeName()).getType();
53          processConfigPersistService.persist(new PipelineContextKey(InstanceType.PROXY), jobType, processConfig);
54      }
55      
56      private PipelineProcessConfiguration convertToProcessConfiguration(final TransmissionRuleSegment segment) {
57          return new PipelineProcessConfiguration(
58                  convertToReadConfiguration(segment.getReadSegment()), convertToWriteConfiguration(segment.getWriteSegment()), convertToAlgorithm(segment.getStreamChannel()));
59      }
60      
61      private PipelineReadConfiguration convertToReadConfiguration(final ReadOrWriteSegment readSegment) {
62          return null == readSegment
63                  ? null
64                  : new PipelineReadConfiguration(readSegment.getWorkerThread(), readSegment.getBatchSize(), readSegment.getShardingSize(), convertToAlgorithm(readSegment.getRateLimiter()));
65      }
66      
67      private PipelineWriteConfiguration convertToWriteConfiguration(final ReadOrWriteSegment writeSegment) {
68          return null == writeSegment ? null : new PipelineWriteConfiguration(writeSegment.getWorkerThread(), writeSegment.getBatchSize(), convertToAlgorithm(writeSegment.getRateLimiter()));
69      }
70      
71      private AlgorithmConfiguration convertToAlgorithm(final AlgorithmSegment algorithmSegment) {
72          return null == algorithmSegment ? null : new AlgorithmConfiguration(algorithmSegment.getName(), algorithmSegment.getProps());
73      }
74      
75      @Override
76      public Class<AlterTransmissionRuleStatement> getType() {
77          return AlterTransmissionRuleStatement.class;
78      }
79  }
