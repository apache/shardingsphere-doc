44  class ValidateRuntimeDatabaseToolHandlerTest {
45      
46      @Test
47      @SuppressWarnings("unchecked")
48      void assertHandle() {
49          try (
50                  MockedConstruction<RuntimeDatabaseValidationService> mocked =
51                          mockConstruction(RuntimeDatabaseValidationService.class, (mock, context) -> when(mock.validate(any(), any())).thenReturn(RuntimeDatabaseValidationResult.ready(
52                                  "logic_db", List.of(RuntimeDatabaseValidationCheckResult.passed("configuration", "Validated the request.")))))) {
53              MCPFeatureRequestContext requestContext = mock(MCPFeatureRequestContext.class);
54              RuntimeDatabaseConfiguration runtimeDatabaseConfig = mock(RuntimeDatabaseConfiguration.class);
55              when(requestContext.findRuntimeDatabaseConfiguration("logic_db")).thenReturn(Optional.of(runtimeDatabaseConfig));
56              MCPSuccessPayload actual = new ValidateRuntimeDatabaseToolHandler().handle(requestContext, Map.of("database", "logic_db"));
57              assertThat(actual.toPayload().get("response_mode"), is("validation"));
58              ArgumentCaptor<RuntimeDatabaseValidationRequest> requestCaptor = ArgumentCaptor.forClass(RuntimeDatabaseValidationRequest.class);
59              ArgumentCaptor<Function<String, Optional<RuntimeDatabaseConfiguration>>> resolverCaptor = ArgumentCaptor.forClass(Function.class);
60              verify(mocked.constructed().getFirst()).validate(requestCaptor.capture(), resolverCaptor.capture());
61              assertThat(requestCaptor.getValue().getDatabase(), is("logic_db"));
62              assertThat(resolverCaptor.getValue().apply("logic_db"), is(Optional.of(runtimeDatabaseConfig)));
63          }
64      }
65  }
