118     private static Map<String, Object> createInputSchemaWithQuery(final Object querySchema, final List<?> required) {
119         return Map.of("type", "object", "properties", Map.of("query", querySchema), "required", required, "additionalProperties", false);
120     }
121     
122     @ParameterizedTest(name = "{0}")
123     @MethodSource("validInputSchemaCases")
124     void assertValidateWithValidInputSchema(final String name, final Map<String, Object> inputSchema) {
125         assertDoesNotThrow(() -> MCPToolInputSchemaValidator.validate(createToolDescriptor(inputSchema)));
126     }
127     
128     private static Stream<Arguments> validInputSchemaCases() {
129         return Stream.of(
130                 Arguments.of("string default", createInputSchemaWithQuery(Map.of("type", "string", "default", "query", "description", "Query."), List.of("query"))),
131                 Arguments.of("integer default without range", createInputSchemaWithQuery(Map.of("type", "integer", "default", 3, "description", "Query."), List.of("query"))),
132                 Arguments.of("BigInteger bounds", createInputSchemaWithQuery(Map.of("type", "integer", "minimum", BigInteger.ZERO, "maximum", BigInteger.TEN, "default", BigInteger.TEN,
133                         "description", "Query."), List.of("query"))),
134                 Arguments.of("schema without type", createInputSchemaWithQuery(Map.of("description", "Query."), List.of("query"))));
135     }
136     
137     private MCPToolDescriptor createToolDescriptor(final Map<String, Object> inputSchema) {
138         Map<String, Object> outputSchema = Map.of("type", "object", "properties", Map.of("status", Map.of("type", "string", "description", "Status.")),
139                 "examples", List.of(Map.of("status", "ok")));
140         MCPToolAnnotations annotations = MCPToolAnnotations.builder().title("Test Tool").readOnlyHint(true).destructiveHint(false).idempotentHint(true).openWorldHint(true).build();
141         return new MCPToolDescriptor("database_gateway_test_tool", "Test Tool", "Run a test tool.", inputSchema, outputSchema, annotations, Map.of());
142     }
