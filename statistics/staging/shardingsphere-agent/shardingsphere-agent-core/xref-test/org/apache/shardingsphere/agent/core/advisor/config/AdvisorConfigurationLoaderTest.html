26  import java.io.IOException;

28  import java.nio.file.Path;
29  import java.util.Arrays;

37  import static org.hamcrest.MatcherAssert.assertThat;
38  import static org.hamcrest.Matchers.is;
39  import static org.junit.jupiter.api.Assertions.assertTrue;
40  
41  class AdvisorConfigurationLoaderTest {
42      
43      private static final String TARGET = YamlTargetObjectFixture.class.getName();
44      
45      private static final String ADVICE = YamlAdviceFixture.class.getName();
46      
47      @Test
48      void assertLoadAndMergeAdvisorConfigurations() throws IOException {
49          Path jarFile = Files.createTempFile("advisor-config", ".jar");
50          try (JarOutputStream jarOutputStream = new JarOutputStream(Files.newOutputStream(jarFile))) {
51              writeAdvisorResource(jarOutputStream, "fixture", "constructor");
52              writeAdvisorResource(jarOutputStream, "another", "method");
53          }
54          try (JarFile jar = new JarFile(jarFile.toFile())) {
55              Map<String, AdvisorConfiguration> actual = AdvisorConfigurationLoader.load(Collections.singleton(jar), Arrays.asList("FIXTURE", "ANOTHER", "MISSING"));
56              assertThat(actual.size(), is(1));
57              AdvisorConfiguration config = actual.get(TARGET);
58              assertThat(config.getTargetClassName(), is(TARGET));
59              assertThat(config.getAdvisors().size(), is(2));
60              assertAdvisor(config.getAdvisors(), ElementMatchers.isConstructor(), "FIXTURE");
61              assertAdvisor(config.getAdvisors(), ElementMatchers.named("call"), "ANOTHER");
62          } finally {
63              Files.deleteIfExists(jarFile);
64          }
65      }
66      
67      private void writeAdvisorResource(final JarOutputStream jarOutputStream, final String pluginType, final String pointcutType) throws IOException {
68          JarEntry entry = new JarEntry(String.join("/", "META-INF", "conf", String.join("-", pluginType, "advisors.yaml")));
69          jarOutputStream.putNextEntry(entry);
70          jarOutputStream.write(createYaml(pointcutType).getBytes());
71          jarOutputStream.closeEntry();
72      }
73      
74      private String createYaml(final String pointcutType) {
75          return String.join(System.lineSeparator(),
76                  "advisors:",
77                  "  - target: " + TARGET,
78                  "    advice: " + ADVICE,
79                  "    pointcuts:",
80                  "      - type: " + pointcutType,
81                  "        name: call");
82      }
83      
84      private void assertAdvisor(final Collection<MethodAdvisorConfiguration> actual, final ElementMatcher<?> pointcut, final String pluginType) {
85          assertTrue(actual.stream().anyMatch(each -> each.getAdviceClassName().equals(ADVICE) && each.getPluginType().equals(pluginType) && each.getPointcut().equals(pointcut)));
86      }
87  }
