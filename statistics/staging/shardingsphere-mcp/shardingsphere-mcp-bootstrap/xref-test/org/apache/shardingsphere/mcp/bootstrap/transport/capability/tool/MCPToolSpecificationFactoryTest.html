65      @Test
66      void assertCreateToolSpecifications() {
67          try (MockedStatic<ToolDefinitionRegistry> mockedToolDefinitionRegistry = mockStatic(ToolDefinitionRegistry.class)) {
68              mockedToolDefinitionRegistry.when(ToolDefinitionRegistry::getSupportedToolDescriptors).thenReturn(List.of(createToolDescriptor("database_gateway_search_metadata")));
69              MCPToolSpecificationFactory actualFactory = new MCPToolSpecificationFactory(mock(MCPRuntimeContext.class, RETURNS_DEEP_STUBS));
70              List<SyncToolSpecification> actual = actualFactory.createToolSpecifications();
71              assertThat(actual.size(), is(1));
72              assertThat(actual.getFirst().tool().name(), is("database_gateway_search_metadata"));
73              assertThat(actual.getFirst().tool().title(), is("Search Metadata"));
74              assertThat(actual.getFirst().tool().description(), is("Search database metadata."));
75              assertThat(actual.getFirst().tool().inputSchema().type(), is("object"));
76              assertThat(actual.getFirst().tool().inputSchema().required(), is(List.of("query")));
77              assertFalse(actual.getFirst().tool().inputSchema().additionalProperties());
78              assertThat(actual.getFirst().tool().inputSchema().properties().get("query"), is(Map.of("type", "string", "description", "Search query.")));
79              assertThat(actual.getFirst().tool().inputSchema().properties().get("object_types"), is(Map.of(
80                      "type", "array",
81                      "description", "Optional object-type filter.",
82                      "items", Map.of("type", "string", "description", "Object type.", "enum", List.of("TABLE", "VIEW")))));
83              assertThat(actual.getFirst().tool().outputSchema(), is(Map.of("type", "object")));
84              assertTrue(actual.getFirst().tool().annotations().readOnlyHint());
85              assertNull(actual.getFirst().tool().annotations().returnDirect());
86              assertThat(actual.getFirst().tool().meta(), is(Map.of(MCPShardingSphereMetadataKeys.RELATED_RESOURCE_URIS, List.of("shardingsphere://databases"))));
87              assertNotNull(actual.getFirst().callHandler());
88          }
89      }

143     @Test
144     void assertCreateToolSpecificationsSanitizeUnexpectedError(final LogCaptureAssertion logCaptureAssertion) {
145         try (MockedStatic<ToolDefinitionRegistry> mockedToolDefinitionRegistry = mockStatic(ToolDefinitionRegistry.class)) {
146             MCPToolDefinition toolDefinition = mockSupportedTool(mockedToolDefinitionRegistry, createToolDescriptorWithoutOutputSchema("fixture_ping"));
147             IllegalStateException expectedException = new IllegalStateException("sensitive detail");
148             mockedToolDefinitionRegistry.when(() -> ToolDefinitionRegistry.dispatch(any(MCPFeatureRuntimeRequestContext.class), eq(toolDefinition), eq(Map.of()))).thenThrow(expectedException);
149             McpError actual = assertThrows(McpError.class, () -> callTool(createToolSpecification(MCPTransportType.STDIO), createExchange(), "fixture_ping", Map.of()));
150             assertThat(actual.getJsonRpcError().code(), is(ErrorCodes.INTERNAL_ERROR));
151             assertThat(actual.getJsonRpcError().message(), is("Service is temporarily unavailable."));
152             assertFalse(String.valueOf(actual.getJsonRpcError().data()).contains("sensitive detail"));
153             logCaptureAssertion.assertErrorLog(actualException -> assertThat(((ThrowableProxy) actualException).getThrowable(), sameInstance(expectedException)));
154         }
155     }

157     @Test
158     void assertCreateToolSpecificationsRejectInvalidInputSchema() {
159         SyncToolSpecification actualSpecification = findToolSpecification(
160                 new MCPToolSpecificationFactory(createRuntimeContext(MCPTransportType.HTTP)).createToolSpecifications(), "database_gateway_search_metadata");
161         CallToolResult actual = callTool(actualSpecification, createExchange(), "database_gateway_search_metadata", Map.of("query", "order", "object_types", List.of("TABLE")));
162         Map<String, Object> actualPayload = getTextContentPayload(actual);
163         Map<?, ?> actualRecovery = (Map<?, ?>) actualPayload.get("recovery");
164         assertThat(actualPayload.get("summary"), is("object_types[0] must be one of [database, schema, table, view, column, index, storage_unit, sequence]."));
165         assertThat(actualRecovery.get("category"), is("invalid_enum_value"));
166         assertThat(actualRecovery.get("field"), is("object_types[0]"));
167         assertThat(actualRecovery.get("allowed_values"), is(List.of("database", "schema", "table", "view", "column", "index", "storage_unit", "sequence")));
168         assertThat(actualRecovery.get("suggested_arguments"), is(Map.of("query", "order")));
169         assertNull(actual.structuredContent());
170         assertTrue(actual.isError());
171     }

186     @Test
187     void assertCreateToolSpecificationsPreservesNestedArgumentsAfterClarification() {
188         try (MockedStatic<ToolDefinitionRegistry> mocked = mockStatic(ToolDefinitionRegistry.class)) {
189             String toolName = "database_gateway_plan_encrypt_rule";
190             MCPToolDefinition definition = mockSupportedTool(mocked,
191                     createPlanningToolDescriptor(toolName, Map.of("plan_id", Map.of("type", "string", "description", "Existing workflow plan identifier."))));
192             mocked.when(() -> ToolDefinitionRegistry.dispatch(any(MCPFeatureRuntimeRequestContext.class), eq(definition), any()))
193                     .thenReturn(new MCPMapPayload(createClarifyingPayload(createClarifyingQuestion("custom_properties.display-name", "string", false, "Provide display name."))),
194                             new MCPMapPayload(Map.of("status", "planned")));
195             McpSyncServerExchange exchange = createElicitationExchange(new McpSchema.ElicitResult(McpSchema.ElicitResult.Action.ACCEPT, Map.of("field_1", "foo_display")));
196             SyncToolSpecification actualSpecification = new MCPToolSpecificationFactory(createRuntimeContext(MCPTransportType.STDIO)).createToolSpecifications().getFirst();
197             CallToolResult actual = actualSpecification.callHandler().apply(exchange, new McpSchema.CallToolRequest(toolName, Map.of("custom_properties", Map.of("foo_key", "foo_value"))));
198             assertThat(actual.structuredContent(), is(Map.of("status", "planned")));
199             mocked.verify(() -> ToolDefinitionRegistry.dispatch(any(MCPFeatureRuntimeRequestContext.class), eq(definition),
200                     eq(Map.of("plan_id", "plan-1", "custom_properties", Map.of("foo_key", "foo_value", "display-name", "foo_display")))));
201         }
