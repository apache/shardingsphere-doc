117     private DataSet getDataSet(final int[] actualUpdateCounts, final Collection<DataSet> dataSets, final String sql) {
118         Collection<DataSet> result = new LinkedList<>();
119         assertThat(actualUpdateCounts.length, is(dataSets.size()));
120         int count = 0;
121         for (DataSet each : dataSets) {
122             if (Statement.SUCCESS_NO_INFO != actualUpdateCounts[count]) {
123                 assertThat(actualUpdateCounts[count], is(each.getUpdateCount()));
124             }
125             result.add(each);
126             count++;
127         }
128         return mergeDataSets(result, sql);
129     }
130     
131     private DataSet mergeDataSets(final Collection<DataSet> dataSets, final String sql) {
132         DataSet result = new DataSet();
133         Set<DataSetRow> existedRows = new HashSet<>();
134         for (DataSet each : dataSets) {
135             mergeMetaData(each, result);
136             if (sql.trim().toUpperCase().startsWith("DELETE")) {
137                 mergeDeleteRow(each, result, existedRows);
138                 continue;
139             } else if (sql.trim().toUpperCase().startsWith("UPDATE")) {
140                 mergeUpdateRow(each, result, existedRows);
141                 continue;
142             }
143             mergeRow(each, result, existedRows);
144         }
145         sortRow(result);
146         return result;
147     }
148     
149     private void mergeDeleteRow(final DataSet original, final DataSet dist, final Set<DataSetRow> existedRows) {
150         Collection<DataSetRow> removedRows = getDifferentRows(existedRows, new HashSet<>(original.getRows()));
151         mergeRow(original, dist, existedRows);
152         dist.getRows().removeAll(removedRows);
153     }
154     
155     private Collection<DataSetRow> getDifferentRows(final Set<DataSetRow> existedRows, final Set<DataSetRow> originalRows) {
156         if (existedRows.isEmpty()) {
157             return Collections.emptyList();
158         }
159         Collection<DataSetRow> result = new HashSet<>();
160         result.addAll(Sets.difference(existedRows, originalRows));
161         result.addAll(Sets.difference(originalRows, existedRows));
162         return result;
163     }
164     
165     private void mergeUpdateRow(final DataSet original, final DataSet dist, final Set<DataSetRow> existedRows) {
166         Collection<DataSetRow> removedRows = getDifferentRows(existedRows, new HashSet<>(original.getRows())).stream().filter(each -> !each.isUpdated()).collect(Collectors.toList());
167         mergeRow(original, dist, existedRows);
168         dist.getRows().removeAll(removedRows);
169     }
170     
171     private void mergeMetaData(final DataSet original, final DataSet dist) {
172         if (dist.getMetaDataList().isEmpty()) {
173             dist.getMetaDataList().addAll(original.getMetaDataList());
174         }
175     }
176     
177     private void mergeRow(final DataSet original, final DataSet dist, final Set<DataSetRow> existedRows) {
178         for (DataSetRow each : original.getRows()) {
179             if (existedRows.add(each)) {
180                 dist.getRows().add(each);
181             }
182         }
183     }
184     
185     private void sortRow(final DataSet dataSet) {
186         dataSet.getRows().sort(Comparator.comparingLong(o -> Long.parseLong(o.splitValues(",").get(0))));
187     }

190         assertThat(actualUpdateCount, is(context.getDataSet().getUpdateCount()));
191         for (DataSetMetaData each : context.getDataSet().getMetaDataList()) {
192             assertDataSet(context, each, testParam);
193         }

196     private void assertDataSet(final SQLE2EITContext context, final DataSetMetaData expectedDataSetMetaData, final AssertionTestParameter testParam) throws SQLException {
197         for (String each : InlineExpressionParserFactory.newInstance(expectedDataSetMetaData.getDataNodes()).splitAndEvaluate()) {
198             DataNode dataNode = new DataNode(each);
199             DataSource dataSource = getEnvironmentEngine().getActualDataSourceMap().get(dataNode.getDataSourceName());
200             DatabaseType databaseType = testParam.getDatabaseType();
201             try (
202                     Connection connection = dataSource.getConnection();
203                     PreparedStatement preparedStatement = connection.prepareStatement(
204                             generateFetchActualDataSQL(getEnvironmentEngine().getActualDataSourceMap(), dataNode, databaseType))) {
205                 assertDataSet(preparedStatement, expectedDataSetMetaData, context.getDataSet().findRows(dataNode), databaseType);
206             }
207         }
208     }

210     private void assertDataSet(final PreparedStatement actualPreparedStatement, final DataSetMetaData expectedDataSetMetaData, final List<DataSetRow> expectedDataSetRows,
211                                final DatabaseType databaseType) throws SQLException {
212         try (ResultSet actualResultSet = actualPreparedStatement.executeQuery()) {
213             assertMetaData(actualResultSet.getMetaData(), expectedDataSetMetaData.getColumns());
214             assertRows(actualResultSet, expectedDataSetRows, databaseType);
215         }

218     protected final void assertDataSet(final int[] actualUpdateCounts, final CaseTestParameter testParam) throws SQLException {
219         Collection<DataSet> dataSets = new LinkedList<>();
220         for (SQLE2ETestCaseAssertion each : testParam.getTestCaseContext().getTestCase().getAssertions()) {
221             dataSets.add(DataSetLoader.load(testParam.getTestCaseContext().getParentPath(), testParam.getScenario(), testParam.getDatabaseType(), testParam.getMode(), each.getExpectedDataFile()));
222         }
223         DataSet dataSet = getDataSet(actualUpdateCounts, dataSets, testParam.getTestCaseContext().getTestCase().getSql());
224         for (DataSetMetaData each : dataSet.getMetaDataList()) {
225             assertDataSet(each, testParam, dataSet);
226         }
227     }

243     private String generateFetchActualDataSQL(final Map<String, DataSource> actualDataSourceMap, final DataNode dataNode, final DatabaseType databaseType) throws SQLException {
244         String tableName = dataNode.getTableName();
245         Optional<String> primaryKeyColumnName = DialectDatabaseAssertionMetaDataFactory.getPrimaryKeyColumnName(databaseType, actualDataSourceMap.get(dataNode.getDataSourceName()), tableName);
246         return primaryKeyColumnName.map(optional -> String.format("SELECT * FROM %s ORDER BY %s ASC", tableName, optional))
247                 .orElseGet(() -> DatabaseTypedSPILoader.findService(DialectQueryBehaviorProvider.class, databaseType)
248                         .flatMap(DialectQueryBehaviorProvider::getFallbackOrderByWhenNoPrimaryKey).map(optional -> String.format("SELECT * FROM %s ORDER BY %s", tableName, optional))
249                         .orElseGet(() -> String.format("SELECT * FROM %s", tableName)));
250     }

252     private void assertMetaData(final ResultSetMetaData actual, final Collection<DataSetColumn> expected) throws SQLException {
253         assertThat(actual.getColumnCount(), is(expected.size()));
254         int index = 1;
255         for (DataSetColumn each : expected) {
256             assertThat(actual.getColumnLabel(index++).toUpperCase(), is(each.getName().toUpperCase()));
257         }
258     }

260     private void assertRows(final ResultSet actual, final List<DataSetRow> expected, final DatabaseType databaseType) throws SQLException {
261         int rowCount = 0;
262         while (actual.next()) {
263             int columnIndex = 1;
264             for (String each : expected.get(rowCount).splitValues(DATA_COLUMN_DELIMITER)) {
265                 assertValue(actual, columnIndex, each, databaseType);
266                 columnIndex++;
267             }
268             rowCount++;
269         }
270         assertThat("Size of actual result set is different with size of expected data set rows.", rowCount, is(expected.size()));
271     }

273     private void assertValue(final ResultSet actual, final int columnIndex, final String expected, final DatabaseType databaseType) throws SQLException {
274         if (SQLE2EITContext.NOT_VERIFY_FLAG.equals(expected)) {
275             return;
276         }
277         if (Types.DATE == actual.getMetaData().getColumnType(columnIndex)) {
278             assertDateTypeValue(actual, columnIndex, expected);
279         } else if (Arrays.asList(Types.TIME, Types.TIME_WITH_TIMEZONE).contains(actual.getMetaData().getColumnType(columnIndex))) {
280             if (isNullValue(expected)) {
281                 assertNull(actual.getTime(columnIndex));
282                 return;
283             }
284             assertThat(DateTimeFormatterFactory.getTimeFormatter().format(actual.getTime(columnIndex).toLocalTime()), is(expected));
285         } else if (Arrays.asList(Types.TIMESTAMP, Types.TIMESTAMP_WITH_TIMEZONE).contains(actual.getMetaData().getColumnType(columnIndex))) {
286             if ("Oracle".equals(databaseType.getType()) && "DATE".equalsIgnoreCase(actual.getMetaData().getColumnTypeName(columnIndex)) || "openGauss".equals(databaseType.getType())) {
287                 assertDateTypeValue(actual, columnIndex, expected);
288             } else {
289                 assertTimestampTypeValue(actual, columnIndex, expected);
290             }
291         } else if (Types.CHAR == actual.getMetaData().getColumnType(columnIndex)
292                 && ("PostgreSQL".equals(databaseType.getType()) || "openGauss".equals(databaseType.getType())
293                         || "Oracle".equals(databaseType.getType()))) {
294             assertThat(String.valueOf(actual.getObject(columnIndex)).trim(), is(expected));
295         } else if (isPostgreSQLOrOpenGaussMoney(actual.getMetaData().getColumnTypeName(columnIndex), databaseType)) {
296             assertThat(actual.getString(columnIndex), is(expected));
297         } else if (Arrays.asList(Types.BINARY, Types.VARBINARY, Types.LONGVARBINARY).contains(actual.getMetaData().getColumnType(columnIndex))) {
298             byte[] actualBytes = actual.getBytes(columnIndex);
299             byte[] expectedBytes = expected.getBytes(StandardCharsets.UTF_8);
300             if (null != actualBytes) {
301                 assertThat(actualBytes.length >= expectedBytes.length ? Arrays.copyOf(actualBytes, expectedBytes.length) : actualBytes, is(expectedBytes));
302             }
303         } else if (Types.BLOB == actual.getMetaData().getColumnType(columnIndex)) {
304             Object actualValue = actual.getObject(columnIndex);
305             if (actualValue instanceof Blob) {
306                 Blob blob = (Blob) actualValue;
307                 assertThat(blob.getBytes(1, (int) blob.length()), is(expected.getBytes(StandardCharsets.UTF_8)));
308             } else if (actualValue instanceof byte[]) {
309                 assertThat(actualValue, is(expected.getBytes(StandardCharsets.UTF_8)));
310             } else {
311                 assertThat(String.valueOf(actualValue), is(expected));
312             }
313         } else if (Types.CLOB == actual.getMetaData().getColumnType(columnIndex)) {
314             assertThat(getClobValue((Clob) actual.getObject(columnIndex)), is(expected));
315         } else if (Types.NCLOB == actual.getMetaData().getColumnType(columnIndex)) {
316             Object actualValue = actual.getObject(columnIndex);
317             if (actualValue instanceof NClob) {
318                 NClob nclob = (NClob) actualValue;
319                 assertThat(nclob.getSubString(1, (int) nclob.length()), is(expected));
320             } else {
321                 assertThat(String.valueOf(actualValue), is(expected));
322             }
323         } else if (Arrays.asList(Types.TINYINT, Types.SMALLINT, Types.INTEGER, Types.BIGINT).contains(actual.getMetaData().getColumnType(columnIndex))) {
324             if (isNullValue(expected)) {
325                 assertNull(actual.getObject(columnIndex));
326                 return;
327             }
328             assertThat(String.valueOf(actual.getObject(columnIndex)), is(expected));
329         } else {
330             assertThat(String.valueOf(actual.getObject(columnIndex)), is(expected));
331         }
332     }

334     private void assertDateTypeValue(final ResultSet actual, final int columnIndex, final String expected) throws SQLException {
335         if (isNullValue(expected)) {
336             assertNull(actual.getDate(columnIndex));
337             return;
338         }
339         assertThat(DateTimeFormatterFactory.getDateFormatter().format(actual.getDate(columnIndex).toLocalDate()), is(expected));
340     }

342     private void assertTimestampTypeValue(final ResultSet actual, final int columnIndex, final String expected) throws SQLException {
343         if (isNullValue(expected)) {
344             assertNull(actual.getTimestamp(columnIndex));
345             return;
346         }
347         assertThat(DateTimeFormatterFactory.getShortMillisDatetimeFormatter().format(actual.getTimestamp(columnIndex).toLocalDateTime()), is(expected));
348     }

350     private boolean isNullValue(final String value) {
351         return null == value || "NULL".equalsIgnoreCase(value);
352     }
353     
354     private boolean isPostgreSQLOrOpenGaussMoney(final String columnTypeName, final DatabaseType databaseType) {
355         return "money".equalsIgnoreCase(columnTypeName) && ("PostgreSQL".equals(databaseType.getType()) || "openGauss".equals(databaseType.getType()));
356     }
357     
358     @SneakyThrows(SQLException.class)
359     private static String getClobValue(final Clob value) {
360         return value.getSubString(1, (int) value.length());
361     }

363     protected void assertGeneratedKeys(final AssertionTestParameter testParam, final ResultSet generatedKeys, final DatabaseType databaseType) throws SQLException {
364         DataSet generatedKeyDataSet = null == testParam.getAssertion() || null == testParam.getAssertion().getExpectedGeneratedKeyDataFile()
365                 ? null
366                 : DataSetLoader.load(
367                         testParam.getTestCaseContext().getParentPath(), testParam.getScenario(), testParam.getDatabaseType(), testParam.getMode(),
368                         testParam.getAssertion().getExpectedGeneratedKeyDataFile());
369         if (null == generatedKeyDataSet) {
370             return;
371         }
372         assertThat("Only support single table for DML.", generatedKeyDataSet.getMetaDataList().size(), is(1));
373         assertMetaData(generatedKeys.getMetaData(), generatedKeyDataSet.getMetaDataList().get(0).getColumns());
374         assertRows(generatedKeys, generatedKeyDataSet.getRows(), databaseType);
375     }

377     protected void executeInitSQLs(final SQLE2ETestCaseAssertion assertion) throws SQLException {
378         if (null == assertion.getInitialSQL()) {
379             return;
380         }
381         try (Connection connection = getEnvironmentEngine().getTargetDataSource(assertion.getTargetDataSourceName()).getConnection()) {
382             executeInitSQLs(assertion, connection);
383         }
384     }
385     
386     private void executeInitSQLs(final SQLE2ETestCaseAssertion assertion, final Connection connection) throws SQLException {
387         if (null == assertion.getInitialSQL().getSql()) {
388             return;
389         }
390         for (String each : Splitter.on(";").trimResults().omitEmptyStrings().splitToList(assertion.getInitialSQL().getSql())) {
391             try (PreparedStatement preparedStatement = connection.prepareStatement(each)) {
392                 preparedStatement.executeUpdate();
393             }
394             waitCompleted();
395         }
396     }
397     
398     private void waitCompleted() {
399         Awaitility.await().pollDelay(1500L, TimeUnit.MILLISECONDS).until(() -> true);
400     }
401     
402     protected void executeDestroySQLs(final SQLE2ETestCaseAssertion assertion) throws SQLException {
403         if (null != assertion.getDestroySQL()) {
404             try (Connection connection = getEnvironmentEngine().getTargetDataSource(assertion.getTargetDataSourceName()).getConnection()) {
405                 executeDestroySQLs(assertion, connection);
406             }
407         }
408     }

410     private void executeDestroySQLs(final SQLE2ETestCaseAssertion assertion, final Connection connection) throws SQLException {
411         if (null == assertion.getDestroySQL().getSql()) {
412             return;
413         }
414         for (String each : Splitter.on(";").trimResults().omitEmptyStrings().splitToList(assertion.getDestroySQL().getSql())) {
415             try (PreparedStatement preparedStatement = connection.prepareStatement(each)) {
416                 preparedStatement.executeUpdate();
417             }
418             waitCompleted();
419         }
