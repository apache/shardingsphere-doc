85      @Test
86      void assertBroadcastWorkflowCanBeAppliedAndValidated() throws IOException, InterruptedException {
87          try (MCPInteractionClient interactionClient = createOpenedInteractionClient()) {
88              Map<String, Object> arguments = Map.of("database", getLogicalDatabaseName(), "operation_type", "create", "tables", "orders");
89              Map<String, Object> actualPlanResponse = interactionClient.call(BROADCAST_PLAN_TOOL_NAME, arguments);
90              assertThat(String.valueOf(actualPlanResponse.get("status")), is("planned"));
91              String planId = String.valueOf(actualPlanResponse.get("plan_id"));
92              Map<String, Object> actualApplyResponse = interactionClient.call(APPLY_TOOL_NAME,
93                      createReviewThenExecuteArguments(planId, getApprovedSteps(previewWorkflow(interactionClient, planId))));
94              assertApplyCompleted(actualApplyResponse);
95              assertThat(getObjectListOrEmpty(actualApplyResponse.get("step_results")).size(), is(1));
96              assertValidationPassed(interactionClient.call(VALIDATE_TOOL_NAME, Map.of("plan_id", planId)));
97              List<Map<String, Object>> actualRules = getPayloadItems(interactionClient.readResource(String.format(BROADCAST_RULES_RESOURCE_URI, getLogicalDatabaseName())));
98              assertThat(actualRules.stream().map(each -> String.valueOf(each.get("broadcast_table"))).toList(), hasItem("orders"));
99              assertDuplicateCreateFails(interactionClient, BROADCAST_PLAN_TOOL_NAME, arguments);
100             applyAndValidateWorkflow(interactionClient, BROADCAST_PLAN_TOOL_NAME,
101                     Map.of("database", getLogicalDatabaseName(), "operation_type", "drop", "tables", "orders"));
102             assertFalse(getPayloadItems(interactionClient.readResource(String.format(BROADCAST_RULES_RESOURCE_URI, getLogicalDatabaseName())))
103                     .stream().anyMatch(each -> "orders".equalsIgnoreCase(String.valueOf(each.get("broadcast_table")))));
104         }

107     @Test
108     void assertReadwriteSplittingWorkflowCanBeAppliedAndValidated() throws IOException, InterruptedException {
109         try (MCPInteractionClient interactionClient = createOpenedInteractionClient()) {
110             assertStorageUnitsExposeWorkflowTopology(interactionClient);
111             planApplyAndValidateWorkflow(interactionClient, READWRITE_SPLITTING_PLAN_TOOL_NAME,
112                     Map.of("database", getLogicalDatabaseName(), "operation_type", "create", "rule", "readwrite_ds", "write_storage_unit", "ds_0",
113                             "read_storage_units", "ds_1", "transactional_read_query_strategy", "DYNAMIC", "load_balancer_type", "ROUND_ROBIN"));
114             List<Map<String, Object>> actualRules = getPayloadItems(interactionClient.readResource(
115                     String.format(READWRITE_SPLITTING_RULES_RESOURCE_URI, getLogicalDatabaseName())));
116             Map<String, Object> actualRule = findItemByField(actualRules, "name", "readwrite_ds");
117             assertThat(String.valueOf(actualRule.get("write_storage_unit_name")), is("ds_0"));
118             assertTrue(String.valueOf(actualRule.get("read_storage_unit_names")).contains("ds_1"));
119             assertThat(String.valueOf(actualRule.get("transactional_read_query_strategy")).toUpperCase(Locale.ENGLISH), is("DYNAMIC"));
120             assertThat(String.valueOf(actualRule.get("load_balancer_type")).toUpperCase(Locale.ENGLISH), is("ROUND_ROBIN"));
121             Map<String, Object> actualStatusPlan = interactionClient.call(READWRITE_SPLITTING_STATUS_PLAN_TOOL_NAME,
122                     Map.of("database", getLogicalDatabaseName(), "rule", "readwrite_ds", "storage_unit", "ds_1", "target_status", "disable"));
123             assertThat(String.valueOf(actualStatusPlan.get("status")), is("failed"));
124             assertThat(getIssueCodes(actualStatusPlan), hasItem(WorkflowIssueCode.CLUSTER_MODE_REQUIRED));
125             Map<String, Object> actualIssueDetails = getObjectOrEmpty(getObjectListOrEmpty(actualStatusPlan.get("issues")).getFirst().get("details"));
126             assertThat(actualIssueDetails.get("required_mode"), is("Cluster"));
127             assertThat(actualIssueDetails.get("actual_mode"), is("Standalone"));
128             assertTrue(getObjectListOrEmpty(actualStatusPlan.get("distsql_artifacts")).isEmpty());
129             assertThat(getObjectListOrEmpty(actualStatusPlan.get("next_actions")).getFirst().get("type"), is("terminal"));
130             assertModelFacingPayloadContract(actualStatusPlan);
131         }

357     private void applyAndValidateWorkflow(final MCPInteractionClient interactionClient, final String toolName,
358                                           final Map<String, Object> arguments) throws IOException, InterruptedException {
359         Map<String, Object> actualPlanResponse = planWorkflow(interactionClient, toolName, arguments);
360         String planId = String.valueOf(actualPlanResponse.get("plan_id"));
361         Map<String, Object> actualApplyResponse = applyReviewedWorkflow(interactionClient, planId);
362         assertApplyCompleted(actualApplyResponse);
363         assertThat(getObjectListOrEmpty(actualApplyResponse.get("step_results")).size(), is(1));
364         assertValidationPassed(interactionClient.call(VALIDATE_TOOL_NAME, Map.of("plan_id", planId)));
365     }
