73  /**
74   * Proxy backend handler factory.
75   */
76  @NoArgsConstructor(access = AccessLevel.PRIVATE)
77  public final class ProxyBackendHandlerFactory {
78      
79      private static final Collection<Class<? extends SQLStatement>> UNSUPPORTED_STANDARD_SQL_STATEMENT_TYPES = Arrays.asList(DCLStatement.class, RenameTableStatement.class);
80      
81      /**
82       * Create new instance of backend handler.
83       *
84       * @param databaseType database type
85       * @param sql SQL to be executed
86       * @param sqlStatement SQL statement
87       * @param connectionSession connection session
88       * @param hintValueContext hint query context
89       * @return created instance
90       * @throws SQLException SQL exception
91       */
92      public static ProxyBackendHandler newInstance(final DatabaseType databaseType, final String sql, final SQLStatement sqlStatement,
93                                                    final ConnectionSession connectionSession, final HintValueContext hintValueContext) throws SQLException {
94          if (sqlStatement instanceof EmptyStatement) {
95              return new SkipProxyBackendHandler(sqlStatement);
96          }
97          ShardingSphereMetaData metaData = ProxyContext.getInstance().getContextManager().getMetaDataContexts().getMetaData();
98          SQLStatementContext sqlStatementContext = sqlStatement instanceof DistSQLStatement
99                  ? new DistSQLStatementContext((DistSQLStatement) sqlStatement)
100                 : new SQLBindEngine(metaData, connectionSession.getCurrentDatabaseName(), hintValueContext).bind(sqlStatement);
101         QueryContext queryContext = new QueryContext(sqlStatementContext, sql, Collections.emptyList(), hintValueContext, connectionSession.getConnectionContext(), metaData);
102         return newInstance(databaseType, queryContext, connectionSession, false);
103     }
104     
105     /**
106      * Create new instance of backend handler.
107      *
108      * @param databaseType database type
109      * @param queryContext query context
110      * @param connectionSession connection session
111      * @param preferPreparedStatement use prepared statement as possible
112      * @return created instance
113      * @throws SQLException SQL exception
114      */
115     public static ProxyBackendHandler newInstance(final DatabaseType databaseType, final QueryContext queryContext,
116                                                   final ConnectionSession connectionSession, final boolean preferPreparedStatement) throws SQLException {
117         connectionSession.setQueryContext(queryContext);
118         SQLStatementContext sqlStatementContext = queryContext.getSqlStatementContext();
119         SQLStatement sqlStatement = sqlStatementContext.getSqlStatement();
120         checkAllowedSQLStatementWhenTransactionFailed(databaseType, sqlStatement, connectionSession.getConnectionContext().getTransactionContext());
121         checkSupportedSQLStatement(databaseType, sqlStatement);
122         checkClusterState(databaseType, sqlStatement);
123         if (sqlStatement instanceof EmptyStatement) {
124             return new SkipProxyBackendHandler(sqlStatement);
125         }
126         if (sqlStatement instanceof DistSQLStatement) {
127             checkSupportedDistSQLStatementInTransaction(sqlStatement, connectionSession);
128             return DistSQLProxyBackendHandlerFactory.newInstance((DistSQLStatement) sqlStatement, queryContext, connectionSession);
129         }
130         handleAutoCommit(sqlStatement, connectionSession);
131         if (sqlStatement instanceof TCLStatement) {
132             return TCLProxyBackendHandlerFactory.newInstance(queryContext, connectionSession);
133         }
134         Optional<ProxyBackendHandler> databaseAdminHandler = DatabaseAdminProxyBackendHandlerFactory.newInstance(
135                 databaseType, sqlStatementContext, connectionSession, queryContext.getSql(), queryContext.getParameters());
136         if (databaseAdminHandler.isPresent()) {
137             return databaseAdminHandler.get();
138         }
139         Optional<ProxyBackendHandler> databaseOperateHandler = findDatabaseOperateProxyBackendHandler(sqlStatement, connectionSession);
140         if (databaseOperateHandler.isPresent()) {
141             return databaseOperateHandler.get();
142         }
143         String databaseName = sqlStatementContext.getTablesContext().getDatabaseName().isPresent()
144                 ? sqlStatementContext.getTablesContext().getDatabaseName().get()
145                 : connectionSession.getUsedDatabaseName();
146         if (null == databaseName) {
147             return DatabaseProxyBackendHandlerFactory.newInstance(queryContext, connectionSession, preferPreparedStatement);
148         }
149         checkSQLExecution(queryContext, connectionSession, databaseName);
150         return DatabaseAdminProxyBackendHandlerFactory.newInstance(databaseType, sqlStatementContext, connectionSession)
151                 .orElseGet(() -> DatabaseProxyBackendHandlerFactory.newInstance(queryContext, connectionSession, preferPreparedStatement));
152     }
153     
154     private static void checkAllowedSQLStatementWhenTransactionFailed(final DatabaseType databaseType, final SQLStatement sqlStatement,
155                                                                       final TransactionConnectionContext transactionContext) throws SQLException {
156         if (transactionContext.isExceptionOccur()
157                 && DatabaseTypedSPILoader.getService(DialectDatabaseMetaData.class, databaseType).getTransactionOption().isAllowCommitAndRollbackOnlyWhenTransactionFailed()) {
158             ShardingSpherePreconditions.checkState(sqlStatement instanceof CommitStatement || sqlStatement instanceof RollbackStatement,
159                     () -> new SQLFeatureNotSupportedException("Current transaction is aborted, commands ignored until end of transaction block."));
160         }
161     }
162     
163     private static void checkSupportedSQLStatement(final DatabaseType databaseType, final SQLStatement sqlStatement) {
164         ShardingSpherePreconditions.checkState(isSupportedSQLStatement(databaseType, sqlStatement),
165                 () -> new UnsupportedSQLOperationException(String.format("unsupported SQL statement `%s`", sqlStatement.getClass().getSimpleName())));
166     }
167     
168     private static boolean isSupportedSQLStatement(final DatabaseType databaseType, final SQLStatement sqlStatement) {
169         Collection<Class<? extends SQLStatement>> unsupportedDialectSQLStatementTypes = DatabaseTypedSPILoader.findService(DialectProxyStateSupportedSQLProvider.class, databaseType)
170                 .map(DialectProxyStateSupportedSQLProvider::getUnsupportedSQLStatementTypesOnReadyState).orElse(Collections.emptyList());
171         return new ProxySQLSupportedJudgeEngine(
172                 Collections.emptyList(), Collections.emptyList(), UNSUPPORTED_STANDARD_SQL_STATEMENT_TYPES, unsupportedDialectSQLStatementTypes).isSupported(sqlStatement);
173     }
174     
175     private static void checkClusterState(final DatabaseType databaseType, final SQLStatement sqlStatement) {
176         ShardingSphereState currentState = ProxyContext.getInstance().getContextManager().getStateContext().getState();
177         TypedSPILoader.findService(ProxyClusterStateChecker.class, currentState).ifPresent(optional -> optional.check(sqlStatement, databaseType));
178     }
179     
180     private static void checkSupportedDistSQLStatementInTransaction(final SQLStatement sqlStatement, final ConnectionSession connectionSession) {
181         ShardingSpherePreconditions.checkState(!connectionSession.getTransactionStatus().isInTransaction() || isSupportedDistSQLStatementInTransaction(sqlStatement),
182                 () -> new UnsupportedSQLOperationException("Non-query DistSQL is not supported within a transaction"));
183     }
184     
185     private static boolean isSupportedDistSQLStatementInTransaction(final SQLStatement sqlStatement) {
186         return sqlStatement instanceof RQLStatement || sqlStatement instanceof QueryableRALStatement || sqlStatement instanceof RULStatement;
187     }
188     
189     private static void handleAutoCommit(final SQLStatement sqlStatement, final ConnectionSession connectionSession) {
190         if (AutoCommitUtils.isNeedStartTransaction(sqlStatement)) {
191             connectionSession.getDatabaseConnectionManager().handleAutoCommit();
192         }
193     }
194     
195     private static Optional<ProxyBackendHandler> findDatabaseOperateProxyBackendHandler(final SQLStatement sqlStatement, final ConnectionSession connectionSession) {
196         return sqlStatement instanceof CreateDatabaseStatement || sqlStatement instanceof DropDatabaseStatement
197                 ? Optional.of(DatabaseOperateProxyBackendHandlerFactory.newInstance(sqlStatement, connectionSession))
198                 : Optional.empty();
199     }
200     
201     private static void checkSQLExecution(final QueryContext queryContext, final ConnectionSession connectionSession, final String databaseName) {
202         Grantee grantee = connectionSession.getConnectionContext().getGrantee();
203         ShardingSphereDatabase database = queryContext.getMetaData().getDatabase(databaseName);
204         for (SQLExecutionChecker each : ShardingSphereServiceLoader.getServiceInstances(SQLExecutionChecker.class)) {
205             each.check(grantee, queryContext, database);
206         }
207     }
