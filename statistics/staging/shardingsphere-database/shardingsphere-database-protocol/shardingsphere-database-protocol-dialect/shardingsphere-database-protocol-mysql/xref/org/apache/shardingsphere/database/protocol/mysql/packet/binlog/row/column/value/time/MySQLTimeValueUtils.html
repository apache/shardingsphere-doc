18  package org.apache.shardingsphere.database.protocol.mysql.packet.binlog.row.column.value.time;
19  
20  import lombok.AccessLevel;
21  import lombok.NoArgsConstructor;
22  
23  import java.util.Locale;
24  
25  /**
26   * Time value utility class of MySQL.
27   */
28  @NoArgsConstructor(access = AccessLevel.PRIVATE)
29  public final class MySQLTimeValueUtils {
30      
31      public static final String ZERO_OF_TIME = "00:00:00";
32      
33      public static final String ZERO_OF_DATE = "0000-00-00";
34      
35      public static final String YEAR_OF_ZERO = "0000";
36      
37      public static final String DATETIME_OF_ZERO = "0000-00-00 00:00:00";
38      
39      /**
40       * Judge whether a date has both a month and a day, which is what {@link java.time.LocalDate} requires.
41       * MySQL stores a zero month or a zero day whenever NO_ZERO_IN_DATE is not in sql_mode.
42       *
43       * @param month month
44       * @param day day
45       * @return whether the date is complete
46       */
47      static boolean isCompleteDate(final int month, final int day) {
48          return month > 0 && day > 0;
49      }
50      
51      /**
52       * Format a date that {@link java.time.LocalDate} cannot hold, in the text form MySQL prints it in, with ASCII digits whatever the default locale.
53       *
54       * @param year year
55       * @param month month
56       * @param day day
57       * @return date in MySQL text form
58       */
59      static String formatIncompleteDate(final int year, final int month, final int day) {
60          return String.format(Locale.ROOT, "%04d-%02d-%02d", year, month, day);
61      }
62      
63      /**
64       * Format a datetime whose date part {@link java.time.LocalDate} cannot hold, in the text form MySQL prints it in, with ASCII digits whatever the default locale.
65       *
66       * @param year year
67       * @param month month
68       * @param day day
69       * @param hour hour
70       * @param minute minute
71       * @param second second
72       * @param nanos nanoseconds, always a whole number of microseconds
73       * @return datetime in MySQL text form
74       */
75      static String formatIncompleteDatetime(final int year, final int month, final int day, final int hour, final int minute, final int second, final int nanos) {
76          String result = String.format(Locale.ROOT, "%04d-%02d-%02d %02d:%02d:%02d", year, month, day, hour, minute, second);
77          return 0 == nanos ? result : result + String.format(Locale.ROOT, ".%06d", nanos / 1000);
78      }
79  }
