56  /**
57   * Authentication engine for Firebird.
58   */
59  public final class FirebirdAuthenticationEngine implements AuthenticationEngine {
60      
61      private FirebirdSRPAuthenticationData authData;
62      
63      private FirebirdAuthenticationMethod plugin;
64      
65      private AuthenticationResult currentAuthResult;
66      
67      private int connectionId;
68      
69      @Override
70      public int handshake(final ChannelHandlerContext context) {
71          connectionId = ConnectionIdGenerator.getInstance().nextId();
72          FirebirdTransactionIdGenerator.getInstance().registerConnection(connectionId);
73          FirebirdStatementIdGenerator.getInstance().registerConnection(connectionId);
74          FirebirdFetchStatementCache.getInstance().registerConnection(connectionId);
75          return connectionId;
76      }
77      
78      @Override
79      public AuthenticationResult authenticate(final ChannelHandlerContext context, final PacketPayload payload) {
80          payload.getByteBuf().resetReaderIndex();
81          AuthorityRule rule = ProxyContext.getInstance().getContextManager().getMetaDataContexts().getMetaData().getGlobalRuleMetaData().getSingleRule(AuthorityRule.class);
82          FirebirdPacketPayload fdbPacketPayload = (FirebirdPacketPayload) payload;
83          FirebirdCommandPacketType type = FirebirdCommandPacketType.valueOf(fdbPacketPayload.readInt4());
84          switch (type) {
85              case CONNECT:
86                  return processConnect(context, fdbPacketPayload, rule);
87              case ATTACH:
88                  return processAttach(context, fdbPacketPayload, rule);
89              case CONT_AUTH:
90                  // TODO implement CONT_AUTH
91              default:
92                  throw new FirebirdProtocolException("Wrong operation %s during authentication phase", type.name());
93          }
94      }
95      
96      private AuthenticationResult processAttach(final ChannelHandlerContext context, final FirebirdPacketPayload payload, final AuthorityRule rule) {
97          FirebirdAttachPacket attachPacket = new FirebirdAttachPacket(payload);
98          context.channel().attr(CommonConstants.CHARSET_ATTRIBUTE_KEY).set(FirebirdCharacterSets.findCharacterSet(attachPacket.getEncoding()));
99          login(currentAuthResult.getDatabase(), currentAuthResult.getUsername(), attachPacket, rule);
100         context.writeAndFlush(new FirebirdGenericResponsePacket());
101         return AuthenticationResultBuilder.finished(currentAuthResult.getUsername(), "", currentAuthResult.getDatabase());
102     }
103     
104     private void login(final String databaseName, final String username, final FirebirdAttachPacket attachPacket, final AuthorityRule rule) {
105         ShardingSpherePreconditions.checkState(
106                 Strings.isNullOrEmpty(databaseName) || ProxyContext.getInstance().getContextManager().getMetaDataContexts().getMetaData().containsDatabase(databaseName),
107                 () -> new UnknownDatabaseException(databaseName));
108         Grantee grantee = new Grantee(username, "");
109         Optional<ShardingSphereUser> user = rule.findUser(grantee);
110         user.ifPresent(shardingSphereUser -> new AuthenticatorFactory<>(FirebirdAuthenticatorType.class, rule)
111                 .newInstance(shardingSphereUser).authenticate(shardingSphereUser, new Object[]{attachPacket.getEncPassword(), authData, attachPacket.getAuthData()}));
112     }
113     
114     private AuthenticationResult processConnect(final ChannelHandlerContext context, final FirebirdPacketPayload payload, final AuthorityRule rule) {
115         FirebirdConnectPacket connectPacket = new FirebirdConnectPacket(payload);
116         FirebirdAcceptPacket acceptPacket = new FirebirdAcceptPacket(connectPacket.getUserProtocols());
117         context.channel().attr(CommonConstants.CHARSET_ATTRIBUTE_KEY).set(FirebirdCharacterSets.findCharacterSet("NONE"));
118         FirebirdConnectionProtocolVersion.getInstance().setProtocolVersion(connectionId, acceptPacket.getProtocol().getVersion());
119         context.channel().attr(FirebirdConstant.CONNECTION_PROTOCOL_VERSION).set(acceptPacket.getProtocol().getVersion());
120         String username = connectPacket.getLogin();
121         Grantee grantee = new Grantee(username, "");
122         Optional<ShardingSphereUser> user = rule.findUser(grantee);
123         plugin = FirebirdAuthenticationMethod.valueOf(getPluginName(rule, user.get()));
124         FirebirdAuthenticationMethod userPlugin = connectPacket.getPlugin();
125         if (plugin != userPlugin) {
126             acceptPacket.setAcceptDataPacket(new byte[0], "", plugin, 0, "");
127         } else if (plugin != FirebirdAuthenticationMethod.LEGACY_AUTH) {
128             authData = new FirebirdSRPAuthenticationData(plugin.getHashAlgorithm(), username, user.get().getPassword(), connectPacket.getAuthData());
129             acceptPacket.setAcceptDataPacket(authData.getSalt(), authData.getPublicKeyHex(), plugin, 0, "");
130         }
131         context.writeAndFlush(acceptPacket);
132         currentAuthResult = AuthenticationResultBuilder.continued(username, connectPacket.getHost(), connectPacket.getDatabase());
133         return currentAuthResult;
134     }
135     
136     private String getPluginName(final AuthorityRule rule, final ShardingSphereUser user) {
137         String pluginName = rule.getAuthenticatorType(user);
138         return pluginName.isEmpty()
139                 // default plugin name
140                 ? Arrays.stream(FirebirdAuthenticatorType.class.getEnumConstants()).filter(AuthenticatorType::isDefault).findAny().orElseThrow(IllegalArgumentException::new).name()
141                 : pluginName;
142     }
