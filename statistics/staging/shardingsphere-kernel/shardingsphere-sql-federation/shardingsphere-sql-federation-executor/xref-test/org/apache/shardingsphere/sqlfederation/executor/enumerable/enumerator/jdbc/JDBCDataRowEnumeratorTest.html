54      @ParameterizedTest(name = "{0}")
55      @MethodSource("timeArguments")
56      void assertEnumerateTimeRow(final String name, final Object value, final Time expected) throws SQLException {
57          MergedResult queryResult = mock(MergedResult.class);
58          QueryResultMetaData metaData = mock(QueryResultMetaData.class);
59          when(metaData.getColumnCount()).thenReturn(2);
60          when(metaData.getColumnType(1)).thenReturn(Types.TIME);
61          when(metaData.getColumnType(2)).thenReturn(Types.INTEGER);
62          when(queryResult.next()).thenReturn(true);
63          when(queryResult.getValue(1, Time.class)).thenReturn(value);
64          when(queryResult.getValue(2, Object.class)).thenReturn(48);
65          JDBCDataRowEnumerator enumerator = new JDBCDataRowEnumerator(queryResult, metaData, Collections.emptyList());
66          assertTrue(enumerator.moveNext());
67          Object actual = enumerator.current();
68          assertThat(actual, isA(Object[].class));
69          Object[] actualValues = (Object[]) actual;
70          assertThat(actualValues[0], isA(Time.class));
71          assertThat(actualValues[0], is(expected));
72          assertThat(actualValues[1], is(48));
73          verify(queryResult).getValue(1, Time.class);
74          verify(queryResult).getValue(2, Object.class);
75          verify(metaData).getColumnType(1);
76          verify(metaData).getColumnType(2);
77      }

114     @Test
115     void assertEnumerateRowsBuildsRows() throws SQLException {
116         MergedResult firstQueryResult = mock(MergedResult.class);
117         LocalDateTime expectedDateTime = LocalDateTime.of(2024, 1, 1, 1, 0);
118         when(firstQueryResult.next()).thenReturn(true, false);
119         when(firstQueryResult.getValue(1, Object.class)).thenReturn(expectedDateTime);
120         when(firstQueryResult.getValue(2, Object.class)).thenReturn("bar_value");
121         QueryResultMetaData firstMetaData = mock(QueryResultMetaData.class);
122         when(firstMetaData.getColumnCount()).thenReturn(2);
123         Statement firstStatement = mock(Statement.class);
124         Statement secondStatement = mock(Statement.class);
125         JDBCDataRowEnumerator enumerator = new JDBCDataRowEnumerator(firstQueryResult, firstMetaData, Arrays.asList(firstStatement, secondStatement));
126         assertTrue(enumerator.moveNext());
127         Object actualRow = enumerator.current();
128         assertThat(actualRow, isA(Object[].class));
129         Object[] actualRowValues = (Object[]) actualRow;
130         assertThat(actualRowValues.length, is(2));
131         assertThat(actualRowValues[0], isA(Timestamp.class));
132         assertThat(actualRowValues[0], is(Timestamp.valueOf(expectedDateTime)));
133         assertThat(actualRowValues[1], is("bar_value"));
134         assertFalse(enumerator.moveNext());
135         enumerator.reset();
136         enumerator.close();
137         verify(firstStatement).close();
138         verify(secondStatement).close();
139         assertNull(enumerator.current());
140     }

158         Statement statement = mock(Statement.class);

165     @Test
166     void assertCloseClosesRemainingStatementsAfterFailure() throws SQLException {
167         Statement failedStatement = mock(Statement.class);
168         SQLException expectedException = new SQLException("failed to close");
169         doThrow(expectedException).when(failedStatement).close();
170         Statement secondStatement = mock(Statement.class);
171         Statement thirdStatement = mock(Statement.class);
172         JDBCDataRowEnumerator enumerator = new JDBCDataRowEnumerator(
173                 mock(MergedResult.class), mock(QueryResultMetaData.class), Arrays.asList(failedStatement, secondStatement, thirdStatement));
174         SQLWrapperException actualException = assertThrows(SQLWrapperException.class, enumerator::close);
175         assertThat(actualException.getCause(), is(expectedException));
176         verify(secondStatement).close();
177         verify(thirdStatement).close();
178         assertNull(enumerator.current());
179     }

181     @Test
182     void assertCloseContinuesWhenTwoStatementsThrowTheSameException() throws SQLException {
183         SQLException sharedException = new SQLException("shared failure");
184         Statement firstStatement = mock(Statement.class);
185         doThrow(sharedException).when(firstStatement).close();
186         Statement secondStatement = mock(Statement.class);
187         doThrow(sharedException).when(secondStatement).close();
188         Statement thirdStatement = mock(Statement.class);
189         JDBCDataRowEnumerator enumerator = new JDBCDataRowEnumerator(
190                 mock(MergedResult.class), mock(QueryResultMetaData.class), Arrays.asList(firstStatement, secondStatement, thirdStatement));
191         SQLWrapperException actualException = assertThrows(SQLWrapperException.class, enumerator::close);
192         assertThat(actualException.getCause(), is(sharedException));
193         verify(thirdStatement).close();
194         assertNull(enumerator.current());
195     }

197     @Test
198     void assertCloseSuppressesLaterFailures() throws SQLException {
199         Statement firstStatement = mock(Statement.class);
200         SQLException firstException = new SQLException("first");
201         doThrow(firstException).when(firstStatement).close();
202         Statement secondStatement = mock(Statement.class);
203         SQLException secondException = new SQLException("second");
204         doThrow(secondException).when(secondStatement).close();
205         JDBCDataRowEnumerator enumerator = new JDBCDataRowEnumerator(
206                 mock(MergedResult.class), mock(QueryResultMetaData.class), Arrays.asList(firstStatement, secondStatement));
207         SQLWrapperException actualException = assertThrows(SQLWrapperException.class, enumerator::close);
208         assertThat(actualException.getCause(), is(firstException));
209         assertThat(actualException.getCause().getSuppressed().length, is(1));
210         assertThat(actualException.getCause().getSuppressed()[0], is(secondException));
211     }
