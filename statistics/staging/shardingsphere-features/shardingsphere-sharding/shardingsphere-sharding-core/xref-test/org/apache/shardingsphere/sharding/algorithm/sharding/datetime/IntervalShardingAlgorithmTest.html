205     void assertFormat() {
206         String inputFormat = "yyyy-MM-dd HH:mm:ss.SSS";
207         String tableFormatByQuarter = "yyyyQQ";
208         String tableFormatByMonth = "yyyyMM";
209         String value = "2020-10-11 00:00:00.000000";
210         LocalDateTime localDateTime = LocalDateTime.parse(value.substring(0, inputFormat.length()), DateTimeFormatter.ofPattern(inputFormat));
211         String tableNameShardedByQuarter = localDateTime.format(DateTimeFormatter.ofPattern(tableFormatByQuarter));
212         String tableNameShardedByMonth = localDateTime.format(DateTimeFormatter.ofPattern(tableFormatByMonth));

214         assertThat(tableNameShardedByMonth, is("202010"));

217     @Test
218     void assertRangeDoShardingByDays() {
219         int expectSize = 24;
220         int stepAmount = 2;
221         IntervalShardingAlgorithm algorithm = createAlgorithm("yyyy-MM-dd HH:mm:ss.SSS", "2021-06-01 00:00:00.000",
222                 "2021-07-31 00:00:00.000", "yyyyMMdd", stepAmount, "DAYS");
223         Collection<String> availableTargetNames = new LinkedList<>();
224         for (int j = 6; j <= 7; j++) {
225             for (int i = 1; j == 6 ? i <= 30 : i <= 31; i = i + stepAmount) {
226                 availableTargetNames.add(String.format("t_order_%04d%02d%02d", 2021, j, i));
227             }
228         }
229         LocalDateTime lower = LocalDateTime.of(2021, 6, 15, 2, 25, 27, 0);
230         LocalDateTime upper = LocalDateTime.of(2021, 7, 31, 2, 25, 27, 0);
231         RangeShardingValue<Comparable<?>> shardingValueAsLocalDateTime = createShardingValue(lower, upper);
232         assertThat(algorithm.doSharding(availableTargetNames, shardingValueAsLocalDateTime).size(), is(expectSize));
233         RangeShardingValue<Comparable<?>> shardingValueAsInstant = createShardingValue(
234                 lower.atZone(ZoneId.systemDefault()).toInstant(),
235                 upper.atZone(ZoneId.systemDefault()).toInstant());
236         assertThat(algorithm.doSharding(availableTargetNames, shardingValueAsInstant).size(), is(expectSize));
237         RangeShardingValue<Comparable<?>> shardingValueAsTimestamp = createShardingValue(Timestamp.valueOf(lower), Timestamp.valueOf(upper));
238         assertThat(algorithm.doSharding(availableTargetNames, shardingValueAsTimestamp).size(), is(expectSize));
239         RangeShardingValue<Comparable<?>> shardingValueAsOffsetDateTime = createShardingValue(
240                 OffsetDateTime.of(lower, OffsetDateTime.now().getOffset()),
241                 OffsetDateTime.of(upper, OffsetDateTime.now().getOffset()));
242         assertThat(algorithm.doSharding(availableTargetNames, shardingValueAsOffsetDateTime).size(), is(expectSize));
243         RangeShardingValue<Comparable<?>> shardingValueAsZonedDateTime = createShardingValue(
244                 ZonedDateTime.of(lower, ZoneId.systemDefault()),
245                 ZonedDateTime.of(upper, ZoneId.systemDefault()));
246         assertThat(algorithm.doSharding(availableTargetNames, shardingValueAsZonedDateTime).size(), is(expectSize));
247         RangeShardingValue<Comparable<?>> shardingValueAsUtilDate = createShardingValue(
248                 Date.from(lower.atZone(ZoneId.systemDefault()).toInstant()),
249                 Date.from(upper.atZone(ZoneId.systemDefault()).toInstant()));
250         assertThat(algorithm.doSharding(availableTargetNames, shardingValueAsUtilDate).size(), is(expectSize));
251         RangeShardingValue<Comparable<?>> shardingValueAsSqlDate = createShardingValue(
252                 new java.sql.Date(lower.atZone(ZoneId.systemDefault()).toInstant().toEpochMilli()),
253                 new java.sql.Date(upper.atZone(ZoneId.systemDefault()).toInstant().toEpochMilli()));
254         assertThrows(UnsupportedTemporalTypeException.class, () -> algorithm.doSharding(availableTargetNames, shardingValueAsSqlDate),
255                 "SQL Date values do not have a time component.");
256         assertThat(createAlgorithm("yyyy-MM-dd", "2021-06-01",
257                 "2021-07-31", "yyyyMMdd", stepAmount, null)
258                 .doSharding(availableTargetNames, shardingValueAsSqlDate).size(),
259                 is(expectSize));
260         RangeShardingValue<Comparable<?>> shardingValueAsString = createShardingValue(
261                 DateTimeFormatterFactory.getDatetimeFormatter().format(lower),
262                 DateTimeFormatterFactory.getDatetimeFormatter().format(upper));
263         assertThat(algorithm.doSharding(availableTargetNames, shardingValueAsString).size(), is(expectSize));
264         assertThat(shardingAlgorithmByDay.doSharding(availableTablesForDayDataSources, shardingValueAsString).size(), is(expectSize));
265     }

345     private IntervalShardingAlgorithm createAlgorithm(final String datetimePattern, final String datetimeLower,
346                                                       final String datetimeUpper, final String shardingSuffixPattern,
347                                                       final Integer datetimeIntervalAmount, final String datetimeIntervalUnit) {
348         Properties props = PropertiesBuilder.build(
349                 new Property("datetime-pattern", datetimePattern),
350                 new Property("datetime-lower", datetimeLower),
351                 new Property("datetime-upper", datetimeUpper),
352                 new Property("sharding-suffix-pattern", shardingSuffixPattern),
353                 new Property("datetime-interval-amount", Integer.toString(datetimeIntervalAmount)));
354         if (null != datetimeIntervalUnit) {
355             props.setProperty("datetime-interval-unit", datetimeIntervalUnit);
356         }
357         return (IntervalShardingAlgorithm) TypedSPILoader.getService(ShardingAlgorithm.class, "INTERVAL", props);
358     }
359     
360     private RangeShardingValue<Comparable<?>> createShardingValue(final Comparable<?> lower, final Comparable<?> upper) {
361         return new RangeShardingValue<>("t_order", "create_time", DATA_NODE_INFO, Range.closed(lower, upper));
362     }
