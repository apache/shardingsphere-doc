43  /**
44   * MCP SQL execution facade.
45   */
46  @AllArgsConstructor(access = AccessLevel.PACKAGE)
47  public final class MCPSQLExecutionFacade implements MCPFeatureExecutionFacade {
48      
49      private final MCPDatabaseCapabilityProvider databaseCapabilityProvider;
50      
51      private final MCPSessionExecutionCoordinator sessionExecutionCoordinator;
52      
53      private final MCPJdbcTransactionStatementExecutor transactionStatementExecutor;
54      
55      private final MCPJdbcStatementExecutor statementExecutor;
56      
57      private final MCPStatementAnalyzer statementAnalyzer;
58      
59      public MCPSQLExecutionFacade(final MCPDatabaseCapabilityProvider databaseCapabilityProvider, final MCPSessionManager sessionManager) {
60          this(databaseCapabilityProvider, new MCPSessionExecutionCoordinator(sessionManager),
61                  new MCPJdbcTransactionStatementExecutor(sessionManager),
62                  new MCPJdbcStatementExecutor(sessionManager.getTransactionResourceManager().getRuntimeDatabases(), sessionManager.getTransactionResourceManager()),
63                  new MCPStatementAnalyzer());
64      }
65      
66      @Override
67      public SQLExecutionResult execute(final SQLExecutionRequest executionRequest) {
68          return sessionExecutionCoordinator.executeWithSessionLock(executionRequest.getSessionId(), () -> {
69              MCPDatabaseCapability databaseCapability = getDatabaseCapability(executionRequest);
70              return executeInternal(executionRequest, statementAnalyzer.analyze(executionRequest.getSql(), databaseCapability), databaseCapability);
71          });
72      }
73      
74      private SQLExecutionResult execute(final SQLExecutionRequest executionRequest, final ClassificationResult classificationResult, final MCPDatabaseCapability databaseCapability) {
75          return sessionExecutionCoordinator.executeWithSessionLock(executionRequest.getSessionId(), () -> executeInternal(executionRequest, classificationResult, databaseCapability));
76      }
77      
78      @Override
79      public SQLExecutionResult executeExplain(final SQLExecutionRequest executionRequest, final String sql) {
80          MCPDatabaseCapability databaseCapability = getDatabaseCapability(executionRequest);
81          ClassificationResult classificationResult = new ExplainSQLCandidateValidator(statementAnalyzer).validate(sql, executionRequest.getSql(), databaseCapability);
82          try {
83              return execute(executionRequest, classificationResult, databaseCapability);
84          } catch (final MCPInvalidRequestException | MCPQueryFailedException ex) {
85              if (MCPJDBCErrorCategory.SYNTAX == MCPJDBCExceptionClassifier.classify(databaseCapability.getDatabaseType(), ex)) {
86                  throw new ExplainSQLSyntaxException(executionRequest.getDatabase(), executionRequest.getSchema(), sql, executionRequest.getSql(), ex);
87              }
88              throw ex;
89          }
90      }
91      
92      private MCPDatabaseCapability getDatabaseCapability(final SQLExecutionRequest executionRequest) {
93          Optional<MCPDatabaseCapability> databaseCapability = databaseCapabilityProvider.provide(executionRequest.getDatabase());
94          ShardingSpherePreconditions.checkState(databaseCapability.isPresent(), DatabaseCapabilityNotFoundException::new);
95          return databaseCapability.orElseThrow();
96      }
97      
98      private SQLExecutionResult executeInternal(final SQLExecutionRequest executionRequest, final ClassificationResult classificationResult,
99                                                 final MCPDatabaseCapability databaseCapability) {
100         ShardingSpherePreconditions.checkContains(databaseCapability.getSupportedStatementClasses(), classificationResult.getStatementClass(),
101                 StatementClassNotSupportedException::new);
102         checkCrossSchemaSql(executionRequest, databaseCapability, classificationResult);
103         return switch (classificationResult.getStatementClass()) {
104             case TRANSACTION_CONTROL, SAVEPOINT -> transactionStatementExecutor.execute(
105                     executionRequest.getSessionId(), executionRequest.getDatabase(), databaseCapability, classificationResult);
106             case QUERY, EXPLAIN, DML, DDL, DCL -> statementExecutor.execute(executionRequest, classificationResult, databaseCapability);
107         };
108     }
109     
110     private void checkCrossSchemaSql(final SQLExecutionRequest executionRequest, final MCPDatabaseCapability databaseCapability, final ClassificationResult classificationResult) {
111         if (SchemaExecutionSemantics.BEST_EFFORT == databaseCapability.getSchemaExecutionSemantics()) {
112             return;
113         }
114         for (SQLStatementObjectName each : classificationResult.getReferencedObjects()) {
115             if (isCrossSchemaReference(each, executionRequest.getDatabase(), databaseCapability.getIdentifierContext())) {
116                 throw new MCPInvalidRequestException(String.format("Cross-schema SQL is not supported for database `%s`: `%s`.", executionRequest.getDatabase(), each.getObjectName()));
117             }
118         }
119     }
120     
121     private boolean isCrossSchemaReference(final SQLStatementObjectName objectName, final String databaseName, final DatabaseIdentifierContext identifierContext) {
122         return (objectName.isQualified() || objectName.isNamespaceTarget())
123                 && !identifierContext.matchesMetaData(IdentifierScope.SCHEMA, databaseName,
124                         new IdentifierValue(objectName.getFirstIdentifier(), objectName.getFirstIdentifierQuoteCharacter()));
125     }
126 }
