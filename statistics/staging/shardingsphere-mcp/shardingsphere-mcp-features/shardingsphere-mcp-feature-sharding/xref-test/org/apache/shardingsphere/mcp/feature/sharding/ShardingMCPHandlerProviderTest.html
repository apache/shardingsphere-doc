37  import static org.hamcrest.MatcherAssert.assertThat;
38  import static org.hamcrest.Matchers.containsInAnyOrder;
39  import static org.hamcrest.Matchers.is;
40  import static org.hamcrest.Matchers.isA;
41  import static org.junit.jupiter.api.Assertions.assertTrue;
42  
43  class ShardingMCPHandlerProviderTest {
44      
45      @Test
46      void assertGetResourceHandlers() {
47          Collection<MCPResourceHandler<?>> actual = new ShardingMCPHandlerProvider().getResourceHandlers();
48          assertThat(actual.size(), is(22));
49          assertTrue(actual.stream().map(MCPResourceHandler::getResourceUriTemplate).toList().contains(ShardingFeatureDefinition.TABLE_RULES_RESOURCE_URI));
50          assertTrue(actual.stream().map(MCPResourceHandler::getResourceUriTemplate).toList().contains(ShardingFeatureDefinition.UNUSED_AUDITORS_RESOURCE_URI));
51          assertTrue(actual.stream().anyMatch(ShardingAlgorithmResourceHandler.class::isInstance));
52          assertTrue(actual.stream().anyMatch(ShardingTableResourceHandler.class::isInstance));
53          assertTrue(actual.stream().anyMatch(ShardingStrategyResourceHandler.class::isInstance));
54          assertTrue(actual.stream().anyMatch(ShardingGovernanceResourceHandler.class::isInstance));
55      }
56      
57      @Test
58      void assertGetToolHandlers() {
59          assertThat(new ShardingMCPHandlerProvider().getToolHandlers().stream().map(MCPToolHandler::getToolName).toList(),
60                  containsInAnyOrder(ShardingFeatureDefinition.PLAN_TABLE_RULE_TOOL_NAME,
61                          ShardingFeatureDefinition.PLAN_TABLE_REFERENCE_TOOL_NAME, ShardingFeatureDefinition.PLAN_DEFAULT_STRATEGY_TOOL_NAME,
62                          ShardingFeatureDefinition.PLAN_KEY_GENERATOR_TOOL_NAME, ShardingFeatureDefinition.PLAN_KEY_GENERATE_STRATEGY_TOOL_NAME,
63                          ShardingFeatureDefinition.PLAN_COMPONENT_CLEANUP_TOOL_NAME));
64      }
65      
66      @Test
67      void assertGetCompletionHandlers() {
68          MCPCompletionHandler<?> actual = new ShardingMCPHandlerProvider().getCompletionHandlers().iterator().next();
69          assertThat(actual, isA(ShardingAlgorithmCompletionHandler.class));
70      }
71      
72      @Test
73      void assertGetWorkflowDefinitions() {
74          Collection<WorkflowRuntimeDefinition> actual = new ShardingMCPHandlerProvider().getWorkflowDefinitions();
75          assertThat(actual.stream().map(each -> each.getWorkflowKind().getValue()).toList(),
76                  containsInAnyOrder("sharding.table.rule", "sharding.table.reference", "sharding.default.strategy",
77                          "sharding.key.generator", "sharding.key.generate.strategy", "sharding.component.cleanup"));
78          assertTrue(actual.stream().allMatch(each -> each.getRuntimeHandler() instanceof ShardingWorkflowValidationService));
79          assertTrue(actual.stream().allMatch(each -> each.getApplyArtifactValidator() instanceof ShardingWorkflowApplyArtifactValidator));
80      }
81      
82      @Test
83      void assertSPIRegistration() {
84          assertTrue(ServiceLoader.load(MCPHandlerProvider.class).stream().anyMatch(each -> ShardingMCPHandlerProvider.class.equals(each.type())));
85      }
86  }
