18  package org.apache.shardingsphere.mcp.support.descriptor;

20  import org.apache.shardingsphere.mcp.api.capability.resource.MCPResourceAnnotations;
21  import org.apache.shardingsphere.mcp.api.capability.resource.MCPResourceDescriptor;
22  import org.junit.jupiter.params.ParameterizedTest;
23  import org.junit.jupiter.params.provider.Arguments;
24  import org.junit.jupiter.params.provider.MethodSource;
25  
26  import java.util.Collection;
27  import java.util.List;
28  import java.util.Map;
29  
30  import static org.hamcrest.MatcherAssert.assertThat;
31  import static org.hamcrest.Matchers.is;
32  import static org.junit.jupiter.api.Assertions.assertThrows;
33  
34  class MCPResourceDescriptorValidatorTest {
35      
36      @ParameterizedTest(name = "{0}")
37      @MethodSource("invalidResources")
38      void assertValidateWithInvalidResource(final String name, final MCPDescriptorCatalog catalog, final String expectedMessage) {
39          IllegalStateException actual = assertThrows(IllegalStateException.class, () -> MCPResourceDescriptorValidator.validate(catalog));
40          assertThat(actual.getMessage(), is(expectedMessage));
41      }
42      
43      private static Collection<Arguments> invalidResources() {
44          MCPResourceDescriptor fixedResource = createResource("shardingsphere://capabilities");
45          MCPResourceDescriptor template = createResource("shardingsphere://databases/{database}");
46          MCPUriVariableDescriptor variable = new MCPUriVariableDescriptor("database", "Database", "Logical database.", true, "database");
47          return List.of(
48                  Arguments.of("template without variables", createCatalog(List.of(fixedResource), List.of(fixedResource), List.of()),
49                          "Resource template `shardingsphere://capabilities` must contain template variables."),
50                  Arguments.of("fixed resource with variables", createCatalog(List.of(template), List.of(), List.of()),
51                          "Fixed resource `shardingsphere://databases/{database}` must not contain template variables."),
52                  Arguments.of("blank URI", createCatalog(List.of(createResource(" ")), List.of(), List.of()), "Resource URI is required."),
53                  Arguments.of("duplicate URI", createCatalog(List.of(fixedResource, fixedResource), List.of(), List.of()), "Duplicate MCP resource descriptor `shardingsphere://capabilities`."),
54                  Arguments.of("missing variable metadata", createCatalog(List.of(), List.of(template), List.of()),
55                          "Resource descriptor `shardingsphere://databases/{database}` must describe URI template variable `database`."),
56                  Arguments.of("optional variable", createCatalog(List.of(), List.of(template), List.of(new MCPUriVariableDescriptor("database", "Database", "Logical database.", false, "database"))),
57                          "Resource parameter `shardingsphere://databases/{database}.database` must be required because URI template variables are required."),
58                  Arguments.of("non-template variable", createCatalog(List.of(), List.of(template), List.of(new MCPUriVariableDescriptor("schema", "Schema", "Logical schema.", true, "schema"))),
59                          "Resource descriptor `shardingsphere://databases/{database}` declares non-template parameter `schema`."),
60                  Arguments.of("duplicate variable metadata", createCatalog(List.of(), List.of(template), List.of(variable, variable)),
61                          "Duplicate MCP resource parameter `shardingsphere://databases/{database}.database`."));
62      }
63      
64      private static MCPResourceDescriptor createResource(final String uri) {
65          return new MCPResourceDescriptor(uri, "foo_resource", "Resource", "Read resource.", "application/json", MCPResourceAnnotations.EMPTY, Map.of());
66      }
67      
68      private static MCPDescriptorCatalog createCatalog(final Collection<MCPResourceDescriptor> resources, final Collection<MCPResourceDescriptor> templates,
69                                                        final Collection<MCPUriVariableDescriptor> variables) {
70          Collection<ShardingSphereMCPResourceMetadata> metadata = templates.stream().map(each -> new ShardingSphereMCPResourceMetadata(each.getUriTemplate(), variables,
71                  "detail", "database", "", List.of(), List.of(), List.of())).toList();
72          return new MCPDescriptorCatalog(new MCPProtocolDescriptorCatalog(resources, templates, List.of(), List.of()),
73                  new MCPShardingSphereDescriptorCatalog(metadata, List.of(), List.of(), List.of(), List.of()));
74      }
75  }
