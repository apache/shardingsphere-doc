152             hintManager.addDatabaseShardingValue("t_hint_test", 1);
153             hintManager.addTableShardingValue("t_hint_test", 1);

177     private static RouteContext routeDatabaseOnlyJoin(final String sql, final List<Object> params) {
178         DatabaseType databaseType = TypedSPILoader.getService(DatabaseType.class, "MySQL");
179         ShardingRuleConfiguration config = new ShardingRuleConfiguration();
180         Collection<ShardingSphereTable> tables = new ArrayList<>(2);
181         for (String each : Arrays.asList("t_order", "t_order_item")) {
182             config.getTables().add(new ShardingTableRuleConfiguration(each, String.format("<LITERAL>ds_0.%s, ds_1.%s, ds_2.%s", each, each, each)));
183             tables.add(new ShardingSphereTable(each, Arrays.asList(new ShardingSphereColumn("order_id", Types.BIGINT, false, false, false, true, false, false),
184                     new ShardingSphereColumn("user_id", Types.INTEGER, false, false, false, true, false, false)), Collections.emptyList(), Collections.emptyList()));
185         }
186         config.setDefaultDatabaseShardingStrategy(new StandardShardingStrategyConfiguration("user_id", "database_mod"));
187         config.getShardingAlgorithms().put("database_mod", new AlgorithmConfiguration("CLASS_BASED", PropertiesBuilder.build(new Property("strategy", "STANDARD"),
188                 new Property("algorithmClassName", ModShardingAlgorithm.class.getName()), new Property("sharding-count", "2"))));
189         Map<String, DataSource> dataSources = new LinkedHashMap<>(3, 1F);
190         for (String each : Arrays.asList("ds_0", "ds_1", "ds_2")) {
191             dataSources.put(each, new MockedDataSource());
192         }
193         ShardingRule shardingRule = new ShardingRule(config, dataSources, mock(ComputeNodeInstanceContext.class), Collections.emptyList());
194         ConfigurationProperties props = new ConfigurationProperties(new Properties());
195         ShardingSphereSchema schema = new ShardingSphereSchema(DATABASE_NAME, databaseType, tables, Collections.emptyList());
196         ShardingSphereDatabase database = new ShardingSphereDatabase(DATABASE_NAME, databaseType, mock(ResourceMetaData.class, RETURNS_DEEP_STUBS),
197                 new RuleMetaData(Collections.singleton(shardingRule)), Collections.singleton(schema), props);
198         return route(sql, params, database, props);
199     }

201     private static RouteContext route(final String sql, final List<Object> params) {
202         DatabaseType databaseType = TypedSPILoader.getService(DatabaseType.class, "MySQL");
203         ShardingRule shardingRule = createShardingRule();
204         ConfigurationProperties props = new ConfigurationProperties(new Properties());
205         ShardingSphereDatabase database = createDatabase(databaseType, shardingRule, props);
206         return route(sql, params, database, props);
207     }
208     
209     private static RouteContext route(final String sql, final List<Object> params, final ShardingSphereDatabase database, final ConfigurationProperties props) {
210         DatabaseType databaseType = database.getProtocolType();
211         RuleMetaData globalRuleMetaData = mock(RuleMetaData.class);
212         ShardingSphereMetaData metaData = new ShardingSphereMetaData(Collections.singleton(database), mock(), globalRuleMetaData, props);
213         SQLParserEngine parserEngine = new SQLParserRule(new DefaultSQLParserRuleConfigurationBuilder().build()).getSQLParserEngine(databaseType);
214         SQLStatementContext sqlStatementContext = new SQLBindEngine(metaData, DATABASE_NAME, new HintValueContext()).bind(parserEngine.parse(sql, false));
215         ConnectionContext connectionContext = new ConnectionContext(Collections::emptySet);
216         connectionContext.setCurrentDatabaseName(DATABASE_NAME);
217         QueryContext queryContext = new QueryContext(sqlStatementContext, sql, params, new HintValueContext(), connectionContext, metaData);
218         return new SQLRouteEngine(database.getRuleMetaData().getRules(), props).route(queryContext, globalRuleMetaData, database);
219     }

221     private static ShardingSphereDatabase createDatabase(final DatabaseType databaseType, final ShardingRule shardingRule, final ConfigurationProperties props) {
222         return new ShardingSphereDatabase(DATABASE_NAME, databaseType, mock(ResourceMetaData.class, RETURNS_DEEP_STUBS),
223                 new RuleMetaData(Collections.singleton(shardingRule)), buildSchemas(databaseType), props);
224     }

226     private static Collection<ShardingSphereSchema> buildSchemas(final DatabaseType databaseType) {
227         Collection<ShardingSphereTable> tables = new LinkedList<>();
228         tables.add(new ShardingSphereTable("t_order", Arrays.asList(
229                 new ShardingSphereColumn("id", Types.INTEGER, true, false, false, true, false, false),
230                 new ShardingSphereColumn("user_id", Types.INTEGER, false, false, false, true, false, false),
231                 new ShardingSphereColumn("status", Types.VARCHAR, false, false, false, true, false, false)), Collections.emptyList(), Collections.emptyList()));
232         tables.add(new ShardingSphereTable("t_order_item", Arrays.asList(
233                 new ShardingSphereColumn("id", Types.INTEGER, true, false, false, true, false, false),
234                 new ShardingSphereColumn("user_id", Types.INTEGER, false, false, false, true, false, false)), Collections.emptyList(), Collections.emptyList()));
235         tables.add(new ShardingSphereTable("t_hint_test", Collections.singleton(
236                 new ShardingSphereColumn("user_id", Types.INTEGER, true, false, false, true, false, false)), Collections.emptyList(), Collections.emptyList()));
237         return Collections.singleton(new ShardingSphereSchema(DATABASE_NAME, databaseType, tables, Collections.emptyList()));
238     }

240     private static ShardingRule createShardingRule() {
241         ShardingRuleConfiguration config = new ShardingRuleConfiguration();
242         config.getTables().add(createInlineTableRule("t_order"));
243         config.getTables().add(createInlineTableRule("t_order_item"));
244         config.getTables().add(createHintTableRule());
245         config.getBindingTableGroups().add(new ShardingTableReferenceRuleConfiguration("foo", "t_order,t_order_item"));
246         config.getShardingAlgorithms().put("ds_inline", createAlgorithm("INLINE", "ds_${user_id % 2}"));
247         config.getShardingAlgorithms().put("t_order_inline", createAlgorithm("INLINE", "t_order_${user_id % 2}"));
248         config.getShardingAlgorithms().put("t_order_item_inline", createAlgorithm("INLINE", "t_order_item_${user_id % 2}"));
249         config.getShardingAlgorithms().put("ds_hint_inline", createAlgorithm("HINT_INLINE", "ds_${value % 2}"));
250         config.getShardingAlgorithms().put("t_hint_test_hint_inline", createAlgorithm("HINT_INLINE", "t_hint_test_${value % 2}"));
251         Map<String, DataSource> dataSources = new LinkedHashMap<>();
252         dataSources.put("ds_0", new MockedDataSource());
253         dataSources.put("ds_1", new MockedDataSource());
254         return new ShardingRule(config, dataSources, mock(ComputeNodeInstanceContext.class), Collections.emptyList());
255     }

257     private static ShardingTableRuleConfiguration createInlineTableRule(final String tableName) {
258         ShardingTableRuleConfiguration result = new ShardingTableRuleConfiguration(tableName, String.format("ds_${0..1}.%s_${0..1}", tableName));
259         result.setDatabaseShardingStrategy(new StandardShardingStrategyConfiguration("user_id", "ds_inline"));
260         result.setTableShardingStrategy(new StandardShardingStrategyConfiguration("user_id", tableName + "_inline"));
261         return result;
262     }

266         result.setDatabaseShardingStrategy(new HintShardingStrategyConfiguration("ds_hint_inline"));
267         result.setTableShardingStrategy(new HintShardingStrategyConfiguration("t_hint_test_hint_inline"));

275     private static Set<String> getRouteUnitSignatures(final RouteContext routeContext) {
276         Set<String> result = new TreeSet<>();
277         for (RouteUnit each : routeContext.getRouteUnits()) {
278             Collection<String> tableMappers = new TreeSet<>();
279             for (RouteMapper tableMapper : each.getTableMappers()) {
280                 tableMappers.add(tableMapper.getLogicName() + "=" + tableMapper.getActualName());
281             }
282             result.add(each.getDataSourceMapper().getActualName() + ":" + String.join(",", tableMappers));
283         }
284         return result;
285     }
286     
287     private static List<Set<String>> getOriginalDataNodeSignatures(final RouteContext routeContext) {
288         List<Set<String>> result = new LinkedList<>();
289         for (Collection<DataNode> each : routeContext.getOriginalDataNodes()) {
290             result.add(each.stream().map(dataNode -> dataNode.getDataSourceName() + "." + dataNode.getTableName()).collect(Collectors.toCollection(TreeSet::new)));
291         }
292         return result;
293     }
294     
295     private static List<Set<String>> createOriginalDataNodes(final String... dataNodes) {
296         return Arrays.stream(dataNodes).map(Collections::singleton).collect(Collectors.toList());
297     }

299     private static Set<String> createRouteUnits(final String... routeUnits) {
300         return new TreeSet<>(Arrays.asList(routeUnits));
