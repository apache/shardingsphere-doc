44  @RequiredArgsConstructor(access = AccessLevel.PACKAGE)
45  final class MetadataSearchCollector {
46      
47      private final MCPMetadataQueryFacade metadataQueryFacade;
48      
49      private final MCPFeatureQueryFacade queryFacade;
50      
51      private final GovernanceMetadataQueryService governanceMetadataQueryService;
52      
53      private final MetadataSearchResourceUriFactory resourceUriFactory;
54      
55      List<MetadataSearchHit> collect(final MetadataSearchRequest request, final Set<SupportedMCPMetadataObjectType> searchObjectTypes) {
56          return request.getDatabase().isEmpty()
57                  ? metadataQueryFacade.queryDatabases().stream().flatMap(each -> collect(each.getDatabase(), request.getSchema(), searchObjectTypes).stream()).toList()
58                  : collect(request.getDatabase(), request.getSchema(), searchObjectTypes);
59      }
60      
61      private List<MetadataSearchHit> collect(final String databaseName, final String schemaName, final Set<SupportedMCPMetadataObjectType> searchObjectTypes) {
62          List<MetadataSearchHit> result = new LinkedList<>();
63          for (SupportedMCPMetadataObjectType each : searchObjectTypes) {
64              result.addAll(collect(databaseName, schemaName, each));
65          }
66          return result;
67      }
68      
69      private List<MetadataSearchHit> collect(final String databaseName, final String schemaName, final SupportedMCPMetadataObjectType objectType) {
70          return switch (objectType) {
71              case DATABASE -> metadataQueryFacade.queryDatabase(databaseName).map(each -> List.of(createSearchHit(each))).orElseGet(List::of);
72              case STORAGE_UNIT -> queryStorageUnitSearchHits(databaseName);
73              case SCHEMA, TABLE, VIEW, COLUMN, INDEX, SEQUENCE -> metadataQueryFacade.isSupportedMetadataObjectType(databaseName, objectType)
74                      ? querySearchHits(databaseName, objectType, schemaName)
75                      : List.of();
76              default -> List.of();
77          };
78      }
79      
80      List<MetadataSearchHit> collectDatabases() {
81          return metadataQueryFacade.queryDatabases().stream().map(this::createSearchHit).toList();
82      }
83      
84      private List<MetadataSearchHit> querySearchHits(final String databaseName, final SupportedMCPMetadataObjectType objectType, final String schemaName) {
85          return switch (objectType) {
86              case SCHEMA -> querySchemaSearchHits(databaseName, schemaName);
87              case TABLE -> queryTables(databaseName, schemaName).stream().map(each -> createTableSearchHit(databaseName, each)).toList();
88              case VIEW -> queryViews(databaseName, schemaName).stream().map(each -> createViewSearchHit(databaseName, each)).toList();
89              case COLUMN -> queryColumnSearchHits(databaseName, schemaName);
90              case INDEX -> queryIndexSearchHits(databaseName, schemaName);
91              case SEQUENCE -> querySequenceSearchHits(databaseName, schemaName);
92              case DATABASE, STORAGE_UNIT -> List.of();
93              default -> List.of();
94          };
95      }
96      
97      private List<MetadataSearchHit> querySchemaSearchHits(final String databaseName, final String schemaName) {
98          List<MetadataSearchHit> result = new LinkedList<>();
99          addSchemaSearchHits(result, databaseName, schemaName);
100         return result;
101     }
102     
103     private void addSchemaSearchHits(final List<MetadataSearchHit> searchHits, final String databaseName, final String schemaName) {
104         if (!schemaName.isEmpty()) {
105             metadataQueryFacade.querySchema(databaseName, schemaName).ifPresent(optional -> searchHits.add(createSearchHit(databaseName, optional)));
106             return;
107         }
108         for (ShardingSphereSchema each : metadataQueryFacade.querySchemas(databaseName)) {
109             searchHits.add(createSearchHit(databaseName, each));
110         }
111     }
112     
113     private List<TableSearchScope> queryTables(final String databaseName, final String schemaName) {
114         if (!schemaName.isEmpty()) {
115             return metadataQueryFacade.queryTables(databaseName, schemaName).stream().map(each -> new TableSearchScope(schemaName, each)).toList();
116         }
117         return metadataQueryFacade.querySchemas(databaseName).stream()
118                 .flatMap(each -> metadataQueryFacade.queryTables(databaseName, each.getName()).stream().map(table -> new TableSearchScope(each.getName(), table))).toList();
119     }
120     
121     private List<TableSearchScope> queryViews(final String databaseName, final String schemaName) {
122         List<TableSearchScope> result = new LinkedList<>();
123         if (!schemaName.isEmpty()) {
124             metadataQueryFacade.queryViews(databaseName, schemaName).forEach(each -> result.add(new TableSearchScope(schemaName, each)));
125             return result;
126         }
127         for (ShardingSphereSchema each : metadataQueryFacade.querySchemas(databaseName)) {
128             metadataQueryFacade.queryViews(databaseName, each.getName()).forEach(view -> result.add(new TableSearchScope(each.getName(), view)));
129         }
130         return result;
131     }
132     
133     private List<MetadataSearchHit> queryColumnSearchHits(final String databaseName, final String schemaName) {
134         List<MetadataSearchHit> result = new LinkedList<>();
135         List<String> schemaNames = schemaName.isEmpty()
136                 ? metadataQueryFacade.querySchemas(databaseName).stream().map(ShardingSphereSchema::getName).toList()
137                 : List.of(schemaName);
138         for (String each : schemaNames) {
139             Map<String, TableType> relationTypes = createRelationTypes(databaseName, each);
140             for (MCPColumnMetadata column : metadataQueryFacade.querySchemaColumns(databaseName, each)) {
141                 TableType relationType = relationTypes.get(column.getRelationName());
142                 if (TableType.TABLE == relationType) {
143                     result.add(createColumnSearchHit(databaseName, each, column.getRelationName(), "", column));
144                 } else if (TableType.VIEW == relationType) {
145                     result.add(createColumnSearchHit(databaseName, each, "", column.getRelationName(), column));
146                 }
147             }
148         }
149         return result;
150     }
151     
152     private Map<String, TableType> createRelationTypes(final String databaseName, final String schemaName) {
153         Map<String, TableType> result = new LinkedHashMap<>();
154         metadataQueryFacade.queryTables(databaseName, schemaName).forEach(each -> result.put(each.getName(), TableType.TABLE));
155         metadataQueryFacade.queryViews(databaseName, schemaName).forEach(each -> result.put(each.getName(), TableType.VIEW));
156         return result;
157     }
158     
159     private List<MetadataSearchHit> queryIndexSearchHits(final String databaseName, final String schemaName) {
160         return queryTables(databaseName, schemaName).stream()
161                 .flatMap(each -> metadataQueryFacade.queryIndexes(databaseName, each.schema, each.table.getName()).stream()
162                         .map(index -> createIndexSearchHit(databaseName, each.schema, each.table.getName(), index)))
163                 .toList();
164     }
165     
166     private List<MetadataSearchHit> querySequenceSearchHits(final String databaseName, final String schemaName) {
167         if (!schemaName.isEmpty()) {
168             return metadataQueryFacade.querySequences(databaseName, schemaName).stream().map(each -> createSearchHit(databaseName, schemaName, each)).toList();
169         }
170         return metadataQueryFacade.querySchemas(databaseName).stream()
171                 .flatMap(each -> metadataQueryFacade.querySequences(databaseName, each.getName()).stream().map(sequence -> createSearchHit(databaseName, each.getName(), sequence))).toList();
172     }
173     
174     private List<MetadataSearchHit> queryStorageUnitSearchHits(final String databaseName) {
175         List<MetadataSearchHit> result = new LinkedList<>();
176         for (Map<String, Object> each : governanceMetadataQueryService.queryStorageUnits(queryFacade, databaseName)) {
177             String storageUnitName = Objects.toString(each.get("name"), "");
178             if (!storageUnitName.isEmpty()) {
179                 result.add(createSearchHit(databaseName, "", SupportedMCPMetadataObjectType.STORAGE_UNIT, "", "", storageUnitName));
180             }
181         }
182         return result;
183     }
184     
185     private MetadataSearchHit createSearchHit(final RuntimeDatabaseProfile databaseProfile) {
186         return createSearchHit(databaseProfile.getDatabase(), "", SupportedMCPMetadataObjectType.DATABASE, "", "", databaseProfile.getDatabase());
187     }
188     
189     private MetadataSearchHit createSearchHit(final String database, final ShardingSphereSchema schema) {
190         return createSearchHit(database, schema.getName(), SupportedMCPMetadataObjectType.SCHEMA, "", "", schema.getName());
191     }
192     
193     private MetadataSearchHit createSearchHit(final String database, final String schema, final MCPSequenceMetadata sequence) {
194         return createSearchHit(database, schema, SupportedMCPMetadataObjectType.SEQUENCE, "", "", sequence.getSequence());
195     }
196     
197     private MetadataSearchHit createSearchHit(final String database, final String schema, final SupportedMCPMetadataObjectType objectType,
198                                               final String table, final String view, final String name) {
199         MetadataSearchResourceUriFactory.MetadataResourceUris resourceUris = resourceUriFactory.create(database, schema, objectType, table, view, name);
200         return MetadataSearchHit.builder()
201                 .database(database)
202                 .schema(schema)
203                 .objectType(objectType.name().toLowerCase(Locale.ENGLISH))
204                 .table(table)
205                 .view(view)
206                 .name(name)
207                 .resource(resourceUris.resource())
208                 .parentResource(resourceUris.parentResource())
209                 .nextResources(resourceUris.nextResources())
210                 .derivationStatus(resourceUris.derivationStatus())
211                 .derivationReason(resourceUris.derivationReason())
212                 .matchKind("")
213                 .matchedFields(List.of())
214                 .matchedValue("")
215                 .build();
216     }
217     
218     private MetadataSearchHit createTableSearchHit(final String database, final TableSearchScope table) {
219         return createSearchHit(database, table.schema, SupportedMCPMetadataObjectType.TABLE, table.table.getName(), "", table.table.getName());
220     }
221     
222     private MetadataSearchHit createViewSearchHit(final String database, final TableSearchScope view) {
223         return createSearchHit(database, view.schema, SupportedMCPMetadataObjectType.VIEW, "", view.table.getName(), view.table.getName());
224     }
225     
226     private MetadataSearchHit createColumnSearchHit(final String database, final String schema, final String table, final String view, final MCPColumnMetadata column) {
227         return createSearchHit(database, schema, SupportedMCPMetadataObjectType.COLUMN, table, view, column.getName());
228     }
229     
230     private MetadataSearchHit createIndexSearchHit(final String database, final String schema, final String table, final ShardingSphereIndex index) {
231         return createSearchHit(database, schema, SupportedMCPMetadataObjectType.INDEX, table, "", index.getName());
232     }
233     
234     @RequiredArgsConstructor(access = AccessLevel.PRIVATE)
235     private static final class TableSearchScope {
236         
237         private final String schema;
238         
239         private final ShardingSphereTable table;
240     }
