18  package org.apache.shardingsphere.mcp.support.protocol.payload;
19  
20  import org.junit.jupiter.params.ParameterizedTest;
21  import org.junit.jupiter.params.provider.Arguments;
22  import org.junit.jupiter.params.provider.MethodSource;
23  
24  import java.util.List;
25  import java.util.Map;
26  import java.util.stream.Stream;
27  
28  import static org.hamcrest.MatcherAssert.assertThat;
29  import static org.hamcrest.Matchers.is;
30  
31  class MCPItemsPayloadTest {
32      
33      @ParameterizedTest(name = "{0}")
34      @MethodSource("assertToPayloadCases")
35      void assertToPayload(final String name, final MCPItemsPayload response, final Map<String, Object> expectedPayload) {
36          Map<String, Object> actual = response.toPayload();
37          assertThat(actual, is(expectedPayload));
38      }
39      
40      private static Stream<Arguments> assertToPayloadCases() {
41          return Stream.of(
42                  Arguments.of("without page token", new MCPItemsPayload(List.of("foo_item")),
43                          Map.of("response_mode", "list", "items", List.of("foo_item"), "count", 1, "continuation_mode", "none")),
44                  Arguments.of("with null page token", new MCPItemsPayload(List.of("foo_item"), (String) null),
45                          Map.of("response_mode", "list", "items", List.of("foo_item"), "count", 1, "continuation_mode", "none")),
46                  Arguments.of("with page token", new MCPItemsPayload(List.of("foo_item"), "foo_token"),
47                          Map.of("response_mode", "list", "items", List.of("foo_item"), "count", 1, "next_page_token", "foo_token",
48                                  "continuation_mode", "pagination")),
49                  Arguments.of("with null items", new MCPItemsPayload(null),
50                          Map.of("response_mode", "list", "items", List.of(), "count", 0, "continuation_mode", "none")),
51                  Arguments.of("with navigation", new MCPItemsPayload(List.of("foo_item"), Map.of("self_resource", Map.of("uri", "shardingsphere://foo"), "next_resources",
52                          List.of(Map.of("uri", "shardingsphere://foo/bar", "resource_kind", "resource", "purpose", "inspect_detail", "reason", "Read child.",
53                                  "source_field", "next_resources")))),
54                          Map.of("response_mode", "list", "items", List.of("foo_item"), "count", 1, "continuation_mode", "none",
55                                  "self_resource", Map.of("uri", "shardingsphere://foo"),
56                                  "next_resources", List.of(Map.of("uri", "shardingsphere://foo/bar", "resource_kind", "resource", "purpose", "inspect_detail",
57                                          "reason", "Read child.", "source_field", "next_resources")))));
58      }
59  }
