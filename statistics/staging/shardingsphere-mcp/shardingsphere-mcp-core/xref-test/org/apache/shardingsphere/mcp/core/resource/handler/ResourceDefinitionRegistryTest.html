67      @ParameterizedTest(name = "{0}")
68      @MethodSource("getSupportedResourcesFailureCases")
69      void assertGetSupportedResourcesFailure(final String name, final Collection<MCPResourceHandler<?>> handlers,
70                                              final Class<? extends Throwable> expectedCauseType, final String expectedMessage) {
71          try (MockedStatic<ShardingSphereServiceLoader> mocked = mockStatic(ShardingSphereServiceLoader.class)) {
72              MCPHandlerProvider provider = createHandlerProvider(handlers);
73              mocked.when(() -> ShardingSphereServiceLoader.getServiceInstances(MCPHandlerProvider.class)).thenReturn(List.of(provider));
74              Class<?> registryClass = assertDoesNotThrow(() -> Class.forName(ResourceDefinitionRegistry.class.getName(), false, createIsolatedResourceDefinitionRegistryClassLoader()));
75              InvocationTargetException actual = assertThrows(InvocationTargetException.class,
76                      () -> Plugins.getMemberAccessor().invoke(registryClass.getMethod("getSupportedResourceDescriptors"), null));
77              assertThat(actual.getCause().getClass(), is(ExceptionInInitializerError.class));
78              Throwable actualCause = actual.getCause().getCause();
79              assertThat(actualCause.getClass(), is(expectedCauseType));
80              assertThat(actualCause.getMessage(), is(expectedMessage));
81          }
82      }

117     private static Stream<Arguments> getSupportedResourcesFailureCases() {
118         MCPResourceHandler<?> nullUriHandler = createResourceHandler(null);
119         MCPResourceHandler<?> emptyUriHandler = createResourceHandler("");
120         MCPResourceHandler<?> blankUriHandler = createResourceHandler("   ");
121         MCPResourceHandler<?> duplicateUriHandler = createResourceHandler("shardingsphere://foo/{id}");
122         MCPResourceHandler<?> otherDuplicateUriHandler = createResourceHandler("shardingsphere://foo/{id}");
123         MCPResourceHandler<?> overlappingUriHandler = createResourceHandler("shardingsphere://foo/{id}");
124         MCPResourceHandler<?> otherOverlappingUriHandler = createResourceHandler("shardingsphere://foo/bar");
125         return Stream.of(
126                 Arguments.of("no resource handlers", Collections.emptyList(), IllegalStateException.class, "No resource handlers are registered."),
127                 Arguments.of("null resource URI", List.of(nullUriHandler), IllegalArgumentException.class,
128                         getRequiredUriMessage()),
129                 Arguments.of("empty resource URI", List.of(emptyUriHandler), IllegalArgumentException.class,
130                         getRequiredUriMessage()),
131                 Arguments.of("blank resource URI", List.of(blankUriHandler), IllegalArgumentException.class,
132                         getRequiredUriMessage()),
133                 Arguments.of("duplicate resource URI template",
134                         List.of(duplicateUriHandler, otherDuplicateUriHandler), IllegalArgumentException.class,
135                         getDuplicateUriTemplateMessage()),
136                 Arguments.of("overlapping resource URI template",
137                         List.of(overlappingUriHandler, otherOverlappingUriHandler), IllegalArgumentException.class,
138                         getOverlappingUriTemplateMessage()),
139                 Arguments.of("unsupported resource handler context", List.of(createUnsupportedResourceHandler()), IllegalArgumentException.class,
140                         getUnsupportedResourceHandlerMessage()));
141     }

163     private static ClassLoader createIsolatedResourceDefinitionRegistryClassLoader() {
164         return new ClassLoader(ResourceDefinitionRegistry.class.getClassLoader()) {
165             
166             @Override
167             protected Class<?> loadClass(final String name, final boolean resolve) throws ClassNotFoundException {
168                 if (!ResourceDefinitionRegistry.class.getName().equals(name)) {
169                     return super.loadClass(name, resolve);
170                 }
171                 synchronized (getClassLoadingLock(name)) {
172                     Class<?> result = findLoadedClass(name);
173                     if (null == result) {
174                         byte[] bytes = readResourceDefinitionRegistryClass(name);
175                         result = defineClass(name, bytes, 0, bytes.length, ResourceDefinitionRegistry.class.getProtectionDomain());
176                     }
177                     if (resolve) {
178                         resolveClass(result);
179                     }
180                     return result;
181                 }
182             }
183         };
184     }
185     
186     private static byte[] readResourceDefinitionRegistryClass(final String name) throws ClassNotFoundException {
187         String resourceName = name.replace('.', '/') + ".class";
188         try (InputStream inputStream = ResourceDefinitionRegistry.class.getClassLoader().getResourceAsStream(resourceName)) {
189             if (null == inputStream) {
190                 throw new ClassNotFoundException(name);
191             }
192             return inputStream.readAllBytes();
193         } catch (final IOException ex) {
194             throw new ClassNotFoundException(name, ex);
195         }
196     }

198     private static String getRequiredUriMessage() {
199         MCPResourceHandler<?> handler = createResourceHandler(null);
200         return String.format("Resource URI template is required for `%s`.", handler.getClass().getName());
201     }
202     
203     private static String getDuplicateUriTemplateMessage() {
204         MCPResourceHandler<?> firstHandler = createResourceHandler("shardingsphere://foo/{id}");
205         MCPResourceHandler<?> secondHandler = createResourceHandler("shardingsphere://foo/{id}");
206         return String.format("Duplicate resource URI template `shardingsphere://foo/{id}` with `%s` and `%s`.",
207                 firstHandler.getClass().getName(), secondHandler.getClass().getName());
208     }
209     
210     private static String getOverlappingUriTemplateMessage() {
211         MCPResourceHandler<?> firstHandler = createResourceHandler("shardingsphere://foo/{id}");
212         MCPResourceHandler<?> secondHandler = createResourceHandler("shardingsphere://foo/bar");
213         return String.format("Overlapping resource URI templates `shardingsphere://foo/{id}` with `%s` and `%s`.",
214                 firstHandler.getClass().getName(), secondHandler.getClass().getName());
215     }
216     
217     private static String getUnsupportedResourceHandlerMessage() {
218         MCPResourceHandler<?> handler = createUnsupportedResourceHandler();
219         return String.format("Unsupported request context type `%s` for `%s`.", MCPFeatureRuntimeRequestContext.class.getName(), handler.getClass().getName());
220     }
