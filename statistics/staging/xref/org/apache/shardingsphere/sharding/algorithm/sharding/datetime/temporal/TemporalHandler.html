18  package org.apache.shardingsphere.sharding.algorithm.sharding.datetime.temporal;
19  
20  import java.time.format.DateTimeFormatter;
21  import java.time.temporal.TemporalAccessor;
22  import java.time.temporal.TemporalUnit;
23  
24  /**
25   * Temporal handler.
26   * 
27   * @param <T> type of temporal
28   */
29  public interface TemporalHandler<T extends TemporalAccessor> {
30      
31      /**
32       * Parse from text.
33       *
34       * @param text temporal text
35       * @param formatter date time formatter
36       * @return parsed temporal
37       */
38      T parse(CharSequence text, DateTimeFormatter formatter);
39      
40      /**
41       * Convert temporal.
42       *
43       * @param temporal to be converted temporal
44       * @return converted temporal
45       */
46      T convertTo(TemporalAccessor temporal);
47      
48      /**
49       * Judge whether temporal1 is after temporal2.
50       *
51       * @param temporal1 temporal1
52       * @param temporal2 temporal2
53       * @param stepAmount step amount
54       * @return is temporal1 after temporal2 or not
55       */
56      boolean isAfter(T temporal1, T temporal2, int stepAmount);
57      
58      /**
59       * Add temporal.
60       *
61       * @param temporal to be added temporal
62       * @param stepAmount to be added step amount
63       * @param unit temporal unit
64       * @return added temporal
65       */
66      T add(T temporal, long stepAmount, TemporalUnit unit);
67  }
