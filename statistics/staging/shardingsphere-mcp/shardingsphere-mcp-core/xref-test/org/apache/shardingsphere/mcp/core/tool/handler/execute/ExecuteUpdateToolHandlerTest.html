60      @ParameterizedTest(name = "{0}")
61      @MethodSource("previewSideEffectCases")
62      void assertHandlePreviewSideEffectScope(final String name, final String sql, final SupportedMCPStatement statementClass, final String statementType, final String savepointName,
63                                              final String expectedClass, final String expectedScope) {
64          MCPFeatureExecutionFacade executionFacade = mock(MCPFeatureExecutionFacade.class);
65          MCPFeatureRequestContext requestContext = mock(MCPFeatureRequestContext.class);
66          when(requestContext.getExecutionFacade()).thenReturn(executionFacade);
67          ClassificationResult classificationResult = new ClassificationResult(statementClass, statementType, sql, savepointName, List.of(), false);
68          try (MockedStatic<SQLExecutionToolHandlerSupport> mocked = mockStatic(SQLExecutionToolHandlerSupport.class)) {
69              mocked.when(() -> SQLExecutionToolHandlerSupport.analyze(eq(requestContext), any(MCPToolArguments.class), eq(sql))).thenReturn(classificationResult);
70              mocked.when(() -> SQLExecutionToolHandlerSupport.checkExecutionArguments(any(MCPToolArguments.class), eq(CoreToolNames.EXECUTE_UPDATE))).thenCallRealMethod();
71              mocked.when(() -> SQLExecutionToolHandlerSupport.isQueryStatement(any(ClassificationResult.class))).thenCallRealMethod();
72              Map<String, Object> actual = new ExecuteUpdateToolHandler().handle(requestContext, Map.of("database", "logic_db", "sql", sql, "execution_mode", "preview")).toPayload();
73              assertThat(actual.get("statement_class"), is(expectedClass));
74              assertThat(actual.get("side_effect_scope"), is(List.of(expectedScope)));
75              assertFalse((boolean) actual.get("would_execute"));
76              verifyNoInteractions(executionFacade);
77          }
78      }

87      @Test
88      void assertHandleUpdateStatement() {
89          MCPFeatureExecutionFacade executionFacade = mock(MCPFeatureExecutionFacade.class);
90          when(executionFacade.execute(any())).thenReturn(createUpdateResult());
91          MCPFeatureRequestContext requestContext = mock(MCPFeatureRequestContext.class);
92          when(requestContext.getSessionIdentity()).thenReturn(new MCPSessionIdentity("session-1", "", "", Map.of()));
93          when(requestContext.getExecutionFacade()).thenReturn(executionFacade);
94          mockDatabaseCapability(requestContext, "logic_db");
95          MCPSuccessPayload actual = new ExecuteUpdateToolHandler().handle(requestContext,
96                  Map.of("database", "logic_db", "schema", "public", "sql", "update orders set status = 'PAID'", "execution_mode", "execute"));
97          assertThat(actual.toPayload().get("response_mode"), is("executed"));
98          assertThat(actual.toPayload().get("execution_mode"), is("execute"));
99          assertThat(actual.toPayload().get("statement_class"), is("dml"));
100         ArgumentCaptor<SQLExecutionRequest> requestCaptor = ArgumentCaptor.forClass(SQLExecutionRequest.class);
101         verify(executionFacade).execute(requestCaptor.capture());
102         assertThat(requestCaptor.getValue().getDatabase(), is("logic_db"));
103         assertThat(requestCaptor.getValue().getSchema(), is("public"));
104         assertThat(requestCaptor.getValue().getSql(), is("update orders set status = 'PAID'"));
105         assertThat(requestCaptor.getValue().getMaxRows(), is(100));
106         assertThat(requestCaptor.getValue().getTimeoutMs(), is(0));
107     }

148     @Test
149     void assertPreviewUpdateStatementWithoutExecuting() {
150         MCPFeatureExecutionFacade executionFacade = mock(MCPFeatureExecutionFacade.class);
151         MCPFeatureRequestContext requestContext = mock(MCPFeatureRequestContext.class);
152         when(requestContext.getExecutionFacade()).thenReturn(executionFacade);
153         mockDatabaseCapability(requestContext, "logic_db");
154         MCPSuccessPayload actual = new ExecuteUpdateToolHandler().handle(requestContext,
155                 Map.of("database", "logic_db", "schema", "public", "sql", "update orders set status = 'PAID'", "execution_mode", "preview"));
156         assertThat(actual.toPayload().get("response_mode"), is("preview"));
157         assertThat(actual.toPayload().get("result_kind"), is("preview"));
158         assertThat(actual.toPayload().get("preview_semantics"), is("classification_only"));
159         assertFalse((boolean) actual.toPayload().get("affected_rows_estimated"));
160         assertFalse((boolean) actual.toPayload().get("would_execute"));
161         assertThat(actual.toPayload().get("status"), is("PREVIEWED"));
162         assertThat(actual.toPayload().get("statement_class"), is("dml"));
163         assertThat(actual.toPayload().get("side_effect_scope"), is(List.of("physical-data")));
164         assertThat(actual.toPayload().get("summary"), is("Previewed UPDATE statement with side-effect scope physical-data. It has not been executed."));
165         assertThat(actual.toPayload().get("review_guidance"),
166                 is("Review normalized_sql and side_effect_scope before execution. "
167                         + "This preview performs database-aware validation and classification; it does not guarantee rule validation, algorithm initialization, affected rows, or runtime success."));
168         assertThat(((Map<?, ?>) actual.toPayload().get("suggested_arguments")).get("execution_mode"), is("execute"));
169         assertThat(((Map<?, ?>) actual.toPayload().get("argument_provenance")).get("sql"), is("server_generated"));
170         assertThat(((Map<?, ?>) actual.toPayload().get("argument_provenance")).get("execution_mode"), is("server_defaulted"));
171         List<?> actualNextActions = (List<?>) actual.toPayload().get("next_actions");
172         assertThat(actualNextActions.size(), is(2));
173         Map<?, ?> actualAskUserAction = (Map<?, ?>) actualNextActions.getFirst();
174         assertThat(actualAskUserAction.get("type"), is("ask_user"));
175         assertThat(actualAskUserAction.get("order"), is(1));
176         assertThat(actualAskUserAction.get("required_inputs"), is(List.of("execution_approved")));
177         Map<?, ?> actualToolCallAction = (Map<?, ?>) actualNextActions.get(1);
178         assertThat(actualToolCallAction.get("type"), is("tool_call"));
179         assertThat(actualToolCallAction.get("order"), is(2));
180         assertThat(actualToolCallAction.get("tool_name"), is("database_gateway_execute_update"));
181         assertThat(actualToolCallAction.get("depends_on"), is(List.of(1)));
182         assertThat(actualToolCallAction.get("reason"),
183                 is("Execute only after reviewing normalized_sql and side_effect_scope; preview did not validate runtime executability."));
184         assertThat(((Map<?, ?>) actualToolCallAction.get("arguments")).get("execution_mode"), is("execute"));
185         assertThat(((Map<?, ?>) ((List<?>) actual.toPayload().get("resources_to_read")).getFirst()).get("uri"),
186                 is("shardingsphere://databases/logic_db/capabilities"));
187         assertFalse(actual.toPayload().containsKey("ask_user_when_uncertain"));
188         verifyNoInteractions(executionFacade);
189     }
