109     @MethodSource("getValuesRowsArguments")
110     void assertValuesRows(final String name, final String sql, final int parameterCount, final int subqueryCount) {
111         SelectStatement actual = (SelectStatement) new SQLStatementVisitorEngine("PostgreSQL").visit(new SQLParserEngine("PostgreSQL", new CacheOption(128, 1024L)).parse(sql, false));
112         assertThat(actual.getValues().size(), is(2));
113         assertThat(actual.getProjections().getProjections().stream().map(ProjectionSegment::getColumnLabel).collect(Collectors.toList()), is(Arrays.asList("column1", "column2")));
114         Collection<ExpressionSegment> expressions = new ArrayList<>(4);
115         for (InsertValuesSegment each : actual.getValues()) {
116             assertThat(each.getValues().size(), is(2));
117             assertThat(sql.charAt(each.getStartIndex()), is('('));
118             assertThat(sql.charAt(each.getStopIndex()), is(')'));
119             expressions.addAll(each.getValues());
120         }
121         assertThat(actual.getParameterCount(), is(parameterCount));
122         assertThat(ExpressionExtractor.getParameterMarkerExpressions(expressions).stream().sorted(Comparator.comparingInt(ParameterMarkerSegment::getStartIndex))
123                 .map(ParameterMarkerSegment::getParameterIndex).collect(Collectors.toList()), is(IntStream.range(0, parameterCount).boxed().collect(Collectors.toList())));
124         assertThat(SubqueryExtractor.extractSubquerySegments(actual, false).size(), is(subqueryCount));
125     }

145     void assertQuotedIdentifierRoundTrip(final String name, final String identifier, final String expectedValue, final String expectedQuotedValue) {
146         String sql = "SELECT " + identifier + " AS " + identifier + " FROM " + identifier + "." + identifier + " AS " + identifier;
147         SelectStatement actual = (SelectStatement) new SQLStatementVisitorEngine("PostgreSQL").visit(new SQLParserEngine("PostgreSQL", new CacheOption(128, 1024L)).parse(sql, false));
148         ColumnProjectionSegment actualProjection = (ColumnProjectionSegment) actual.getProjections().getProjections().iterator().next();
149         SimpleTableSegment actualTable = (SimpleTableSegment) actual.getFrom().get();
150         ColumnSegment actualColumn = actualProjection.getColumn();
151         assertThat(sql.substring(actualColumn.getStartIndex(), actualColumn.getStopIndex() + 1), is(identifier));
152         assertThat(sql.substring(actualTable.getTableName().getStartIndex(), actualTable.getTableName().getStopIndex() + 1), is(identifier));
153         Collection<IdentifierValue> actualIdentifiers = Arrays.asList(actualColumn.getIdentifier(), actualProjection.getAlias().get(), actualTable.getOwner().get().getIdentifier(),

155         for (IdentifierValue each : actualIdentifiers) {
156             assertThat(each.getValue(), is(expectedValue));
157             assertThat(each.getValueWithQuoteCharacters(), is(expectedQuotedValue));
158             String roundTripSQL = "SELECT " + each.getValueWithQuoteCharacters();
159             SelectStatement actualRoundTrip = (SelectStatement) new SQLStatementVisitorEngine("PostgreSQL")
160                     .visit(new SQLParserEngine("PostgreSQL", new CacheOption(128, 1024L)).parse(roundTripSQL, false));
161             assertThat(((ColumnProjectionSegment) actualRoundTrip.getProjections().getProjections().iterator().next()).getColumn().getIdentifier().getValue(), is(expectedValue));
162         }
