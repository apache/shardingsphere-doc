132     @ParameterizedTest(name = "{0}")
133     @MethodSource("assertCheckBeforeUpdateWithFailureArguments")
134     void assertCheckBeforeUpdateWithFailure(final String name, final Collection<String> allTables, final Collection<String> singleTables, final Collection<DataNode> dataNodes,
135                                             final Collection<String> configuredTables, final Class<? extends RuntimeException> expectedException) {
136         Map<String, Collection<DataNode>> tableDataNodes = new HashMap<>(1, 1F);
137         tableDataNodes.put("foo_tbl", dataNodes);
138         prepareCheckBeforeUpdateContext(allTables, singleTables, tableDataNodes, configuredTables);
139         assertThrows(expectedException, () -> executor.checkBeforeUpdate(new UnloadSingleTableStatement(false, Collections.singletonList("foo_tbl"))));
140     }

158     @ParameterizedTest(name = "{0}")
159     @MethodSource("assertBuildToBeAlteredRuleConfigurationArguments")
160     void assertBuildToBeAlteredRuleConfiguration(final String name, final boolean schemaAvailable, final Collection<String> currentTables,
161                                                  final Map<String, Collection<DataNode>> tableDataNodes, final UnloadSingleTableStatement sqlStatement,
162                                                  final Collection<String> expectedTables) {
163         when(rule.getConfiguration()).thenReturn(new SingleRuleConfiguration(new LinkedList<>(currentTables), null));
164         when(dataNodeRuleAttribute.getDataNodesByTableName(anyString())).thenAnswer(invocation -> tableDataNodes.getOrDefault(invocation.getArgument(0), Collections.emptyList()));
165         try (MockedConstruction<DatabaseTypeRegistry> ignored = mockDatabaseTypeRegistry(schemaAvailable)) {
166             SingleRuleConfiguration actual = executor.buildToBeAlteredRuleConfiguration(sqlStatement);
167             assertThat(new HashSet<>(actual.getTables()), is(new HashSet<>(expectedTables)));
168         }
169     }

191     private MockedConstruction<DatabaseTypeRegistry> mockDatabaseTypeRegistry(final boolean storageSchemaAvailable) {
192         DialectDatabaseMetaData protocolMetaData = mock(DialectDatabaseMetaData.class, RETURNS_DEEP_STUBS);
193         when(protocolMetaData.getSchemaOption().isSchemaAvailable()).thenReturn(true);
194         DialectDatabaseMetaData storageMetaData = mock(DialectDatabaseMetaData.class, RETURNS_DEEP_STUBS);
195         when(storageMetaData.getSchemaOption().isSchemaAvailable()).thenReturn(storageSchemaAvailable);
196         return mockConstruction(DatabaseTypeRegistry.class, (mock, context) -> when(mock.getDialectDatabaseMetaData())
197                 .thenReturn(storageDatabaseType == context.arguments().get(0) ? storageMetaData : protocolMetaData));
198     }

200     private void prepareCheckBeforeUpdateContext(final Collection<String> allTables, final Collection<String> singleTables,
201                                                  final Map<String, Collection<DataNode>> tableDataNodes, final Collection<String> configuredTables) {
202         ShardingSphereSchema schema = mock(ShardingSphereSchema.class);
203         when(schema.getAllTables()).thenReturn(
204                 allTables.stream().map(each -> new ShardingSphereTable(each, Collections.emptyList(), Collections.emptyList(), Collections.emptyList())).collect(Collectors.toList()));
205         when(database.findDefaultSchema()).thenReturn(Optional.of(schema));
206         when(tableMapperRuleAttribute.getLogicTableNames()).thenReturn(singleTables);
207         when(dataNodeRuleAttribute.getDataNodesByTableName(anyString())).thenAnswer(invocation -> tableDataNodes.getOrDefault(invocation.getArgument(0), Collections.emptyList()));
208         when(rule.getConfiguration()).thenReturn(new SingleRuleConfiguration(new LinkedList<>(configuredTables), null));
209     }

211     private static Stream<Arguments> assertCheckBeforeUpdateWithFailureArguments() {
212         return Stream.of(
213                 Arguments.of("missing table throws NoSuchTableException", Collections.emptyList(), Collections.singleton("foo_tbl"), Collections.singleton(new DataNode("foo_ds.foo_tbl")),
214                         Collections.singleton("foo_ds.foo_tbl"), NoSuchTableException.class),
215                 Arguments.of("non single table throws SingleTableNotFoundException", Collections.singleton("foo_tbl"), Collections.emptyList(), Collections.singleton(new DataNode("foo_ds.foo_tbl")),
216                         Collections.singleton("foo_ds.foo_tbl"), SingleTableNotFoundException.class),
217                 Arguments.of("missing data node throws MissingRequiredRuleException", Collections.singleton("foo_tbl"), Collections.singleton("foo_tbl"), Collections.emptyList(),
218                         Collections.singleton("foo_ds.foo_tbl"), MissingRequiredRuleException.class),
219                 Arguments.of("missing rule config throws MissingRequiredRuleException", Collections.singleton("foo_tbl"), Collections.singleton("foo_tbl"),
220                         Collections.singleton(new DataNode("foo_ds.foo_tbl")),
221                         Collections.emptyList(), MissingRequiredRuleException.class));
222     }

224     private static Stream<Arguments> assertCheckBeforeUpdateWithSuccessArguments() {
225         Map<String, Collection<DataNode>> singleTableNodes = new HashMap<>(1, 1F);
226         singleTableNodes.put("foo_tbl", Collections.singleton(new DataNode("foo_ds.foo_tbl")));
227         Map<String, Collection<DataNode>> multipleTableNodes = new HashMap<>(2, 1F);
228         multipleTableNodes.put("foo_tbl", Collections.singleton(new DataNode("foo_ds.foo_tbl")));
229         multipleTableNodes.put("bar_tbl", Collections.singleton(new DataNode("foo_ds.foo_schema.bar_tbl")));
230         Map<String, Collection<DataNode>> schemaTableNodes = new HashMap<>(1, 1F);
231         schemaTableNodes.put("foo_tbl", Collections.singleton(new DataNode("foo_ds.foo_schema.foo_tbl")));
232         return Stream.of(
233                 Arguments.of("unload all tables bypasses table checks", false, new UnloadSingleTableStatement(true, Collections.emptyList()),
234                         Collections.emptyList(), Collections.emptyList(), Collections.emptyMap(), Collections.emptyList()),
235                 Arguments.of("valid single table passes checks", false, new UnloadSingleTableStatement(false, Collections.singletonList("foo_tbl")),
236                         Collections.singleton("foo_tbl"), Collections.singleton("foo_tbl"), singleTableNodes, Collections.singleton("foo_ds.foo_tbl")),
237                 Arguments.of("valid multiple tables pass checks", false, new UnloadSingleTableStatement(false, Arrays.asList("foo_tbl", "bar_tbl")),
238                         Arrays.asList("foo_tbl", "bar_tbl"), Arrays.asList("foo_tbl", "bar_tbl"), multipleTableNodes, Arrays.asList("foo_ds.foo_tbl", "foo_ds.bar_tbl")),
239                 Arguments.of("valid schema table passes checks", true, new UnloadSingleTableStatement(false, Collections.singletonList("foo_tbl")),
240                         Collections.singleton("foo_tbl"), Collections.singleton("foo_tbl"), schemaTableNodes, Collections.singleton("foo_ds.foo_schema.foo_tbl")));
241     }

243     private static Stream<Arguments> assertBuildToBeAlteredRuleConfigurationArguments() {
244         Map<String, Collection<DataNode>> twoSegmentTableNodes = Collections.singletonMap("foo_tbl", Collections.singleton(new DataNode("foo_ds.foo_tbl")));
245         Map<String, Collection<DataNode>> schemaTableNodes = Collections.singletonMap("bar_tbl", Collections.singleton(new DataNode("foo_ds.foo_schema.bar_tbl")));
246         Map<String, Collection<DataNode>> dottedTableNodes = Collections.singletonMap(
247                 "foo.bar", Arrays.asList(new DataNode("readwrite_ds", "foo_schema", "foo.bar"), new DataNode("bar_ds", "foo_schema", "foo.bar")));
248         Map<String, Collection<DataNode>> suffixTableNodes = Collections.singletonMap("bar", Collections.singleton(new DataNode("foo_ds", "foo_schema", "bar")));
249         return Stream.of(
250                 Arguments.of("unload all tables keeps altered config empty", false, Arrays.asList("foo_ds.foo_tbl", "foo_ds.bar_tbl"), Collections.emptyMap(),
251                         new UnloadSingleTableStatement(true, Collections.emptyList()), Collections.emptyList()),
252                 Arguments.of("unload one table removes matching two-segment node", false, Arrays.asList("foo_ds.foo_tbl", "foo_ds.bar_tbl"), twoSegmentTableNodes,
253                         new UnloadSingleTableStatement(false, Collections.singletonList("foo_tbl")), Collections.singleton("foo_ds.bar_tbl")),
254                 Arguments.of("unload one table removes matching three-segment node", true, Arrays.asList("foo_ds.foo_schema.bar_tbl", "foo_ds.foo_tbl"), schemaTableNodes,
255                         new UnloadSingleTableStatement(false, Collections.singletonList("bar_tbl")), Collections.singleton("foo_ds.foo_tbl")),
256                 Arguments.of("unload dotted table removes all exact nodes", false, Arrays.asList("readwrite_ds.foo.bar", "bar_ds.foo.bar", "foo_ds.bar"), dottedTableNodes,
257                         new UnloadSingleTableStatement(false, Collections.singletonList("foo.bar")), Collections.singleton("foo_ds.bar")),
258                 Arguments.of("unload suffix table preserves dotted table", false, Arrays.asList("foo_ds.bar", "foo_ds.foo.bar"), suffixTableNodes,
259                         new UnloadSingleTableStatement(false, Collections.singletonList("bar")), Collections.singleton("foo_ds.foo.bar")));
260     }
