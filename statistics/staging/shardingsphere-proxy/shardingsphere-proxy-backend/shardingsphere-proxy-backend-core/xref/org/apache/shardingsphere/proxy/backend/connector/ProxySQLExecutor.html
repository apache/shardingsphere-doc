109     private final Map<ShardingSphereRule, TransactionHook> transactionHooks = OrderedSPILoader.getServices(

112     public ProxySQLExecutor(final JDBCDriverType type,
113                             final ProxyDatabaseConnectionManager databaseConnectionManager, final DatabaseProxyConnector databaseProxyConnector, final SQLStatementContext sqlStatementContext) {
114         this.type = type;
115         this.databaseConnectionManager = databaseConnectionManager;
116         ExecutorEngine executorEngine = BackendExecutorContext.getInstance().getExecutorEngine();
117         ConnectionContext connectionContext = databaseConnectionManager.getConnectionSession().getConnectionContext();
118         JDBCExecutor jdbcExecutor = new JDBCExecutor(executorEngine, connectionContext);
119         regularExecutor = new ProxyJDBCExecutor(type, databaseConnectionManager.getConnectionSession(), databaseProxyConnector, jdbcExecutor);
120         rawExecutor = new RawExecutor(executorEngine, connectionContext);
121         MetaDataContexts metaDataContexts = ProxyContext.getInstance().getContextManager().getMetaDataContexts();
122         String currentDatabaseName = Strings.isNullOrEmpty(databaseConnectionManager.getConnectionSession().getCurrentDatabaseName())
123                 ? databaseConnectionManager.getConnectionSession().getUsedDatabaseName()
124                 : databaseConnectionManager.getConnectionSession().getCurrentDatabaseName();
125         ShardingSphereDatabase currentDatabase = metaDataContexts.getMetaData().getDatabase(currentDatabaseName);
126         String currentSchemaName = getSchemaName(sqlStatementContext, currentDatabase);
127         sqlFederationEngine = new SQLFederationEngine(currentDatabaseName, currentSchemaName, metaDataContexts.getMetaData(), metaDataContexts.getStatistics(), jdbcExecutor);
128     }
129     
130     private String getSchemaName(final SQLStatementContext sqlStatementContext, final ShardingSphereDatabase database) {
131         return sqlStatementContext.getTablesContext().getSchemaName().orElseGet(database::getDefaultSchemaName);
132     }
133     
134     /**
135      * Check execute prerequisites.
136      *
137      * @param sqlStatementContext execution context
138      */
139     public void checkExecutePrerequisites(final SQLStatementContext sqlStatementContext) {
140         ShardingSpherePreconditions.checkState(
141                 isValidExecutePrerequisites(sqlStatementContext.getSqlStatement()), () -> new TableModifyInTransactionException(getTableName(sqlStatementContext.getTablesContext())));
142     }
143     
144     private boolean isValidExecutePrerequisites(final SQLStatement sqlStatement) {
145         return !(sqlStatement instanceof DDLStatement) || isSupportDDLInTransaction(sqlStatement.getDatabaseType(), (DDLStatement) sqlStatement);
146     }
147     
148     private boolean isSupportDDLInTransaction(final DatabaseType databaseType, final DDLStatement sqlStatement) {
149         DialectTransactionOption transactionOption = new DatabaseTypeRegistry(databaseType).getDialectDatabaseMetaData().getTransactionOption();
150         boolean isDDLWithoutMetaDataChanged = isDDLWithoutMetaDataChanged(sqlStatement);
151         if (isInXATransaction()) {
152             return transactionOption.isSupportDDLInXATransaction() && (isDDLWithoutMetaDataChanged || transactionOption.isSupportMetaDataRefreshInTransaction());
153         }
154         if (isInLocalTransaction()) {
155             return transactionOption.isSupportMetaDataRefreshInTransaction() || isDDLWithoutMetaDataChanged;
156         }
157         return true;
158     }
159     
160     private boolean isInXATransaction() {
161         TransactionType transactionType = TransactionUtils.getTransactionType(databaseConnectionManager.getConnectionSession().getConnectionContext().getTransactionContext());
162         TransactionStatus transactionStatus = databaseConnectionManager.getConnectionSession().getTransactionStatus();
163         return TransactionType.XA == transactionType && transactionStatus.isInTransaction();
164     }
165     
166     private boolean isInLocalTransaction() {
167         return databaseConnectionManager.getConnectionSession().getTransactionStatus().isInTransaction();
168     }
169     
170     // TODO should be removed after metadata refresh supported for all database.
171     private boolean isDDLWithoutMetaDataChanged(final DDLStatement sqlStatement) {
172         return isCursorStatement(sqlStatement) || sqlStatement instanceof TruncateStatement;
173     }
174     
175     private boolean isCursorStatement(final DDLStatement sqlStatement) {
176         return sqlStatement instanceof CursorStatement || sqlStatement instanceof CloseStatement || sqlStatement instanceof MoveStatement || sqlStatement instanceof FetchStatement;
177     }
178     
179     private String getTableName(final TablesContext tablesContext) {
180         return tablesContext.getSimpleTables().isEmpty() ? "unknown_table" : tablesContext.getSimpleTables().iterator().next().getTableName().getIdentifier().getValue();
181     }

183     /**
184      * Execute SQL.
185      *
186      * @param executionContext execution context
187      * @return execute results
188      * @throws SQLException SQL exception
189      */
190     public List<ExecuteResult> execute(final ExecutionContext executionContext) throws SQLException {
191         String databaseName = databaseConnectionManager.getConnectionSession().getUsedDatabaseName();
192         Collection<ShardingSphereRule> rules = ProxyContext.getInstance().getContextManager().getMetaDataContexts().getMetaData().getDatabase(databaseName).getRuleMetaData().getRules();
193         int maxConnectionsSizePerQuery = ProxyContext.getInstance()
194                 .getContextManager().getMetaDataContexts().getMetaData().getProps().<Integer>getValue(ConfigurationPropertyKey.MAX_CONNECTIONS_SIZE_PER_QUERY);
195         DialectDatabaseMetaData dialectDatabaseMetaData = new DatabaseTypeRegistry(executionContext.getSqlStatementContext().getSqlStatement().getDatabaseType()).getDialectDatabaseMetaData();
196         boolean isReturnGeneratedKeys = isReturnGeneratedKeys(executionContext.getSqlStatementContext(), dialectDatabaseMetaData, executionContext.getQueryContext().getParameters());
197         return hasRawExecutionRule(rules)
198                 ? rawExecute(executionContext, rules, maxConnectionsSizePerQuery)
199                 : useDriverToExecute(executionContext, rules, maxConnectionsSizePerQuery, isReturnGeneratedKeys, SQLExecutorExceptionHandler.isExceptionThrown());
200     }
201     
202     private boolean hasRawExecutionRule(final Collection<ShardingSphereRule> rules) {
203         for (ShardingSphereRule each : rules) {
204             if (each.getAttributes().findAttribute(RawExecutionRuleAttribute.class).isPresent()) {
205                 return true;
206             }
207         }
208         return false;
209     }
210     
211     private List<ExecuteResult> rawExecute(final ExecutionContext executionContext, final Collection<ShardingSphereRule> rules, final int maxConnectionsSizePerQuery) throws SQLException {
212         RawExecutionPrepareEngine prepareEngine = new RawExecutionPrepareEngine(maxConnectionsSizePerQuery, rules);
213         ExecutionGroupContext<RawSQLExecutionUnit> executionGroupContext;
214         try {
215             String databaseName = databaseConnectionManager.getConnectionSession().getUsedDatabaseName();
216             executionGroupContext = prepareEngine.prepare(databaseName, executionContext, executionContext.getExecutionUnits(),
217                     new ExecutionGroupReportContext(databaseConnectionManager.getConnectionSession().getProcessId(),
218                             databaseName, databaseConnectionManager.getConnectionSession().getConnectionContext().getGrantee()));
219         } catch (final SQLException ex) {
220             return getSaneExecuteResults(executionContext, ex);
221         }
222         // TODO handle query header
223         return rawExecutor.execute(executionGroupContext, executionContext.getQueryContext(), new RawSQLExecutorCallback());
224     }
225     
226     private List<ExecuteResult> useDriverToExecute(final ExecutionContext executionContext, final Collection<ShardingSphereRule> rules,
227                                                    final int maxConnectionsSizePerQuery, final boolean isReturnGeneratedKeys, final boolean isExceptionThrown) throws SQLException {
228         JDBCBackendStatement statementManager = (JDBCBackendStatement) databaseConnectionManager.getConnectionSession().getStatementManager();
229         String databaseName = databaseConnectionManager.getConnectionSession().getUsedDatabaseName();
230         DriverExecutionPrepareEngine<JDBCExecutionUnit, Connection> prepareEngine = new DriverExecutionPrepareEngine<>(
231                 type, maxConnectionsSizePerQuery, databaseConnectionManager, statementManager, new StatementOption(isReturnGeneratedKeys), rules,
232                 ProxyContext.getInstance().getContextManager().getMetaDataContexts().getMetaData());
233         ExecutionGroupContext<JDBCExecutionUnit> executionGroupContext;
234         try {
235             executionGroupContext = prepareEngine.prepare(databaseName, executionContext, executionContext.getExecutionUnits(),
236                     new ExecutionGroupReportContext(databaseConnectionManager.getConnectionSession().getProcessId(),
237                             databaseName, databaseConnectionManager.getConnectionSession().getConnectionContext().getGrantee()));
238         } catch (final SQLException ex) {
239             return getSaneExecuteResults(executionContext, ex);
240         }
241         if (executionContext.getExecutionUnits().size() > 1) {
242             DatabaseType databaseType = ProxyContext.getInstance().getContextManager().getDatabase(databaseName).getProtocolType();
243             Optional<DialectJDBCResultMetadataChecker> checker = DatabaseTypedSPILoader.findService(DialectJDBCResultMetadataChecker.class, databaseType);
244             if (checker.isPresent()) {
245                 JDBCExecutionUnit sample = executionGroupContext.getInputGroups().iterator().next().getInputs().get(0);
246                 checker.get().check(executionContext.getExecutionUnits(), sample.getStorageResource(), sample.getExecutionUnit().getSqlUnit().getSql());
247             }
248         }
249         executeTransactionHooksBeforeExecuteSQL(databaseConnectionManager.getConnectionSession());
250         return regularExecutor.execute(executionContext.getQueryContext(), executionGroupContext, isReturnGeneratedKeys, isExceptionThrown);
251     }
252     
253     @SuppressWarnings({"unchecked", "rawtypes"})
254     private void executeTransactionHooksBeforeExecuteSQL(final ConnectionSession connectionSession) throws SQLException {
255         if (!getTransactionContext(connectionSession).isInTransaction()) {
256             return;
257         }
258         for (Entry<ShardingSphereRule, TransactionHook> entry : transactionHooks.entrySet()) {
259             entry.getValue().beforeExecuteSQL(entry.getKey(), ProxyContext.getInstance().getContextManager().getDatabaseType(),
260                     connectionSession.getDatabaseConnectionManager().getCachedConnections().values(), getTransactionContext(connectionSession),
261                     connectionSession.getIsolationLevel().orElse(TransactionIsolationLevel.READ_COMMITTED));
262         }
263     }
264     
265     private TransactionConnectionContext getTransactionContext(final ConnectionSession connectionSession) {
266         return connectionSession.getDatabaseConnectionManager().getConnectionSession().getConnectionContext().getTransactionContext();
267     }
268     
269     private List<ExecuteResult> getSaneExecuteResults(final ExecutionContext executionContext, final SQLException originalException) throws SQLException {
270         DatabaseType databaseType = ProxyContext.getInstance().getContextManager().getDatabase(databaseConnectionManager.getConnectionSession().getUsedDatabaseName()).getProtocolType();
271         Optional<ExecuteResult> executeResult = DatabaseTypedSPILoader.findService(DialectSaneQueryResultEngine.class, databaseType)
272                 .flatMap(optional -> optional.getSaneQueryResult(executionContext.getSqlStatementContext().getSqlStatement(), originalException));
273         return executeResult.map(Collections::singletonList).orElseThrow(() -> originalException);
274     }

276     /**
277      * Judge whether to return generated keys.
278      *
279      * @param sqlStatementContext SQL statement context
280      * @param dialectDatabaseMetaData dialect database meta data
281      * @param params SQL parameters
282      * @return whether to return generated keys
283      */
284     public static boolean isReturnGeneratedKeys(final SQLStatementContext sqlStatementContext, final DialectDatabaseMetaData dialectDatabaseMetaData, final List<Object> params) {
285         if (!(sqlStatementContext instanceof InsertStatementContext) || !dialectDatabaseMetaData.getGeneratedKeyOption().isPresent()) {
286             return false;
287         }
288         InsertStatementContext insertStatementContext = (InsertStatementContext) sqlStatementContext;
289         if (!insertStatementContext.getGeneratedKeyContext().isPresent()) {
290             return false;
291         }
292         
293         GeneratedKeyContext generatedKeyContext = insertStatementContext.getGeneratedKeyContext().get();
294         if (generatedKeyContext.isGenerated()) {
295             return true;
296         }
297         
298         DialectGeneratedKeyOption generatedKeyOption = dialectDatabaseMetaData.getGeneratedKeyOption().get();
299         return isReturnGeneratedKeysForExplicit(insertStatementContext, params, generatedKeyOption, generatedKeyContext.getColumnName());
300     }
301     
302     private static boolean isReturnGeneratedKeysForExplicit(final InsertStatementContext insertStatementContext, final List<Object> params,
303                                                             final DialectGeneratedKeyOption generatedKeyOption, final String columnName) {
304         int columnIndex = -1;
305         int index = 0;
306         for (String each : insertStatementContext.getColumnNames()) {
307             if (each.equalsIgnoreCase(columnName)) {
308                 columnIndex = index;
309                 break;
310             }
311             index++;
312         }
313         if (-1 != columnIndex) {
314             for (InsertValueContext each : insertStatementContext.getInsertValueContexts()) {
315                 if (isReturnGeneratedKeysForExplicit(each, columnIndex, params, generatedKeyOption)) {
316                     return true;
317                 }
318             }
319         }
320         return false;
321     }
322     
323     private static boolean isReturnGeneratedKeysForExplicit(final InsertValueContext insertValueContext, final int columnIndex, final List<Object> params,
324                                                             final DialectGeneratedKeyOption generatedKeyOption) {
325         List<ExpressionSegment> expressions = insertValueContext.getValueExpressions();
326         if (null != expressions && columnIndex < expressions.size()) {
327             ExpressionSegment expr = expressions.get(columnIndex);
328             return isReturnGeneratedKeysFromExpression(expr, params, generatedKeyOption);
329         }
330         return false;
331     }
332     
333     private static boolean isReturnGeneratedKeysFromExpression(final ExpressionSegment expr, final List<Object> params, final DialectGeneratedKeyOption generatedKeyOption) {
334         if (expr instanceof ParameterMarkerExpressionSegment) {
335             int markerIndex = ((ParameterMarkerExpressionSegment) expr).getParameterMarkerIndex();
336             if (params.size() > markerIndex) {
337                 return generatedKeyOption.isGeneratedKeyTriggerValue(params.get(markerIndex));
338             }
339             return false;
340         }
341         if (expr instanceof LiteralExpressionSegment) {
342             return generatedKeyOption.isGeneratedKeyTriggerValue(((LiteralExpressionSegment) expr).getLiterals());
343         }
344         if (expr instanceof CommonExpressionSegment) {
345             return generatedKeyOption.isGeneratedKeyTriggerValue(expr.getText());
346         }
347         return false;
348     }
